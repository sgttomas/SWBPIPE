use open_pipe_stress_frame_kernel::*;
use open_pipe_stress_straight_pipe::*;
use open_pipe_stress_curved_bend::*;
use open_pipe_stress_linear_supports::*;
use serde_json::{json,Value};
fn mv(k:&Matrix12,d:&[f64;12])->[f64;12]{std::array::from_fn(|i|(0..12).map(|j|k[i][j]*d[j]).sum())}
fn dot(a:&[f64],b:&[f64])->f64{a.iter().zip(b).map(|(x,y)|x*y).sum()}
fn max(a:&[f64])->f64{a.iter().map(|x|x.abs()).fold(0.,f64::max)}
fn cross(a:[f64;3],b:[f64;3])->[f64;3]{[a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0]]}
fn section()->FrameSection{FrameSection::new(2e11,8e10,0.003,5e-6,7e-6,9e-6).unwrap()}
fn pipe(i:FrameNode,j:FrameNode)->StraightPipeElement{StraightPipeElement::new("P",i,j,StraightPipeSectionProperties::new(2e11,8e10,0.003,5e-6,7e-6,9e-6,Some(23.)).unwrap(),[0.,1.,0.]).unwrap()}
fn rigid_d(i:[f64;3],j:[f64;3],axis:usize)->[f64;12]{let mut d=[0.;12]; for (n,x) in [i,j].iter().enumerate(){if axis<3{d[6*n+axis]=1.;}else{let mut w=[0.;3];w[axis-3]=1.;d[6*n..6*n+3].copy_from_slice(&cross(w,*x));d[6*n+axis]=1.;}}d}
fn resultants(p:[f64;12],chord:[f64;3])->Value{let m=cross(chord,[p[6],p[7],p[8]]);json!({"force":[p[0]+p[6],p[1]+p[7],p[2]+p[8]],"moment_about_i":[p[3]+p[9]+m[0],p[4]+p[10]+m[1],p[5]+p[11]+m[2]]})}
fn main(){
let i=FrameNode::new(0,[0.,0.,0.]).unwrap();let j=FrameNode::new(1,[2.,0.,0.]).unwrap();let e=FrameElement::new(i,j,section(),[0.,1.,0.]).unwrap();let k=e.global_stiffness().unwrap();let p=pipe(i,j);
let f=vec![0.,0.,0.,0.,0.,0.,120.,350.,-240.,80.,-60.,45.];let kd=k.iter().map(|x|x.to_vec()).collect::<Vec<_>>();let r=reduce_system(&kd,&f,&[0,1,2,3,4,5]).unwrap();let x=solve_dense(&r.stiffness,&r.force).unwrap();let mut d=[0.;12];for (a,b) in r.free_dofs.iter().zip(x){d[*a]=b;}let mut reactions=mv(&k,&d);for a in 0..12{reactions[a]-=f[a];}
let rigid=(0..6).map(|a|{let v=rigid_d(i.coordinates,j.coordinates,a);json!({"axis":a,"max_force":max(&mv(&k,&v)),"energy":0.5*dot(&v,&mv(&k,&v))})}).collect::<Vec<_>>();
let kr=FrameElement::new(j,i,section(),[0.,1.,0.]).unwrap().global_stiffness().unwrap();let reverse=(0..12).flat_map(|a|(0..12).map(move|b|(k[a][b]-kr[(a+6)%12][(b+6)%12]).abs())).fold(0.,f64::max);
let user=UserStiffnessElement::new(i,j,[0.,1.,0.],1000.,2000.,3000.,4000.).unwrap();let uk=user.global_stiffness().unwrap();let uv=rigid_d(i.coordinates,j.coordinates,5);let uf=mv(&uk,&uv);
let loads=[UniformLocalLoad::new(LocalLoadDirection::Y,-100.).unwrap()];let stations=[0.,0.25,0.5,0.75,1.].map(|s|{let r=p.recover_station_resultants(&[0.;12],s,&loads,&[]).unwrap();json!({"x":r.distance_from_i,"vy":r.shear_force_y,"mz":r.bending_moment_z})});
let loadcases=[LocalLoadDirection::X,LocalLoadDirection::Y,LocalLoadDirection::Z].map(|a|{let q=SpannedUniformLocalLoad::new(a,-100.,UniformLoadSpan::new(0.2,0.7).unwrap()).unwrap();let z=p.equivalent_nodal_loads_with_spans(&[q],&[PointLocalForce::new(0.4,a,73.).unwrap()]).unwrap();json!({"direction":format!("{:?}",a),"nodal":z,"resultants":resultants(z,[2.,0.,0.])})});
let mut bends=Vec::new();for phi in [0.05_f64,0.2,0.7,std::f64::consts::FRAC_PI_2,2.5]{for factors in [[1.,1.],[2.,3.]]{let bi=FrameNode::new(0,[2.,0.,0.]).unwrap();let bj=FrameNode::new(1,[2.*phi.cos(),2.*phi.sin(),0.]).unwrap();let b=CurvedBendMacroElement::new(bi,bj,[0.;3],2e11,8e10,0.003,5e-6,9e-6,factors[0],factors[1]).unwrap();let flex=b.end_flexibility().unwrap();let bk=b.global_stiffness();let chord=[bj.coordinates[0]-2.,bj.coordinates[1],0.];let uniform=b.consistent_uniform_nodal_loads([37.,-61.,83.]);let pressure=b.consistent_radial_pressure_nodal_loads(120.);bends.push(json!({"phi":phi,"factors":factors,"flexibility":flex,"stiffness_error":bk.as_ref().err().map(|e|e.to_string()),"rigid":bk.as_ref().ok().map(|a|(0..6).map(|z|max(&mv(a,&rigid_d(bi.coordinates,bj.coordinates,z)))).collect::<Vec<_>>()),"uniform_resultants":uniform.ok().map(|p|resultants(p,chord)),"pressure_resultants":pressure.ok().map(|p|resultants(p,chord)),"stations":[0.,0.25,0.5,0.75,1.].map(|s|b.arc_section_resultants_with_radial_pressure(s,[3.,5.,7.,11.,13.,17.],[37.,-61.,83.],120.).unwrap())}));}}
let spring=LinearSupport::spring("S",1,FrameDof::Uy,Some(SupportQuantity::positive(2e6,QuantityDimension::TranslationalStiffness).unwrap()));let app=apply_linear_supports(&kd,&f,&[LinearSupport::anchor("A",0),spring]).unwrap();let sx=solve_dense(&app.reduced_system.stiffness,&app.reduced_system.force).unwrap();
let mut ns=LinearSupport::spring("negative",1,FrameDof::Uy,Some(SupportQuantity::new(-2e6,QuantityDimension::TranslationalStiffness).unwrap()));ns.restrained_dofs.push(FrameDof::Uz);let na=apply_linear_supports(&kd,&f,&[LinearSupport::anchor("A",0),ns]);
let overflow=local_stiffness(FrameProperties::new(FrameSection::new(1e308,1e308,10.,10.,10.,10.).unwrap(),2.).unwrap()).unwrap();let invalid_point=p.station_resultants_from_i_end(PipeEndResultants::from_local_forces(&[0.;12],PipeEnd::I),0.5,&[],&[PointLocalForce{station_fraction:2.,direction:LocalLoadDirection::Y,force:10.}]);
let out=json!({"tip_displacements":&d[6..],"reactions":reactions,"rigid_frame":rigid,"reversal_max_abs":reverse,"user_rigid_rotation":{"displacements":uv,"forces":uf,"energy":0.5*dot(&uv,&uf),"net":resultants(uf,[2.,0.,0.])},"clamped_uniform_stations":stations,"partial_and_point":loadcases,"bends":bends,"spring_tip_displacements":sx,"negative_multidof_spring":{"accepted":na.is_ok(),"error":na.as_ref().err().map(|e|e.to_string()),"prepared_spring_count":na.as_ref().ok().map(|a|a.prepared_boundary.springs.len()),"uy_diagonal":na.as_ref().ok().map(|a|a.stiffness_with_springs[7][7])},"overflow_stiffness_nonfinite_count":overflow.iter().flatten().filter(|x|!x.is_finite()).count(),"invalid_station_point_accepted":invalid_point.is_ok()});println!("{}",serde_json::to_string_pretty(&out).unwrap());
}
