560: /// Power-of-two scale in bounded steps; reject underflow/overflow instead of erasing terms.
561: pub fn radix_scale(mut value: f64, mut exponent: i32) -> Result<f64, StructuralError> {
562:     if !value.is_finite() {
563:         return Err(StructuralError::Range("nonfinite radix input"));
564:     }
565:     if value == 0.0 {
566:         return Ok(value);
567:     }
568:     while exponent != 0 {
569:         let step = exponent.clamp(-512, 512);
570:         value *= 2.0_f64.powi(step);
571:         exponent -= step;
572:         if !value.is_finite() || value == 0.0 || value.is_subnormal() {
573:             return Err(StructuralError::Range("radix scaling loses normal range"));
574:         }
575:     }
576:     Ok(value)
577: }
578: pub(crate) fn binary_exponent(value: f64) -> i32 {
579:     let bits = value.abs().to_bits();
580:     let e = ((bits >> 52) & 0x7ff) as i32;
581:     if e != 0 {
582:         e - 1023
583:     } else {
584:         -1074 + (63 - (bits & ((1_u64 << 52) - 1)).leading_zeros() as i32)
585:     }
586: }

589:     let x = a * b;

591:         Err(StructuralError::Range("product overflow or underflow"))
592:     } else {
593:         Ok(x)
594:     }

611: fn validate(system: &StructuralSystem<'_>) -> Result<(), StructuralError> {
612:     let n = system.force.len();
613:     if system.stiffness.len() != n || system.stiffness.iter().any(|r| r.len() != n) {
614:         return Err(StructuralError::InvalidInput("matrix dimensions"));
615:     }
616:     if system
617:         .force
618:         .iter()
619:         .chain(system.stiffness.iter().flatten())
620:         .any(|x| !x.is_finite())
621:     {
622:         return Err(StructuralError::InvalidInput(
623:             "nonfinite original equations",
624:         ));
625:     }
626:     if let Some(evidence) = &system.symmetry {
627:         if evidence.basis.is_empty()
628:             || evidence.absolute_roundoff.len() != n
629:             || evidence.operation_counts.len() != n
630:             || evidence
631:                 .absolute_roundoff
632:                 .iter()
633:                 .any(|r| r.len() != n || r.iter().any(|v| !v.is_finite() || *v < 0.0))
634:             || evidence.operation_counts.iter().any(|r| r.len() != n)
635:         {
636:             return Err(StructuralError::InvalidInput("symmetry provenance"));
637:         }
638:     }
639:     // Audit BOTH triangles of full retained equations, including coupling to prescribed DOFs.
640:     for i in 0..n {
641:         for j in 0..i {
642:             let a = system.stiffness[i][j];
643:             let b = system.stiffness[j][i];
644:             if a == b {
645:                 continue;
646:             }
647:             let scale = a.abs().max(b.abs());
648:             let skew = (a / scale - b / scale).abs();
649:             let allowed = system.symmetry.as_ref().map_or(0.0, |e| {
650:                 e.absolute_roundoff[i][j] / scale + e.absolute_roundoff[j][i] / scale
651:             });
652:             if skew > allowed {
653:                 return Err(StructuralError::Asymmetric {
654:                     row: i,
655:                     col: j,
656:                     relative_skew: skew,
657:                 });
658:             }
659:         }
660:     }
661:     let mut seen = vec![false; n];
662:     for &i in system.free_dofs {
663:         if i >= n || seen[i] {
664:             return Err(StructuralError::InvalidInput("free map"));
665:         }
666:         seen[i] = true;
667:     }
668:     for &(i, v) in system.prescribed {
669:         if i >= n || seen[i] || !v.is_finite() {
670:             return Err(StructuralError::InvalidInput("prescribed map"));
671:         }
672:         seen[i] = true;
673:     }
674:     if seen.iter().any(|x| !*x) {
675:         return Err(StructuralError::InvalidInput("incomplete constraint map"));
676:     }
677:     Ok(())
678: }
679: fn validate_prepared(prepared: &PreparedSystem<'_>) -> Result<(), StructuralError> {
680:     validate(prepared.source)?;
681:     let n = prepared.source.free_dofs.len();
682:     if prepared.matrix.len() != n
683:         || prepared.rhs.len() != n
684:         || prepared.scale_exponents.len() != n
685:         || prepared
686:             .matrix
687:             .iter()
688:             .any(|r| r.len() != n || r.iter().any(|v| !v.is_finite()))
689:         || prepared.rhs.iter().any(|v| !v.is_finite())
690:     {
691:         return Err(StructuralError::InvalidInput(
692:             "prepared shape/map/finiteness",
693:         ));
694:     }

697:             if prepared.matrix[i][j] != prepared.matrix[j][i] {

699:             }
700:         }
701:     }

703: }
704: /// Exact nonoverlapping expansion of represented inputs within checked IEEE range.

706: #[derive(Debug, Clone, Default)]
707: pub(crate) struct Expansion {
708:     pub(crate) terms: Vec<f64>,
709:     pub(crate) operations: usize,
710: }
711: impl Expansion {
712:     pub(crate) fn add(&mut self, value: f64) -> Result<(), StructuralError> {
713:         if !value.is_finite() {
714:             return Err(StructuralError::Range("expansion input"));
715:         }
716:         let mut q = value;
717:         let mut next = Vec::new();
718:         for &e in &self.terms {
719:             let sum = q + e;
720:             let z = sum - q;
721:             let low = (q - (sum - z)) + (e - z);
722:             if !sum.is_finite() || !z.is_finite() || !low.is_finite() {
723:                 return Err(StructuralError::Range("expansion addition"));
724:             }
725:             if low != 0.0 {
726:                 next.push(low);
727:             }
728:             q = sum;
729:             self.operations += 6;
730:         }
731:         if q != 0.0 {
732:             next.push(q);
733:         }
734:         self.terms = next;
735:         Ok(())
736:     }
737:     pub(crate) fn add_product(
738:         &mut self,
739:         a: f64,
740:         b: f64,
741:         exponent: i32,
742:     ) -> Result<(), StructuralError> {
743:         if a == 0.0 || b == 0.0 {
744:             return Ok(());
745:         }
746:         if !a.is_finite() || !b.is_finite() {
747:             return Err(StructuralError::Range("expansion product input"));
748:         }
749:         let ea = binary_exponent(a);
750:         let eb = binary_exponent(b);
751:         let ma = exact_radix(a, -ea)?;
752:         let mb = exact_radix(b, -eb)?;
753:         let hi = ma * mb;
754:         let lo = ma.mul_add(mb, -hi);
755:         self.operations += 2;
756:         self.add(exact_radix(lo, ea + eb + exponent)?)?;
757:         self.add(exact_radix(hi, ea + eb + exponent)?)?;
758:         Ok(())
759:     }
760:     pub(crate) fn is_zero(&self) -> bool {
761:         self.terms.is_empty()
762:     }
763:     /// The expansion's exact value, correctly rounded once (+0.0 for zero).
764:     pub(crate) fn round(&self) -> Result<f64, StructuralError> {
765:         crate::exact_sum::exact_rounded_sum(self.terms.iter().copied()).map_err(sum_range)
766:     }
767:     pub(crate) fn exact_scalar(&self) -> Result<Option<f64>, StructuralError> {
768:         let value = self.round()?;
769:         let mut difference = self.clone();
770:         difference.add(-value)?;
771:         Ok(if difference.is_zero() {
772:             Some(value)
773:         } else {
774:             None
775:         })
776:     }
777: }
778: fn sum_range(error: SumError) -> StructuralError {
779:     match error {
780:         SumError::NonFinite => StructuralError::Range("exact sum input"),
781:         SumError::AccumulatorOverflow => StructuralError::Range("exact sum accumulator"),
782:         SumError::NonRepresentable => StructuralError::Range("exact sum outside binary64 range"),
783:     }
784: }
785: pub(crate) fn exact_radix(mut value: f64, mut exponent: i32) -> Result<f64, StructuralError> {
786:     if !value.is_finite() {
787:         return Err(StructuralError::Range("exact radix input"));
788:     }
789:     while exponent != 0 && value != 0.0 {
790:         let step = exponent.clamp(-512, 512);
791:         let next = value * 2.0_f64.powi(step);
792:         if !next.is_finite() || next == 0.0 || next * 2.0_f64.powi(-step) != value {
793:             return Err(StructuralError::Range("exact radix loses represented bits"));
794:         }
795:         value = next;
796:         exponent -= step;
797:     }
798:     Ok(value)
799: }
800: fn contribution_sums(
801:     system: &StructuralSystem<'_>,
802: ) -> Result<Option<Vec<Vec<Expansion>>>, StructuralError> {
803:     let Some(entries) = system.contributions else {
804:         return Ok(None);
805:     };
806:     let n = system.force.len();
807:     let mut sums = vec![vec![Expansion::default(); n]; n];
808:     for entry in entries {
809:         if entry.row >= n || entry.col >= n || !entry.value.is_finite() {
810:             return Err(StructuralError::InvalidInput("contribution"));
811:         }
812:         let sum = &mut sums[entry.row][entry.col];
813:         let old = sum.round()?;
814:         sum.add(entry.value)?;
815:         if entry.row == entry.col && entry.value > 0.0 && old != 0.0 && sum.round()? == old {
816:             return Err(unresolved(
817:                 "positive diagonal contribution absorbed by assembly; stabilization unresolved",
818:                 Some(entry.row),
819:             ));
820:         }
821:     }
822:     Ok(Some(sums))
823: }
824: fn contribution_differences(
825:     system: &StructuralSystem<'_>,
826:     sums: &[Vec<Expansion>],
827: ) -> Result<Vec<Vec<Expansion>>, StructuralError> {
828:     let mut differences = sums.to_vec();
829:     for (i, row) in differences.iter_mut().enumerate() {
830:         for (j, value) in row.iter_mut().enumerate() {
831:             value.add(-system.stiffness[i][j])?;
832:         }
833:     }
834:     Ok(differences)
835: }
836: fn audit_contributions(
837:     system: &StructuralSystem<'_>,

840: ) -> Result<(f64, f64, Vec<ContributionRounding>), StructuralError> {
841:     let Some(sums) = contribution_sums(system)? else {

843:     };
844:     let differences = contribution_differences(system, &sums)?;
845:     let mut rounding = Vec::new();
846:     for (i, row) in differences.iter().enumerate() {

855:                     stored_difference_high: difference.terms.last().copied().unwrap_or(0.0),

857:                     accumulated_expansion: terms.clone(),
858:                     difference_expansion: difference.terms.clone(),

860:             }
861:         }
862:     }
863:     let mut perturbation_norm: f64 = 0.0;
864:     let mut matrix_norm: f64 = 0.0;
865:     for (c, &j) in system.free_dofs.iter().enumerate() {
866:         let mut delta_column = 0.0;
867:         let mut matrix_column = 0.0;
868:         for (r, &i) in system.free_dofs.iter().enumerate() {
869:             for &part in &differences[i][j].terms {
870:                 delta_column = checked_value(
871:                     delta_column + exact_radix(part, exponents[r] + exponents[c])?.abs(),
872:                 )?;
873:             }
874:             matrix_column = checked_value(
875:                 matrix_column
876:                     + radix_scale(system.stiffness[i][j], exponents[r] + exponents[c])?.abs(),
877:             )?;
878:         }
879:         perturbation_norm = perturbation_norm.max(delta_column);
880:         matrix_norm = matrix_norm.max(matrix_column);
881:     }
882:     let mut delta_rhs_norm = 0.0;
883:     let mut rhs_norm = 0.0;
884:     for (r, &i) in system.free_dofs.iter().enumerate() {
885:         let mut delta = Expansion::default();
886:         for &(j, value) in system.prescribed {
887:             for &part in &differences[i][j].terms {
888:                 delta.add_product(-part, value, exponents[r])?;
889:             }
890:         }
891:         delta_rhs_norm =
892:             checked_value(delta_rhs_norm + delta.terms.iter().map(|v| v.abs()).sum::<f64>())?;
893:         rhs_norm = checked_value(rhs_norm + rhs[r].abs())?;
894:     }
895:     if delta_rhs_norm != 0.0 && rhs_norm == 0.0 {
896:         return Err(unresolved(
897:             "contribution-preserved prescribed coupling changes zero reduced load",
898:             None,
899:         ));
900:     }
901:     Ok((
902:         if matrix_norm == 0.0 {
903:             0.0
904:         } else {
905:             perturbation_norm / matrix_norm
906:         },
907:         if rhs_norm == 0.0 {
908:             0.0
909:         } else {
910:             delta_rhs_norm / rhs_norm
911:         },
912:         rounding,
913:     ))
914: }
915: /// Componentwise action of the complete intended contributions, including K_fc*u_c.
916: /// Exact expansion numerator; denominator and guard use their separately recorded
917: /// operation count. This does not establish forward accuracy of a soft mode.
918: fn audit_intended_action<R: Represented + ?Sized>(
919:     equations: &R,
920:     u: &[f64],
921: ) -> Result<Vec<ResidualRow>, StructuralError> {
922:     let Some(sums) = equations.contribution_sums()? else {
923:         return Ok(Vec::new());
924:     };
925:     let system = equations.view();
926:     let mut rows = Vec::new();
927:     for &i in system.free_dofs {
928:         // K1: `sums_row(i)` yields (j, exact expansion of K_ij) in ascending
929:         // column order; an empty expansion takes no part below.
930:         let mut exponent = if system.force[i] == 0.0 {
931:             i32::MIN
932:         } else {
933:             binary_exponent(system.force[i])
934:         };
935:         for (j, coefficient) in equations.sums_row(&sums, i) {
936:             if u[j] != 0.0 {
937:                 for &part in &coefficient.terms {
938:                     exponent = exponent.max(binary_exponent(part) + binary_exponent(u[j]) + 1);
939:                 }
940:             }
941:         }
942:         if exponent == i32::MIN {
943:             exponent = 0;
944:         }
945:         let mut residual = Expansion::default();
946:         residual.add(exact_radix(-system.force[i], -exponent)?)?;
947:         let mut denominator = exact_radix(system.force[i].abs(), -exponent)?;
948:         let mut denominator_operations = 0;
949:         for (j, coefficient) in equations.sums_row(&sums, i) {
950:             for &part in &coefficient.terms {
951:                 residual.add_product(part, u[j], -exponent)?;
952:                 // Positive sum over exact expansion components is an explicit estimate
953:                 // of |coalesced intended K|*|u|, not original unconsolidated contribution magnitudes.
954:                 denominator = checked_value(
955:                     denominator + normalized_product(part.abs(), u[j].abs(), exponent)?,
956:                 )?;
957:                 denominator_operations += 2;
958:             }
959:         }
960:         let operations = (residual.operations + denominator_operations + 2).max(2);
961:         let g = gamma(operations);
962:         // Zero witness (S11 D-S11-1): an exact-zero intended-action residual keeps
963:         // the -0.0 that committed diagnostics publish (245 `normalized_residual:
964:         // -0.0` values in 31 fixture files, e.g. n05-dense_scrutiny.raw.json).
965:         // It is only rendered in Debug diagnostics and gated through abs(); it
966:         // never feeds a bit-equality check or a receipt. Every other zero is +0.0.
967:         let r = if residual.is_zero() {
968:             -0.0
969:         } else {
970:             residual.round()?
971:         };
972:         let allowance = g * denominator / (1.0 - g);
973:         let ratio = if denominator == 0.0 {
974:             if residual.is_zero() {
975:                 0.0
976:             } else {
977:                 f64::INFINITY
978:             }
979:         } else {
980:             (r.abs() / denominator + g / (1.0 - g)) * (1.0 + g) * (1.0 + 4.0 * f64::EPSILON)
981:         };
982:         let target = 64.0 * g;
983:         rows.push(ResidualRow {global_dof:i,residual:physical_residual_record(r,exponent)?,denominator:physical_residual_record(denominator,exponent)?,row_scale_exponent:exponent,normalized_residual:r,normalized_denominator:denominator,normalized_evaluation_allowance:allowance,normalization_basis:"exact contribution expansions and full intended free action including prescribed DOFs",operation_count:operations,evaluation_allowance:physical_residual_record(allowance,exponent)?,guarded_ratio:ratio,target,passed:ratio<=target});
984:     }
985:     Ok(rows)
986: }

1360:             let average = a[i][j] / 2.0 + a[j][i] / 2.0;

1386: fn normalized_product(a: f64, b: f64, row_exponent: i32) -> Result<f64, StructuralError> {
1387:     if a == 0.0 || b == 0.0 {
1388:         return Ok(0.0);
1389:     }
1390:     let ea = binary_exponent(a);
1391:     let eb = binary_exponent(b);
1392:     let ma = radix_scale(a, -ea)?;
1393:     let mb = radix_scale(b, -eb)?;
1394:     radix_scale(checked_product(ma, mb)?, ea + eb - row_exponent)
1395: }

1428: fn evaluate_original_residual_bound<R: Represented + ?Sized>(
1429:     equations: &R,
1430:     binding: ForceBinding<'_>,
1431:     u: &[f64],
1432: ) -> Result<Vec<ResidualRow>, StructuralError> {
1433:     equations.validate_equations()?;
1434:     let system = equations.view();
1435:     if u.len() != system.force.len() || u.iter().any(|x| !x.is_finite()) {
1436:         return Err(StructuralError::InvalidInput("displacement vector"));
1437:     }
1438:     if system.prescribed.iter().any(|&(i, v)| u[i] != v) {
1439:         return Err(StructuralError::InvalidInput(
1440:             "prescribed displacement incompatibility",
1441:         ));
1442:     }
1443:     let mut rows = Vec::new();
1444:     for &i in system.free_dofs {
1445:         // K1: `row(i)` yields (j, K_ij) in ascending column order, zeros
1446:         // included (dense) or left out (sparse); a zero takes no part below.
1447:         let mut row_exponent = if system.force[i] == 0.0 {
1448:             i32::MIN
1449:         } else {
1450:             binary_exponent(system.force[i])
1451:         };
1452:         for (j, k) in equations.row(i) {
1453:             let x = u[j];
1454:             if k != 0.0 && x != 0.0 {
1455:                 row_exponent = row_exponent.max(binary_exponent(k) + binary_exponent(x) + 1);
1456:             }
1457:         }
1458:         if row_exponent == i32::MIN {
1459:             row_exponent = 0;
1460:         }
1461:         let mut r = -radix_scale(system.force[i], -row_exponent)?;
1462:         let mut d = r.abs();
1463:         let mut count = 0;
1464:         // KS3: on a row coupled to a nonzero prescribed value the numerator is
1465:         // one exact sum (ledger terms negated, plus K_ij * u_j over the full
1466:         // row), scaled before its single rounding. Other rows: today's code.
1467:         let exact_numerator = !binding.binary64() && equations.prescribed_coupled(i);
1468:         let mut numerator = ExactAccumulator::new();
1469:         if exact_numerator {
1470:             match binding.assembled() {
1471:                 Some(force) => force
1472:                     .accumulate_dof(i, &mut numerator, true)
1473:                     .map_err(sum_range)?,
1474:                 None => numerator.add(-system.force[i]).map_err(sum_range)?,
1475:             }
1476:         }
1477:         for (j, k) in equations.row(i) {
1478:             let x = u[j];
1479:             if k == 0.0 {
1480:                 continue;
1481:             }
1482:             count += 1;
1483:             let p = normalized_product(k, x, row_exponent)?;
1484:             if exact_numerator {
1485:                 numerator.add_product(k, x).map_err(sum_range)?;
1486:             } else {
1487:                 r = checked_value(r + p)?;
1488:             }
1489:             d = checked_value(d + p.abs())?;
1490:         }
1491:         if exact_numerator {
1492:             r = checked_value(numerator.round_scaled(-row_exponent).map_err(sum_range)?)?;
1493:         }
1494:         let operations = 2 * count + 2;
1495:         let g = gamma(operations);
1496:         let allowance = g * d / (1.0 - g);
1497:         let guarded = if d == 0.0 {
1498:             if r == 0.0 {
1499:                 0.0
1500:             } else {
1501:                 f64::INFINITY
1502:             }
1503:         } else {
1504:             (r.abs() / d + g / (1.0 - g)) * (1.0 + g) * (1.0 + 4.0 * f64::EPSILON)
1505:         };
1506:         let target = 64.0 * g;
1507:         rows.push(ResidualRow {
1508:             global_dof: i,
1509:             residual: physical_residual_record(r, row_exponent)?,
1510:             denominator: physical_residual_record(d, row_exponent)?,
1511:             row_scale_exponent: row_exponent,
1512:             normalized_residual: r,
1513:             normalized_denominator: d,
1514:             normalized_evaluation_allowance: allowance,
1515:             normalization_basis:
1516:                 "radix mantissas; one multiply/add per term; physical records descriptive",
1517:             operation_count: operations,
1518:             evaluation_allowance: physical_residual_record(allowance, row_exponent)?,
1519:             guarded_ratio: guarded,
1520:             target,
1521:             passed: guarded <= target,
1522:         });
1523:     }
1524:     Ok(rows)
1525: }

1553: pub fn estimate_rcond<F>(a: &[Vec<f64>], solve: &F) -> Result<f64, StructuralError>

1555:     F: Fn(&[f64]) -> Result<Vec<f64>, StructuralError>,

1559: /// K1: the matrix the condition estimate reads, dense or sparse. The estimate
1560: /// itself (`estimate_rcond` below, the body of the public function above) is
1561: /// written once; a representation supplies its shape check and its columns.
1562: trait ConditionMatrix {
1563:     fn dimension(&self) -> usize;
1564:     /// Every row has `dimension()` finite entries.
1565:     fn square_and_finite(&self) -> bool;
1566:     /// Column `j`'s entries in ascending row order. Zeros may be left out:
1567:     /// they add nothing to the column's 1-norm.
1568:     fn column(&self, j: usize) -> impl Iterator<Item = f64> + '_;
1569:     fn estimate_rcond<F>(&self, solve: &F) -> Result<f64, StructuralError>
1570:     where
1571:         F: Fn(&[f64]) -> Result<Vec<f64>, StructuralError>,
1572:     {
1573:         let n = self.dimension();
1574:         if !self.square_and_finite() {
1575:             return Err(StructuralError::InvalidInput("condition matrix"));
1576:         }
1577:         if n == 0 {
1578:             return Ok(1.0);
1579:         }
1580:         let mut norm: f64 = 0.0;
1581:         for j in 0..n {
1582:             let mut sum = 0.0;
1583:             for value in self.column(j) {
1584:                 sum = checked_value(sum + value.abs())?;
1585:             }
1586:             norm = norm.max(sum);
1587:         }
1588:         let mut x = vec![1.0 / n as f64; n];
1589:         let mut estimate: f64 = 0.0;
1590:         let mut previous = n;
1591:         for _ in 0..5 {
1592:             let y = solve(&x)?;
1593:             if y.len() != n || y.iter().any(|v| !v.is_finite()) {
1594:                 return Err(StructuralError::InvalidInput("condition solve output"));
1595:             }
1596:             let current = checked_value(y.iter().map(|v| v.abs()).sum())?;
1597:             if current <= estimate && estimate != 0.0 {
1598:                 break;
1599:             }
1600:             estimate = current;
1601:             let signs: Vec<f64> = y
1602:                 .iter()
1603:                 .map(|&v| if v >= 0.0 { 1.0 } else { -1.0 })
1604:                 .collect();
1605:             let z = solve(&signs)?;
1606:             if z.len() != n || z.iter().any(|v| !v.is_finite()) {
1607:                 return Err(StructuralError::InvalidInput(
1608:                     "condition transpose solve output",
1609:                 ));
1610:             }
1611:             let j = (0..n)
1612:                 .max_by(|&i, &j| z[i].abs().total_cmp(&z[j].abs()))
1613:                 .unwrap();
1614:             let dot: f64 = z.iter().zip(&x).map(|(a, b)| a * b).sum();
1615:             checked_value(dot)?;
1616:             if z[j].abs() <= dot || j == previous {
1617:                 break;
1618:             }
1619:             previous = j;
1620:             x.fill(0.0);
1621:             x[j] = 1.0;
1622:         }
1623:         let alt: Vec<f64> = (0..n)
1624:             .map(|i| {
1625:                 if n == 1 {
1626:                     1.0
1627:                 } else {
1628:                     (if i % 2 == 0 { 1.0 } else { -1.0 }) * (1.0 + i as f64 / (n - 1) as f64)
1629:                 }
1630:             })
1631:             .collect();
1632:         let y = solve(&alt)?;
1633:         if y.len() != n || y.iter().any(|v| !v.is_finite()) {
1634:             return Err(StructuralError::InvalidInput("condition safeguard output"));
1635:         }
1636:         let alt_est =
1637:             checked_value(y.iter().map(|v| v.abs()).sum::<f64>() * 2.0 / (3.0 * n as f64))?;
1638:         estimate = estimate.max(alt_est);
1639:         let rcond = 1.0 / checked_product(norm, estimate)?;
1640:         if !rcond.is_finite() || rcond <= f64::EPSILON {
1641:             return Err(unresolved(
1642:                 "scaled condition estimate at working-precision boundary",
1643:                 None,
1644:             ));
1645:         }
1646:         Ok(rcond.min(1.0))
1647:     }
1648: }

1663: /// Common completion for EVERY structural backend; no finite-LU bypass.
1664: /// K1: written once over the representation (`Represented` for the source
1665: /// equations, `PreparedGate` for the prepared system), so the dense and the
1666: /// sparse paths run the same completion.
1667: fn finish_checked_factor<R, P, F>(
1668:     equations: &R,
1669:     prepared: &P,
1670:     pivots: Vec<PivotEvidence>,
1671:     factorization: &'static str,
1672:     solve: F,
1673: ) -> Result<StructuralSolution, StructuralError>
1674: where
1675:     R: Represented + ?Sized,
1676:     P: PreparedGate + ?Sized,
1677:     F: Fn(&[f64]) -> Result<Vec<f64>, StructuralError>,
1678: {
1679:     equations.validate_equations()?;
1680:     let system = equations.view();
1681:     let n = system.free_dofs.len();
1682:     if !prepared.gate_shape_matches(n) {
1683:         return Err(StructuralError::InvalidInput("prepared dimensions"));
1684:     }
1685:     let rcond = prepared.gate_rcond(&solve)?;
1686:     let (relative_perturbation, load_perturbation) = prepared.gate_assembly();
1687:     let amplification = (relative_perturbation + load_perturbation) / rcond;
1688:     if !amplification.is_finite() || amplification >= 1.0 {
1689:         return Err(unresolved(
1690:             "assembly perturbation amplification unresolved",
1691:             None,
1692:         ));
1693:     }
1694:     let scale_exponents = prepared.gate_scale_exponents();
1695:     let force_binding = prepared.gate_binding();
1696:     let mut y = solve(prepared.gate_rhs())?;
1697:     if y.len() != n || y.iter().any(|v| !v.is_finite()) {
1698:         return Err(StructuralError::InvalidInput("structural backend output"));
1699:     }
1700:     let mut attempts = 0;
1701:     let mut prior = f64::INFINITY;
1702:     loop {
1703:         let mut u = vec![0.0; system.force.len()];
1704:         for &(i, v) in system.prescribed {
1705:             u[i] = v;
1706:         }
1707:         for (r, &i) in system.free_dofs.iter().enumerate() {
1708:             u[i] = radix_scale(y[r], scale_exponents[r])?;
1709:         }
1710:         let residual = evaluate_original_residual_bound(equations, force_binding, &u)?;
1711:         let worst = residual
1712:             .iter()
1713:             .map(|r| r.guarded_ratio / r.target)
1714:             .fold(0.0, f64::max);
1715:         if residual.iter().all(|r| r.passed) {
1716:             let intended = audit_intended_action(equations, &u)?;
1717:             if intended.iter().any(|r| !r.passed) {
1718:                 return Err(unresolved(
1719:                     "contribution-preserved intended free action failed",
1720:                     None,
1721:                 ));
1722:             }
1723:             // S11 section 6: a detected load loss is Sensitive, never a refusal.
1724:             let load_fidelity = match force_binding.audit_terms() {
1725:                 // RV1-N5: an audit failure is never propagated; it is a
1726:                 // Sensitive report (section 6).
1727:                 Some(terms) => match equations.load_fidelity(&u, terms) {
1728:                     Ok(report) => (!report.rows.is_empty()).then_some(report),
1729:                     Err(error) => Some(LoadFidelityReport {
1730:                         rows: Vec::new(),
1731:                         audit_error: Some(format!("{error:?}")),
1732:                     }),
1733:                 },
1734:                 None => None,
1735:             };
1736:             let ordinary_sensitive = rcond < f64::EPSILON.sqrt() || load_fidelity.is_some();
1737:             // K-D5 (D1 §4.3.1): a case that would publish Passed is demoted to
1738:             // Sensitive when the formation check's estimate exceeds the
1739:             // criterion or the check cannot re-form it. Values never change.
1740:             let formation_check = match prepared.gate_formation() {
1741:                 Some(source) if !ordinary_sensitive => equations.formation_check(
1742:                     source,
1743:                     force_binding.audit_terms(),
1744:                     scale_exponents,
1745:                     &u,
1746:                     &solve,
1747:                 ),
1748:                 _ => None,
1749:             };
1750:             let (symmetry_projection_performed, maximum_scaled_skew, symmetry_basis) =
1751:                 prepared.gate_symmetry();
1752:             return Ok(StructuralSolution {
1753:                 displacements: u,
1754:                 load_fidelity: load_fidelity.clone(),
1755:                 formation_check: formation_check.clone(),
1756:                 report: StructuralReport {
1757:                     policy: POLICY,
1758:                     quality: if ordinary_sensitive || formation_check.is_some() {
1759:                         SolveQuality::Sensitive
1760:                     } else {
1761:                         SolveQuality::Passed
1762:                     },
1763:                     factorization,
1764:                     scale_exponents: scale_exponents.to_vec(),
1765:                     pivots,
1766:                     condition_estimator: CONDITION_ESTIMATOR,
1767:                     reciprocal_condition_estimate: rcond,
1768:                     residual_rows: residual,
1769:                     refinement_attempts: attempts,
1770:                     contribution_audit_performed: equations.contributions_supplied(),
1771:                     contribution_rounding: prepared.gate_contribution_rounding().to_vec(),
1772:                     assembly_relative_perturbation_estimate: relative_perturbation,
1773:                     assembly_amplification_estimate: amplification,
1774:                     assembly_load_perturbation_estimate: load_perturbation,
1775:                     intended_residual_rows: intended,
1776:                     symmetry_projection_performed,
1777:                     maximum_scaled_skew,
1778:                     symmetry_basis,
1779:                 },
1780:             });
1781:         }
1782:         if attempts == 3 || worst >= prior {
1783:             return Err(unresolved(
1784:                 "original-equation residual failed after bounded refinement",
1785:                 None,
1786:             ));
1787:         }
1788:         prior = worst;
1789:         let mut correction = Vec::new();
1790:         for (i, row) in residual.iter().enumerate() {
1791:             correction.push(radix_scale(
1792:                 -row.normalized_residual,
1793:                 row.row_scale_exponent + scale_exponents[i],
1794:             )?);
1795:         }
1796:         let delta = solve(&correction)?;
1797:         if delta.len() != n || delta.iter().any(|v| !v.is_finite()) {
1798:             return Err(StructuralError::InvalidInput("refinement backend output"));
1799:         }
1800:         for (v, d) in y.iter_mut().zip(delta) {
1801:             *v = checked_value(*v + d)?;
1802:         }
1803:         attempts += 1;
1804:     }
1805: }

1810: impl CholeskyFactor {
1811:     pub fn solve(&self, rhs: &[f64]) -> Result<Vec<f64>, StructuralError> {
1812:         let n = self.lower.len();
1813:         if rhs.len() != n {
1814:             return Err(StructuralError::InvalidInput("triangular RHS"));
1815:         }
1816:         let mut x = rhs.to_vec();
1817:         for i in 0..n {
1818:             for j in 0..i {
1819:                 x[i] = checked_value(x[i] - checked_product(self.lower[i][j], x[j])?)?;
1820:             }
1821:             x[i] = checked_quotient(x[i], self.lower[i][i])?;
1822:         }
1823:         for i in (0..n).rev() {
1824:             for j in i + 1..n {
1825:                 x[i] = checked_value(x[i] - checked_product(self.lower[j][i], x[j])?)?;
1826:             }
1827:             x[i] = checked_quotient(x[i], self.lower[i][i])?;
1828:         }
1829:         Ok(x)
1830:     }
1831: }
1832: fn cholesky(
1833:     prepared: &PreparedSystem,
1834:     free_dofs: &[usize],
1835: ) -> Result<(CholeskyFactor, Vec<PivotEvidence>), StructuralError> {
1836:     let n = prepared.matrix.len();
1837:     if free_dofs.len() != n
1838:         || prepared
1839:             .matrix
1840:             .iter()
1841:             .any(|row| row.len() != n || row.iter().any(|v| !v.is_finite()))
1842:     {
1843:         return Err(StructuralError::InvalidInput("Cholesky matrix/free map"));
1844:     }
1845:     for i in 0..n {
1846:         for j in 0..i {
1847:             if prepared.matrix[i][j] != prepared.matrix[j][i] {
1848:                 return Err(StructuralError::InvalidInput(
1849:                     "Cholesky requires audited symmetric matrix",
1850:                 ));
1851:             }
1852:         }
1853:     }
1854:     let mut l = vec![vec![0.0; n]; n];
1855:     let mut evidence = Vec::new();
1856:     for i in 0..n {
1857:         for j in 0..=i {
1858:             let mut sum = prepared.matrix[i][j];
1859:             let mut scale = sum.abs();
1860:             for k in 0..j {
1861:                 let term = checked_product(l[i][k], l[j][k])?;
1862:                 sum = checked_value(sum - term)?;
1863:                 scale = checked_value(scale + term.abs())?;
1864:             }
1865:             if i == j {
1866:                 evidence.push(screen_pivot(i, free_dofs[i], sum, scale, 2 * j + 2)?);
1867:                 l[i][j] = sum.sqrt();
1868:             } else {
1869:                 l[i][j] = checked_quotient(sum, l[j][j])?;
1870:             }
1871:         }
1872:     }
1873:     Ok((CholeskyFactor { lower: l }, evidence))
1874: }

2204: /// KF2: `negative_pair_witness` in O(n^2), with its work counts (pairs
2205: /// visited, cell terms evaluated, verifications). It visits the pairs (i, j < i)
2206: /// in the order of the search it replaces, and for each builds no direction:
2207: /// the direction e_i + sign*e_j has only four cells, (j,j), (j,i), (i,j) and
2208: /// (i,i), and they are exactly the cells `verify_negative_direction` evaluates

2210: /// it fails with the verifier's error, or reaches the verifier's verdict, bit
2211: /// for bit. Only a pair whose verdict is a witness is built and passed to the
2212: /// unchanged verifier, which publishes the witness (once, in O(n^2)).
2213: fn negative_pair_witness_counted(
2214:     prepared: &PreparedSystem<'_>,
2215: ) -> Result<(Option<StructuralError>, usize, usize, usize), StructuralError> {
2216:     validate_prepared(prepared)?;
2217:     let system = prepared.source;
2218:     let n = prepared.matrix.len();
2219:     let mut visited = 0;
2220:     let mut evaluated = 0;
2221:     let mut verified = 0;
2222:     for i in 0..n {
2223:         for j in 0..i {
2224:             visited += 1;
2225:             let sign = if prepared.matrix[i][j] >= 0.0 {
2226:                 -1.0
2227:             } else {
2228:                 1.0
2229:             };
2230:             // (cell, v[a], v[b]) in the verifier's row-major order (j < i).
2231:             let cells = [
2232:                 (j, j, sign, sign),
2233:                 (j, i, sign, 1.0),
2234:                 (i, j, 1.0, sign),
2235:                 (i, i, 1.0, 1.0),
2236:             ];
2237:             let mut energy = 0.0;
2238:             let mut magnitude = 0.0;
2239:             let mut terms = 0;
2240:             for (a, b, da, db) in cells {
2241:                 if prepared.matrix[a][b] == 0.0 {
2242:                     continue;
2243:                 }
2244:                 let original = radix_scale(
2245:                     system.stiffness[system.free_dofs[a]][system.free_dofs[b]],
2246:                     prepared.scale_exponents[a] + prepared.scale_exponents[b],
2247:                 )?;
2248:                 let term = checked_product(checked_product(da, original)?, db)?;
2249:                 energy = checked_value(energy + term)?;
2250:                 magnitude = checked_value(magnitude + term.abs())?;
2251:                 terms += 1;
2252:             }
2253:             evaluated += terms;
2254:             let allowance = 64.0 * gamma(3 * terms + 2) * magnitude;
2255:             if energy < -allowance {
2256:                 verified += 1;
2257:                 let mut v = vec![0.0; n];
2258:                 v[i] = 1.0;
2259:                 v[j] = sign;
2260:                 if let Some(witness) = verify_negative_direction(prepared, &v)? {
2261:                     return Ok((Some(witness), visited, evaluated, verified));
2262:                 }
2263:             }
2264:         }
2265:     }
