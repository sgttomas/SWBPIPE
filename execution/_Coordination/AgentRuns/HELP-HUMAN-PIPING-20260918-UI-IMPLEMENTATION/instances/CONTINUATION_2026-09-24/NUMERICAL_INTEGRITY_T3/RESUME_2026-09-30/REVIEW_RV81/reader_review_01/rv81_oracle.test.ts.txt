// arithmetic with RV81's Python exact-rational vectors and runs targeted gate probes.
import { describe, expect, it } from 'vitest';
import { readFileSync, writeFileSync } from 'node:fs';
import corpusText from '../../../../../fixtures/results/retained_precision_cases.json?raw';
import { absoluteBound, upwardProduct, upwardSmallSum, binary64Bits, decodeBinary64, phi512, eHat, stopFeasible, validateRetainedPrecision, accountingRules } from './retainedPrecision';
import { canonicalSha256HexCheckedV1 } from '../../services/hashService';
const SCRATCH = process.env.RV81_SCRATCH!;
const vectors = JSON.parse(readFileSync(SCRATCH + '/oracle_vectors.json', 'utf8'));
const corpus = JSON.parse(corpusText);
const fromWord = (h: string) => { const v = new DataView(new ArrayBuffer(8)); v.setBigUint64(0, BigInt('0x' + h)); return v.getFloat64(0); };
const tryBits = (f: () => number) => { try { return binary64Bits(f()); } catch { return null; } };

describe('RV81 independent arithmetic oracle', () => {
  it('compares every vector and records mismatches', () => {
    const mismatches: unknown[] = []; const counts: Record<string, number> = {};
    for (const [a, b, want] of vectors.products) { counts.products = (counts.products ?? 0) + 1; const got = tryBits(() => upwardProduct(fromWord(a), fromWord(b))); if (got !== want) mismatches.push(['product', a, b, want, got]); }
    for (const [a, b, want] of vectors.small_sums) { counts.small_sums = (counts.small_sums ?? 0) + 1; const got = tryBits(() => upwardSmallSum(fromWord(a), fromWord(b))); if (got !== want) mismatches.push(['small_sum', a, b, want, got]); }
    for (const [v, s, want] of vectors.bounds) { counts.bounds = (counts.bounds ?? 0) + 1; const got = tryBits(() => absoluteBound(fromWord(v), fromWord(s))); if (got !== want) mismatches.push(['bound', v, s, want, got]); }
    for (const [h, want] of vectors.phi512) { counts.phi512 = (counts.phi512 ?? 0) + 1; const got = binary64Bits(phi512(fromWord(h))); if (got !== want) mismatches.push(['phi512', h, want, got]); }
    for (const [f, m, L, w0, w1] of vectors.ehat) { counts.ehat = (counts.ehat ?? 0) + 1; const got = eHat(fromWord(f), fromWord(m), fromWord(L)).map(binary64Bits); if (got[0] !== w0 || got[1] !== w1) mismatches.push(['ehat', f, m, L, [w0, w1], got]); }
    const bitsOf = (n: number) => [0, 1, 2, 3].map(k => ((n >> k) & 1) === 1);
    for (const [present, nonin, L, floor, s, want] of vectors.stop_feasible) {
      counts.stop_feasible = (counts.stop_feasible ?? 0) + 1;
      const got = stopFeasible(bitsOf(s), bitsOf(present), bitsOf(nonin), L === 1, [floor ? floor.map((x: number) => x === 1) : [false, false]]);
      if (got !== want) mismatches.push(['stop_feasible', present, nonin, L, floor, s, want, got]);
    }
    // decode/encode round trip and refusals
    for (const w of ['0000000000000000', '8000000000000000', '0000000000000001', '7fefffffffffffff']) if (binary64Bits(decodeBinary64(w)) !== w) mismatches.push(['roundtrip', w]);
    for (const bad of ['7ff0000000000000', 'fff8000000000000', '7FF0000000000000', '00000000000000', 'zz00000000000000']) { let threw = false; try { decodeBinary64(bad); } catch { threw = true; } if (!threw) mismatches.push(['decode_accepts', bad]); }
    writeFileSync(SCRATCH + '/oracle_results.json', JSON.stringify({ counts, mismatches }, null, 1));
    expect(mismatches).toEqual([]);
  });
});

async function rehash(source: any) {
  const body = source.retained_precision.body;
  for (const s of body.sources) if (s.preparation) {
    const a = body.product_attempts[s.preparation.attempt_ref];
    if (a.preparation.members.every((m: any) => m.result.kind === 'prepared')) s.preparation.sha256 = await canonicalSha256HexCheckedV1({ domain: 'retained_precision_preparation_v1', payload: {
      definition_id: a.definition_id, definition_sha256: 'a7ed7ca0bf0bba6e8b821ca4befa00a0fa9541a83694be8b28ac63e39b1d0349', owner_ref: a.owner_ref, ordinary_attempt_ref: a.ordinary_attempt_ref, material_basis_ref: a.material_basis_ref,
      members: a.preparation.members.map((m: any) => ({ member: m.member, old_source: m.old_source, old_facts: m.old_facts, section: m.result.section })) } });
  }
  for (const c of body.cases) if (c.status === 'selected') { const { index: _, ...s } = body.sources[c.source_ref]; c.source_identity_sha256 = await canonicalSha256HexCheckedV1({ domain: 'retained_precision_source_mp_v2', payload: s }); }
  const { retained_precision: _, ...publication } = source;
  body.publication_sha256 = await canonicalSha256HexCheckedV1({ domain: 'retained_precision_publication_mp_v2', payload: publication });
  source.retained_precision.receipt_sha256 = await canonicalSha256HexCheckedV1({ domain: 'retained_precision_receipt_mp_v2', payload: body });
}
async function outcome(source: any, invocation?: any) {
  try { const r = await validateRetainedPrecision(source, invocation); return { result: 'pass', eligible: r.numerical_eligible, standing: r.standing }; }
  catch (e: any) { return { gate: e.gate, code: e.code }; }
}
const base = (id: string) => structuredClone(corpus.cases.find((c: any) => c.id === id));
describe('RV81 gate probes (rehashed single defects)', () => {
  it('records probe outcomes', async () => {
    const results: Record<string, unknown> = {};
    // P1: an old operational member index outside 0..n-1 on a source-less attempt (C3:302 G3 member-prefix coverage).
    { const c = base('two_case_preparation_failure_synthetic'); c.source.retained_precision.body.product_attempts[1].operational.old[0].member = 5; await rehash(c.source);
      results.P1_old_member_out_of_range_with_invocation = await outcome(structuredClone(c.source), structuredClone(c.invocation));
      results.P1_old_member_out_of_range_without_invocation = await outcome(structuredClone(c.source)); }
    // P2: an ordinary diagnostic_refs entry that resolves but names another case (I64 known difference 4).
    { const c = base('two_case_facade_after_certificate_synthetic'), b = c.source.retained_precision.body;
      results.P2_base_diagnostic_refs = b.ordinary_attempts.map((o: any) => o.diagnostic_refs);
      b.ordinary_attempts[0].diagnostic_refs.push(b.cases[1].diagnostic_ref); await rehash(c.source);
      results.P2_foreign_listed_diagnostic = await outcome(c.source, c.invocation); }
    // P3: non-array product_attempts (a G1 shape defect) -- which gate reports first.
    { const c = base('ordinary_prepared_synthetic'); c.source.retained_precision.body.product_attempts = {};
      results.P3_product_attempts_object = await outcome(c.source, c.invocation); }
    // P4: body.work removed (G1 shape defect touching a G0-pinned threshold).
    { const c = base('ordinary_prepared_synthetic'); delete c.source.retained_precision.body.work;
      results.P4_work_removed = await outcome(c.source, c.invocation); }
    // P5: a candidate-role record carrying a verification shared build (I64 known difference 3).
    { const c = base('ordinary_prepared_synthetic'), b = c.source.retained_precision.body, r0 = b.cases[0].run.records[0];
      results.P5_base_record0 = { role: r0.role, v: r0.verification === null, vref: r0.verification_shared_build_ref, vlme: r0.work.verification_lme };
      r0.work.verification_lme = 1; r0.work.own_stages.scale += 1; r0.work.wide_lme += 1; r0.work.own_lme += 1;
      await rehash(c.source); results.P5_candidate_record_verification_lme = await outcome(c.source, c.invocation); }
    // P6: accountingRules on hostile inputs (exported surface): returns booleans or throws, never mutates.
    { const a = structuredClone(base('ordinary_prepared_synthetic').source.retained_precision.body.product_attempts[0]); const before = JSON.stringify(a);
      const r = accountingRules(a); const proto = accountingRules({ adapter: { fault: null }, x: { kind: 'work_accounting', fault: 'constructor' } });
      let threw = false; try { accountingRules({} as any); } catch { threw = true; }
      results.P6_accounting_rules = { base: r, unmutated: JSON.stringify(a) === before, proto_key_rejected: proto, missing_adapter_throws: threw }; }
    // P7: selected case whose ordinary quality is checks_passed (with a consistent report outcome).
    { const c = base('ordinary_prepared_synthetic'), b = c.source.retained_precision.body, o = b.ordinary_attempts[0];
      results.P7_base_initial = { kind: o.initial.kind, outcome: o.initial.outcome, q: c.source.numerical_quality.cases[0].solve_quality };
      if (o.initial.kind === 'report') { o.initial.outcome = 'checks_passed'; c.source.numerical_quality.cases[0].solve_quality = 'checks_passed'; await rehash(c.source); results.P7_selected_checks_passed = await outcome(c.source, c.invocation); } }
    // P8/P9: base producer identity fields that Python pins at G0 (schema_version, component_version).
    { const c = base('ordinary_prepared_synthetic'); c.source.producer.component_version = '9.9.9'; await rehash(c.source); results.P8_component_version = await outcome(c.source, c.invocation); }
    { const c = base('ordinary_prepared_synthetic'); results.P9_base_schema_version = c.source.schema_version; c.source.schema_version = '0.3.0'; await rehash(c.source); results.P9_schema_version = await outcome(c.source, c.invocation); }
    // P10: N8 boundary -- a Ceiling right after an escalating p256 verification-solve failure of the p128 candidate.
    { const c = base('verification_failure_skip_synthetic'), run = c.source.retained_precision.body.cases[0].run;
      results.P10_base_shape = run.attempts.map((a: any) => [a.precision, a.verification?.phase ?? null, a.outcome.kind]); }
    // P11: captured_prefix attempt (must-pass prefix_captured shape) given a non-null run_ref -- G3 or G5 (checklist P7)?
    { const m = corpus.must_pass.find((x: any) => x.id === 'prefix_captured'), c = base(m.base);
      for (const edit of m.edits ?? []) { let v = c.source; for (const k of edit.path.slice(0, -1)) v = v[k]; const k = edit.path.at(-1); if (edit.op === 'remove') delete v[k]; else v[k] = structuredClone(edit.value); }
      if (m.rehash) await rehash(c.source);
      results.P11_prefix_captured_base = await outcome(structuredClone(c.source), structuredClone(c.invocation));
      const b = c.source.retained_precision.body, a = b.product_attempts.find((x: any) => x.operational.old_coverage === 'captured_prefix');
      results.P11_attempt = { id: a?.id, source_ref: a?.source_ref, run_ref: a?.run_ref, pm: a?.preparation.members.length, new: a?.operational.new.length };
      a.run_ref = 0; await rehash(c.source); results.P11_captured_prefix_with_run_ref = await outcome(c.source, c.invocation); }
    // P12: S06 s1 -- prepared_product_failure is permitted only when the case's own product_attempt_ref is the same U.
    { const c = base('two_case_preparation_failure_synthetic'), b = c.source.retained_precision.body;
      results.P12_base_case1 = { status: b.cases[1].status, par: b.cases[1].product_attempt_ref, cause: b.cases[1].reason.cause, run: b.cases[1].run, source_ref: b.cases[1].source_ref };
      b.product_attempts.splice(1, 1); b.cases[1].product_attempt_ref = null; await rehash(c.source);
      results.P12_orphan_cause_dangling_attempt = await outcome(structuredClone(c.source), structuredClone(c.invocation));
      b.cases[1].reason.cause.product_attempt_ref = 0; await rehash(c.source);
      results.P12_orphan_cause_foreign_ready_attempt = await outcome(c.source, c.invocation); }
    writeFileSync(SCRATCH + '/probe_results.json', JSON.stringify(results, null, 1));
    expect(true).toBe(true);
  });
});
