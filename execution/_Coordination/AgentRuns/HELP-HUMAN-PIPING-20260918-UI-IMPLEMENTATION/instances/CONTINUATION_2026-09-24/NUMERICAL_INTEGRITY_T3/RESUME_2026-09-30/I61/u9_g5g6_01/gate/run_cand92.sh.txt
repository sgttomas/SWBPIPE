#!/bin/bash
# I61 U9 G6 on the moved PR head C2 = 92a5a9da1c (ROOT: both-entry uses C2). Build the candidate probe from C2, rerun part 1's
# candidate side with the same unchanged driver, gate_check it, and compare with the existing base run. One cargo job at a time.
set -u
WT=WT; G=$WT/scratch/u9_gate; PY=REPO_ROOT/projects/chirality-piping/.venv/bin/python
pgrep -f memguard.sh >/dev/null || { echo "MEMGUARD NOT RUNNING"; exit 9; }
echo "== probe cand92 start $(date -u +%FT%TZ)"
( cd $G/probe_cand92 && env -u RUSTFLAGS CARGO_BUILD_JOBS=4 cargo build --release --offline --target-dir $WT/targets/i61-u9g6/cand92 > $G/logs/build_probe_cand92.log 2>&1 ); echo "== probe cand92 exit=$? $(date -u +%FT%TZ)"
diff $G/probe_cand92_Cargo.lock.before $G/probe_cand92/Cargo.lock > $G/lock_diff_probe_cand92.txt; echo "lock diff lines $(wc -l < $G/lock_diff_probe_cand92.txt)"
shasum -a 256 $WT/targets/i61-u9g6/cand92/release/t3_p1_probe | sed "s#$WT#WT#"
cd $G
echo "== part1 cand92 start $(date -u +%FT%TZ)"
$PY -B scripts/gate_run_base_full.py $WT/targets/i61-u9g6/cand92/release/t3_p1_probe gen_out part1_cand92 4 > logs/part1_cand92_driver.log 2>&1; echo "== part1 cand92 exit=$? $(date -u +%FT%TZ)"
$PY -B scripts/gate_check.py gen_out part1_cand92/runs.jsonl inputs/references.json part1_cand92/result_part1_cand92.json inputs/S11_EXCEPTIONS.json inputs/FORMATION_EXCEPTIONS.json > logs/part1_cand92_gate_check.log 2>&1; echo "== gate_check cand92 exit=$?"
$PY -B scripts/compare_gate_kf2.py part1_base part1_cand92 scripts part1_base/result_part1_base.json part1_cand92/result_part1_cand92.json > logs/compare_part1_cand92.log 2>&1; echo "== compare cand92 exit=$?"
for d in full envelopes stderr; do echo "diff -rq $d: $(diff -rq part1_base/$d part1_cand92/$d 2>&1 | wc -l) differing; files $(ls part1_cand92/$d | wc -l)"; done
echo CAND92-DONE $(date -u +%FT%TZ)
