575: /// The kernel's sparse assembly, per modulus basis (D1 §4.8): the global
576: /// stiffness of `frames`, `users`, `blocks` and ground `springs` (global DOF,
577: /// stiffness) on its connectivity pattern.
578: ///
579: /// It is the sparse sibling of `assemble_global_stiffness_with_user_elements`
580: /// followed by the product's curved and spring additions, and it keeps their
581: /// semantics:
582: /// - each element is formed by the same call (`global_stiffness`) in the same
583: ///   order, after the same node check, so a formation refusal is the same
584: ///   error for the same first element, before any value is accumulated;
585: /// - values accumulate in the dense order (frames, users, blocks, springs);
586: /// - after the frames and users, the first non-finite entry in row-major order
587: ///   is refused as `NonFiniteInput { name: "assembled stiffness" }`; blocks
588: ///   and springs are then added unchecked, as the product adds them.
589: ///
590: /// A block or spring outside the model is refused as `InvalidNodeIndex`
591: /// (the dense additions would index out of bounds).
592: pub fn assemble_sparse_stiffness(
593:     node_count: usize,
594:     frames: &[FrameElement],
595:     users: &[UserStiffnessElement],
596:     blocks: &[StiffnessBlock],
597:     springs: &[(usize, f64)],
598:     options: &SparseAssemblyOptions,
599: ) -> Result<SparseStiffness, FrameKernelError> {
600:     let SparseAssemblyOptions { force_scale } = options;
601:     if !force_scale.is_unscaled() {
602:         // K2b: the same assembly, of the inputs formed at 2^b.
603:         let (frames, users, blocks, springs) =
604:             force_scaled_inputs(frames, users, blocks, springs, *force_scale)?;
605:         return assemble_sparse_stiffness(
606:             node_count,
607:             &frames,
608:             &users,
609:             &blocks,
610:             &springs,
611:             &SparseAssemblyOptions::new(),
612:         );
613:     }
614:     let dimension = node_count * DOF_PER_NODE;
615:     let mut formed: Vec<(usize, usize, Matrix12)> = Vec::with_capacity(frames.len() + users.len());
616:     for element in frames {
617:         check_node(element.node_i.index, node_count)?;
618:         check_node(element.node_j.index, node_count)?;
619:         formed.push((
620:             element.node_i.index,
621:             element.node_j.index,
622:             element.global_stiffness()?,
623:         ));
624:     }
625:     for element in users {
626:         check_node(element.node_i.index, node_count)?;
627:         check_node(element.node_j.index, node_count)?;
628:         formed.push((
629:             element.node_i.index,
630:             element.node_j.index,
631:             element.global_stiffness()?,
632:         ));
633:     }
634:     for block in blocks {
635:         check_node(block.node_i, node_count)?;
636:         check_node(block.node_j, node_count)?;
637:     }
638:     for &(dof, _) in springs {
639:         check_node(dof / DOF_PER_NODE, node_count)?;
640:     }
641:     let connectivity: Vec<(usize, usize)> = formed
642:         .iter()
643:         .map(|&(a, b, _)| (a, b))
644:         .chain(blocks.iter().map(|b| (b.node_i, b.node_j)))
645:         .collect();
646:     let spring_dofs: Vec<usize> = springs.iter().map(|&(dof, _)| dof).collect();
647:     let pattern = SparsePattern::from_connectivity(node_count, &connectivity, &spring_dofs)
648:         .expect("nodes and DOFs checked above");
649:     debug_assert_eq!(pattern.dimension(), dimension);
650:     let mut values = vec![0.0; pattern.entry_count()];
651:     for (a, b, matrix) in &formed {
652:         scatter_block(&pattern, &mut values, *a, *b, matrix);
653:     }
654:     // V-K seeded fault VK-F08 (§7.3-8): member 1's i–j coupling block left
655:     // out, in sparse mode only (the dense assembly keeps it).
656:     #[cfg(any(test, feature = "mutation-controls"))]
657:     if super::retained::seeded::active(super::retained::seeded::Fault::F08) {
658:         if let Some(&(a, b, _)) = formed.first() {
659:             let map = element_dof_map(a, b);
660:             for r in 0..6 {
661:                 for c in 6..ELEMENT_DOF {
662:                     for (row, col) in [(map[r], map[c]), (map[c], map[r])] {
663:                         let index = pattern.find(row, col).expect("element block in pattern");
664:                         values[index] = 0.0;
665:                     }
666:                 }
667:             }
668:         }
669:     }
670:     for row in 0..dimension {
671:         for index in pattern.row_range(row) {
672:             let value = values[index];
673:             if !value.is_finite() {
674:                 return Err(FrameKernelError::NonFiniteInput {
675:                     name: "assembled stiffness",
676:                     value,
677:                 });
678:             }
679:         }
680:     }
681:     for block in blocks {
682:         scatter_block(
683:             &pattern,
684:             &mut values,
685:             block.node_i,
686:             block.node_j,
687:             &block.stiffness,
688:         );
689:     }
690:     for &(dof, stiffness) in springs {
691:         let index = pattern.find(dof, dof).expect("spring diagonal in pattern");
692:         values[index] += stiffness;
693:     }
694:     Ok(SparseStiffness { pattern, values })
695: }

699: fn scatter_block(
700:     pattern: &SparsePattern,
701:     values: &mut [f64],
702:     node_i: usize,
703:     node_j: usize,
704:     matrix: &Matrix12,
705: ) {
706:     let map = element_dof_map(node_i, node_j);

1031: fn validate_sparse(system: &SparseStructuralSystem<'_>) -> Result<(), StructuralError> {
1032:     let n = system.force.len();
1033:     let stiffness = system.stiffness;
1034:     let pattern = &stiffness.pattern;
1035:     if pattern.dimension != n {
1036:         return Err(StructuralError::InvalidInput("matrix dimensions"));
1037:     }
1038:     if system
1039:         .force
1040:         .iter()
1041:         .chain(stiffness.values.iter())
1042:         .any(|x| !x.is_finite())
1043:     {
1044:         return Err(StructuralError::InvalidInput(
1045:             "nonfinite original equations",
1046:         ));
1047:     }
1048:     let nnz = pattern.entry_count();
1049:     if let Some(evidence) = &system.symmetry {
1050:         if evidence.basis.is_empty()
1051:             || evidence.absolute_roundoff.len() != nnz
1052:             || evidence.operation_counts.len() != nnz
1053:             || evidence
1054:                 .absolute_roundoff
1055:                 .iter()
1056:                 .any(|v| !v.is_finite() || *v < 0.0)
1057:         {
1058:             return Err(StructuralError::InvalidInput("symmetry provenance"));
1059:         }
1060:     }
1061:     for i in 0..n {
1062:         for index in pattern.row_range(i) {
1063:             let j = pattern.columns[index];
1064:             if j >= i {
1065:                 break;
1066:             }
1067:             let mirror = pattern.transpose[index];
1068:             let a = stiffness.values[index];
1069:             let b = stiffness.values[mirror];
1070:             if a == b {
1071:                 continue;
1072:             }
1073:             let scale = a.abs().max(b.abs());
1074:             let skew = (a / scale - b / scale).abs();
1075:             let allowed = system.symmetry.as_ref().map_or(0.0, |e| {
1076:                 e.absolute_roundoff[index] / scale + e.absolute_roundoff[mirror] / scale
1077:             });
1078:             if skew > allowed {
1079:                 return Err(StructuralError::Asymmetric {
1080:                     row: i,
1081:                     col: j,
1082:                     relative_skew: skew,
1083:                 });
1084:             }
1085:         }
1086:     }
1087:     let mut seen = vec![false; n];
1088:     for &i in system.free_dofs {
1089:         if i >= n || seen[i] {
1090:             return Err(StructuralError::InvalidInput("free map"));
1091:         }
1092:         seen[i] = true;
1093:     }
1094:     for &(i, v) in system.prescribed {
1095:         if i >= n || seen[i] || !v.is_finite() {
1096:             return Err(StructuralError::InvalidInput("prescribed map"));
1097:         }
1098:         seen[i] = true;
1099:     }
1100:     if seen.iter().any(|x| !*x) {
1101:         return Err(StructuralError::InvalidInput("incomplete constraint map"));
1102:     }
1103:     Ok(())
1104: }

1182: /// `contribution_sums` per pattern entry (O(nnz) memory), with the same
1183: /// absorbed-diagonal refusal, in contribution order.
1184: fn sparse_contribution_sums(
1185:     system: &SparseStructuralSystem<'_>,
1186: ) -> Result<Option<Vec<Expansion>>, StructuralError> {
1187:     let Some(entries) = system.contributions else {
1188:         return Ok(None);
1189:     };
1190:     let n = system.force.len();
1191:     let pattern = &system.stiffness.pattern;
1192:     let mut sums = vec![Expansion::default(); pattern.entry_count()];
1193:     for entry in entries {
1194:         if entry.row >= n || entry.col >= n || !entry.value.is_finite() {
1195:             return Err(StructuralError::InvalidInput("contribution"));
1196:         }
1197:         let index = pattern
1198:             .find(entry.row, entry.col)
1199:             .ok_or(StructuralError::InvalidInput(
1200:                 "contribution outside pattern",
1201:             ))?;
1202:         let sum = &mut sums[index];
1203:         let old = sum.round()?;
1204:         sum.add(entry.value)?;
1205:         if entry.row == entry.col && entry.value > 0.0 && old != 0.0 && sum.round()? == old {
1206:             return Err(unresolved(
1207:                 "positive diagonal contribution absorbed by assembly; stabilization unresolved",
1208:                 Some(entry.row),
1209:             ));
1210:         }
1211:     }
1212:     Ok(Some(sums))
1213: }

1299: fn prepare_sparse_bound<'s>(
1300:     system: &'s SparseStructuralSystem<'s>,
1301:     force_binding: ForceBinding<'s>,
1302: ) -> Result<SparsePreparedSystem<'s>, StructuralError> {
1303:     if force_binding.binary64() {
1304:         // The option-(c) binary64 fold is not offered on the pattern: the
1305:         // nonlinear loop stays on the dense `_binary64` path until F1b.
1306:         return Err(StructuralError::InvalidInput(
1307:             "binary64 binding on the sparse representation",
1308:         ));
1309:     }
1310:     validate_sparse(system)?;
1311:     let stiffness = system.stiffness;
1312:     let pattern = &stiffness.pattern;
1313:     let n = system.free_dofs.len();
1314:     let mut exponents = Vec::with_capacity(n);
1315:     for &i in system.free_dofs {
1316:         let d = stiffness.get(i, i);
1317:         if d < 0.0 {
1318:             let mut direction = vec![0.0; system.force.len()];
1319:             direction[i] = 1.0;
1320:             return Err(StructuralError::NegativeEnergy {
1321:                 direction,
1322:                 energy: d,
1323:                 allowance: 0.0,
1324:             });
1325:         }
1326:         if d == 0.0 {
1327:             return Err(unresolved(
1328:                 "zero original diagonal; no physical nullity inferred",
1329:                 Some(i),
1330:             ));
1331:         }
1332:         exponents.push(-binary_exponent(d).div_euclid(2));
1333:     }
1334:     let mut free_position = vec![usize::MAX; system.force.len()];
1335:     for (r, &i) in system.free_dofs.iter().enumerate() {
1336:         free_position[i] = r;
1337:     }
1338:     let mut row_starts = Vec::with_capacity(n + 1);
1339:     row_starts.push(0);
1340:     let mut columns = Vec::new();
1341:     let mut values = Vec::new();
1342:     let mut source_entries = Vec::new();
1343:     let mut diagonal = Vec::with_capacity(n);
1344:     let mut rhs = vec![0.0; n];
1345:     for (r, &i) in system.free_dofs.iter().enumerate() {
1346:         let mut entries: Vec<(usize, usize)> = pattern
1347:             .row_range(i)
1348:             .filter_map(|index| {
1349:                 let c = free_position[pattern.columns[index]];
1350:                 (c != usize::MAX).then_some((c, index))
1351:             })
1352:             .collect();
1353:         entries.sort_unstable_by_key(|&(c, _)| c);
1354:         let start = columns.len();
1355:         for (c, index) in entries {
1356:             values.push(radix_scale(
1357:                 stiffness.values[index],
1358:                 exponents[r] + exponents[c],
1359:             )?);
1360:             columns.push(c);
1361:             source_entries.push(index);
1362:         }
1363:         diagonal.push(
1364:             start
1365:                 + columns[start..].binary_search(&r).map_err(|_| {
1366:                     unresolved(
1367:                         "zero original diagonal; no physical nullity inferred",
1368:                         Some(i),
1369:                     )
1370:                 })?,
1371:         );
1372:         row_starts.push(columns.len());
1373:         let couplings = system.prescribed_couplings(i);
1374:         let coupled = couplings.iter().any(|&(_, k, g)| k != 0.0 && g != 0.0);
1375:         if force_binding.assembled().is_some() || coupled {
1376:             rhs[r] = exact_scaled_rhs(
1377:                 system.view(),
1378:                 force_binding,
1379:                 i,
1380:                 couplings.iter().map(|&(_, k, g)| (k, g)),
1381:                 exponents[r],
1382:             )?;
1383:         } else {
1384:             // No nonzero prescribed product: every `b - K_ij g_j` of the
1385:             // dense fold subtracts a signed zero, which is exact. Its only
1386:             // effects are refusing a subnormal b at the first subtraction and
1387:             // turning b = -0.0 into +0.0 when some product is -0.0.
1388:             let b = legacy_zero_product_fold(system, i, &couplings)?;
1389:             rhs[r] = radix_scale(b, exponents[r])?;
1390:         }
1391:     }
1392:     let mut transpose = Vec::with_capacity(columns.len());
1393:     for r in 0..n {
1394:         for index in row_starts[r]..row_starts[r + 1] {
1395:             let c = columns[index];
1396:             let position = columns[row_starts[c]..row_starts[c + 1]]
1397:                 .binary_search(&r)
1398:                 .map_err(|_| StructuralError::InvalidInput("prepared pattern symmetry"))?;
1399:             transpose.push(row_starts[c] + position);
1400:         }
1401:     }
1402:     let nnz = pattern.entry_count();
1403:     if let Some(evidence) = &system.symmetry {
1404:         if evidence.basis.is_empty()
1405:             || evidence.absolute_roundoff.len() != nnz
1406:             || evidence.operation_counts.len() != nnz
1407:             || evidence
1408:                 .absolute_roundoff
1409:                 .iter()
1410:                 .any(|v| !v.is_finite() || *v < 0.0)
1411:         {
1412:             return Err(StructuralError::InvalidInput("symmetry provenance"));
1413:         }
1414:     }
1415:     let mut projection = false;
1416:     let mut max_skew: f64 = 0.0;
1417:     for r in 0..n {
1418:         for index in row_starts[r]..row_starts[r + 1] {
1419:             let c = columns[index];
1420:             if c >= r {
1421:                 break;
1422:             }
1423:             let mirror = transpose[index];
1424:             let skew = checked_value(values[index] - values[mirror])?.abs();
1425:             max_skew = max_skew.max(skew);
1426:             if skew == 0.0 {
1427:                 continue;
1428:             }
1429:             let allowed = if let Some(e) = &system.symmetry {
1430:                 checked_value(
1431:                     radix_scale(
1432:                         e.absolute_roundoff[source_entries[index]],
1433:                         exponents[r] + exponents[c],
1434:                     )? + radix_scale(
1435:                         e.absolute_roundoff[source_entries[mirror]],
1436:                         exponents[r] + exponents[c],
1437:                     )?,
1438:                 )?
1439:             } else {
1440:                 0.0
1441:             };
1442:             if skew > allowed {
1443:                 return Err(StructuralError::Asymmetric {
1444:                     row: system.free_dofs[r],
1445:                     col: system.free_dofs[c],
1446:                     relative_skew: skew / values[index].abs().max(values[mirror].abs()),
1447:                 });
1448:             }
1449:             let average = values[index] / 2.0 + values[mirror] / 2.0;
1450:             values[index] = average;
1451:             values[mirror] = average;
1452:             projection = true;
1453:         }
1454:     }
1455:     let mut prepared = SparsePreparedSystem {
1456:         source: system,
1457:         force_binding,
1458:         formation: None,
1459:         row_starts,
1460:         columns,
1461:         values,
1462:         transpose,
1463:         source_entries,
1464:         diagonal,
1465:         rhs,
1466:         scale_exponents: exponents,
1467:         assembly_relative_perturbation_estimate: 0.0,
1468:         assembly_load_perturbation_estimate: 0.0,
1469:         contribution_rounding: Vec::new(),
1470:         symmetry_projection_performed: projection,
1471:         maximum_scaled_skew: max_skew,
1472:         symmetry_basis: system.symmetry.as_ref().map(|e| e.basis.to_owned()),
1473:     };
1474:     let (perturbation, load_perturbation, rounding) = sparse_audit_contributions(&prepared)?;
1475:     prepared.assembly_relative_perturbation_estimate = perturbation;
1476:     prepared.assembly_load_perturbation_estimate = load_perturbation;
1477:     prepared.contribution_rounding = rounding;
1478:     Ok(prepared)
1479: }

1481: /// The dense fold `b = checked_value(b - checked_product(K_ij, g_j)?)?` over
1482: /// every prescribed (j, g_j), on a row where every product is a signed zero
1483: /// (no stored coupling has K_ij != 0 and g_j != 0; an absent K_ij is +0.0).
1484: /// Subtracting a signed zero is exact, so the fold reduces to: the first
1485: /// subtraction refuses a subnormal b, as `checked_value` does; and b = -0.0
1486: /// becomes +0.0 if some product is -0.0 (-0.0 - (-0.0) = +0.0). No other value
1487: /// changes, so no binary64 accumulation is formed.
1488: fn legacy_zero_product_fold(
1489:     system: &SparseStructuralSystem<'_>,
1490:     i: usize,
1491:     couplings: &[(usize, f64, f64)],
1492: ) -> Result<f64, StructuralError> {
1493:     let b = system.force[i];
1494:     if system.prescribed.is_empty() {
1495:         return Ok(b);
1496:     }
1497:     if b.is_subnormal() {
1498:         return Err(StructuralError::Range("arithmetic outside normal range"));
1499:     }
1500:     if b == 0.0 && b.is_sign_negative() {
1501:         // Products K_ij * g_j with the sign bit set: every absent column
1502:         // contributes g_j's sign; a stored one the product's own sign.
1503:         let all_negative = system.negative_prescribed;
1504:         let stored_negative_g = couplings.iter().filter(|c| c.2.is_sign_negative()).count();
1505:         let stored_negative_product = couplings
1506:             .iter()
1507:             .filter(|c| c.1.is_sign_negative() != c.2.is_sign_negative())
1508:             .count();
1509:         if all_negative - stored_negative_g + stored_negative_product > 0 {
1510:             return Ok(0.0);
1511:         }
1512:     }
1513:     Ok(b)
1514: }

1516: /// `audit_contributions` over the pattern, reading every term in the dense
1517: /// order: the rounding list row-major, each free column's norms down its free
1518: /// rows, and each row's prescribed coupling in `prescribed` order.
1519: fn sparse_audit_contributions(
1520:     prepared: &SparsePreparedSystem<'_>,
1521: ) -> Result<(f64, f64, Vec<ContributionRounding>), StructuralError> {
1522:     let system = prepared.source;
1523:     let Some(sums) = sparse_contribution_sums(system)? else {
1524:         return Ok((0.0, 0.0, Vec::new()));
1525:     };
1526:     let stiffness = system.stiffness;
1527:     let pattern = &stiffness.pattern;
1528:     let exponents = &prepared.scale_exponents;
1529:     let mut differences = sums.clone();
1530:     for (index, value) in differences.iter_mut().enumerate() {
1531:         value.add(-stiffness.values[index])?;
1532:     }
1533:     let mut rounding = Vec::new();
1534:     for i in 0..pattern.dimension {
1535:         for index in pattern.row_range(i) {
1536:             let difference = &differences[index];
1537:             if !difference.is_zero() {
1538:                 let terms = &sums[index].terms;
1539:                 rounding.push(ContributionRounding {
1540:                     row: i,
1541:                     col: pattern.columns[index],
1542:                     accumulated_high: terms.last().copied().unwrap_or(0.0),
1543:                     accumulated_low: terms.iter().rev().skip(1).copied().sum(),
1544:                     stored_difference_high: difference.terms.last().copied().unwrap_or(0.0),
1545:                     stored_difference_low: difference.terms.iter().rev().skip(1).copied().sum(),
1546:                     accumulated_expansion: terms.clone(),
1547:                     difference_expansion: difference.terms.clone(),
1548:                 });
1549:             }
1550:         }
1551:     }
1552:     let mut perturbation_norm: f64 = 0.0;
1553:     let mut matrix_norm: f64 = 0.0;
1554:     for c in 0..prepared.dimension() {
1555:         let mut delta_column = 0.0;
1556:         let mut matrix_column = 0.0;
1557:         // Column c's free rows r, ascending: the prepared row c (by symmetry
1558:         // of the pattern), whose source entry (free[c], free[r]) has the
1559:         // transpose (free[r], free[c]).
1560:         for index in prepared.row_range(c) {
1561:             let r = prepared.columns[index];
1562:             let global = pattern.transpose[prepared.source_entries[index]];
1563:             for &part in &differences[global].terms {
1564:                 delta_column = checked_value(
1565:                     delta_column + exact_radix(part, exponents[r] + exponents[c])?.abs(),
1566:                 )?;
1567:             }
1568:             matrix_column = checked_value(
1569:                 matrix_column
1570:                     + radix_scale(stiffness.values[global], exponents[r] + exponents[c])?.abs(),
1571:             )?;
1572:         }
1573:         perturbation_norm = perturbation_norm.max(delta_column);
1574:         matrix_norm = matrix_norm.max(matrix_column);
1575:     }
1576:     let mut delta_rhs_norm = 0.0;
1577:     let mut rhs_norm = 0.0;
1578:     for (r, &i) in system.free_dofs.iter().enumerate() {
1579:         let mut delta = Expansion::default();
1580:         let mut couplings: Vec<(usize, usize)> = pattern
1581:             .row_range(i)
1582:             .filter_map(|index| {
1583:                 let position = system.prescribed_position[pattern.columns[index]];
1584:                 (position != usize::MAX).then_some((position, index))
1585:             })
1586:             .collect();
1587:         couplings.sort_unstable_by_key(|&(position, _)| position);
1588:         for (position, index) in couplings {
1589:             let value = system.prescribed[position].1;
1590:             for &part in &differences[index].terms {
1591:                 delta.add_product(-part, value, exponents[r])?;
1592:             }
1593:         }
1594:         delta_rhs_norm =
1595:             checked_value(delta_rhs_norm + delta.terms.iter().map(|v| v.abs()).sum::<f64>())?;
1596:         rhs_norm = checked_value(rhs_norm + prepared.rhs[r].abs())?;
1597:     }
1598:     if delta_rhs_norm != 0.0 && rhs_norm == 0.0 {
1599:         return Err(unresolved(
1600:             "contribution-preserved prescribed coupling changes zero reduced load",
1601:             None,
1602:         ));
1603:     }
1604:     Ok((
1605:         if matrix_norm == 0.0 {
1606:             0.0
1607:         } else {
1608:             perturbation_norm / matrix_norm
1609:         },
1610:         if rhs_norm == 0.0 {
1611:             0.0
1612:         } else {
1613:             delta_rhs_norm / rhs_norm
1614:         },
1615:         rounding,
1616:     ))
1617: }

1619: /// `validate_prepared` over the pattern, including the internal structure, so
1620: /// a corrupted prepared system is an error, never a panic.
1621: fn validate_sparse_prepared(prepared: &SparsePreparedSystem<'_>) -> Result<(), StructuralError> {
1622:     validate_sparse(prepared.source)?;
1623:     let n = prepared.source.free_dofs.len();
1624:     let nnz = prepared.columns.len();
1625:     let source_nnz = prepared.source.stiffness.pattern.entry_count();
1626:     let shape = StructuralError::InvalidInput("prepared shape/map/finiteness");
1627:     if prepared.row_starts.len() != n + 1
1628:         || prepared.row_starts.first() != Some(&0)
1629:         || prepared.row_starts.last() != Some(&nnz)
1630:         || prepared.row_starts.windows(2).any(|w| w[0] > w[1])
1631:         || prepared.values.len() != nnz
1632:         || prepared.transpose.len() != nnz
1633:         || prepared.source_entries.len() != nnz
1634:         || prepared.diagonal.len() != n
1635:         || prepared.rhs.len() != n
1636:         || prepared.scale_exponents.len() != n
1637:         || prepared.values.iter().any(|v| !v.is_finite())
1638:         || prepared.rhs.iter().any(|v| !v.is_finite())
1639:         || prepared.source_entries.iter().any(|&k| k >= source_nnz)
1640:     {
1641:         return Err(shape);
1642:     }
1643:     for r in 0..n {
1644:         let row = &prepared.columns[prepared.row_range(r)];
1645:         if row.iter().any(|&c| c >= n) || row.windows(2).any(|w| w[0] >= w[1]) {
1646:             return Err(shape);
1647:         }
1648:         let d = prepared.diagonal[r];
1649:         if !prepared.row_range(r).contains(&d) || prepared.columns[d] != r {
1650:             return Err(shape);
1651:         }
1652:         for index in prepared.row_range(r) {
1653:             let t = prepared.transpose[index];
1654:             if t >= nnz
1655:                 || prepared.transpose[t] != index
1656:                 || !prepared.row_range(prepared.columns[index]).contains(&t)
1657:                 || prepared.columns[t] != r
1658:             {
1659:                 return Err(shape);
1660:             }
1661:         }
1662:     }
1663:     if (0..nnz).any(|index| prepared.values[index] != prepared.values[prepared.transpose[index]]) {
1664:         return Err(StructuralError::InvalidInput("prepared symmetry"));
1665:     }
1666:     Ok(())
1667: }

1739: /// `factor_structural_profile` on the sparse prepared system: the caller
1740: /// supplies only the ordering and the skyline; the profile rows are filled
1741: /// from the stored entries (zeros elsewhere), and the factorization is the
1742: /// shared `ProfileFactor::factor_structural_profile`.
1743: pub fn factor_sparse_structural_profile<'p, 's>(
1744:     prepared: &'p SparsePreparedSystem<'s>,
1745:     order: &[usize],
1746:     first: &[usize],
1747: ) -> Result<SparsePositiveFactor<'p, 's>, StructuralError> {
1748:     validate_sparse_prepared(prepared)?;
1749:     let n = prepared.dimension();
1750:     if order.len() != n || first.len() != n {
1751:         return Err(StructuralError::InvalidInput("profile dimensions"));
1752:     }
1753:     let mut seen = vec![false; n];
1754:     for &i in order {
1755:         if i >= n || seen[i] {
1756:             return Err(StructuralError::InvalidInput("profile permutation"));
1757:         }
1758:         seen[i] = true;
1759:     }
1760:     let mut position = vec![0; n];
1761:     for (k, &i) in order.iter().enumerate() {
1762:         position[i] = k;
1763:     }
1764:     let mut rows = Vec::with_capacity(n);
1765:     for i in 0..n {
1766:         if first[i] > i {
1767:             return Err(StructuralError::InvalidInput("profile first column"));
1768:         }
1769:         let mut row = vec![0.0; i - first[i] + 1];
1770:         for index in prepared.row_range(order[i]) {
1771:             let k = position[prepared.columns[index]];
1772:             let value = prepared.values[index];
1773:             if k < first[i] {
1774:                 if value != 0.0 {
1775:                     return Err(StructuralError::InvalidInput(
1776:                         "profile omits represented coefficient",
1777:                     ));
1778:                 }
1779:             } else if k <= i {
1780:                 row[k - first[i]] = value;
1781:             }
1782:         }
1783:         rows.push(row);
1784:     }
1785:     let factor = ProfileFactor {
1786:         first: first.to_vec(),
1787:         rows,
1788:         order: order.to_vec(),
1789:     };
1790:     let (factor, pivots) = factor.factor_structural_profile(prepared.source.free_dofs)?;
1791:     Ok(SparsePositiveFactor {
1792:         prepared,
1793:         factor,
1794:         pivots,
1795:     })
1796: }

1875: pub fn verify_sparse_negative_direction(
1876:     prepared: &SparsePreparedSystem<'_>,
1877:     direction: &[f64],
1878: ) -> Result<Option<StructuralError>, StructuralError> {
1879:     validate_sparse_prepared(prepared)?;
1880:     let n = prepared.dimension();
1881:     if direction.len() != n || direction.iter().any(|v| !v.is_finite()) {
1882:         return Err(StructuralError::InvalidInput("negative direction"));
1883:     }
1884:     let original = &prepared.source.stiffness.values;
1885:     let mut energy = 0.0;
1886:     let mut magnitude = 0.0;
1887:     let mut terms = 0;
1888:     for r in 0..n {
1889:         for index in prepared.row_range(r) {
1890:             let c = prepared.columns[index];
1891:             if prepared.values[index] == 0.0 || direction[r] == 0.0 || direction[c] == 0.0 {
1892:                 continue;
1893:             }
1894:             let k = radix_scale(
1895:                 original[prepared.source_entries[index]],
1896:                 prepared.scale_exponents[r] + prepared.scale_exponents[c],
1897:             )?;
1898:             let term = checked_product(checked_product(direction[r], k)?, direction[c])?;
1899:             energy = checked_value(energy + term)?;
1900:             magnitude = checked_value(magnitude + term.abs())?;
1901:             terms += 1;
1902:         }
1903:     }
1904:     negative_verdict(prepared, direction, energy, magnitude, terms)
1905: }

1907: /// The verdict of `verify_negative_direction` from its energy sums.
1908: fn negative_verdict(
1909:     prepared: &SparsePreparedSystem<'_>,
1910:     direction: &[f64],
1911:     energy: f64,
1912:     magnitude: f64,
1913:     terms: usize,
1914: ) -> Result<Option<StructuralError>, StructuralError> {
1915:     let allowance = 64.0 * gamma(3 * terms + 2) * magnitude;
1916:     if energy < -allowance {
1917:         let system = prepared.source;
1918:         let mut mapped = vec![0.0; system.force.len()];
1919:         for (i, &global) in system.free_dofs.iter().enumerate() {
1920:             mapped[global] = radix_scale(direction[i], prepared.scale_exponents[i])?;
1921:         }
1922:         Ok(Some(StructuralError::NegativeEnergy {
1923:             direction: mapped,
1924:             energy,
1925:             allowance,
1926:         }))
1927:     } else {
1928:         Ok(None)
1929:     }
1930: }

1932: /// `negative_pair_witness` over the pattern, in O(nnz): it visits only the
1933: /// stored pairs below the diagonal, in the dense (row-major) order. A pair
1934: /// with a zero coupling has energy a_ii + a_jj > 0 (prepared diagonals are
1935: /// positive), so it can never be the dense search's witness; skipping the
1936: /// unstored pairs therefore returns the dense search's first witness.
1937: pub fn sparse_negative_pair_witness(
1938:     prepared: &SparsePreparedSystem<'_>,
1939: ) -> Result<Option<StructuralError>, StructuralError> {
1940:     sparse_negative_pair_witness_counted(prepared).map(|(witness, _)| witness)
1941: }

1943: /// `sparse_negative_pair_witness` with the number of pairs it visited.
1944: pub(crate) fn sparse_negative_pair_witness_counted(
1945:     prepared: &SparsePreparedSystem<'_>,
1946: ) -> Result<(Option<StructuralError>, usize), StructuralError> {
1947:     validate_sparse_prepared(prepared)?;
1948:     let n = prepared.dimension();
1949:     let mut visited = 0;
1950:     for r in 0..n {
1951:         for index in prepared.row_range(r) {
1952:             let c = prepared.columns[index];
1953:             if c >= r {
1954:                 break;
1955:             }
1956:             visited += 1;
1957:             let sign = if prepared.values[index] >= 0.0 {
1958:                 -1.0
1959:             } else {
1960:                 1.0
1961:             };
1962:             let (energy, magnitude, terms) = pair_energy(prepared, r, c, index, sign)?;
1963:             let allowance = 64.0 * gamma(3 * terms + 2) * magnitude;
1964:             if energy < -allowance {
1965:                 let mut direction = vec![0.0; n];
1966:                 direction[r] = 1.0;
1967:                 direction[c] = sign;
1968:                 return Ok((
1969:                     negative_verdict(prepared, &direction, energy, magnitude, terms)?,
1970:                     visited,
1971:                 ));
1972:             }
1973:         }
1974:     }
1975:     Ok((None, visited))
1976: }

1978: /// The energy sums of the direction e_r + sign·e_c (c < r) in O(1): its only
1979: /// terms, in the dense row-major order, are (c,c), (c,r), (r,c) and (r,r).
1980: fn pair_energy(
1981:     prepared: &SparsePreparedSystem<'_>,
1982:     r: usize,
1983:     c: usize,
1984:     lower: usize,
1985:     sign: f64,
1986: ) -> Result<(f64, f64, usize), StructuralError> {
1987:     let original = &prepared.source.stiffness.values;
1988:     let upper = prepared.transpose[lower];
1989:     let cells = [
1990:         (c, c, prepared.diagonal[c], sign, sign),
1991:         (c, r, upper, sign, 1.0),
1992:         (r, c, lower, 1.0, sign),
1993:         (r, r, prepared.diagonal[r], 1.0, 1.0),
1994:     ];
1995:     let mut energy = 0.0;
1996:     let mut magnitude = 0.0;
1997:     let mut terms = 0;
1998:     for (a, b, index, da, db) in cells {
1999:         if prepared.values[index] == 0.0 {
2000:             continue;
2001:         }
2002:         let k = radix_scale(
2003:             original[prepared.source_entries[index]],
2004:             prepared.scale_exponents[a] + prepared.scale_exponents[b],
2005:         )?;
2006:         let term = checked_product(checked_product(da, k)?, db)?;
2007:         energy = checked_value(energy + term)?;
2008:         magnitude = checked_value(magnitude + term.abs())?;
2009:         terms += 1;
2010:     }
2011:     Ok((energy, magnitude, terms))
2012: }
