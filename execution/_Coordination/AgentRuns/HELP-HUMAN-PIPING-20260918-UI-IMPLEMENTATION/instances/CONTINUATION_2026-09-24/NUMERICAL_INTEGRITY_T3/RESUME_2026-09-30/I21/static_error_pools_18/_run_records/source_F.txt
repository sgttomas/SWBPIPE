499:         for axis in local_axes {

501:         }

507:             || dot(local_axes[0], local_axes[2]).abs() > AXIS_TOLERANCE
508:             || dot(local_axes[1], local_axes[2]).abs() > AXIS_TOLERANCE

511:                 detail: "local axes must be mutually orthogonal",
512:             });
513:         }
514:

516:         if max_abs_component(subtract(right_handed_z, local_axes[2])) > AXIS_TOLERANCE {
517:             return Err(FrameKernelError::InvalidOrientation {
518:                 detail: "local axes must form a right-handed basis",
519:             });
520:         }

525:     pub fn from_x_axis_and_y_reference(
526:         local_x_axis: [f64; 3],

533:         let projection = dot(y_reference, x_axis);

711: pub fn local_stiffness(properties: FrameProperties) -> Result<Matrix12, FrameKernelError> {
712:     let section = properties.section;
713:     validate_positive_finite("elastic_modulus", section.elastic_modulus)?;
714:     validate_positive_finite("shear_modulus", section.shear_modulus)?;
715:     validate_positive_finite("area", section.area)?;
716:     validate_positive_finite("second_moment_y", section.second_moment_y)?;
717:     validate_positive_finite("second_moment_z", section.second_moment_z)?;
718:     validate_positive_finite("torsion_constant", section.torsion_constant)?;
719:     validate_positive_finite("length", properties.length)?;
720: 
721:     let e = section.elastic_modulus;
722:     let g = section.shear_modulus;
723:     let area = section.area;
724:     let iy = section.second_moment_y;
725:     let iz = section.second_moment_z;
726:     let j = section.torsion_constant;
727:     let length = properties.length;
728: 
729:     // K2a (T3 D1 revision 5a.2 section 4.7, formation range, step 1): each
730:     // coefficient is formed with the same binary64 operations, in the same
731:     // order, as before (`12.0 * e * iy / length3` is ((12*E)*Iy)/L^3), and
732:     // every intermediate product and quotient is checked, not only the final
733:     // coefficient. A zero, subnormal or non-finite intermediate of nonzero
734:     // finite operands is refused with `NumericalRange`; an accepted
735:     // coefficient has exactly the unchecked bits. k*E is formed once and
736:     // shared by the y and z coefficients (the same operation on the same
737:     // operands, so the same bits). Until formation-time scaling (K2b, F1)
738:     // lands, such a formation is refused, never scaled.
739:     let length2 = checked_formation_product("L^2 (for 6EI/L^2 and 12EI/L^3): L*L", length, length)?;
740:     let length3 = checked_formation_product("L^3 (for 12EI/L^3): L^2*L", length2, length)?;
741: 
742:     let axial = checked_formation_quotient(
743:         "EA/L: (E*A)/L",
744:         checked_formation_product("EA/L: E*A", e, area)?,
745:         length,
746:     )?;
747:     let torsion = checked_formation_quotient(
748:         "GJ/L: (G*J)/L",
749:         checked_formation_product("GJ/L: G*J", g, j)?,
750:         length,
751:     )?;
752: 
753:     let twelve_e = checked_formation_product("12EIy/L^3 and 12EIz/L^3: 12*E", 12.0, e)?;
754:     let bend_y_12 = checked_formation_quotient(
755:         "12EIy/L^3: (12*E*Iy)/L^3",
756:         checked_formation_product("12EIy/L^3: (12*E)*Iy", twelve_e, iy)?,
757:         length3,
758:     )?;
759:     let six_e = checked_formation_product("6EIy/L^2 and 6EIz/L^2: 6*E", 6.0, e)?;
760:     let bend_y_6 = checked_formation_quotient(
761:         "6EIy/L^2: (6*E*Iy)/L^2",
762:         checked_formation_product("6EIy/L^2: (6*E)*Iy", six_e, iy)?,
763:         length2,
764:     )?;
765:     let four_e = checked_formation_product("4EIy/L and 4EIz/L: 4*E", 4.0, e)?;
766:     let bend_y_4 = checked_formation_quotient(
767:         "4EIy/L: (4*E*Iy)/L",
768:         checked_formation_product("4EIy/L: (4*E)*Iy", four_e, iy)?,
769:         length,
770:     )?;
771:     let two_e = checked_formation_product("2EIy/L and 2EIz/L: 2*E", 2.0, e)?;
772:     let bend_y_2 = checked_formation_quotient(
773:         "2EIy/L: (2*E*Iy)/L",
774:         checked_formation_product("2EIy/L: (2*E)*Iy", two_e, iy)?,
775:         length,
776:     )?;
777:     let bend_z_12 = checked_formation_quotient(
778:         "12EIz/L^3: (12*E*Iz)/L^3",
779:         checked_formation_product("12EIz/L^3: (12*E)*Iz", twelve_e, iz)?,
780:         length3,
781:     )?;
782:     let bend_z_6 = checked_formation_quotient(
783:         "6EIz/L^2: (6*E*Iz)/L^2",
784:         checked_formation_product("6EIz/L^2: (6*E)*Iz", six_e, iz)?,
785:         length2,
786:     )?;
787:     let bend_z_4 = checked_formation_quotient(
788:         "4EIz/L: (4*E*Iz)/L",
789:         checked_formation_product("4EIz/L: (4*E)*Iz", four_e, iz)?,
790:         length,
791:     )?;
792:     let bend_z_2 = checked_formation_quotient(
793:         "2EIz/L: (2*E*Iz)/L",
794:         checked_formation_product("2EIz/L: (2*E)*Iz", two_e, iz)?,
795:         length,
796:     )?;
797: 
798:     let mut stiffness = [[0.0; ELEMENT_DOF]; ELEMENT_DOF];
799: 
800:     set_symmetric(&mut stiffness, UX, UX + DOF_PER_NODE, -axial);
801:     stiffness[UX][UX] = axial;
802:     stiffness[UX + DOF_PER_NODE][UX + DOF_PER_NODE] = axial;
803: 
804:     set_symmetric(&mut stiffness, RX, RX + DOF_PER_NODE, -torsion);
805:     stiffness[RX][RX] = torsion;
806:     stiffness[RX + DOF_PER_NODE][RX + DOF_PER_NODE] = torsion;
807: 
808:     add_bending_z(&mut stiffness, bend_z_12, bend_z_6, bend_z_4, bend_z_2);
809:     add_bending_y(&mut stiffness, bend_y_12, bend_y_6, bend_y_4, bend_y_2);
810: 
811:     for row in &stiffness {
812:         validate_named_finite_slice("computed local stiffness", row)?;
813:     }
814:     Ok(stiffness)
815: }

837: /// K2a: returns `value`, the unchecked binary64 result, bit for bit, unless
838: /// both operands are nonzero and finite and `value` is zero, subnormal or
839: /// non-finite, which is `NumericalRange { name }`. An exactly zero (or a
840: /// non-finite) operand is not a range error: such inputs keep their existing
841: /// handling (`validate_positive_finite` refuses them before formation).
842: fn checked_formation_value(
843:     name: &'static str,
844:     left: f64,
845:     right: f64,
846:     value: f64,
847: ) -> Result<f64, FrameKernelError> {
848:     let operands_in_range = left != 0.0 && right != 0.0 && left.is_finite() && right.is_finite();
849:     if operands_in_range && !value.is_normal() {
850:         return Err(FrameKernelError::NumericalRange { name });
851:     }
852:     Ok(value)
853: }
