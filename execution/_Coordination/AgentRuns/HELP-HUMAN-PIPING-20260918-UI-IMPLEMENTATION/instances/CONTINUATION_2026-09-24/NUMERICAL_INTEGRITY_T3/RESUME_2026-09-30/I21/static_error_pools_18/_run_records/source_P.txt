1: //! The binary64 sparse gate's parity checks on RF-MECH and RF-LARGE (brief
2: //! Scope 3; T3 D1 §4.8's parity protocol items 1–3; ROOT's C4 ruling: at up
3: //! to 100 members; plan §5.3).
4: //!
5: //! The same adapted model is formed through FK's binary64 path (FK's
6: //! `FrameElement`, global-axis springs, rigid DOFs, loads) in both modes, with
7: //! the same M03 evidence: the stiffness contributions and the formation
8: //! allowances in the form the structural adapter records them (as K1's
9: //! pattern-path tests build them).
10: //! - **Item 1:** every stored entry of the pattern assembly equals the dense
11: //!   assembly's entry bit for bit, and every entry outside the pattern is +0.
12: //! - **Item 2:** the M03 outcome class (Passed, Sensitive, or the refusal)
13: //!   is the same in both modes.
14: //! - **Item 3:** where both publish, max|u_sparse − u_dense| ≤ 1e-9·max|u_dense|
15: //!   (the DEC-053 basis), decided exactly; asserted where both are Passed and
16: //!   recorded otherwise.
17: //!
18: //! W1 is the lane's method; the binary64 values are not compared with R1 here.
19: use crate::cases::Model;

23:     SparseAssemblyOptions, SparseStiffness, SparseStructuralSystem, SparseSymmetryEvidence,
24:     StiffnessContribution, StructuralError, StructuralSolution, StructuralSystem, SymmetryEvidence,

33: const BASIS: &str = "V-K parity: the structural adapter's formation allowances";

35: /// A model in FK's binary64 terms.
36: pub struct Binary64Model {
37:     pub node_count: usize,
38:     pub frames: Vec<FrameElement>,
39:     pub springs: Vec<(usize, f64)>,
40:     pub force: Vec<f64>,
41:     pub free: Vec<usize>,
42:     pub prescribed: Vec<(usize, f64)>,
43: }
44: 
45: pub fn binary64_model(model: &Model) -> Result<Binary64Model, String> {
46:     let nodes: Vec<FrameNode> = model
47:         .nodes
48:         .iter()
49:         .enumerate()
50:         .map(|(k, p)| FrameNode::new(k, *p).map_err(|e| format!("{e:?}")))
51:         .collect::<Result<_, _>>()?;
52:     let frames = model
53:         .members
54:         .iter()
55:         .map(|m| {
56:             let section = FrameSection::new(
57:                 m.elastic_modulus,
58:                 m.shear_modulus,
59:                 m.area,
60:                 m.second_moment_y,
61:                 m.second_moment_z,
62:                 m.torsion_constant,
63:             )
64:             .map_err(|e| format!("{e:?}"))?;
65:             FrameElement::new(
66:                 nodes[m.node_i as usize],
67:                 nodes[m.node_j as usize],
68:                 section,
69:                 m.y_reference,
70:             )
71:             .map_err(|e| format!("{e:?}"))
72:         })
73:         .collect::<Result<Vec<_>, _>>()?;
74:     let mut springs = Vec::new();
75:     for s in &model.springs {
76:         match s.axis {
77:             Some(a) => springs.push((s.node as usize * 6 + a, s.stiffness)),
78:             None => return Err("a directional spring has no binary64 form".into()),
79:         }
80:     }
81:     let n = model.nodes.len() * 6;
82:     let mut force = vec![0.0; n];
83:     for (node, c, v, _) in &model.loads {
84:         force[*node as usize * 6 + c] += v;
85:     }
86:     let restrained: Vec<usize> = model
87:         .constraints
88:         .iter()
89:         .map(|&(node, c)| node as usize * 6 + c)
90:         .collect();
91:     let free = (0..n).filter(|g| !restrained.contains(g)).collect();
92:     let mut prescribed: Vec<(usize, f64)> = restrained.iter().map(|&g| (g, 0.0)).collect();
93:     prescribed.sort_by_key(|p| p.0);
94:     Ok(Binary64Model {
95:         node_count: model.nodes.len(),
96:         frames,
97:         springs,
98:         force,
99:         free,
100:         prescribed,
101:     })
102: }

104: /// Both triangles of every element's and spring's contribution, before
105: /// coalescence.
106: fn contributions(m: &Binary64Model) -> Vec<StiffnessContribution> {
107:     let mut out = Vec::new();
108:     for e in &m.frames {
109:         let k = e.global_stiffness().expect("formed");
110:         let map = element_dof_map(e.node_i.index, e.node_j.index);
111:         for i in 0..12 {
112:             for j in 0..12 {
113:                 out.push(StiffnessContribution {
114:                     row: map[i],
115:                     col: map[j],
116:                     value: k[i][j],
117:                 });
118:             }
119:         }
120:     }
121:     for &(dof, value) in &m.springs {
122:         out.push(StiffnessContribution {
123:             row: dof,
124:             col: dof,
125:             value,
126:         });
127:     }
128:     out
129: }

131: /// The formation allowances and operation counts per (row, col), in the
132: /// structural adapter's form: each element's `transform_roundoff`, one count
133: /// per spring, and γ(scatter)·Σ|contributions|.
134: fn allowances(
135:     m: &Binary64Model,
136:     c: &[StiffnessContribution],
137: ) -> BTreeMap<(usize, usize), (f64, usize)> {
138:     let mut out: BTreeMap<(usize, usize), (f64, usize)> = BTreeMap::new();
139:     for e in &m.frames {
140:         let t = e.orientation().expect("oriented").transformation_matrix();
141:         let evidence =
142:             transform_roundoff(&e.local_stiffness().expect("formed"), &t).expect("finite");
143:         let map = element_dof_map(e.node_i.index, e.node_j.index);
144:         for i in 0..12 {
145:             for j in 0..12 {
146:                 let slot = out.entry((map[i], map[j])).or_default();
147:                 slot.0 += evidence.absolute_roundoff[i][j];
148:                 slot.1 += evidence.operation_counts[i][j] + 1;
149:             }
150:         }
151:     }
152:     for &(dof, _) in &m.springs {
153:         out.entry((dof, dof)).or_default().1 += 1;
154:     }
155:     let mut magnitude: BTreeMap<(usize, usize), (f64, usize)> = BTreeMap::new();
156:     for x in c {
157:         let slot = magnitude.entry((x.row, x.col)).or_default();
158:         slot.0 += x.value.abs();
159:         slot.1 += 1;
160:     }
161:     for (key, (mag, scatter)) in magnitude {
162:         out.entry(key).or_default().0 += gamma(scatter) * mag;
163:     }
164:     out
165: }

167: /// The M03 outcome class of a result.
168: pub fn class(r: &Result<StructuralSolution, StructuralError>) -> String {
169:     match r {
170:         Ok(s) => format!("{:?}", s.report.quality),
171:         Err(StructuralError::InvalidInput(w)) => format!("InvalidInput({w})"),
172:         Err(StructuralError::Range(w)) => format!("Range({w})"),
173:         Err(StructuralError::Asymmetric { .. }) => "Asymmetric".into(),
174:         Err(StructuralError::NumericallyUnresolved { reason, .. }) => {
175:             format!("NumericallyUnresolved({reason})")
176:         }
177:         Err(StructuralError::NegativeEnergy { .. }) => "NegativeEnergy".into(),
178:         Err(StructuralError::Mechanism { .. }) => "Mechanism".into(),
179:     }
180: }

182: #[derive(Clone, Debug)]
183: pub struct Parity {
184:     /// Item 1: None when the pattern and dense assemblies agree bit for bit
185:     /// (only where the dense mode runs).
186:     pub bitwise_k: Option<Option<String>>,
187:     pub sparse: String,
188:     pub dense: Option<String>,
189:     /// Item 3 where both modes publish: within the DEC-053 basis, and the
190:     /// ratio max|Δu|/(1e-9·max|u_dense|) for display.
191:     pub delta: Option<(bool, f64)>,
192:     pub both_passed: bool,
193: }
194: 
195: fn max_abs(v: &[f64]) -> f64 {
196:     v.iter().fold(0.0f64, |m, x| m.max(x.abs()))
197: }
198: 
199: /// Items 1–3 on one model; `dense` runs the dense mode too (≤ ~100 members).
200: pub fn parity(model: &Model, dense: bool) -> Result<Parity, String> {
201:     let m = binary64_model(model)?;
202:     let c = contributions(&m);
203:     let allow = allowances(&m, &c);
204:     let sparse_k: SparseStiffness = assemble_sparse_stiffness(
205:         m.node_count,
206:         &m.frames,
207:         &[],
208:         &[],
209:         &m.springs,
210:         &SparseAssemblyOptions::new(),
211:     )
212:     .map_err(|e| format!("{e:?}"))?;
213:     let n = m.node_count * 6;
214:     let pattern = sparse_k.pattern();
215:     let (mut s_round, mut s_count) = (Vec::new(), Vec::new());
216:     for row in 0..n {
217:         for e in pattern.row_range(row) {
218:             let (r, k) = allow
219:                 .get(&(row, pattern.column(e)))
220:                 .copied()
221:                 .unwrap_or((0.0, 0));
222:             s_round.push(r);
223:             s_count.push(k);
224:         }
225:     }
226:     let ss = SparseStructuralSystem::new(
227:         &sparse_k,
228:         &m.force,
229:         &m.free,
230:         &m.prescribed,
231:         Some(&c),
232:         Some(SparseSymmetryEvidence {
233:             absolute_roundoff: &s_round,
234:             operation_counts: &s_count,
235:             basis: BASIS,
236:         }),
237:     );
238:     let sparse = solve_sparse_structural(&ss);
239:     let mut out = Parity {
240:         bitwise_k: None,
241:         sparse: class(&sparse),
242:         dense: None,
243:         delta: None,
244:         both_passed: false,
245:     };
246:     if !dense {
247:         return Ok(out);
248:     }
249:     let mut dense_k =
250:         assemble_global_stiffness(m.node_count, &m.frames).map_err(|e| format!("{e:?}"))?;
251:     for &(dof, v) in &m.springs {
252:         dense_k[dof][dof] += v;
253:     }
254:     let mut mismatch = None;
255:     'rows: for (row, dense_row) in dense_k.iter().enumerate() {
256:         for (col, &d) in dense_row.iter().enumerate() {
257:             let stored = pattern.find(row, col).map(|_| sparse_k.get(row, col));
258:             let ok = match stored {
259:                 Some(v) => v.to_bits() == d.to_bits(),
260:                 None => d.to_bits() == 0.0f64.to_bits(),
261:             };
262:             if !ok {
263:                 mismatch = Some(format!("({row}, {col}): pattern {stored:?}, dense {d:e}"));
264:                 break 'rows;
265:             }
266:         }
267:     }
268:     out.bitwise_k = Some(mismatch);
269:     let mut d_round = vec![vec![0.0; n]; n];
270:     let mut d_count = vec![vec![0usize; n]; n];
271:     for (&(i, j), &(r, k)) in &allow {
272:         d_round[i][j] = r;
273:         d_count[i][j] = k;
274:     }
275:     let ds = StructuralSystem {
276:         stiffness: &dense_k,
277:         force: &m.force,
278:         free_dofs: &m.free,
279:         prescribed: &m.prescribed,
280:         contributions: Some(&c),
281:         symmetry: Some(SymmetryEvidence {
282:             absolute_roundoff: &d_round,
283:             operation_counts: &d_count,
284:             basis: BASIS,
285:         }),
286:     };
287:     let dense_r = solve_structural_dense(&ds);
288:     out.dense = Some(class(&dense_r));
289:     if let (Ok(s), Ok(d)) = (&sparse, &dense_r) {
290:         let scale = Exact::from_f64(max_abs(&d.displacements)).scaled(0, -9);
291:         let (mut worst, mut worst_x) = (Exact::zero(), 0.0f64);
292:         for (a, b) in s.displacements.iter().zip(&d.displacements) {
293:             let diff = Exact::from_f64(*a).sub(&Exact::from_f64(*b)).abs();
294:             if diff.cmp(&worst) == Ordering::Greater {
295:                 worst_x = (a - b).abs();
296:                 worst = diff;
297:             }
298:         }
299:         let within = worst.cmp(&scale) != Ordering::Greater;
300:         let ratio = worst_x / (1e-9 * max_abs(&d.displacements));
301:         out.delta = Some((within, ratio));
302:         out.both_passed = out.sparse == "Passed" && out.dense.as_deref() == Some("Passed");
303:     }
304:     Ok(out)
305: }
