async(kind,id,index=0)=>{
 if(load("pg3_current"))throw Error("owned process active");
 if(Date.now()>=Date.parse(load("pg3_end")))throw Error("G3 boundary");
 if(kind==="build"&&!load("pg3_releases").some(r=>r.id===id))throw Error("one-copy manager release absent");
 const g=load("pg3_groups").find(g=>g.runtime_id===id),b=load("pg3_bindings")[id];
 const name=kind==="build"?"build":kind+"_"+String(index+1).padStart(2,"0");
 const log=b.logs+"/"+name;
 const mk=await tools.exec_command({cmd:"mkdir -p '"+b.logs+"'",max_output_tokens:100});if(mk.exit_code!==0)throw Error("logdir: "+mk.output);
 const guard=await tools.exec_command({cmd:"ps -p 5387 -o pid=,ppid=,etime=,command=",max_output_tokens:300});if(guard.exit_code!==0||!guard.output.includes("memguard.sh"))throw Error("guard");
 const sid=kind==="control"?"BASELINE":id;
 const bid=kind==="control"?(g.target==="--lib"?"BASELINE_LIB":"BASELINE_S11"):id;
 const sb=load("pg3_bindings")[sid];if(!sb)throw Error("cwd unbound");
 let argv,filter=null;
 if(kind==="build")argv=g.compile.map(a=>a.replace("<wt>",load("cs_wt")));
 else{const bin=load("pg3_builds")[bid];if(!bin)throw Error("binary unbound");filter=g.filters[index];argv=[bin.binary.path,...(kind==="list"?["--list","--exact",filter]:["--exact",filter,"--test-threads=2","--nocapture"])];}
 const q=s=>"'"+s.replace(/'/g,"'\\''")+"'";
 const cmd="/usr/bin/time -l env -u RUSTFLAGS -u CARGO_ENCODED_RUSTFLAGS -u RUSTC_WRAPPER -u RUSTC_WORKSPACE_WRAPPER GIT_OPTIONAL_LOCKS=0 RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2 CARGO_NET_OFFLINE=true FK_SEEDED_FAULT=NONE "+argv.map(a=>a==="<CARGO>"?'"$HOME/.cargo/bin/cargo"':q(a)).join(" ")+" > "+q(log+".stdout")+" 2> "+q(log+".stderr");
 const r={kind,id,registration:g.group+"::"+g.id,index,name,source_id:sid,binary_id:bid,filter,argv,cmd,workdir:sb.source+"/projects/chirality-piping/core/solver/frame_kernel",log,guard_before:guard,launch_utc:new Date().toISOString()};
 r.tool_initial=await tools.exec_command({cmd,workdir:r.workdir,tty:true,yield_time_ms:1000,max_output_tokens:300});store("pg3_current",r);return {kind,id,index,launch:r.launch_utc,session:r.tool_initial.session_id,exit:r.tool_initial.exit_code};
}
