4: //! dense-derived one. `order_sparse_structural` takes the adjacency from the
5: //! prepared system's stored entries (`adjacency_from_symmetric_entries`) and
6: //! the skyline from `SymmetricProfileMatrix::from_entries_with_order`. Both
7: //! skip an exactly zero value, as `adjacency_from_dense` and

9: //! prepared matrix the order and the profile are today's.

19: pub fn factor_structural_ldlt<'p, 's>(
20:     prepared: &'p PreparedSystem<'s>,
21: ) -> Result<PositiveFactor<'p, 's>, StructuralError> {
22:     let matrix = prepared.matrix();
23:     let adjacency = adjacency_from_dense(matrix)
24:         .map_err(|_| StructuralError::InvalidInput("sparse adjacency"))?;
25:     let order = reverse_cuthill_mckee(&adjacency)
26:         .map_err(|_| StructuralError::InvalidInput("sparse order"))?;
27:     let profile = SymmetricProfileMatrix::from_dense_with_order(matrix, &order)
28:         .map_err(|_| StructuralError::InvalidInput("sparse profile"))?;
29:     structural::factor_structural_profile(prepared, &order, &profile.first_columns)
30: }
31: pub fn solve_structural_sparse(
32:     system: &StructuralSystem<'_>,
33: ) -> Result<StructuralSolution, StructuralError> {
34:     let prepared = structural::prepare_structural(system)?;
35:     let factor = match factor_structural_ldlt(&prepared) {
36:         Ok(f) => f,
37:         Err(error) => return Err(structural::negative_pair_witness(&prepared)?.unwrap_or(error)),
38:     };
39:     structural::finish_structural(&factor)
40: }

42: /// The ordering and skyline of a sparse prepared system, with its
43: /// deterministic storage counts (the resource guard reads
44: /// `profile_entry_count` before the factor allocates its rows).
45: #[derive(Debug, Clone, PartialEq, Eq)]
46: pub struct SparseStructuralOrdering {
47:     /// `order[k]` is the free position of the `k`-th ordered row.
48:     pub order: Vec<usize>,
49:     pub first_columns: Vec<usize>,
50:     pub profile_entry_count: usize,
51:     pub max_half_bandwidth: usize,
52: }
53: 
54: /// The pattern path's ordering: RCM on the adjacency of the stored nonzero
55: /// entries, and the skyline of those entries under that order.
56: pub fn order_sparse_structural(
57:     prepared: &SparsePreparedSystem<'_>,
58: ) -> Result<SparseStructuralOrdering, StructuralError> {
59:     let n = prepared.dimension();
60:     let entries: Vec<SymmetricMatrixEntry> = prepared
61:         .lower_entries()
62:         .map(|(row, col, value)| SymmetricMatrixEntry { row, col, value })
63:         .collect();
64:     let adjacency = adjacency_from_symmetric_entries(n, &entries)
65:         .map_err(|_| StructuralError::InvalidInput("sparse adjacency"))?;
66:     let order = reverse_cuthill_mckee(&adjacency)
67:         .map_err(|_| StructuralError::InvalidInput("sparse order"))?;
68:     let profile = SymmetricProfileMatrix::from_entries_with_order(n, &entries, &order)
69:         .map_err(|_| StructuralError::InvalidInput("sparse profile"))?;
70:     Ok(SparseStructuralOrdering {
71:         profile_entry_count: profile.profile_entry_count(),
72:         max_half_bandwidth: profile.max_half_bandwidth(),
73:         first_columns: profile.first_columns,
74:         order,
75:     })
76: }

78: /// The pattern path's sibling of `factor_structural_ldlt`.
79: pub fn factor_sparse_structural_ldlt<'p, 's>(
80:     prepared: &'p SparsePreparedSystem<'s>,
81: ) -> Result<SparsePositiveFactor<'p, 's>, StructuralError> {
82:     let ordering = order_sparse_structural(prepared)?;
83:     structural::factor_sparse_structural_profile(prepared, &ordering.order, &ordering.first_columns)
84: }

86: /// Factor, or the witness on a refused factor, then the completion: the
