import json,sys,collections
d=json.load(open(sys.argv[1]))
cnt=collections.defaultdict(collections.Counter)
for c in d['cases']:
    for mode,m in c.get('modes',{}).items():
        for n in m.get('negative_controls',[]):
            o=n['outcome']
            k='matches wrong' if o.startswith('MAIN_MATCHES') else ('partial' if o.startswith('partially') else ('distinct' if o.startswith('distinct') else 'not evaluable'))
            cnt[c['family']][k]+=1
print('| Family | NC evaluations (both modes) | main matches the wrong answer | partially | distinct from the wrong answer | not evaluable |')
print('|---|---:|---:|---:|---:|---:|')
for f,v in cnt.items():
    print('| %s | %d | %d | %d | %d | %d |'%(f,sum(v.values()),v['matches wrong'],v['partial'],v['distinct'],v['not evaluable']))
print()
for c in d['cases']:
    for mode,m in c.get('modes',{}).items():
        for n in m.get('negative_controls',[]):
            if n['outcome'].startswith('MAIN_MATCHES') and c['family'] not in ('RF-CANCEL',):
                print(c['id'],mode,n['id'],n['outcome'][:60], 'trusted' if m.get('published_as_trusted') else '')
