"""T3 P1 follow-up: typed entry on the captured S11-class cases plus the skew case (committed as run_typed.py.txt).
Usage: python3 -B run_typed.py <probe-binary> <gen_out-dir> <results.json> <runs.jsonl>"""
import json, os, sys
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from run import run_one
binary, gen, resp, runs = sys.argv[1:5]
r = json.load(open(resp))
ids = sorted({e['case'] for e in r['exceptions']['captured']['s11_class']}) + ['RF-SKEW-T-CANT-OFF-122-r1e-04']
cases = {c['id']: c for c in json.load(open(os.path.join(gen, 'cases.json')))}
with open(runs, 'a') as fh:
    for cid in ids:
        for mode in ('sparse_interactive', 'dense_scrutiny'):
            rec = run_one(binary, os.path.join(gen, cases[cid]['request_file']), mode, True, 600)
            rec.update(case=cid, mode=mode, entry='typed_on_captured')
            fh.write(json.dumps(rec) + '\n')
            run = rec.get('probe', {}).get('run', {})
            print(cid, mode, 'ok' if run.get('ok') else 'ERR', rec['wall_seconds'], rec['peak_rss_kib'], flush=True)
print(len(ids), 'cases')
