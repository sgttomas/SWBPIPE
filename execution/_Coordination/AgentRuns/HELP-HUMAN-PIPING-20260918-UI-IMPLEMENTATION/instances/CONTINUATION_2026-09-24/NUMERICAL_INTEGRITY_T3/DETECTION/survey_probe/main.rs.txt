//! T3 P1 realistic-fixture survey probe. Scratch copy only; never part of a product crate.
//! Usage: t3_p1_survey <request.json> <sparse_interactive|dense_scrutiny>
//! Runs one committed request through the captured public entry and prints one JSON line with
//! the producer, numerical_quality, standing, receipt summary, every diagnostic (integrity text
//! in full), the summary headline and every published result row (id, kind, unit, value, basis).
use open_pipe_stress_product_physics::{run_linear_static_preview_value_with_mode, PreviewSolverMode};
use open_pipe_stress_result_export::semantic_contract::numerical_use_standing_with_context;
use serde_json::{json, Value};

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let request: Value = serde_json::from_str(&std::fs::read_to_string(&args[1]).expect("file")).expect("json");
    let mode = PreviewSolverMode::from_wire(&args[2]).expect("mode");
    let out = match run_linear_static_preview_value_with_mode(request.clone(), mode) {
        Err(e) => json!({"ok": false, "error": e}),
        Ok(e) => {
            let v = serde_json::to_value(&e).unwrap();
            let requested: Vec<Value> = request["model"]["load_cases"].as_array().map(|cs| cs.iter()
                .map(|c| json!({"ref_type":"load_case","ref_id":c["id"]})).collect()).unwrap_or_default();
            let invocation = json!({"request": request, "solver_mode": mode.as_str()});
            let standing = numerical_use_standing_with_context(&v, &requested, Some(&invocation));
            let rows: Vec<Value> = v["results"].as_array().unwrap().iter().map(|r| json!({
                "id": r["id"], "kind": r["kind"], "unit": r["unit"], "value": r["value"],
                "basis": r["basis_ref"]["ref_id"], "component": r["metadata"]["component"]})).collect();
            let diags: Vec<Value> = e.diagnostics.iter().map(|d| json!({"id": d.id, "code": d.code, "severity": d.severity, "message": d.message})).collect();
            let receipt = v.get("source_block_recovery").filter(|r| !r.is_null()).map(|r| json!({
                "status": r["body"]["status"],
                "cases": r["body"]["cases"].as_array().map(|cs| cs.iter().map(|c| json!({"basis_ref": c["basis_ref"], "outcome": c["outcome"], "selected_method": c["selected_method"]})).collect::<Vec<_>>())}));
            json!({"ok": true, "producer": v["producer"], "numerical_quality": v["numerical_quality"], "status": v["status"],
                   "standing": standing, "receipt": receipt, "summary": v["summary"], "diagnostics": diags, "results": rows})
        }
    };
    println!("{}", serde_json::to_string(&json!({"mode": mode.as_str(), "run": out})).unwrap());
}
