import json, re, subprocess, collections
t = open('out/tables.md').read()
sec = {}
for part in t.split('## ')[1:]:
    name, body = part.split('\n', 1)
    sec[name.strip()] = body.strip()
tpl = open('RETURN.template.md').read()
tpl = tpl.replace('{{FAMILY_TABLE}}', sec['family'])
tpl = tpl.replace('{{SILENT_TABLE}}', 'Captured entry, both modes. Each row is one case and mode; every breaching quantity, with observed, expected, scale and ratio, is in `results.json` under `exceptions`. Counts: S11 class, 100 distinct (case, quantity) pairs in 15 cases (200 case, mode and quantity rows); other class, 4 distinct pairs in 1 case (6 rows).\n\n' + sec['silent'])
tpl = tpl.replace('{{UDL_NC}}', 'R1 fold values 2.2754050722e-8 (A-node-B, R1 ratio 1.01) and 2.2754050756e-8 (A-B-node, R1 ratio 2.53); main is 46.5 times the criterion from the reference, which implies a net moment at S1 about 4.6e-8 relative high')
nc = subprocess.run(['python3', 'ncsum.py', 'res2.json'], capture_output=True, text=True).stdout.split('\n\n')[0]
tpl = tpl.replace('{{NC_TABLE}}', nc.strip())
tpl = tpl.replace('{{MEMORY_TABLE}}', sec['memory'])
d = json.load(open('res2.json'))
rows = ['| Case | Mode | Outcome | Quality | Standing (with a P1-supplied invocation) | pass / mismatch | Worst ratio |', '|---|---|---|---|---|---|---|']
for c in d['cases']:
    for mode, m in c.get('modes', {}).items():
        tt = m.get('typed_secondary')
        if not tt:
            continue
        i = tt.get('identity') or {}
        rows.append('| %s | %s | %s | %s | %s | %s | %s |' % (c['id'], mode, tt['outcome'], i.get('numerical_quality_status', ''), i.get('standing', ''),
                    '%d / %d' % (tt['counts']['pass'], tt['counts']['mismatch']) if tt.get('counts') else '', tt.get('max_ratio', '')))
tpl = tpl.replace('{{TYPED_TABLE}}', '\n'.join(rows))
# survey: solved rows in full; blocked or error runs grouped
sv = json.load(open('survey.json'))
lines = sec['survey'].split('\n')
head, body = lines[:2], lines[2:]
solved = [l for l in body if '| solved' in l]
grp = collections.defaultdict(list)
for f in sv['fixtures']:
    for mode, m in f['modes'].items():
        if m.get('outcome') != 'solved':
            key = ', '.join(m.get('blocking', [])) or ('Err: ' + m.get('error', '')[:60])
            grp[key].append(f['fixture'])
blk = ['', 'Runs that did not solve (both modes each, grouped by blocking code):', '']
for k, v in grp.items():
    names = sorted(set(v))
    blk.append('- `%s`: %d fixtures (%s)' % (k, len(names), '; '.join(n.split('/')[-1] for n in names[:6]) + ('; ...' if len(names) > 6 else '')))
tpl = tpl.replace('{{SURVEY_TABLE}}', '\n'.join(head + solved + blk))
open('RETURN.md', 'w').write(tpl)
print(len(tpl), re.findall(r'\{\{[A-Z_]+\}\}', tpl))
