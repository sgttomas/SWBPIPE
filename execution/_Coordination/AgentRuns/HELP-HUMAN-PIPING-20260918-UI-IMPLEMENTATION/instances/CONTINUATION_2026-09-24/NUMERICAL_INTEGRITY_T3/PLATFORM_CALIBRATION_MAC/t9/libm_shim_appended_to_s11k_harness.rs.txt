// ---- calibration only (not S11-K's harness): every libm entry the product reaches is shimmed.
// CALIB_LIBM_LOG=<file> logs each call (name, arg bits, platform result bits);
// CALIB_LIBM_TABLE=<file> replays results from a table (misses fall back to the platform libm).
mod libm_shim {
    use std::collections::HashMap;
    use std::ffi::c_void;
    use std::io::Write;
    use std::sync::{Mutex, OnceLock};
    extern "C" { fn dlsym(handle: *mut c_void, symbol: *const std::ffi::c_char) -> *mut c_void; }
    const RTLD_NEXT: *mut c_void = -1isize as *mut c_void;
    static LOG: OnceLock<Option<Mutex<std::fs::File>>> = OnceLock::new();
    static TABLE: OnceLock<HashMap<(String, u64, u64), u64>> = OnceLock::new();
    fn log() -> &'static Option<Mutex<std::fs::File>> {
        LOG.get_or_init(|| std::env::var("CALIB_LIBM_LOG").ok().map(|p| Mutex::new(std::fs::OpenOptions::new().create(true).append(true).open(p).unwrap())))
    }
    fn table() -> &'static HashMap<(String, u64, u64), u64> {
        TABLE.get_or_init(|| {
            let mut m = HashMap::new();
            if let Ok(p) = std::env::var("CALIB_LIBM_TABLE") {
                for line in std::fs::read_to_string(p).unwrap().lines() {
                    let f: Vec<&str> = line.split_whitespace().collect();
                    m.insert((f[0].to_string(), u64::from_str_radix(f[1], 16).unwrap(), u64::from_str_radix(f[2], 16).unwrap()), u64::from_str_radix(f[3], 16).unwrap());
                }
            }
            m
        })
    }
    fn sym(name: &str) -> *mut c_void {
        let c = std::ffi::CString::new(name).unwrap();
        unsafe { dlsym(RTLD_NEXT, c.as_ptr()) }
    }
    pub fn call(name: &str, x: f64, y: f64, two: bool) -> f64 {
        let native = unsafe {
            if two { std::mem::transmute::<_, extern "C" fn(f64, f64) -> f64>(sym(name))(x, y) }
            else { std::mem::transmute::<_, extern "C" fn(f64) -> f64>(sym(name))(x) }
        };
        let key = (name.to_string(), x.to_bits(), if two { y.to_bits() } else { 0 });
        if let Some(l) = log() { writeln!(l.lock().unwrap(), "{} {:016x} {:016x} {:016x}", name, key.1, key.2, native.to_bits()).unwrap(); }
        match table().get(&key) { Some(bits) => f64::from_bits(*bits), None => native }
    }
}
macro_rules! shim1 { ($($n:ident),*) => { $( #[no_mangle] pub extern "C" fn $n(x: f64) -> f64 { libm_shim::call(stringify!($n), x, 0.0, false) } )* } }
macro_rules! shim2 { ($($n:ident),*) => { $( #[no_mangle] pub extern "C" fn $n(x: f64, y: f64) -> f64 { libm_shim::call(stringify!($n), x, y, true) } )* } }
shim1!(exp, expm1, log, log1p, log10, log2, exp2, sin, cos, tan, asin, acos, atan, sinh, cosh, tanh, cbrt);
shim2!(hypot, atan2, pow);
#[repr(C)]
pub struct SinCos { s: f64, c: f64 }
// macOS: LLVM fuses sin/cos pairs into __sincos_stret.
#[no_mangle]
pub extern "C" fn __sincos_stret(x: f64) -> SinCos {
    SinCos { s: libm_shim::call("sin", x, 0.0, false), c: libm_shim::call("cos", x, 0.0, false) }
}
