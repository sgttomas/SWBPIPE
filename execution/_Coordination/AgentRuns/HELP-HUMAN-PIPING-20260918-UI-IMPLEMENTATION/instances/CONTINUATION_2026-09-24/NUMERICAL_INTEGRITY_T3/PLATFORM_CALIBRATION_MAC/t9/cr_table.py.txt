"""Correctly rounded results for every logged libm call (exact rationals; Decimal at 100 digits)."""
import struct, sys
from fractions import Fraction as F
from decimal import Decimal, getcontext
getcontext().prec = 100
b2f = lambda h: struct.unpack('<d', struct.pack('<Q', int(h, 16)))[0]
f2b = lambda x: '%016x' % struct.unpack('<Q', struct.pack('<d', x))[0]
ul = lambda a, b: abs(struct.unpack('<q', struct.pack('<d', a))[0] - struct.unpack('<q', struct.pack('<d', b))[0])
dec = lambda x: Decimal(F(x).numerator) / Decimal(F(x).denominator)
def cr(n, x, y):
    if n == 'hypot':
        s = F(x) ** 2 + F(y) ** 2
        return float((Decimal(s.numerator) / Decimal(s.denominator)).sqrt())
    if n == 'exp': return float(dec(x).exp())
    if n == 'expm1': return float(dec(x).exp() - 1)
    raise SystemExit('no reference for ' + n)
src, dst = sys.argv[1:3]
tot, bad = {}, {}
with open(dst, 'w') as out:
    for line in open(src):
        n, xb, yb, rb = line.split()
        x, y, nat = b2f(xb), b2f(yb), b2f(rb)
        c = cr(n, x, y)
        tot[n] = tot.get(n, 0) + 1
        if c != nat: bad.setdefault(n, []).append((x, y, nat, c))
        out.write(f'{n} {xb} {yb} {f2b(c)}\n')
for n in sorted(tot):
    d = bad.get(n, [])
    print(f'{n}: macOS differs from correctly rounded on {len(d)} of {tot[n]} distinct arguments (max {max([ul(a[2], a[3]) for a in d], default=0)} ulp)')
