// ---- calibration only (not P1's probe): macOS cannot set RLIMIT_AS, so the heap is capped at
// 6 GiB here instead. A refused allocation returns null, and Rust aborts with "memory allocation
// of N bytes failed", as the Linux runs do when RLIMIT_AS (6 GiB) refuses the mapping.
mod heap_cap {
    use std::alloc::{GlobalAlloc, Layout, System};
    use std::sync::atomic::{AtomicUsize, Ordering};
    const CAP: usize = 6 * 1024 * 1024 * 1024;
    static USED: AtomicUsize = AtomicUsize::new(0);
    pub struct Capped;
    fn reserve(n: usize) -> bool {
        let mut cur = USED.load(Ordering::Relaxed);
        loop {
            let Some(next) = cur.checked_add(n) else { return false };
            if next > CAP { return false; }
            match USED.compare_exchange_weak(cur, next, Ordering::Relaxed, Ordering::Relaxed) { Ok(_) => return true, Err(c) => cur = c }
        }
    }
    fn release(n: usize) { USED.fetch_sub(n, Ordering::Relaxed); }
    unsafe impl GlobalAlloc for Capped {
        unsafe fn alloc(&self, l: Layout) -> *mut u8 {
            if !reserve(l.size()) { return std::ptr::null_mut(); }
            let p = System.alloc(l); if p.is_null() { release(l.size()); } p
        }
        unsafe fn alloc_zeroed(&self, l: Layout) -> *mut u8 {
            if !reserve(l.size()) { return std::ptr::null_mut(); }
            let p = System.alloc_zeroed(l); if p.is_null() { release(l.size()); } p
        }
        unsafe fn dealloc(&self, p: *mut u8, l: Layout) { System.dealloc(p, l); release(l.size()); }
        unsafe fn realloc(&self, p: *mut u8, l: Layout, new: usize) -> *mut u8 {
            if new > l.size() && !reserve(new - l.size()) { return std::ptr::null_mut(); }
            let q = System.realloc(p, l, new);
            if q.is_null() { if new > l.size() { release(new - l.size()); } } else if new < l.size() { release(l.size() - new); }
            q
        }
    }
    #[global_allocator]
    static A: Capped = Capped;
}
