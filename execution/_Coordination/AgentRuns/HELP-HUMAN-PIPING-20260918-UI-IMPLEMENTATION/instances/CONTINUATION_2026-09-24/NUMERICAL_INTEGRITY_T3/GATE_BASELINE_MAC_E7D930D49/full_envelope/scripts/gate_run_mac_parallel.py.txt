"""Calibration-only parallel driver for gate part 1 on macOS (scratch; not a gate of record).

Case set, ordering key, modes, entries and timeouts as gate_run_parts.py part1 (the 4 dense
timeouts excluded). run.py's run_one is imported unchanged, except that resource.setrlimit is a
no-op: macOS cannot set RLIMIT_AS. The probe binary instead caps its own heap at 6 GiB (see its
main.rs). Runs execute in a small thread pool; records are appended as they finish.
"""
import json, os, sys, time
from concurrent.futures import ThreadPoolExecutor, as_completed
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import run as R  # noqa: E402
R.resource.setrlimit = lambda *a, **k: None
binary, gen, runs_path, workers = sys.argv[1], sys.argv[2], sys.argv[3], int(sys.argv[4])
TIMEOUTS = {('RF-LARGE-CHAIN-n01000-ROT', 'dense_scrutiny'), ('RF-LARGE-TREE-n01000-AX', 'dense_scrutiny')}
cases = [c for c in json.load(open(os.path.join(gen, 'cases.json'))) if 'request_file' in c]
jobs = []
for c in cases:
    large = c.get('members', 0) >= 1000
    for mode in ('sparse_interactive', 'dense_scrutiny'):
        if (c['id'], mode) in TIMEOUTS:
            continue
        for entry, typed in (('captured', False), ('typed', True)):
            jobs.append((c, mode, entry, typed, R.TIMEOUT_LARGE if large else R.TIMEOUT_SMALL))
jobs.sort(key=lambda j: -j[0].get('members', 0))
def go(j):
    c, mode, entry, typed, to = j
    rec = R.run_one(binary, os.path.join(gen, c['request_file']), mode, typed, to)
    rec.update(case=c['id'], mode=mode, entry=entry, rlimit_as_enforced=False, heap_cap_bytes=6 * 1024 ** 3,
               memorystatus_level_end=int(os.popen('sysctl -n kern.memorystatus_level').read().strip() or -1))
    return rec
t0 = time.time()
with open(runs_path, 'w') as fh, ThreadPoolExecutor(workers) as ex:
    futs = [ex.submit(go, j) for j in jobs]
    for n, f in enumerate(as_completed(futs), 1):
        rec = f.result()
        fh.write(json.dumps(rec) + '\n'); fh.flush()
        run = rec.get('probe', {}).get('run', {})
        if n % 50 == 0 or rec['wall_seconds'] > 60:
            print(n, '/', len(jobs), rec['case'], rec['mode'], rec['entry'], 'ok' if run.get('ok') else 'not-ok', rec['wall_seconds'], 'mem%', rec['memorystatus_level_end'], flush=True)
print('done', len(jobs), 'runs in', round(time.time() - t0), 's', flush=True)
