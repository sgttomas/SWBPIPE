"""K-D5 (I3) no-Passed-breach gate: evaluation (scratch; committed as gate_check.py.txt).

Usage: python3 -B gate_check.py <gen_out-dir> <runs.jsonl> <references.json> <out.json> <exceptions.json>...
(the pinned exception files: GATE/S11_EXCEPTIONS.json and, since ROOT's F12 ruling db665f2cb,
GATE/FORMATION_EXCEPTIONS.json)

For every (entry, case, mode) run of a frozen-reference case (ids in references.json), a quantity
breaches when compare.py's predicate fails (P1's comparator, imported unchanged: exact rationals,
|obs - exp| <= 1e-9 max(|exp|, scale)). A breach counts for the gate when the envelope is published
as trusted (P1's rule: numerical_quality status checks_passed or standing numerically_eligible).
Triples (entry, case, quantity) are merged over modes and compared with the pinned exceptions:
the gate passes iff every trusted breach is a pinned triple. Pinned triples no longer breached are
listed (informational).
"""
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import compare as C  # noqa: E402

gen, runs_path, refs_path, out_path = sys.argv[1:5]
exc_paths = sys.argv[5:]
frozen = set(json.load(open(refs_path))['cases'])
pinned = set()
for exc_path in exc_paths:
    d = json.load(open(exc_path))
    pinned |= {tuple(t) for t in d.get('triples', [])}
    pinned |= {tuple(r[:3]) for r in d.get('rows', [])}
cases = {c['id']: c for c in json.load(open(os.path.join(gen, 'cases.json')))}
actual, trusted_runs, runs_seen, per_run = set(), 0, 0, []
for line in open(runs_path):
    rec = json.loads(line)
    c = cases[rec['case']]
    if rec['case'] not in frozen:
        continue
    runs_seen += 1
    outcome, why = C.classify(rec)
    row = {'case': rec['case'], 'mode': rec['mode'], 'entry': rec['entry'], 'outcome': outcome}
    run = rec.get('probe', {}).get('run', {})
    if not (outcome == 'solved' and run.get('ok')):
        per_run.append(row)
        continue
    env = run['envelope']
    ident = C.identity(env)
    E = C.Envelope(env)
    has_solution = any(k.startswith('result:disp:') for k in E.by_id)
    trusted = has_solution and (ident.get('numerical_quality_status') == 'checks_passed'
                                or ident.get('standing') == C.TRUSTED_STANDING)
    row.update(quality=ident.get('numerical_quality_status'), standing=ident.get('standing'), trusted=bool(trusted))
    breaches = []
    if has_solution:
        for (k, exp_s, cls, scale_s, gov) in c['rows']:
            obs, note = E.observe(k, c['meta'])
            if obs is None:
                continue
            f = C.case_unit_factor(cls, c['meta']['units'])
            ok = all(C.compare_value(x / f, C.dfr(exp_s), C.dfr(scale_s))[1] for x in obs)
            if not ok:
                breaches.append(k)
    row['breaches'] = len(breaches)
    if trusted:
        trusted_runs += 1
        for k in breaches:
            actual.add((rec['entry'], rec['case'], k))
    per_run.append(row)
violations = sorted(actual - pinned)
fixed = sorted(pinned - actual)
out = {'runs_evaluated': runs_seen, 'trusted_runs': trusted_runs,
       'trusted_breach_triples': len(actual),
       'by_entry': {e: sum(1 for t in actual if t[0] == e) for e in ('captured', 'typed')},
       'violations_outside_pinned_exceptions': [list(t) for t in violations],
       'pinned_triples_not_breached': len(fixed),
       'gate': 'PASS' if not violations else 'FAIL',
       'runs': per_run}
json.dump(out, open(out_path, 'w'), indent=1)
print(json.dumps({k: v for k, v in out.items() if k != 'runs'}, indent=1)[:3000])
