"""T3 P1 detection: run driver (scratch; committed as run.py.txt).

Usage: python3 -B run.py <probe-binary> <gen_out-dir> <runs.jsonl> [case-id-prefix ...]

Each (case, mode) runs in a FRESH process of the probe, with RLIMIT_AS set in the child
before exec and a wall-clock timeout.  Wall time and peak RSS (ru_maxrss of that child,
from wait4) are recorded for every run.  Where the captured entry refuses at capture,
the historical typed entry is run as a secondary observation.  Before every run of a case
with >= 1000 members the driver waits while any cargo process is running (shared host).
"""
import json
import os
import resource
import signal
import subprocess
import sys
import time

MODES = ('sparse_interactive', 'dense_scrutiny')
RLIMIT_AS_BYTES = 6 * 1024 ** 3
TIMEOUT_SMALL = 600
TIMEOUT_LARGE = 1800


def cargo_busy():
    return subprocess.run(['pgrep', '-x', 'cargo'], capture_output=True).returncode == 0


def run_one(binary, req, mode, typed, timeout):
    def limit():
        resource.setrlimit(resource.RLIMIT_AS, (RLIMIT_AS_BYTES, RLIMIT_AS_BYTES))
        os.setsid()
    args = [binary, req, mode] + (['typed'] if typed else [])
    t0 = time.monotonic()
    p = subprocess.Popen(args, stdout=subprocess.PIPE, stderr=subprocess.PIPE, preexec_fn=limit)
    out_chunks, err_chunks = [], []
    import threading

    def pump(src, dst):
        for chunk in iter(lambda: src.read(1 << 16), b''):
            dst.append(chunk)
    th = [threading.Thread(target=pump, args=(p.stdout, out_chunks)), threading.Thread(target=pump, args=(p.stderr, err_chunks))]
    for t in th:
        t.start()
    timed_out = False
    while True:
        pid, status, ru = os.wait4(p.pid, os.WNOHANG)
        if pid:
            break
        if time.monotonic() - t0 > timeout:
            os.killpg(p.pid, signal.SIGKILL)
            timed_out = True
            pid, status, ru = os.wait4(p.pid, 0)
            break
        time.sleep(0.02)
    wall = time.monotonic() - t0
    for t in th:
        t.join()
    p.returncode = os.waitstatus_to_exitcode(status)
    out = b''.join(out_chunks).decode('utf-8', 'replace').strip()
    err = b''.join(err_chunks).decode('utf-8', 'replace')
    rec = {'wall_seconds': round(wall, 4), 'peak_rss_kib': ru.ru_maxrss, 'exit_code': p.returncode,
           'timed_out': timed_out, 'rlimit_as_bytes': RLIMIT_AS_BYTES, 'timeout_seconds': timeout,
           'stderr_tail': err[-1500:]}
    if out:
        try:
            rec['probe'] = json.loads(out.splitlines()[-1])
        except json.JSONDecodeError:
            rec['stdout_tail'] = out[-1500:]
    return rec


def main():
    binary, gen, runs_path = sys.argv[1:4]
    prefixes = sys.argv[4:]
    cases = json.load(open(os.path.join(gen, 'cases.json')))
    todo = [c for c in cases if 'request_file' in c and (not prefixes or any(c['id'].startswith(p) for p in prefixes))]
    todo.sort(key=lambda c: (c.get('members', 0) >= 1000, c.get('members', 0), c['id']))
    with open(runs_path, 'a') as fh:
        for c in todo:
            large = c.get('members', 0) >= 1000
            req = os.path.join(gen, c['request_file'])
            for mode in MODES:
                if large:
                    while cargo_busy():
                        time.sleep(30)
                rec = run_one(binary, req, mode, False, TIMEOUT_LARGE if large else TIMEOUT_SMALL)
                rec.update(case=c['id'], mode=mode, entry='captured')
                fh.write(json.dumps(rec) + '\n')
                fh.flush()
                run = rec.get('probe', {}).get('run', {})
                print(c['id'], mode, 'ok' if run.get('ok') else ('ERR ' + str(run.get('error', rec.get('exit_code')))[:80]),
                      rec['wall_seconds'], rec['peak_rss_kib'], flush=True)
                refused_at_capture = run and not run.get('ok') and 'CHECKED-JSON' in str(run.get('error', ''))
                if refused_at_capture:
                    rec2 = run_one(binary, req, mode, True, TIMEOUT_LARGE if large else TIMEOUT_SMALL)
                    rec2.update(case=c['id'], mode=mode, entry='typed_secondary')
                    fh.write(json.dumps(rec2) + '\n')
                    fh.flush()


if __name__ == '__main__':
    main()
