"""Per-run comparison of the full-envelope re-run with G1's first base run (both on e7d930d49).

Usage: python3 -B compare_with_first_run.py <gen_out> <first_dir> <full_dir> <out.json>

Per (case, mode, entry):
- summary envelope bytes: envelopes/<key>.json byte-equal (the exact span of run.envelope), and presence;
- classification: compare.py's classify() outcome and reason (P1's comparator, imported unchanged);
- ok, exit code, timed_out, stderr_tail (from runs.jsonl);
- error text (not-ok runs);
- the probe line as parsed, with the only expected differences removed (run.solve_seconds, and
  run.envelope_sha256 which only the variant emits);
- gate rows (outcome, quality, standing, trusted, breaches) from each gate_check result.
Also: full-envelope hash presence (exactly the ok runs), and file hash == run.envelope_sha256.
"""
import hashlib
import json
import os
import sys
from collections import Counter

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import compare as C  # noqa: E402

gen, first, full, out_path = sys.argv[1:5]
fam = {c['id']: c.get('family', '?') for c in json.load(open(os.path.join(gen, 'cases.json')))}


def runs(d):
    m = {}
    for line in open(os.path.join(d, 'runs.jsonl')):
        r = json.loads(line)
        k = (r['case'], r['mode'], r['entry'])
        assert k not in m, k
        m[k] = r
    return m


def rows(p):
    return {(r['case'], r['mode'], r['entry']): r for r in json.load(open(p))['runs']}


def stripped(r):
    p = json.loads(json.dumps(r.get('probe')))
    if p and 'run' in p:
        p['run'].pop('solve_seconds', None)
        p['run'].pop('envelope_sha256', None)
    return p


A, B = runs(first), runs(full)
GA = rows(os.path.join(first, 'result_part1_base_e7d930d49.json'))
GB = rows(os.path.join(full, 'result_part1_base_e7d930d49_full.json'))
counts, diffs = Counter(), []
full_hashes, full_checked = 0, 0
for k in sorted(set(A) | set(B)):
    if k not in A or k not in B:
        diffs.append({'key': list(k), 'presence': [k in A, k in B]})
        counts['presence'] += 1
        continue
    a, b = A[k], B[k]
    key = '%s__%s__%s' % k
    d = {}
    ea = os.path.join(first, 'envelopes', key + '.json')
    eb = os.path.join(full, 'envelopes', key + '.json')
    if os.path.exists(ea) != os.path.exists(eb):
        d['summary_envelope_presence'] = [os.path.exists(ea), os.path.exists(eb)]
    elif os.path.exists(ea) and open(ea, 'rb').read() != open(eb, 'rb').read():
        d['summary_envelope_bytes'] = 'differ'
    ca, cb = C.classify(a), C.classify(b)
    if ca != cb:
        d['classify'] = [list(ca), list(cb)]
    ra, rb = a.get('probe', {}).get('run', {}), b.get('probe', {}).get('run', {})
    for f, va, vb in (('ok', bool(ra.get('ok')), bool(rb.get('ok'))), ('exit_code', a.get('exit_code'), b.get('exit_code')),
                      ('timed_out', a.get('timed_out'), b.get('timed_out')), ('stderr_tail', a.get('stderr_tail'), b.get('stderr_tail')),
                      ('error', ra.get('error'), rb.get('error'))):
        if va != vb:
            d[f] = [va, vb]
    if stripped(a) != stripped(b):
        d['probe_line_other_than_solve_seconds_and_envelope_sha256'] = 'differ'
    ga, gb = GA.get(k), GB.get(k)
    if (ga is None) != (gb is None):
        d['gate_row_presence'] = [ga is not None, gb is not None]
    elif ga is not None:
        for f in ('outcome', 'quality', 'standing', 'trusted', 'breaches'):
            if ga.get(f) != gb.get(f):
                d['gate_' + f] = [ga.get(f), gb.get(f)]
    has = 'envelope_sha256' in rb
    if has != bool(rb.get('ok')):
        d['full_hash_presence_vs_ok'] = [has, bool(rb.get('ok'))]
    if has:
        full_hashes += 1
        fp = os.path.join(full, 'full', key + '.json')
        if os.path.exists(fp):
            h = hashlib.sha256()
            with open(fp, 'rb') as fh:
                for chunk in iter(lambda: fh.read(1 << 20), b''):
                    h.update(chunk)
            if h.hexdigest() != rb['envelope_sha256']:
                d['full_file_hash'] = [h.hexdigest(), rb['envelope_sha256']]
            else:
                full_checked += 1
        else:
            d['full_file_missing'] = True
    elif os.path.exists(os.path.join(full, 'full', key + '.json')):
        d['full_file_without_hash'] = True
    for f in d:
        counts[f] += 1
    if d:
        d.update(key=list(k), family=fam[k[0]])
        diffs.append(d)
out = {'orientation': 'each difference is [first run 8525c06d..., full-envelope re-run]',
       'runs_first': len(A), 'runs_full': len(B), 'gate_rows_first': len(GA), 'gate_rows_full': len(GB),
       'ok_first': sum(bool(r.get('probe', {}).get('run', {}).get('ok')) for r in A.values()),
       'ok_full': sum(bool(r.get('probe', {}).get('run', {}).get('ok')) for r in B.values()),
       'full_envelope_hashes': full_hashes, 'full_envelope_files_matching_hash': full_checked,
       'difference_counts': dict(counts), 'runs_with_any_difference': len(diffs), 'differences': diffs}
json.dump(out, open(out_path, 'w'), indent=1)
print(json.dumps({k: v for k, v in out.items() if k != 'differences'}, indent=1))
for d in diffs[:20]:
    print(d)
