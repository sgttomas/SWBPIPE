"""Gate part 1 on macOS with per-run envelope capture (scratch; G1 gate-baseline TASK, 2026-09-28).

Derived from the calibration driver gate_run_mac_parallel.py (sha256 5a2029a6...). Unchanged from it:
the case set, the 4 part-2 dense timeouts excluded, the modes, the entries, the timeouts
(run.py's TIMEOUT_SMALL / TIMEOUT_LARGE), the ordering key (descending members), run.py's run_one
imported unchanged with resource.setrlimit a no-op (the probe caps its own heap at 6 GiB), and the
fields added to each record (case, mode, entry, rlimit_as_enforced, heap_cap_bytes,
memorystatus_level_end). Records are appended to runs.jsonl as they finish.

Added, none of which changes what run_one returns:
1. Envelope capture. run.py's module-level name `subprocess` is rebound, for run.py only, to a copy
   of the subprocess module whose Popen wraps the child's stdout and stderr pipes in a pass-through
   reader: every read returns the pipe's bytes unchanged and also keeps them. After run_one returns
   (its pump threads are joined), the kept bytes are written verbatim to stdout/<key>.out and
   stderr/<key>.err, and, when the probe printed an envelope, the envelope's exact byte span of
   that stdout line to envelopes/<key>.json. The driver checks per run that json.loads of the kept
   stdout's last line equals the record's `probe`.
2. Worker rule (ROOT, Q13 / G1 brief). Phase A: the runs of cases with >= 10,000 members, at most
   2 at a time, with nothing else running. Phase B: every other part-1 run, at most `workers` (<= 4).
3. Host sampling every 5 s to host_samples.tsv (UTC, load averages, kern.memorystatus_level,
   running t3_p1_probe / cargo / rustc process counts) and per-run start/end times to schedule.tsv.

Usage: python3 -B gate_run_base_envelopes.py <probe-binary> <gen_out-dir> <out-dir> <workers> [case-id ...]
(optional case ids restrict the run, for a smoke test only)
"""
import hashlib
import json
import os
import subprocess
import sys
import threading
import time
import types
from concurrent.futures import ThreadPoolExecutor, as_completed

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import run as R  # noqa: E402

R.resource.setrlimit = lambda *a, **k: None  # as the calibration driver: macOS cannot set RLIMIT_AS

_tls = threading.local()


class _Tee:
    """Pass-through reader: returns the wrapped pipe's bytes unchanged and keeps a copy."""

    def __init__(self, raw):
        self._raw = raw
        self.kept = []

    def read(self, n=-1):
        b = self._raw.read(n)
        if b:
            self.kept.append(b)
        return b

    def close(self):
        return self._raw.close()

    def __getattr__(self, name):
        return getattr(self._raw, name)


class _TeePopen(subprocess.Popen):
    def __init__(self, *a, **k):
        super().__init__(*a, **k)
        if self.stdout is not None:
            self.stdout = _Tee(self.stdout)
        if self.stderr is not None:
            self.stderr = _Tee(self.stderr)
        _tls.proc = self


_sp = types.ModuleType('subprocess')
_sp.__dict__.update(subprocess.__dict__)
_sp.Popen = _TeePopen
R.subprocess = _sp  # run.py's own reference only; the global subprocess module is untouched

binary, gen, outdir, workers = sys.argv[1], sys.argv[2], sys.argv[3], int(sys.argv[4])
only = set(sys.argv[5:])
assert 1 <= workers <= 4, 'at most 4 workers'
TIMEOUTS = {('RF-LARGE-CHAIN-n01000-ROT', 'dense_scrutiny'), ('RF-LARGE-TREE-n01000-AX', 'dense_scrutiny')}
HUGE = 10000
cases = [c for c in json.load(open(os.path.join(gen, 'cases.json'))) if 'request_file' in c]
jobs = []
for c in cases:
    if only and c['id'] not in only:
        continue
    large = c.get('members', 0) >= 1000
    for mode in ('sparse_interactive', 'dense_scrutiny'):
        if (c['id'], mode) in TIMEOUTS:
            continue
        for entry, typed in (('captured', False), ('typed', True)):
            jobs.append((c, mode, entry, typed, R.TIMEOUT_LARGE if large else R.TIMEOUT_SMALL))
jobs.sort(key=lambda j: -j[0].get('members', 0))
phase_a = [j for j in jobs if j[0].get('members', 0) >= HUGE]
phase_b = [j for j in jobs if j[0].get('members', 0) < HUGE]

for d in ('stdout', 'stderr', 'envelopes'):
    os.makedirs(os.path.join(outdir, d), exist_ok=True)
runs_path = os.path.join(outdir, 'runs.jsonl')
assert not os.path.exists(runs_path), 'refusing to overwrite ' + runs_path


def utc():
    return time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())


def memlevel():
    return int(os.popen('sysctl -n kern.memorystatus_level').read().strip() or -1)


def envelope_span(text, parsed):
    """Exact span of run.envelope in the probe's compact, key-sorted serde_json line."""
    run = parsed.get('run', {})
    if 'envelope' not in run:
        return None
    prefix = '{"mode":' + json.dumps(parsed['mode']) + ',"run":{"entry":' + json.dumps(run['entry']) + ',"envelope":'
    if not text.startswith(prefix):
        raise ValueError('unexpected probe line layout')
    env, end = json.JSONDecoder().raw_decode(text, len(prefix))
    if env != run['envelope']:
        raise ValueError('envelope span does not parse to the recorded envelope')
    return text[len(prefix):end]


def go(j, phase):
    c, mode, entry, typed, to = j
    key = '%s__%s__%s' % (c['id'], mode, entry)
    t_start = utc()
    _tls.proc = None
    rec = R.run_one(binary, os.path.join(gen, c['request_file']), mode, typed, to)
    proc = _tls.proc
    t_end = utc()
    rec.update(case=c['id'], mode=mode, entry=entry, rlimit_as_enforced=False, heap_cap_bytes=6 * 1024 ** 3,
               memorystatus_level_end=memlevel())
    out_b = b''.join(proc.stdout.kept)
    err_b = b''.join(proc.stderr.kept)
    with open(os.path.join(outdir, 'stdout', key + '.out'), 'wb') as fh:
        fh.write(out_b)
    with open(os.path.join(outdir, 'stderr', key + '.err'), 'wb') as fh:
        fh.write(err_b)
    check = 'no_probe_line'
    env_sha = '-'
    text = out_b.decode('utf-8', 'replace').strip()
    if text:
        last = text.splitlines()[-1]
        try:
            parsed = json.loads(last)
        except json.JSONDecodeError:
            parsed = None
        if parsed is not None:
            if parsed != rec.get('probe'):
                raise RuntimeError('tee mismatch for ' + key)
            check = 'probe_equal'
            span = envelope_span(out_b.decode('utf-8').strip().splitlines()[-1], parsed)
            if span is not None:
                sb = span.encode('utf-8')
                with open(os.path.join(outdir, 'envelopes', key + '.json'), 'wb') as fh:
                    fh.write(sb)
                env_sha = hashlib.sha256(sb).hexdigest()
    meta = dict(key=key, phase=phase, start=t_start, end=t_end, wall=rec['wall_seconds'],
                stdout_sha256=hashlib.sha256(out_b).hexdigest(), stdout_bytes=len(out_b),
                stderr_bytes=len(err_b), stderr_within_tail=len(err_b.decode('utf-8', 'replace')) <= 1500,
                envelope_sha256=env_sha, tee_check=check)
    return rec, meta


stop = threading.Event()


def sampler():
    with open(os.path.join(outdir, 'host_samples.tsv'), 'w') as fh:
        fh.write('utc\tload1\tload5\tload15\tmemorystatus_level\tprobes\tcargo\trustc\n')
        while not stop.is_set():
            l1, l5, l15 = os.getloadavg()
            n = [len(subprocess.run(['pgrep', '-x', p], capture_output=True, text=True).stdout.split())
                 for p in ('t3_p1_probe', 'cargo', 'rustc')]
            fh.write('%s\t%.2f\t%.2f\t%.2f\t%d\t%d\t%d\t%d\n' % (utc(), l1, l5, l15, memlevel(), *n))
            fh.flush()
            stop.wait(5)


th = threading.Thread(target=sampler, daemon=True)
th.start()
t0 = time.time()
total = len(jobs)
n = 0
print(utc(), 'start', total, 'runs; phase A', len(phase_a), '(2 workers); phase B', len(phase_b), '(%d workers)' % workers, flush=True)
with open(runs_path, 'w') as fh, open(os.path.join(outdir, 'schedule.tsv'), 'w') as sch:
    sch.write('key\tphase\tstart_utc\tend_utc\twall_seconds\tstdout_sha256\tstdout_bytes\tstderr_bytes\tstderr_within_1500\tenvelope_sha256\ttee_check\n')
    for phase, pj, w in (('A', phase_a, 2), ('B', phase_b, workers)):
        if not pj:
            continue
        print(utc(), 'phase', phase, len(pj), 'runs,', w, 'workers', flush=True)
        with ThreadPoolExecutor(w) as ex:
            futs = [ex.submit(go, j, phase) for j in pj]
            for f in as_completed(futs):
                rec, meta = f.result()
                n += 1
                fh.write(json.dumps(rec) + '\n')
                fh.flush()
                sch.write('\t'.join(str(meta[k]) for k in ('key', 'phase', 'start', 'end', 'wall', 'stdout_sha256', 'stdout_bytes',
                                                           'stderr_bytes', 'stderr_within_tail', 'envelope_sha256', 'tee_check')) + '\n')
                sch.flush()
                run = rec.get('probe', {}).get('run', {})
                if n % 50 == 0 or rec['wall_seconds'] > 60 or phase == 'A':
                    print(utc(), n, '/', total, rec['case'], rec['mode'], rec['entry'], 'ok' if run.get('ok') else 'not-ok',
                          rec['wall_seconds'], 'exit', rec['exit_code'], 'mem%', rec['memorystatus_level_end'],
                          'load1 %.2f' % os.getloadavg()[0], flush=True)
stop.set()
th.join()
print(utc(), 'done', n, 'runs in', round(time.time() - t0), 's', flush=True)
