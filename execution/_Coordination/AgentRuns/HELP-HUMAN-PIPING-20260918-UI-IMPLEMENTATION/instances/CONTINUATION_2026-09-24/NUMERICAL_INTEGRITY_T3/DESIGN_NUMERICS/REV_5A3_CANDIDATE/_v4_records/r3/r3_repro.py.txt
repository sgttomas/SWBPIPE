"""V4 delta check at R3: R3's specification in V4's emulator (W residual over the contributions at q,
on the final state; best-state hybrid gate). (1) LEVER2 at 2^90, 2^100, 2^110; (2) the 22 controls
and K4's 20 probe cases; (3) SEEDED-COMMON."""
import os, sys, json, time, math
HERE = os.path.dirname(os.path.abspath(__file__)); ROOT = os.path.dirname(HERE)
sys.path.insert(0, ROOT); sys.path.insert(0, os.path.join(ROOT, "delta_r2"))
from fractions import Fraction as Fr
import v4emu_r2 as R2
from v4emu import load, flog2, model, member, fix, spring, station
import r2_lever2 as LV
R2.W_K = "contributions"; R2.GATE_STATE = "best"
def main():
    out = {"lever2": {}, "controls": {}, "probe": {}, "seeded": {}}
    def lever(k, s):
        m = LV.lever2(k, s); ok, K = LV.ldlt_dyadic(m); kt = 2.0 ** (flog2(K[(1, 1)]) - 580)
        m["springs"].append((1, kt)); st = R2.solve_r2(m, 512, "hybrid"); E, _ = R2.E_r2(m, st)
        efo = float(R2.ehat(E, 0, R2.extents(m)[0])[0]); P = 2.0 ** (math.floor(math.log2(efo)) - 438)
        m = LV.lever2(k, s, Pexp=P); m["springs"].append((1, kt)); load(m, 4, 2, P); return m
    for (k, s) in [(90, 40), (100, 40), (110, 20)]:
        m = lever(k, s); t = time.time(); r = R2.run_r2(m)
        out["lever2"][m["name"]] = r
        print("LEVER2", m["name"], r["selected"], r["attempts"], r.get("honesty"), "(%.1fs)" % (time.time() - t), flush=True)
    from r2_controls import ms as controls
    for m in controls:
        t = time.time(); r = R2.run_r2(m)
        out["controls"][m["name"]] = dict(selected=r["selected"], classes=r.get("classes"), honesty=r.get("honesty"), g5a=r.get("g5a_failures"), last=r["attempts"][-1] if r["attempts"] else None)
        print("control", m["name"], r["selected"], r.get("classes"), r.get("honesty"), r.get("g5a_failures"), r["attempts"][-1] if r["attempts"] else None, "(%.1fs)" % (time.time() - t), flush=True)
    from gate_probe_nsec import S as NSEC
    def run_345(y, members, tip):
        m = model("PROBE-y%s-%dm-%s" % ("".join("%g" % c for c in y), members, tip[0]), [(3.0*s_, 4.0*s_, 0.0) for s_ in range(members+1)])
        for s_ in range(members): member(m, s_, s_+1, y=y, **NSEC)
        fix(m, 0, range(6))
        for c, v in enumerate(tip[1]):
            if v: load(m, members, c, v)
        return m
    loads = [("general", [1000.0, -500.0, 2000.0, 100.0, 200.0, 300.0]), ("inplane", [-400.0, 300.0, 0, 0, 0, 0]),
             ("outofplane", [0, 0, 1000.0, 0, 0, 0]), ("axial", [600.0, 800.0, 0, 0, 0, 0]), ("torque", [0, 0, 0, 60.0, 80.0, 0])]
    for y in ((3.0,4.0,5.0), (0.0,0.0,1.0)):
        for members in (1, 3):
            for L in loads:
                m = run_345(y, members, L); t = time.time()
                st = None
                r = R2.run_r2(m)
                try:
                    st = R2.solve_r2(m, 128, "hybrid", estimate=False)["info"]
                except Exception as e:
                    st = str(e)
                out["probe"][m["name"]] = dict(selected=r["selected"], honesty=r.get("honesty"), g5a=r.get("g5a_failures"), gate128=st)
                print("probe", m["name"], r["selected"], r.get("honesty"), r.get("g5a_failures"), st, "(%.1fs)" % (time.time() - t), flush=True)
    def n05():
        m = model("SEEDED-COMMON", [(0,0,0),(2,0,0)]); member(m, 0, 1, **NSEC)
        fix(m, 0, (0,1,2,4,5)); spring(m, 0, 3, 1e-4); load(m, 1, 3, 1e-8); station(m, 0, 0.5); return m
    R2.SEED["SEEDED-COMMON"] = [(7, Fr(1, 2**110))]
    r = R2.run_r2(n05()); out["seeded"] = r
    print("SEEDED-COMMON under R3:", r["selected"], r["attempts"], r.get("honesty"), flush=True)
    json.dump(out, open(os.path.join(HERE, "r3_repro.json"), "w"), indent=1, default=str)

if __name__ == "__main__":
    main()
