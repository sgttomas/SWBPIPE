"""V4: a runtime bound for R3's unmeasured term e_q (derived):
   |e_q| <= 62.5 * 2^-qW * Gamma_q,  Gamma_q = |a_q| |K^-1| A-bar |u|
         <= 62.5 * 2^-qW * ||a_q S||_1 * ||K~^-1||_1 * ||S A-bar |u| ||_inf   (K~ = S K S, symmetric)
Every factor is formed by the design already (S: radix equilibration; ||K~^-1||_1: the Hager-Higham
estimate, here exact; A-bar |u|: E's reaction pass; a_q: the recovery row). Reports, per model and
verification precision P = 2p, the worst ratio of that charge to e_q's allowance 124 * 2^-2p * e-hat
(p = 128, 256) or to 2^-22 * b (p = 512, b >= 2^-502 e-hat), for qW = 2p + 64 (R3) and 3p + 64 (the option)."""
import os, sys, json, time, math
HERE = os.path.dirname(os.path.abspath(__file__)); ROOT = os.path.dirname(HERE)
sys.path.insert(0, ROOT); sys.path.insert(0, os.path.join(ROOT, "delta_r2"))
from fractions import Fraction as Fr
import v4emu_r2 as R2
from v4emu import Ar, recover, layout, bodies, extents, row_body, flog2, exact_inverse, Stop, load
R2.W_K = "contributions"; R2.GATE_STATE = "best"

def charge(m, P):
    st = R2.solve_r2(m, P, "hybrid", estimate=False)
    p = P // 2
    K, n, u, ops = st["K"], st["n"], st["u"], st["ops"]
    free = [g for g in range(n) if g not in m["cons"]]
    if not free:
        return {}, 0.0
    s = {g: Fr(2) ** (-(flog2(K[(g, g)]) // 2)) for g in free}
    Kt = [[s[a] * K.get((a, b), Fr(0)) * s[b] for b in free] for a in free]
    Ki = exact_inverse(Kt)
    inv1 = max(sum(abs(Ki[i][j]) for i in range(len(free))) for j in range(len(free)))
    A = R2.abar_r2(Ar(P), m, ops)
    SAu = max(s[i] * sum(A.get((i, j), 0) * abs(u[j]) for j in range(n)) for i in free)
    # recovery rows a_q over the free DOFs (exact linear recovery with the P-formed operators, no ledger)
    m0 = dict(m); m0["loads"] = {}
    cols = {}
    for g in free:
        e = [Fr(0)] * n; e[g] = Fr(1)
        cols[g], _ = recover(Ar(None), m0, ops, K, n, e)
    E, _ = R2.E_r2(m, st)
    lay = layout(m); nb = bodies(m); ext = extents(m)
    worst = {}
    for idx, (key, kind, inp) in enumerate(lay):
        if kind not in ("fo", "mo"): continue
        b = row_body(m, key, nb)
        eh = R2.ehat(E, b, ext[b])[0 if kind == "fo" else 1]
        if eh == 0: continue
        aS = sum(abs(cols[g][idx]) * s[g] for g in free)
        N = aS * SAu
        for label, qW in (("2p+64", min(2 * p + 64, 1024)), ("3p+64", min(3 * p + 64, 1024))):
            C = Fr(125, 2) * Fr(2) ** -qW * inv1 * N
            allow = 124 * Fr(2) ** (-2 * p) * eh if p < 512 else Fr(2) ** -524 * eh
            r = float(C / allow)
            worst[label] = max(worst.get(label, 0.0), r)
    return worst, float(inv1)

if __name__ == "__main__":
    from r2_controls import ms as controls
    from gate_probe_nsec import S as NSEC
    from v4emu import model, member, fix
    import r2_lever2 as LV
    def lever(k, s_):
        m = LV.lever2(k, s_); ok, K = LV.ldlt_dyadic(m); kt = 2.0 ** (flog2(K[(1, 1)]) - 580)
        m["springs"].append((1, kt)); st = R2.solve_r2(m, 512, "hybrid"); E, _ = R2.E_r2(m, st)
        efo = float(R2.ehat(E, 0, R2.extents(m)[0])[0]); Pl = 2.0 ** (math.floor(math.log2(efo)) - 438)
        m = LV.lever2(k, s_, Pexp=Pl); m["springs"].append((1, kt)); load(m, 4, 2, Pl); return m
    def run_345(y, members, tip):
        m = model("PROBE-y%s-%dm-%s" % ("".join("%g" % c for c in y), members, tip[0]), [(3.0*i, 4.0*i, 0.0) for i in range(members+1)])
        for i in range(members): member(m, i, i+1, y=y, **NSEC)
        fix(m, 0, range(6))
        for c, v in enumerate(tip[1]):
            if v: load(m, members, c, v)
        return m
    loads = [("general", [1000.0, -500.0, 2000.0, 100.0, 200.0, 300.0]), ("inplane", [-400.0, 300.0, 0, 0, 0, 0]),
             ("outofplane", [0, 0, 1000.0, 0, 0, 0]), ("axial", [600.0, 800.0, 0, 0, 0, 0]), ("torque", [0, 0, 0, 60.0, 80.0, 0])]
    models = list(controls) + [run_345(y, mm, L) for y in ((3.0,4.0,5.0),) for mm in (1, 3) for L in loads] + [lever(90, 40)]
    out = {}
    agg = {"2p+64": 0.0, "3p+64": 0.0}
    for m in models:
        t = time.time(); rec = {}
        for P in (256, 512, 1024):
            try:
                w, inv1 = charge(m, P)
            except Stop as e:
                rec[P] = "stop:%s" % e; continue
            rec[P] = dict(w, inv1_log2=math.log2(inv1) if inv1 else None)
            if not m["name"].startswith("LEVER"):
                for k_, v in w.items(): agg[k_] = max(agg[k_], v)
        out[m["name"]] = rec
        print(m["name"], {P: (v if isinstance(v, str) else {k_: ("%.3g" % x if x is not None else None) for k_, x in v.items()}) for P, v in rec.items()}, "(%.1fs)" % (time.time() - t), flush=True)
    print("worst charge/allowance over non-LEVER models:", agg)
    json.dump(dict(per_model=out, worst_non_lever=agg), open(os.path.join(HERE, "r3_charge.json"), "w"), indent=1, default=str)
