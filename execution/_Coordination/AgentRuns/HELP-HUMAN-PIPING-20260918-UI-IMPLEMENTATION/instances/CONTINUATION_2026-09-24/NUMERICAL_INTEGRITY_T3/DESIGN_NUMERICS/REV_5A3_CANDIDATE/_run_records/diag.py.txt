#!/usr/bin/env python3
"""Per-kind noise diagnostics: for each attempt pair (p, 2p) and each body and
kind, the largest |q_p - q_2p| in units of 2^-p * E(body, kind) (uncoupled) and
of 2^-p * Ecoupled(body, kind) (coupled through L_b as S* is)."""
import sys, math
from fractions import Fraction as Fr
from emu import Case, solve_at, body_E, body_kind_max, Stop, KINDS
from models import models

names = set(sys.argv[1:])
for m in models():
    if names and m["name"] not in names:
        continue
    case = Case(m)
    st = {}
    for P in (128, 256, 512, 1024):
        try:
            st[P] = solve_at(case, P)
        except Stop as s:
            st[P] = s
    print(m["name"])
    for p in (128, 256, 512):
        a, b = st[p], st[2 * p]
        if isinstance(a, Stop) or isinstance(b, Stop):
            print("  p", p, "stop", a if isinstance(a, Stop) else "", b if isinstance(b, Stop) else "")
            continue
        Eb = body_E(case, b["E"])
        s2 = body_kind_max(case, b["values"])
        worst = {}
        for (rid, kind, body, inp), qp, q2 in zip(case.layout, a["values"], b["values"]):
            if inp:
                continue
            d = abs(qp - q2)
            key = (body, kind)
            if d > worst.get(key, (Fr(0), None))[0]:
                worst[key] = (d, rid)
        for (body, kind), (d, rid) in sorted(worst.items()):
            if kind not in ("fo", "mo") or d == 0:
                continue
            L = case.extent[body]
            E = Eb[body][kind]
            Ec = max(E, Eb[body]["mo"] / L) if kind == "fo" else max(E, L * Eb[body]["fo"])
            S = float(s2[body][kind])
            r1 = float(d) / (2.0 ** -p * E) if E else float("inf")
            r2 = float(d) / (2.0 ** -p * Ec) if Ec else float("inf")
            print("  p %4d body %d %s  |dq|/(2^-p E)=%9.3g  |dq|/(2^-p Ec)=%9.3g  S*2p/E=2^%.1f  row %s" % (
                p, body, kind, r1, r2, math.log2(S / E) if S and E else float("-inf"), rid))
