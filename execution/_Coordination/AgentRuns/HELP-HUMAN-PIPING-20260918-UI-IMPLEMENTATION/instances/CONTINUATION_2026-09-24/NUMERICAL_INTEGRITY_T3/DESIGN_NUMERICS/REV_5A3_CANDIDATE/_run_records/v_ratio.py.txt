#!/usr/bin/env python3
"""For each control and each compared pair (p, 2p): the largest
log2(V_q / (2^-64 M_q)) over force and moment rows, V_q = 2^(8-2p) E^, and
the smallest margin log2((2^-64 M_q - |dq|) / V_q) over rows where the pair
agrees; they show how far V is from changing any decision."""
import math, json
from fractions import Fraction as Fr
from emu import Case, solve_at, body_E, body_kind_max, coupled_wide, Stop, KINDS
from models import models

def flog2(x):
    x = Fr(x)
    return math.log2(x.numerator) - math.log2(x.denominator)


out = {}
for m in models():
    case = Case(m)
    st = {}
    for P in (128, 256, 512, 1024):
        try:
            st[P] = solve_at(case, P)
        except Stop as s:
            st[P] = s.what
    rec = {}
    for p in (128, 256, 512):
        a, b = st[p], st[2 * p]
        if isinstance(a, str) or isinstance(b, str):
            continue
        Eb = body_E(case, b["E"])
        s2 = body_kind_max(case, b["values"])
        sc = [coupled_wide(s2[x], case.extent[x], 2 * p) for x in range(case.nbody)]
        worst = -1e9
        for (rid, kind, body, inp), qa, qb in zip(case.layout, a["values"], b["values"]):
            if inp or kind not in ("fo", "mo") or Eb[body][kind] == 0:
                continue
            M = max(abs(qb), sc[body][kind])
            V = Fr(2) ** (8 - 2 * p) * Fr(Eb[body][kind])
            if M == 0:
                worst = float("inf")
                continue
            worst = max(worst, flog2(V) - flog2(Fr(1, 2 ** 64) * M))
        rec[p] = round(worst, 1) if worst != -1e9 else None
    out[m["name"]] = rec
for k, v in out.items():
    print("%-22s %s" % (k, v))
json.dump(out, open("v_ratio.stdout.json", "w"), indent=1)
