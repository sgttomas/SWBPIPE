#!/usr/bin/env python3
"""V4, delta check at R6, check 4: ||K~_P^-1||_1 for R1's RF-LARGE references at 1,000
members (4,500 to 6,000 free DOFs), which R6 did not compute, and the looseness of R6's
shift bound S there. K~ is formed as in r5_large (emu5's assembly at P = 256, radix
equilibration, K4's order); the norm is the largest column 1-norm of K~^-1, from one
binary64 solve per column on the binary64 LDL^T factor. binary64 is an approximation
(the solves' relative error grows with the condition number, here up to about 2^50). S is DS1's recorded log2 S at the same state
(`_run_records_r6/large6_1000.json`, R6 run, the P = 256 verification).
Usage: python3 r6_exact1000.py [case ids ...] (default: all six)
Output: r6_exact1000.json."""
import json, math, os, sys, time
HERE = os.path.dirname(os.path.abspath(__file__))
V4 = os.path.dirname(HERE)
sys.path.insert(0, os.path.join(V4, "r5"))
sys.dont_write_bytecode = True
from r5_large import r1_adapter, Case, setup, ldl, T3

RR6 = os.path.join(T3, "DESIGN_NUMERICS", "REV_5A3_CANDIDATE", "_run_records_r6", "large6_1000.json")


def norm_inv(Lr, D, n):
    best = 0.0
    for jc in range(n):
        x = [0.0] * n
        x[jc] = 1.0
        for i in range(jc + 1, n):
            s = 0.0
            for j, l in Lr[i]:
                v = x[j]
                if v:
                    s += l * v
            x[i] = -s if s else 0.0
        for i in range(n):
            x[i] /= D[i]
        for i in range(n - 1, -1, -1):
            v = x[i]
            if v:
                for j, l in Lr[i]:
                    x[j] -= l * v
        s1 = sum(abs(v) for v in x)
        if s1 > best:
            best = s1
    return best


def ds1_S(cid):
    for r in json.load(open(RR6)):
        if r["case"] == cid:
            runs = r["runs"]
            b = runs["R6"]["bounds"]
            key = "256" if "256" in b else sorted(b)[0]
            blk = b[key]["blocks"][0]
            return key, blk.get("log2_S"), blk.get("log2_B"), blk.get("log2_est_c")
    return None


if __name__ == "__main__":
    cases = r1_adapter.load_refs(T3)
    ids = sys.argv[1:] or [k for k, c in cases.items() if c["family"] == "RF-LARGE" and "generator" not in c["model"]
                           and len(c["model"]["members"]) == 1000]
    out = []
    for cid in ids:
        t0 = time.time()
        m, index = r1_adapter.adapt(cid, cases[cid], cases[cid]["model"])
        case = Case(m)
        rows, first, n = setup(case, 256, True)
        L, D = ldl(rows, first)
        Lr = [sorted(L[i].items()) for i in range(n)]
        # x[i] = e_jc[i] - sum_j l_ij x[j]: rows i > jc start from 0 (unit lower L)
        ex = norm_inv(Lr, D, n)
        rec = dict(case=cid, free=n, log2_exact_binary64=math.log2(ex), seconds=round(time.time() - t0, 1))
        d = ds1_S(cid)
        if d:
            rec.update(ds1_state=d[0], ds1_log2_S=d[1], ds1_log2_B=d[2], ds1_log2_est=d[3])
            if d[2] is not None:
                rec["log2_B_over_exact"] = d[2] - rec["log2_exact_binary64"]
            if d[3] is not None:
                rec["log2_exact_over_est"] = rec["log2_exact_binary64"] - d[3]
        out.append(rec)
        print(json.dumps(rec), flush=True)
    json.dump(out, open(os.path.join(HERE, "r6_exact1000.json"), "w"), indent=1)
