#!/usr/bin/env python3
"""R6: emu6's regression against emu5. With ORDER = "natural", emu6 run with R5's
switches (BOUND "Uc", SCOPE "body", option r5) and with R4's (BOUND "Fest", SCOPE "all",
option r4) must reproduce run_controls5.stdout.json's R5 and R4 records exactly:
selected precision, attempts, class counts, honesty, worst ratio, and (R5) the charge
and theta at every verification. Everything emu6 changed (sparse row loops, per-block
passes, the new ingredients function) is exercised; the R6 bound is not.
Output: reproduce6.stdout.json."""
import json, time
import emu6, models5
from emu6 import Case, schedule, classify, honesty, solve_at, charge_report, Stop

emu6.ORDER = "natural"
MODES = {"R5": ("r5", dict(BOUND="Uc", SCOPE="body")), "R4": ("r4", dict(BOUND="Fest", SCOPE="all"))}
ref5 = {rec["model"]: rec for rec in json.load(open("run_controls5.stdout.json"))}
out = []
mism = 0
for m in models5.models():
    t0 = time.time()
    case = Case(m)
    emu6.SEED.clear()
    emu6.GATE = "absolute"
    ref = solve_at(case, 2048, ref=True, estimate=False)
    emu6.SEED.update(models5.SEEDS)
    rec = dict(model=m["name"], diffs=[])
    for tag, (opt, sw) in MODES.items():
        saved = {k: getattr(emu6, k) for k in sw}
        for k, v in sw.items():
            setattr(emu6, k, v)
        emu6.GATE = "hybrid"
        cache = {}
        try:
            r = schedule(case, opt, cache)
            o = dict(selected=r["selected"], attempts=[list(a) for a in r["attempts"]], solved=r["solved"])
            if r["selected"] is not None:
                pub, cl, sc, g5 = classify(case, r, opt)
                w, c = honesty(case, pub, cl, ref)
                o.update(counts=c, worst=w[0], worst_row=str(w[1]), honest=w[0] <= 1.0,
                         g5a_fail=[str(x) for x in g5], w_over_v=r["info"]["w_over_v"])
            if tag == "R5":
                o["charge"] = {}
                for p in (128, 256, 512):
                    try:
                        cand = cache[p] if p in cache else solve_at(case, p)
                        ver = cache[2 * p] if 2 * p in cache else solve_at(case, 2 * p)
                        if isinstance(cand, Stop) or isinstance(ver, Stop):
                            o["charge"][str(p)] = "stop"
                            continue
                        ch, th = charge_report(case, cand, ver, p)
                        o["charge"][str(p)] = dict(charge=ch, theta=th)
                    except Stop as e:
                        o["charge"][str(p)] = "stop:%s" % e.what
        finally:
            for k, v in saved.items():
                setattr(emu6, k, v)
            emu6.GATE = "absolute"
        old = ref5[m["name"]]["runs"][tag]
        for key in ("selected", "attempts", "counts", "honest", "worst", "g5a_fail", "w_over_v"):
            if o.get(key) != old.get(key):
                rec["diffs"].append((tag, key, o.get(key), old.get(key)))
        if tag == "R5":
            for p_, v in o["charge"].items():
                ov = old["charge"].get(p_)
                if isinstance(v, dict) and isinstance(ov, dict):
                    if v["charge"] != ov["charge"] or v["theta"] != ov["theta"]:
                        rec["diffs"].append((tag, "charge" + p_, v, ov))
                elif v != ov:
                    rec["diffs"].append((tag, "charge" + p_, v, ov))
    mism += bool(rec["diffs"])
    rec["seconds"] = round(time.time() - t0, 2)
    out.append(rec)
    print("%-22s %s %.2fs" % (m["name"], "same" if not rec["diffs"] else "DIFF %s" % rec["diffs"][:2], rec["seconds"]), flush=True)
print("models with any difference:", mism, "of", len(out))
json.dump(dict(models=out, mismatched=mism), open("reproduce6.stdout.json", "w"), indent=1, default=str)
