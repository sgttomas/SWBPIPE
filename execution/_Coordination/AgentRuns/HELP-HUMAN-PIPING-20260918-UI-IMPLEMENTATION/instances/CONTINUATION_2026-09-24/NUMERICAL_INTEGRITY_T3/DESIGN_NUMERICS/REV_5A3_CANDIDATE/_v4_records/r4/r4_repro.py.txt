#!/usr/bin/env python3
"""V4, delta check at R4, check 1: re-execute DS1's emulator emu4 (read-only import,
R4's defaults) on the controls that carry V4-T1's closure, and compare with DS1's
recorded run_controls4 / mutants4 outcomes: CHARGE-SLENDER (R4, and mutants M17 no
charge, M19 F = 1), LEVER2 and TILT-LEVER at three gains, N05, SEEDED-COMMON.
Output: r4_repro.json."""
import json, os, sys
HERE = os.path.dirname(os.path.abspath(__file__))
DS1 = os.path.join(os.path.dirname(os.path.dirname(HERE)), "ds1")
sys.path.insert(0, DS1)
import emu4, models4
from emu4 import Case, schedule, classify, honesty, solve_at

emu4.SEED.update(models4.SEEDS)
M = {m["name"]: m for m in models4.models()}
NAMES = ["N05", "SEEDED-COMMON", "CHARGE-SLENDER", "LEVER2-k90-s40", "LEVER2-k100-s40", "LEVER2-k110-s20",
         "TILT-LEVER-k90-s40", "TILT-LEVER-k100-s40", "TILT-LEVER-k110-s20"]


def run(name):
    case = Case(M[name])
    g0 = emu4.GATE
    emu4.GATE = "absolute"
    ref = solve_at(case, 2048, ref=True, estimate=False)
    emu4.GATE = g0
    r = schedule(case, "r4", {})
    o = dict(selected=r["selected"], attempts=[list(a) for a in r["attempts"]])
    if r["selected"] is not None:
        pub, cl, sc, g5 = classify(case, r, "r4")
        w, c = honesty(case, pub, cl, ref)
        o.update(honest=w[0] <= 1.0, worst=w[0])
    return o


if __name__ == "__main__":
    emu4.GATE = "hybrid"
    out = {}
    for n in NAMES:
        out[n] = dict(R4=run(n))
        print(n, json.dumps(out[n]), flush=True)
    for tag, sw, val in (("M17", "CHARGE", False), ("M19", "F_ALLOW", 1)):
        saved = getattr(emu4, sw)
        setattr(emu4, sw, val)
        try:
            out["CHARGE-SLENDER"][tag] = run("CHARGE-SLENDER")
        finally:
            setattr(emu4, sw, saved)
        print("CHARGE-SLENDER", tag, json.dumps(out["CHARGE-SLENDER"][tag]), flush=True)
    rec = json.load(open(os.path.join(DS1, "run_controls4.stdout.json")))
    recorded = {r["model"]: r["runs"]["R4"]["selected"] for r in rec}
    out["_agree_with_DS1_run_controls4"] = {n: (out[n]["R4"]["selected"] == recorded.get(n)) for n in NAMES}
    print(json.dumps(out["_agree_with_DS1_run_controls4"]))
    json.dump(out, open(os.path.join(HERE, "r4_repro.json"), "w"), indent=1, default=str)
