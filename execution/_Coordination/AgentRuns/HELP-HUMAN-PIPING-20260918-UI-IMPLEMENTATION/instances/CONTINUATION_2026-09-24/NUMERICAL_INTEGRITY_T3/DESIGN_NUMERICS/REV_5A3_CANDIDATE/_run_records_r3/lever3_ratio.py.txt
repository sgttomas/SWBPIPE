#!/usr/bin/env python3
"""LEVER2 in emu3: the 256 candidate against its 512 verification, W/V of the
worst row under R3 (W at contribution level) and under W_Q = 3p+64; and the
first-order size of the lost spring's effect. Output: lever3_ratio.stdout.json."""
import json
import emu3, lever3
from emu3 import Case, solve_at, stop_rule

out = {}
for k, s in ((90, 40), (100, 40), (110, 20)):
    m, meta = lever3.build(k, s)
    case = Case(m)
    rec = {}
    for tag, wq in (("R3", "P+64"), ("W_Q=3p+64", "3p+64")):
        emu3.W_Q = wq
        emu3.GATE = "hybrid"
        c256 = solve_at(case, 256)
        v512 = solve_at(case, 512)
        ok, info = stop_rule(case, c256, v512, 256, "fin")
        rec[tag] = dict(accepted=ok, reason=str(info["reason"]), w_over_v=info["w_over_v"])
    emu3.W_Q = "P+64"
    out[m["name"]] = rec
    print(m["name"], rec, flush=True)
json.dump(out, open("lever3_ratio.stdout.json", "w"), indent=1)
