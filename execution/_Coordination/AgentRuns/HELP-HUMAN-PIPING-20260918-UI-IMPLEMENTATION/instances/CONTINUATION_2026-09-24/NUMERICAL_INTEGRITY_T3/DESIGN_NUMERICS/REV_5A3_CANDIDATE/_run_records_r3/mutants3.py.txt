#!/usr/bin/env python3
"""R3: every mutant of R3's section 7 (M1-M16), run on the controls, against the
unmutated R3 (fin, hybrid gate, W's residual at contribution level on the final
state, the bounded test on the best state). A kill is a change of selected
precision, of class counts, or of honesty. The controls are R2's plus LEVER2
(gain 2^90) and EXACT-RIGID.
Output: mutants3.stdout.json."""
import json, random
from fractions import Fraction as Fr
import emu3, models3, models2, lever3
from emu3 import Case, schedule, classify, honesty, solve_at
from sweep import gen

emu3.SEED.update(models2.SEEDS)
M = {m["name"]: m for m in models3.models()}
rng = random.Random(3)
for i in range(116):
    g = gen(rng, i)
R115 = dict(g, name="R115-SEED3")
M["R115-SEED3"] = R115
CONTROLS = ["F-2", "F-2-SPOS", "F-2-CEIL", "PRESCRIBED-TAIL", "RIGID-UNLOADED", "F-3-FREE", "F-3-ROT",
            "REACTIONS-ONLY", "SKEW6-K1E-12", "N06", "B1-L", "LEDGER-AT-RESTRAINT", "DEMOTION2", "ASSEMBLY-SAT",
            "LOADONLY-y345", "GS-TRANS-y345", "GS-ROT-y345", "SEEDED-COMMON", "M7-GS1", "M10-ANISO", "M10-G", "R115-SEED3",
            "TWO-SPAN", "LEVER2-k90-s40", "EXACT-RIGID"]
REFS = {}


def run(option="fin"):
    out = {}
    for name in CONTROLS:
        case = Case(M[name])
        if name not in REFS:
            g0 = emu3.GATE
            emu3.GATE = "absolute"
            REFS[name] = solve_at(case, 2048, ref=True, estimate=False)
            emu3.GATE = g0
            emu3.GATE = g0
        r = schedule(case, option, {})
        rec = dict(selected=r["selected"])
        if r["selected"] is not None:
            pub, cl, sc, g5 = classify(case, r, option)
            w, c = honesty(case, pub, cl, REFS[name])
            rec.update(honest=w[0] <= 1.0, relative=c["relative"], g5a=len(g5), corrections=r["cand"]["corrections"])
        out[name] = rec
    return out


def setter(**kw):
    def on():
        saved = {k: getattr(emu3, k) for k in kw}
        for k, v in kw.items():
            setattr(emu3, k, v)
        return saved
    return on


MUTANTS = [
    ("M1 drop V", setter(USE_V=False), "fin"),
    ("M2 uncoupled E", setter(E_COUPLED=False), "fin"),
    ("M3 floor at every p", setter(FLOOR_AT={128, 256, 512}), "fin"),
    ("M4 no floor", setter(FLOOR_AT=set()), "fin"),
    ("M5 E without prescribed |u|", setter(E_PRESCRIBED=False), "fin"),
    ("M6 E from sum |ledger terms|", setter(E_LEDGER="terms"), "fin"),
    ("M7 entrywise E", setter(E_OPERATOR="entrywise"), "fin"),
    ("M8 floor 2^-(2p-74)", setter(FLOOR_EXP_FN=lambda p: 2 * p - 74), "fin"),
    ("M9 floor at 256", setter(FLOOR_AT={256}), "fin"),
    ("M10 omit g", setter(G_MODE="one"), "fin"),
    ("M11 skip the estimator", setter(), "iii"),
    ("M12 coalesced gate (K4 at A2)", setter(GATE="k4"), "fin"),
    ("M13 bounded test drives refinement (pure form)", setter(GATE="absolute"), "fin"),
    ("M14 W residual over K^q's assembled entries (R2's quantity)", setter(W_RES="assembled"), "fin"),
    ("M15 W residual reused from the gate (R2's text)", setter(W_RES="reuse"), "fin"),
    ("M16 bounded test on the last state (R2)", setter(BEST_STATE=False), "fin"),
]
base = run("fin")
results = {"revision (fin, bounded gate)": base}
for tag, on, option in MUTANTS:
    saved = on()
    try:
        results[tag] = run(option)
    finally:
        for k, v in saved.items():
            setattr(emu3, k, v)
for tag, res in results.items():
    print(tag)
    for k, v in res.items():
        if tag.startswith("revision") or v != base[k]:
            print("   ", k, v)
json.dump(results, open("mutants3.stdout.json", "w"), indent=1)
