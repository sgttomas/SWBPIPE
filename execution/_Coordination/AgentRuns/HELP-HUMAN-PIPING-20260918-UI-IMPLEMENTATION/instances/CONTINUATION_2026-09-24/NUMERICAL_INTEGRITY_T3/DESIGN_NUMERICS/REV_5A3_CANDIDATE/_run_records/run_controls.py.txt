#!/usr/bin/env python3
"""Runs every model under the current stop rule and the addendum's options,
and checks each published classification against the P = 2048 reference.
Output: run_controls.stdout.json (one record per model and option)."""
import json, sys, time
from fractions import Fraction as Fr
from emu import Case, solve_at, schedule, classify, honesty, body_E, Stop
from models import models

OPTIONS = ["current", "ia", "ib", "ii", "iii", "iiig", "ig"]
only = set(sys.argv[1:])
out = []
for m in models():
    if only and m["name"] not in only:
        continue
    t0 = time.time()
    case = Case(m)
    ref = solve_at(case, 2048, ref=True)
    cache = {}
    rec = dict(model=m["name"], free=len(case.free), bodies=case.nbody, options={})
    for opt in OPTIONS:
        r = schedule(case, opt, cache)
        o = dict(selected=r["selected"], attempts=r["attempts"])
        if r["selected"] is not None:
            pub, classes, scales = classify(case, r, opt)
            worst, counts = honesty(case, pub, classes, ref["values"], ref["E"])
            o["counts"] = counts
            o["worst_claim_ratio"] = worst[0]
            o["worst_row"] = str(worst[1])
            o["honest"] = worst[0] <= 1.0
            Eb = body_E(case, r["ver"]["E"])
            o["S*_pub"] = [{k: v for k, v in s.items()} for s in scales]
            o["E_body"] = Eb
            o["gated"] = r["info"]["gated"]
        rec["options"][opt] = o
    rec["seconds"] = round(time.time() - t0, 2)
    out.append(rec)
    c = rec["options"]
    print("%-18s " % m["name"] + " ".join("%s:%s%s" % (k, c[k]["selected"], "" if c[k].get("honest", True) else "!DISHONEST") for k in OPTIONS), rec["seconds"], flush=True)
json.dump(out, open("run_controls.stdout.json" if not only else "run_controls.partial.json", "w"), indent=1, default=str)
