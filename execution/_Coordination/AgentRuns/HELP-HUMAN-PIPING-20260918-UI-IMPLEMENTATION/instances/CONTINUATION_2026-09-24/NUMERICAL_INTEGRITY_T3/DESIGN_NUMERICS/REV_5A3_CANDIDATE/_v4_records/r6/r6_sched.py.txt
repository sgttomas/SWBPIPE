#!/usr/bin/env python3
"""V4, delta check at R6, check 4: re-execute DS1's emu6 (read-only import; R6's defaults:
K4's order, per-block B = min(Uc_c, S_c), block scope) on R1's RF-LARGE references, with
honesty against emu6's P = 2048 reference, and compare the selection with DS1's large6
record. V4's own bound figures for the same states are in r6_bounds and r6_exact1000.
Usage: python3 r6_sched.py <case id> [...]
Output: r6_sched.json (appended per case in the log)."""
import json, os, sys, time
HERE = os.path.dirname(os.path.abspath(__file__))
WT = os.path.dirname(os.path.dirname(os.path.dirname(HERE)))
DS1 = os.path.join(WT, "scratch", "ds1")
T3 = os.path.join(WT, "numerics", "projects", "chirality-piping", "execution", "_Coordination", "AgentRuns",
                  "HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION", "instances", "CONTINUATION_2026-09-24",
                  "NUMERICAL_INTEGRITY_T3")
sys.path.insert(0, DS1)
sys.dont_write_bytecode = True
import emu6, r1_adapter
from emu6 import Case, schedule, classify, honesty, solve_at

cases = r1_adapter.load_refs(T3)
out = {}
for cid in sys.argv[1:]:
    t0 = time.time()
    m, index = r1_adapter.adapt(cid, cases[cid], cases[cid]["model"])
    case = Case(m)
    emu6.GATE = "absolute"
    ref = solve_at(case, 2048, ref=True, estimate=False)
    emu6.GATE = "hybrid"
    r = schedule(case, "r6", {})
    o = dict(selected=r["selected"], attempts=[list(a) for a in r["attempts"]])
    if r["selected"] is not None:
        pub, cl, sc, g5 = classify(case, r, "r6")
        w, c = honesty(case, pub, cl, ref)
        o.update(honest=w[0] <= 1.0, worst=w[0], counts=c, g5a=len(g5),
                 charge=r["info"].get("charge_over_allowance"), theta=r["info"].get("theta"))
    o["seconds"] = round(time.time() - t0, 1)
    out[cid] = o
    print(cid, json.dumps(o, default=str), flush=True)
json.dump(out, open(os.path.join(HERE, "r6_sched_%s.json" % sys.argv[1]), "w"), indent=1, default=str)
