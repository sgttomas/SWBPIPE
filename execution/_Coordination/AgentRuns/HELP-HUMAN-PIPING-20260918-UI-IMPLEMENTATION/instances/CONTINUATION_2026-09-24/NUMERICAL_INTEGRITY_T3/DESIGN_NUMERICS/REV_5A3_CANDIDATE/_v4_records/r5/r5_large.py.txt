#!/usr/bin/env python3
"""V4, delta check at R5, check 3 (Uc's looseness on large models): R1's RF-LARGE
families (CHAIN, TREE, CONT; AX and ROT) through R1's own reference models
(DS1's r1_adapter, read-only), assembled by DS1's emu5 at P = 256, radix-equilibrated,
ordered by K4's reverse Cuthill-McKee on the free-free pattern (factor.rs
`reverse_cuthill_mckee` and `order_free`, ported line by line) and, for comparison,
in natural order; then K4's profile LDL^T loop in binary64 (the multipliers'
magnitudes are what U depends on), and:
  U   = ||M(L)^-T D^-1 M(L)^-1 e||_inf,   N_L = || |L| D |L^T| ||_1,
  exact ||K~^-1||_1 by one solve per column (binary64; up to max_exact_n free DOFs),
  log2 t = log2(U gamma_m N_L) at P = 256, 512, 1024 (t >= 1 means Uc does not exist).
Large magnitudes are carried in the decimal module (no overflow).
Usage: python3 r5_large.py [sizes, default 10,100] [max_exact_n, default 700]
Output: r5_large.json."""
import json, math, os, sys, time
from decimal import Decimal, Context, localcontext
from collections import deque
from fractions import Fraction as Fr

HERE = os.path.dirname(os.path.abspath(__file__))
WT = os.path.dirname(os.path.dirname(os.path.dirname(HERE)))
DS1 = os.path.join(WT, "scratch", "ds1")
T3 = os.path.join(WT, "numerics", "projects", "chirality-piping", "execution", "_Coordination", "AgentRuns",
                  "HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION", "instances", "CONTINUATION_2026-09-24",
                  "NUMERICAL_INTEGRITY_T3")
sys.path.insert(0, DS1)
sys.dont_write_bytecode = True
import emu5, r1_adapter
from emu5 import Case, assemble, floor_log2

CTX = Context(prec=34, Emax=10 ** 9, Emin=-10 ** 9)


def rcm(adjacency):
    """factor.rs reverse_cuthill_mckee, line by line."""
    n = len(adjacency)
    nb = [set() for _ in range(n)]
    for node, raw in enumerate(adjacency):
        for other in raw:
            if other != node:
                nb[node].add(other)
                nb[other].add(node)
    neighbors = [sorted(s) for s in nb]
    degrees = [len(s) for s in neighbors]
    neighbors = [sorted(s, key=lambda x: (degrees[x], x)) for s in neighbors]

    def reach(seed):
        marked = [False] * n
        marked[seed] = True
        out = [seed]
        q = deque([seed])
        while q:
            v = q.popleft()
            for w in neighbors[v]:
                if not marked[w]:
                    marked[w] = True
                    out.append(w)
                    q.append(w)
        return out

    def ecc(start):
        marked = [False] * n
        marked[start] = True
        level = [start]
        depth = 0
        while True:
            nxt = []
            for v in level:
                for w in neighbors[v]:
                    if not marked[w]:
                        marked[w] = True
                        nxt.append(w)
            if not nxt:
                return depth, level
            depth += 1
            level = nxt

    def mindeg(nodes):
        return min(nodes, key=lambda v: (degrees[v], v)) if nodes else None

    def pps(seed):
        comp = reach(seed)
        cand = mindeg(comp)
        ce, last = ecc(cand)
        while True:
            nx = mindeg(last)
            if nx is None:
                break
            ne, nl = ecc(nx)
            if ne > ce:
                cand, ce, last = nx, ne, nl
            else:
                break
        return cand

    visited = [False] * n
    order = []
    for seed in range(n):
        if visited[seed]:
            continue
        start = pps(seed)
        visited[start] = True
        q = deque([start])
        while q:
            v = q.popleft()
            order.append(v)
            for w in neighbors[v]:
                if not visited[w]:
                    visited[w] = True
                    q.append(w)
    order.reverse()
    return order


def setup(case, P, use_rcm):
    K, Abar, pattern, members, dblocks = assemble(case.m, P)
    free = case.free
    pos = {g: a for a, g in enumerate(free)}
    sc = [-(floor_log2(abs(K[(g, g)])) // 2) for g in free]
    adj = [[] for _ in free]
    for (r, c) in pattern:
        if r != c and r in pos and c in pos:
            adj[pos[r]].append(pos[c])
    n = len(free)
    order = rcm(adj) if use_rcm else list(range(n))
    rank = [0] * n
    for k, a in enumerate(order):
        rank[a] = k
    first = [min([rank[b] for b in adj[a]] + [i]) for i, a in enumerate(order)]
    rows = []
    for i, a in enumerate(order):
        g = free[a]
        row = {}
        for b in adj[a] + [a]:
            j = rank[b]
            if j <= i:
                v = K.get((g, free[b]), Fr(0))
                row[j] = float(v * Fr(2) ** (sc[a] + sc[b])) if v else 0.0
        rows.append(row)
    return rows, first, n


def ldl(rows, first):
    n = len(rows)
    L = [dict() for _ in range(n)]
    D = [0.0] * n
    work = [0.0] * n
    for i in range(n):
        for j in range(first[i], i):
            s = rows[i].get(j, 0.0)
            for kk in range(max(first[i], first[j]), j):
                s -= work[kk] * L[j].get(kk, 0.0)
            work[j] = s
            L[i][j] = s / D[j]
        piv = rows[i].get(i, 0.0)
        for kk in range(first[i], i):
            piv -= work[kk] * L[i].get(kk, 0.0)
        if piv <= 0:
            raise RuntimeError("nonpositive pivot at %d" % i)
        D[i] = piv
    return L, D


def u_and_nl(L, D, first):
    n = len(D)
    with localcontext(CTX):
        a = [Decimal(1)] * n
        for i in range(n):
            acc = Decimal(1)
            for j in range(first[i], i):
                l = L[i].get(j, 0.0)
                if l:
                    acc += Decimal(abs(l)) * a[j]
            a[i] = acc
        c = [a[i] / Decimal(D[i]) for i in range(n)]
        for i in reversed(range(n)):
            for j in range(first[i], i):
                l = L[i].get(j, 0.0)
                if l:
                    c[j] += Decimal(abs(l)) * c[i]
        U = max(c)
        at = [Decimal(1)] * n
        for i in range(n):
            for j in range(first[i], i):
                l = L[i].get(j, 0.0)
                if l:
                    at[j] += Decimal(abs(l))
        bt = [Decimal(D[i]) * at[i] for i in range(n)]
        ct = list(bt)
        for i in range(n):
            for j in range(first[i], i):
                l = L[i].get(j, 0.0)
                if l:
                    ct[i] += Decimal(abs(l)) * bt[j]
        NL = max(ct)
        return U, NL


def exact_norm(L, D, first, cols=None):
    n = len(D)
    best = 0.0
    for jcol in (cols if cols is not None else range(n)):
        x = [0.0] * n
        x[jcol] = 1.0
        for i in range(jcol + 1, n):
            s = x[i]
            for j in range(first[i], i):
                l = L[i].get(j)
                if l and x[j]:
                    s -= l * x[j]
            x[i] = s
        for i in range(n):
            x[i] /= D[i]
        for i in reversed(range(n)):
            v = x[i]
            if v:
                for j in range(first[i], i):
                    l = L[i].get(j)
                    if l:
                        x[j] -= l * v
        best = max(best, sum(abs(v) for v in x))
    return best


def log2d(x):
    with localcontext(CTX):
        return float(x.ln() / Decimal(2).ln())


def run(cid, c, max_exact_n):
    m, index = r1_adapter.adapt(cid, c, c["model"])
    case = Case(m)
    rec = dict(case=cid, members=len(m["members"]), free=len(case.free))
    for tag, use in (("rcm", True), ("natural", False)):
        t0 = time.time()
        rows, first, n = setup(case, 256, use)
        L, D = ldl(rows, first)
        U, NL = u_and_nl(L, D, first)
        r = dict(profile=sum(i - f + 1 for i, f in enumerate(first)), log2_U=log2d(U), log2_NL=log2d(NL))
        mm = 2 * n + 2
        for P in (256, 512, 1024):
            r["log2_t_P%d" % P] = r["log2_U"] + r["log2_NL"] + math.log2(mm) - P
        if n <= max_exact_n:
            ex = exact_norm(L, D, first)
            r["log2_exact"] = math.log2(ex)
            r["U_over_exact"] = float(U / Decimal(ex)) if r["log2_U"] < 1000 else None
            r["log2_U_over_exact"] = r["log2_U"] - r["log2_exact"]
        r["seconds"] = round(time.time() - t0, 1)
        rec[tag] = r
    return rec


if __name__ == "__main__":
    sizes = [int(x) for x in (sys.argv[1] if len(sys.argv) > 1 else "10,100").split(",")]
    max_exact_n = int(sys.argv[2]) if len(sys.argv) > 2 else 700
    cases = r1_adapter.load_refs(T3)
    out = []
    for cid, c in cases.items():
        if c["family"] != "RF-LARGE" or "generator" in c["model"]:
            continue
        n = len(c["model"]["members"])
        if n not in sizes:
            continue
        rec = run(cid, c, max_exact_n)
        out.append(rec)
        print(json.dumps(rec), flush=True)
    name = "r5_large_%s.json" % "_".join(str(s) for s in sizes)
    json.dump(out, open(os.path.join(HERE, name), "w"), indent=1)
