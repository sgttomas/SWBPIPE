#!/usr/bin/env python3
"""Revision 2 measurements, for every model and precision P with a state:
- total:  max |q_P - q*| / (2^-P e-hat)        over force and moment rows;
- res:    max |q_P - R*(u_P)| / (2^-P E_q)     (R* the recovery with the P = 2048
          operators: the resolution part the count bounds);
- solve:  max |R*(u_P) - q*| / (2^-P e-hat)    (the conditioned part);
- est:    over rows whose solve part exceeds 2^-(P+20) e-hat, the range of
          W_q / |R*(u_P) - q*| (how well the estimator measures the solve part).
Usage: python3 measure2.py <seed> <random count>  -> measure2_<seed>.json"""
import json, sys, random
from fractions import Fraction as Fr
import emu2, models2
from emu2 import Case, solve_at, body_E, e_hat, recover, assemble, Stop
from sweep import gen


def run(m):
    case = Case(m)
    try:
        ref = solve_at(case, 2048, ref=True, estimate=False)
    except Stop:
        return None
    K2, A2, pat2, mem2, db2 = assemble(case.m, 2048)
    out = {}
    for P in (128, 256, 512, 1024):
        try:
            st = solve_at(case, P)
        except Stop as s:
            out[P] = s.what
            continue
        eh = e_hat(case, body_E(case, st))
        rstar = recover(case, 2048, K2, mem2, db2, st["u"], with_loads=True)
        tot = res = sol = 0.0
        est_lo, est_hi = None, None
        for (rid, kind, body, inp), q, qs, rs, E, W in zip(case.layout, st["values"], ref["values"], rstar, st["E"], st["W"]):
            if inp or kind not in ("fo", "mo"):
                continue
            h = Fr(eh[body][kind])
            if h == 0:
                continue
            unit = h / Fr(2) ** P
            tot = max(tot, float(abs(q - qs) / unit))
            sol = max(sol, float(abs(rs - qs) / unit))
            if E:
                res = max(res, float(abs(q - rs) / (E / Fr(2) ** P)))
            actual = abs(rs - qs)
            if actual > unit / Fr(2) ** 20:
                r = float(W / actual)
                est_lo = r if est_lo is None else min(est_lo, r)
                est_hi = r if est_hi is None else max(est_hi, r)
        out[P] = dict(total=tot, res=res, solve=sol, est_lo=est_lo, est_hi=est_hi)
    return out


seed, count = int(sys.argv[1]), int(sys.argv[2])
emu2.SEED.clear()
res = {}
for m in models2.models():
    if m["name"] == "SEEDED-COMMON":
        continue
    res[m["name"]] = run(m)
rng = random.Random(seed)
for i in range(count):
    m = gen(rng, i)
    res["S%d-%s" % (seed, m["name"])] = run(m)
agg = dict(total=0.0, res=0.0, solve=0.0)
arg = {}
elo, ehi = None, None
n = 0
for name, rec in res.items():
    if not rec:
        continue
    for P, v in rec.items():
        if not isinstance(v, dict):
            continue
        n += 1
        for k in agg:
            if v[k] > agg[k]:
                agg[k] = v[k]
                arg[k] = (name, P)
        if v["est_lo"] is not None:
            elo = v["est_lo"] if elo is None else min(elo, v["est_lo"])
            ehi = v["est_hi"] if ehi is None else max(ehi, v["est_hi"])
summary = dict(states=n, max=agg, at=arg, estimator_ratio_range=[elo, ehi])
print(json.dumps(summary, indent=1))
json.dump(dict(summary=summary, per_model=res), open("measure2_%d.json" % seed, "w"), indent=1, default=str)
