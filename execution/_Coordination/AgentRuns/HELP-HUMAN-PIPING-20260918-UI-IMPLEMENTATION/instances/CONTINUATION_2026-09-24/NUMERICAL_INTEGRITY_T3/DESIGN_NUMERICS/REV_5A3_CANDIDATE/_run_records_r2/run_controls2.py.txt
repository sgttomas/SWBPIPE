#!/usr/bin/env python3
"""Revision 2: every control under the current rule (K4's gate), the
candidate (iii, K4's gate), the revision (fin: iii + estimator, bounded-operator
gate) and fin with K4's gate (to isolate S3). Honesty against the P = 2048
reference; G5a's published-data checks; W/V at the accepted pair.
Output: run_controls2.stdout.json."""
import json, sys, time
import emu2, models2
from emu2 import Case, schedule, classify, honesty, solve_at

emu2.SEED.update(models2.SEEDS)
RUNS = [("current", "k4"), ("iii", "k4"), ("fin", "k4"), ("fin", "absolute"), ("fin", "hybrid")]
only = set(sys.argv[1:])
out = []
for m in models2.models():
    if only and m["name"] not in only:
        continue
    t0 = time.time()
    case = Case(m)
    emu2.GATE = "absolute"
    ref = solve_at(case, 2048, ref=True, estimate=False)
    caches = {}
    rec = dict(model=m["name"], runs={})
    for opt, gate in RUNS:
        emu2.GATE = gate
        r = schedule(case, opt, caches.setdefault(gate, {}))
        o = dict(selected=r["selected"], attempts=r["attempts"], solved=r["solved"])
        if r["selected"] is not None:
            pub, cl, sc, g5 = classify(case, r, opt)
            w, c = honesty(case, pub, cl, ref)
            o.update(counts=c, worst=w[0], worst_row=str(w[1]), honest=w[0] <= 1.0, g5a_fail=[str(x) for x in g5],
                     w_over_v=r["info"]["w_over_v"])
        rec["runs"]["%s/%s" % (opt, gate)] = o
    emu2.GATE = "absolute"
    rec["seconds"] = round(time.time() - t0, 2)
    out.append(rec)
    f = lambda k: "%s%s" % (rec["runs"][k]["selected"], "" if rec["runs"][k].get("honest", True) else "!FALSE")
    print("%-22s " % m["name"] + "  ".join("%s:%s" % (k, f(k)) for k in rec["runs"]), rec["seconds"], flush=True)
json.dump(out, open("run_controls2.stdout.json" if not only else "run_controls2.partial.json", "w"), indent=1, default=str)
