#!/usr/bin/env python3
"""R5 (V4-U3): the claim ratio of the false claims on SEEDED-SOFT under mutants M22
(translation and rotation rows without W+) and M15 (W's residual reused from the
gate), and of today's rule; SEEDED-COMMON under M11 and M15 for comparison. The
reference (P = 2048) carries no test-only seed. Output: m22_ratio5.stdout.json."""
import json
import emu5, models5
from emu5 import Case, schedule, classify, honesty, solve_at

M = {m["name"]: m for m in models5.models()}
RUNS = [("SEEDED-SOFT", "M22", "r5", "hybrid", dict(TRRO="zero")),
        ("SEEDED-SOFT", "M15", "r5", "hybrid", dict(W_RES="reuse")),
        ("SEEDED-SOFT", "today", "current", "k4", {}),
        ("SEEDED-SOFT", "R5", "r5", "hybrid", {}),
        ("SEEDED-COMMON", "M11", "r5", "hybrid", dict(W_TEST=False)),
        ("SEEDED-COMMON", "M15", "r5", "hybrid", dict(W_RES="reuse"))]
out = []
refs = {}
for name, tag, opt, gate, sw in RUNS:
    case = Case(M[name])
    if name not in refs:
        emu5.SEED.clear()
        emu5.GATE = "absolute"
        refs[name] = solve_at(case, 2048, ref=True, estimate=False)
    emu5.SEED.clear()
    emu5.SEED.update(models5.SEEDS)
    saved = {k: getattr(emu5, k) for k in sw}
    for k, v in sw.items():
        setattr(emu5, k, v)
    emu5.GATE = gate
    try:
        r = schedule(case, opt, {})
        rec = dict(model=name, run=tag, selected=r["selected"], attempts=[list(a) for a in r["attempts"]])
        if r["selected"] is not None:
            pub, cl, sc, g5 = classify(case, r, opt)
            w, c = honesty(case, pub, cl, refs[name])
            rec.update(worst_ratio=w[0], worst_row=str(w[1]), honest=w[0] <= 1.0, counts=c)
    finally:
        for k, v in saved.items():
            setattr(emu5, k, v)
        emu5.GATE = "absolute"
    out.append(rec)
    print(name, tag, rec.get("selected"), rec.get("worst_ratio"), rec.get("worst_row"), flush=True)
json.dump(out, open("m22_ratio5.stdout.json", "w"), indent=1, default=str)
