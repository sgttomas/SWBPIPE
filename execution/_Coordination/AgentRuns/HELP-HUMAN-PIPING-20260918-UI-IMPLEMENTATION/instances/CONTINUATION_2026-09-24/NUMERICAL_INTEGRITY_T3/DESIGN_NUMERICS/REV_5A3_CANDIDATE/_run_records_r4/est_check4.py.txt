#!/usr/bin/env python3
"""R4 section 4.1.6.3 item 9 (F): how far the Hager-Higham estimate est, as the
condition screen forms it, falls below the exact ||K~_P^-1||_1 (K~_P = S K_P S over
the free DOFs, S K4's radix equilibration), at every verification precision
P = 256, 512, 1024 a control reaches. Exact inverse by Gauss-Jordan in Fractions;
models with more than 40 free DOFs are skipped.
Output: est_check4.stdout.json."""
import json, sys, time
from fractions import Fraction as Fr
import emu4, models4
from emu4 import Case, solve_at, assemble, floor_log2, Stop
from gate_probe3_cases import LOADS, run_345


def exact_inv_norm1(A):
    n = len(A)
    M = [list(row) + [Fr(int(i == j)) for j in range(n)] for i, row in enumerate(A)]
    for c in range(n):
        piv = next(r for r in range(c, n) if M[r][c] != 0)
        M[c], M[piv] = M[piv], M[c]
        pv = M[c][c]
        M[c] = [x / pv for x in M[c]]
        for r in range(n):
            if r != c and M[r][c] != 0:
                f = M[r][c]
                M[r] = [x - f * y for x, y in zip(M[r], M[c])]
    inv = [row[n:] for row in M]
    return max(sum(abs(inv[i][j]) for i in range(n)) for j in range(n))


def check(m):
    case = Case(m)
    free = case.free
    rec = {}
    if not free or len(free) > 40:
        return "skipped (%d free)" % len(free)
    for P in (256, 512, 1024):
        try:
            st = solve_at(case, P)
        except Stop as e:
            rec[P] = "stop:%s" % e.what
            continue
        K = assemble(case.m, P)[0]
        sc = [-(floor_log2(abs(K[(g, g)])) // 2) for g in free]
        Kt = [[K.get((a, b), Fr(0)) * Fr(2) ** (sa + sb) for b, sb in zip(free, sc)] for a, sa in zip(free, sc)]
        ex = exact_inv_norm1(Kt)
        rec[P] = dict(exact=float(ex), est=float(st["est"]), exact_over_est=float(ex / st["est"]))
    return rec


if __name__ == "__main__":
    ms = list(models4.models())
    for y in ((3.0, 4.0, 5.0), (0.0, 0.0, 1.0)):
        for members in (1, 3):
            for label, tip in LOADS:
                m = run_345(y, members, tip)
                m["name"] = "PROBE-%s-%dm-%s" % ("".join("%g" % c for c in y), members, label.split()[0])
                ms.append(m)
    out = {}
    worst = (0.0, None)
    for m in ms:
        t = time.time()
        r = check(m)
        out[m["name"]] = r
        if isinstance(r, dict):
            for P, v in r.items():
                if isinstance(v, dict) and v["exact_over_est"] > worst[0]:
                    worst = (v["exact_over_est"], (m["name"], P))
        print(m["name"], r if isinstance(r, str) else {P: (v if isinstance(v, str) else "%.6g" % v["exact_over_est"]) for P, v in r.items()}, "(%.1fs)" % (time.time() - t), flush=True)
    print("worst exact/est:", worst)
    json.dump(dict(per_model=out, worst=worst), open("est_check4.stdout.json", "w"), indent=1, default=str)
