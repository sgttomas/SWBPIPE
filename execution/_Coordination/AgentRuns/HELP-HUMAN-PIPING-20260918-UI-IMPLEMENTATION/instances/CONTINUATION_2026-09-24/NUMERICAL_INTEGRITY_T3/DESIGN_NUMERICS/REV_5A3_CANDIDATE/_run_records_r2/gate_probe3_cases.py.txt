def run_345(y, members, tip):
    m = new_model("RUN345-%s-%d" % (y, members), [(3.0 * s, 4.0 * s, 0.0) for s in range(members + 1)])
    for s in range(members):
        add_member(m, s + 1, s, s + 1, y=y)
    fix(m, 0, range(6))
    for c, v in enumerate(tip):
        if v:
            load(m, members, c, v)
    return m
