#!/usr/bin/env python3
"""Summary of sweep_<seed>.json files: per option, selected / unresolved /
dishonest counts, and against the current rule: changes of selected p and of
classes at the same p; the attempts (solves) per option."""
import json, sys
OPTIONS = ["current", "ii", "iii", "ib", "iiig", "ig"]
recs = []
for f in sys.argv[1:]:
    recs += [r for r in json.load(open(f)) if r.get("options")]
refstop = 0
summ = {}
for opt in OPTIONS:
    sel = [r for r in recs if opt in r["options"]]
    n_sel = sum(1 for r in sel if r["options"][opt]["selected"] is not None)
    dis = [(r["model"]["name"], r["options"][opt].get("worst"), r["options"][opt].get("row")) for r in sel if r["options"][opt].get("honest") is False]
    at = {}
    for r in sel:
        k = str(r["options"][opt]["selected"])
        at[k] = at.get(k, 0) + 1
    solves = sum(len(r["options"][opt]["solved"]) for r in sel)
    work = sum(sum((p // 128) ** 2 for p in r["options"][opt]["solved"]) for r in sel)
    summ[opt] = dict(models=len(sel), selected=n_sel, unresolved=len(sel) - n_sel, selected_at=at,
                     dishonest=len(dis), dishonest_examples=dis[:8], total_solves=solves,
                     work_units_p_squared=work)
diff = {}
for opt in OPTIONS[1:]:
    ch = []; cls = []
    for r in recs:
        a, b = r["options"]["current"], r["options"][opt]
        if a["selected"] != b["selected"]:
            ch.append((r["model"]["name"], a["selected"], b["selected"], a.get("honest")))
        elif a["selected"] is not None and a["counts"] != b["counts"]:
            cls.append((r["model"]["name"], a["counts"], b["counts"]))
    kinds = {}
    for name, x, y, h in ch:
        key = "%s->%s%s" % (x, y, "" if h is not False else " (current dishonest)")
        kinds[key] = kinds.get(key, 0) + 1
    diff[opt] = dict(selected_p_changed=len(ch), transitions=kinds, same_p_classes_changed=len(cls), class_examples=cls[:4])
print(json.dumps(dict(models=len(recs), summary=summ, versus_current=diff), indent=1))
