"""V4: deliberate lever constructions against the conjecture.
L1: a beam lever (pivot node 0, link at x=b, tip at x=a) moved rigidly in y by prescribed
values, with the tip joined to a prescribed node by a lateral member (a big coupling in the
tip's y row) and a tiny load at the tip (saturated at 128 and 256). Statics gain a/b to the link.
L2: n sites, each a free node on a stiff spine, each with a lateral spring-member to its own
prescribed node (rigid translation), and a tiny load at every site (coherent), collected into a
single root member."""
import math, json, time
from fractions import Fraction as Fr
from v4emu import *
from adversary import score

def lever(k_ratio, a_over_b, lat_E, EI_lever):
    b = 1.0; a = float(a_over_b)
    m = model("L1-a/b=%g-lat=%g-EI=%g" % (a_over_b, lat_E, EI_lever),
              [(0,0,0),(b,0,0),(a,0,0),(b,-1,0),(a+1,0,0)])
    member(m, 0, 1, E=EI_lever, A=1.0, Iy=1.0, Iz=1.0, J=1.0, G=EI_lever/2)
    member(m, 1, 2, E=EI_lever, A=1.0, Iy=1.0, Iz=1.0, J=1.0, G=EI_lever/2)
    member(m, 1, 3, E=k_ratio, A=1.0, Iy=2.0**-30, Iz=2.0**-30, J=1.0, G=k_ratio/2, y=(1,0,0))  # the link
    member(m, 2, 4, E=lat_E, A=1.0, Iy=1.0, Iz=1.0, J=1.0, G=lat_E/2)                              # lateral
    for nd in (0, 3, 4):
        fix(m, nd, (0,), 0.0); fix(m, nd, (1,), 1.0); fix(m, nd, (2,))
    fix(m, 0, (3, 4)); fix(m, 3, (3, 4, 5)); fix(m, 4, (3, 4, 5))
    for nd in (1, 2):
        fix(m, nd, (2, 3, 4))   # in-plane problem
    load(m, 2, 1, 2.0 ** -400)
    return m

def spine(n, s_E, k_lat):
    nodes = [(float(i), 0.0, 0.0) for i in range(n + 1)] + [(float(i), -1.0, 0.0) for i in range(1, n + 1)]
    m = model("L2-n=%d-s=%g-k=%g" % (n, s_E, k_lat), nodes)
    for i in range(n):
        member(m, i, i + 1, E=s_E, A=1.0, Iy=1.0, Iz=1.0, J=1.0, G=s_E/2)
    for i in range(1, n + 1):
        member(m, i, n + i, E=k_lat, A=1.0, Iy=2.0**-30, Iz=2.0**-30, J=1.0, G=k_lat/2, y=(1,0,0))
    fix(m, 0, (0,), 1.0); fix(m, 0, (1, 2, 3, 4, 5))
    for i in range(1, n + 1):
        fix(m, n + i, (0,), 1.0); fix(m, n + i, (1, 2, 3, 4, 5))
        fix(m, i, (2, 3, 4))
        load(m, i, 1, 2.0 ** -400)
    return m

out = {}
for args in [(1.0, 16, 1.0, 1.0), (1.0, 2**10, 1.0, 2.0**20), (1.0, 2**10, 2.0**-10, 2.0**20), (2.0**10, 2**16, 1.0, 2.0**40),
             (1.0, 2**20, 2.0**-20, 2.0**40)]:
    m = lever(*args); t = time.time()
    s = score(m, precs=(128, 256))
    out[m["name"]] = [s[0], s[1], str(s[2])]
    print(m["name"], s, "(%.1fs)" % (time.time() - t), flush=True)
for args in [(4, 1.0, 1.0), (8, 2.0**10, 1.0), (8, 1.0, 2.0**10), (16, 2.0**20, 1.0)]:
    m = spine(*args); t = time.time()
    s = score(m, precs=(128, 256))
    out[m["name"]] = [s[0], s[1], str(s[2])]
    print(m["name"], s, "(%.1fs)" % (time.time() - t), flush=True)
json.dump(out, open("lever.json", "w"), indent=1)
