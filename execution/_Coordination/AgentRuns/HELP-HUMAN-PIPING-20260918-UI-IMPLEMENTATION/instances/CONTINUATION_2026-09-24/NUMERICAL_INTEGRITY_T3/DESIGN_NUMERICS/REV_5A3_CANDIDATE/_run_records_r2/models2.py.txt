#!/usr/bin/env python3
"""DS1 revision-2 models: the candidate's set (models.py) plus V4's DEMOTION2,
ASSEMBLY-SAT and the loaded y_ref (3,4,5) cantilever, and DS1's new controls
(SEEDED-COMMON, LEDGER-AT-RESTRAINT, the M2, M7 and M10 behavioural controls).
Invented inputs; every property is stated here or in models.py."""
from fractions import Fraction as Fr
from emu2 import new_model, add_member, fix, spring, load, n_section
import models as m1


def unit(E=1.0, **over):
    s = dict(E=E, G=1.0, A=1.0, Iy=1.0, Iz=1.0, J=1.0)
    s.update(over)
    return s


def demotion(name, E_big=2.0 ** 480, I_soft=1.0):
    """V4's DEMOTION2 (I_soft = 2^400) and ASSEMBLY-SAT (I_soft = 1): member 1
    (0,0,0)-(2,0,0), E = E_big, unit otherwise, y_ref (0,1,0), translated rigidly
    (ux = 1 at both nodes); member 2 (2,0,0)-(2,2,0), unit section with
    Iy = Iz = I_soft, y_ref (1,0,0); node 2 ux = 1 prescribed, uz, rx, ry, rz
    fixed, uy free and loaded 1 N."""
    m = new_model(name, [(0, 0, 0), (2, 0, 0), (2, 2, 0)])
    add_member(m, 1, 0, 1, y=(0.0, 1.0, 0.0), section=unit(E_big))
    add_member(m, 2, 1, 2, y=(1.0, 0.0, 0.0), section=unit(1.0, Iy=I_soft, Iz=I_soft))
    fix(m, 0, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (0,), 1.0)
    fix(m, 1, (1, 2, 3, 4, 5))
    fix(m, 2, (0,), 1.0)
    fix(m, 2, (2, 3, 4, 5))
    load(m, 2, 1, 1.0)
    return m


def run345(name, y, rigid=None, tip=None, section=None, members=3):
    """A (3,4,0) run of `members` members, N section unless given; root fully
    fixed (rigid=None), translated (rigid='t': ux = 1) or rotated (rigid='r':
    rz = 1e-3); optional tip load 1 N at the last node's uy."""
    m = new_model(name, [(3.0 * s, 4.0 * s, 0.0) for s in range(members + 1)])
    for s in range(members):
        add_member(m, s + 1, s, s + 1, y=y, section=section)
    if rigid is None:
        fix(m, 0, range(6))
    elif rigid == "t":
        fix(m, 0, (0,), 1.0)
        fix(m, 0, (1, 2, 3, 4, 5))
    else:
        fix(m, 0, (0, 1, 2, 3, 4))
        fix(m, 0, (5,), 1e-3)
    if tip:
        load(m, members, 1, tip)
    return m


def models():
    out = list(m1.models())
    out.append(demotion("DEMOTION2", I_soft=2.0 ** 400))
    out.append(demotion("ASSEMBLY-SAT", I_soft=1.0))
    # V4's LOADONLY cantilever: y_ref (3,4,5) on the (3,4,0) run, root fixed, 1 N tip
    out.append(run345("LOADONLY-y345", (3.0, 4.0, 5.0), rigid=None, tip=1.0))
    out.append(run345("LOADONLY-y001", (0.0, 0.0, 1.0), rigid=None, tip=1.0))
    out.append(run345("GS-TRANS-y345", (3.0, 4.0, 5.0), rigid="t"))
    out.append(run345("GS-ROT-y345", (3.0, 4.0, 5.0), rigid="r"))
    out.append(run345("GS-ROT-y345-LOADED", (3.0, 4.0, 5.0), rigid="r", tip=1.0))
    # M6's control: cancelling contributions at a restrained DOF
    m = m1.pin_case("LEDGER-AT-RESTRAINT", m1.SKEW, 1e-4, (0.0, 0.0, 0.0))
    for v in (1e80, 1e-8, -1e80):
        load(m, 0, 0, v)
    load(m, 1, 4, 2e-8)
    out.append(m)
    # S2's seeded-fault control: N05 whose solved state carries the same
    # displacement 2^-110 m at node 1 uy at every precision (emu2.SEED; a test hook)
    m = new_model("SEEDED-COMMON", [(0, 0, 0), (2, 0, 0)])
    add_member(m, 1, 0, 1)
    fix(m, 0, (0, 1, 2, 4, 5))
    spring(m, 1, 0, 3, 1e-4)
    load(m, 1, 3, 1e-8)
    m["stations"].append(dict(id=1, member=1, t=0.5))
    out.append(m)
    # M7's control: a single (3,4,0) member with y_ref (3,4,5) translated rigidly
    # (both nodes prescribed ux = 1); its end actions carry noise-formed axes.
    m = new_model("M7-GS1", [(0, 0, 0), (3, 4, 0)])
    add_member(m, 1, 0, 1, y=(3.0, 4.0, 5.0))
    fix(m, 0, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (0,), 1.0)
    fix(m, 1, (1, 2, 3, 4, 5))
    out.append(m)
    # M10's control: an anisotropic member (Iy = 2^10 Iz) with y_ref 2^-12 from
    # its chord (g = 2^13), rotated rigidly about z by a prescribed root.
    sec = n_section()
    sec["Iy"] = sec["Iz"] * 1024.0
    m = new_model("M10-ANISO", [(0, 0, 0), (3, 4, 0), (6, 8, 0)])
    for s in range(2):
        add_member(m, s + 1, s, s + 1, y=(3.0, 4.0, 5.0 * 2.0 ** -12), section=sec)
    fix(m, 0, (0, 1, 2, 3, 4))
    fix(m, 0, (5,), 1e-3)
    out.append(m)
    # M10's behavioural control: DEMOTION2's layout with member 1's y_ref
    # (1, 2^-12, 0) on its (2,0,0) chord, formed exactly at every p, g = 2^13;
    # E_big = 2^173 (EA/L = 2^172) and member 2's I = 2^100 (no assembled
    # saturation). With g the 128 candidate is rejected by V (E-hat/M = 2^186 >
    # 2^184); without g it is accepted at 128.
    m = new_model("M10-G", [(0, 0, 0), (2, 0, 0), (2, 2, 0)])
    add_member(m, 1, 0, 1, y=(1.0, 2.0 ** -12, 0.0), section=unit(2.0 ** 173))
    add_member(m, 2, 1, 2, y=(1.0, 0.0, 0.0), section=unit(1.0, Iy=2.0 ** 100, Iz=2.0 ** 100))
    fix(m, 0, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (0,), 1.0)
    fix(m, 1, (1, 2, 3, 4, 5))
    fix(m, 2, (0,), 1.0)
    fix(m, 2, (2, 3, 4, 5))
    load(m, 2, 1, 1.0)
    out.append(m)
    return out

def exact_rigid():
    """EXACT-RIGID (V4): one unit-section member (0,0,0)-(2,0,0), y_ref (0,1,0),
    both nodes prescribed ux = 1 (the rest 0): forces exactly zero at every p."""
    m = new_model("EXACT-RIGID", [(0, 0, 0), (2, 0, 0)])
    add_member(m, 1, 0, 1, y=(0.0, 1.0, 0.0), section=unit(1.0))
    fix(m, 0, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (0,), 1.0)
    fix(m, 1, (1, 2, 3, 4, 5))
    return m
