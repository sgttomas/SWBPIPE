#!/usr/bin/env python3
"""R5: every control (models5: R4's 48, V4's R4 controls and DS1's R5 controls) under
today's rule (K4's gate), R4 (emu5 with R4's switches: F*est, theta and the g check
unscoped) and R5 (option r5: Uc, scoped theta and g check). Honesty against the
P = 2048 reference (without any test-only seed); G5a; W/V; for R5 the worst
charge/allowance and theta at every verification the case reaches, and Uc/est.
Output: run_controls5.stdout.json."""
import json, sys, time
import emu5, models5
from emu5 import Case, schedule, classify, honesty, solve_at, charge_report, Stop

R4SW = dict(BOUND="Fest", THETA_SCOPE="all", G_SCOPE="all")
RUNS = [("today", "current", "k4", {}), ("R4", "r4", "hybrid", R4SW), ("R5", "r5", "hybrid", {})]
only = set(sys.argv[1:])
out = []
for m in models5.models():
    if only and m["name"] not in only:
        continue
    t0 = time.time()
    case = Case(m)
    emu5.SEED.clear()
    emu5.GATE = "absolute"
    ref = solve_at(case, 2048, ref=True, estimate=False)
    emu5.SEED.update(models5.SEEDS)
    rec = dict(model=m["name"], runs={})
    for tag, opt, gate, sw in RUNS:
        saved = {k: getattr(emu5, k) for k in sw}
        for k, v in sw.items():
            setattr(emu5, k, v)
        emu5.GATE = gate
        cache = {}
        try:
            r = schedule(case, opt, cache)
            o = dict(selected=r["selected"], attempts=[list(a) for a in r["attempts"]], solved=r["solved"])
            if r["selected"] is not None:
                pub, cl, sc, g5 = classify(case, r, opt)
                w, c = honesty(case, pub, cl, ref)
                o.update(counts=c, worst=w[0], worst_row=str(w[1]), honest=w[0] <= 1.0, g5a_fail=[str(x) for x in g5],
                         w_over_v=r["info"]["w_over_v"])
            if tag == "R5":
                o["charge"] = {}
                for p in (128, 256, 512):
                    try:
                        cand = cache[p] if p in cache else solve_at(case, p)
                        ver = cache[2 * p] if 2 * p in cache else solve_at(case, 2 * p)
                        if isinstance(cand, Stop) or isinstance(ver, Stop):
                            o["charge"][p] = "stop"
                            continue
                        ch, th = charge_report(case, cand, ver, p)
                        uc = ver.get("uc")
                        o["charge"][p] = dict(charge=ch, theta=th,
                                              Uc_over_est=(float(uc["Uc"] / ver["est_hh"]) if uc and uc["Uc"] and ver.get("est_hh") else None))
                    except Stop as e:
                        o["charge"][p] = "stop:%s" % e.what
            rec["runs"][tag] = o
        finally:
            for k, v in saved.items():
                setattr(emu5, k, v)
    emu5.GATE = "absolute"
    rec["seconds"] = round(time.time() - t0, 2)
    out.append(rec)
    f = lambda k: "%s%s" % (rec["runs"][k]["selected"], "" if rec["runs"][k].get("honest", True) else "!FALSE")
    print("%-22s " % m["name"] + "  ".join("%s:%s" % (k, f(k)) for k in rec["runs"]), rec["seconds"], flush=True)
json.dump(out, open("run_controls5.stdout.json" if not only else "run_controls5.partial.json", "w"), indent=1, default=str)
