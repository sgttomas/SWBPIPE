#!/usr/bin/env python3
"""V4, delta check at R4, checks 2 and 8: small exact checks in DS1's emulator emu4
(read-only import).

 (a) Abar symmetry (the controls, and one member with Iy != Iz). emu4's element_bounded forms Bbar^T (|D| Bbar) with each factor
     product rounded once, for every (a, b); the (a, b) and (b, a) entries are
     rounded along different paths. Reported: the largest relative asymmetry of
     the assembled Abar over the controls at P = 256, and ||S Abar S||_1 against
     ||S Abar S||_inf (Theorem step 5 uses the latter; N_u carries the former).
 (b) Gram-Schmidt non-orthogonality (R4 section 7, M10's paragraph: "Non-orthogonality
     enters only at second order, g^2 2^-2P. So rigid motions leak no g-amplified
     force"). A member on the skew chord (3, 4, 12)/13 * 13 with y_ref = chord +
     2^-k (a perpendicular), k = 0, 20, 40, so g ~ 2^(k+4); at P = 256: e_x.e_y, and the
     force the formed K_e produces from the exact rigid rotation about the member
     axis (translations 0, both rotations = the chord), in units of
     2^-P * (Abar |u|) (E's own scale, which carries g).
Output: r4_checks.json."""
import json, os, sys
from fractions import Fraction as Fr

HERE = os.path.dirname(os.path.abspath(__file__))
DS1 = os.path.join(os.path.dirname(os.path.dirname(HERE)), "ds1")
sys.path.insert(0, DS1)
import emu4, models4
from emu4 import Case, assemble, floor_log2, form_member, new_model, add_member


def abar_symmetry(P=256):
    worst = (Fr(0), None)
    worst_norm = (0.0, None)
    for m in models4.models():
        case = Case(m)
        if not case.free:
            continue
        K, Abar, pattern, members, dblocks = assemble(m, P)
        for (i, j), v in Abar.items():
            w = Abar.get((j, i), Fr(0))
            if v and w != v:
                rel = abs(v - w) / max(v, w)
                if rel > worst[0]:
                    worst = (rel, (m["name"], i, j))
        free = case.free
        s = {g: Fr(2) ** (-(floor_log2(abs(K[(g, g)])) // 2)) for g in free}
        col = max(s[j] * sum(s[i] * Abar.get((i, j), Fr(0)) for i in free) for j in free)
        row = max(s[i] * sum(Abar.get((i, j), Fr(0)) * s[j] for j in free) for i in free)
        if col != row:
            r = float(abs(col - row) / max(col, row))
            if r > worst_norm[0]:
                worst_norm = (r, m["name"])
    return dict(worst_entry_relative_asymmetry=float(worst[0]), at=worst[1],
                worst_norm_relative_difference=worst_norm[0], norm_at=worst_norm[1])


def gs_leak(k, P=256):
    chord = (3.0, 4.0, 12.0)
    # a unit perpendicular to the chord, (4, -3, 0)/5, scaled by 2^-k, added to the chord
    yr = (3.0 + 4.0 / 5 * 2.0 ** -k, 4.0 - 3.0 / 5 * 2.0 ** -k, 12.0)
    m = new_model("GS-%d" % k, [(0, 0, 0), chord])
    add_member(m, 1, 0, 1, y=yr, section=dict(E=1.0, G=1.0, A=1.0, Iy=1.0, Iz=1.0, J=1.0))
    op = form_member(m["nodes"], m["members"][0], P)
    ex, ey, ez = op["axes"]
    exey = sum(a * b for a, b in zip(ex, ey))
    d = [Fr(c) for c in chord]
    u = [Fr(0)] * 3 + d + [Fr(0)] * 3 + d          # exact rigid rotation about the axis
    f = [sum(op["ke"][(a, b)] * u[b] for b in range(12)) for a in range(12)]
    Ab = op["Abar"]
    scale = [sum(Ab[a][b] * abs(u[b]) for b in range(12)) for a in range(12)]
    ratio = max(abs(fa) / (Fr(2) ** -P * sa) for fa, sa in zip(f, scale) if sa)
    return dict(k=k, g=op["g"].bit_length() - 1 if isinstance(op["g"], int) else str(op["g"]),
                ex_dot_ey=float(exey), ex_dot_ey_over_2mP=float(abs(exey) * Fr(2) ** P),
                leak_over_2mP_Abar_u=float(ratio),
                leak_over_2mP_Bbar_u_without_g=float(ratio * op["g"]))


def abar_unequal_section(P=256):
    """One member on the chord (1, 2, 3) with Iy = 0.3, Iz = 0.7 (binary64 values):
    the element's Abar (a, b) against (b, a)."""
    m = new_model("ABAR-IY-NE-IZ", [(0, 0, 0), (1, 2, 3)])
    add_member(m, 1, 0, 1, section=dict(E=1.0, G=1.0, A=1.0, Iy=0.3, Iz=0.7, J=1.0))
    op = form_member(m["nodes"], m["members"][0], P)
    A = op["Abar"]
    worst = (Fr(0), None)
    for a in range(12):
        for b in range(12):
            if A[a][b] and A[b][a] != A[a][b]:
                rel = abs(A[a][b] - A[b][a]) / max(A[a][b], A[b][a])
                if rel > worst[0]:
                    worst = (rel, (a, b))
    return dict(worst_relative_asymmetry=float(worst[0]), worst_relative_asymmetry_over_2mP=float(worst[0] * Fr(2) ** P), at=worst[1])


if __name__ == "__main__":
    out = dict(abar_symmetry=abar_symmetry(), abar_unequal_section=abar_unequal_section())
    print(json.dumps(out["abar_unequal_section"], default=str), flush=True)
    print(json.dumps(out["abar_symmetry"], default=str), flush=True)
    out["gs_leak"] = []
    for k in (0, 20, 40):
        r = gs_leak(k)
        out["gs_leak"].append(r)
        print(json.dumps(r), flush=True)
    json.dump(out, open(os.path.join(HERE, "r4_checks.json"), "w"), indent=1, default=str)
