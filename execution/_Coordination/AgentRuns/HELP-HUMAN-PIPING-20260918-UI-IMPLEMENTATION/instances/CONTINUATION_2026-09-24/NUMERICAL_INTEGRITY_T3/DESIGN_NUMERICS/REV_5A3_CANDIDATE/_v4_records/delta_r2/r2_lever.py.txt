"""V4 delta check, the open lead: an assembled stiffness entry saturated at P + 64 at a DOF whose
grounding is soft (a lever's tip), with rigid prescribed motion. Everything else is exact dyadic
arithmetic, so the stop rule sees no difference and W (whose K^q also lacks the entry) sees nothing.
Question: does R2 accept a claim the truth violates, or do V / the floor / the estimate catch it?"""
import os, sys, json, time, math
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from fractions import Fraction as Fr
from v4emu import model, member, fix, load, spring, exact_state, layout, flog2, Stop
import v4emu_r2 as R2

def lever(k, EI1, EI2, EAl, Il, P, ktiny_exp_below, a=1.0):
    L2 = 2.0 ** k
    m = model("LEVER-k%d" % k, [(0,0,0), (-1,0,0), (L2,0,0), (-1,-1,0), (0,1,0)])
    A, B, C, G, D = 0, 1, 2, 3, 4
    member(m, B, A, E=EI1, A=1.0, Iy=1.0, Iz=1.0, J=1.0, G=1.0)      # E*I = EI1 (I = 1)
    member(m, A, C, E=EI2, A=1.0, Iy=1.0, Iz=1.0, J=1.0, G=1.0)
    member(m, G, B, E=EAl, A=1.0, Iy=Il, Iz=Il, J=1.0, G=1.0, y=(1,0,0))  # the link (vertical)
    member(m, A, D, E=1.0, A=1.0, Iy=1.0, Iz=1.0, J=1.0, G=1.0, y=(1,0,0))  # a branch carrying P axially
    for nd in range(5):
        fix(m, nd, (2, 3, 4))                  # in-plane problem
    fix(m, A, (0,), 0.0); fix(m, A, (1,), a)   # pin at A, moved rigidly by a in y
    fix(m, G, (0,), 0.0); fix(m, G, (1,), a); fix(m, G, (5,), 0.0)
    fix(m, B, (0,), 0.0); fix(m, C, (0,), 0.0); fix(m, D, (0,), 0.0); fix(m, D, (5,), 0.0)
    # the tiny global spring at the tip's uy, to fixed ground: its exponent is set below the
    # saturation threshold of the tip's assembled diagonal at 576 bits (after a dry run)
    m["springs"].append((6 * C + 1, 0.0))
    if P: load(m, D, 1, P)
    return m

def tip_diag(m):
    from v4emu import Ar, form_member, assemble
    ar = Ar(None)
    ops = [form_member(ar, m["nodes"], mb) for mb in m["members"]]
    K, n = assemble(ar, m, ops)
    return K[(13, 13)]

out = {}
for (k, EI1, EI2, EAl, Il, Pexp) in [(60, 2.0**40, 2.0**200, 2.0**40, 2.0**-10, None), (90, 2.0**40, 2.0**290, 2.0**40, 2.0**-10, None)]:
    m = lever(k, EI1, EI2, EAl, Il, 0.0, 0)
    Kcc = tip_diag(m)
    e = flog2(Kcc)
    ktiny = 2.0 ** (e - 580)              # below half an ulp at 576 bits
    m["springs"][-1] = (13, ktiny)
    # dry run without P to read e-hat, then set P so that V <= eps*M at 256 with margin
    st = R2.solve_r2(m, 512, "hybrid")
    E, _ = R2.E_r2(m, st)
    efo = float(E[(0, "fo")])
    P = 2.0 ** (math.floor(math.log2(efo)) - 425)
    load(m, 4, 1, P)
    t = time.time()
    res = R2.schedule_r2(m, "hybrid", True)
    rec = dict(k=k, Kcc=float(Kcc), ktiny=ktiny, efo=efo, P=P, selected=res["selected"], attempts=[str(a) for a in res["attempts"]])
    tr = exact_state(m)
    if res["selected"] is not None:
        rows, Sp, Sc = R2.publish_r2(m, res)
        h = R2.honesty_rows(rows, tr)
        rec["honesty"] = [h[0], str(h[1]), h[2]]
        lay = layout(m)
        link = [(str(key), float(qp), cls, None if b is None else float(b), float(qt)) for (key, kind, qp, cls, b), qt in zip(rows, tr["vals"]) if key in (("end", 2, "i", 0), ("end", 2, "j", 0), ("rea", 7))]
        rec["link_rows"] = link
        rec["g5a"] = [str(x) for x in R2.g5a(m, res, rows, Sc)]
        rec["W/V"] = str(res["attempts"][-1])
    # the true link force (member 2 axial) and the error the solve cannot see
    out[m["name"]] = rec
    print(json.dumps(rec, indent=1), "(%.1fs)" % (time.time() - t), flush=True)
json.dump(out, open("delta_r2/r2_lever.json", "w"), indent=1, default=str)
