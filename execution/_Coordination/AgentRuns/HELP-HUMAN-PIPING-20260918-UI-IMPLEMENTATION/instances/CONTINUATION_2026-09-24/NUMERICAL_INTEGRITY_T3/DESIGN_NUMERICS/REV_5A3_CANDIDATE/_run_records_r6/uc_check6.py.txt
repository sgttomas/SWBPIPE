#!/usr/bin/env python3
"""R6 (V4-V1, V4-V2): every certified bound per block against the exact norm.
Every verification state (P = 256, 512, 1024) of every model with at most 40 free
DOFs among models6's controls, the 20-case probe set and HH-FOOL at m = 80 and 200
(R5's 231 model states, now in K4's order), with emu6's shift forced for every block
(SHIFT_ALWAYS, SCOPE "all"): per block of the free-free pattern, Uc_c, the shift
bound S_c (sigma from est, halving, at most three factorizations) and B_c = min(Uc_c, S_c)
against the exact ||K~_c^-1||_1 (Gauss-Jordan in Fractions). Required: Uc_c, S_c and
B_c >= exact whenever they exist. Reports the ratios and est's.
Output: uc_check6.stdout.json."""
import json, sys, time
from fractions import Fraction as Fr
import emu6, models6, models5
from emu6 import Case, solve_at, assemble, floor_log2, Stop
from gate_probe3_cases import LOADS, run_345
from bounds6 import lg

emu6.SCOPE = "all"
emu6.SHIFT_ALWAYS = True


def exact_inv_norm1(A):
    n = len(A)
    M = [list(row) + [Fr(int(i == j)) for j in range(n)] for i, row in enumerate(A)]
    for c in range(n):
        piv = next(r for r in range(c, n) if M[r][c] != 0)
        M[c], M[piv] = M[piv], M[c]
        pv = M[c][c]
        M[c] = [x / pv for x in M[c]]
        for r in range(n):
            if r != c and M[r][c] != 0:
                f = M[r][c]
                M[r] = [x - f * y for x, y in zip(M[r], M[c])]
    inv = [row[n:] for row in M]
    return max(sum(abs(inv[i][j]) for i in range(n)) for j in range(n))


def model_states(m):
    case = Case(m)
    free = case.free
    if not free or len(free) > 40:
        return "skipped (%d free)" % len(free)
    rec = {}
    for P in (256, 512, 1024):
        try:
            st = solve_at(case, P)
        except Stop as e:
            rec[P] = "stop:%s" % e.what
            continue
        K = assemble(case.m, P)[0]
        sc = [-(floor_log2(abs(K[(g, g)])) // 2) for g in free]
        est = st["est_hh"]
        blocks = []
        for bi in st["uc"]["blocks"]:
            pl = bi["pos"]
            Kt = [[K.get((free[a], free[b]), Fr(0)) * Fr(2) ** (sc[a] + sc[b]) for b in pl] for a in pl]
            ex = exact_inv_norm1(Kt)
            r = dict(n=bi["n"], U_over_exact=float(bi["U"] / ex), log2_exact=lg(ex), log2_est=lg(est),
                     Uc_ok=(bi["Uc"] >= ex) if bi["Uc"] is not None else None,
                     S_ok=(bi["S"] >= ex) if bi["S"] is not None else None,
                     B_ok=(bi["B"] >= ex) if bi["B"] is not None else None,
                     Uc_over_exact=(float(bi["Uc"] / ex) if bi["Uc"] is not None else None),
                     S_over_exact=(float(bi["S"] / ex) if bi["S"] is not None else None),
                     B_over_exact=(float(bi["B"] / ex) if bi["B"] is not None else None),
                     tries=bi["tries"])
            blocks.append(r)
        rec[P] = blocks
    return rec


if __name__ == "__main__":
    ms = list(models6.models())
    ms.append(models5.hh_fool(80, False))
    ms.append(models5.hh_fool(200, False))
    for y in ((3.0, 4.0, 5.0), (0.0, 0.0, 1.0)):
        for members in (1, 3):
            for label, tip in LOADS:
                m = run_345(y, members, tip)
                m["name"] = "PROBE-%s-%dm-%s" % ("".join("%g" % c for c in y), members, label.split()[0])
                ms.append(m)
    out = dict(models={})
    agg = dict(states=0, blocks=0, fail_Uc=0, fail_S=0, fail_B=0, S_exists=0, S_missing=0,
               worst_U_over_exact=(0.0, None), worst_S_over_exact=(0.0, None), worst_B_over_exact=(0.0, None),
               min_S_over_exact=(9e99, None), min_B_over_exact=(9e99, None))
    for m in ms:
        t = time.time()
        r = model_states(m)
        out["models"][m["name"]] = r
        if isinstance(r, dict):
            for P, v in r.items():
                if not isinstance(v, list):
                    continue
                agg["states"] += 1
                for k, b in enumerate(v):
                    agg["blocks"] += 1
                    where = (m["name"], P, k)
                    agg["fail_Uc"] += b["Uc_ok"] is False
                    agg["fail_S"] += b["S_ok"] is False
                    agg["fail_B"] += b["B_ok"] is False
                    agg["S_exists"] += b["S_over_exact"] is not None
                    agg["S_missing"] += b["S_over_exact"] is None
                    for key, k2 in (("worst_U_over_exact", "U_over_exact"), ("worst_S_over_exact", "S_over_exact"), ("worst_B_over_exact", "B_over_exact")):
                        if b[k2] is not None and b[k2] > agg[key][0]:
                            agg[key] = (b[k2], where)
                    for key, k2 in (("min_S_over_exact", "S_over_exact"), ("min_B_over_exact", "B_over_exact")):
                        if b[k2] is not None and b[k2] < agg[key][0]:
                            agg[key] = (b[k2], where)
        brief = r if isinstance(r, str) else {P: (v if isinstance(v, str) else ["U/ex=%.3g S/ex=%s B/ex=%.3g" % (b["U_over_exact"], ("%.3g" % b["S_over_exact"]) if b["S_over_exact"] else "-", b["B_over_exact"] or -1) for b in v]) for P, v in r.items()}
        print(m["name"], brief, "(%.1fs)" % (time.time() - t), flush=True)
    out["aggregate"] = agg
    out["shift_factorizations"] = emu6.STATS["shift_factorizations"]
    print("aggregate:", agg, flush=True)
    json.dump(out, open("uc_check6.stdout.json", "w"), indent=1, default=str)
