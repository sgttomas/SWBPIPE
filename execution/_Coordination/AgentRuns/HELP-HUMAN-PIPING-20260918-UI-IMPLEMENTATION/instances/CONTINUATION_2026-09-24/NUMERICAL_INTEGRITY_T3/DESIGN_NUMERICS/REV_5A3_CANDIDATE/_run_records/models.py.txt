#!/usr/bin/env python3
"""DS1 models: K4's survey controls (re-declared from K4's generator at
`cef218a10`, `tests/retained_k4/gen_k4_vectors.py` `models()`, invented inputs),
I12's F-2 probe (`<wt>/scratch/i12/a1_probes/combine_tests_probes.rs`
`probe_single_case_lost_load`) and F-3 (`RIGID-UNLOADED`), and DS1's own
variants. Every property is stated here."""
from emu import new_model, add_member, fix, spring, load, n_section


def pin_case(name, direction, k, moment, members=1, y=(0.0, 0.0, 1.0)):
    nodes = [tuple(c * s for c in direction) for s in range(members + 1)]
    m = new_model(name, nodes)
    for s in range(members):
        add_member(m, s + 1, s, s + 1, y)
    fix(m, 0, (0, 1, 2))
    for c in (3, 4, 5):
        spring(m, c, 0, c, k)
    for c in range(3):
        if moment[c] != 0:
            load(m, members, 3 + c, moment[c])
    return m


def mom(k):
    return (1e-8 * k / 1e-4, 2e-8 * k / 1e-4, 0.0)


SKEW = (3.0, 4.0, 0.0)


def unit_section(E, **over):
    s = dict(E=E, G=1.0, A=1.0, Iy=1.0, Iz=1.0, J=1.0)
    s.update(over)
    return s


def f2(name="F-2", E=1024.0, load_value=2.0 ** -300):
    """I12's F-2: one member (0,0,0)-(2,0,0), E = 1024, G = A = I = J = 1,
    y_ref (0,1,0), so EA/L = 512; node 0 fully fixed with ux = 1; node 1 fixed
    in components 1..5; a load at node 1 ux."""
    m = new_model(name, [(0, 0, 0), (2, 0, 0)])
    add_member(m, 1, 0, 1, y=(0.0, 1.0, 0.0), section=unit_section(E))
    fix(m, 0, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (1, 2, 3, 4, 5))
    load(m, 1, 0, load_value)
    return m


def models():
    out = []
    sec = n_section()
    for name, k, T in (("N05", 1e-4, 1e-8), ("N06", 1e-12, 1e-16)):
        m = new_model(name, [(0, 0, 0), (2, 0, 0)])
        add_member(m, 1, 0, 1)
        fix(m, 0, (0, 1, 2, 4, 5))
        spring(m, 1, 0, 3, k)
        load(m, 1, 3, T)
        m["stations"].append(dict(id=1, member=1, t=0.5))
        out.append(m)
    out.append(pin_case("SKEW-K1E-28", SKEW, 1e-28, mom(1e-28)))
    out.append(pin_case("SKEW6-K1E-12", SKEW, 1e-12, mom(1e-12), members=6))
    out.append(pin_case("SKEW-K1E-12", SKEW, 1e-12, mom(1e-12)))
    out.append(pin_case("SKEW-K1E-4", SKEW, 1e-4, mom(1e-4)))
    out.append(pin_case("AXIS-K1E-4", (5.0, 0.0, 0.0), 1e-4, mom(1e-4)))
    out.append(pin_case("OBLIQUE-K1E-4", (2.0, 3.0, 6.0), 1e-4, mom(1e-4), y=(6.0, 2.0, -3.0)))
    m = pin_case("B1-L", SKEW, 1e-4, (0.0, 0.0, 0.0))
    for v in (1e80, 1e-8, -1e80):
        load(m, 1, 3, v)
    load(m, 1, 4, 2e-8)
    out.append(m)
    for s in (1e-10, 1e-14):
        m = new_model("S8-W-%g" % s, [(0, 0, 0), (3, 4, 0), (6, 8, 0)])
        add_member(m, 1, 0, 1)
        add_member(m, 2, 1, 2, E=sec["E"] * s, G=sec["G"] * s)
        fix(m, 0, range(6))
        for c in range(6):
            spring(m, 1 + c, 2, c, 1e12)
        load(m, 1, 1, 1000.0)
        load(m, 1, 5, 300.0)
        out.append(m)
    m = new_model("DUPLICATE", [(0, 0, 0), (2, 0, 0), (4, 0, 0), (5, 4, 0)])
    add_member(m, 1, 0, 1)
    add_member(m, 2, 1, 3, E=sec["E"] * 2.0 ** -300, G=sec["G"] * 2.0 ** -300)
    add_member(m, 3, 1, 2)
    fix(m, 0, range(6))
    fix(m, 2, range(6))
    fix(m, 3, range(6))
    load(m, 1, 5, 1.0)
    load(m, 1, 1, 1.0)
    out.append(m)
    m = new_model("PRESCRIBED", [(0, 0, 0), (2, 0, 0), (4, 0, 0)])
    add_member(m, 1, 0, 1)
    add_member(m, 2, 1, 2)
    fix(m, 0, (0, 1, 2, 3, 4))
    fix(m, 0, (5,), 1e-3)
    fix(m, 2, (1,), -2e-4)
    fix(m, 2, (0, 2))
    out.append(m)
    a = sec["G"] * sec["J"] / 2
    m = new_model("PIVOT", [(0, 0, 0), (2, 0, 0)])
    add_member(m, 1, 0, 1)
    fix(m, 0, (0, 1, 2, 4, 5))
    spring(m, 1, 0, 3, a * 2.0 ** -120)
    load(m, 1, 3, a * 2.0 ** -120 * 1e-4)
    out.append(m)
    for name, tip in (("REACTIONS-ONLY", 2.0 ** -45), ("MIXED-2^-200", 2.0 ** -200)):
        m = new_model(name, [(0, 0, 0), (3, 4, 0), (3, 4, 2)])
        add_member(m, 1, 0, 1)
        fix(m, 0, (0,), 1.0)
        fix(m, 1, (0,), 1.0)
        fix(m, 0, (1, 2, 3, 4, 5))
        fix(m, 1, (1, 2, 3, 4, 5))
        add_member(m, 2, 1, 2, y=(1.0, 0.0, 0.0), Iy=2.0 ** -26, Iz=2.0 ** -26)
        load(m, 2, 0, tip)
        out.append(m)
    m = new_model("RIGID-UNLOADED", [(0, 0, 0), (3, 4, 0), (0, 0, 5), (0, 0, 7)])
    add_member(m, 1, 0, 1)
    fix(m, 0, (0,), 1.0)
    fix(m, 1, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (1, 2, 3, 4, 5))
    add_member(m, 2, 2, 3, y=(1.0, 0.0, 0.0))
    fix(m, 2, range(6))
    load(m, 3, 0, 1e-30)
    out.append(m)
    m = new_model("TWO-SPAN", [(0, 0, 0), (1, 0, 0), (2 + 2.0 ** -40, 0, 0)])
    add_member(m, 1, 0, 1)
    add_member(m, 2, 1, 2)
    fix(m, 0, range(6))
    fix(m, 2, range(6))
    load(m, 1, 5, 1.0)
    out.append(m)
    m = new_model("ZERO-TORSION-345", [(0, 0, 0), (3, 4, 0)])
    add_member(m, 1, 0, 1)
    fix(m, 0, range(6))
    load(m, 1, 3, 3.0)
    load(m, 1, 4, 4.0)
    out.append(m)
    m = new_model("ALL-ZERO-BODY", [(0, 0, 0), (2, 0, 0), (0, 5, 0), (2, 5, 0)])
    add_member(m, 1, 0, 1)
    add_member(m, 2, 2, 3)
    fix(m, 0, range(6))
    fix(m, 2, range(6))
    load(m, 1, 2, 10.0)
    out.append(m)
    out.append(pin_case("SKEW-K1E-60", SKEW, 1e-60, mom(1e-60)))
    # ---- DS1's F-2 / F-3 controls
    out.append(f2())
    out.append(f2("F-2-CEIL", E=2.0 ** 41, load_value=2.0 ** -1000))
    # F-3 with free nodes: a skew three-member run whose root node is fully
    # prescribed (ux = 1, the rest 0); the free run follows rigidly; unloaded.
    m = new_model("F-3-FREE", [(0, 0, 0), (3, 4, 0), (6, 8, 0), (9, 12, 0)])
    for s in range(3):
        add_member(m, s + 1, s, s + 1)
    fix(m, 0, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    out.append(m)
    # F-3 as a rigid rotation: root prescribed rz = 1e-3 (the rest 0); the free
    # run rotates rigidly about the z axis through node 0; unloaded.
    m = new_model("F-3-ROT", [(0, 0, 0), (3, 4, 0), (6, 8, 0)])
    for s in range(2):
        add_member(m, s + 1, s, s + 1)
    fix(m, 0, (0, 1, 2, 3, 4))
    fix(m, 0, (5,), 1e-3)
    out.append(m)
    # F-2 with a real force elsewhere in the body (S* > 0): member 1 is stiff
    # (E = 2^200, EA/L = 2^199) with F-2's pattern and a load 2^-60 at node 1
    # (N1 = 2^-60, lost at 128 and 256 since u1 = 1 + 2^-259); member 2 (unit
    # section, along y) carries 1 N axially; node 2 is free in ux, uy, rz so
    # member 2 follows node 1's ux rigidly.
    m = new_model("F-2-SPOS", [(0, 0, 0), (2, 0, 0), (2, 2, 0)])
    add_member(m, 1, 0, 1, y=(0.0, 1.0, 0.0), section=unit_section(2.0 ** 200))
    add_member(m, 2, 1, 2, y=(1.0, 0.0, 0.0), section=unit_section(1.0))
    fix(m, 0, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (1, 2, 3, 4, 5))
    fix(m, 2, (2, 3, 4))
    load(m, 1, 0, 2.0 ** -60)
    load(m, 2, 1, 1.0)
    out.append(m)
    # ROOT's A2 point 1: a combined prescribed value 1 + 2^-1100 (an exact
    # combination of binary64 prescriptions, e.g. (1 + 2^-1100) = a - b with a,
    # b binary64 is impossible, so it is given here directly as the exact
    # combined value), rounded once at p, loses its tail at every p <= 1024.
    from fractions import Fraction as Fr
    m = new_model("PRESCRIBED-TAIL", [(0, 0, 0), (2, 0, 0)])
    add_member(m, 1, 0, 1, y=(0.0, 1.0, 0.0), section=unit_section(1024.0))
    fix(m, 0, (0,), 1.0)
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (0,), Fr(1) + Fr(1, 2 ** 1100))
    fix(m, 1, (1, 2, 3, 4, 5))
    out.append(m)
    m = new_model("PRESCRIBED-TAIL-FREE", [(0, 0, 0), (2, 0, 0), (4, 0, 0)])
    add_member(m, 1, 0, 1, y=(0.0, 1.0, 0.0), section=unit_section(1024.0))
    add_member(m, 2, 1, 2, y=(0.0, 1.0, 0.0), section=unit_section(1024.0))
    fix(m, 0, (0,), Fr(1) + Fr(1, 2 ** 1100))
    fix(m, 0, (1, 2, 3, 4, 5))
    fix(m, 1, (1, 2, 3, 4, 5))
    fix(m, 2, (0,), 1.0)
    fix(m, 2, (1, 2, 3, 4, 5))
    out.append(m)
    return out
