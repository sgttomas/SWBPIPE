#!/usr/bin/env python3
"""S3: K4's Rust probe `probe_v4_s3_y_reference_with_a_chord_component`
(K4 `5ad1b6174`, RETURN section 21) re-built in emu2: a (3,4,0) run of 1 or 3
N-section members, root fixed, a tip load (five cases), y_ref (3,4,5) or
(0,0,1); K4's coalesced gate against the bounded-operator gate, under the
revision's rule (fin). Honesty against the P = 2048 reference.
Output: gate_probe3.stdout.json."""
import json
import emu2
from emu2 import Case, schedule, classify, honesty, solve_at, new_model, add_member, fix, load

LOADS = [("general", [1000.0, -500.0, 2000.0, 100.0, 200.0, 300.0]),
         ("in-plane transverse F", [-400.0, 300.0, 0.0, 0.0, 0.0, 0.0]),
         ("out-of-plane F", [0.0, 0.0, 1000.0, 0.0, 0.0, 0.0]),
         ("axial F", [600.0, 800.0, 0.0, 0.0, 0.0, 0.0]),
         ("torque", [0.0, 0.0, 0.0, 60.0, 80.0, 0.0])]


def run_345(y, members, tip):
    m = new_model("RUN345-%s-%d" % (y, members), [(3.0 * s, 4.0 * s, 0.0) for s in range(members + 1)])
    for s in range(members):
        add_member(m, s + 1, s, s + 1, y=y)
    fix(m, 0, range(6))
    for c, v in enumerate(tip):
        if v:
            load(m, members, c, v)
    return m


out = []
for y in ((3.0, 4.0, 5.0), (0.0, 0.0, 1.0)):
    for members in (1, 3):
        for label, tip in LOADS:
            m = run_345(y, members, tip)
            case = Case(m)
            emu2.GATE = "absolute"
            ref = solve_at(case, 2048, ref=True, estimate=False)
            rec = dict(y=y, members=members, load=label)
            for gate in ("k4", "absolute", "hybrid"):
                emu2.GATE = gate
                r = schedule(case, "fin", {})
                o = dict(selected=r["selected"], attempts=[a[1][:32] for a in r["attempts"]])
                if r["selected"]:
                    pub, cl, sc, g5 = classify(case, r, "fin")
                    w, c = honesty(case, pub, cl, ref)
                    o.update(honest=w[0] <= 1.0, worst=w[0], relative=c["relative"])
                rec[gate] = o
            emu2.GATE = "absolute"
            out.append(rec)
            print(y, members, "%-22s" % label, "k4:", rec["k4"]["selected"], rec["k4"]["attempts"][:1],
                  " absolute:", rec["absolute"]["selected"], rec["absolute"].get("honest"),
                  " hybrid:", rec["hybrid"]["selected"], rec["hybrid"].get("honest"), flush=True)
json.dump(out, open("gate_probe3.stdout.json", "w"), indent=1)
