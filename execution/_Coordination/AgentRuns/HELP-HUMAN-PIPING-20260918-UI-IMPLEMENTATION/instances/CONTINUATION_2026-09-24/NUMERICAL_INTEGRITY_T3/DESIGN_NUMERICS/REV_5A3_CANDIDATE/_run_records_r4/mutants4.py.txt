#!/usr/bin/env python3
"""R4: every mutant of R4's section 7 (M1-M23), run on the controls, against the
unmutated R4 (option r4, hybrid gate, W at q_W = 3p + 64 with exact prescribed values,
the charge, theta and the g check, translation and rotation rows carrying W+ + C).
A kill is a change of selected precision, of class counts, or of honesty (M13: of the
correction count, at evidence level). The controls are R3's plus CHARGE-SLENDER,
TILT-LEVER (gain 2^90) and the conditioned controls SKEW-K1E-28, PIVOT and SKEW6.
Output: mutants4.stdout.json."""
import json, random
from fractions import Fraction as Fr
import emu4, models4, models2
from emu4 import Case, schedule, classify, honesty, solve_at
from sweep import gen

emu4.SEED.update(models2.SEEDS)
M = {m["name"]: m for m in models4.models()}
rng = random.Random(3)
for i in range(116):
    g = gen(rng, i)
R115 = dict(g, name="R115-SEED3")
M["R115-SEED3"] = R115
CONTROLS = ["F-2", "F-2-SPOS", "F-2-CEIL", "PRESCRIBED-TAIL", "RIGID-UNLOADED", "F-3-FREE", "F-3-ROT",
            "REACTIONS-ONLY", "SKEW6-K1E-12", "N06", "B1-L", "LEDGER-AT-RESTRAINT", "DEMOTION2", "ASSEMBLY-SAT",
            "LOADONLY-y345", "GS-TRANS-y345", "GS-ROT-y345", "SEEDED-COMMON", "M7-GS1", "M10-ANISO", "M10-G", "R115-SEED3",
            "TWO-SPAN", "LEVER2-k90-s40", "EXACT-RIGID", "CHARGE-SLENDER", "TILT-LEVER-k90-s40",
            "SKEW-K1E-28", "PIVOT"]
REFS = {}


def run(option="r4"):
    out = {}
    for name in CONTROLS:
        case = Case(M[name])
        if name not in REFS:
            g0 = emu4.GATE
            emu4.GATE = "absolute"
            REFS[name] = solve_at(case, 2048, ref=True, estimate=False)
            emu4.GATE = g0
            emu4.GATE = g0
        r = schedule(case, option, {})
        rec = dict(selected=r["selected"])
        if r["selected"] is not None:
            pub, cl, sc, g5 = classify(case, r, option)
            w, c = honesty(case, pub, cl, REFS[name])
            rec.update(honest=w[0] <= 1.0, relative=c["relative"], g5a=len(g5), corrections=r["cand"]["corrections"])
        out[name] = rec
    return out


def setter(**kw):
    def on():
        saved = {k: getattr(emu4, k) for k in kw}
        for k, v in kw.items():
            setattr(emu4, k, v)
        return saved
    return on


MUTANTS = [
    ("M1 drop V", setter(USE_V=False), "r4"),
    ("M2 uncoupled E", setter(E_COUPLED=False), "r4"),
    ("M3 floor at every p", setter(FLOOR_AT={128, 256, 512}), "r4"),
    ("M4 no floor", setter(FLOOR_AT=set()), "r4"),
    ("M5 E without prescribed |u|", setter(E_PRESCRIBED=False), "r4"),
    ("M6 E from sum |ledger terms|", setter(E_LEDGER="terms"), "r4"),
    ("M7 entrywise E", setter(E_OPERATOR="entrywise"), "r4"),
    ("M8 floor 2^-(2p-74)", setter(FLOOR_EXP_FN=lambda p: 2 * p - 74), "r4"),
    ("M9 floor at 256", setter(FLOOR_AT={256}), "r4"),
    ("M10 omit g", setter(G_MODE="one"), "r4"),
    ("M11 skip the estimator's test", setter(W_TEST=False), "r4"),
    ("M12 coalesced gate (K4 at A2)", setter(GATE="k4"), "r4"),
    ("M13 bounded test drives refinement (pure form)", setter(GATE="absolute"), "r4"),
    ("M14 W residual over K^q's assembled entries (R2's quantity)", setter(W_RES="assembled"), "r4"),
    ("M15 W residual reused from the gate (R2's text)", setter(W_RES="reuse"), "r4"),
    ("M16 bounded test on the last state (R2)", setter(BEST_STATE=False), "r4"),
    ("M17 drop the charge", setter(CHARGE=False), "r4"),
    ("M18 W formed at q = 2p + 64 (R3)", setter(W_Q="P+64"), "r4"),
    ("M19 F omitted (F = 1)", setter(F_ALLOW=1), "r4"),
    ("M20 drop the theta check", setter(THETA=False), "r4"),
    ("M21 prescribed values rounded in W's residual (R3)", setter(W_PRESC="rounded"), "r4"),
    ("M22 translation and rotation rows without W+ + C (R3)", setter(TRRO="zero"), "r4"),
    ("M23 drop the g check", setter(G_CHECK=False), "r4"),
]
base = run("r4")
results = {"R4 (r4, hybrid gate)": base}
for tag, on, option in MUTANTS:
    saved = on()
    try:
        results[tag] = run(option)
    finally:
        for k, v in saved.items():
            setattr(emu4, k, v)
for tag, res in results.items():
    print(tag)
    for k, v in res.items():
        if tag.startswith("R4") or v != base[k]:
            print("   ", k, v)
json.dump(results, open("mutants4.stdout.json", "w"), indent=1)
