#!/usr/bin/env python3
"""V4, delta check at R5: a single-body kill of M17 (drop the charge) under R5's Uc.
R5 kills M17 only with HH-SLENDER-m40, where another body's hidden mode lifts the global
Uc. CHARGE-SLENDER (models4.charge_slender, Iy = Iz = 2^-180) no longer does. Here the
same cantilever with Iy = Iz = 2^-200 and 2^-220, in DS1's emu5 (read-only), under R5
and under M17 (CHARGE = False), with honesty against the P = 2048 reference.
Usage: python3 r5_m17.py [exponents, default -180,-200,-220]
Output: r5_m17.json."""
import json, os, sys
HERE = os.path.dirname(os.path.abspath(__file__))
DS1 = os.path.join(os.path.dirname(os.path.dirname(HERE)), "ds1")
sys.path.insert(0, DS1)
sys.dont_write_bytecode = True
import emu5, models4
from emu5 import Case, schedule, classify, honesty, solve_at, charge_report

out = {}
for e in [int(x) for x in (sys.argv[1] if len(sys.argv) > 1 else '-180,-200,-220').split(',')]:
    m = models4.charge_slender(I_exp=e)
    m["name"] = "CHARGE-SLENDER-I%d" % e
    case = Case(m)
    emu5.GATE = "absolute"
    ref = solve_at(case, 2048, ref=True, estimate=False)
    emu5.GATE = "hybrid"
    rec = {}
    for tag, val in (("R5", True), ("M17", False)):
        emu5.CHARGE = val
        r = schedule(case, "r5", {})
        o = dict(selected=r["selected"], attempts=[list(a) for a in r["attempts"]])
        if r["selected"] is not None:
            pub, cl, sc, g5 = classify(case, r, "r5")
            w, c = honesty(case, pub, cl, ref)
            o.update(honest=w[0] <= 1.0, worst=w[0], charge_theta=charge_report(case, r["cand"], r["ver"], r["selected"]))
        rec[tag] = o
    emu5.CHARGE = True
    out[m["name"]] = rec
    print(m["name"], json.dumps(rec), flush=True)
json.dump(out, open(os.path.join(HERE, "r5_m17.json"), "w"), indent=1)
