#!/usr/bin/env python3
"""R3 spot check for section 5.5's derived claim (i): every formed element
stiffness annihilates rigid translations exactly, at every precision, because
its translation columns are exact negatives after each rounding to nearest.
For every member of every R3 control and P in (128, 320, 576, 1024), the exact
products K_e^formed . t for t = unit translation along x, y, z at both ends,
and (for contrast) for a unit rigid rotation about z through the i end.
Output: checks3.stdout.json."""
import json
from fractions import Fraction as Fr
import models3
from emu3 import form_member

out = dict(members=0, translation_nonzero=0, rotation_nonzero=0, examples=[])
for m in models3.models():
    nodes = m["nodes"]
    for mm in m["members"]:
        xi, xj = nodes[mm["i"]], nodes[mm["j"]]
        for P in (128, 320, 576, 1024):
            op = form_member(nodes, mm, P)
            ke = op["ke"]
            out["members"] += 1
            for c in range(3):
                t = [Fr(0)] * 12
                t[c] = t[6 + c] = Fr(1)
                if any(sum(ke[(a, b)] * t[b] for b in range(12)) != 0 for a in range(12)):
                    out["translation_nonzero"] += 1
                    out["examples"].append((m["name"], mm["id"], P, "translation", c))
            # rigid rotation theta_z = 1 about node i: u = z x (x - x_i), rz = 1
            r = [Fr(0)] * 12
            d = [Fr(xj[k]) - Fr(xi[k]) for k in range(3)]
            r[5] = r[11] = Fr(1)
            r[6], r[7] = -d[1], d[0]
            if any(sum(ke[(a, b)] * r[b] for b in range(12)) != 0 for a in range(12)):
                out["rotation_nonzero"] += 1
print(json.dumps({k: v for k, v in out.items() if k != "examples"}), out["examples"][:5])
json.dump(out, open("checks3.stdout.json", "w"), indent=1, default=str)
