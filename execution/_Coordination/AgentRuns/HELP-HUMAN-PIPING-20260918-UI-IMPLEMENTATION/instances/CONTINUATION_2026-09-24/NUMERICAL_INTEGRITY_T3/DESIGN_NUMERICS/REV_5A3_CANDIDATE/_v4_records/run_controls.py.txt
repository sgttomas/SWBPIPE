import json, sys, time
from fractions import Fraction as Fr
from v4emu import run
import cases
ms = [cases.f2(), cases.f2_spos(), cases.f2("F-2-CEIL", E=2.0**41, lv=2.0**-1000), cases.ptail(), cases.ptail(True),
      cases.rigid_unloaded(), cases.f3_free(), cases.f3_rot(), cases.exact_rigid(), cases.demotion()]
res = {}
for m in ms:
    t = time.time()
    out, tr = run(m, rules=("current", "iii"))
    res[m["name"]] = {r: {k: v for k, v in rec.items() if k != "_rows"} for r, rec in out.items()}
    print("  (%.1fs)" % (time.time() - t), flush=True)
json.dump(res, open("run_controls.json", "w"), indent=1, default=str)
