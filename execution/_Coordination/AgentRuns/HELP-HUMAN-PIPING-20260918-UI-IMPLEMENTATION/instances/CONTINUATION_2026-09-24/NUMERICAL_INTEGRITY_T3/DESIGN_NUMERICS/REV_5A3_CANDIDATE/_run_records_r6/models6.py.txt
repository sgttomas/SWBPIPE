import copy
from fractions import Fraction as Fr
import models5
from models5 import UNIT, clone
from emu4 import new_model, add_member, fix, spring, load

SEEDS = dict(models5.SEEDS)


def theta_stub_coupled():
    m = new_model("THETA-STUB-COUPLED", [(0, 0, 0), (1, 0, 0), (2, 1, 0), (2, 0, 0)])
    add_member(m, 1, 0, 1, section=UNIT)
    add_member(m, 3, 1, 3, section=UNIT)
    I = 2.0 ** -204
    add_member(m, 2, 3, 2, y=(0.0, 0.0, 1.0), section=dict(UNIT, A=2.0 ** 50, Iy=I, Iz=I))
    fix(m, 0, range(6))
    fix(m, 3, (0, 1, 2))                # node 3: rotations free, translations fixed
    fix(m, 2, (0, 1, 3, 4, 5))          # node 2: only uz free
    load(m, 1, 1, 1.0)
    return m


R6_CONTROLS = ["THETA-STUB-COUPLED"]


def models():
    out = list(models5.models())
    out += [theta_stub_coupled()]
    return out
