#!/usr/bin/env python3
"""V4, delta check at R6, checks 1 and 3: Lemma E (the shifted-factorization bound)
stressed at low precision, with V4's own implementation (not DS1's).

For random radix-equilibrated SPD matrices K~ (r5_stress's four kinds, n = 3 to 10) and
P = 8, 10, 12, 16 and 24 bits:
  - sigma is swept across lambda_min(K~): lambda * {0.25, 0.5, 0.9, 0.99, 0.999, 1, 1.001,
    1.01, 1.1, 1.5, 3}, with lambda a binary64 power-iteration estimate (an availability
    input only, as est is in R6), and near sigma' = 0: sigma = e (1 + 2^-k), k = 1, 4, 10,
    where e = gamma_m N'_L + delta from a first shift at lambda/2;
  - the shifted diagonal d~_i = fl(K~_ii - sigma) is rounded to nearest; K4's factor loop
    (factor.rs, row by row with the work vector, every operation to nearest at P) runs
    with the pivot test d'_i > 0;
  - where every pivot is positive: N'_L (upward), delta = 2^(1-P) max |d~_i| (exact),
    gamma_m (m = 2n + 2, upward), sigma' = sigma - (gamma_m N'_L + delta) (bracket upward,
    difference downward), S = ceil(sqrt n)/sigma' (upward), all by V4's own nearest-then-
    one-ulp rounding.
Checked exactly (Fractions) wherever sigma' > 0:
  (1) lambda_min(K~) > sigma'  <=>  K~ - sigma' I is positive definite (exact LDL^T);
  (2) S >= ||K~^-1||_1 (exact inverse).
The same checks for M27 (sigma' = sigma, no backward-error or rounding term) show the
test's sensitivity.
Usage: python3 r6_lemmaE.py [matrices, default 1500]
Output: r6_lemmaE.json."""
import json, math, os, random, sys
from fractions import Fraction as Fr
from math import isqrt

HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, os.path.join(os.path.dirname(HERE), "r5"))
sys.dont_write_bytecode = True
from r5_stress import rn, up, down, flog2, equilibrate, gen


def ceil_sqrt(k):
    r = isqrt(k)
    return r if r * r == k else r + 1


def factor_first(Kt):
    n = len(Kt)
    return [min(j for j in range(i + 1) if Kt[i][j] != 0 or j == i) for i in range(n)]


def shifted_factor(G, first, p):
    """K4's loop on G with the pivot test d' > 0. None if a pivot is not positive."""
    n = len(G)
    L = [[Fr(0)] * n for _ in range(n)]
    D = [Fr(0)] * n
    work = [Fr(0)] * n
    for i in range(n):
        for j in range(first[i], i):
            s = G[i][j]
            for kk in range(max(first[i], first[j]), j):
                s = rn(s - rn(work[kk] * L[j][kk], p), p)
            work[j] = s
            L[i][j] = rn(s / D[j], p)
        piv = G[i][i]
        for kk in range(first[i], i):
            piv = rn(piv - rn(work[kk] * L[i][kk], p), p)
        if piv <= 0:
            return None
        D[i] = piv
    return L, D


def nl_up(L, D, first, p):
    n = len(D)
    at = [Fr(1)] * n
    for i in range(n):
        for j in range(first[i], i):
            if L[i][j]:
                at[j] = up(at[j] + abs(L[i][j]), p)
    bt = [up(D[i] * at[i], p) for i in range(n)]
    ct = list(bt)
    for i in range(n):
        for j in range(first[i], i):
            if L[i][j]:
                ct[i] = up(ct[i] + up(abs(L[i][j]) * bt[j], p), p)
    return max(ct)


def shift_bound(Kt, first, sigma, p):
    n = len(Kt)
    dt = [rn(Kt[i][i] - sigma, p) for i in range(n)]
    G = [list(r) for r in Kt]
    for i in range(n):
        G[i][i] = dt[i]
    f = shifted_factor(G, first, p)
    if f is None:
        return None
    L, D = f
    NL = nl_up(L, D, first, p)
    m = 2 * n + 2
    gam = up(Fr(m) / (Fr(2) ** p - m), p)
    delta = Fr(2) ** (1 - p) * max(abs(x) for x in dt)
    e = up(up(gam * NL, p) + delta, p)
    sp = down(sigma - e, p) if sigma - e > 0 else None
    S = up(Fr(ceil_sqrt(n)) / sp, p) if sp and sp > 0 else None
    S27 = up(Fr(ceil_sqrt(n)) / sigma, p)
    return dict(e=e, sp=sp, S=S, S27=S27)


def exact_inverse(A):
    n = len(A)
    M = [list(r) + [Fr(int(i == j)) for j in range(n)] for i, r in enumerate(A)]
    for c in range(n):
        piv = next(r for r in range(c, n) if M[r][c] != 0)
        M[c], M[piv] = M[piv], M[c]
        pv = M[c][c]
        M[c] = [v / pv for v in M[c]]
        for r in range(n):
            if r != c and M[r][c] != 0:
                f = M[r][c]
                M[r] = [x - f * y for x, y in zip(M[r], M[c])]
    return [r[n:] for r in M]


def pd_exact(A, s):
    """K~ - s I positive definite, by exact LDL^T (all pivots > 0)."""
    n = len(A)
    M = [[A[i][j] - (s if i == j else 0) for j in range(n)] for i in range(n)]
    for k in range(n):
        if M[k][k] <= 0:
            return False
        for i in range(k + 1, n):
            f = M[i][k] / M[k][k]
            if f:
                for j in range(k + 1, n):
                    M[i][j] -= f * M[k][j]
    return True


def lam_min_estimate(inv):
    n = len(inv)
    B = [[float(x) for x in r] for r in inv]
    x = [1.0 + 0.1 * i for i in range(n)]
    mu = 0.0
    for _ in range(200):
        y = [sum(B[i][j] * x[j] for j in range(n)) for i in range(n)]
        ny = math.sqrt(sum(v * v for v in y))
        mu = ny / math.sqrt(sum(v * v for v in x))
        x = [v / ny for v in y]
    return 1.0 / mu


if __name__ == "__main__":
    count = int(sys.argv[1]) if len(sys.argv) > 1 else 1500
    rng = random.Random(20260929 + 6)
    st = dict(matrices=0, shifted_factors=0, positive=0, with_S=0, viol_lambda=0, viol_S=0,
              m27_with_S=0, m27_viol_lambda=0, m27_viol_S=0, sigma_above_lambda_positive=0,
              near_zero_cases=0, min_S_over_exact=None, min_m27_S_over_exact=None, examples_m27=[],
              violations=[])
    for it in range(count):
        kind = ("gram", "lap", "lever", "near")[it % 4]
        K = gen(rng, kind)
        Kt = equilibrate(K)
        if any(Kt[i][i] <= 0 for i in range(len(Kt))):
            continue
        n = len(Kt)
        first = factor_first(Kt)
        inv = exact_inverse(Kt)
        ex = max(sum(abs(inv[i][j]) for i in range(n)) for j in range(n))
        lam = lam_min_estimate(inv)
        st["matrices"] += 1
        for P in (8, 10, 12, 16, 24):
            sigmas = [Fr(lam * f) for f in (0.25, 0.5, 0.9, 0.99, 0.999, 1.0, 1.001, 1.01, 1.1, 1.5, 3.0)]
            base = shift_bound(Kt, first, rn(Fr(lam / 2), P), P)
            if base is not None and base["e"] > 0:
                sigmas += [base["e"] * (1 + Fr(1, 2 ** k)) for k in (1, 4, 10)]
            for sg in sigmas:
                sg = rn(sg, P)
                if sg <= 0:
                    continue
                r = shift_bound(Kt, first, sg, P)
                st["shifted_factors"] += 1
                if r is None:
                    continue
                st["positive"] += 1
                above = not pd_exact(Kt, sg)        # sigma >= lambda_min, yet every shifted pivot passed
                if above:
                    st["sigma_above_lambda_positive"] += 1
                if r["sp"] is not None and r["sp"] > 0:
                    st["with_S"] += 1
                    if r["sp"] < sg * Fr(1, 2 ** 4):
                        st["near_zero_cases"] += 1
                    okl = pd_exact(Kt, r["sp"])
                    oks = r["S"] >= ex
                    if not okl:
                        st["viol_lambda"] += 1
                    if not oks:
                        st["viol_S"] += 1
                    if not (okl and oks):
                        st["violations"].append(dict(it=it, kind=kind, P=P, n=n, sigma=float(sg)))
                    q = r["S"] / ex
                    if st["min_S_over_exact"] is None or q < st["min_S_over_exact"][0]:
                        st["min_S_over_exact"] = (q, dict(it=it, kind=kind, P=P, n=n, sigma_over_lam=float(sg) / lam))
                st["m27_with_S"] += 1
                l27 = pd_exact(Kt, sg)
                s27 = r["S27"] >= ex
                if not l27:
                    st["m27_viol_lambda"] += 1
                if not s27:
                    st["m27_viol_S"] += 1
                    if len(st["examples_m27"]) < 5:
                        st["examples_m27"].append(dict(it=it, kind=kind, P=P, n=n, S27_over_exact=float(r["S27"] / ex),
                                                       sigma_over_lam=float(sg) / lam))
                q27 = r["S27"] / ex
                if st["min_m27_S_over_exact"] is None or q27 < st["min_m27_S_over_exact"][0]:
                    st["min_m27_S_over_exact"] = (q27, dict(it=it, kind=kind, P=P))
    for k in ("min_S_over_exact", "min_m27_S_over_exact"):
        if st[k]:
            st[k] = (float(st[k][0]), st[k][1])
    print(json.dumps(st, indent=1))
    json.dump(st, open(os.path.join(HERE, "r6_lemmaE.json"), "w"), indent=1)
