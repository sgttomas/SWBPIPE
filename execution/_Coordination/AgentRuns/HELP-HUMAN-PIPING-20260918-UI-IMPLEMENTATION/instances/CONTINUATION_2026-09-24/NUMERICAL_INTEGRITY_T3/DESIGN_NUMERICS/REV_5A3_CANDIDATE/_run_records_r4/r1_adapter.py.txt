#!/usr/bin/env python3
"""R1 adapter (as in r1_ratios.py, following K4's `r1_adapt` at `3ed6c0e26`),
shared by the revision-2 R1 scripts. Usage: load(T3 path) -> (r1 module, cases)."""
import importlib.util, json, sys
from fractions import Fraction as Fr
from pathlib import Path
sys.dont_write_bytecode = True
from emu2 import new_model, add_member, fix, spring, load
AX = {"UX": 0, "UY": 1, "UZ": 2, "RX": 3, "RY": 4, "RZ": 5}
r1 = None


def load_refs(T3):
    global r1
    spec = importlib.util.spec_from_file_location("r1ref", Path(T3) / "REFERENCES" / "references.py")
    r1 = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(r1)
    return json.loads((Path(T3) / "REFERENCES" / "references.json").read_text())["cases"]


def ref_u(case_json, index, n):
    rows = case_json.get("expected_represented") if case_json.get("basis") == "represented" else case_json.get("expected")
    u = [Fr(0)] * n
    for row in rows or []:
        f = row[0].split(".")
        if f[0] in ("u", "th"):
            u[6 * index[f[1]] + AX[f[2]]] = r1.parse_input(row[1])
    return u


def adapt(cid, c, model):
    basis = c.get("basis") or "intended"
    num = r1.parse_input
    dec = (lambda x: num(x)) if basis == "intended" else (lambda x: r1.rep(num(x)))
    names = list(model["nodes_m"])
    index = {n: k for k, n in enumerate(names)}
    m = new_model(cid, [tuple(float(num(x)) for x in model["nodes_m"][n]) for n in names])
    secs = {}
    for sid, sec in model["sections"].items():
        od, idd = dec(sec["OD"]), dec(sec["ID"])
        A = r1.PI_Q * (od * od - idd * idd) / 4
        I = r1.PI_Q * (od ** 4 - idd ** 4) / 64
        secs[sid] = dict(E=float(num(sec["E"])), G=float(num(sec["G"])), A=float(A), Iy=float(I), Iz=float(I), J=2.0 * float(I))
    for k, (mid, ni, nj, sid) in enumerate(model["members"]):
        xi, xj = m["nodes"][index[ni]], m["nodes"][index[nj]]
        d = [xj[a] - xi[a] for a in range(3)]
        y = (1.0, 0.0, 0.0) if d[0] == 0.0 and d[1] == 0.0 else (0.0, 0.0, 1.0)
        add_member(m, k + 1, index[ni], index[nj], y=y, section=secs[sid])
        m["stations"].append(dict(id=k + 1, member=k + 1, t=0.5))
    nid = 1
    for node, sup in model.get("supports", {}).items():
        n = index[node]
        for dof in sup.get("rigid", []):
            fix(m, n, (AX[dof],))
        sps = sup.get("springs", [])
        for kind in ("translation", "rotation"):
            mine = [sp for sp in sps if sp["kind"] == kind]
            dirs = [[num(x) for x in sp["direction"]] for sp in mine]
            axis_only = all(sum(1 for x in dv if x != 0) == 1 for dv in dirs)
            off = 0 if kind == "translation" else 3
            for sp, dv in zip(mine, dirs):
                kk = float(dec(sp["k"]))
                if axis_only:
                    comp = next(a for a in range(3) if dv[a] != 0)
                    m["springs"].append(dict(id=nid, node=n, c=off + comp, k=kk))
                else:
                    m["dsprings"].append(dict(id=nid, node=n, kind="t" if kind == "translation" else "r",
                                              n=tuple(float(x) for x in dv), k=kk))
                nid += 1
    for node, ld in model.get("loads", {}).items():
        for key, off in (("F", 0), ("M", 3)):
            for a, x in enumerate(ld.get(key, ())):
                v = float(dec(x))
                if v != 0.0:
                    load(m, index[node], off + a, v)
    for (node, dof, x) in model.get("load_contributions_in_authored_order", []):
        load(m, index[node], AX[dof], float(dec(x)))
    return m, index
