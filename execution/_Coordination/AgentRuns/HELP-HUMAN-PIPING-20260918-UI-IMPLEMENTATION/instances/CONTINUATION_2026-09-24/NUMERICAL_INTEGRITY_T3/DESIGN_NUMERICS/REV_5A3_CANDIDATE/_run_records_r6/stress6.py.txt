#!/usr/bin/env python3
"""R6, Lemma E at low precision (the design's precisions make the backward-error term
negligible, so the lemma's rounding argument is exercised where rounding is large).

For random radix-equilibrated symmetric positive definite matrices (n = 3 to 10; Gram,
Laplacian, mixed-sign lever and near-singular kinds, after V4's R5 stress kinds), with
entries rounded to P bits, at P = 10, 12, 16, 20, 24 and 32, and for V4's F2 family at
P = 256 and 512: shift by sigma = f / ||K~^-1||_1 (f drawn from 0.3 to 10, so sigma lies on
both sides of lambda_min and past the point where M27's bound falls below the norm), factor K~ - sigma I with emu6's loop (the same statements as
emu6's shifted_bounds, in natural order), and form
  S     = ceil(sqrt(n)) / sigma',  sigma' = sigma - delta - gamma_m N'_L (downward; R6), and
  S_M27 = ceil(sqrt(n)) / sigma                               (mutant M27),
whenever every shifted pivot is positive. Required: S >= exact ||K~^-1||_1 (Fractions).
M27 is killed if S_M27 < exact occurs. Also Uc and min(Uc, S) against exact.
Usage: python3 stress6.py <count> [seed]. Output: stress6.stdout.json."""
import json, random, sys, time
from fractions import Fraction as Fr
from emu6 import rp, ru, rd, floor_log2, u_pass, nl_pass, ceil_sqrt


def inv_exact(A):
    n = len(A)
    M = [list(row) + [Fr(int(i == j)) for j in range(n)] for i, row in enumerate(A)]
    for c in range(n):
        piv = next(r for r in range(c, n) if M[r][c] != 0)
        M[c], M[piv] = M[piv], M[c]
        pv = M[c][c]
        M[c] = [x / pv for x in M[c]]
        for r in range(n):
            if r != c and M[r][c] != 0:
                f = M[r][c]
                M[r] = [x - f * y for x, y in zip(M[r], M[c])]
    return [row[n:] for row in M]


def norm1(A):
    n = len(A)
    return max(sum(abs(A[i][j]) for i in range(n)) for j in range(n))


def gen(rng, kind, n):
    if kind == "gram":
        B = [[Fr(rng.randint(-8, 8), 8) for _ in range(n)] for _ in range(n)]
        A = [[sum(B[k][i] * B[k][j] for k in range(n)) for j in range(n)] for i in range(n)]
        for i in range(n):
            A[i][i] += Fr(1, 2 ** rng.randint(2, 12))
    elif kind == "laplacian":
        A = [[Fr(0)] * n for _ in range(n)]
        for i in range(n):
            for j in range(i + 1, n):
                if rng.random() < 0.6:
                    w = Fr(rng.randint(1, 16), 4)
                    A[i][i] += w; A[j][j] += w; A[i][j] -= w; A[j][i] -= w
        for i in range(n):
            A[i][i] += Fr(1, 2 ** rng.randint(0, 14))
    elif kind == "lever":
        A = [[Fr(0)] * n for _ in range(n)]
        for i in range(n - 1):
            k = Fr(2) ** rng.randint(-6, 6)
            s = rng.choice((1, -1))
            A[i][i] += k; A[i + 1][i + 1] += k; A[i][i + 1] += s * k; A[i + 1][i] += s * k
            A[i][i] += k * Fr(rng.randint(1, 4), 64)
        A[n - 1][n - 1] += Fr(1, 2 ** rng.randint(0, 10))
    else:  # near-singular: Gram of a nearly rank-deficient matrix
        B = [[Fr(rng.randint(-8, 8), 8) for _ in range(n)] for _ in range(n - 1)]
        row = [sum(B[k][j] * rng.randint(-2, 2) for k in range(n - 1)) for j in range(n)]
        eps = Fr(1, 2 ** rng.randint(4, 20))
        B.append([x + eps * rng.randint(-4, 4) for x in row])
        A = [[sum(B[k][i] * B[k][j] for k in range(n)) for j in range(n)] for i in range(n)]
        for i in range(n):
            A[i][i] += Fr(1, 2 ** rng.randint(10, 30))
    return A


def equilibrate_round(A, P):
    n = len(A)
    s = [-(floor_log2(A[i][i]) // 2) for i in range(n)]
    K = [[rp(A[i][j], P) * Fr(2) ** (s[i] + s[j]) for j in range(n)] for i in range(n)]
    for i in range(n):
        for j in range(i):
            K[i][j] = K[j][i]
    return K


def factor_rows(K, P, sig=None):
    """emu6's loop (natural order, profile from the nonzeros); sig shifts every diagonal.
    Returns (get, first, ok, dsh) where ok is False if a pivot is not positive."""
    n = len(K)
    first = [min(j for j in range(i + 1) if K[i][j] != 0) for i in range(n)]
    sr = [dict() for _ in range(n)]
    dsh = [None] * n
    for i in range(n):
        for j in range(first[i], i + 1):
            sr[i][j] = K[i][j]
        if sig is not None:
            dsh[i] = rp(sr[i][i] - sig, P)
            sr[i][i] = dsh[i]
    get = lambda i, j: sr[i].get(j, Fr(0)) if j >= first[i] else Fr(0)
    wk = [Fr(0)] * n
    for i in range(n):
        for j in range(first[i], i):
            s = get(i, j)
            for kk in range(max(first[i], first[j]), j):
                s = rp(s - rp(wk[kk] * get(j, kk), P), P)
            wk[j] = s
            sr[i][j] = rp(s / get(j, j), P)
        piv = get(i, i)
        for kk in range(first[i], i):
            piv = rp(piv - rp(wk[kk] * get(i, kk), P), P)
        if piv <= 0:
            return get, first, False, dsh
        sr[i][i] = piv
    return get, first, True, dsh


def bounds(K, P, sig):
    n = len(K)
    mm = 2 * n + 2
    gam = ru(Fr(mm) / (Fr(2) ** P - mm), P)
    out = {}
    get, first, ok, _ = factor_rows(K, P)
    if ok:
        U = max(u_pass(get, n, first, P))
        NL = max(nl_pass(get, n, first, P))
        t = ru(ru(U * gam, P) * NL, P)
        den = rd(1 - t, P) if t < 1 else Fr(0)
        out["Uc"] = ru(U / den, P) if den > 0 else None
    get, first, ok, dsh = factor_rows(K, P, sig)
    out["shift_ok"] = ok
    if ok:
        NLp = max(nl_pass(get, n, first, P))
        delta = Fr(2) ** (1 - P) * max(abs(d) for d in dsh)
        e = ru(ru(gam * NLp, P) + delta, P)
        sp = rd(sig - e, P)
        r = ceil_sqrt(n)
        out["S"] = ru(Fr(r) / sp, P) if sp > 0 else None
        out["S_M27"] = ru(Fr(r) / sig, P)
    return out


def f2(m, nv=12):
    n = nv + 4
    hpos = [0, 2, 4, 6]
    vpos = [i for i in range(n) if i not in hpos]
    K = [[Fr(0)] * n for _ in range(n)]
    ks = [Fr(1) if t % 2 == 0 else Fr(3) for t in range(nv - 1)]
    for t, k in enumerate(ks):
        a, b = vpos[t], vpos[t + 1]
        K[a][a] += k; K[b][b] += k; K[a][b] -= k; K[b][a] -= k
    K[vpos[0]][vpos[0]] += Fr(1, 4)
    K[vpos[-1]][vpos[-1]] += Fr(1, 4)
    w = [Fr(1), Fr(-1), Fr(-1), Fr(1)]
    d = Fr(1, 2 ** m)
    for a in range(4):
        for b in range(4):
            K[hpos[a]][hpos[b]] = Fr(int(a == b)) - w[a] * w[b] / 4 + d * w[a] * w[b] / 4
    return K


if __name__ == "__main__":
    count = int(sys.argv[1])
    seed = int(sys.argv[2]) if len(sys.argv) > 2 else 6
    rng = random.Random(seed)
    agg = dict(cases=0, shift_ok=0, S_exists=0, S_below_exact=0, M27_below_exact=0, Uc_exists=0, Uc_below_exact=0,
               B_below_exact=0, min_S_over_exact=None, min_M27_over_exact=None, by_P={})
    examples = []
    t0 = time.time()
    for k in range(count):
        kind = ("gram", "laplacian", "lever", "near")[k % 4]
        n = rng.randint(3, 10)
        P = rng.choice((10, 12, 16, 20, 24, 32))
        A = gen(rng, kind, n)
        K = equilibrate_round(A, P)
        try:
            ex = norm1(inv_exact(K))
        except StopIteration:
            continue
        f = Fr(rng.randint(30, 1000), 100)
        sig = rp(f / ex, P)
        if sig <= 0:
            continue
        b = bounds(K, P, sig)
        agg["cases"] += 1
        bp = agg["by_P"].setdefault(str(P), dict(cases=0, S_exists=0, M27_below=0))
        bp["cases"] += 1
        agg["shift_ok"] += b["shift_ok"]
        if b.get("Uc") is not None:
            agg["Uc_exists"] += 1
            agg["Uc_below_exact"] += b["Uc"] < ex
        if b.get("S") is not None:
            agg["S_exists"] += 1
            bp["S_exists"] += 1
            agg["S_below_exact"] += b["S"] < ex
            r = float(b["S"] / ex)
            agg["min_S_over_exact"] = r if agg["min_S_over_exact"] is None else min(agg["min_S_over_exact"], r)
        cands = [x for x in (b.get("Uc"), b.get("S")) if x is not None]
        if cands and min(cands) < ex:
            agg["B_below_exact"] += 1
        if b["shift_ok"]:
            r = float(b["S_M27"] / ex)
            agg["min_M27_over_exact"] = r if agg["min_M27_over_exact"] is None else min(agg["min_M27_over_exact"], r)
            if b["S_M27"] < ex:
                agg["M27_below_exact"] += 1
                bp["M27_below"] += 1
                if len(examples) < 8:
                    examples.append(dict(kind=kind, n=n, P=P, f=float(f), M27_over_exact=r,
                                         S_over_exact=(float(b["S"] / ex) if b.get("S") else None)))
    # V4's F2 family at the design's precisions
    f2rec = []
    for P in (256, 512):
        for mexp in (20, 60, 100, 110, 120, 200, 230):
            K = equilibrate_round(f2(mexp), P)
            ex = norm1(inv_exact(K))
            for f in (Fr(1, 2), Fr(9, 10), Fr(11, 10), Fr(2)):
                sig = rp(f / ex, P)
                b = bounds(K, P, sig)
                f2rec.append(dict(P=P, m=mexp, f=float(f), shift_ok=b["shift_ok"],
                                  S_over_exact=(float(b["S"] / ex) if b.get("S") else None),
                                  S_ge_exact=(b["S"] >= ex) if b.get("S") else None,
                                  Uc_ge_exact=(b["Uc"] >= ex) if b.get("Uc") else None))
    agg["F2"] = dict(runs=len(f2rec), S_exists=sum(1 for r in f2rec if r["S_over_exact"]),
                     S_below=sum(1 for r in f2rec if r["S_ge_exact"] is False),
                     Uc_below=sum(1 for r in f2rec if r["Uc_ge_exact"] is False))
    agg["seconds"] = round(time.time() - t0, 1)
    print(json.dumps(agg, indent=1))
    print(json.dumps(examples, indent=1))
    json.dump(dict(aggregate=agg, M27_examples=examples, F2=f2rec), open("stress6.stdout.json", "w"), indent=1)
