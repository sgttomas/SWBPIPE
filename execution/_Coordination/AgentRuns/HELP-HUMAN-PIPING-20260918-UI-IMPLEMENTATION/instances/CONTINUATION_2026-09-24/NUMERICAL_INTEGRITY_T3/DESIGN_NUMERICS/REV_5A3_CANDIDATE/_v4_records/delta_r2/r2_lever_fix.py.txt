"""V4: does a contribution-level residual for W catch LEVER2, without false alarms on the controls?"""
import os, sys, json, time
HERE = os.path.dirname(os.path.abspath(__file__)); sys.path.insert(0, os.path.dirname(HERE)); sys.path.insert(0, HERE)
import v4emu_r2 as R2
from v4emu import load, flog2
import r2_lever2 as LV
from r2_controls import ms as controls
R2.W_K = "contributions"
out = {}
for (k, s) in [(90, 40), (110, 20)]:
    m = LV.lever2(k, s)
    ok, K = LV.ldlt_dyadic(m)
    kt = 2.0 ** (flog2(K[(1, 1)]) - 580)
    m["springs"].append((1, kt))
    st = R2.solve_r2(m, 512, "hybrid"); E, _ = R2.E_r2(m, st)
    import math
    efo = float(R2.ehat(E, 0, R2.extents(m)[0])[0]); P = 2.0 ** (math.floor(math.log2(efo)) - 438)
    m = LV.lever2(k, s, Pexp=P); m["springs"].append((1, kt)); load(m, 4, 2, P)
    r = R2.run_r2(m)
    out[m["name"]] = r
    print("contribution-level W:", m["name"], r["selected"], r["attempts"], r.get("honesty"), flush=True)
out["controls"] = {}
for m in controls:
    t = time.time()
    r = R2.run_r2(m, truth=False)
    out["controls"][m["name"]] = dict(selected=r["selected"], last=r["attempts"][-1] if r["attempts"] else None)
    print("contribution-level W:", m["name"], r["selected"], r["attempts"][-1] if r["attempts"] else None, "(%.1fs)" % (time.time()-t), flush=True)
json.dump(out, open(os.path.join(HERE, "r2_lever_fix.json"), "w"), indent=1, default=str)
