import json, time
from gate_probe_nsec import S
from v4emu import model, member, fix, load, run
def cant(name, y, fixed_root=True):
    m = model(name, [(0,0,0),(3,4,0),(6,8,0),(9,12,0)])
    for s in range(3): member(m, s, s+1, y=y, **S)
    fix(m, 0, range(6))
    load(m, 3, 1, 1.0)
    return m
out = {}
for m in (cant("GS-LOADONLY-y345-N", (3.0,4.0,5.0)), cant("GS-LOADONLY-y001-N", (0.0,0.0,1.0))):
    for gate in ("k4", "abs"):
        t = time.time()
        o, tr = run(m, rules=("current","iii"), gate=gate, verbose=False)
        out.setdefault(m["name"], {})[gate] = {r: {k: v for k, v in rec.items() if k in ("selected","attempts","honesty")} for r, rec in o.items()}
        print(m["name"], gate, {r: (v["selected"], v["attempts"], v.get("honesty")) for r, v in out[m["name"]][gate].items()}, "(%.1fs)" % (time.time()-t), flush=True)
json.dump(out, open("gate_probe_loadonly.json","w"), indent=1, default=str)
