#!/usr/bin/env python3
"""R5 (emu5, option r5 under the hybrid gate), R1 (N9 and S3's reach): (a) every non-generator R1 case of the
families K4 and V-K compare, the formation-scale ratio log2(e-hat/S*_c) from
R1's own displacements (recovery at 128, emu2's E); (b) K4's R1 lane families
(RF-CHAIN, RF-SKEW, RF-WEAK, RF-FINITE, RF-CANCEL nodal) and RF-MECH's
companion through the full schedule under the current rule (K4's gate) and the
revision (fin, hybrid gate): selected precision, class counts, G5a failures.
Usage: python3 r1_lane2.py <T3 path> [ratios|lane]"""
import json, math, sys, time
from fractions import Fraction as Fr
import emu5
from emu5 import Case, schedule, classify, assemble, recover, formation_scale, body_E, e_hat, body_kind_max, coupled_wide, rp, Stop
import r1_adapter

T3, what = sys.argv[1], sys.argv[2]
cases = r1_adapter.load_refs(T3)
LANE = ("RF-CHAIN", "RF-SKEW", "RF-WEAK", "RF-FINITE", "RF-CANCEL")
if what == "ratios":
    out = {}
    for cid, c in cases.items():
        if "generator" in c["model"] or "member_uniform_loads_N_per_m_global" in c["model"]:
            continue
        if c["family"] == "RF-LARGE" and len(c["model"]["members"]) > 200:
            continue
        try:
            m, index = r1_adapter.adapt(cid, c, c["model"])
            case = Case(m)
            u = r1_adapter.ref_u(c, index, case.n)
            p = 128
            K, Abar, pat, members, dblocks = assemble(m, p)
            uP = [rp(x, p) for x in u]
            vals = recover(case, p, K, members, dblocks, uP, with_loads=True)
            E = formation_scale(case, p, Abar, members, dblocks, uP)
            st = dict(values=vals, E=E)
            eh = e_hat(case, body_E(case, st))
            s = body_kind_max(case, vals)
            worst = -1e9
            for b in range(case.nbody):
                sc = coupled_wide(s[b], case.extent[b], 256)
                for k in ("fo", "mo"):
                    if eh[b][k] == 0:
                        continue
                    r = math.inf if sc[k] == 0 else math.log2(eh[b][k]) - (math.log2(sc[k].numerator) - math.log2(sc[k].denominator))
                    worst = max(worst, r)
            f = out.setdefault(c["family"], dict(cases=0, max_log2_ehat_over_S=-1e9, at=None, errors=0))
            f["cases"] += 1
            if worst > f["max_log2_ehat_over_S"]:
                f["max_log2_ehat_over_S"], f["at"] = worst, cid
        except Exception as e:
            out.setdefault(c["family"], dict(cases=0, max_log2_ehat_over_S=-1e9, at=None, errors=0))["errors"] += 1
    print(json.dumps(out, indent=1))
    json.dump(out, open("r1_ratios5.stdout.json", "w"), indent=1)
else:
    out = []
    t0 = time.time()
    for cid, c in cases.items():
        if "generator" in c["model"] or "member_uniform_loads_N_per_m_global" in c["model"]:
            continue
        if c["family"] not in LANE and cid != "RF-MECH-LINE345-RX-COMPANION":
            continue
        m, index = r1_adapter.adapt(cid, c, c["model"])
        case = Case(m)
        rec = dict(case=cid, family=c["family"], runs={})
        for opt, gate in (("current", "k4"), ("r5", "hybrid")):
            emu5.GATE = gate
            r = schedule(case, opt, {})
            o = dict(selected=r["selected"], attempts=[a[1][:30] for a in r["attempts"]])
            if r["selected"]:
                pub, cl, sc, g5 = classify(case, r, opt)
                o["classes"] = [x[0] for x in cl]
                o["g5a"] = [str(x) for x in g5]
                o["w_over_v"] = r["info"]["w_over_v"]
                o["charge"] = r["info"].get("charge_over_allowance")
                o["theta"] = r["info"].get("theta")
            rec["runs"][opt] = o
        a, b = rec["runs"]["current"], rec["runs"]["r5"]
        rec["same"] = a["selected"] == b["selected"] and a.get("classes") == b.get("classes")
        out.append(rec)
        print(cid, a["selected"], b["selected"], "same" if rec["same"] else "DIFFERENT", len(b.get("g5a", [])), round(time.time() - t0), flush=True)
    json.dump(out, open("r1_lane5.stdout.json", "w"), indent=1)
