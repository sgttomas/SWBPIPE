#!/usr/bin/env python3
"""R7: emu7's regressions. (a) With ORDER "natural" (where rank and free position
coincide), emu7 with R5's switches (BOUND "Uc", SCOPE "body", option r5) and with R4's
(BOUND "Fest", SCOPE "all", option r4) must reproduce run_controls5.stdout.json's R5 and R4
records exactly. (b) With ORDER "rcm" and EST_INDEX "rank" (emu6's indexing), emu7 with
R6's switches must reproduce run_controls6.stdout.json's R6 records exactly: selected
precision, attempts, class counts, honesty, worst ratio, G5a, W/V, and the charge and theta
at every verification. So the only change from emu6 is the estimator's indexing.
Output: reproduce7.stdout.json."""
import json, time
import emu7, models5, models6
from emu7 import Case, schedule, classify, honesty, solve_at, charge_report, Stop

ref5 = {rec["model"]: rec for rec in json.load(open("run_controls5.stdout.json"))}
ref6 = {rec["model"]: rec for rec in json.load(open("run_controls6.stdout.json"))}
PARTS = [("natural", "position", models5, ref5, {"R5": ("r5", dict(BOUND="Uc", SCOPE="body")), "R4": ("r4", dict(BOUND="Fest", SCOPE="all"))}),
         ("rcm", "rank", models6, ref6, {"R6": ("r6", {})})]
out = []
mism = 0
for order, idx, mods, refs, MODES in PARTS:
    emu7.ORDER = order
    emu7.EST_INDEX = idx
    for m in mods.models():
        t0 = time.time()
        case = Case(m)
        emu7.SEED.clear()
        emu7.GATE = "absolute"
        ref = solve_at(case, 2048, ref=True, estimate=False)
        emu7.SEED.update(mods.SEEDS)
        rec = dict(part=order + "/" + idx, model=m["name"], diffs=[])
        for tag, (opt, sw) in MODES.items():
            saved = {k: getattr(emu7, k) for k in sw}
            for k, v in sw.items():
                setattr(emu7, k, v)
            emu7.GATE = "hybrid"
            cache = {}
            try:
                r = schedule(case, opt, cache)
                o = dict(selected=r["selected"], attempts=[list(a) for a in r["attempts"]], solved=r["solved"])
                if r["selected"] is not None:
                    pub, cl, sc, g5 = classify(case, r, opt)
                    w, c = honesty(case, pub, cl, ref)
                    o.update(counts=c, worst=w[0], worst_row=str(w[1]), honest=w[0] <= 1.0,
                             g5a_fail=[str(x) for x in g5], w_over_v=r["info"]["w_over_v"])
                o["charge"] = {}
                for p in (128, 256, 512):
                    try:
                        cand = cache[p] if p in cache else solve_at(case, p)
                        ver = cache[2 * p] if 2 * p in cache else solve_at(case, 2 * p)
                        if isinstance(cand, Stop) or isinstance(ver, Stop):
                            o["charge"][str(p)] = "stop"
                            continue
                        ch, th = charge_report(case, cand, ver, p)
                        o["charge"][str(p)] = dict(charge=ch, theta=th)
                    except Stop as e:
                        o["charge"][str(p)] = "stop:%s" % e.what
            finally:
                for k, v in saved.items():
                    setattr(emu7, k, v)
                emu7.GATE = "absolute"
            old = refs[m["name"]]["runs"][tag]
            for key in ("selected", "attempts", "counts", "honest", "worst", "g5a_fail", "w_over_v"):
                if o.get(key) != old.get(key):
                    rec["diffs"].append((tag, key, o.get(key), old.get(key)))
            if "charge" in old:
                for p_, v in o["charge"].items():
                    ov = old["charge"].get(p_)
                    if isinstance(v, dict) and isinstance(ov, dict):
                        if v["charge"] != ov["charge"] or v["theta"] != ov["theta"]:
                            rec["diffs"].append((tag, "charge" + p_, v, ov))
                    elif v != ov and not (isinstance(ov, str) and isinstance(v, str)):
                        rec["diffs"].append((tag, "charge" + p_, v, ov))
        mism += bool(rec["diffs"])
        rec["seconds"] = round(time.time() - t0, 2)
        out.append(rec)
        print("%-14s %-24s %s %.2fs" % (rec["part"], m["name"], "same" if not rec["diffs"] else "DIFF %s" % rec["diffs"][:2], rec["seconds"]), flush=True)
print("models with any difference:", mism, "of", len(out))
json.dump(dict(models=out, mismatched=mism), open("reproduce7.stdout.json", "w"), indent=1, default=str)
