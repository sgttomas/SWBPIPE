#!/usr/bin/env python3
"""R6 helpers: JSON-friendly summaries of emu6's bound records (log2 values, so that
bounds above binary64's range stay representable)."""
import math
from fractions import Fraction as Fr


def lg(x):
    """log2 of a positive Fraction (or None)."""
    if x is None:
        return None
    x = Fr(x)
    if x <= 0:
        return None
    num, den = x.numerator, x.denominator
    sn = max(num.bit_length() - 60, 0)
    sd = max(den.bit_length() - 60, 0)
    return math.log2(num >> sn) + sn - math.log2(den >> sd) - sd


def bound_summary(st):
    """Per tested block of a verification state: n, log2 of U, Uc, S and B, the number
    of shifted factorizations, t; and est. From st["uc"] (emu6) and the ingredients."""
    uc = st.get("uc")
    if not uc:
        return None
    est = st.get("est_hh")
    tested = set()
    dblk = st.get("data_blocks") or []
    out = dict(log2_est=lg(est), log2_U_global=lg(uc["U"]), log2_Uc_global=lg(uc["Uc"]),
               shift_runs=uc.get("shift_runs", 0), blocks=[])
    for b, bi in enumerate(uc["blocks"]):
        if bi.get("B") is None and bi.get("S") is None and not (b < len(dblk) and dblk[b]):
            continue
        out["blocks"].append(dict(block=b, n=bi["n"], data=(dblk[b] if b < len(dblk) else None),
                                  log2_U=lg(bi["U"]), log2_Uc=lg(bi["Uc"]), log2_S=lg(bi["S"]), log2_est_c=lg(bi.get("est")),
                                  log2_B=lg(bi["B"]), tries=bi["tries"], log2_t=lg(bi["t"])))
    return out
