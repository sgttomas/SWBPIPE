#!/usr/bin/env python3
"""R6: DS1's port of K4's ordering (k4order.py, from factor.rs at cef218a10) against
V4's independent port (_v4_records/r5/r5_large.py.txt, function rcm, whose text is
read from V4's committed record and executed here, unchanged). Compared on the
free-free structural adjacency of every control (models6), the probe set and R1's
RF-LARGE references at 10, 100 and 1,000 members. Also records each RF-LARGE model's
profile size in K4's order and in natural order.
Usage: python3 k4order_check6.py <T3 path> <V4 r5_large.py.txt>. Output: k4order_check6.stdout.json."""
import json, sys, ast
from collections import deque
import k4order, models6, r1_adapter
from gate_probe3_cases import LOADS, run_345

T3, V4FILE = sys.argv[1], sys.argv[2]
src = open(V4FILE).read()
tree = ast.parse(src)
fn = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == "rcm"][0]
ns = {"deque": deque}
exec(compile(ast.Module(body=[fn], type_ignores=[]), "v4_rcm", "exec"), ns)
v4_rcm = ns["rcm"]


def check(m):
    pat = k4order.structural_pattern(m)
    nn = len(m["nodes"])
    fixed = {6 * c["node"] + c["c"] for c in m["constraints"]}
    free = [g for g in range(6 * nn) if g not in fixed]
    adj = k4order.free_adjacency(free, pat)
    a = k4order.reverse_cuthill_mckee(adj)
    b = v4_rcm(adj)
    rank = [0] * len(free)
    for k, x in enumerate(a):
        rank[x] = k
    first = [min([rank[y] for y in adj[x]] + [i]) for i, x in enumerate(a)]
    prof = sum(i - f + 1 for i, f in enumerate(first))
    firstn = [min([y for y in adj[i]] + [i]) for i in range(len(free))]
    profn = sum(i - f + 1 for i, f in enumerate(firstn))
    return a == b, len(free), prof, profn


out = {}
ms = list(models6.models())
for y in ((3.0, 4.0, 5.0), (0.0, 0.0, 1.0)):
    for members in (1, 3):
        for label, tip in LOADS:
            m = run_345(y, members, tip)
            m["name"] = "PROBE-%s-%dm-%s" % ("".join("%g" % c for c in y), members, label.split()[0])
            ms.append(m)
cases = r1_adapter.load_refs(T3)
for cid, c in cases.items():
    if c["family"] == "RF-LARGE" and "generator" not in c["model"] and len(c["model"]["members"]) in (10, 100, 1000):
        ms.append(r1_adapter.adapt(cid, c, c["model"])[0])
same = 0
for m in ms:
    ok, n, prof, profn = check(m)
    out[m["name"]] = dict(same=ok, free=n, profile_k4=prof, profile_natural=profn)
    same += ok
    print(m["name"], ok, n, prof, profn, flush=True)
print("identical orders:", same, "of", len(ms))
json.dump(dict(models=out, identical=same, total=len(ms)), open("k4order_check6.stdout.json", "w"), indent=1)
