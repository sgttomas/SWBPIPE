"""Locate the GS-ROT-y345 discrepancy: compare DS1's emulator's K and u at 128 with V4's."""
import sys
sys.path.insert(0, "..")
from fractions import Fraction as Fr
import emu as ds1
from emu import new_model, add_member, fix, Case
import v4emu, cases
def gs_ds1(y, rot):
    m = new_model("GS", [(0, 0, 0), (3, 4, 0), (6, 8, 0), (9, 12, 0)])
    for s in range(3):
        add_member(m, s + 1, s, s + 1, y=y)
    if rot:
        fix(m, 0, (0, 1, 2, 3, 4)); fix(m, 0, (5,), 1e-3)
    else:
        fix(m, 0, (0,), 1.0); fix(m, 0, (1, 2, 3, 4, 5))
    return m
md = gs_ds1((3.0,4.0,5.0), True)
case = Case(md)
K1, absK, pattern, members = ds1.assemble(case.m, 128)
mv = cases.gs("GS-ROT-y345", (3.0,4.0,5.0), True)
# V4's section: DS1 default section is n_section() (pi from math); V4's cases use unit sections
print("DS1 member section:", {k: md["members"][0][k] for k in ("E","G","A","Iy","Iz","J")})
print("V4 member section:", {k: mv["members"][0][k] for k in ("E","G","A","Iy","Iz","J")})
