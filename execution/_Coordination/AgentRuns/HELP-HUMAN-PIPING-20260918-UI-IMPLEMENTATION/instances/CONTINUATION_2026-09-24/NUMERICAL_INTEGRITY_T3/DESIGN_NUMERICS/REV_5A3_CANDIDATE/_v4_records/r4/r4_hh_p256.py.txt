#!/usr/bin/env python3
"""V4, delta check at R4, check 3: the F2 family at p = 256 (verified at 512) and the
HH-FOOL frame at m = 200 (its 128 candidate fails the pivot screen; 256 onward
solve). Output: r4_hh_p256.json."""
import json, os, sys
HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, HERE)
from r4_hh import examine, f2
from r4_hhframe import run
out = [examine("F2-m200-p256", f2(200), 256), examine("F2-m230-p256", f2(230), 256), run(200, False)]
for r in out:
    print(json.dumps(r, default=str), flush=True)
json.dump(out, open(os.path.join(HERE, "r4_hh_p256.json"), "w"), indent=1, default=str)
