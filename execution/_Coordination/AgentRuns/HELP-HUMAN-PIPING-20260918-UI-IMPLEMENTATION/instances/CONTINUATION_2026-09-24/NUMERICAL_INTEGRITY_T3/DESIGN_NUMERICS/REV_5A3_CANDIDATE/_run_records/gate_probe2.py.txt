#!/usr/bin/env python3
"""Section 8 probe 2: realistic-looking sloped runs (members with a vertical
component, y_ref (0,0,1) per the R1 adapter rule, so not perpendicular to the
chord), N-section, under (a) gravity-like loads only, (b) a support
settlement plus loads, (c) a settlement only. K4's residual-gate denominator
against the chain-absolute (bounded) one."""
import json
import emu
from emu import Case, schedule, new_model, add_member, fix, load

def sloped(name, settle, loaded):
    pts = [(0, 0, 0), (3, 0, 4), (6, 0, 8), (9, 4, 8)]
    m = new_model(name, pts)
    for s in range(3):
        add_member(m, s + 1, s, s + 1, y=(0.0, 0.0, 1.0))
    fix(m, 0, range(6))
    fix(m, 3, (0, 1, 3, 4, 5))
    fix(m, 3, (2,), settle)
    if loaded:
        load(m, 1, 2, -1000.0)
        load(m, 2, 2, -1000.0)
    return m

cases = [sloped("SLOPED-LOADS", 0.0, True), sloped("SLOPED-SETTLE+LOADS", -0.01, True),
         sloped("SLOPED-SETTLE", -0.01, False), sloped("SLOPED-SETTLE-1m", -1.0, False)]
out = {}
for mode in ("k4", "absolute"):
    emu.GATE_DENOMINATOR = mode
    for m in cases:
        case = Case(m)
        cache = {}
        r = schedule(case, "iii", cache)
        out.setdefault(m["name"], {})[mode] = (r["selected"], [a[1][:28] for a in r["attempts"]])
for k, v in out.items():
    print(k, v)
json.dump(out, open("gate_probe2.stdout.json", "w"), indent=1)

# F-3 on the sloped run: the whole run translated (root fully prescribed,
# rest free, unloaded), and translated with the far support following.
def sloped_rigid(name, t, far):
    pts = [(0, 0, 0), (3, 0, 4), (6, 0, 8), (9, 4, 8)]
    m = new_model(name, pts)
    for s in range(3):
        add_member(m, s + 1, s, s + 1, y=(0.0, 0.0, 1.0))
    for c in range(6):
        fix(m, 0, (c,), t[c] if c < 3 else 0.0)
    if far:
        for c in range(6):
            fix(m, 3, (c,), t[c] if c < 3 else 0.0)
    return m

more = [sloped_rigid("SLOPED-RIGID-X", (0.05, 0.0, 0.0), False), sloped_rigid("SLOPED-RIGID-Z-BOTH", (0.0, 0.0, -0.02), True),
        sloped_rigid("SLOPED-RIGID-XYZ-BOTH", (0.01, 0.02, -0.03), True)]
out2 = {}
for mode in ("k4", "absolute"):
    emu.GATE_DENOMINATOR = mode
    for m in more:
        case = Case(m)
        cache = {}
        r = {opt: schedule(case, opt, cache) for opt in ("current", "iii", "ig")}
        out2.setdefault(m["name"], {})[mode] = {o: (v["selected"], [a[1][:22] for a in v["attempts"]]) for o, v in r.items()}
for k, v in out2.items():
    print(k, v)
json.dump(dict(first=out, rigid=out2), open("gate_probe2.stdout.json", "w"), indent=1)
