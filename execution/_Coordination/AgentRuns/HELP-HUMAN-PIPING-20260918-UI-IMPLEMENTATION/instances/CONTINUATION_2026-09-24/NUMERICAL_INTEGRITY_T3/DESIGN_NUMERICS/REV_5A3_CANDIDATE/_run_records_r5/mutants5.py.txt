#!/usr/bin/env python3
"""R5: every mutant of R5's section 7 (M1-M18, M20-M26; M19 retired with F), run on
the controls, against the unmutated R5 (option r5: Uc in place of F*est, theta per body
with data, the g check scoped to members in bodies with data). A kill is a change of
selected precision, of class counts, or of honesty (M13: of the correction count, at
evidence level). The controls are R4's plus V4's R4 controls and DS1's R5 controls.
The reference solve carries no test-only seed.
Output: mutants5.stdout.json."""
import json, random
from fractions import Fraction as Fr
import emu5, models5, models2
from emu5 import Case, schedule, classify, honesty, solve_at
from sweep import gen

emu5.SEED.update(models5.SEEDS)
M = {m["name"]: m for m in models5.models()}
rng = random.Random(3)
for i in range(116):
    g = gen(rng, i)
R115 = dict(g, name="R115-SEED3")
M["R115-SEED3"] = R115
CONTROLS = ["F-2", "F-2-SPOS", "F-2-CEIL", "PRESCRIBED-TAIL", "RIGID-UNLOADED", "F-3-FREE", "F-3-ROT",
            "REACTIONS-ONLY", "SKEW6-K1E-12", "N06", "B1-L", "LEDGER-AT-RESTRAINT", "DEMOTION2", "ASSEMBLY-SAT",
            "LOADONLY-y345", "GS-TRANS-y345", "GS-ROT-y345", "SEEDED-COMMON", "M7-GS1", "M10-ANISO", "M10-G", "R115-SEED3",
            "TWO-SPAN", "LEVER2-k90-s40", "EXACT-RIGID", "CHARGE-SLENDER", "TILT-LEVER-k90-s40",
            "SKEW-K1E-28", "PIVOT", "SEEDED-SOFT", "THETA-ZERO-BODY", "G-FIXED-MEMBER", "THETA-STUB",
            "G-PRESC-MEMBER", "HH-SLENDER-m40", "HH-FOOL-m40-LOADED", "HH-FOOL-m100-LOADED"]
REFS = {}


def run(option="r5"):
    out = {}
    for name in CONTROLS:
        case = Case(M[name])
        if name not in REFS:
            g0 = emu5.GATE
            emu5.GATE = "absolute"
            seeds = dict(emu5.SEED)
            emu5.SEED.clear()
            REFS[name] = solve_at(case, 2048, ref=True, estimate=False)
            emu5.SEED.update(seeds)
            emu5.GATE = g0
            emu5.GATE = g0
        r = schedule(case, option, {})
        rec = dict(selected=r["selected"])
        if r["selected"] is not None:
            pub, cl, sc, g5 = classify(case, r, option)
            w, c = honesty(case, pub, cl, REFS[name])
            rec.update(honest=w[0] <= 1.0, relative=c["relative"], g5a=len(g5), corrections=r["cand"]["corrections"])
        out[name] = rec
    return out


def setter(**kw):
    def on():
        saved = {k: getattr(emu5, k) for k in kw}
        for k, v in kw.items():
            setattr(emu5, k, v)
        return saved
    return on


MUTANTS = [
    ("M1 drop V", setter(USE_V=False), "r5"),
    ("M2 uncoupled E", setter(E_COUPLED=False), "r5"),
    ("M3 floor at every p", setter(FLOOR_AT={128, 256, 512}), "r5"),
    ("M4 no floor", setter(FLOOR_AT=set()), "r5"),
    ("M5 E without prescribed |u|", setter(E_PRESCRIBED=False), "r5"),
    ("M6 E from sum |ledger terms|", setter(E_LEDGER="terms"), "r5"),
    ("M7 entrywise E", setter(E_OPERATOR="entrywise"), "r5"),
    ("M8 floor 2^-(2p-74)", setter(FLOOR_EXP_FN=lambda p: 2 * p - 74), "r5"),
    ("M9 floor at 256", setter(FLOOR_AT={256}), "r5"),
    ("M10 omit g", setter(G_MODE="one"), "r5"),
    ("M11 skip the estimator's test", setter(W_TEST=False), "r5"),
    ("M12 coalesced gate (K4 at A2)", setter(GATE="k4"), "r5"),
    ("M13 bounded test drives refinement (pure form)", setter(GATE="absolute"), "r5"),
    ("M14 W residual over K^q's assembled entries (R2's quantity)", setter(W_RES="assembled"), "r5"),
    ("M15 W residual reused from the gate (R2's text)", setter(W_RES="reuse"), "r5"),
    ("M16 bounded test on the last state (R2)", setter(BEST_STATE=False), "r5"),
    ("M17 drop the charge", setter(CHARGE=False), "r5"),
    ("M18 W formed at q = 2p + 64 (R3)", setter(W_Q="P+64"), "r5"),
    ("M20 drop the theta check", setter(THETA=False), "r5"),
    ("M21 prescribed values rounded in W's residual (R3)", setter(W_PRESC="rounded"), "r5"),
    ("M22 translation and rotation rows without W+ + C (R3)", setter(TRRO="zero"), "r5"),
    ("M23 drop the g check", setter(G_CHECK=False), "r5"),
    ("M24 est in place of Uc", setter(BOUND="est"), "r5"),
    ("M25 theta and the g check unscoped (R4)", setter(THETA_SCOPE="all", G_SCOPE="all"), "r5"),
    ("M26 F*est in place of Uc (R4)", setter(BOUND="Fest"), "r5"),
]
base = run("r5")
results = {"R5 (r5, hybrid gate)": base}
for tag, on, option in MUTANTS:
    saved = on()
    try:
        results[tag] = run(option)
    finally:
        for k, v in saved.items():
            setattr(emu5, k, v)
for tag, res in results.items():
    print(tag)
    for k, v in res.items():
        if tag.startswith("R5") or v != base[k]:
            print("   ", k, v)
json.dump(results, open("mutants5.stdout.json", "w"), indent=1)
