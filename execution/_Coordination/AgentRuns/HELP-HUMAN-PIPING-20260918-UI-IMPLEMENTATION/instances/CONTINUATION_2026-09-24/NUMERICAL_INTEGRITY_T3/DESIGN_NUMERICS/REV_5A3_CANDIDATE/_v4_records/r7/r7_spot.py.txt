#!/usr/bin/env python3
"""V4, delta check at R7, check 2: spot-reproduce DS1's R7 claims with V4's own K4-indexed
copy of emu6 (`_v4_records/r6/emu6_k4hh.py.txt`, regenerated by r6_k4hh.py; it is not
emu7), against DS1's R7 records.

 (a) The 59 controls (models6, with its test-only seeds) under today's rule, R4, R5 and
     R6 (the switches of DS1's run_controls7): selected precision and attempts, against
     run_controls7.stdout.json.
 (b) Mutants M17, M20, M22, M23, M24, M26, M28 and M29 on ten controls (HH-SLENDER-m40,
     HH-FOOL-m40-LOADED, HH-FOOL-m100-LOADED, CHARGE-SLENDER, THETA-STUB-COUPLED,
     SEEDED-SOFT, G-PRESC-MEMBER, SKEW6-K1E-12 and R1's RF-LARGE-CHAIN-n00100-AX and
     TREE-n00100-AX), with honesty against the P = 2048 reference: selected, honest,
     relative-class count, G5a failures and corrections, against mutants7.stdout.json.
 (c) The HH-FOOL estimate at m = 40, 80, 100 and 200: est_c of the hidden block and the
     global est against the exact per-block norm, at P = 256.
Output: r7_spot.json."""
import json, math, os, sys
from fractions import Fraction as Fr

HERE = os.path.dirname(os.path.abspath(__file__))
V4 = os.path.dirname(HERE)
WT = os.path.dirname(os.path.dirname(V4))
DS1 = os.path.join(WT, "scratch", "ds1")
T3 = os.path.join(WT, "numerics", "projects", "chirality-piping", "execution", "_Coordination", "AgentRuns",
                  "HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION", "instances", "CONTINUATION_2026-09-24",
                  "NUMERICAL_INTEGRITY_T3")
sys.path.insert(0, os.path.join(V4, "r6"))
sys.path.insert(0, DS1)
sys.dont_write_bytecode = True
import emu6_k4hh as E
import models6, models5, r1_adapter
from r6_k4hh import ests

RR7 = os.path.join(T3, "DESIGN_NUMERICS", "REV_5A3_CANDIDATE", "_run_records_r7")
RUNS = [("today", "current", "k4", {}),
        ("R4", "r4", "hybrid", dict(BOUND="Fest", SCOPE="all")),
        ("R5", "r5", "hybrid", dict(BOUND="Uc", SCOPE="body")),
        ("R6", "r6", "hybrid", {})]


def with_sw(sw, fn):
    saved = {k: getattr(E, k) for k in sw}
    for k, v in sw.items():
        setattr(E, k, v)
    try:
        return fn()
    finally:
        for k, v in saved.items():
            setattr(E, k, v)


E.SEED.clear()
E.SEED.update(models6.SEEDS)
M = {m["name"]: m for m in models6.models()}
out = dict(controls={}, controls_differ=[], mutants={}, mutants_differ=[], hhfool={})

# (a)
rc7 = {r["model"]: r["runs"] for r in json.load(open(os.path.join(RR7, "run_controls7.stdout.json")))}
for name, m in M.items():
    rec = {}
    for tag, opt, gate, sw in RUNS:
        def go():
            E.GATE = gate
            r = E.schedule(E.Case(m), opt, {})
            return dict(selected=r["selected"], attempts=[list(a) for a in r["attempts"]])
        rec[tag] = with_sw(sw, go)
        ds1 = rc7[name][tag]
        if rec[tag]["selected"] != ds1["selected"] or rec[tag]["attempts"] != [list(a) for a in ds1["attempts"]]:
            out["controls_differ"].append((name, tag))
    out["controls"][name] = rec
    print(name, {t: rec[t]["selected"] for t in rec}, flush=True)

# (b)
refs = r1_adapter.load_refs(T3)
for cid in ("RF-LARGE-CHAIN-n00100-AX", "RF-LARGE-TREE-n00100-AX"):
    M[cid] = r1_adapter.adapt(cid, refs[cid], refs[cid]["model"])[0]
SUB = ["HH-SLENDER-m40", "HH-FOOL-m40-LOADED", "HH-FOOL-m100-LOADED", "CHARGE-SLENDER", "THETA-STUB-COUPLED",
       "SEEDED-SOFT", "G-PRESC-MEMBER", "SKEW6-K1E-12", "RF-LARGE-CHAIN-n00100-AX", "RF-LARGE-TREE-n00100-AX"]
MUT = [("R6 (r6, hybrid gate)", {}),
       ("M17 drop the charge", dict(CHARGE=False)),
       ("M20 drop the theta check", dict(THETA=False)),
       ("M22 translation and rotation rows without W+ + C (R3)", dict(TRRO="zero")),
       ("M23 drop the g check", dict(G_CHECK=False)),
       ("M24 est in place of the bound", dict(BOUND="est")),
       ("M26 F*est in place of the bound (R4)", dict(BOUND="Fest")),
       ("M28 no shift (Uc_c alone)", dict(BOUND="Uc_block")),
       ("M29 R5's form (one global Uc, body scope)", dict(BOUND="Uc", SCOPE="body"))]
mu7 = json.load(open(os.path.join(RR7, "mutants7.stdout.json")))
REFS = {}
for name in SUB:
    case = E.Case(M[name])
    E.GATE = "absolute"
    seeds = dict(E.SEED)
    E.SEED.clear()
    REFS[name] = E.solve_at(case, 2048, ref=True, estimate=False)
    E.SEED.update(seeds)
E.GATE = "hybrid"
for tag, sw in MUT:
    res = {}
    for name in SUB:
        def go():
            case = E.Case(M[name])
            r = E.schedule(case, "r6", {})
            rec = dict(selected=r["selected"])
            if r["selected"] is not None:
                pub, cl, sc, g5 = E.classify(case, r, "r6")
                w, c = E.honesty(case, pub, cl, REFS[name])
                rec.update(honest=w[0] <= 1.0, relative=c["relative"], g5a=len(g5), corrections=r["cand"]["corrections"])
            return rec
        res[name] = with_sw(sw, go)
        if res[name] != mu7[tag][name]:
            out["mutants_differ"].append((tag, name, res[name], mu7[tag][name]))
    out["mutants"][tag] = res
    print(tag, {k: v["selected"] for k, v in res.items()}, flush=True)

# (c)
for mexp in (40, 80, 100, 200):
    e = ests(E, models5.hh_fool(mexp, False))
    H = next(b for b in e["blocks"] if b["n"] == 4)      # the hidden block (four DOFs)
    V = next(b for b in e["blocks"] if b["n"] == 8)      # the grounded chain
    out["hhfool"][mexp] = dict(log2_est_global=e["log2_est"], log2_est_c_H=H["log2_est_c"], log2_exact_H=H["log2_exact"],
                               miss_H=H["log2_exact"] - H["log2_est_c"], miss_global=H["log2_exact"] - e["log2_est"],
                               log2_exact_V=V["log2_exact"])
    print("HH-FOOL-m%d" % mexp, out["hhfool"][mexp], flush=True)

print("controls differ:", out["controls_differ"])
print("mutants differ:", out["mutants_differ"])
json.dump(out, open(os.path.join(HERE, "r7_spot.json"), "w"), indent=1, default=str)
