"""V4's case definitions (own, from the addendum's prose and ROOT's rulings)."""
from v4emu import model, member, fix, load, spring, station
from fractions import Fraction as Fr

def f2(name="F-2", E=1024.0, lv=2.0**-300):
    m = model(name, [(0,0,0),(2,0,0)])
    member(m, 0, 1, E=E, y=(0,1,0))
    fix(m, 0, (0,), 1.0); fix(m, 0, (1,2,3,4,5)); fix(m, 1, (1,2,3,4,5))
    load(m, 1, 0, lv)
    return m

def f2_spos():
    m = model("F-2-SPOS", [(0,0,0),(2,0,0),(2,2,0)])
    member(m, 0, 1, E=2.0**200, y=(0,1,0))
    member(m, 1, 2, E=1.0, y=(1,0,0))
    fix(m, 0, (0,), 1.0); fix(m, 0, (1,2,3,4,5)); fix(m, 1, (1,2,3,4,5)); fix(m, 2, (2,3,4))
    load(m, 1, 0, 2.0**-60); load(m, 2, 1, 1.0)
    return m

def ptail(free=False):
    # combination (1, 2^-100) of operand A (prescribes 1 at both ux) and operand B
    # (prescribes 2^-1000 at node 1 ux, 0 at node 0 ux): exact terms kept per DOF
    if not free:
        m = model("PTAIL", [(0,0,0),(2,0,0)])
        member(m, 0, 1, E=1024.0, y=(0,1,0))
        fix(m, 0, (0,), [(1.0, 1.0), (2.0**-100, 0.0)])
        fix(m, 0, (1,2,3,4,5)); fix(m, 1, (1,2,3,4,5))
        fix(m, 1, (0,), [(1.0, 1.0), (2.0**-100, 2.0**-1000)])
        return m
    m = model("PTAIL-FREE", [(0,0,0),(2,0,0),(4,0,0)])
    member(m, 0, 1, E=1024.0, y=(0,1,0)); member(m, 1, 2, E=1024.0, y=(0,1,0))
    fix(m, 0, (0,), [(1.0, 1.0), (2.0**-100, 2.0**-1000)])
    fix(m, 0, (1,2,3,4,5)); fix(m, 1, (1,2,3,4,5))
    fix(m, 2, (0,), [(1.0, 1.0), (2.0**-100, 0.0)]); fix(m, 2, (1,2,3,4,5))
    return m

def rigid_unloaded():
    m = model("RIGID-UNLOADED", [(0,0,0),(3,4,0),(0,0,5),(0,0,7)])
    member(m, 0, 1)
    fix(m, 0, (0,), 1.0); fix(m, 1, (0,), 1.0); fix(m, 0, (1,2,3,4,5)); fix(m, 1, (1,2,3,4,5))
    member(m, 2, 3, y=(1,0,0))
    fix(m, 2, range(6))
    load(m, 3, 0, 1e-30)
    return m

def f3_free():
    m = model("F-3-FREE", [(0,0,0),(3,4,0),(6,8,0),(9,12,0)])
    for s in range(3): member(m, s, s+1)
    fix(m, 0, (0,), 1.0); fix(m, 0, (1,2,3,4,5))
    return m

def f3_rot():
    m = model("F-3-ROT", [(0,0,0),(3,4,0),(6,8,0)])
    for s in range(2): member(m, s, s+1)
    fix(m, 0, (0,1,2,3,4)); fix(m, 0, (5,), 1e-3)
    return m

def gs(name, y, rot, loaded=False):
    m = model(name, [(0,0,0),(3,4,0),(6,8,0),(9,12,0)])
    for s in range(3): member(m, s, s+1, y=y)
    if rot:
        fix(m, 0, (0,1,2,3,4)); fix(m, 0, (5,), 1e-3)
    else:
        fix(m, 0, (0,), 1.0); fix(m, 0, (1,2,3,4,5))
    if loaded: load(m, 3, 1, 1.0)
    return m

def demotion(E=2.0**480):
    """V4: a rigidly translated stiff axis-aligned member (exact zero actions at
    every P) and, in the same body, a unit-section member carrying 1 N."""
    m = model("DEMOTION-2^%d" % int(__import__('math').log2(E)), [(0,0,0),(2,0,0),(2,2,0)])
    member(m, 0, 1, E=E, y=(0,1,0))
    member(m, 1, 2, E=1.0, y=(1,0,0))
    fix(m, 0, (0,), 1.0); fix(m, 0, (1,2,3,4,5))
    fix(m, 1, (0,), 1.0); fix(m, 1, (1,2,3,4,5))
    fix(m, 2, (0,), 1.0); fix(m, 2, (2,3,4,5))
    load(m, 2, 1, 1.0)
    return m

def exact_rigid():
    m = model("EXACT-RIGID", [(0,0,0),(2,0,0)])
    member(m, 0, 1, y=(0,1,0))
    fix(m, 0, (0,), 1.0); fix(m, 0, (1,2,3,4,5))
    fix(m, 1, (0,), 1.0); fix(m, 1, (1,2,3,4,5))
    return m

def assembly_sat(Ebig=2.0**480):
    """V4: the stiff member's EA/L and the unit member's 12EI/L^3 share K(node1 ux):
    2^479 + 1.5 is saturated at 128 and 256, so R(1, ux) leaks -1.5 identically."""
    m = demotion(Ebig); m["name"] = "ASSEMBLY-SAT"
    return m

def demotion2(Ebig=2.0**480, Isoft=2.0**400):
    """V4: as demotion(), but the unit member's bending sections are 2^400 so that
    no assembled entry is saturated; every row is exact at 128."""
    from v4emu import model, member, fix, load
    m = model("DEMOTION2", [(0,0,0),(2,0,0),(2,2,0)])
    member(m, 0, 1, E=Ebig, y=(0,1,0))
    member(m, 1, 2, E=1.0, Iy=Isoft, Iz=Isoft, y=(1,0,0))
    fix(m, 0, (0,), 1.0); fix(m, 0, (1,2,3,4,5))
    fix(m, 1, (0,), 1.0); fix(m, 1, (1,2,3,4,5))
    fix(m, 2, (0,), 1.0); fix(m, 2, (2,3,4,5))
    load(m, 2, 1, 1.0)
    return m
