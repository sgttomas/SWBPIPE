"""V4: adversarial search against the conjecture |e_2p(q)| <= 2^8 * 2^-2p * E-hat (force/moment).
Axis-aligned dyadic frames (formation exact at every P, so saturation, not formation noise,
dominates), rigid prescribed translations (big right-hand sides), tiny loads adjacent to the
prescribed nodes (saturated at 128 and 256), wide stiffness contrasts (levers).
Objective: max over rows and P in {128, 256, 512} of |q_P - q*| / (2^-P * E-hat)."""
import json, math, random, sys, time
from fractions import Fraction as Fr
from v4emu import *

def gen(rng, idx):
    # nodes on a small dyadic grid in the x-y plane (plus optional z offsets)
    nn = rng.randint(3, 6)
    pts = set()
    pts.add((0.0, 0.0, 0.0))
    while len(pts) < nn:
        base = rng.choice(sorted(pts))
        ax = rng.randrange(3 if rng.random() < 0.3 else 2)
        step = float(2 ** rng.randint(-3, 5)) * rng.choice((1, -1))
        p = list(base); p[ax] += step
        pts.add(tuple(p))
    nodes = sorted(pts)
    m = model("ADV-%d" % idx, nodes)
    # members: connect along axes (a spanning tree plus extras)
    def axis_pair(a, b):
        return sum(1 for k in range(3) if a[k] != b[k]) == 1
    pairs = [(i, j) for i in range(len(nodes)) for j in range(i + 1, len(nodes)) if axis_pair(nodes[i], nodes[j])]
    rng.shuffle(pairs)
    parent = list(range(len(nodes)))
    def f(x):
        while parent[x] != x: x = parent[x]
        return x
    chosen = []
    for i, j in pairs:
        if f(i) != f(j):
            parent[f(i)] = f(j); chosen.append((i, j))
        elif rng.random() < 0.3:
            chosen.append((i, j))
    if len({f(i) for i in range(len(nodes))}) != 1:
        return None
    for i, j in chosen:
        d = [nodes[j][k] - nodes[i][k] for k in range(3)]
        y = (0.0, 0.0, 1.0) if d[2] == 0 else (1.0, 0.0, 0.0)
        e = 2.0 ** rng.randint(-10, 60)
        sec = dict(E=e, G=e / 2, A=2.0 ** rng.randint(-10, 10), Iy=2.0 ** rng.randint(-20, 10),
                   Iz=2.0 ** rng.randint(-20, 10), J=2.0 ** rng.randint(-20, 10))
        member(m, i, j, y=y, **sec)
    # prescribed nodes: a rigid translation
    npres = rng.randint(1, max(1, len(nodes) - 1))
    pres = rng.sample(range(len(nodes)), npres)
    t = [float(rng.choice((0, 1, 2 ** rng.randint(-5, 20)))) for _ in range(3)]
    if all(c == 0 for c in t): t[1] = 1.0
    for nd in pres:
        for c in range(3): fix(m, nd, (c,), t[c])
        fix(m, nd, (3, 4, 5))
    # tiny loads at free DOFs of nodes adjacent to prescribed nodes, plus random tiny loads
    adj = {j if i in pres else i for i, j in chosen if (i in pres) != (j in pres)}
    free_nodes = [x for x in range(len(nodes)) if x not in pres]
    if not free_nodes: return None
    for _ in range(rng.randint(1, 4)):
        nd = rng.choice(sorted(adj) or free_nodes)
        c = rng.randrange(6)
        load(m, nd, c, rng.choice((1, -1)) * 2.0 ** -rng.randint(150, 700))
    return m

def score(m, precs=(128, 256, 512)):
    tr = exact_state(m)
    lay = layout(m); nb = bodies(m); ext = extents(m)
    best = (0.0, None, None)
    for P in precs:
        try:
            st = solve_at(m, P)
        except Stop:
            continue
        E, _ = formation_scale(m, st)
        for idx, (key, kind, inp) in enumerate(lay):
            if kind not in ("fo", "mo") or inp:
                continue
            b = row_body(m, key, nb)
            efo, emo = e_hat(E, b, ext[b])
            Eh = efo if kind == "fo" else emo
            dt = abs(st["vals"][idx] - tr["vals"][idx])
            if dt == 0: continue
            r = math.inf if Eh == 0 else float(dt / (Fr(2) ** -P * Eh))
            if r > best[0]: best = (r, P, key)
    return best

if __name__ == "__main__":
    seed = int(sys.argv[1]); count = int(sys.argv[2]); budget = float(sys.argv[3]) if len(sys.argv) > 3 else 1e9
    rng = random.Random(seed)
    t0 = time.time(); results = []; worst = (0.0,)
    k = 0; ran = 0
    while ran < count and time.time() - t0 < budget:
        k += 1
        m = gen(rng, k)
        if m is None: continue
        try:
            s = score(m)
        except (Stop, ZeroDivisionError, StopIteration) as e:
            continue
        ran += 1
        results.append((m["name"], s[0], s[1], str(s[2])))
        if s[0] > worst[0]:
            worst = s + (m["name"],)
            print("new worst", worst, flush=True)
    print("ran", ran, "models; worst ratio", worst)
    json.dump(dict(seed=seed, ran=ran, worst=[str(x) for x in worst], results=results), open("adversary_%d.json" % seed, "w"), indent=0)
