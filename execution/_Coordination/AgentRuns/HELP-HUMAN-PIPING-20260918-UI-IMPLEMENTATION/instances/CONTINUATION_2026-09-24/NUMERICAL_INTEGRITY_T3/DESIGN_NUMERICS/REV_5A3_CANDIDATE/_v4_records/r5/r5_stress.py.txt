#!/usr/bin/env python3
"""V4, delta check at R5, checks 1 and 3(a)(b): Lemma D stressed at low precision,
where the factor's backward error is large and t = U gamma_m N_L approaches 1.

V4's own implementation (not DS1's): K4's profile LDL^T loop (factor.rs, row by row
with the work vector, every operation rounded to nearest at P), then item 7 of R5's
section 4.1.6.3 with directed rounding in V4's own ru/rd (nearest, then one ulp in the
required direction, unconditionally, as item 7 permits), and gamma_m with m = 2n + 2.

Matrices (random, seed pinned): radix-equilibrated SPD with a profile, of four kinds:
  gram:    B B^T + eps I, B integer in [-3, 3] with a random band;
  lap:     a grounded spring network with a random weak spring (an M-matrix);
  lever:   a mixed-sign 'sum spring' block (w = (1,-1,-1,1) soft) in a chain;
  near:    Q diag(1, ..., 2^-k) Q^T with Q from integer Householder reflections.
For every factor with all pivots positive: exact ||K~^-1||_1 (Fractions); U, N_L, t, Uc.
Checks: Uc >= exact whenever Uc exists (Lemma D); the largest t with Uc existing; how
often U alone is below exact; the smallest Uc/exact. The pivot screen is recorded but
not required (Lemma D needs only d_i > 0).
Usage: python3 r5_stress.py [count] [m mode: 2n+2 (R5's), n+1, 1].
Output: r5_stress.json (m = 2n + 2), r5_stress_mnp1.json, r5_stress_m1.json."""
import json, os, random, sys
from fractions import Fraction as Fr

HERE = os.path.dirname(os.path.abspath(__file__))
M_MODE = sys.argv[2] if len(sys.argv) > 2 else "2n+2"   # sensitivity: "n+1" (V4's derived gamma_(t+1), t <= n - 1) or "1"


def flog2(a):
    n, d = a.numerator, a.denominator
    e = n.bit_length() - d.bit_length()
    if e >= 0:
        if n < (d << e):
            e -= 1
    else:
        if (n << -e) < d:
            e -= 1
    return e


def rn(x, p):
    x = Fr(x)
    if x == 0:
        return Fr(0)
    s = -1 if x < 0 else 1
    a = abs(x)
    e = flog2(a)
    sh = p - 1 - e
    n, d = a.numerator, a.denominator
    num, den = (n << sh, d) if sh >= 0 else (n, d << (-sh))
    q, r = divmod(num, den)
    if 2 * r > den or (2 * r == den and (q & 1)):
        q += 1
    return s * (Fr(q, 1 << sh) if sh >= 0 else Fr(q << (-sh)))


def up(x, p):
    """Round up (x > 0): nearest, then one ulp up, unconditionally."""
    y = rn(x, p)
    return y + Fr(2) ** (flog2(y) - p + 1)


def down(x, p):
    """Round down (x > 0): nearest, then one ulp down, unconditionally."""
    y = rn(x, p)
    e = flog2(y)
    step = Fr(2) ** (e - p) if y == Fr(2) ** e else Fr(2) ** (e - p + 1)
    return y - step


def equilibrate(K):
    n = len(K)
    s = [-(flog2(K[i][i]) // 2) for i in range(n)]
    return [[K[i][j] * Fr(2) ** (s[i] + s[j]) for j in range(n)] for i in range(n)]


def factor(Kt, p):
    n = len(Kt)
    first = [min(j for j in range(i + 1) if Kt[i][j] != 0) for i in range(n)]
    L = [[Fr(0)] * n for _ in range(n)]
    D = [Fr(0)] * n
    work = [Fr(0)] * n
    screen = True
    for i in range(n):
        for j in range(first[i], i):
            s = Kt[i][j]
            for kk in range(max(first[i], first[j]), j):
                s = rn(s - rn(work[kk] * L[j][kk], p), p)
            work[j] = s
            L[i][j] = rn(s / D[j], p)
        piv = Kt[i][i]
        canc = abs(piv)
        for kk in range(first[i], i):
            t = rn(work[kk] * L[i][kk], p)
            piv = rn(piv - t, p)
            canc = rn(canc + abs(t), p)
        m = 2 * (i - first[i]) + 2
        if not (piv * (2 ** p - m) - 64 * m * canc > 0):
            screen = False
        if piv <= 0:
            return None
        D[i] = piv
    return L, D, first, screen


def uc(L, D, first, p):
    n = len(D)
    a = [Fr(1)] * n
    for i in range(n):
        acc = Fr(1)
        for j in range(first[i], i):
            if L[i][j]:
                acc = up(acc + up(abs(L[i][j]) * a[j], p), p)
        a[i] = acc
    c = [up(a[i] / D[i], p) for i in range(n)]
    for i in reversed(range(n)):
        for j in range(first[i], i):
            if L[i][j]:
                c[j] = up(c[j] + up(abs(L[i][j]) * c[i], p), p)
    U = max(c)
    at = [Fr(1)] * n
    for i in range(n):
        for j in range(first[i], i):
            if L[i][j]:
                at[j] = up(at[j] + abs(L[i][j]), p)
    bt = [up(D[i] * at[i], p) for i in range(n)]
    ct = list(bt)
    for i in range(n):
        for j in range(first[i], i):
            if L[i][j]:
                ct[i] = up(ct[i] + up(abs(L[i][j]) * bt[j], p), p)
    NL = max(ct)
    m = {'2n+2': 2 * n + 2, 'n+1': n + 1, '1': 1}[M_MODE]
    gam = up(Fr(m) / (Fr(2) ** p - m), p)
    t = up(up(U * gam, p) * NL, p)
    if t >= 1:
        return U, None, t
    return U, up(U / down(1 - t, p), p), t


def exact_norm(A):
    n = len(A)
    M = [list(r) + [Fr(int(i == j)) for j in range(n)] for i, r in enumerate(A)]
    for c in range(n):
        piv = next(r for r in range(c, n) if M[r][c] != 0)
        M[c], M[piv] = M[piv], M[c]
        pv = M[c][c]
        M[c] = [v / pv for v in M[c]]
        for r in range(n):
            if r != c and M[r][c] != 0:
                f = M[r][c]
                M[r] = [x - f * y for x, y in zip(M[r], M[c])]
    return max(sum(abs(M[i][n + j]) for i in range(n)) for j in range(n))


def gen(rng, kind):
    n = rng.randint(3, 10)
    if kind == "gram":
        bw = rng.randint(1, n - 1)
        B = [[Fr(rng.randint(-3, 3)) if abs(i - j) <= bw else Fr(0) for j in range(n)] for i in range(n)]
        eps = Fr(1, 2 ** rng.randint(0, 30))
        K = [[sum(B[i][k] * B[j][k] for k in range(n)) + (eps if i == j else 0) for j in range(n)] for i in range(n)]
    elif kind == "lap":
        K = [[Fr(0)] * n for _ in range(n)]
        for i in range(n - 1):
            k = Fr(rng.randint(1, 5)) if rng.random() > 0.2 else Fr(1, 2 ** rng.randint(5, 40))
            K[i][i] += k; K[i + 1][i + 1] += k; K[i][i + 1] -= k; K[i + 1][i] -= k
            if rng.random() < 0.3 and i + 2 < n:
                k2 = Fr(rng.randint(1, 3))
                K[i][i] += k2; K[i + 2][i + 2] += k2; K[i][i + 2] -= k2; K[i + 2][i] -= k2
        K[0][0] += Fr(1, 2 ** rng.randint(0, 30))
    elif kind == "lever":
        n = max(n, 5)
        K = [[Fr(0)] * n for _ in range(n)]
        for i in range(n - 1):
            k = Fr(rng.randint(1, 4))
            sgn = rng.choice((1, -1))
            K[i][i] += k; K[i + 1][i + 1] += k; K[i][i + 1] += sgn * k; K[i + 1][i] += sgn * k
        d = Fr(1, 2 ** rng.randint(0, 40))
        for i in range(n):
            K[i][i] += d * rng.randint(1, 3)
    else:
        k = rng.randint(2, 40)
        lam = [Fr(1)] * (n - 1) + [Fr(1, 2 ** k)]
        K = [[lam[i] if i == j else Fr(0) for j in range(n)] for i in range(n)]
        for _ in range(2):
            v = [Fr(rng.randint(-2, 2)) for _ in range(n)]
            if all(x == 0 for x in v):
                continue
            vv = sum(x * x for x in v)
            H = [[Fr(int(i == j)) - 2 * v[i] * v[j] / vv for j in range(n)] for i in range(n)]
            K = [[sum(H[i][a] * sum(K[a][b] * H[j][b] for b in range(n)) for a in range(n)) for j in range(n)] for i in range(n)]
    return K


if __name__ == "__main__":
    rng = random.Random(20260929)
    count = int(sys.argv[1]) if len(sys.argv) > 1 else 400
    stats = dict(factors=0, uc_exists=0, uc_missing=0, violations=[], U_below_exact=0,
                 min_uc_over_exact=None, max_t_with_uc=0.0, screen_pass=0)
    examples = []
    for it in range(count):
        kind = ("gram", "lap", "lever", "near")[it % 4]
        K = gen(rng, kind)
        try:
            Kt = equilibrate(K)
        except Exception:
            continue
        if any(Kt[i][i] <= 0 for i in range(len(Kt))):
            continue
        ex = exact_norm(Kt)
        for P in (10, 12, 16, 20, 24, 32):
            f = factor(Kt, P)
            if f is None:
                continue
            L, D, first, screen = f
            stats["factors"] += 1
            stats["screen_pass"] += int(screen)
            U, Uc, t = uc(L, D, first, P)
            if U < ex:
                stats["U_below_exact"] += 1
            if Uc is None:
                stats["uc_missing"] += 1
                continue
            stats["uc_exists"] += 1
            r = Uc / ex
            if Uc < ex:
                stats["violations"].append(dict(it=it, kind=kind, P=P, n=len(Kt), Uc_over_exact=float(r), t=float(t)))
            if stats["min_uc_over_exact"] is None or r < stats["min_uc_over_exact"][0]:
                stats["min_uc_over_exact"] = (r, dict(it=it, kind=kind, P=P, t=float(t), U_over_exact=float(U / ex)))
            if t > stats["max_t_with_uc"]:
                stats["max_t_with_uc"] = float(t)
                examples.append(dict(it=it, kind=kind, P=P, n=len(Kt), t=float(t), U_over_exact=float(U / ex), Uc_over_exact=float(r)))
    stats["min_uc_over_exact"] = (float(stats["min_uc_over_exact"][0]), stats["min_uc_over_exact"][1]) if stats["min_uc_over_exact"] else None
    stats["largest_t_examples"] = examples[-5:]
    print(json.dumps(stats, indent=1))
    stats["m_mode"] = M_MODE
    json.dump(stats, open(os.path.join(HERE, "r5_stress%s.json" % ("" if M_MODE == "2n+2" else "_m" + M_MODE.replace("+", "p"))), "w"), indent=1)
