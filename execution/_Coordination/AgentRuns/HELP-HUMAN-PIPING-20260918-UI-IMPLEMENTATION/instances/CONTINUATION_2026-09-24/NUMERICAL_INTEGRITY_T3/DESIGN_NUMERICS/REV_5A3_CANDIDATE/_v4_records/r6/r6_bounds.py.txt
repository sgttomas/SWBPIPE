#!/usr/bin/env python3
"""V4, delta check at R6, check 4 (availability, in V4's own implementation): the
certified bounds of R6's section 4.1.6.3 item 7 on R1's RF-LARGE references at the
P = 256 verification state, computed by V4's code, not emu6's:
  - K~_P from emu5's assembly at P = 256 (the same formation as emu6), radix-equilibrated,
    in K4's order (V4's port of factor.rs's reverse Cuthill-McKee, r5_large.rcm);
  - K4's factor loop at P bits (V4's rounding, r5_stress.rn);
  - est: K4's `condition` estimate, ported line by line (r4_hh), on that factor;
  - Uc = U/(1 - U gamma_m N_L) and S = ceil(sqrt n)/sigma' with sigma = 1/(2 est) rounded
    downward, sigma' = sigma - (gamma_m N'_L + delta), at most three shifts (halving),
    every step with V4's nearest-then-one-ulp directed rounding; B = min(Uc, S);
  - the exact norm: rationals up to 60 free DOFs; above, binary64 column solves on the
    binary64 factor (r5_large.exact_norm), an approximation labelled as such.
Compared with DS1's large6 bound summaries (the same state).
Usage: python3 r6_bounds.py [sizes, default 10,100]
Output: r6_bounds_<sizes>.json."""
import json, math, os, sys, time
from fractions import Fraction as Fr

HERE = os.path.dirname(os.path.abspath(__file__))
V4 = os.path.dirname(HERE)
sys.path.insert(0, os.path.join(V4, "r5"))
sys.path.insert(0, os.path.join(V4, "r4"))
sys.dont_write_bytecode = True
import r5_large
from r5_large import r1_adapter, Case, assemble, floor_log2, rcm, T3, ldl as ldl64, exact_norm as exact64
from r5_stress import rn, up, down
from r6_lemmaE import ceil_sqrt, exact_inverse

P = 256


def scaled_rcm(case):
    K, _, pattern, _, _ = assemble(case.m, P)
    free = case.free
    pos = {g: a for a, g in enumerate(free)}
    sc = [-(floor_log2(abs(K[(g, g)])) // 2) for g in free]
    adj = [[] for _ in free]
    for (r, c) in pattern:
        if r != c and r in pos and c in pos:
            adj[pos[r]].append(pos[c])
    order = rcm(adj)
    n = len(free)
    rank = [0] * n
    for k, a in enumerate(order):
        rank[a] = k
    first = [min([rank[b] for b in adj[a]] + [i]) for i, a in enumerate(order)]
    rows = []
    for i, a in enumerate(order):
        row = {}
        for b in adj[a] + [a]:
            j = rank[b]
            if j <= i:
                v = K.get((free[a], free[b]), Fr(0))
                if v:
                    row[j] = v * Fr(2) ** (sc[a] + sc[b])
        rows.append(row)
    return rows, first, n


def factor(rows, first, n, shift=None):
    """K4's loop at P; shift = sigma subtracts sigma from the diagonal (rounded to
    nearest) and replaces the pivot screen by d' > 0. Returns (L, D, dtilde) or None."""
    L = [dict() for _ in range(n)]
    D = [Fr(0)] * n
    work = [Fr(0)] * n
    dt = []
    for i in range(n):
        for j in range(first[i], i):
            s = rows[i].get(j, Fr(0))
            for kk in range(max(first[i], first[j]), j):
                l = L[j].get(kk)
                if l and work[kk]:
                    s = rn(s - rn(work[kk] * l, P), P)
            work[j] = s
            if s:
                L[i][j] = rn(s / D[j], P)
        piv = rows[i].get(i, Fr(0))
        if shift is not None:
            piv = rn(piv - shift, P)
            dt.append(piv)
        for kk in range(first[i], i):
            l = L[i].get(kk)
            if l and work[kk]:
                piv = rn(piv - rn(work[kk] * l, P), P)
        if piv <= 0:
            return None
        D[i] = piv
    return L, D, dt


def solve(L, D, n, b):
    x = list(b)
    for i in range(n):
        for j, l in L[i].items():
            if x[j]:
                x[i] = rn(x[i] - rn(l * x[j], P), P)
    for i in range(n):
        x[i] = rn(x[i] / D[i], P)
    for i in reversed(range(n)):
        v = x[i]
        if v:
            for j, l in L[i].items():
                x[j] = rn(x[j] - rn(l * v, P), P)
    return x


def hh_est(L, D, n):
    """factor.rs `condition`'s estimate (r4_hh.hager_higham), on this factor."""
    x = [rn(Fr(1, n), P)] * n
    est = Fr(0)
    prev = n
    for _ in range(5):
        y = solve(L, D, n, x)
        cur = rn(sum(abs(v) for v in y), P)
        if cur <= est and est != 0:
            break
        est = cur
        z = solve(L, D, n, [Fr(1) if v >= 0 else Fr(-1) for v in y])
        j = 0
        for i in range(1, n):
            if abs(z[i]) >= abs(z[j]):
                j = i
        dot = rn(sum(z[i] * x[i] for i in range(n)), P)
        if abs(z[j]) <= dot or j == prev:
            break
        prev = j
        x = [Fr(0)] * n
        x[j] = Fr(1)
    alt = [(Fr(1) if n == 1 else rn(Fr(n - 1 + i, n - 1), P)) * (1 if i % 2 == 0 else -1) for i in range(n)]
    y = solve(L, D, n, alt)
    return max(est, rn(rn(sum(abs(v) for v in y), P) * 2 / (3 * n), P))


def passes(L, D, n):
    a = [Fr(1)] * n
    for i in range(n):
        acc = Fr(1)
        for j, l in L[i].items():
            acc = up(acc + up(abs(l) * a[j], P), P)
        a[i] = acc
    c = [up(a[i] / D[i], P) for i in range(n)]
    for i in reversed(range(n)):
        for j, l in L[i].items():
            c[j] = up(c[j] + up(abs(l) * c[i], P), P)
    at = [Fr(1)] * n
    for i in range(n):
        for j, l in L[i].items():
            at[j] = up(at[j] + abs(l), P)
    bt = [up(D[i] * at[i], P) for i in range(n)]
    ct = list(bt)
    for i in range(n):
        for j, l in L[i].items():
            ct[i] = up(ct[i] + up(abs(l) * bt[j], P), P)
    return max(c), max(ct)


def lg(x):
    return None if x is None else (math.log2(x.numerator) - math.log2(x.denominator))


def run(cid, c):
    m, index = r1_adapter.adapt(cid, c, c["model"])
    case = Case(m)
    rows, first, n = scaled_rcm(case)
    t0 = time.time()
    L, D, _ = factor(rows, first, n)
    est = hh_est(L, D, n)
    U, NL = passes(L, D, n)
    m_ = 2 * n + 2
    gam = up(Fr(m_) / (Fr(2) ** P - m_), P)
    t = up(up(U * gam, P) * NL, P)
    Uc = up(U / down(1 - t, P), P) if t < 1 else None
    rec = dict(case=cid, free=n, log2_est=lg(est), log2_U=lg(U), log2_Uc=lg(Uc), log2_t=lg(t))
    S = None
    tries = 0
    if Uc is None or Uc > 2 * ceil_sqrt(n) * est:
        sg = down(1 / (2 * est), P)
        while tries < 3:
            tries += 1
            f = factor(rows, first, n, shift=sg)
            if f is None:
                sg = sg / 2
                continue
            L2, D2, dt = f
            _, NL2 = passes(L2, D2, n)
            delta = Fr(2) ** (1 - P) * max(abs(x) for x in dt)
            e = up(up(gam * NL2, P) + delta, P)
            if sg - e > 0:
                sp = down(sg - e, P)
                S = up(Fr(ceil_sqrt(n)) / sp, P)
            break
    rec.update(tries=tries, log2_S=lg(S))
    B = min([x for x in (Uc, S) if x is not None]) if (Uc or S) else None
    rec["log2_B"] = lg(B)
    rec["bound_seconds"] = round(time.time() - t0, 1)
    if n <= 60:
        Kt = [[Fr(0)] * n for _ in range(n)]
        for i, r in enumerate(rows):
            for j, v in r.items():
                Kt[i][j] = v
                Kt[j][i] = v
        inv = exact_inverse(Kt)
        ex = max(sum(abs(inv[i][j]) for i in range(n)) for j in range(n))
        rec["log2_exact"] = lg(ex)
        rec["exact_kind"] = "rational"
        rec["B_ge_exact"] = B >= ex
    else:
        rf = [{j: float(v) for j, v in r.items()} for r in rows]
        Lf, Df = ldl64(rf, first)
        rec["log2_exact"] = math.log2(exact64(Lf, Df, first))
        rec["exact_kind"] = "binary64 column solves"
    rec["log2_B_over_exact"] = rec["log2_B"] - rec["log2_exact"]
    return rec


if __name__ == "__main__":
    sizes = [int(x) for x in (sys.argv[1] if len(sys.argv) > 1 else "10,100").split(",")]
    cases = r1_adapter.load_refs(T3)
    out = []
    for cid, c in cases.items():
        if c["family"] != "RF-LARGE" or "generator" in c["model"] or len(c["model"]["members"]) not in sizes:
            continue
        rec = run(cid, c)
        out.append(rec)
        print(json.dumps(rec), flush=True)
    json.dump(out, open(os.path.join(HERE, "r6_bounds_%s.json" % "_".join(map(str, sizes))), "w"), indent=1)
