from v4emu import *
from fractions import Fraction as Fr
def f2(name="F-2", E=1024.0, lv=2.0**-300):
    m = model(name, [(0,0,0),(2,0,0)])
    member(m, 0, 1, E=E, y=(0,1,0))
    fix(m, 0, (0,), 1.0); fix(m, 0, (1,2,3,4,5)); fix(m, 1, (1,2,3,4,5))
    load(m, 1, 0, lv)
    return m
import time
t=time.time()
out, tr = run(f2(), rules=("current","ii","iii"))
print("time", time.time()-t)
# show N row
lay = layout(f2())
for rule in ("current","iii"):
    rows = out[rule].get("_rows")
    if rows:
        for (key,kind,qp,cls,b),qt in zip(rows, tr["vals"]):
            if key[0]=="end" and key[3]==0:
                print(rule, key, float(qp), cls, None if b is None else float(b), "truth", float(qt))
