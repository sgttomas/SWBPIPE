#!/usr/bin/env python3
"""V4, delta check at R6: does emu6's condition estimate index its vectors as K4 does?

K4's `condition` (factor.rs) forms x, the sign vector, the argmax (last index on ties)
and the alternating vector (-1)^i (n-1+i)/(n-1) over FREE POSITIONS (ascending global
DOF); only `solve_scaled` maps them into the RCM order. emu6 sets case.free to the RCM
order and runs the same loops over that list, so its alternating vector and its tie-break
are indexed by RCM rank, not by free position.

This script writes emu6_k4hh.py, a copy of DS1's emu6.py (read, not modified) in which
only those two loops are re-indexed to free-position order, as K4's are, and compares:
  (a) est (global) and est_c per block against the exact per-block norm at P = 256, for
      HH-FOOL-m40, -m100 (unloaded and loaded) and HH-SLENDER-m40 (models5);
  (b) R6's selection and M24's (est in place of the bound) on HH-SLENDER-m40, and R6's on
      the HH-FOOL frames, under emu6 and under the K4-indexed copy.
Output: r6_k4hh.json."""
import importlib, json, os, sys
from fractions import Fraction as Fr

HERE = os.path.dirname(os.path.abspath(__file__))
WT = os.path.dirname(os.path.dirname(os.path.dirname(HERE)))
DS1 = os.path.join(WT, "scratch", "ds1")
sys.path.insert(0, HERE)
sys.path.insert(0, DS1)
sys.dont_write_bytecode = True

src = open(os.path.join(DS1, "emu6.py")).read()
old_arg = """            j = 0
            for i in range(1, n):
                if abs(z[i]) >= abs(z[j]):
                    j = i"""
new_arg = """            # V4 patch: K4's argmax runs over free positions (ascending global DOF)
            posorder = sorted(range(n), key=lambda a_: free[a_])
            j = posorder[0]
            for i in posorder[1:]:
                if abs(z[i]) >= abs(z[j]):
                    j = i"""
old_alt = """        alt = []
        for i in range(n):
            mag = one if n == 1 else rp(Fr(n - 1 + i) / Fr(n - 1), p)
            alt.append(mag if i % 2 == 0 else -mag)"""
new_alt = """        # V4 patch: K4's alternating vector is indexed by free position (ascending global DOF)
        posorder = sorted(range(n), key=lambda a_: free[a_])
        alt = [Fr(0)] * n
        for k_, i in enumerate(posorder):
            mag = one if n == 1 else rp(Fr(n - 1 + k_) / Fr(n - 1), p)
            alt[i] = mag if k_ % 2 == 0 else -mag"""
assert src.count(old_arg) == 1 and src.count(old_alt) == 1
open(os.path.join(HERE, "emu6_k4hh.py"), "w").write(src.replace(old_arg, new_arg).replace(old_alt, new_alt))

import emu6
import emu6_k4hh
import models5
from r6_lemmaE import exact_inverse


def lg(x):
    import math
    return None if x is None else (math.log2(x.numerator) - math.log2(x.denominator))


def block_exact(mod, case, P):
    K = mod.assemble(case.m, P)[0]
    free = case.free
    sc = [-(mod.floor_log2(abs(K[(g, g)])) // 2) for g in free]
    rank = {g: a for a, g in enumerate(free)}
    nbr = [[rank[c] for c in case.adj.get(g, []) if c in rank] for g in free]
    blocks, blk = mod.free_blocks(len(free), nbr)
    out = []
    for pl in blocks:
        A = [[K.get((free[a], free[b]), Fr(0)) * Fr(2) ** (sc[a] + sc[b]) for b in pl] for a in pl]
        inv = exact_inverse(A)
        out.append(max(sum(abs(inv[i][j]) for i in range(len(pl))) for j in range(len(pl))))
    return out


def ests(mod, m, P=256):
    case = mod.Case(m)
    st = mod.solve_at(case, P)
    ex = block_exact(mod, case, P)
    bl = st["uc"]["blocks"]
    return dict(log2_est=lg(st["est_hh"]),
                blocks=[dict(n=b["n"], log2_est_c=lg(b["est"]), log2_exact=lg(e), log2_Uc=lg(b["Uc"]),
                             log2_S=lg(b.get("S")), log2_B=lg(b.get("B"))) for b, e in zip(bl, ex)])


def sel(mod, m, **sw):
    saved = {k: getattr(mod, k) for k in sw}
    for k, v in sw.items():
        setattr(mod, k, v)
    try:
        case = mod.Case(m)
        mod.GATE = "hybrid"
        r = mod.schedule(case, "r6", {})
        return dict(selected=r["selected"], attempts=[list(a) for a in r["attempts"]])
    finally:
        for k, v in saved.items():
            setattr(mod, k, v)


if __name__ == "__main__":
    models = [models5.hh_fool(40, False), models5.hh_fool(100, False), models5.hh_fool(40, True),
              models5.hh_fool(100, True), models5.hh_slender(40)]
    out = {}
    for m in models:
        rec = {}
        for tag, mod in (("emu6", emu6), ("k4_indexed", emu6_k4hh)):
            rec[tag] = dict(ests=ests(mod, m), R6=sel(mod, m))
            if m["name"].startswith("HH-SLENDER"):
                rec[tag]["M24"] = sel(mod, m, BOUND="est")
        out[m["name"]] = rec
        print(m["name"], json.dumps(rec, default=str), flush=True)
    json.dump(out, open(os.path.join(HERE, "r6_k4hh.json"), "w"), indent=1, default=str)
