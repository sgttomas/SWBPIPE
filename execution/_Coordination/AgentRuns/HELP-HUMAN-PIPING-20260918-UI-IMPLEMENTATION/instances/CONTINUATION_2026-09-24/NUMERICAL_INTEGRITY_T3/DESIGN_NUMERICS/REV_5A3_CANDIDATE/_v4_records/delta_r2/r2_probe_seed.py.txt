"""V4 delta check: (1) K4's Rust probe set (run_345, 5 loads x 1,3 members x 2 y_ref) under K4's
gate and R2's hybrid gate; (2) SEEDED-COMMON with W's residual reused (R2's text) and recomputed
(DS1's emu2); (3) TWO-SPAN's corrections under the coalesced, hybrid and pure gates."""
import os, sys, json, time
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from fractions import Fraction as Fr
from v4emu import model, member, fix, load, spring, station, run, solve_at, Stop
import v4emu_r2 as R2
from gate_probe_nsec import S as NSEC
out = {}
def run_345(y, members, tip):
    m = model("PROBE-y%s-%dm-%s" % ("".join("%g" % c for c in y), members, tip[0]), [(3.0*s, 4.0*s, 0.0) for s in range(members+1)])
    for s in range(members): member(m, s, s+1, y=y, **NSEC)
    fix(m, 0, range(6))
    for c, v in enumerate(tip[1]):
        if v: load(m, members, c, v)
    return m
loads = [("general", [1000.0, -500.0, 2000.0, 100.0, 200.0, 300.0]), ("inplane", [-400.0, 300.0, 0, 0, 0, 0]),
         ("outofplane", [0, 0, 1000.0, 0, 0, 0]), ("axial", [600.0, 800.0, 0, 0, 0, 0]), ("torque", [0, 0, 0, 60.0, 80.0, 0])]
out["probe"] = {}
for y in ((3.0,4.0,5.0), (0.0,0.0,1.0)):
    for members in (1, 3):
        for L in loads:
            m = run_345(y, members, L); t = time.time()
            k4 = R2.run_r2(m, gate="k4", use_W=True, truth=False)
            hy = R2.run_r2(m, gate="hybrid", use_W=True, truth=True)
            out["probe"][m["name"]] = dict(k4_gate=k4["selected"], k4_attempts=k4["attempts"], hybrid=hy["selected"], hybrid_honesty=hy.get("honesty"), g5a=hy.get("g5a_failures"), hybrid_attempts=hy["attempts"])
            print(m["name"], "| coalesced:", k4["selected"], [a[1][:40] for a in k4["attempts"]], "| hybrid:", hy["selected"], hy.get("honesty"), hy.get("g5a_failures"), "(%.1fs)" % (time.time()-t), flush=True)
# (2) SEEDED-COMMON: N05 with 2^-110 m added to node 1 uy after the gate at every precision
def n05():
    m = model("SEEDED-COMMON", [(0,0,0),(2,0,0)]); member(m, 0, 1, **NSEC)
    fix(m, 0, (0,1,2,4,5)); spring(m, 0, 3, 1e-4); load(m, 1, 3, 1e-8); station(m, 0, 0.5)
    return m
R2.SEED["SEEDED-COMMON"] = [(6 + 1, Fr(1, 2**110))]
out["seeded"] = {}
for mode in ("reuse", "recompute"):
    R2.W_RESIDUAL = mode
    r = R2.run_r2(n05(), gate="hybrid", use_W=True)
    out["seeded"][mode] = r
    print("SEEDED-COMMON W residual =", mode, "->", r["selected"], r["attempts"], r.get("honesty"), flush=True)
r = R2.run_r2(n05(), gate="hybrid", use_W=False); out["seeded"]["no_W"] = r
print("SEEDED-COMMON without W ->", r["selected"], r.get("honesty"), flush=True)
R2.W_RESIDUAL = "reuse"; R2.SEED.clear()
# (3) TWO-SPAN corrections under each gate
m = model("TWO-SPAN", [(0,0,0),(1,0,0),(2 + 2.0**-40, 0, 0)]); member(m, 0, 1, **NSEC); member(m, 1, 2, **NSEC)
fix(m, 0, range(6)); fix(m, 2, range(6)); load(m, 1, 5, 1.0)
out["two_span"] = {}
for gate in ("k4", "hybrid", "pure"):
    st = R2.solve_r2(m, 128, gate)
    out["two_span"][gate] = st["info"]
    print("TWO-SPAN at 128, gate", gate, st["info"], flush=True)
json.dump(out, open("delta_r2/r2_probe_seed.json", "w"), indent=1, default=str)
