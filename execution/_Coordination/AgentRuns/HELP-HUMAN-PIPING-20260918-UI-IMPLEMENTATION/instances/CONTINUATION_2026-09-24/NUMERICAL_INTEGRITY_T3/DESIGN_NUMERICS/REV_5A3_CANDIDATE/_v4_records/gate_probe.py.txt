import json, time
from v4emu import run, solve_at, Stop
import cases
probes = [cases.gs("GS-ROT-y345", (3.0,4.0,5.0), True), cases.gs("GS-TRANS-y345", (3.0,4.0,5.0), False),
          cases.gs("GS-ROT-y001", (0.0,0.0,1.0), True), cases.gs("GS-ROT-y345-LOADED", (3.0,4.0,5.0), True, loaded=True)]
out = {}
for m in probes:
    for gate in ("k4", "abs"):
        t = time.time()
        o, tr = run(m, rules=("current","iii"), gate=gate, verbose=False)
        out.setdefault(m["name"], {})[gate] = {r: {k: v for k, v in rec.items() if k in ("selected","attempts","honesty")} for r, rec in o.items()}
        print(m["name"], gate, {r: (v["selected"], v["attempts"], v.get("honesty")) for r, v in out[m["name"]][gate].items()}, "(%.1fs)" % (time.time()-t), flush=True)
# the residual detail at 128 under K4's gate
m = probes[0]
try:
    solve_at(m, 128, "k4")
except Stop as s:
    print("GS-ROT-y345 at 128, K4 gate:", s)
json.dump(out, open("gate_probe.json","w"), indent=1, default=str)
