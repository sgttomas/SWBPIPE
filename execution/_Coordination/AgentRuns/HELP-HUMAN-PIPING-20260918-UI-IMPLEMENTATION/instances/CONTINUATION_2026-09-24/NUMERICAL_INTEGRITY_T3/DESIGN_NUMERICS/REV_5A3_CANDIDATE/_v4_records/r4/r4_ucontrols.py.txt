#!/usr/bin/env python3
"""V4, delta check at R4, check 3 (the safeguard): on DS1's est_check4 state set
(models4's controls and the 20-case probe set, every model with at most 40 free
DOFs, at every verification precision P = 256, 512, 1024 that solves), compare

  est   K4's Hager-Higham estimate as emu4's solve_at forms it,
  exact ||K~_P^-1||_1 (Gauss-Jordan in Fractions),
  U     the comparison-matrix bound ||M(L)^-T D^-1 M(L)^-1 e||_inf from the P-bit
        LDL^T factor (the same loops as emu4), a certified upper bound on
        ||(L D L^T)^-1||_1 at the cost of one substitution pair on nonnegative data,
  Uc    U / (1 - U gamma_m || |L| D |L^T| ||_1), m = 2n + 2: with the factor's
        backward error, a certified upper bound on ||K~_P^-1||_1 itself.

Reports exact/est, U/exact, Uc/exact and Uc/(F est), F = 2^16: where Uc <= F est,
replacing F est by Uc in theta, the charge and W+ would make them derived at no
loss of availability. Output: r4_ucontrols.json."""
import json, os, sys, time
from fractions import Fraction as Fr

HERE = os.path.dirname(os.path.abspath(__file__))
DS1 = os.path.join(os.path.dirname(os.path.dirname(HERE)), "ds1")
sys.path.insert(0, DS1)
sys.path.insert(0, HERE)
import emu4, models4
from emu4 import Case, solve_at, assemble, floor_log2, Stop
from gate_probe3_cases import LOADS, run_345
from r4_hh import factor, exact_inverse, norm1, comparison_bound

F = Fr(2) ** 16


def ldl_abs_norm1(L, D):
    n = len(D)
    # || |L| D |L^T| ||_1 = max column sum; the matrix is symmetric and nonnegative, so
    # it is || |L| (D (|L^T| e)) ||_inf.
    a = [Fr(1)] * n
    for i in range(n):          # |L^T| e: a_j = 1 + sum_{i > j} |l_ij|
        for j in range(i):
            if L[i][j]:
                a[j] += abs(L[i][j])
    b = [D[i] * a[i] for i in range(n)]
    c = list(b)
    for i in range(n):          # |L| b: c_i = b_i + sum_{j < i} |l_ij| b_j
        for j in range(i):
            if L[i][j]:
                c[i] += abs(L[i][j]) * b[j]
    return max(c)


def check(m):
    case = Case(m)
    free = case.free
    if not free or len(free) > 40:
        return "skipped (%d free)" % len(free)
    rec = {}
    for P in (256, 512, 1024):
        try:
            st = solve_at(case, P)
        except Stop as e:
            rec[P] = "stop:%s" % e.what
            continue
        K = assemble(case.m, P)[0]
        sc = [-(floor_log2(abs(K[(g, g)])) // 2) for g in free]
        Kt = [[K.get((a, b), Fr(0)) * Fr(2) ** (sa + sb) for b, sb in zip(free, sc)] for a, sa in zip(free, sc)]
        ex = norm1(exact_inverse(Kt))
        L, D, ok = factor(Kt, P)
        U = comparison_bound(L, D)
        n = len(free)
        mm = 2 * n + 2
        gam = Fr(mm, 2 ** P - mm)
        denom = 1 - U * gam * ldl_abs_norm1(L, D)
        Uc = U / denom if denom > 0 else None
        est = st["est"]
        rec[P] = dict(exact_over_est=float(ex / est), U_over_exact=float(U / ex),
                      Uc_over_exact=(float(Uc / ex) if Uc else None),
                      Uc_over_F_est=(float(Uc / (F * est)) if Uc else None),
                      Uc_ge_exact=(Uc >= ex if Uc else None))
    return rec


if __name__ == "__main__":
    ms = list(models4.models())
    for y in ((3.0, 4.0, 5.0), (0.0, 0.0, 1.0)):
        for members in (1, 3):
            for label, tip in LOADS:
                m = run_345(y, members, tip)
                m["name"] = "PROBE-%s-%dm-%s" % ("".join("%g" % c for c in y), members, label.split()[0])
                ms.append(m)
    out = {}
    agg = dict(states=0, worst_U_over_exact=(0.0, None), worst_Uc_over_F_est=(0.0, None),
               Uc_le_F_est=0, Uc_fail=0, worst_exact_over_est=(0.0, None))
    for m in ms:
        t = time.time()
        r = check(m)
        out[m["name"]] = r
        if isinstance(r, dict):
            for P, v in r.items():
                if not isinstance(v, dict):
                    continue
                agg["states"] += 1
                if v["U_over_exact"] > agg["worst_U_over_exact"][0]:
                    agg["worst_U_over_exact"] = (v["U_over_exact"], (m["name"], P))
                if v["exact_over_est"] > agg["worst_exact_over_est"][0]:
                    agg["worst_exact_over_est"] = (v["exact_over_est"], (m["name"], P))
                if v["Uc_over_F_est"] is None or not v["Uc_ge_exact"]:
                    agg["Uc_fail"] += 1
                else:
                    if v["Uc_over_F_est"] <= 1:
                        agg["Uc_le_F_est"] += 1
                    if v["Uc_over_F_est"] > agg["worst_Uc_over_F_est"][0]:
                        agg["worst_Uc_over_F_est"] = (v["Uc_over_F_est"], (m["name"], P))
        print(m["name"], r if isinstance(r, str) else {P: (v if isinstance(v, str) else "U/ex=%.4g Uc/(F est)=%.3g" % (v["U_over_exact"], v["Uc_over_F_est"] or -1)) for P, v in r.items()}, "(%.1fs)" % (time.time() - t), flush=True)
    print("aggregate:", agg)
    json.dump(dict(per_model=out, aggregate=agg), open(os.path.join(HERE, "r4_ucontrols.json"), "w"), indent=1, default=str)
