#!/usr/bin/env python3
"""V4, delta check at R5, check 3 (a certified alternative with polynomial looseness,
illustrated): on R1's RF-LARGE models, in binary64 and K4's RCM order (r5_large's
setup), the shifted-factorization bound on ||K~^-1||_1:

  if K~ - sigma I + delta = L' D' L'^T with every d' > 0 and |delta| <= gamma_m |L'| D' |L'^T|,
  then lambda_min(K~) > sigma - gamma_m N_L'  (Weyl; the 2-norm of a symmetric nonnegative
  matrix is at most its 1-norm), so ||K~^-1||_1 <= sqrt(n) ||K~^-1||_2 <= sqrt(n)/(sigma - gamma_m N_L').

sigma is half of an inverse-iteration estimate of lambda_min (5 steps on the unshifted
factor); gamma_m is taken at P = 256 with m = 2n + 2. This run is an illustration in
binary64, not a certified evaluation: it shows the bound's looseness, B/exact <= 2 sqrt(n)
in principle, against U's on the same factors (r5_large).
Usage: python3 r5_shift.py [sizes, default 10,100] [max_exact_n, default 700]
Output: r5_shift_<sizes>.json."""
import json, math, os, sys, time
HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, HERE)
sys.dont_write_bytecode = True
import r5_large
from r5_large import r1_adapter, Case, setup, ldl, exact_norm, T3


def solve(L, D, first, b):
    n = len(D)
    x = list(b)
    for i in range(n):
        s = x[i]
        for j in range(first[i], i):
            l = L[i].get(j)
            if l:
                s -= l * x[j]
        x[i] = s
    for i in range(n):
        x[i] /= D[i]
    for i in reversed(range(n)):
        v = x[i]
        if v:
            for j in range(first[i], i):
                l = L[i].get(j)
                if l:
                    x[j] -= l * v
    return x


def nl(L, D, first):
    n = len(D)
    at = [1.0] * n
    for i in range(n):
        for j in range(first[i], i):
            l = L[i].get(j)
            if l:
                at[j] += abs(l)
    bt = [D[i] * at[i] for i in range(n)]
    ct = list(bt)
    for i in range(n):
        for j in range(first[i], i):
            l = L[i].get(j)
            if l:
                ct[i] += abs(l) * bt[j]
    return max(ct)


def run(cid, c, max_exact_n):
    m, index = r1_adapter.adapt(cid, c, c["model"])
    case = Case(m)
    rows, first, n = setup(case, 256, True)
    L, D = ldl(rows, first)
    x = [1.0 + 0.001 * (i % 7) for i in range(n)]
    lam = None
    for _ in range(5):
        y = solve(L, D, first, x)
        ny = math.sqrt(sum(v * v for v in y))
        nx = math.sqrt(sum(v * v for v in x))
        lam = nx / ny
        x = [v / ny for v in y]
    sigma = lam / 2
    rows2 = [dict(r) for r in rows]
    for i in range(n):
        rows2[i][i] = rows2[i].get(i, 0.0) - sigma
    rec = dict(case=cid, free=n, lambda_est=lam, sigma=sigma)
    try:
        L2, D2 = ldl(rows2, first)
        g = (2 * n + 2) * 2.0 ** -256
        margin = sigma - g * nl(L2, D2, first)
        rec["shift_factor_positive"] = True
        rec["log2_B"] = math.log2(math.sqrt(n) / margin) if margin > 0 else None
    except RuntimeError:
        rec["shift_factor_positive"] = False
        rec["log2_B"] = None
    if n <= max_exact_n:
        rec["log2_exact"] = math.log2(exact_norm(L, D, first))
        if rec["log2_B"] is not None:
            rec["log2_B_over_exact"] = rec["log2_B"] - rec["log2_exact"]
            rec["log2_2sqrt_n"] = math.log2(2 * math.sqrt(n))
    return rec


if __name__ == "__main__":
    sizes = [int(x) for x in (sys.argv[1] if len(sys.argv) > 1 else "10,100").split(",")]
    max_exact_n = int(sys.argv[2]) if len(sys.argv) > 2 else 700
    cases = r1_adapter.load_refs(T3)
    out = []
    for cid, c in cases.items():
        if c["family"] != "RF-LARGE" or "generator" in c["model"] or len(c["model"]["members"]) not in sizes:
            continue
        t0 = time.time()
        rec = run(cid, c, max_exact_n)
        rec["seconds"] = round(time.time() - t0, 1)
        out.append(rec)
        print(json.dumps(rec), flush=True)
    json.dump(out, open(os.path.join(HERE, "r5_shift_%s.json" % "_".join(map(str, sizes))), "w"), indent=1)
