#!/usr/bin/env python3
"""DS1, R6: K4's elimination order, ported from `retained/factor.rs` at `cef218a10`
(`bfs_reachable`, `bfs_eccentricity`, `min_by_degree_then_index`,
`pseudo_peripheral_start`, `reverse_cuthill_mckee`, `order_free`), function by
function. V4 ported the same functions for its R5 check (`_v4_records/r5/r5_large`);
`k4order_check6.py` compares the two ports on every model it orders.

structural_pattern(model): the (row, column) pairs K4's structure.pattern holds:
every member's 12 x 12 DOF pairs, each spring's diagonal and each directional
block's 3 x 3 pairs (the keys of emu's assembled K).
order_free(free, pattern): the free DOFs in K4's order (reverse Cuthill-McKee on
the free-free sub-pattern), as a list of global DOFs.
Standard library only."""
from collections import deque


def member_dofs(m):
    return [6 * m["i"] + k for k in range(6)] + [6 * m["j"] + k for k in range(6)]


def structural_pattern(model):
    pat = set()
    for m in model["members"]:
        d = member_dofs(m)
        for a in d:
            for b in d:
                pat.add((a, b))
    for s in model["springs"]:
        g = 6 * s["node"] + s["c"]
        pat.add((g, g))
    for sp in model.get("dsprings", []):
        base = 6 * sp["node"] + (0 if sp["kind"] == "t" else 3)
        for a in range(3):
            for b in range(3):
                pat.add((base + a, base + b))
    return pat


def bfs_reachable(seed, neighbors):
    marked = [False] * len(neighbors)
    marked[seed] = True
    reachable = [seed]
    queue = deque([seed])
    while queue:
        node = queue.popleft()
        for nxt in neighbors[node]:
            if not marked[nxt]:
                marked[nxt] = True
                reachable.append(nxt)
                queue.append(nxt)
    return reachable


def bfs_eccentricity(start, neighbors):
    marked = [False] * len(neighbors)
    marked[start] = True
    level = [start]
    depth = 0
    while True:
        next_level = []
        for node in level:
            for nxt in neighbors[node]:
                if not marked[nxt]:
                    marked[nxt] = True
                    next_level.append(nxt)
        if not next_level:
            return depth, level
        depth += 1
        level = next_level


def min_by_degree_then_index(nodes, degrees):
    return min(nodes, key=lambda node: (degrees[node], node)) if nodes else None


def pseudo_peripheral_start(seed, neighbors, degrees):
    component = bfs_reachable(seed, neighbors)
    candidate = min_by_degree_then_index(component, degrees)
    if candidate is None:
        candidate = seed
    cand_ecc, last_level = bfs_eccentricity(candidate, neighbors)
    while True:
        nxt = min_by_degree_then_index(last_level, degrees)
        if nxt is None:
            break
        next_ecc, next_last = bfs_eccentricity(nxt, neighbors)
        if next_ecc > cand_ecc:
            candidate, cand_ecc, last_level = nxt, next_ecc, next_last
        else:
            break
    return candidate


def reverse_cuthill_mckee(adjacency):
    node_count = len(adjacency)
    neighbors = [[] for _ in range(node_count)]
    for node, raw in enumerate(adjacency):
        for other in raw:
            if other != node:
                neighbors[node].append(other)
                neighbors[other].append(node)
    neighbors = [sorted(set(lst)) for lst in neighbors]      # sort_unstable + dedup
    degrees = [len(lst) for lst in neighbors]
    neighbors = [sorted(lst, key=lambda node: (degrees[node], node)) for lst in neighbors]
    visited = [False] * node_count
    order = []
    for seed in range(node_count):
        if visited[seed]:
            continue
        start = pseudo_peripheral_start(seed, neighbors, degrees)
        visited[start] = True
        queue = deque([start])
        while queue:
            node = queue.popleft()
            order.append(node)
            for nxt in neighbors[node]:
                if not visited[nxt]:
                    visited[nxt] = True
                    queue.append(nxt)
    order.reverse()
    return order


def free_adjacency(free, pattern):
    """Per free position a (free ascending), the free positions of pattern row free[a]."""
    position = {g: a for a, g in enumerate(free)}
    rows = {}
    for (r, c) in pattern:
        if r != c and r in position and c in position:
            rows.setdefault(r, []).append(position[c])
    return [rows.get(g, []) for g in free]


def order_free(free, pattern):
    """K4's order_free: RCM on the free-free sub-pattern of the ascending free DOFs.
    Returns the global DOFs in elimination order."""
    free = sorted(free)
    adjacency = free_adjacency(free, pattern)
    order = reverse_cuthill_mckee(adjacency)
    return [free[a] for a in order]
