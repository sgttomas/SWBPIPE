"""V4 emulator fidelity: K4's recorded survey outcomes (RETURN §11 G; ROOT rulings)."""
import math, json, time
from fractions import Fraction as Fr
from v4emu import model, member, fix, load, spring, station, run
PI = Fr(math.pi)
def nsec():
    od, idd = Fr("0.2"), Fr("0.18")
    A = PI * (od*od - idd*idd) / 4; I = PI * (od**4 - idd**4) / 64
    return dict(E=200e9, G=80e9, A=float(A), Iy=float(I), Iz=float(I), J=float(2*I))
S = nsec()
def mem(m, i, j, y=(0.0,0.0,1.0), **over):
    s = dict(S); s.update(over); member(m, i, j, y=y, **s)
def pin_case(name, d, k, mo, members=1, y=(0.0,0.0,1.0)):
    m = model(name, [tuple(c*s for c in d) for s in range(members+1)])
    for s in range(members): mem(m, s, s+1, y)
    fix(m, 0, (0,1,2))
    for c in (3,4,5): spring(m, 0, c, k)
    for c in range(3):
        if mo[c] != 0: load(m, members, 3+c, mo[c])
    return m
mom = lambda k: (1e-8*k/1e-4, 2e-8*k/1e-4, 0.0)
ms = []
ms.append(pin_case("SKEW-K1E-28", (3.0,4.0,0.0), 1e-28, mom(1e-28)))
ms.append(pin_case("SKEW6-K1E-12", (3.0,4.0,0.0), 1e-12, mom(1e-12), members=6))
a = float(Fr(S["G"]) * Fr(S["J"]) / 2)
m = model("PIVOT", [(0,0,0),(2,0,0)]); mem(m, 0, 1); fix(m, 0, (0,1,2,4,5)); spring(m, 0, 3, a*2.0**-120); load(m, 1, 3, a*2.0**-120*1e-4); ms.append(m)
m = model("REACTIONS-ONLY", [(0,0,0),(3,4,0),(3,4,2)]); mem(m, 0, 1)
fix(m, 0, (0,), 1.0); fix(m, 1, (0,), 1.0); fix(m, 0, (1,2,3,4,5)); fix(m, 1, (1,2,3,4,5))
mem(m, 1, 2, y=(1.0,0.0,0.0), Iy=2.0**-26, Iz=2.0**-26); load(m, 2, 0, 2.0**-45); ms.append(m)
m = model("ZERO-TORSION-345", [(0,0,0),(3,4,0)]); mem(m, 0, 1); fix(m, 0, range(6)); load(m, 1, 3, 3.0); load(m, 1, 4, 4.0); ms.append(m)
m = model("ALL-ZERO-BODY", [(0,0,0),(2,0,0),(0,5,0),(2,5,0)]); mem(m, 0, 1); mem(m, 2, 3); fix(m, 0, range(6)); fix(m, 2, range(6)); load(m, 1, 2, 10.0); ms.append(m)
out = {}
for m in ms:
    t = time.time()
    o, tr = run(m, rules=("current","iii"), truth=False, verbose=False)
    out[m["name"]] = {r: (v["selected"], v["attempts"]) for r, v in o.items()}
    print(m["name"], out[m["name"]], "(%.1fs)" % (time.time()-t), flush=True)
json.dump(out, open("fidelity.json","w"), indent=1, default=str)
