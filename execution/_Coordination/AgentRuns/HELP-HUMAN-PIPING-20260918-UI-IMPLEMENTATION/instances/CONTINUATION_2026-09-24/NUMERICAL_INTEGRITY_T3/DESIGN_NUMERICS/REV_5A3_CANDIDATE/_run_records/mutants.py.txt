#!/usr/bin/env python3
"""Section 7's mutants, emulated where DS1's emulator can express them: each
runs the relevant controls under the recommended option (iii) with one change,
and reports selected p and honesty against the unmutated (iii)."""
import json
from fractions import Fraction as Fr
import emu
from emu import Case, solve_at, schedule, classify, honesty, new_model, add_member, fix, load
from models import models, pin_case, SKEW

M = {m["name"]: m for m in models()}
# M6's control: cancelling loads (1e80, 1e-8, -1e80) at a restrained DOF of B1-L's geometry.
m = pin_case("LEDGER-AT-RESTRAINT", SKEW, 1e-4, (0.0, 0.0, 0.0))
for v in (1e80, 1e-8, -1e80):
    load(m, 0, 0, v)
load(m, 1, 4, 2e-8)
M[m["name"]] = m
CONTROLS = ["F-2", "F-2-SPOS", "F-2-CEIL", "PRESCRIBED-TAIL", "RIGID-UNLOADED", "F-3-FREE", "F-3-ROT",
            "REACTIONS-ONLY", "SKEW6-K1E-12", "N06", "B1-L", "LEDGER-AT-RESTRAINT"]


def run(tag):
    out = {}
    for name in CONTROLS:
        case = Case(M[name])
        ref = solve_at(case, 2048, ref=True)
        r = schedule(case, "iii", {})
        rec = dict(selected=r["selected"])
        if r["selected"] is not None:
            pub, classes, _ = classify(case, r, "iii")
            w, c = honesty(case, pub, classes, ref["values"], ref["E"])
            rec.update(honest=w[0] <= 1.0, worst=w[0], relative=c["relative"])
        out[name] = rec
    return out


base = run("base")
results = {"baseline (iii)": base}
orig_floors, orig_body_E = emu.floors, emu.body_E
variants = []


def v_M1():
    emu.LAMBDA = 0
def u_M1():
    emu.LAMBDA = 256
variants.append(("M1 drop V", v_M1, u_M1))


def v_M2():
    emu.body_E = lambda case, E, coupled=True: orig_body_E(case, E, coupled=False)
def u_M2():
    emu.body_E = orig_body_E
variants.append(("M2 uncoupled E", v_M2, u_M2))


def v_M3():
    def f(option, p, Eb):
        return orig_floors("ib", p, Eb)
    emu.floors = f
def u_M3():
    emu.floors = orig_floors
variants.append(("M3 floor at every p", v_M3, u_M3))


def v_M4():
    def f(option, p, Eb):
        return orig_floors("ii", p, Eb)
    emu.floors = f
variants.append(("M4 no floor", v_M4, u_M3))


def v_M5():
    emu.E_PRESCRIBED = False
def u_M5():
    emu.E_PRESCRIBED = True
variants.append(("M5 E without prescribed |u|", v_M5, u_M5))


def v_M6():
    emu.E_LEDGER = "terms"
def u_M6():
    emu.E_LEDGER = "net"
variants.append(("M6 E from sum |ledger terms|", v_M6, u_M6))


def v_M8():
    def f(option, p, Eb):
        out = {k: Fr(0) for k in emu.KINDS}
        if p == 512:
            for k in ("fo", "mo"):
                out[k] = Fr(Eb[k]) * Fr(2) ** (74 - 2 * p)
        return out
    emu.floors = f
variants.append(("M8 floor at the verification resolution", v_M8, u_M3))


def v_M9():
    def f(option, p, Eb):
        if p != 256:
            return {k: Fr(0) for k in emu.KINDS}
        return {k: Fr(Eb[k]) * Fr(2) ** (74 - 256) for k in emu.KINDS}
    emu.floors = f
variants.append(("M9 floor at 256 instead of 512", v_M9, u_M3))

# classification must follow the same floor as the stop rule for M3/M8/M9: the
# emulator's classify() applies Phi for option "iii" only at p = 512; for these
# mutants the killing observation is the selected p (and, where noted, honesty).
for tag, on, off in variants:
    on()
    try:
        results[tag] = run(tag)
    finally:
        off()
for tag, res in results.items():
    diff = {k: v for k, v in res.items() if tag.startswith("baseline") or v != base[k]}
    print(tag)
    for k, v in diff.items():
        print("   ", k, v)
json.dump(results, open("mutants.stdout.json", "w"), indent=1)
