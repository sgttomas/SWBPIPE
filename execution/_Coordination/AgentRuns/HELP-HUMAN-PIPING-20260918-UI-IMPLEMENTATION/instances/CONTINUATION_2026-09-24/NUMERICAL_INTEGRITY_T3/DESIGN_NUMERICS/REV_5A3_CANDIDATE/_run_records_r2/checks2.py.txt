#!/usr/bin/env python3
"""Revision 2 spot checks for the document's claims:
- EXACT-RIGID under today's rule and the revision (selected p, largest b on a
  force row, honesty);
- TWO-SPAN's correction count at 128 under each gate (K4-M11's evidence, O5);
- the correction count of every K4 probe case under the hybrid gate;
- the kinds of G5a failure under mutants M7 (entrywise operator), M2 and M5;
- DEMOTION2's largest b.
Output: checks2.stdout.json."""
import json
import emu2, models2
from emu2 import Case, schedule, classify, honesty, solve_at, Stop
import gate_probe3_cases as gp

out = {}


def run(m, opt, gate):
    case = Case(m)
    emu2.GATE = "absolute"
    ref = solve_at(case, 2048, ref=True, estimate=False)
    emu2.GATE = gate
    r = schedule(case, opt, {})
    o = dict(selected=r["selected"], attempts=[a[1][:40] for a in r["attempts"]])
    if r["selected"]:
        pub, cl, sc, g5 = classify(case, r, opt)
        w, c = honesty(case, pub, cl, ref)
        bmax = max((b for (rid, kind, body, inp), (k, b) in zip(case.layout, cl)
                    if kind in ("fo", "mo") and k == "absolute"), default=None)
        o.update(honest=w[0] <= 1.0, worst=w[0], counts=c, b_max_force_moment=bmax,
                 g5a=[str(x[:3]) for x in g5], corrections=r["cand"]["corrections"])
    emu2.GATE = "absolute"
    return o


M = {m["name"]: m for m in models2.models()}
M["EXACT-RIGID"] = models2.exact_rigid()
out["EXACT-RIGID"] = {k: run(M["EXACT-RIGID"], *k.split("/")) for k in ("current/k4", "fin/hybrid")}
out["DEMOTION2"] = run(M["DEMOTION2"], "fin", "hybrid")

tw = {}
for gate in ("k4", "absolute", "hybrid"):
    emu2.GATE = gate
    st = solve_at(Case(M["TWO-SPAN"]), 128)
    tw[gate] = st["corrections"]
emu2.GATE = "absolute"
out["TWO-SPAN corrections at 128"] = tw

pc = []
for y in ((3.0, 4.0, 5.0), (0.0, 0.0, 1.0)):
    for members in (1, 3):
        for label, tip in gp.LOADS:
            emu2.GATE = "hybrid"
            case = Case(gp.run_345(y, members, tip))
            try:
                st = solve_at(case, 128)
                pc.append((y, members, label, st["corrections"], st["gate_worst"]))
            except Stop as s:
                pc.append((y, members, label, s.what, None))
emu2.GATE = "absolute"
out["probe corrections at 128 (hybrid)"] = pc

saved = emu2.E_OPERATOR
emu2.E_OPERATOR = "entrywise"
m7 = {}
for name in ("RIGID-UNLOADED", "F-3-FREE", "F-3-ROT", "REACTIONS-ONLY", "SKEW6-K1E-12", "B1-L", "LEDGER-AT-RESTRAINT"):
    o = run(M[name], "fin", "hybrid")
    m7[name] = o.get("g5a")
emu2.E_OPERATOR = saved
out["M7 G5a failure kinds"] = m7

# M2 (uncoupled E) and M5 (E without the prescribed |u|): the kinds of G5a failure
for tag, attr, val, names in (("M2", "E_COUPLED", False, ("TWO-SPAN",)),
                              ("M5", "E_PRESCRIBED", False, ("PRESCRIBED-TAIL", "REACTIONS-ONLY", "DEMOTION2",
                                                             "ASSEMBLY-SAT", "M10-G"))):
    saved = getattr(emu2, attr)
    setattr(emu2, attr, val)
    out["%s G5a failure kinds" % tag] = {n: run(M[n], "fin", "hybrid").get("g5a") for n in names}
    setattr(emu2, attr, saved)

print(json.dumps(out, indent=1, default=str))
json.dump(out, open("checks2.stdout.json", "w"), indent=1, default=str)
