"""V4: realized resolution and total error ratios on V4's own models.
res  = max over force/moment rows |q_P - R*(u_P)| / (2^-P * E_q(u_P))  (the count's quantity, <= 139 claimed)
tot  = max over force/moment rows |q_P - q*| / (2^-P * E-hat(body, kind))  (the conjecture's quantity; DS1: <= 2.73)
R* = exact recovery with exact operators; q* = exact truth."""
import json, math, time, sys
from fractions import Fraction as Fr
from v4emu import *
import cases
from fidelity import ms as fid_models
from gate_probe_nsec import S as NSEC

def exact_ops(m):
    ar = Ar(None)
    ops = [form_member(ar, m["nodes"], mb) for mb in m["members"]]
    K, n = assemble(ar, m, ops)
    return ar, ops, K, n

def measure(m, gate="k4"):
    tr = exact_state(m)
    ar0, ops0, K0, n0 = exact_ops(m)
    lay = layout(m); nb = bodies(m); ext = extents(m)
    out = {}
    for P in (128, 256, 512, 1024):
        try:
            st = solve_at(m, P, gate)
        except Stop as s:
            out[P] = "stop:%s" % s
            continue
        E, Erow = formation_scale(m, st)
        Rstar, _ = recover(ar0, m, ops0, K0, n0, st["u"])
        res = 0.0; tot = 0.0; argres = argtot = None
        for idx, (key, kind, inp) in enumerate(lay):
            if kind not in ("fo", "mo") or inp:
                continue
            Eq = Erow[idx]
            b = row_body(m, key, nb)
            efo, emo = e_hat(E, b, ext[b])
            Eh = efo if kind == "fo" else emo
            dq = abs(st["vals"][idx] - Rstar[idx])
            if dq:
                r = float(dq / (Fr(2) ** -P * Eq)) if Eq else math.inf
                if r > res: res, argres = r, key
            dt = abs(st["vals"][idx] - tr["vals"][idx])
            if dt:
                r = float(dt / (Fr(2) ** -P * Eh)) if Eh else math.inf
                if r > tot: tot, argtot = r, key
        out[P] = dict(res=res, res_at=str(argres), tot=tot, tot_at=str(argtot))
    return out

def nsec_gs(name, y, loaded, rot=True):
    m = model(name, [(0,0,0),(3,4,0),(6,8,0),(9,12,0)])
    for s in range(3): member(m, s, s+1, y=y, **NSEC)
    if rot:
        fix(m, 0, (0,1,2,3,4)); fix(m, 0, (5,), 1e-3)
    else:
        fix(m, 0, range(6))
    if loaded: load(m, 3, 1, 1.0)
    return m

if __name__ == "__main__":
    models = [cases.f2(), cases.f2_spos(), cases.f2("F-2-CEIL", E=2.0**41, lv=2.0**-1000), cases.ptail(), cases.ptail(True),
              cases.rigid_unloaded(), cases.f3_free(), cases.f3_rot(), cases.exact_rigid(), cases.demotion2(),
              cases.gs("GS-ROT-y345", (3.0,4.0,5.0), True), nsec_gs("GS-ROT-y001-N", (0.0,0.0,1.0), False),
              nsec_gs("GS-LOADONLY-y001-N", (0.0,0.0,1.0), True, rot=False)] + fid_models
    allres = {}
    worst_res = worst_tot = 0.0
    for m in models:
        t = time.time()
        r = measure(m)
        allres[m["name"]] = r
        for P, v in r.items():
            if isinstance(v, dict):
                worst_res = max(worst_res, v["res"]); worst_tot = max(worst_tot, v["tot"])
        print(m["name"], {P: (v if isinstance(v, str) else (round(v["res"], 3), round(v["tot"], 3))) for P, v in r.items()}, "(%.1fs)" % (time.time()-t), flush=True)
    print("worst res ratio (count claims <= 139):", worst_res)
    print("worst total ratio (DS1 measured <= 2.73):", worst_tot)
    json.dump(dict(per_model=allres, worst_res=worst_res, worst_tot=worst_tot), open("measure.json", "w"), indent=1)
