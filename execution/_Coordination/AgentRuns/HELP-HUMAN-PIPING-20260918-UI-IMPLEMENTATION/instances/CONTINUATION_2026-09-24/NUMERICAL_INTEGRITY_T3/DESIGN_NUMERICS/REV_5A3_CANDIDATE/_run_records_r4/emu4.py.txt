# ---------------------------------------------------------------- rounding

def floor_log2(a):
    n, d = a.numerator, a.denominator
    e = n.bit_length() - d.bit_length()
    if e >= 0:
        if n < (d << e):
            e -= 1
    else:
        if (n << -e) < d:
            e -= 1
    return e


def rp(x, p):
    """Fraction x rounded once to p bits (nearest, ties to even); 0 -> 0."""
    if x == 0:
        return Fr(0)
    neg = x < 0
    a = -x if neg else x
    e = floor_log2(a)
    sh = p - 1 - e
    n, d = a.numerator, a.denominator
    if sh >= 0:
        num, den = n << sh, d
    else:
        num, den = n, d << (-sh)
    q, r = divmod(num, den)
    if 2 * r > den or (2 * r == den and q & 1):
        q += 1
    v = Fr(q, 1 << sh) if sh >= 0 else Fr(q << (-sh))
    return -v if neg else v


def sqrt_p(x, p):
    n, d = x.numerator, x.denominator
    k = (p + 4) - (n.bit_length() - d.bit_length()) // 2
    num = n << (2 * k) if k >= 0 else n
    den = d if k >= 0 else d << (-2 * k)
    t = num // den
    s = isqrt(t)
    exact = s * s == t and t * den == num
    val = Fr(2 * s + (0 if exact else 1), 2)
    val = val / (Fr(2) ** k)
    return rp(val, p)


def fl(x):
    """binary64 of a Fraction, correctly rounded; overflow -> inf."""
    if x == 0:
        return 0.0
    try:
        return x.numerator / x.denominator
    except OverflowError:
        return math.copysign(math.inf, x)


def next_up(x):
    if x == 0.0:
        return 5e-324
    b = struct.unpack("<Q", struct.pack("<d", x))[0]
    return struct.unpack("<d", struct.pack("<Q", b + 1))[0]


def fl_up(x):
    """The least binary64 >= x (x >= 0 Fraction)."""
    y = fl(x)
    if y != math.inf and Fr(y) < x:
        y = next_up(y)
    return y


def publishable(v):
    """K3's outcome: normal or subnormal is published; underflow/overflow are not."""
    y = fl(v)
    if y == math.inf or y == -math.inf:
        return False
    if y == 0.0 and v != 0:
        return False
    return True

def n_section():
    E, G = 200e9, 80e9
    od, idd = 0.2, 0.18
    A = math.pi * (od * od - idd * idd) / 4
    I = math.pi * (od ** 4 - idd ** 4) / 64
    return dict(E=E, G=G, A=A, Iy=I, Iz=I, J=2 * I)


def new_model(name, nodes):
    return dict(name=name, nodes=[tuple(float(c) for c in p) for p in nodes],
                members=[], springs=[], dsprings=[], constraints=[], loads=[], stations=[])

    s = dict(section or n_section())
    s.update(over)
    model["members"].append(dict(id=mid, i=i, j=j, y=tuple(float(c) for c in y), **s))

# ---------------------------------------------------------------- formation

def g_exact(yr, yc):
    """g = 2^k, k the least integer >= 0 with 4^k (yc.yc) >= (y_ref.y_ref);
    both dot products exact."""
    if G_MODE == "one":
        return 1
    yy = sum((c * c for c in yr), Fr(0))
    cc = sum((c * c for c in yc), Fr(0))
    k = 0
    while Fr(4) ** k * cc < yy:
        k += 1
    return 2 ** k


def form_member(nodes, m, p):
    """assemble.rs `form_member` at p (every rounding mirrored)."""
    F = Fr
    xi, xj = nodes[m["i"]], nodes[m["j"]]

    def dot(a, b):
        return rp(sum((x * y for x, y in zip(a, b)), Fr(0)), p)

    def normalize(v):
        n = sqrt_p(dot(v, v), p)
        return [rp(c / n, p) for c in v], n

    d = [rp(F(xj[k]) - F(xi[k]), p) for k in range(3)]
    ex, L = normalize(d)
    yr = [F(c) for c in m["y"]]
    proj = dot(yr, ex)
    yc = [rp(yr[k] - proj * ex[k], p) for k in range(3)]
    ey, _ = normalize(yc)
    zc = [rp(ex[a] * ey[b] - ex[b] * ey[a], p) for a, b in ((1, 2), (2, 0), (0, 1))]
    ez, _ = normalize(zc)
    inv = rp(Fr(1) / L, p)
    axial = rp(rp(F(m["E"]) * F(m["A"]), p) / L, p)
    torsion = rp(rp(F(m["G"]) * F(m["J"]), p) / L, p)
    bz = rp(rp(F(m["E"]) * F(m["Iz"]), p) / L, p)
    by = rp(rp(F(m["E"]) * F(m["Iy"]), p) / L, p)
    B = [[Fr(0)] * 12 for _ in range(6)]
    for k in range(3):
        iy = rp(inv * ey[k], p)
        iz = rp(inv * ez[k], p)
        B[0][k], B[0][6 + k] = -ex[k], ex[k]
        B[1][3 + k], B[1][9 + k] = -ex[k], ex[k]
        for row, rot in ((2, 3), (3, 9)):
            B[row][k], B[row][6 + k], B[row][rot + k] = iy, -iy, ez[k]
        for row, rot in ((4, 3), (5, 9)):
            B[row][k], B[row][6 + k], B[row][rot + k] = -iz, iz, ey[k]
    drows = [[(0, axial)], [(1, torsion)], [(2, 4 * bz), (3, 2 * bz)], [(2, 2 * bz), (3, 4 * bz)],
             [(4, 4 * by), (5, 2 * by)], [(4, 2 * by), (5, 4 * by)]]
    DB = [[rp(sum((c * B[s][col] for s, c in drows[r]), Fr(0)), p) for col in range(12)] for r in range(6)]
    ke = {}
    for a in range(12):
        for b in range(a, 12):
            ke[(a, b)] = ke[(b, a)] = rp(sum((B[r][a] * DB[r][b] for r in range(6)), Fr(0)), p)
    op = dict(axes=[ex, ey, ez], inv=inv, axial=axial, torsion=torsion, bz=bz, by=by,
              ke=ke, B=B, drows=drows, g=g_exact(yr, yc))
    op["Abar"] = element_bounded(op, p)
    return op


def element_bounded(op, p):
    """g * Bbar^T |D| Bbar (12x12), stage-rounded at p: DBbar = |D| Bbar (one
    exact sum rounded once per entry), then Bbar^T DBbar (likewise), then the
    exact power-of-two g."""
    inv = op["inv"]
    if E_OPERATOR == "entrywise":
        Bb = [[abs(x) for x in row] for row in op["B"]]
    else:
        Bb = [[Fr(0)] * 12 for _ in range(6)]
        for k in range(3):
            Bb[0][k] = Bb[0][6 + k] = Fr(1)
            Bb[1][3 + k] = Bb[1][9 + k] = Fr(1)
            for row, rot in ((2, 3), (3, 9), (4, 3), (5, 9)):
                Bb[row][k] = Bb[row][6 + k] = abs(inv)
                Bb[row][rot + k] = Fr(1)
    Dabs = [[Fr(0)] * 6 for _ in range(6)]
    for r, terms in enumerate(op["drows"]):
        for s, c in terms:
            Dabs[r][s] = abs(c)
    DB = [[rp(sum((Dabs[r][s] * Bb[s][col] for s in range(6)), Fr(0)), p) for col in range(12)] for r in range(6)]
    A = [[rp(sum((Bb[r][a] * DB[r][b] for r in range(6)), Fr(0)), p) * op["g"] for b in range(12)] for a in range(12)]
    return A


def directional_block(s, p):
    n = [Fr(c) for c in s["n"]]
    norm2 = rp(sum((c * c for c in n), Fr(0)), p)
    k = Fr(s["k"])
    out = [[Fr(0)] * 3 for _ in range(3)]
    for a in range(3):
        for b in range(a, 3):
            mm = rp(n[a] * n[b], p)
            v = rp(rp(k * mm, p) / norm2, p)
            out[a][b] = out[b][a] = v
    return out


def member_dofs(m):
    return [6 * m["i"] + k for k in range(6)] + [6 * m["j"] + k for k in range(6)]


def assemble(model, p):
    """K (exact sum per entry, rounded once), the assembled bounded operator
    A-bar (exact sum of the element blocks, springs |k| and directional blocks
    entrywise, rounded once), the pattern, members and directional blocks."""
    nodes = model["nodes"]
    members = [form_member(nodes, m, p) for m in model["members"]]
    contrib = {}
    acontrib = {}
    for m, op in zip(model["members"], members):
        dofs = member_dofs(m)
        for a in range(12):
            for b in range(12):
                contrib.setdefault((dofs[a], dofs[b]), []).append(op["ke"][(a, b)])
                if op["Abar"][a][b]:
                    acontrib.setdefault((dofs[a], dofs[b]), []).append(op["Abar"][a][b])
    for s in model["springs"]:
        d = 6 * s["node"] + s["c"]
        contrib.setdefault((d, d), []).append(Fr(s["k"]))
        acontrib.setdefault((d, d), []).append(abs(Fr(s["k"])))
    dblocks = []
    for sp in model.get("dsprings", []):
        blk = directional_block(sp, p)
        base = 6 * sp["node"] + (0 if sp["kind"] == "t" else 3)
        for a in range(3):
            for b in range(3):
                contrib.setdefault((base + a, base + b), []).append(blk[a][b])
                if blk[a][b]:
                    acontrib.setdefault((base + a, base + b), []).append(abs(blk[a][b]))
        dblocks.append((base, blk))
    K = {rc: rp(sum(vs, Fr(0)), p) for rc, vs in contrib.items()}
    Abar = {rc: rp(sum(vs, Fr(0)), p) for rc, vs in acontrib.items()}
    return K, Abar, set(contrib.keys()), members, dblocks

class Case:
    def __init__(self, model):
        self.m = model
        self.nn = len(model["nodes"])
        self.n = 6 * self.nn
        self.fixed = {}
        for c in model["constraints"]:
            self.fixed[6 * c["node"] + c["c"]] = Fr(c["v"])
        self.free = [g for g in range(self.n) if g not in self.fixed]
        self.f = {}
        self.f_terms = {}
        for l in model["loads"]:
            g = 6 * l["node"] + l["c"]
            self.f[g] = self.f.get(g, Fr(0)) + Fr(l["v"])
            self.f_terms[g] = self.f_terms.get(g, Fr(0)) + abs(Fr(l["v"]))
        parent = list(range(self.nn))

        def find(a):
            while parent[a] != a:
                parent[a] = parent[parent[a]]
                a = parent[a]
            return a
        for mm in model["members"]:
            ra, rb = find(mm["i"]), find(mm["j"])
            if ra != rb:
                parent[max(ra, rb)] = min(ra, rb)
        roots = sorted(set(find(a) for a in range(self.nn)))
        self.body_of = [roots.index(find(a)) for a in range(self.nn)]
        self.nbody = len(roots)
        self.extent = []
        for b in range(self.nbody):
            pts = [model["nodes"][a] for a in range(self.nn) if self.body_of[a] == b]
            d = [max(p[k] for p in pts) - min(p[k] for p in pts) for k in range(3)]
            self.extent.append(math.sqrt((d[0] * d[0] + d[1] * d[1]) + d[2] * d[2]))
        self.layout = self._layout()

    def _layout(self):
        out = []
        m = self.m
        for g in range(self.n):
            node, c = divmod(g, 6)
            out.append((("u", node, c), "tr" if c < 3 else "ro", self.body_of[node], g in self.fixed))
        for node in range(self.nn):
            out.append((("mag", node), "tr", self.body_of[node], False))
        for mm in m["members"]:
            for end in ("i", "j"):
                for c in range(6):
                    out.append((("end", mm["id"], end, c), "fo" if c < 3 else "mo", self.body_of[mm["i"]], False))
        for st in m["stations"]:
            mm = next(x for x in m["members"] if x["id"] == st["member"])
            for c in range(6):
                out.append((("st", st["id"], c), "fo" if c < 3 else "mo", self.body_of[mm["i"]], False))
        for s in m["springs"]:
            out.append((("spr", s["id"]), "fo" if s["c"] < 3 else "mo", self.body_of[s["node"]], False))
        for sp in m.get("dsprings", []):
            for a in range(3):
                out.append((("dspr", sp["id"], a), "fo" if sp["kind"] == "t" else "mo", self.body_of[sp["node"]], False))
        for g in sorted(self.fixed):
            node, c = divmod(g, 6)
            out.append((("R", node, c), "fo" if c < 3 else "mo", self.body_of[node], False))
        return out

class Stop(Exception):

def solve_at(case, p, ref=False, estimate=True):
    q = p + 64 if (p < 1024 or ref) else 1024
    K, Abar, pattern, members, dblocks = assemble(case.m, p)
    if q == p:
        Kq, Abarq, membersq, dblocksq = K, Abar, members, dblocks
    else:
        Kq, Abarq, _, membersq, dblocksq = assemble(case.m, q)
    free = case.free
    n = len(free)
    scale = []
    for g in free:
        d = K.get((g, g), Fr(0))
        if d == 0:
            raise Stop("ZeroDiagonal")
        scale.append(-(floor_log2(abs(d)) // 2))

    def kt(a, b):
        v = K.get((free[a], free[b]), Fr(0))
        return v * Fr(2) ** (scale[a] + scale[b]) if v else Fr(0)
    first = []
    for i in range(n):
        f_i = i
        for j in range(i):
            if (free[i], free[j]) in pattern:
                f_i = j
                break
        first.append(f_i)
    rows = [dict() for _ in range(n)]
    for i in range(n):
        for j in range(first[i], i + 1):
            rows[i][j] = kt(i, j)

    def get(i, j):
        return rows[i].get(j, Fr(0)) if j >= first[i] else Fr(0)
    work = [Fr(0)] * n
    for i in range(n):
        for j in range(first[i], i):
            s = get(i, j)
            for kk in range(max(first[i], first[j]), j):
                s = rp(s - rp(work[kk] * get(j, kk), p), p)
            work[j] = s
            rows[i][j] = rp(s / get(j, j), p)
        pivot = get(i, i)
        canc = abs(pivot)
        for kk in range(first[i], i):
            term = rp(work[kk] * get(i, kk), p)
            pivot = rp(pivot - term, p)
            canc = rp(canc + abs(term), p)
        ops = 2 * (i - first[i]) + 2
        if not (pivot * (2 ** p - ops) - 64 * ops * canc > 0):
            raise Stop("Pivot")
        rows[i][i] = pivot

    def solve_scaled(b):
        x = list(b)
        for i in range(n):
            for j in range(first[i], i):
                x[i] = rp(x[i] - rp(get(i, j) * x[j], p), p)
        for i in range(n):
            x[i] = rp(x[i] / get(i, i), p)
        for i in reversed(range(n)):
            v = x[i]
            for j in range(first[i], i):
                x[j] = rp(x[j] - rp(get(i, j) * v, p), p)
        return x

    def solve(b):
        y = solve_scaled([bi * Fr(2) ** s for bi, s in zip(b, scale)])
        return [yi * Fr(2) ** s for yi, s in zip(y, scale)]
    est = None
    if n > 0 and not ref:
        norm = Fr(0)
        for a in range(n):
            col = rp(sum((abs(kt(a, b)) for b in range(n)), Fr(0)), p)
            norm = max(norm, col)
        one = Fr(1)
        x = [rp(Fr(1, n), p)] * n
        est = Fr(0)
        prev = n
        for _ in range(5):
            y = solve_scaled(x)
            cur = rp(sum((abs(v) for v in y), Fr(0)), p)
            if cur <= est and est != 0:
                break
            est = cur
            signs = [one if v >= 0 else -one for v in y]
            z = solve_scaled(signs)
            j = 0
            for i in range(1, n):
                if abs(z[i]) >= abs(z[j]):
                    j = i
            dot = rp(sum((z[i] * x[i] for i in range(n)), Fr(0)), p)
            if abs(z[j]) <= dot or j == prev:
                break
            prev = j
            x = [Fr(0)] * n
            x[j] = one
        alt = []
        for i in range(n):
            mag = one if n == 1 else rp(Fr(n - 1 + i) / Fr(n - 1), p)
            alt.append(mag if i % 2 == 0 else -mag)
        y = solve_scaled(alt)
        total = rp(sum((abs(v) for v in y), Fr(0)), p) * 2
        alternative = rp(total / (3 * n), p)
        est = max(est, alternative)
        product = rp(norm * est, p)
        if product == 0 or product >= Fr(2) ** (p - 1):
            raise Stop("Condition")
    u = [Fr(0)] * case.n
    for g, v in case.fixed.items():
        u[g] = rp(v, p)
    rhs = []
    for g in free:
        s = case.f.get(g, Fr(0))
        for c in case.fixed:
            if u[c] != 0 and (g, c) in K:
                s -= K[(g, c)] * u[c]
        rhs.append(rp(s, p))
    uf = solve(rhs) if n else []

    def residual(uu):
        """Exact r_i = f_i - sum_j K^q_ij u_j per free row, the coalesced
        denominator (M03: |f| + sum |K^q_ij u_j|), the bounded one
        (|f| + sum A-bar^q_ij |u_j|) and m_i."""
        out = []
        for g in free:
            r = case.f.get(g, Fr(0))
            dc = abs(case.f.get(g, Fr(0)))
            db = dc
            cnt = 0
            for j in range(case.n):
                kij = Kq.get((g, j))
                if kij is None or kij == 0 or uu[j] == 0:
                    continue
                cnt += 1
                r -= kij * uu[j]
                dc += abs(kij * uu[j])
            for j in range(case.n):
                a = Abarq.get((g, j))
                if a and uu[j] != 0:
                    db += a * abs(uu[j])
            out.append((r, dc, db, 2 * cnt + 2))
        return out

    if W_Q == "3p+64" and not ref:
        qw = min(p + p // 2 + 64, 1024) if p < 1024 else 1024
        _, _, _, membersw, dblocksw = assemble(case.m, qw)
    else:
        qw, membersw, dblocksw = q, membersq, dblocksq
    Kc = {}
    if not ref:
        for mm, op in zip(case.m["members"], membersw):
            dofs = member_dofs(mm)
            for a in range(12):
                for b in range(12):
                    v = op["ke"][(a, b)]
                    if v:
                        Kc[(dofs[a], dofs[b])] = Kc.get((dofs[a], dofs[b]), Fr(0)) + v
        for s_ in case.m["springs"]:
            d_ = 6 * s_["node"] + s_["c"]
            Kc[(d_, d_)] = Kc.get((d_, d_), Fr(0)) + Fr(s_["k"])
        for base, blk in dblocksw:
            for a in range(3):
                for b in range(3):
                    if blk[a][b]:
                        Kc[(base + a, base + b)] = Kc.get((base + a, base + b), Fr(0)) + blk[a][b]

    def residual_contrib(uu):
        """V4-R1: r_i = f_i - sum over every element entry, spring and
        directional-block entry of its contribution times u_j: one exact
        expansion, never rounded (the per-(i, j) sums above are exact)."""
        out = []
        for g in free:
            r = case.f.get(g, Fr(0))
            for j in range(case.n):
                v = Kc.get((g, j))
                if v and uu[j] != 0:
                    r -= v * uu[j]
            out.append(r)
        return out

    def ratio(r, d, mm):
        try:
            return float(abs(r) * (2 ** p - mm) / (64 * mm * d)) if d else (0.0 if r == 0 else float("inf"))
        except OverflowError:
            return float("inf")
    def bounded_worst(rows_):
        """The exact worst bounded ratio |r|(2^p - m)/(64 m d^b) of a state."""
        w = Fr(0)
        for x in rows_:
            num = abs(x[0]) * (2 ** p - x[3])
            if num == 0:
                continue
            if x[2] == 0:
                return None
            w = max(w, num / (64 * x[3] * x[2]))
        return w
    corrections = 0
    prior = None
    worst = 0.0
    seen = []                  # (uf, rows) of every evaluated state, in order
    gate_rows = []
    gate_state = None          # (index of the chosen state, states evaluated, accepted by)
    while n:
        for a, g in enumerate(free):
            u[g] = uf[a]
        rows_ = residual(u)
        seen.append((list(uf), rows_))
        gate_rows = rows_
        # the refinement driver: the coalesced ratio (GATE "k4" and "hybrid") or
        # the bounded one (GATE "absolute")
        dsel = (lambda x: x[1]) if GATE in ("k4", "hybrid") else (lambda x: x[2])
        drive_pass = all(abs(x[0]) * (2 ** p - x[3]) <= 64 * x[3] * dsel(x) for x in rows_)
        worst = max((ratio(x[0], dsel(x), x[3]) for x in rows_), default=0.0)
        if drive_pass:
            gate_state = (len(seen) - 1, len(seen), "coalesced" if GATE != "absolute" else "bounded")
            break
        if corrections == 3 or (prior is not None and worst >= prior):
            if GATE == "hybrid":
                # V4-R5: the bounded test on the best state seen (smallest worst
                # bounded ratio, the earliest on a tie); R2: on the last state
                if BEST_STATE:
                    cands = [(bounded_worst(r_), k) for k, (_, r_) in enumerate(seen)]
                    cands = [(w_, k) for w_, k in cands if w_ is not None]
                    pick = min(cands)[1] if cands else len(seen) - 1
                else:
                    pick = len(seen) - 1
                uf_b, rows_b = seen[pick]
                if all(abs(x[0]) * (2 ** p - x[3]) <= 64 * x[3] * x[2] for x in rows_b):
                    uf = uf_b
                    gate_rows = rows_b
                    gate_state = (pick, len(seen), "bounded")
                    worst = max((ratio(x[0], x[2], x[3]) for x in rows_b), default=0.0)
                    break
            raise Stop("ResidualGate(%.3g)" % worst)
        prior = worst
        delta = solve([rp(x[0], p) for x in rows_])
        uf = [rp(v + dv, p) for v, dv in zip(uf, delta)]
        corrections += 1
    for a, g in enumerate(free):
        u[g] = uf[a]
    if not ref:
        for dof, val in SEED.get(case.m["name"], []):
            u[dof] = rp(u[dof] + val, p)
    vals = recover(case, p, K, members, dblocks, u, with_loads=True)
    st = dict(p=p, values=vals, u=u, corrections=corrections, gate_worst=worst, gate_state=gate_state)
    st["E"] = formation_scale(case, p, Abar, members, dblocks, u)
    if estimate and not ref:
        # S2 / V4-R1 / V4-R2: the verification estimate's residual on the final state
        ux = list(u)
        if W_PRESC == "exact":
            for g_, v_ in case.fixed.items():
                ux[g_] = v_
        rfull = []
        if not n:
            rhat = []
        elif W_RES == "contrib":
            rfull = residual_contrib(ux)
            rhat = [rp(r_, p) for r_ in rfull]
        elif W_RES == "assembled":
            rfull = [x[0] for x in residual(u)]
            rhat = [rp(r_, p) for r_ in rfull]
        else:
            rfull = [x[0] for x in gate_rows]
            rhat = [rp(r_, p) for r_ in rfull]
        u_for_charge = ux
        delta = solve(rhat) if n else []
        dfull = [Fr(0)] * case.n
        for a, g in enumerate(free):
            dfull[g] = delta[a]
        st["W"] = [abs(v) for v in recover(case, p, K, members, dblocks, dfull, with_loads=False)]
        st.update(r4_ingredients(case, p, qw, est, scale, Abar, members, dblocks, u_for_charge, rfull, Kc, dfull))
    return st


def bounded_rows(case, p, Abar, members, dblocks, w):
    """Per row, the bounded recovery row applied to the nonnegative operand w,
    without the ledger: sum_j a-bar_qj w_j (E's expansion with w in place of |u|
    and f omitted). With w = S on the free DOFs and 0 elsewhere it is
    ||a-bar_q S||_1."""
    saved = (case.f, case.f_terms)
    case.f, case.f_terms = {}, {}
    try:
        return formation_scale(case, p, Abar, members, dblocks, w)
    finally:
        case.f, case.f_terms = saved


def r4_ingredients(case, p, qw, est, scale, Abar, members, dblocks, ux, r, Kc, dfull):
    """R4 section 4.1.6.3 items 8-10 at the verification state (precision p = P):
    theta, the per-row charge C_q (force and moment rows) and, for translation and
    rotation rows, W+ + C. All exact Fractions. K is block-diagonal by body (bodies
    are the element graph's components; springs and directional blocks do not
    connect them), so every norm is taken over the row's own body; est, the
    estimate of ||K~^-1||_1 over the whole matrix, bounds each block's."""
    free = case.free
    n = len(free)
    out = dict(qw=qw, est=est, theta=Fr(0), g_ok=True, C=[Fr(0)] * len(case.layout),
               Wplus_trro=[Fr(0)] * len(case.layout), charge_terms=None)
    gmax = max((op["g"] for op in members), default=1)
    out["g_ok"] = gmax <= Fr(2) ** (p - 16)
    if not n or est is None:
        return out
    F = F_ALLOW
    s = {g: Fr(2) ** sc for g, sc in zip(free, scale)}
    fset = set(free)
    bof = lambda g: case.body_of[g // 6]
    nb = case.nbody
    SAu = [Fr(0)] * nb
    for g in free:
        acc = Fr(0)
        for j in range(case.n):
            a = Abar.get((g, j))
            if a and ux[j] != 0:
                acc += a * abs(ux[j])
        SAu[bof(g)] = max(SAu[bof(g)], s[g] * acc)
    SAS = [Fr(0)] * nb
    for j in free:
        acc = Fr(0)
        for i in free:
            a = Abar.get((i, j))
            if a:
                acc += s[i] * a
        SAS[bof(j)] = max(SAS[bof(j)], acc * s[j])
    Sr = [Fr(0)] * nb
    for g, ri in zip(free, r):
        Sr[bof(g)] = max(Sr[bof(g)], s[g] * abs(ri))
    Sr2 = [Fr(0)] * nb
    for g, ri in zip(free, r):
        acc = ri
        for j in free:
            v = Kc.get((g, j))
            if v and dfull[j] != 0:
                acc -= v * dfull[j]
        Sr2[bof(g)] = max(Sr2[bof(g)], s[g] * abs(acc))
    Sid = [Fr(0)] * nb
    for g in free:
        Sid[bof(g)] = max(Sid[bof(g)], abs(dfull[g]) / s[g])
    t1, t2, t3 = [], [], []
    for b in range(nb):
        Nu = SAu[b] + 2 * F * est * SAS[b] * Sr[b]
        t1.append(Fr(2) ** (7 - qw) * F * est * Nu)
        t2.append(70 * Fr(2) ** (-p) * Sid[b])
        t3.append(3 * F * est * Sr2[b])
    out["theta"] = F * est * Fr(2) ** (7 - p) * max(SAS)
    out["charge_terms"] = dict(t1=[float(x) for x in t1], t2=[float(x) for x in t2], t3=[float(x) for x in t3],
                               SAu=[float(x) for x in SAu], SAS=[float(x) for x in SAS], est=float(est))
    w = [s[g] if g in fset else Fr(0) for g in range(case.n)]
    aS = bounded_rows(case, p, Abar, members, dblocks, w)
    C = [Fr(0)] * len(case.layout)
    Wp = [Fr(0)] * len(case.layout)
    for idx, (rid, kind, body, inp) in enumerate(case.layout):
        if kind in ("fo", "mo"):
            if aS[idx] is not None:
                C[idx] = aS[idx] * (t1[body] + t2[body] + t3[body])
        elif rid[0] == "u" and not inp:
            g = 6 * rid[1] + rid[2]
            C[idx] = s[g] * (t1[body] + t3[body])
            Wp[idx] = abs(dfull[g]) + C[idx]
    for idx, (rid, kind, body, inp) in enumerate(case.layout):
        if rid[0] == "mag":
            node = rid[1]
            acc = Fr(0)
            for c in range(3):
                g = 6 * node + c
                if g in fset:
                    acc += abs(dfull[g]) + s[g] * (t1[body] + t3[body])
                elif g in case.fixed:
                    acc += abs(rp(case.fixed[g], p) - case.fixed[g])
            Wp[idx] = acc      # the magnitude's own rounding is added in the stop rule
    out["C"] = C
    out["Wplus_trro"] = Wp
    return out


def recover(case, p, K, members, dblocks, u, with_loads=True):
    """recover.rs at p. With with_loads=False the ledger is omitted (the linear
    recovery of a correction)."""
    m = case.m
    vals = []
    for g in range(case.n):
        vals.append(u[g])
    for node in range(case.nn):
        b = 6 * node
        sq = rp(sum((u[b + k] * u[b + k] for k in range(3)), Fr(0)), p)
        vals.append(sqrt_p(sq, p) if sq else Fr(0))
    Qs = []
    ends = []
    for mm, op in zip(m["members"], members):
        dofs = member_dofs(mm)
        d = [Fr(0)] * 12
        for blk in range(4):
            for r in range(3):
                d[3 * blk + r] = rp(sum((op["axes"][r][c] * u[dofs[3 * blk + c]] for c in range(3)), Fr(0)), p)
        inv = op["inv"]
        e = [Fr(0)] * 6
        for idx, (plus, minus) in enumerate(((6, 0), (9, 3))):
            e[idx] = rp(d[plus] - d[minus], p)
        for idx, rot, tr, sign in ((2, 5, 1, True), (3, 11, 1, True), (4, 4, 2, False), (5, 10, 2, False)):
            s = d[rot]
            if sign:
                s = s + inv * d[tr] - inv * d[tr + 6]
            else:
                s = s - inv * d[tr] + inv * d[tr + 6]
            e[idx] = rp(s, p)
        bz, by = op["bz"], op["by"]
        coeffs = [[(op["axial"], 0)], [(op["torsion"], 1)], [(4 * bz, 2), (2 * bz, 3)], [(2 * bz, 2), (4 * bz, 3)],
                  [(4 * by, 4), (2 * by, 5)], [(2 * by, 4), (4 * by, 5)]]
        Q = [rp(sum((c * e[at] for c, at in t), Fr(0)), p) for t in coeffs]
        vy = rp(inv * Q[2] + inv * Q[3], p)
        vz = -rp(inv * Q[4] + inv * Q[5], p)
        ends.append([-Q[0], vy, vz, -Q[1], Q[4], Q[2], Q[0], -vy, -vz, Q[1], Q[5], Q[3]])
        Qs.append(Q)
    for acts in ends:
        vals.extend(acts)
    ids = [mm["id"] for mm in m["members"]]
    for st in m["stations"]:
        k = ids.index(st["member"])
        Q, acts = Qs[k], ends[k]
        t = Fr(st["t"])
        my = rp(t * Q[5] + t * Q[4] - Q[4], p)
        mz = rp(t * Q[3] + t * Q[2] - Q[2], p)
        vals.extend([acts[6], acts[7], acts[8], acts[9], my, mz])
    for s in m["springs"]:
        g = 6 * s["node"] + s["c"]
        vals.append(rp(-Fr(s["k"]) * u[g], p))
    for base, blk in dblocks:
        for a in range(3):
            vals.append(rp(-sum((blk[a][b] * u[base + b] for b in range(3)), Fr(0)), p))
    for g in sorted(case.fixed):
        s = Fr(0)
        for j in range(case.n):
            kij = K.get((g, j))
            if kij is not None and u[j] != 0:
                s += kij * u[j]
        if with_loads:
            s -= case.f.get(g, Fr(0))
        vals.append(rp(s, p))
    return vals


def formation_scale(case, p, Abar, members, dblocks, u):
    """E_q per row (Fractions; None for rows without E), revision section 4.1.6.2."""
    m = case.m
    au = [Fr(0) if (not E_PRESCRIBED and g in case.fixed) else abs(x) for g, x in enumerate(u)]
    E = [None] * (case.n + case.nn)
    Qs, ends = [], []
    for mm, op in zip(m["members"], members):
        dofs = member_dofs(mm)
        # d-bar per block and local axis
        db = [Fr(0)] * 12
        for blk in range(4):
            for r in range(3):
                if E_OPERATOR == "entrywise":
                    db[3 * blk + r] = rp(sum((abs(op["axes"][r][c]) * au[dofs[3 * blk + c]] for c in range(3)), Fr(0)), p)
                else:
                    db[3 * blk + r] = rp(sum((au[dofs[3 * blk + c]] for c in range(3)), Fr(0)), p)
        inv = abs(op["inv"])
        eb = [rp(db[6] + db[0], p), rp(db[9] + db[3], p)]
        for rot, tr in ((5, 1), (11, 1), (4, 2), (10, 2)):
            eb.append(rp(db[rot] + inv * db[tr] + inv * db[tr + 6], p))
        bz, by = abs(op["bz"]), abs(op["by"])
        coeffs = [[(abs(op["axial"]), 0)], [(abs(op["torsion"]), 1)], [(4 * bz, 2), (2 * bz, 3)], [(2 * bz, 2), (4 * bz, 3)],
                  [(4 * by, 4), (2 * by, 5)], [(2 * by, 4), (4 * by, 5)]]
        Qb = [rp(sum((c * eb[at] for c, at in t), Fr(0)), p) for t in coeffs]
        vy = rp(inv * Qb[2] + inv * Qb[3], p)
        vz = rp(inv * Qb[4] + inv * Qb[5], p)
        g = op["g"]
        acts = [Qb[0], vy, vz, Qb[1], Qb[4], Qb[2], Qb[0], vy, vz, Qb[1], Qb[5], Qb[3]]
        ends.append([g * x for x in acts])
        Qs.append(Qb)
    for acts in ends:
        E.extend(acts)
    ids = [mm["id"] for mm in m["members"]]
    for st in m["stations"]:
        k = ids.index(st["member"])
        Qb, acts = Qs[k], ends[k]
        g = members[k]["g"]
        t = abs(Fr(st["t"]))
        E.extend([acts[6], acts[7], acts[8], acts[9], g * rp(t * Qb[5] + t * Qb[4] + Qb[4], p),
                  g * rp(t * Qb[3] + t * Qb[2] + Qb[2], p)])
    for s in m["springs"]:
        g = 6 * s["node"] + s["c"]
        E.append(rp(abs(Fr(s["k"])) * au[g], p))
    for base, blk in dblocks:
        for a in range(3):
            E.append(rp(sum((abs(blk[a][b]) * au[base + b] for b in range(3)), Fr(0)), p))
    for g in sorted(case.fixed):
        f = abs(case.f.get(g, Fr(0))) if E_LEDGER == "net" else case.f_terms.get(g, Fr(0))
        s = f + sum((Abar.get((g, j), Fr(0)) * au[j] for j in range(case.n)), Fr(0))
        E.append(rp(s, p))
    return E


# ---------------------------------------------------------------- published E, e-hat, V, Phi

def body_E(case, st):
    """E(body, kind) for fo and mo: the maximum over the body's force or moment
    rows (none is input-derived), unpublishable rows included (revision
    section 4.1.6.2 item 4), rounded upward to binary64."""
    out = [{"fo": Fr(0), "mo": Fr(0)} for _ in range(case.nbody)]
    for (rid, kind, body, inp), v, e in zip(case.layout, st["values"], st["E"]):
        if inp or kind not in ("fo", "mo") or e is None:
            continue
        if e > out[body][kind]:
            out[body][kind] = e
    return [{k: fl_up(v) for k, v in b.items()} for b in out]


def e_hat(case, Eb):
    """Item 6a's binary64 coupling of the published E bits (floats)."""
    out = []
    for b in range(case.nbody):
        fo, mo = Eb[b]["fo"], Eb[b]["mo"]
        L = case.extent[b]
        if E_COUPLED and L != 0.0:
            out.append(dict(fo=max(fo, mo / L), mo=max(mo, L * fo)))
        else:
            out.append(dict(fo=fo, mo=mo))
    return out


def phi512(eh):
    """fl_up(2^-438 * e-hat) as a binary64."""
    return fl_up(Fr(eh) / Fr(2) ** 438)


def body_kind_max(case, values):
    s = [{k: Fr(0) for k in KINDS} for _ in range(case.nbody)]
    for (rid, kind, body, inp), v in zip(case.layout, values):
        if inp or not publishable(v):
            continue
        a = abs(v)
        if a > s[body][kind]:
            s[body][kind] = a
    return s


def coupled_wide(s, L, P):
    if L == 0:
        return dict(s)
    Lb = Fr(L)
    return dict(tr=max(s["tr"], rp(Lb * s["ro"], P)), ro=max(s["ro"], rp(s["tr"] / Lb, P)),
                fo=max(s["fo"], rp(s["mo"] / Lb, P)), mo=max(s["mo"], rp(Lb * s["fo"], P)))


# options: "current"; "iii" (V + Phi at 512); "fin" (iii + estimator W <= 2^(6-2p) e-hat)
FLOOR_AT = {512}          # mutants M3 (every p), M4 (none), M9 ({256})
FLOOR_EXP_FN = lambda p: p - 74   # Phi = fl_up(2^-(p-74) e-hat) at p in FLOOR_AT (2^-438 at 512); mutant M8: 2p - 74
USE_V = True              # mutant M1


def stop_rule(case, cand, ver, p, option):
    P2 = 2 * p
    s = body_kind_max(case, ver["values"])
    Eb = body_E(case, ver)
    eh = e_hat(case, Eb)
    sstar = []
    phis = []
    for b in range(case.nbody):
        c = coupled_wide(s[b], case.extent[b], P2)
        ph = {}
        if option in ("iii", "fin", "r4") and p in FLOOR_AT:
            for k in ("fo", "mo"):
                ph[k] = fl_up(Fr(eh[b][k]) / Fr(2) ** FLOOR_EXP_FN(p))
                c[k] = max(c[k], Fr(ph[k]))
        sstar.append(c)
        phis.append(ph)
    info = dict(Eb=Eb, eh=eh, phis=phis, reason=None, w_over_v=0.0)
    Ms = []
    for idx, ((rid, kind, body, inp), qp, q2) in enumerate(zip(case.layout, cand["values"], ver["values"])):
        M = max(abs(q2), sstar[body][kind])
        Ms.append(M)
        lhs = abs(qp - q2)
        if option in ("iii", "fin", "r4") and USE_V and kind in ("fo", "mo"):
            V = Fr(eh[body][kind]) * Fr(2) ** (8 - P2)
            lhs += V
        if option == "r4" and TRRO == "measured" and kind in ("tr", "ro") and not inp:
            # R4: translation and rotation rows carry the verification's measured bound
            vq = ver["Wplus_trro"][idx]
            if rid[0] == "mag":
                vq += abs(q2) * Fr(2) ** (1 - P2)
            lhs += vq
        if lhs > EPS * M:
            info["reason"] = ("stop_rule", rid)
            return False, info
    if option in ("fin", "r4") and W_TEST:
        for idx, ((rid, kind, body, inp), w) in enumerate(zip(case.layout, ver["W"])):
            if kind not in ("fo", "mo"):
                continue
            V = Fr(eh[body][kind]) * Fr(2) ** (8 - P2)
            if w and V == 0:
                info["reason"] = ("verification_estimate", rid)
                return False, info
            if V:
                r = float(w / V)
                info["w_over_v"] = max(info["w_over_v"], r)
                if w > V / 4:
                    info["reason"] = ("verification_estimate", rid)
                    return False, info
    if option == "r4":
        info["theta"] = float(ver["theta"])
        if THETA and ver["theta"] > Fr(1, 2):
            info["reason"] = ("theta", None)
            return False, info
        if G_CHECK and not ver["g_ok"]:
            info["reason"] = ("g_validity", None)
            return False, info
        worst = 0.0
        for idx, ((rid, kind, body, inp), c) in enumerate(zip(case.layout, ver["C"])):
            if kind not in ("fo", "mo") or not c:
                continue
            if p < 512:
                allow = 60 * Fr(2) ** (-P2) * Fr(eh[body][kind])
            else:
                allow = Fr(2) ** -22 * EPS * Ms[idx]
            try:
                ratio_ = float(c / allow) if allow else float("inf")
            except OverflowError:
                ratio_ = float("inf")
            worst = max(worst, ratio_)
            if CHARGE and c > allow:
                info["reason"] = ("charge", rid)
                info["charge_over_allowance"] = worst
                return False, info
        info["charge_over_allowance"] = worst
    return True, info


def charge_report(case, cand, ver, p):
    """R4: the worst C_q / allowance over force and moment rows for the pair
    (cand at p, ver at 2p), whatever the other tests decide; and theta."""
    P2 = 2 * p
    s = body_kind_max(case, ver["values"])
    eh = e_hat(case, body_E(case, ver))
    worst = 0.0
    for idx, ((rid, kind, body, inp), q2, c) in enumerate(zip(case.layout, ver["values"], ver["C"])):
        if kind not in ("fo", "mo") or not c:
            continue
        if p < 512:
            allow = 60 * Fr(2) ** (-P2) * Fr(eh[body][kind])
        else:
            cw = coupled_wide(s[body], case.extent[body], P2)
            Mk = max(cw[kind], fl_up(Fr(eh[body][kind]) / Fr(2) ** FLOOR_EXP_FN(p)))
            allow = Fr(2) ** -22 * EPS * max(abs(q2), Fr(Mk))
        try:
            worst = max(worst, float(c / allow) if allow else float("inf"))
        except OverflowError:
            worst = float("inf")
    return worst, float(ver["theta"])


def schedule(case, option, cache):
    PRE = [128, 256, 512, 1024]

    def get(P):
        if P not in cache:
            try:
                cache[P] = solve_at(case, P)
            except Stop as s:
                cache[P] = s
        return cache[P]
    attempts = []
    solved = set()
    c = 0
    pending = None
    while c < 3:
        p = PRE[c]
        cand = pending if pending is not None else get(p)
        solved.add(p)
        pending = None
        if isinstance(cand, Stop):
            attempts.append((p, "failed:" + cand.what))
            c += 1
            continue
        ver = get(PRE[c + 1])
        solved.add(PRE[c + 1])
        if isinstance(ver, Stop):
            attempts.append((p, "rejected:verification_failed"))
            attempts.append((PRE[c + 1], "failed:" + ver.what))
            c += 2
            continue
        ok, info = stop_rule(case, cand, ver, p, option)
        if ok:
            attempts.append((p, "accepted"))
            attempts.append((PRE[c + 1], "verified"))
            return dict(selected=p, attempts=attempts, cand=cand, ver=ver, info=info, solved=sorted(solved))
        attempts.append((p, "rejected:%s:%s" % info["reason"]))
        pending = ver
        c += 1
    return dict(selected=None, attempts=attempts, solved=sorted(solved))


# ---------------------------------------------------------------- classification and G5a

def coupled_f64(s, L):
    if L == 0.0:
        return dict(s)
    return dict(tr=max(s["tr"], L * s["ro"]), ro=max(s["ro"], s["tr"] / L),
                fo=max(s["fo"], s["mo"] / L), mo=max(s["mo"], L * s["fo"]))


def classify(case, result, option):
    p = result["selected"]
    vals = result["cand"]["values"]
    pub = [fl(v) if publishable(v) else None for v in vals]
    s = [{k: 0.0 for k in KINDS} for _ in range(case.nbody)]
    for (rid, kind, body, inp), v in zip(case.layout, pub):
        if not inp and v is not None:
            s[body][kind] = max(s[body][kind], abs(v))
    Eb = result["info"]["Eb"]
    eh = result["info"]["eh"]
    scales = []
    for b in range(case.nbody):
        c = coupled_f64(s[b], case.extent[b])
        if option in ("iii", "fin", "r4") and p in FLOOR_AT:
            for k in ("fo", "mo"):
                c[k] = max(c[k], result["info"]["phis"][b][k])
        scales.append(c)
    classes = []
    for (rid, kind, body, inp), v in zip(case.layout, pub):
        if inp:
            classes.append(("input_derived", None))
            continue
        if v is None:
            classes.append(("unpublishable", None))
            continue
        S = scales[body][kind]
        if S < 2.0 ** -988 or abs(v) < R_FLOOR * S:
            classes.append(("absolute", fl_up(Fr(S) / Fr(2 ** 64))))
        else:
            classes.append(("relative", None))
    g5a = g5a_checks(case, pub, s, Eb, eh)
    return pub, classes, scales, g5a


def g5a_checks(case, pub, s_uncoupled, Eb, eh):
    """S7 and N10: the zero rule, the sanity bound and the published-data lower
    bound, in binary64 exactly as pinned. Returns a list of failures."""
    fails = []
    sc = [coupled_f64(s_uncoupled[b], case.extent[b]) for b in range(case.nbody)]
    for b in range(case.nbody):
        for k in ("fo", "mo"):
            if Eb[b][k] == 0.0:
                for (rid, kind, body, inp), v in zip(case.layout, pub):
                    if body == b and kind == k and not inp and v not in (None, 0.0):
                        fails.append(("zero_rule", b, k, rid))
            if not (eh[b][k] * ONE_PLUS_2M40 >= sc[b][k]):
                fails.append(("sanity", b, k))
    # lower bound from published displacements and the receipt's k_a, k_t
    uv = {}
    for (rid, kind, body, inp), v in zip(case.layout, pub):
        if rid[0] == "u":
            uv[(rid[1], rid[2])] = abs(v) if v is not None else 0.0
    lb = [{"fo": 0.0, "mo": 0.0} for _ in range(case.nbody)]
    for mm in case.m["members"]:
        b = case.body_of[mm["i"]]
        xi, xj = case.m["nodes"][mm["i"]], case.m["nodes"][mm["j"]]
        dx, dy, dz = (xj[0] - xi[0]), (xj[1] - xi[1]), (xj[2] - xi[2])
        L = math.sqrt((dx * dx + dy * dy) + dz * dz)
        ka = (mm["E"] * mm["A"]) / L
        kt = (mm["G"] * mm["J"]) / L

        def nsum(node, off):
            return (uv[(node, off)] + uv[(node, off + 1)]) + uv[(node, off + 2)]
        Nt = nsum(mm["i"], 0) + nsum(mm["j"], 0)
        Nr = nsum(mm["i"], 3) + nsum(mm["j"], 3)
        for k, N, kk, S in (("fo", Nt, ka, sc[b]["tr"]), ("mo", Nr, kt, sc[b]["ro"])):
            if N > 2.0 ** -59 * S:
                lb[b][k] = max(lb[b][k], kk * (N - 2.0 ** -60 * S))
    for b in range(case.nbody):
        for k in ("fo", "mo"):
            if not (eh[b][k] * ONE_PLUS_2M40 >= lb[b][k]):
                fails.append(("lower_bound", b, k, eh[b][k], lb[b][k]))
    return fails


def honesty(case, pub, classes, ref):
    """Worst claim ratio against the P = 2048 reference, allowing the
    reference's own resolution 2^(8-2048) E-hat_ref as slack."""
    Ebr = e_hat(case, body_E(case, ref))
    worst = (0.0, None)
    counts = dict(absolute=0, relative=0, input_derived=0, unpublishable=0)
    for (rid, kind, body, inp), q, (cls, b), qs in zip(case.layout, pub, classes, ref["values"]):
        counts[cls] += 1
        if cls not in ("absolute", "relative"):
            continue
        sc = Ebr[body]["fo"] if kind == "fo" else Ebr[body]["mo"] if kind == "mo" else 0.0
        slack = Fr(sc) * Fr(2) ** (8 - 2048) if sc else Fr(0)
        err = max(Fr(0), abs(Fr(q) - qs) - slack)
        try:
            if cls == "absolute":
                r = float(err / Fr(b)) if b else (0.0 if err == 0 else float("inf"))
            else:
                r = float(err / (Fr(1, 10 ** 9) * abs(qs))) if qs else (0.0 if err == 0 else float("inf"))
        except OverflowError:
            r = float("inf")
        if r > worst[0]:
            worst = (r, rid)
    return worst, counts
