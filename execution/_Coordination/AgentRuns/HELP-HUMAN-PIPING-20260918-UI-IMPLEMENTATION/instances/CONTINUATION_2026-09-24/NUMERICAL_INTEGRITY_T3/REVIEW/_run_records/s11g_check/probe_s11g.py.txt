#!/usr/bin/env python3
"""V1 S11G_CHECK: independent checks of D1's S11-G note (standard library only; no product code built or run).

Usage (from T3/):  nice -n 19 python3 <this> <v1_dir> <out.json>
<v1_dir> holds V1's probe_d5_check.py (REVIEW/_run_records/d5_check/probe_d5_check.py.txt), used only for
its binary64 frame and local stiffness emulation (product_frame, local_K, section).

Inputs read from T3/: REFERENCES/references.json (R1), DETECTION/results.json (P1 at main c61a540ea).
Nothing from D1's s11g_forecast.py is imported.

Part A   UDL-W1e5/-W1e8/-W1e80: SP's fixed-end rotation terms written from P/core/solver/straight_pipe
         src/lib.rs `spanned_uniform_equivalent_terms` (operation order as the source), the exact defect of
         each term against the same formula in rationals on the same held operands (q, L, a, b), the
         exact signed sum, and the DESIGN §4.1.6 coupled free-row scale; the represented net compared bit
         for bit with I4's post-S11-F ledger row.
Part A2  S11-F probe A (one 2 m cantilever, loads (G, 0.3, -G) and (0.3, G, -G), G = 1e8, 1e80).
Part C   Row-relative realized error of every published row in P1's passing case-modes against R1's exact
         value, restricted to rows at or above the DESIGN floor 2^-34 * S*, with S* the DESIGN §4.1.6
         coupled body scale formed from R1's exact values (translation/extension with rotation/twist; force
         with moment; L_b the body's bounding-box diagonal). This measures the INPLANE mechanism
         (a small quantity in a large body, binary64 solve noise) wherever it occurs.
Part R   Candidate R-b re-implemented from the note's definition: per member end,
         B = gamma_16 * sum_k |Kloc_rk| * sum_j |T_kj u_j| over the two bending rows, q = published |Mb|,
         fires when B > 1e-9 |q| and |q| > 2^10 B; u from P1's published displacements. Each firing row is
         classified by its realized row-relative error and by the DESIGN floor.
"""
import json
import math
import sys
from fractions import Fraction as Fr

sys.path.insert(0, sys.argv[1])
import probe_d5_check as p  # noqa: E402

U = 2.0 ** -53
R_FLOOR = Fr(1, 2 ** 34)


def gamma(k):
    return k * U / (1 - k * U)


def num(x):
    """R1 literals: decimal, a/b, or m*2^k (also m*10^k)."""
    x = str(x)
    for base in ("2", "10"):
        tag = "*" + base + "^"
        if tag in x:
            mm, k = x.split(tag)
            return Fr(mm) * Fr(int(base)) ** int(k)
    return Fr(x)


# ------------------------------------------------------------------ Part A
def sp_rot(q, L, a, b):
    """rotation_i, rotation_j exactly as spanned_uniform_equivalent_terms writes them (binary64)."""
    ri = q * L * L * ((0.5 * b * b - (2.0 / 3.0) * b ** 3 + 0.25 * b ** 4) - (0.5 * a * a - (2.0 / 3.0) * a ** 3 + 0.25 * a ** 4))
    rj = q * L * L * (((-b ** 3 / 3.0) + 0.25 * b ** 4) - ((-a ** 3 / 3.0) + 0.25 * a ** 4))
    return ri, rj


def sp_rot_exact(q, L, a, b):
    q, L, a, b = Fr(q), Fr(L), Fr(a), Fr(b)
    P = lambda x: x * x / 2 - Fr(2, 3) * x ** 3 + x ** 4 / 4
    Q = lambda x: -x ** 3 / 3 + x ** 4 / 4
    return q * L * L * (P(b) - P(a)), q * L * L * (Q(b) - Q(a))


def sp_trans(q, L, a, b):
    ti = q * L * ((b - b ** 3 + 0.5 * b ** 4) - (a - a ** 3 + 0.5 * a ** 4))
    tj = q * L * ((b ** 3 - 0.5 * b ** 4) - (a ** 3 - 0.5 * a ** 4))
    return ti, tj


def sp_trans_exact(q, L, a, b):
    q, L, a, b = Fr(q), Fr(L), Fr(a), Fr(b)
    return q * L * ((b - b ** 3 + b ** 4 / 2) - (a - a ** 3 + a ** 4 / 2)), q * L * ((b ** 3 - b ** 4 / 2) - (a ** 3 - a ** 4 / 2))


def part_a(ref):
    i4_terms = {"RF-CANCEL-UDL-W1e8": (0.2, -33333333.333333325, 33333333.625000015)}
    out = {}
    for cid in ("RF-CANCEL-UDL-W1e5", "RF-CANCEL-UDL-W1e8", "RF-CANCEL-UDL-W1e80"):
        m = ref[cid]["model"]
        wA = float(num(m["member_uniform_loads_N_per_m_global"]["A"][1]))
        wB = float(num(m["member_uniform_loads_N_per_m_global"]["B"][1]))
        M = float(num(m["loads"]["S1"]["M"][2]))
        L = 2.0
        _, rjA = sp_rot(wA, L, 0.0, 1.0)
        riB, _ = sp_rot(wB, L, 0.0, 1.0)
        _, xjA = sp_rot_exact(wA, L, 0.0, 1.0)
        xiB, _ = sp_rot_exact(wB, L, 0.0, 1.0)
        terms = (M, rjA, riB)
        net_rep = Fr(M) + Fr(rjA) + Fr(riB)
        E = (Fr(rjA) - xjA) + (Fr(riB) - xiB)        # exact signed defect (inputs: zero)
        n_int = Fr(M) + xjA + xiB
        # free rows of the body: S1 RX, RY, RZ; only S1.RZ loaded -> F = 0, S*_moment = M_free = |n_int|
        thr = Fr(1, 10 ** 9) * abs(n_int)
        rec = {"terms": terms, "net_represented_rounded": float(net_rep), "intended_net": str(n_int),
               "exact_defect": float(E), "threshold": float(thr), "stat_over_threshold": float(abs(E) / thr),
               "fires": abs(E) > thr}
        if cid in i4_terms:
            rec["matches_I4_ledger_terms_bitwise"] = terms == i4_terms[cid]
            rec["net_equals_0.4916666902601719"] = float(net_rep) == 0.4916666902601719
        # restrained rows (reaction side) against the smaller free-row scale (conservative)
        tiA, tjA = sp_trans(wA, L, 0.0, 1.0)
        tiB, tjB = sp_trans(wB, L, 0.0, 1.0)
        eiA, ejA = sp_trans_exact(wA, L, 0.0, 1.0)
        eiB, ejB = sp_trans_exact(wB, L, 0.0, 1.0)
        riA, _ = sp_rot(wA, L, 0.0, 1.0)
        _, rjB = sp_rot(wB, L, 0.0, 1.0)
        xiA, _ = sp_rot_exact(wA, L, 0.0, 1.0)
        _, xjB = sp_rot_exact(wB, L, 0.0, 1.0)
        rows = {"S0.UY": [(tiA, eiA)], "S0.RZ": [(riA, xiA)], "S1.UY": [(tjA, ejA), (tiB, eiB)],
                "S2.UY": [(tjB, ejB)], "S2.RZ": [(rjB, xjB)]}
        rr = {}
        for k, tl in rows.items():
            d = sum(Fr(t) - x for t, x in tl)
            ni = sum(x for _, x in tl)
            th = Fr(1, 10 ** 9) * max(abs(ni), abs(n_int))
            rr[k] = {"defect": float(d), "threshold_vs_free_scale": float(th), "fires": abs(d) > th}
        rec["restrained_rows"] = rr
        # the a-priori alternative as D1 states it (c = 16 on the monomial magnitudes)
        apri = 16 * U * (abs(wA) * L * L * (1 / 3 + 1 / 4) + abs(wB) * L * L * (0.5 + 2 / 3 + 0.25))
        rec["apriori_c16_over_threshold"] = apri / float(thr)
        out[cid] = rec
    return out


def part_a2():
    out = {}
    L = 2.0
    for g in (1e8, 1e80):
        for name, qs in (("GnG", (g, 0.3, -g)), ("nGG", (0.3, g, -g))):
            rows = {}
            for key, fn, fx, slot in (("tip.UY", sp_trans, sp_trans_exact, 1), ("tip.RZ", sp_rot, sp_rot_exact, 1)):
                vals = [fn(q, L, 0.0, 1.0)[slot] for q in qs]
                exs = [fx(q, L, 0.0, 1.0)[slot] for q in qs]
                indiv = [float(Fr(v) - x) for v, x in zip(vals, exs)]
                d = sum(Fr(v) - x for v, x in zip(vals, exs))
                ni = sum(exs)
                rows[key] = {"individual_defects": indiv, "signed_sum": float(d), "net_intended": float(ni)}
            F, Mm = abs(rows["tip.UY"]["net_intended"]), abs(rows["tip.RZ"]["net_intended"])
            fo, mo = max(F, Mm / L), max(Mm, L * F)
            for key in rows:
                sc = fo if key.endswith("UY") else mo
                th = 1e-9 * max(abs(rows[key]["net_intended"]), sc)
                rows[key]["threshold"] = th
                rows[key]["margin"] = th / abs(rows[key]["signed_sum"]) if rows[key]["signed_sum"] else math.inf
                rows[key]["fires"] = abs(rows[key]["signed_sum"]) > th
            out[f"G={g:g} {name}"] = rows
    return out


# ------------------------------------------------------------------ Part C and R
KIND = {"translation": "t", "extension": "t", "rotation": "r", "twist": "r", "force": "F", "moment": "M"}


def body_scales(case_ref):
    m = case_ref["model"]
    pts = [[float(num(v)) for v in xyz] for xyz in m["nodes_m"].values()]
    ext = [max(q[k] for q in pts) - min(q[k] for q in pts) for k in range(3)]
    Lb = math.sqrt(sum(e * e for e in ext))
    mx = {"t": Fr(0), "r": Fr(0), "F": Fr(0), "M": Fr(0)}
    for e in case_ref["expected"]:
        k = KIND.get(e[2])
        if k is None or e[1] in (None, ""):
            continue
        try:
            v = abs(num(e[1]))
        except (ValueError, ZeroDivisionError):
            continue
        mx[k] = max(mx[k], v)
    Lbf = Fr(Lb)
    S = {"t": max(mx["t"], Lbf * mx["r"]), "r": max(mx["r"], mx["t"] / Lbf if Lb else Fr(0)),
         "F": max(mx["F"], mx["M"] / Lbf if Lb else Fr(0)), "M": max(mx["M"], Lbf * mx["F"])}
    return S, Lb


def part_c(ref, res):
    out = []
    for case in res["cases"]:
        cr = ref.get(case["id"])
        if cr is None or "nodes_m" not in cr.get("model", {}) or "expected" not in cr:
            continue
        S, Lb = body_scales(cr)
        for mode, x in case["modes"].items():
            if x.get("verdict") != "pass":
                continue
            hits = []
            for q in x.get("quantities", []):
                key, cls, exp, obs = q[0], q[1], q[2], q[3]
                k = KIND.get(cls)
                if k is None or not obs or obs[0] in (None, ""):
                    continue
                try:
                    e, o = num(exp), num(obs[0])
                except (ValueError, ZeroDivisionError):
                    continue
                if e == 0 or abs(e) < R_FLOOR * S[k]:
                    continue
                ratio = abs(o - e) / (Fr(1, 10 ** 9) * abs(e))
                if ratio > 1:
                    hits.append([key, cls, float(e), float(S[k]), float(ratio)])
            if hits:
                out.append({"case": case["id"], "family": case.get("family"), "mode": mode, "L_b": Lb,
                            "rows_above_floor_over_criterion": len(hits), "max_ratio": max(h[-1] for h in hits),
                            "rows": sorted(hits, key=lambda h: -h[-1])[:8]})
    return out


def member_end_rows(m, u):
    rows = []
    for (mid, a, b, sid) in m["members"]:
        xi = [float(num(v)) for v in m["nodes_m"][a]]
        xj = [float(num(v)) for v in m["nodes_m"][b]]
        d = [xj[k] - xi[k] for k in range(3)]
        yref = (0.0, 1.0, 0.0) if d[0] == 0 and d[1] == 0 else (0.0, 0.0, 1.0)
        R, L = p.product_frame(xi, xj, yref)
        s = m["sections"][sid]
        E = float(num(s["E"]))
        G = float(num(s["G"])) if "G" in s else E / (2.0 * (1.0 + float(num(s["nu"]))))
        sec = p.section(float(num(s["OD"])), float(num(s["ID"])), E, G)
        K = p.local_K(sec, L)
        ue = list(u[a]) + list(u[b])
        tabs = [sum(abs(R[r][c] * ue[3 * blk + c]) for c in range(3)) for blk in range(4) for r in range(3)]
        for end, (ry, rz) in (("i", (4, 5)), ("j", (10, 11))):
            B = gamma(16) * sum((abs(K[ry][k]) + abs(K[rz][k])) * tabs[k] for k in range(12))
            rows.append((f"Mb.{mid}.{end}", B))
    return rows


def part_r(ref, res):
    fired = []
    passing = 0
    for case in res["cases"]:
        cr = ref.get(case["id"])
        if cr is None or "members" not in cr.get("model", {}) or "expected" not in cr or "nodes_m" not in cr["model"]:
            continue
        S, _ = body_scales(cr)
        expd = {e[0]: e[1] for e in cr["expected"]}
        for mode, x in case["modes"].items():
            if x.get("verdict") != "pass":
                continue
            obs = {q[0]: q[3][0] for q in x.get("quantities", []) if q[3] and q[3][0] not in (None, "")}
            m = cr["model"]
            try:
                u = {nd: [float(num(obs[("u." if c[0] == "U" else "th.") + nd + "." + c])) for c in ("UX", "UY", "UZ", "RX", "RY", "RZ")]
                     for nd in m["nodes_m"]}
            except KeyError:
                continue
            passing += 1
            for key, B in member_end_rows(m, u):
                if key not in obs:
                    continue
                qo = abs(float(num(obs[key])))
                if B > 1e-9 * qo and qo > 1024 * B:
                    e = num(expd[key]) if expd.get(key) not in (None, "") else None
                    rr = float(abs(num(obs[key]) - e) / (Fr(1, 10 ** 9) * abs(e))) if e else None
                    fired.append({"case": case["id"], "mode": mode, "row": key, "q": qo, "B": B,
                                  "realized_row_relative_ratio": rr,
                                  "above_design_floor": bool(e) and abs(e) >= R_FLOOR * S["M"],
                                  "S_star_moment": float(S["M"])})
    cms = sorted({(f["case"], f["mode"]) for f in fired})
    return {"passing_case_modes_evaluated": passing, "firing_rows": len(fired), "firing_case_modes": [list(c) for c in cms],
            "rows": fired}


def main():
    ref = json.load(open("REFERENCES/references.json"))["cases"]
    res = json.load(open("DETECTION/results.json"))
    out = {"python": sys.version.split()[0], "A": part_a(ref), "A2": part_a2(), "C": part_c(ref, res), "R": part_r(ref, res)}
    # INPLANE force identity: after S11-F the N2 row is the correctly rounded exact net of (1e80, 1e-8, -1e80)
    net = Fr(1e80) + Fr(float("1e-8")) - Fr(1e80)
    out["inplane_force_identity"] = {"s11f_net_equals_plain_authored_1e-8": float(net) == float("1e-8"),
                                     "note": "identical AssembledForce, so identical solve and published rows"}
    json.dump(out, open(sys.argv[2], "w"), indent=1)
    summ = {"A": {k: {kk: v[kk] for kk in ("exact_defect", "threshold", "stat_over_threshold", "fires", "apriori_c16_over_threshold")}
                  | {kk: v[kk] for kk in v if kk.startswith("matches") or kk.startswith("net_equals")}
                  | {"restrained_fire": [r for r, x in v["restrained_rows"].items() if x["fires"]]} for k, v in out["A"].items()},
            "A2": {k: {r: [x["individual_defects"], x["signed_sum"], x["threshold"], x["fires"]] for r, x in v.items()} for k, v in out["A2"].items()},
            "C": [[c["case"], c["mode"], c["rows_above_floor_over_criterion"], round(c["max_ratio"], 3)] for c in out["C"]],
            "R": {"passing": out["R"]["passing_case_modes_evaluated"], "rows": out["R"]["firing_rows"],
                  "case_modes": out["R"]["firing_case_modes"],
                  "rows_detail": [[f["case"], f["mode"], f["row"], f["q"], f["B"], f["realized_row_relative_ratio"], f["above_design_floor"]]
                                  for f in out["R"]["rows"]]},
            "inplane_force_identity": out["inplane_force_identity"]}
    print(json.dumps(summ, indent=1))


if __name__ == "__main__":
    main()
