import csv, sys
base_p, cand_p, c1_p = sys.argv[1:4]
def load(p):
    with open(p) as f:
        r = csv.DictReader(f, delimiter='\t')
        return {(x['case'], x['mode'], x['entry']): x for x in r}
b, c = load(base_p), load(cand_p)
c1 = set()
with open(c1_p) as f:
    for x in csv.DictReader(f, delimiter='\t'):
        c1.add((x['case'], x['mode'], x['entry']))
keys_equal = set(b) == set(c)
c2 = {k for k in b if 'n10000' in k[0] and k[0].startswith('RF-LARGE')}
fields = ['outcome', 'ok', 'exit_code', 'full_envelope_sha256', 'summary_envelope_sha256', 'error_text_sha256']
diff = {k for k in b if any(b[k][f] != c[k][f] for f in fields)}
print('runs', len(b), len(c), 'keys_equal', keys_equal)
print('C1', len(c1), 'C2', len(c2), 'C1&C2', len(c1 & c2))
print('C3 runs', len(set(b) - c1 - c2))
print('changed total', len(diff))
print('changed outside C1 and C2', sorted(diff - c1 - c2))
print('C1 unchanged (should be empty)', sorted(c1 - diff))
print('C2 changed', len(diff & c2))
