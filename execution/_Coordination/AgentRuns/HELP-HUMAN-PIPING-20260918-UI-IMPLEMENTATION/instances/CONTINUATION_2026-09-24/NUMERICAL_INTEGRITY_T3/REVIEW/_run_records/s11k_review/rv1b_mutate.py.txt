def pristine(path):
    return subprocess.run(["git", "-C", WT, "show", "1251df986:" + path], capture_output=True, text=True, check=True).stdout


def patch(edits):
    for path, old, new in edits:
        full = os.path.join(WT, path)
        text = open(full).read()
        assert text.count(old) == 1, (path, old[:70], text.count(old))
        open(full, "w").write(text.replace(old, new, 1))


def run(crate_dir, args):
    while subprocess.run(["pgrep", "-x", "cargo"], capture_output=True).returncode == 0:
        subprocess.run(["sleep", "10"])
    cmd = ["cargo", "test", "--offline", "--locked", "--no-fail-fast"] + args
    p = subprocess.run(cmd, cwd=os.path.join(WT, crate_dir), env=ENV, capture_output=True, text=True)
    out = p.stdout + p.stderr
    failed = sorted(set(re.findall(r"^test (\S+) \.\.\. FAILED", out, re.M)))
    passed = sum(int(x) for x in re.findall(r"test result: \w+\. (\d+) passed", out))
    compile_error = ("error[" in out or "could not compile" in out) and not failed
    return {"command": "(cd " + crate_dir + " && " + " ".join(cmd) + ")", "exit": p.returncode,
            "passed": passed, "failed_tests": failed, "compile_error": compile_error,
            "tail": out[-1500:] if compile_error else ""}


results = []
only = sys.argv[1:]
for label, what, edits, tests in MUTANTS:
    if only and label not in only:
        continue
    touched = sorted({e[0] for e in edits})
    saved = {t: pristine(t) for t in touched}
    entry = {"mutant": label, "mutation": what,
             "patch": [{"file": e[0], "old": e[1], "new": e[2]} for e in edits], "runs": {}}
    try:
        patch(edits)
        for crate_dir, args in tests:
            entry["runs"][crate_dir + " " + " ".join(args)] = run(crate_dir, args)
    finally:
        for t, text in saved.items():
            open(os.path.join(WT, t), "w").write(text)
    results.append(entry)
    print(label, {k: (v["exit"], v["failed_tests"], v["compile_error"]) for k, v in entry["runs"].items()}, flush=True)
json.dump(results, open("<scratch>/rv1b_mutation_results.json", "w"), indent=1)
