"""RV22: the K6b adapter's K4SRC, decoded, against R1's model (references.py model_json(full=True)).

Independent of I16's k6b_adapter_check.py and runner/k6b_sources.py: this DECODES the bytes the
binary emits (per FK source.rs encode_stiffness_part/encoding) and compares each decoded field with
R1's model semantically: node coordinates bit-equal to R1's inputs rounded once; member end nodes by
R1's labels; E and G bit-equal; A, Iy, Iz, J against R1's exact rationals (pi via R1's PI_Q) within a
few ulps, with Iy == Iz and J == 2*Iy exactly; y_reference a unit global axis not parallel to the member;
the constrained DOF set equal to R1's rigid supports, every value 0; loads equal to R1's nonzero load
components (bits), with source id k6:<global DOF>; one station per member at 0.5; no springs, directional
springs or support groups; and R1's model carries nothing K4SRC does not (no springs, member loads or
load contributions). Usage: rv22_adapter_check.py <references.py> <k6_observe> <case id>...
"""
import importlib.util, math, struct, subprocess, sys
from fractions import Fraction as Fr
sys.dont_write_bytecode = True
DOFS = ('UX', 'UY', 'UZ', 'RX', 'RY', 'RZ')

class Reader:
    def __init__(self, b): self.b, self.i = b, 0
    def take(self, n):
        v = self.b[self.i:self.i + n]; self.i += n
        if len(v) != n: raise ValueError('truncated')
        return v
    def u32(self): return struct.unpack('<I', self.take(4))[0]
    def u8(self): return self.take(1)[0]
    def f64(self): return struct.unpack('<d', self.take(8))[0]
    def dof(self): n = self.u32(); c = self.u8(); assert c < 6; return n * 6 + c

def decode(b):
    r = Reader(b)
    assert r.take(6) == b'K4SRC\x01'
    out = {}
    out['nodes'] = [(r.f64(), r.f64(), r.f64()) for _ in range(r.u32())]
    out['members'] = []
    for _ in range(r.u32()):
        mid, i, j = r.u32(), r.u32(), r.u32()
        props = [r.f64() for _ in range(6)]
        y = (r.f64(), r.f64(), r.f64())
        out['members'].append((mid, i, j, props, y))
    out['springs'] = r.u32(); assert out['springs'] == 0
    out['dsprings'] = r.u32(); assert out['dsprings'] == 0
    out['constraints'] = [(r.dof(), r.f64()) for _ in range(r.u32())]
    loads = []
    for _ in range(r.u32()):
        g = r.dof(); n = r.u32(); sid = r.take(n).decode('ascii'); v = r.f64(); loads.append((g, sid, v))
    out['loads'] = loads
    out['stations'] = [(r.u32(), r.u32(), r.f64()) for _ in range(r.u32())]
    out['supports'] = r.u32()
    assert r.i == len(b), 'trailing bytes'
    return out

def ulps(a, exact):
    return abs(Fr(a) - exact) / Fr(math.ulp(a))

def main():
    refs_path, binary, ids = sys.argv[1], sys.argv[2], sys.argv[3:]
    spec = importlib.util.spec_from_file_location('references', refs_path)
    R = importlib.util.module_from_spec(spec); sys.modules['references'] = R; spec.loader.exec_module(R)
    R.build_large()
    PI = R.PI_Q
    fails = 0
    for cid in ids:
        case = next(c for c in R.CASES if c['id'] == cid)
        m = R.model_json(case['defn'], full=True)
        problems = []
        for k in m:
            if k not in ('nodes_m', 'sections', 'members', 'supports', 'loads'):
                problems.append('R1 model carries %s' % k)
        hexs = subprocess.run([binary, '--emit-source', '--model', cid], check=True, capture_output=True, text=True).stdout.strip()
        d = decode(bytes.fromhex(hexs))
        labels = list(m['nodes_m'])
        idx = {l: k for k, l in enumerate(labels)}
        if len(d['nodes']) != len(labels): problems.append('node count')
        for k, l in enumerate(labels):
            want = tuple(float(Fr(c)) for c in m['nodes_m'][l])
            if tuple(x.hex() for x in d['nodes'][k]) != tuple(x.hex() for x in want):
                problems.append('node %s' % l)
        (sec,) = m['sections'].values()
        E, G, OD, ID = (Fr(sec[k]) for k in ('E', 'G', 'OD', 'ID'))
        # R1 realizes the binary64-represented basis: each input rounded once, then exact.
        rE, rG, rOD, rID = (Fr(float(x)) for x in (E, G, OD, ID))
        A_ex = PI * (rOD * rOD - rID * rID) / 4
        I_ex = PI * (rOD ** 4 - rID ** 4) / 64
        worst = 0
        worst_by = {}
        if len(d['members']) != len(m['members']): problems.append('member count')
        for k, (mid, a, b, sec_id) in enumerate(m['members']):
            got = d['members'][k]
            if got[0] != k + 1: problems.append('member id %d' % k)
            if (got[1], got[2]) != (idx[a], idx[b]): problems.append('member ends %s' % mid)
            e, g, A, Iy, Iz, J = got[3]
            if Fr(e) != rE or Fr(g) != rG: problems.append('E/G %s' % mid)
            if not (Iy == Iz and J == 2 * Iy): problems.append('Iy/Iz/J relation %s' % mid)
            for name, v, ex in (('A', A, A_ex), ('I', Iy, I_ex), ('J', J, 2 * I_ex)):
                u = ulps(v, ex); worst = max(worst, u); worst_by[name] = max(worst_by.get(name, 0), u)
                # K6's binary64 section formula (plan 3.2, by design) is not fl() of the exact value;
                # the tolerance is 8 ulps (about 1e-15 relative), far inside R1's 1e-9 predicate.
                if u > 8: problems.append('section %s %s %.2f ulps' % (mid, name, float(u)))
            y = got[4]
            if sorted(y) != [0.0, 0.0, 1.0]: problems.append('y_ref not a unit axis %s' % mid)
            pa, pb = [tuple(Fr(c) for c in m['nodes_m'][x]) for x in (a, b)]
            ax = y.index(1.0)
            dvec = [pb[t] - pa[t] for t in range(3)]
            if all(dvec[t] == 0 for t in range(3) if t != ax): problems.append('y_ref parallel to member %s' % mid)
        want_c = sorted(idx[n] * 6 + DOFS.index(dn) for n, s in m['supports'].items() for dn in s['rigid'])
        for n, s in m['supports'].items():
            if s.get('springs'): problems.append('R1 springs at %s' % n)
        if sorted(g for g, _ in d['constraints']) != want_c: problems.append('constraint set')
        if any(v != 0.0 or math.copysign(1, v) < 0 for _, v in d['constraints']): problems.append('constraint value')
        want_l = {}
        for n, ld in m['loads'].items():
            for kk, off in (('F', 0), ('M', 3)):
                for t, c in enumerate(ld.get(kk, [])):
                    if Fr(c) != 0: want_l[idx[n] * 6 + off + t] = float(Fr(c))
        got_l = {g: v for g, _, v in d['loads']}
        if len(got_l) != len(d['loads']): problems.append('duplicate load DOF')
        if set(got_l) != set(want_l) or any(got_l[g].hex() != want_l[g].hex() for g in want_l): problems.append('loads')
        if any(sid != 'k6:%d' % g for g, sid, _ in d['loads']): problems.append('load source ids')
        if d['stations'] != [(k + 1, k + 1, 0.5) for k in range(len(m['members']))]: problems.append('stations')
        if d['supports'] != 0: problems.append('support groups')
        fails += bool(problems)
        print('%s %s nodes=%d members=%d constraints=%d loads=%d worst_section_ulps=%.3f %s' % (
            'OK' if not problems else 'DIFFERS', cid, len(d['nodes']), len(d['members']), len(d['constraints']),
            len(d['loads']), float(worst), ' '.join('%s=%.3f' % (k, float(v)) for k, v in sorted(worst_by.items())) + ' ' + '; '.join(problems[:5])))
    print('summary: %d of %d agree with R1' % (len(ids) - fails, len(ids)))
    return 1 if fails else 0

if __name__ == '__main__':
    sys.exit(main())
