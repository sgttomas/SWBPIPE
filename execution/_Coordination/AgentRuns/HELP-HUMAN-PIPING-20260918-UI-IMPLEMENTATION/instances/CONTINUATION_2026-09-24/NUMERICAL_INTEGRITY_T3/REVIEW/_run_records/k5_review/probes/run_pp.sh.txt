#!/bin/bash
W=<wt>
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2
for t in base cand; do
  PPD=$W/scratch/rv14/$t/projects/chirality-piping/core/product_physics
  cp $W/scratch/rv14/probe/rv14_pp_probe.rs $PPD/tests/
  echo "$(date -u +%FT%TZ) start $t $(sysctl -n kern.memorystatus_level) $(sysctl -n vm.loadavg)" >> $W/scratch/rv14/probe/pp_run.log
  (cd $PPD && CARGO_TARGET_DIR=$W/rv14-target/pp-$t cargo test --offline --locked -j 4 --test rv14_pp_probe -- --nocapture > $W/scratch/rv14/probe/pp_probe_$t.log 2>&1; echo "exit=$?" >> $W/scratch/rv14/probe/pp_probe_$t.log)
  echo "$(date -u +%FT%TZ) end $t" >> $W/scratch/rv14/probe/pp_run.log
done
