#!/usr/bin/env python3
"""RV5 mutation and evasion patches. Usage: rv5_mutate.py <core-dir> <id>.
M23, M27, M31a, M31b, M32a, M32b: I3's committed patches (mutate.py.txt), applied unchanged.
E1-E4: RV5's evasion attempts against the folded nonlinear pins."""
import re, subprocess, sys
core, mid = sys.argv[1], sys.argv[2]
NL = core + "/solver/nonlinear_integration/src/lib.rs"
SA = core + "/solver/nonlinear_integration/src/structural_adapter.rs"
if mid in ("M23", "M26", "M27", "M28", "M31a", "M31b", "M31b0", "M32a", "M32b"):
    sys.exit(subprocess.call([sys.executable, sys.argv[0].replace("rv5_mutate.py", "i3_mutate.py"), core, mid]))

def rep(path, old, new, count=1):
    t = open(path).read()
    assert t.count(old) == count, (mid, path, t.count(old))
    open(path, "w").write(t.replace(old, new))

CALL = """        assembly.solve_binary64(
            stiffness,
            force,
            &reduced.free_dofs,
            &prescribed,
            linear_solve_mode,
        )?"""
HELPER = """
    /// RV5 evasion helper: the typed formation-checked entry behind a neutral name.
    pub(crate) fn solve_binary64_audited(
        &self,
        k: &[Vec<f64>],
        f: &[f64],
        free: &[usize],
        prescribed: &[(usize, f64)],
        mode: LinearSolveMode,
    ) -> Result<StructuralSolution, StructuralError> {
        let mut ledger = open_pipe_stress_frame_kernel::load_ledger::LoadLedger::new();
        for (dof, &v) in f.iter().enumerate() {
            if v != 0.0 {
                ledger.push("rv5", dof, v);
            }
        }
        let typed = ledger.finish(f.len()).unwrap();
        self.solve_assembled_with_formation_check(k, &typed, free, prescribed, mode, &[], true)
    }
"""
ANCHOR = "    /// Named, unchanged binary64 variant of `solve` for the nonlinear active-set\n"
if mid == "E1":
    # alias/helper: the loop calls a neutral-named SA helper; the legacy call text
    # is kept in an unreachable branch so the lexed source pins still see it.
    rep(SA, ANCHOR, HELPER + ANCHOR)
    rep(NL, CALL, """        if linear_solve_mode == LinearSolveMode::DenseScrutiny || true {
            assembly.solve_binary64_audited(stiffness, force, &reduced.free_dofs, &prescribed, linear_solve_mode)?
        } else {
""" + CALL + "\n        }")
elif mid == "E2":
    # UFCS: fully qualified call of the checked entry in the loop.
    rep(NL, CALL, """        {
            let mut ledger = open_pipe_stress_frame_kernel::load_ledger::LoadLedger::new();
            for (dof, &v) in force.iter().enumerate() {
                if v != 0.0 {
                    ledger.push("rv5", dof, v);
                }
            }
            let typed = ledger.finish(force.len()).unwrap();
            crate::structural_adapter::AssemblyEvidence::solve_assembled_with_formation_check(
                assembly, stiffness, &typed, &reduced.free_dofs, &prescribed, linear_solve_mode, &[], true,
            )?
        }""")
elif mid == "E3":
    # comment/string text satisfying a source pin: the call becomes the helper,
    # and the legacy call text survives only in a comment and a string literal.
    rep(SA, ANCHOR, HELPER + ANCHOR)
    rep(NL, CALL, """        // assembly.solve_binary64(
        {
            let _pin = "assembly.solve_binary64(";
            assembly.solve_binary64_audited(stiffness, force, &reduced.free_dofs, &prescribed, linear_solve_mode)?
        }""")
elif mid == "E4":
    # sibling module on a path the behavioural pin does not exercise: the
    # derived-friction unit-force solves go through a copy of
    # solve_linearized_system_evidence in a new module that calls the checked
    # entry via the SA helper; lib.rs names no formation entry point.
    rep(SA, ANCHOR, HELPER + ANCHOR)
    t = open(NL).read()
    m = re.search(r"\nfn solve_linearized_system_evidence\(.*?\n}\n", t, re.S)
    body = m.group(0).replace("fn solve_linearized_system_evidence(", "pub(crate) fn solve_linearized_system_evidence_unit(")
    assert body.count(CALL) == 1
    body = body.replace(CALL, """        assembly.solve_binary64_audited(
            stiffness,
            force,
            &reduced.free_dofs,
            &prescribed,
            linear_solve_mode,
        )?""")
    open(core + "/solver/nonlinear_integration/src/unit_bridge.rs", "w").write("#![allow(unused_imports)]\nuse super::*;\n" + body)
    rep(NL, "pub mod product_equilibrium;\n", "pub mod product_equilibrium;\nmod unit_bridge;\n")
    rep(NL, """            let unit = solve_linearized_system_evidence(
                Some(assembly),""", """            let unit = unit_bridge::solve_linearized_system_evidence_unit(
                Some(assembly),""")
else:
    raise SystemExit("unknown " + mid)
print("applied", mid)
