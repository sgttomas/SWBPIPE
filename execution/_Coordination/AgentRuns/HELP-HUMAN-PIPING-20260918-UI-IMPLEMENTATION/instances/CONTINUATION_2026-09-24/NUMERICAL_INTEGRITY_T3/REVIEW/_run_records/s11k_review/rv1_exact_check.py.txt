"""RV1 independent exactness check of S11-K (standard library only).

Generates cases, runs the Rust probe (argv[1]) on them, and compares every
output with a correctly rounded reference computed here from
fractions.Fraction with an explicit round-half-even to binary64 (independent
of the code under review). S11 rule: an exact zero, or a nonzero value that
rounds to zero, is +0.0.
"""
import itertools
import json
import math
import random
import struct
import subprocess
import sys
from fractions import Fraction as F

MAXF = F(sys.float_info.max)


def hx(x):
    return struct.pack(">d", x).hex()


def fromhx(s):
    return struct.unpack(">d", bytes.fromhex(s))[0]


def rne(q):
    """Round a Fraction to binary64, ties to even; +0.0 for any zero result.
    Returns None when outside the binary64 range."""
    if q == 0:
        return 0.0
    sign = -1 if q < 0 else 1
    x = abs(q)
    # e with 2^e <= x < 2^(e+1)
    e = x.numerator.bit_length() - x.denominator.bit_length()
    if F(2) ** e > x:
        e -= 1
    elif F(2) ** (e + 1) <= x:
        e += 1
    quantum = max(e - 52, -1074)
    m = x / (F(2) ** quantum)
    n = m.numerator // m.denominator
    rem = m - n
    if rem > F(1, 2) or (rem == F(1, 2) and n % 2 == 1):
        n += 1
    if n == 0:
        return 0.0
    if F(n) * F(2) ** quantum > MAXF:
        return None
    value = math.ldexp(n, quantum)
    return sign * value


def bits_eq(a, b):
    return hx(a) == hx(b)


random.seed(20260926)
cases = []  # (line, expected_list_or_callable, label, naive_differs)


def rnd_float(lo=-300, hi=300):
    return random.choice([-1, 1]) * random.random() * 10.0 ** random.randint(lo, hi)


def subnormal():
    return random.choice([-1, 1]) * math.ldexp(random.randint(1, 2**52 - 1), -1074)


def fold(vals):
    s = 0.0
    for v in vals:
        s += v
    return s


def add_sum(vals, label):
    exp = rne(sum((F(v) for v in vals), F(0)))
    naive = fold(vals)
    for perm_i, order in enumerate(orders_of(vals)):
        cases.append(("SUM " + " ".join(hx(v) for v in order), [exp], f"{label} perm{perm_i}",
                      exp is not None and not bits_eq(naive, exp)))


def orders_of(vals, k=3):
    out = [list(vals), list(reversed(vals))]
    for _ in range(k):
        v = list(vals)
        random.shuffle(v)
        out.append(v)
    return out

# ---- B. dots: products, underflow nets, ties at the subnormal quantum
def add_dot(pairs, label):
    exact = sum((F(a) * F(b) for a, b in pairs), F(0))
    exp = rne(exact)
    naive = fold([a * b for a, b in pairs])
    for perm_i, order in enumerate(orders_of(pairs, 2)):
        cases.append(("DOT " + " ".join(hx(a) + " " + hx(b) for a, b in order), [exp],
                      f"{label} perm{perm_i}", exp is not None and not bits_eq(naive, exp)))

add_dot([(2.0**-540, 2.0**-535), (2.0**-600, 2.0**-500)], "2^-1075 + 2^-1100 (up)")

add_dot([(1.7e308, 1.7e308), (-1.7e308, 1.7e308)], "(1.7e308)^2 cancel")
add_dot([(1.0, 1.3), (1.35, 4.1e7), (-1.35, 4.1e7)], "thermal pair products")

    pairs += [(a, b), (-a, b)]

             for _ in range(random.randint(1, 5))]

# ---- D. E13 combinations
def add_e13(pairs, label):
    exact = sum((F(v) * F(c) for v, c in pairs), F(0))
    exp = rne(exact)
    if exp is None:
        exp = math.inf if exact > 0 else -math.inf
    naive = fold([v * c for v, c in pairs])
    for perm_i, order in enumerate(orders_of(pairs, 2)):
        cases.append(("E13 " + " ".join(hx(v) + " " + hx(c) for v, c in order), [exp],
                      f"{label} perm{perm_i}", not bits_eq(naive, exp)))


for g, n in [(1e7, 0.3), (1e8, 0.3), (1e80, 1e-8)]:
    add_e13([(g, 1.0), (n, 1.0), (g, -1.0)], f"E13 A+B-A2 G={g:g}")
    add_e13([(g, 1.35), (n, 1.0), (g, -1.35)], f"E13 non-unit G={g:g}")
add_e13([(1.3, 1.0), (4.1e7, 1.35), (4.1e7, -1.35)], "E13 thermal")
for i in range(40):
    pairs = [(rnd_float(-50, 50), rnd_float(-3, 3)) for _ in range(random.randint(2, 6))]
    add_e13(pairs, f"E13 random {i}")
add_e13([(5.0, -1.0), (5.0, 1.0)], "E13 zero")

# ---- E. straight_pipe E1-E4 and E6
L = 2.0


def uniform_station_terms(direction, a, b, w, distance):
    span_start = a * L
    span_end = b * L
    active_end = min(distance, span_end)
    if active_end <= span_start:
        return {}
    active_length = active_end - span_start
    lever_integral = distance * active_length - 0.5 * (active_end * active_end - span_start * span_start)
    if direction == "X":
        return {"N": [w * active_length]}
    if direction == "Y":
        return {"Vy": [w * active_length], "Mz": [-(w * lever_integral)]}
    return {"Vz": [w * active_length], "My": [w * lever_integral]}


def point_station_terms(direction, r, force, distance):
    load_distance = r * L
    if load_distance > distance:
        return {}
    lever = distance - load_distance
    if direction == "X":
        return {"N": [force]}
    if direction == "Y":
        return {"Vy": [force], "Mz": [-(force * lever)]}
    return {"Vz": [force], "My": [force * lever]}


def tok_u(d, a, b, w):
    return f"U:{d}:{hx(a)}:{hx(b)}:{hx(w)}"


def tok_p(d, r, force):
    return f"P:{d}:{hx(r)}:{hx(force)}"


e_cases = []  # (line, checker, label)


def e1_check(out, naive_differs_flag):
    parts = out.split("|")
    total = [fromhx(x) for x in parts[0].split(",")]
    terms = [[fromhx(x) for x in p.split(",")] for p in parts[1:]]
    bad = []
    naive_diff = False
    for dof in range(12):
        exp = rne(sum((F(t[dof]) for t in terms), F(0)))
        if not bits_eq(total[dof], exp):
            bad.append((dof, total[dof], exp))
        if not bits_eq(fold([t[dof] for t in terms]), exp):
            naive_diff = True
    return bad, naive_diff


for g, n in [(1e7, 0.3), (1e8, 0.3), (1e80, 1e-8)]:
    for d in "XYZ":
        for order in ([g, n, -g], [n, g, -g]):

            toks = [tok_p(d, 0.37, q) for q in order] + [tok_u(d, 0.25, 0.75, q) for q in order]
            e_cases.append(("E1 " + " ".join(toks), "E1", f"E1 point+partial {d} G={g:g} {order[0]:g}"))

for g, n in [(1e7, 0.3), (1e8, 0.3), (1e80, 1e-8)]:

        e_cases.append((line, ("E3", order, loads, axial), f"E3 G={g:g} {order[0]:g}"))


def station_line(i_end, fraction, loads):
    return "E4 " + " ".join(hx(v) for v in i_end + [fraction]) + " ; " + " ".join(loads)


for g, n in [(1e7, 0.3), (1e8, 0.3), (1e80, 1e-8)]:
    for order in ([g, n, -g], [n, g, -g]):
        for fraction in (0.25, 0.5, 0.75, 1.0):
            i_end = [2.5, -0.6, 0.4, 0.125, 0.3, -0.6]
            specs = [("U", "Y", 0.0, 1.0, q) for q in order] + [("U", "Z", 0.1, 0.9, q) for q in order] + \
                    [("P", "Y", 0.2, q) for q in order] + [("U", "X", 0.0, 1.0, q) for q in order]
            e_cases.append((specs, i_end, fraction, f"E4 G={g:g} {order[0]:g} x={fraction}"))
# zero-sign station cases: load-free station (+0 from zero i-end), cancelling loads.

e_cases.append(([("U", "Y", 0.0, 1.0, 3.0), ("U", "Y", 0.0, 1.0, -3.0)], [0.0] * 6, 0.5, "E4 exactly cancelling loads"))

# ---- run the probe
lines = [c[0] for c in cases]
e_lines = []
for c in e_cases:
    if isinstance(c[0], list):
        specs, i_end, fraction, label = c
        toks = [tok_u(s[1], s[2], s[3], s[4]) if s[0] == "U" else tok_p(s[1], s[2], s[3]) for s in specs]
        e_lines.append(station_line(i_end, fraction, toks))
    else:
        e_lines.append(c[0])
proc = subprocess.run([sys.argv[1]], input="\n".join(lines + e_lines) + "\n", capture_output=True, text=True)
if proc.returncode != 0:
    print(proc.stderr)
    sys.exit(2)
outs = proc.stdout.splitlines()
assert len(outs) == len(lines) + len(e_lines), (len(outs), len(lines), len(e_lines))

summary = {"accumulator_and_e13": {"cases": 0, "mismatches": [], "naive_fold_differs": 0},
           "straight_pipe": {"cases": 0, "mismatches": [], "naive_fold_differs": 0}}
for (line, exp, label, naive_differs), out in zip(cases, outs):
    s = summary["accumulator_and_e13"]
    s["cases"] += 1
    s["naive_fold_differs"] += int(naive_differs)
    e = exp[0]
    if e is None:
        ok = out.startswith("ERR")
    else:
        ok = out == hx(e)
    if not ok:
        s["mismatches"].append({"label": label, "got": out, "expected": None if e is None else hx(e)})

for c, out in zip(e_cases, outs[len(lines):]):
    s = summary["straight_pipe"]
    s["cases"] += 1
    kind = c[1] if not isinstance(c[0], list) else "E4"
    label = c[-1]
    bad = []
    naive = False
    if kind == "E1":
        bad, naive = e1_check(out, False)
    elif isinstance(kind, tuple) and kind[0] == "E2":
        vals = [fromhx(x) for x in out.split(",")]
        order = kind[1]
        exp_i = rne(sum((F(-v) for v in order), F(0)))
        exp_j = rne(sum((F(v) for v in order), F(0)))
        if not bits_eq(vals[0], exp_i):
            bad.append(("UXi", vals[0], exp_i))
        if not bits_eq(vals[6], exp_j):
            bad.append(("UXj", vals[6], exp_j))
        naive = not bits_eq(fold([-v for v in order]), exp_i)
    elif isinstance(kind, tuple) and kind[0] == "E3":
        _, order, loads, axial = kind
        parts = out.split("|")
        local = [fromhx(x) for x in parts[0].split(",")]
        ends = [[fromhx(x) for x in p.split(",")] for p in parts[1:]]
        # load terms from the E1 API of the same loads
        e1 = subprocess.run([sys.argv[1]], input="E1 " + " ".join(loads) + "\n", capture_output=True, text=True).stdout.strip()
        terms = [[fromhx(x) for x in p.split(",")] for p in e1.split("|")[1:]]
        pairs = []
        for nval in axial:
            p = [0.0] * 12
            p[0] = -nval
            p[6] = nval
            pairs.append(p)
        for end_i, off in ((0, 0), (1, 6)):
            for k in range(6):
                dof = off + k
                exact = F(local[dof]) - sum((F(t[dof]) for t in terms), F(0)) - sum((F(p[dof]) for p in pairs), F(0))
                exp = rne(exact)
                if not bits_eq(ends[end_i][k], exp):
                    bad.append((dof, ends[end_i][k], exp))
                f_ = local[dof]
                for t in terms:
                    f_ -= t[dof]
                g_ = f_
                for p in pairs:
                    g_ -= p[dof]
                if not bits_eq(g_, exp):
                    naive = True
    else:  # E4
        specs, i_end, fraction, _ = c
        distance = fraction * L
        comp = {"N": [i_end[0]], "Vy": [i_end[1]], "Vz": [i_end[2]],
                "My": [i_end[4], i_end[2] * distance], "Mz": [i_end[5], -(i_end[1] * distance)]}
        for s_ in specs:
            t = uniform_station_terms(s_[1], s_[2], s_[3], s_[4], distance) if s_[0] == "U" else \
                point_station_terms(s_[1], s_[2], s_[3], distance)
            for k, v in t.items():
                comp[k].extend(v)
        vals = [fromhx(x) for x in out.split(",")]
        idx = {"N": 0, "Vy": 1, "Vz": 2, "My": 4, "Mz": 5}
        for k, terms in comp.items():
            exp = rne(sum((F(v) for v in terms), F(0)))
            if not bits_eq(vals[idx[k]], exp):
                bad.append((k, vals[idx[k]], exp))
            if not bits_eq(fold(terms) if k in ("My", "Mz") or len(terms) > 1 else terms[0] + 0.0, exp):
                naive = True
        if not bits_eq(vals[3], i_end[3]):
            bad.append(("T passthrough", vals[3], i_end[3]))
    s["naive_fold_differs"] += int(naive)
    if bad:
        s["mismatches"].append({"label": label, "bad": [(str(a), repr(b), repr(cc)) for a, b, cc in bad]})

print(json.dumps(summary, indent=1))
