"""RV22 confirmation: replay b3's admissions with the head's runner and the regenerated counts.
Each b3 row is re-decided against the rows recorded before it (its tier's baseline as recorded),
exactly as run_tier calls admission(); the decision and the binary's backstop are compared."""
import importlib.util, json, os, sys
sys.dont_write_bytecode = True
runner_dir, rec_dir, counts_path = sys.argv[1:4]
spec = importlib.util.spec_from_file_location('k6_runner', os.path.join(runner_dir, 'k6_runner.py'))
r = importlib.util.module_from_spec(spec); spec.loader.exec_module(r)
counts = r.read_counts(counts_path)
recs = [json.loads(l) for l in open(os.path.join(rec_dir, 'records.jsonl'))]
changed = 0; worst_back = 0; worst_proj = 0
for i, rec in enumerate(recs):
    run = next(x for x in r.schedule() if x['order'] == rec['order'])
    d = r.admission(run, counts, recs[:i], rec.get('baseline_rss_bytes'), baseline_footprint_bytes=rec.get('baseline_footprint_bytes'))
    if d['decision'] != rec['admission']['decision']:
        changed += 1; print('CHANGED', rec['run_id'], rec['admission']['decision'], '->', d['decision'], d['reason'])
    if rec['mode'] == 'w1a':
        worst_back = max(worst_back, d['estimate_adm_bytes'] / (run['heap_cap_bytes'] // 2))
        worst_proj = max(worst_proj, d['projected_rss_bytes'] or 0)
print('rows replayed %d; decisions changed %d; largest W1 E_max/(heap cap/2) %.3f; largest projected RSS %.2f GB (0.8 C = %.2f GB)' % (
    len(recs), changed, worst_back, worst_proj / 1e9, 0.8 * recs[0]['rss_cap_bytes'] / 1e9))
