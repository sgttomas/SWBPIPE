#!/bin/bash
# RV17 delta check: the platform-independence of the two new hash pins (CX-F, m1), by inspection.
W=<wt>; R=$W/scratch/rv17
echo "RV17: platform-independence of the new full-envelope pins at 6fa422979 (inspection; hosted Linux CI is the arbiter)."
echo
echo "1. Non-test calls of functions of unspecified precision on the exact-block publication path (source_recovery.rs,"
echo "   source_receipt.rs and its non-test submodules, exact_boundary.rs and exact_boundary/*.rs); test modules excluded"
echo "   by hand after the scan (exact_boundary.rs's hits are all at or after its test module):"
git -C $W/f1b grep -nE "\.(hypot|exp|exp_m1|ln|ln_1p|log10|log2|powf|powi|sin|cos|tan|asin|acos|atan|atan2|cbrt|sin_cos)\(" 6fa422979 -- \
  projects/chirality-piping/core/product_physics/src/source_recovery.rs projects/chirality-piping/core/product_physics/src/source_receipt.rs \
  'projects/chirality-piping/core/product_physics/src/source_receipt/*.rs' \
  projects/chirality-piping/core/solver/frame_kernel/src/structural/exact_boundary.rs \
  'projects/chirality-piping/core/solver/frame_kernel/src/structural/exact_boundary/*.rs' | grep -v "_tests.rs\|/tests.rs" | sed 's/^/   /'
echo "   (every hit is in exact_boundary.rs's test module; the non-test path uses only +, -, *, /, sqrt and next_up/next_down, all correctly rounded; source_receipt's"
echo "   scaled_norm is m * sqrt((a*a + b*b) + c*c) with a, b, c divided by the maximum)"
echo "   test-module start of exact_boundary.rs: line $(git -C $W/f1b show 6fa422979:projects/chirality-piping/core/solver/frame_kernel/src/structural/exact_boundary.rs | grep -n '^#\[cfg(test)\]' | head -1 | cut -d: -f1)"
echo
echo "2. CX-F (captured; main's envelope, sparse; the dense one has the same rows): every nonzero published row."
python3 - <<'PY'
import json, collections
p = '<wt>/scratch/rv17/coex/out2/CX-F-small-axial-3x2m1016__sparse_interactive__captured__main.json'
e = json.load(open(p))
nz = [(r['kind'], r['id'], r['value']) for r in e['results'] if r['value'] != 0.0]
print(f"   {len(e['results'])} rows, {len(nz)} nonzero:")
for k in nz: print('    ', k)
print("   Every magnitude row (displacement_magnitude, reaction_resultant, open_formula_stress_summary) has exactly one")
print("   nonzero component (N1 UX; anchor Fx; axial stress), so hypot(x, +-0) = |x| is exact under C Annex F; there is")
print("   no thermal load (no exp or expm1). The receipt holds sha256 digests, integer work counts and f64 bits.")
PY
echo
echo "3. m1: a blocked envelope, no result row, one diagnostic with a fixed text: no computed value at all."
