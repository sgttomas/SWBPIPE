"""RV5 M31b attack: exact-arithmetic emulation (standard library only).

For a curved-bend model, forms in Decimal (80 digits):
  K_act : the check's intended system (objective element, H from the actual chord),
  K_frm : the M31b intended system (same R, phi, K_t, axes; H from the product's
          binary64 formula chord R(cos phi - 1), R sin phi, 0),
and the product's element with exact arithmetic except the binary64 chord (== K_frm).
u_prod is approximated by the exact solution of the K_frm system rounded to binary64
(the product's element differs from K_frm only by its own binary64 rounding, which is
negligible on the well-conditioned models used here and is measured separately in the
Rust run).  Then, to first order with K~^-1 ~= K_act^-1:
  EF_correct = K_act^-1 (f - K_act u)      (what the correct check estimates)
  EF_M31b    = K_act^-1 (f - K_frm u)      (what M31b estimates)
  actual     = u - u_act                    (published error vs the exact reference)
Ratios are |w|/(1e-9 * max(|q_i|, S*_kind)) per free nodal row, with S* formed exactly
as formation_check.rs body_scales does.  Trigger = 2 * ratio.
"""
import json
import math
import sys
from decimal import Decimal as D, getcontext

getcontext().prec = 80
ZERO, ONE = D(0), D(1)


def dsqrt(x):
    return x.sqrt()


def datan(t):
    # atan for t >= 0 by halving reduction and series.
    k = 0
    while t > D("0.05"):
        t = t / (ONE + dsqrt(ONE + t * t))
        k += 1
    s, term, n, t2 = ZERO, t, 1, t * t
    while True:
        add = term / n
        if abs(add) < D(10) ** -78:
            break
        s += add
        term = -term * t2
        n += 2
    return s * (2 ** k)


def included_angle(sin, cos):
    # phi = 2 atan(sin / (1 + cos)), 0 < phi < pi
    return 2 * datan(sin / (ONE + cos))


def sub(a, b):
    return [a[i] - b[i] for i in range(3)]


def dot(a, b):
    return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]


def cross(a, b):
    return [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]]


def norm(a):
    return dsqrt(dot(a, a))


def inv(m):
    n = len(m)
    a = [row[:] + [ONE if i == j else ZERO for j in range(n)] for i, row in enumerate(m)]
    for c in range(n):
        p = max(range(c, n), key=lambda r: abs(a[r][c]))
        a[c], a[p] = a[p], a[c]
        d = a[c][c]
        a[c] = [v / d for v in a[c]]
        for r in range(n):
            if r != c and a[r][c] != 0:
                f = a[r][c]
                a[r] = [a[r][k] - f * a[c][k] for k in range(2 * n)]
    return [row[n:] for row in a]


def solve(m, b):
    n = len(m)
    a = [m[i][:] + [b[i]] for i in range(n)]
    for c in range(n):
        p = max(range(c, n), key=lambda r: abs(a[r][c]))
        a[c], a[p] = a[p], a[c]
        d = a[c][c]
        for r in range(c + 1, n):
            if a[r][c] != 0:
                f = a[r][c] / d
                for k in range(c, n + 1):
                    a[r][k] -= f * a[c][k]
    x = [ZERO] * n
    for r in range(n - 1, -1, -1):
        s = a[r][n] - sum(a[r][k] * x[k] for k in range(r + 1, n))
        x[r] = s / a[r][r]
    return x


def b64_formula_chord(xi, xj, c):
    """The product's binary64 geometry() and local_stiffness() chord."""
    ri = [xi[k] - c[k] for k in range(3)]
    rj = [xj[k] - c[k] for k in range(3)]
    n = lambda v: math.sqrt(v[0] * v[0] + v[1] * v[1] + v[2] * v[2])
    radius_i, radius_j = n(ri), n(rj)
    ok = abs(radius_i - radius_j) <= 1.0e-9 * max(radius_i, radius_j)
    radius = 0.5 * (radius_i + radius_j)
    pn = [ri[1] * rj[2] - ri[2] * rj[1], ri[2] * rj[0] - ri[0] * rj[2], ri[0] * rj[1] - ri[1] * rj[0]]
    ang = math.atan2(n(pn), ri[0] * rj[0] + ri[1] * rj[1] + ri[2] * rj[2])
    ok = ok and (1e-9 <= ang <= math.pi - 1e-9)
    return [radius * (math.cos(ang) - 1.0), radius * math.sin(ang), 0.0], ok, ang, radius_i, radius_j


def curved_element(xi, xj, c, sec, factor, chord_mode):
    E, G, A, I, J = [D(repr(v)) for v in sec]
    fin = fout = D(repr(factor))
    Xi, Xj, C = [[D(repr(v)) for v in p] for p in (xi, xj, c)]
    ri, rj = sub(Xi, C), sub(Xj, C)
    ni, nj = norm(ri), norm(rj)
    R = (ni + nj) / 2
    nrm = cross(ri, rj)
    nn = norm(nrm)
    cos = dot(ri, rj) / (ni * nj)
    sin = nn / (ni * nj)
    phi = included_angle(sin, cos)
    sin2 = 2 * sin * cos
    gram = [[phi, sin, ONE - cos],
            [sin, phi / 2 + sin2 / 4, sin * sin / 2],
            [ONE - cos, sin * sin / 2, phi / 2 - sin2 / 4]]
    z = ZERO
    rs, rc = R * sin, R * cos
    z3 = [z, z, z]
    cases = [
        [[-rs, z, R], z3, z3, [z, z, -ONE]],
        [[rc, -R, z], z3, z3, [z, ONE, z]],
        [z3, [z, rs, -rc], [R, -rc, -rs], z3],
        [z3, [z, ONE, z], [z, z, -ONE], z3],
        [z3, [z, z, ONE], [z, ONE, z], z3],
        [[ONE, z, z], z3, z3, z3],
    ]

    def quad(l, r):
        return sum(l[a] * gram[a][b] * r[b] for a in range(3) for b in range(3))

    EI, GJ, EA = E * I, G * J, E * A
    F = [[ZERO] * 6 for _ in range(6)]
    for a in range(6):
        for b in range(6):
            F[a][b] = R * (fin * quad(cases[a][0], cases[b][0]) / EI + fout * quad(cases[a][1], cases[b][1]) / EI
                           + quad(cases[a][2], cases[b][2]) / GJ + quad(cases[a][3], cases[b][3]) / EA)
    Kt = inv(F)
    ex = [v / ni for v in ri]
    ez = [v / nn for v in nrm]
    ey = cross(ez, ex)
    axes = [ex, ey, ez]
    if chord_mode == "actual":
        d = sub(Xj, Xi)
        ch = [dot(axes[k], d) for k in range(3)]
    elif chord_mode == "formula_b64":
        ch = [D(repr(v)) for v in b64_formula_chord(xi, xj, c)[0]]
    elif chord_mode == "formula_exact":
        # R (cos phi - 1), R sin phi evaluated exactly
        ch = [R * (cos - 1), R * sin, ZERO]
    else:
        raise ValueError(chord_mode)
    H = [[ONE if i == j else ZERO for j in range(6)] for i in range(6)]
    H[3][1], H[3][2], H[4][0], H[4][2], H[5][0], H[5][1] = -ch[2], ch[1], ch[2], -ch[0], -ch[1], ch[0]
    HK = [[sum(H[r][k] * Kt[k][cc] for k in range(6)) for cc in range(6)] for r in range(6)]
    HKH = [[sum(HK[r][k] * H[cc][k] for k in range(6)) for cc in range(6)] for r in range(6)]
    K = [[ZERO] * 12 for _ in range(12)]
    for r in range(6):
        for cc in range(6):
            K[r][cc] = HKH[r][cc]
            K[r][cc + 6] = -HK[r][cc]
            K[r + 6][cc] = -HK[cc][r]
            K[r + 6][cc + 6] = Kt[r][cc]
    # T^T K T
    T = [[ZERO] * 12 for _ in range(12)]
    for blk in range(4):
        for p in range(3):
            for q in range(3):
                T[3 * blk + p][3 * blk + q] = axes[p][q]
    KT = [[sum(K[r][k] * T[k][cc] for k in range(12) if T[k][cc] != 0) for cc in range(12)] for r in range(12)]
    G_ = [[sum(T[k][r] * KT[k][cc] for k in range(12) if T[k][r] != 0) for cc in range(12)] for r in range(12)]
    return G_, phi


def frame_element(xi, xj, sec, yref):
    E, G, A, I, J = [D(repr(v)) for v in sec]
    Xi, Xj = [[D(repr(v)) for v in p] for p in (xi, xj)]
    d = sub(Xj, Xi)
    L = norm(d)
    ex = [v / L for v in d]
    yr = [D(repr(v)) for v in yref]
    pr = dot(yr, ex)
    yc = [yr[k] - pr * ex[k] for k in range(3)]
    ym = norm(yc)
    ey = [v / ym for v in yc]
    ez = cross(ex, ey)
    axes = [ex, ey, ez]
    k = [[ZERO] * 12 for _ in range(12)]

    def pair(i, j, v):
        k[i][i] = v; k[j][j] = v; k[i][j] = -v; k[j][i] = -v

    pair(0, 6, E * A / L)
    pair(3, 9, G * J / L)
    z12, z6, z4, z2 = 12 * E * I / L ** 3, 6 * E * I / L ** 2, 4 * E * I / L, 2 * E * I / L
    bz = [[z12, z6, -z12, z6], [z6, z4, -z6, z2], [-z12, -z6, z12, -z6], [z6, z2, -z6, z4]]
    by = [[z12, -z6, -z12, -z6], [-z6, z4, z6, z2], [-z12, z6, z12, z6], [-z6, z2, z6, z4]]
    for idx, t in (([1, 5, 7, 11], bz), ([2, 4, 8, 10], by)):
        for r in range(4):
            for c in range(4):
                k[idx[r]][idx[c]] = t[r][c]
    T = [[ZERO] * 12 for _ in range(12)]
    for blk in range(4):
        for p in range(3):
            for q in range(3):
                T[3 * blk + p][3 * blk + q] = axes[p][q]
    KT = [[sum(k[r][kk] * T[kk][cc] for kk in range(12) if T[kk][cc] != 0) for cc in range(12)] for r in range(12)]
    return [[sum(T[kk][r] * KT[kk][cc] for kk in range(12) if T[kk][r] != 0) for cc in range(12)] for r in range(12)]


def assemble(model, chord_mode):
    nn = len(model["nodes"])
    n = 6 * nn
    K = [[ZERO] * n for _ in range(n)]
    phis = []
    for m in model["members"]:
        i, j = m["i"], m["j"]
        if m.get("bend"):
            c, fac = m["bend"]
            ke, phi = curved_element(model["nodes"][i], model["nodes"][j], c, model["section"], fac, chord_mode)
            phis.append(phi)
        else:
            ke = frame_element(model["nodes"][i], model["nodes"][j], model["section"], m["y_reference"])
        mp = list(range(6 * i, 6 * i + 6)) + list(range(6 * j, 6 * j + 6))
        for r in range(12):
            for cc in range(12):
                K[mp[r]][mp[cc]] += ke[r][cc]
    for dof, v in model["springs"]:
        K[dof][dof] += D(repr(v))
    return K, phis


def body_scales(model, free, u):
    # One body when the element graph is connected (all models here are).
    coords = model["nodes"]
    s_tr = max((abs(u[i]) for i in free if i % 6 < 3), default=0.0)
    s_ro = max((abs(u[i]) for i in free if i % 6 >= 3), default=0.0)
    used = sorted({m["i"] for m in model["members"]} | {m["j"] for m in model["members"]})
    d = [max(coords[k][a] for k in used) - min(coords[k][a] for k in used) for a in range(3)]
    lb = math.sqrt((d[0] * d[0] + d[1] * d[1]) + d[2] * d[2])
    if lb > 0:
        return max(s_tr, lb * s_ro), max(s_ro, s_tr / lb)
    return s_tr, s_ro


def analyse(model):
    nn = len(model["nodes"])
    n = 6 * nn
    free = [d for d in range(n) if d not in model["rigid"]]
    f = [ZERO] * n
    for dof, v in model["loads"]:
        f[dof] += D(repr(v))
    Ka, phis = assemble(model, "actual")
    Kf, _ = assemble(model, "formula_b64")
    sub_ = lambda K: [[K[a][b] for b in free] for a in free]
    fa = [f[a] for a in free]
    ua = solve(sub_(Ka), fa)
    uf = solve(sub_(Kf), fa)
    # u_prod proxy: formula-chord system rounded to binary64
    u = [0.0] * n
    uact = [ZERO] * n
    for r, a in enumerate(free):
        u[a] = float(uf[r])
        uact[a] = ua[r]
    U = [D(repr(v)) for v in u]
    Kaf, Kff = sub_(Ka), sub_(Kf)
    Uf = [U[a] for a in free]
    rho_c = [fa[r] - sum(Kaf[r][k] * Uf[k] for k in range(len(free))) for r in range(len(free))]
    rho_b = [fa[r] - sum(Kff[r][k] * Uf[k] for k in range(len(free))) for r in range(len(free))]
    w_c = solve(Kaf, rho_c)
    w_b = solve(Kaf, rho_b)
    tr, ro = body_scales(model, free, u)
    rows = []
    for r, a in enumerate(free):
        scale = max(abs(u[a]), ro if a % 6 >= 3 else tr)
        crit = D("1e-9") * D(repr(scale))
        rows.append({
            "dof": a,
            "ef_correct": float(abs(w_c[r]) / crit),
            "ef_m31b": float(abs(w_b[r]) / crit),
            "delta": float(abs(w_c[r] - w_b[r]) / crit),
            "actual": float(abs(U[a] - uact[a]) / crit),
        })
    best = lambda key: max(rows, key=lambda x: x[key])
    chord, ok, ang, ri, rj = b64_formula_chord(model["nodes"][model["members"][0]["i"]] if model["members"][0].get("bend") else None, None, None) if False else (None, None, None, None, None)
    admissible = []
    for m in model["members"]:
        if m.get("bend"):
            ch, ok, ang, ri, rj = b64_formula_chord(model["nodes"][m["i"]], model["nodes"][m["j"]], m["bend"][0])
            admissible.append({"ok": ok, "phi": ang, "rel_mismatch": (rj - ri) / max(ri, rj)})
    return {
        "name": model["name"],
        "admissible": admissible,
        "max_trigger_correct": 2 * best("ef_correct")["ef_correct"],
        "max_trigger_m31b": 2 * best("ef_m31b")["ef_m31b"],
        "max_delta_row": best("delta"),
        "max_actual_row": best("actual"),
        "demote_correct": 2 * best("ef_correct")["ef_correct"] > 1,
        "demote_m31b": 2 * best("ef_m31b")["ef_m31b"] > 1,
    }


SEC = [200000000000.0, 80000000000.0, 0.005969026041820614, 2.700984283923829e-05, 5.401968567847658e-05]


def cantilever(name, R, phi, plane_normal, shift_frac, springs, load):
    """One elbow, node 0 at the origin with rigid translations and rotational
    springs, node 1 free; centre on the plane, then shifted along the chord by
    shift_frac * R (the radius-mismatch distortion)."""
    nx, ny, nz = plane_normal
    L = math.sqrt(nx * nx + ny * ny + nz * nz)
    zh = [nx / L, ny / L, nz / L]
    # an in-plane unit vector a (radial at node 0 points from centre to node 0)
    t = [1.0, 0.0, 0.0] if abs(zh[0]) < 0.9 else [0.0, 1.0, 0.0]
    a = [t[1] * zh[2] - t[2] * zh[1], t[2] * zh[0] - t[0] * zh[2], t[0] * zh[1] - t[1] * zh[0]]
    la = math.sqrt(sum(v * v for v in a))
    a = [v / la for v in a]
    b = [zh[1] * a[2] - zh[2] * a[1], zh[2] * a[0] - zh[0] * a[2], zh[0] * a[1] - zh[1] * a[0]]
    centre = [-R * a[k] for k in range(3)]  # node 0 at origin, radial x = a
    x1 = [centre[k] + R * (math.cos(phi) * a[k] + math.sin(phi) * b[k]) for k in range(3)]
    ch = [x1[k] for k in range(3)]
    lc = math.sqrt(sum(v * v for v in ch))
    centre = [centre[k] + shift_frac * R * ch[k] / lc for k in range(3)]
    return {
        "name": name,
        "section": SEC,
        "nodes": [[0.0, 0.0, 0.0], x1],
        "members": [{"i": 0, "j": 1, "y_reference": None, "bend": (centre, 1.0)}],
        "rigid": [0, 1, 2],
        "springs": [(3, springs[0]), (4, springs[1]), (5, springs[2])],
        "loads": load,
    }


def max_shift(R, phi, normal, sign=-1.0):
    """Largest |shift_frac| (bisection) whose binary64 geometry is admissible."""
    lo, hi = 0.0, 5e-9 / max(math.sin(phi / 2), 1e-12)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        m = cantilever("x", R, phi, normal, sign * mid, (1, 1, 1), [])
        _, ok, _, _, _ = b64_formula_chord(m["nodes"][0], m["nodes"][1], m["members"][0]["bend"][0])
        if ok:
            lo = mid
        else:
            hi = mid
    return sign * lo


if __name__ == "__main__":
    out = []
    specs = json.loads(sys.argv[1]) if len(sys.argv) > 1 else None
    for spec in specs:
        R, phi, normal, springs, load = spec["R"], spec["phi"], spec["normal"], spec["springs"], spec["load"]
        s = spec.get("shift")
        if s == "max":
            s = max_shift(R, phi, normal)
        m = cantilever(spec["name"], R, phi, normal, s or 0.0, springs, load)
        r = analyse(m)
        r["shift_frac"] = s
        r["model"] = {"nodes": m["nodes"], "centre": m["members"][0]["bend"][0], "springs": m["springs"], "loads": m["loads"]}
        out.append(r)
        print(json.dumps(r), flush=True)
