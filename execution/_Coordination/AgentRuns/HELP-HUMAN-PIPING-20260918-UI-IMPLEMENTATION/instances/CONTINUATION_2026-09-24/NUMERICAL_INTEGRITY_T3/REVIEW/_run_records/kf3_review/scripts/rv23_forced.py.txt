#!/usr/bin/env python3
"""RV23: checks the forced-refusal probe (`RV23 F` lines of rv23_forced_refusals).

For every model, the unforced 256 verification dumps each block's K̃_c (the
verification's scaled matrix at P = 256, from K4's own `scaled_profile`). Each
forced run (a block's Uc refused in the forward, pivot or backward pass, or
its S refused in the shift's formation) prints per block Uc, S, B and the
refusal. Checked with exact Fractions:
- B_c >= ‖K̃_c⁻¹‖₁ (the exact norm) on every block with a B, forced or not;
- B_c = min over the formed bounds (Uc_c when not refused, S_c when formed);
- a block with data, a refused bound and no B: the verification stopped;
- other blocks' Uc, S and B equal their unforced values bit for bit.
usage: rv23_forced.py <probe stdout>
"""
import sys
from fractions import Fraction as Fr

import math

from rv23_oracle import floor_log2, inv_norm1_dense, opt

recs = {}
K = {}
stops = {}
w1 = {}
for line in open(sys.argv[1]):
    if "RV23 F " not in line:
        continue
    f = line[line.index("RV23 F "):].split()
    name, tag = f[2], f[3]
    kind = f[4]
    if kind == "vblk":
        d = dict(zip(f[6::2], f[7::2]))
        recs.setdefault((name, tag), {})[int(f[5])] = d
    elif kind == "K":
        b, i, first = int(f[5]), int(f[6]), int(f[7])
        K.setdefault(name, {}).setdefault(b, {})[i] = (first, [opt(t) for t in f[8:]])
    elif kind == "verify_stop":
        stops[(name, tag)] = " ".join(f[5:])
    elif kind == "W1":
        w1[(name, tag)] = " ".join(f[5:])


def lg(x):
    if x is None:
        return "-"
    e = floor_log2(x)
    return "%.2f" % (e + math.log2(float(x / Fr(2) ** e)))


norms = {}
for name, blocks in K.items():
    for b, rows in blocks.items():
        idx = sorted(rows)
        pos = {g: k for k, g in enumerate(idx)}
        n = len(idx)
        M = [[Fr(0)] * n for _ in range(n)]
        for g in idx:
            first, vals = rows[g]
            for off, v in enumerate(vals):
                j = first + off
                if v:
                    M[pos[g]][pos[j]] = v
                    M[pos[j]][pos[g]] = v
        norms[(name, b)] = inv_norm1_dense(M)[0]

fails = 0
checked = 0
for (name, tag), blocks in sorted(recs.items()):
    base = recs.get((name, "none"), {})
    for b, d in sorted(blocks.items()):
        if "B" not in d:
            # The verification stopped: the shared build's Uc only.
            print("%-28s %-5s blk %d (stopped pass) uc %s refused %s" % (name, tag, b, lg(opt(d["uc"])), d["refused"]))
            continue
        uc, S, B = opt(d["uc"]), opt(d["S"]), opt(d["B"])
        nrm = norms.get((name, b))
        msgs = []
        if B is not None:
            cands = [x for x in (uc, S) if x is not None]
            if B != min(cands):
                msgs.append("B != min(formed)")
            if nrm is not None:
                checked += 1
                if B < nrm:
                    msgs.append("B BELOW NORM")
        if d["refused"] != "-" and uc is not None:
            msgs.append("refused Uc still formed")
        if tag != "none" and b in base:
            # A block other than the forced one keeps its values.
            forced_block = int(tag.split(":")[0])
            if b != forced_block and (d["uc"], d["S"], d["B"]) != (base[b]["uc"], base[b]["S"], base[b]["B"]):
                msgs.append("OTHER BLOCK CHANGED")
        ok = not msgs
        fails += 0 if ok else 1
        print("%-28s %-5s blk %d n %-3s data %-5s refused %-28s srefused %-30s log2 Uc %6s S %6s B %6s norm %6s %s" % (
            name, tag, b, d["n"], d["data"], d["refused"], d["srefused"], lg(uc), lg(S), lg(B),
            lg(nrm), "ok" if ok else "FAIL " + ";".join(msgs)))
    if (name, tag) in stops:
        print("%-28s %-5s verification stopped: %s" % (name, tag, stops[(name, tag)]))
print("blocks with B checked against the exact norm: %d; failures: %d" % (checked, fails))
