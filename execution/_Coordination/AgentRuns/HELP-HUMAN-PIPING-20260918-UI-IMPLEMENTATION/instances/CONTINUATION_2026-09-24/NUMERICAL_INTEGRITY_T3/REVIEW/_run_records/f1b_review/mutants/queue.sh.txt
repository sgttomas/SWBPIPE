#!/bin/bash
# RV17: run the queued mutants one at a time (the second of my two cargo slots; NONE holds the first).
cd <wt>/scratch/rv17/runs
while read keep id patcher crates; do
  [ -z "$id" ] && continue
  KEEP=$keep JOBS=4 ./run_mutant.sh $id $patcher $crates > logs/$id.driver 2>&1
done <<LIST
1 RV17-M1 mutate_rv17.py solver/nonlinear_integration product_physics
1 RV17-M2 mutate_rv17.py solver/nonlinear_integration product_physics
1 RV17-M3 mutate_rv17.py solver/nonlinear_integration product_physics
1 RV17-M4 mutate_rv17.py solver/nonlinear_integration product_physics
0 F1B-PIN-SECOND mutate_f1b.py solver/nonlinear_integration check:product_physics
0 F1B-PIN-HELPER mutate_f1b.py solver/nonlinear_integration check:product_physics
0 F1B-PIN-NEVER mutate_f1b.py solver/nonlinear_integration check:product_physics
LIST
echo QUEUE1-DONE >> MUTANTS.txt
