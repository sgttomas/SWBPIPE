#!/usr/bin/env python3
"""RV23's independent oracles for KF3 (review only; standard library).

- `rv23lib`: exact dyadic parsing of K4's value tokens, directed rounding at
  P bits, the exact accumulator's span rule, and R7 7b's passes with
  amendment A2 as the reviewer reads the design and ROOT's ruling. Written
  from the design, not from GEN or the Rust.
- `syn`: the synthetic multi-block probe (`rv23_synthetic_multiblock`):
  per block the exact ‖K̃_c⁻¹‖₁ (Fraction inverse, or X with K̃·X = I checked
  exactly), the reviewer's own passes against the Rust's tokens, GEN's A2
  passes against the same factor, and 7d's outcome.
- `forced`: the W1-level forced-refusal probe (`rv23_forced_refusals`): per
  block the exact norm of the verification's K̃_c and B_c against it.
"""
import sys
from fractions import Fraction as Fr

SPAN_LIMIT = 8128


# ------------------------------------------------------------------ values
def parse_tok(s):
    if s in ("Z+", "Z-"):
        return Fr(0)
    sign = -1 if s[0] == "-" else 1
    hexs, e = s[1:].split("p")
    e = int(e)
    return sign * Fr(int(hexs, 16)) * Fr(2) ** (e + 1 - 4 * len(hexs))


def opt(s):
    return None if s == "-" else parse_tok(s)


def floor_log2(a):
    assert a > 0
    e = a.numerator.bit_length() - a.denominator.bit_length()
    if Fr(2) ** e > a:
        e -= 1
    if Fr(2) ** (e + 1) <= a:
        e += 1
    return e


def low_bit(a):
    a = abs(a)
    n, d = a.numerator, a.denominator
    assert d & (d - 1) == 0, "not dyadic"
    return (n & -n).bit_length() - 1 - (d.bit_length() - 1)


def span(terms):
    t = [x for x in terms if x != 0]
    if not t:
        return 0
    return max(floor_log2(abs(x)) for x in t) - min(low_bit(x) for x in t) + 1


def rnd(x, p, mode):
    """x at p significant bits: 'n' nearest-even, 'u' up, 'd' down."""
    if x == 0:
        return Fr(0)
    neg = x < 0
    a = -x if neg else x
    e = floor_log2(a)
    unit = Fr(2) ** (e - p + 1)
    q = a / unit
    fl = q.numerator // q.denominator
    rem = q - fl
    if mode == "n":
        r = fl + (1 if (rem > Fr(1, 2) or (rem == Fr(1, 2) and fl % 2 == 1)) else 0)
    else:
        want_up = (mode == "u") != neg
        r = fl + (1 if (rem != 0 and want_up) else 0)
    v = r * unit
    return -v if neg else v


class Refused(Exception):
    pass


def add_dir(a, b, p, up=True):
    s = a + b
    if span([a, b, rnd(s, p, "n")]) > SPAN_LIMIT:
        raise Refused()
    return rnd(s, p, "u" if up else "d")


def mul_up(a, b, p):
    return rnd(a * b, p, "u")


def div_up(a, b, p):
    return rnd(a / b, p, "u")


def gamma(nf, p):
    m = 2 * nf + 2
    return rnd(Fr(m) / (Fr(2) ** p - m), p, "u")


# ------------------------------------------------------------------ 7b with A2
def passes(L, first, blk, p):
    """R7 7b's two passes over a factor L (L[i][j] for first[i] <= j <= i,
    L[i][i] = d_i), with A2: a block's first refusal is kept, its later rows
    are skipped. Returns (c, c', refused{block: (pass, row)})."""
    n = len(first)
    refused = {}
    a = [Fr(1)] * n
    for i in range(n):
        b = blk[i]
        if b in refused:
            continue
        acc = Fr(1)
        ok = True
        for j in range(first[i], i):
            l = abs(L[i][j])
            if l == 0:
                continue
            t = mul_up(l, a[j], p)
            try:
                acc = add_dir(acc, t, p)
            except Refused:
                refused[b] = ("forward", i)
                ok = False
                break
        if ok:
            a[i] = acc
    c = [div_up(a[i], L[i][i], p) if blk[i] not in refused else Fr(0) for i in range(n)]
    for i in reversed(range(n)):
        b = blk[i]
        if b in refused:
            continue
        for j in range(first[i], i):
            l = abs(L[i][j])
            if l == 0:
                continue
            t = mul_up(l, c[i], p)
            try:
                c[j] = add_dir(c[j], t, p)
            except Refused:
                refused[b] = ("backward", j)
                break
    at = [Fr(1)] * n
    for i in range(n):
        b = blk[i]
        if b in refused:
            continue
        for j in range(first[i], i):
            l = abs(L[i][j])
            if l == 0:
                continue
            try:
                at[j] = add_dir(at[j], l, p)
            except Refused:
                refused[b] = ("nl_column", j)
                break
    bt = [mul_up(L[i][i], at[i], p) if blk[i] not in refused else Fr(0) for i in range(n)]
    ct = list(bt)
    for i in range(n):
        b = blk[i]
        if b in refused:
            continue
        for j in range(first[i], i):
            l = abs(L[i][j])
            if l == 0:
                continue
            t = mul_up(l, bt[j], p)
            try:
                ct[i] = add_dir(ct[i], t, p)
            except Refused:
                refused[b] = ("nl_row", i)
                break
    return c, ct, refused


def bounds_from(U, NL, gam, p):
    t = mul_up(mul_up(U, gam, p), NL, p)
    if t < 1:
        den = add_dir(Fr(1), -t, p, up=False)
        return t, div_up(U, den, p)
    return t, None


# ------------------------------------------------------------------ exact norms
def inv_norm1_dense(K):
    """‖K⁻¹‖₁ by Fraction Gauss–Jordan (small symmetric K)."""
    n = len(K)
    A = [row[:] + [Fr(int(i == j)) for j in range(n)] for i, row in enumerate(K)]
    for col in range(n):
        piv = next(r for r in range(col, n) if A[r][col] != 0)
        A[col], A[piv] = A[piv], A[col]
        pv = A[col][col]
        A[col] = [x / pv for x in A[col]]
        for r in range(n):
            if r != col and A[r][col] != 0:
                f = A[r][col]
                A[r] = [x - f * y for x, y in zip(A[r], A[col])]
    X = [row[n:] for row in A]
    return max(sum(abs(X[i][j]) for i in range(n)) for j in range(n)), X


def full_from_profile(first, rows, idx):
    """Dense symmetric K̃ on the local rows `idx` (global row indices)."""
    pos = {g: k for k, g in enumerate(idx)}
    n = len(idx)
    K = [[Fr(0)] * n for _ in range(n)]
    for g in idx:
        for off, v in enumerate(rows[g]):
            j = first[g] + off
            if v != 0:
                assert j in pos, "cross-block entry"
                K[pos[g]][pos[j]] = v
                K[pos[j]][pos[g]] = v
    return K


def inv_norm1_nilpotent(first, rows, L, idx):
    """‖K̃⁻¹‖₁ for a block whose factor has N = I − L with N² = 0: X =
    (I + N)ᵀ D⁻¹ (I + N), accepted only after K̃·X = I is checked exactly."""
    pos = {g: k for k, g in enumerate(idx)}
    n = len(idx)
    G = [dict() for _ in range(n)]  # G[r][c], lower triangular, I + N
    for g in idx:
        r = pos[g]
        G[r][r] = Fr(1)
        for off, v in enumerate(L[g][:-1]):
            j = first[g] + off
            if v != 0:
                G[r][pos[j]] = -v
    D = [L[g][-1] for g in idx]
    # X = Gᵀ D⁻¹ G: X[i][j] = Σ_k G[k][i] G[k][j] / D[k]
    X = [dict() for _ in range(n)]
    for k in range(n):
        items = list(G[k].items())
        for i, gi in items:
            for j, gj in items:
                X[i][j] = X[i].get(j, Fr(0)) + gi * gj / D[k]
    # K̃·X = I, exactly.
    K = [dict() for _ in range(n)]
    for g in idx:
        for off, v in enumerate(rows[g]):
            j = first[g] + off
            if v != 0:
                K[pos[g]][pos[j]] = v
                K[pos[j]][pos[g]] = v
    Xc = [dict() for _ in range(n)]  # columns
    for i in range(n):
        for j, v in X[i].items():
            Xc[j][i] = v
    for i in range(n):
        for j in range(n):
            s = sum(kv * Xc[j].get(k, Fr(0)) for k, kv in K[i].items())
            if s != (1 if i == j else 0):
                raise AssertionError("K·X != I at (%d,%d)" % (i, j))
    return max(sum(abs(v) for v in Xc[j].values()) for j in range(n))
