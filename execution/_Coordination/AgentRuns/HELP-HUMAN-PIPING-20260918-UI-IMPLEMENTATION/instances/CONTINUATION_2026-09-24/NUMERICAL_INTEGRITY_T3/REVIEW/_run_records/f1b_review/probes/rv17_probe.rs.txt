//! RV17 (F1b review) scratch probe, never committed to the product: writes full envelopes of
//! invented models to $RV17_OUT so that a clean tree and a mutant tree can be compared byte for
//! byte. Invented inputs only.
use open_pipe_stress_product_physics::{
    run_linear_static_preview_value_with_mode, run_linear_static_preview_with_mode,
    LinearStaticPreviewRequest, PreviewSolverMode,
};
use serde_json::{json, Value};

const PROV: &str = "invented_rv17_f1b_review_probe_no_library_data";
const ALL: [&str; 6] = ["UX", "UY", "UZ", "RX", "RY", "RZ"];

fn out(name: &str, bytes: &[u8]) {
    let dir = std::env::var("RV17_OUT").expect("RV17_OUT");
    std::fs::create_dir_all(&dir).unwrap();
    std::fs::write(format!("{dir}/{name}.json"), bytes).unwrap();
}

fn run_all(name: &str, request: &Value, captured: bool) {
    for (mode, tag) in [
        (PreviewSolverMode::SparseInteractive, "sparse"),
        (PreviewSolverMode::DenseScrutiny, "dense"),
    ] {
        let typed: LinearStaticPreviewRequest = serde_json::from_value(request.clone()).unwrap();
        let envelope = run_linear_static_preview_with_mode(typed, mode);
        out(&format!("{name}__{tag}__typed"), &serde_json::to_vec(&envelope).unwrap());
        if captured {
            match run_linear_static_preview_value_with_mode(request.clone(), mode) {
                Ok(envelope) => out(&format!("{name}__{tag}__captured"), &serde_json::to_vec(&envelope).unwrap()),
                Err(error) => out(&format!("{name}__{tag}__captured"), format!("{{\"error\":{:?}}}", format!("{error:?}")).as_bytes()),
            }
        }
    }
}

#[allow(clippy::too_many_arguments)]
fn member(id: &str, length: f64, (od, wall): (f64, f64), (e, g): (f64, f64), supports: Vec<Value>, loads: Vec<Value>) -> Value {
    json!({"model": {
        "schema_version": "0.1.0", "document_kind": "openpipestress.product_preview.model",
        "analysis_status": {"mechanics": "ready_for_preview_diagnostics",
                            "rule_check": "not_performed_user_rule_inputs_missing",
                            "professional_acceptance": "not_provided"},
        "project": {"id": format!("invented:rv17-probe:{id}"),
                    "units": {"length": "m", "force": "N", "angle": "rad", "pressure": "Pa", "temperature": "degC", "stress": "Pa"}},
        "nodes": [{"id": "N0", "position": {"x": 0.0, "y": 0.0, "z": 0.0}, "provenance": PROV},
                  {"id": "N1", "position": {"x": length, "y": 0.0, "z": 0.0}, "provenance": PROV}],
        "pipe_segments": [{"id": "M1", "from": "N0", "to": "N1", "material": "mat:P",
                           "y_reference": {"x": 0.0, "y": 1.0, "z": 0.0},
                           "section": {"outside_diameter": {"value": od, "unit": "m"}, "wall_thickness": {"value": wall, "unit": "m"}},
                           "provenance": PROV}],
        "materials": [{"id": "mat:P", "elastic_modulus": {"value": e, "unit": "Pa"},
                       "shear_modulus": {"value": g, "unit": "Pa"}, "provenance": PROV}],
        "supports": supports,
        "load_cases": [{"id": "case", "label": id, "kind": "primitive_user_load", "primitive_loads": loads, "provenance": PROV}],
        "combinations": []}, "materials": []})
}

fn anchor(node: &str, restraints: &[&str]) -> Value {
    json!({"id": format!("anchor:{node}"), "node": node, "family": "anchor", "restraints": restraints, "provenance": PROV})
}

fn spring(node: &str, dof: &str, k: f64) -> Value {
    let unit = if dof.starts_with('R') { "N*m/rad" } else { "N/m" };
    json!({"id": format!("spring:{node}:{dof}"), "node": node, "family": "spring", "restraints": [dof],
           "stiffness": {"dof": dof, "value": {"value": k, "unit": unit}}, "provenance": PROV})
}

fn force(id: &str, dof: &str, value: f64) -> Value {
    if dof.starts_with('R') {
        json!({"id": id, "category": "concentrated_moment", "target": {"type": "node", "node": "N1"},
               "direction": dof, "magnitude": {"value": value, "unit": "N*m"}, "dimension": "moment", "provenance": PROV})
    } else {
        json!({"id": id, "category": "concentrated_force", "target": {"type": "node", "node": "N1"},
               "direction": format!("global_{}", dof[1..].to_lowercase()),
               "magnitude": {"value": value, "unit": "N"}, "dimension": "force", "provenance": PROV})
    }
}

/// RV17-M1: K2a's partial-underflow member with an open gap on N1 UY (a nonlinear invocation whose
/// basis formation leaves the range): main blocks at the basis assembly.
#[test]
fn rv17_probe_m1_nonlinear_formation_range() {
    let supports = vec![
        anchor("N0", &ALL),
        anchor("N1", &["UX", "UZ", "RX", "RY", "RZ"]),
        json!({"id": "support:open-gap", "node": "N1", "family": "nonlinear", "restraints": [],
               "nonlinear": {"behavior": "gap", "dof": "UY", "initial_state": "inactive",
                             "closes_when": "positive_displacement", "gap": {"value": 1000.0, "unit": "mm"}},
               "provenance": PROV}),
    ];
    let request = member("m1", 9.5367431640625e-07, (3.0e-8, 3.0e-9), (1.3e-292, 1.0e-200), supports,
                         vec![force("load:0", "UY", 1.0e-307)]);
    run_all("m1-partial-underflow-open-gap", &request, true);
}

/// RV17-M2: K2a's exact-zero member (a formation-range basis that W2 publishes at b = 734 in
/// F1b's tests), free in UY and UZ, each held by a ground spring of a different stiffness and
/// loaded, so that the two published spring actions differ.
#[test]
fn rv17_probe_m2_two_springs() {
    for (tag, k1, k2) in [("a", 1.0e-289, 3.0e-289), ("b", 2.0e-289, 5.0e-290)] {
        let supports = vec![anchor("N0", &ALL), anchor("N1", &["UX", "RX", "RY", "RZ"]),
                            spring("N1", "UY", k1), spring("N1", "UZ", k2)];
        let request = member(&format!("m2{tag}"), 1.8189894035458565e-12, (1.0e-11, 1.0e-12), (6.4e-280, 1.0e-100),
                             supports, vec![force("load:y", "UY", 2.05e-289), force("load:z", "UZ", 4.1e-289)]);
        run_all(&format!("m2{tag}-exact-zero-two-springs"), &request, true);
    }
}

/// RV17-M4: a stored stiffness entry that overflows when a spring is added after the frames
/// (FK's sparse assembly checks the frame sum only): EA/L ~ 6e304 N/m (12E = 1.2e308 still finite) plus a 1.7976931348623157e308
/// N/m spring on N1 UX. Typed only (capture refuses integral floats above 2^53 - 1).
#[test]
fn rv17_probe_m4_spring_overflow() {
    let supports = vec![anchor("N0", &ALL), anchor("N1", &["UY", "UZ", "RX", "RY", "RZ"]), spring("N1", "UX", f64::MAX)];
    let request = member("m4", 1.0, (0.2, 0.01), (1.0e307, 4.0e306), supports, vec![force("load:0", "UX", 1.0)]);
    run_all("m4-spring-overflow", &request, false);
}
