#!/usr/bin/env python3
"""RV19's independent honesty oracle for K4's selected controls (review only).

Written from D1 revision 5a.2 §4.1 and K4's documented published set, not from
K4's generator. Each model's intended system is formed and solved in decimal
arithmetic at 1600 significant digits (about 5300 bits; every binary64 input is
lifted exactly), and each published row is compared with the claim K4 attaches
to it:
  abs  |q_pub - q*| <= b*(1 + 2^-22)   (1 + 2^-21 when selected at 512)
  rel  |q_pub - q*| <= 2^-64*max(|q_pub|, S*_pub)*(1 + 2^-21) + 2^-53*|q_pub|
  inp  q_pub is q* rounded once to binary64
  unp  Underflow: |q*| below half the least subnormal; Overflow: |q*| beyond MAX
Standard library only.
"""
import sys
import struct
from decimal import Decimal, getcontext, ROUND_HALF_EVEN
from fractions import Fraction

ctx = getcontext()
ctx.prec = 1600
ctx.Emax = 999999999
ctx.Emin = -999999999
ctx.rounding = ROUND_HALF_EVEN

TWO = Decimal(2)


def hf(h):
    return struct.unpack(">d", bytes.fromhex(h))[0]


def dec(x):
    return Decimal(x)  # exact for binary64


def p2(e):
    return TWO ** e


class Model:
    def __init__(self, name):
        self.name = name
        self.nodes = []
        self.members = []
        self.springs = []
        self.dsprings = []
        self.constraints = []
        self.loads = []
        self.stations = []
        self.supports = []


def parse_models(text):
    out = {}
    combos = {}
    cur = None
    combo = None
    for line in text.splitlines():
        f = line.split()
        if not f:
            continue
        if f[0] == "model":
            cur = Model(f[1])
            combo = None
            continue
        if f[0] == "combo":
            combo = (f[1], [(hf(o.split(":")[0]), o.split(":")[1]) for o in f[2:]])
            combos[f[1]] = combo[1]
            cur = None
            continue
        if f[0] == "end":
            if cur is not None:
                out[cur.name] = cur
            cur = None
            combo = None
            continue
        if cur is None:
            continue
        if f[0] == "node":
            cur.nodes.append([hf(f[1]), hf(f[2]), hf(f[3])])
        elif f[0] == "member":
            cur.members.append(dict(id=int(f[1]), i=int(f[2]), j=int(f[3]), E=hf(f[4]), G=hf(f[5]),
                                    A=hf(f[6]), Iy=hf(f[7]), Iz=hf(f[8]), J=hf(f[9]),
                                    yr=[hf(f[10]), hf(f[11]), hf(f[12])]))
        elif f[0] == "spring":
            cur.springs.append(dict(id=int(f[1]), node=int(f[2]), comp=int(f[3]), k=hf(f[4])))
        elif f[0] == "dspring":
            cur.dsprings.append(dict(id=int(f[1]), node=int(f[2]), off=0 if f[3] == "t" else 3,
                                     n=[hf(f[4]), hf(f[5]), hf(f[6])], k=hf(f[7])))
        elif f[0] == "constraint":
            cur.constraints.append((int(f[1]) * 6 + int(f[2]), hf(f[3])))
        elif f[0] == "load":
            cur.loads.append((int(f[1]) * 6 + int(f[2]), hf(f[3])))
        elif f[0] == "station":
            cur.stations.append(dict(id=int(f[1]), member=int(f[2]), t=hf(f[3])))
        elif f[0] == "support":
            ids = lambda s: [] if s == "-" else [int(x) for x in s.split(",")]
            cur.supports.append(dict(id=int(f[1]), node=int(f[2]),
                                     restrained=[ch == "1" for ch in f[3]],
                                     springs=ids(f[4]), dsprings=ids(f[5])))
    return out, combos


def dot(a, b):
    return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]


def cross(a, b):
    return [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]]


def unit(v):
    n = dot(v, v).sqrt()
    return [x / n for x in v], n


def member_ops(m, nodes):
    xi = [dec(x) for x in nodes[m["i"]]]
    xj = [dec(x) for x in nodes[m["j"]]]
    d = [xj[k] - xi[k] for k in range(3)]
    ex, L = unit(d)
    yr = [dec(x) for x in m["yr"]]
    pr = dot(yr, ex)
    yc = [yr[k] - pr * ex[k] for k in range(3)]
    ey, _ = unit(yc)
    ez = cross(ex, ey)
    ka = dec(m["E"]) * dec(m["A"]) / L
    kt = dec(m["G"]) * dec(m["J"]) / L
    bz = dec(m["E"]) * dec(m["Iz"]) / L
    by = dec(m["E"]) * dec(m["Iy"]) / L
    Z = Decimal(0)
    B = [[Z] * 12 for _ in range(6)]
    inv = 1 / L
    for k in range(3):
        B[0][k] = -ex[k]
        B[0][6 + k] = ex[k]
        B[1][3 + k] = -ex[k]
        B[1][9 + k] = ex[k]
        for row, rot in ((2, 3), (3, 9)):
            B[row][k] = inv * ey[k]
            B[row][6 + k] = -inv * ey[k]
            B[row][rot + k] = ez[k]
        for row, rot in ((4, 3), (5, 9)):
            B[row][k] = -inv * ez[k]
            B[row][6 + k] = inv * ez[k]
            B[row][rot + k] = ey[k]
    D = [[Z] * 6 for _ in range(6)]
    D[0][0] = ka
    D[1][1] = kt
    D[2][2] = 4 * bz; D[2][3] = 2 * bz; D[3][2] = 2 * bz; D[3][3] = 4 * bz
    D[4][4] = 4 * by; D[4][5] = 2 * by; D[5][4] = 2 * by; D[5][5] = 4 * by
    DB = [[sum(D[r][s] * B[s][c] for s in range(6)) for c in range(12)] for r in range(6)]
    Ke = [[sum(B[r][a] * DB[r][b] for r in range(6)) for b in range(12)] for a in range(12)]
    dofs = [m["i"] * 6 + k for k in range(6)] + [m["j"] * 6 + k for k in range(6)]
    return dict(ex=ex, ey=ey, ez=ez, L=L, inv=inv, ka=ka, kt=kt, bz=bz, by=by, Ke=Ke, dofs=dofs)


def dspring_block(s):
    n = [dec(x) for x in s["n"]]
    nn = dot(n, n)
    k = dec(s["k"])
    return [[k * n[a] * n[b] / nn for b in range(3)] for a in range(3)]


def solve_model(m, loads, prescribed):
    """loads: dict dof -> Decimal; prescribed: dict dof -> Decimal."""
    ndof = 6 * len(m.nodes)
    Z = Decimal(0)
    K = [[Z] * ndof for _ in range(ndof)]
    ops = []
    for mem in m.members:
        o = member_ops(mem, m.nodes)
        ops.append(o)
        for a in range(12):
            for b in range(12):
                K[o["dofs"][a]][o["dofs"][b]] += o["Ke"][a][b]
    for s in m.springs:
        g = s["node"] * 6 + s["comp"]
        K[g][g] += dec(s["k"])
    dblocks = []
    for s in m.dsprings:
        blk = dspring_block(s)
        dblocks.append(blk)
        base = s["node"] * 6 + s["off"]
        for a in range(3):
            for b in range(3):
                K[base + a][base + b] += blk[a][b]
    free = [g for g in range(ndof) if g not in prescribed]
    pos = {g: i for i, g in enumerate(free)}
    n = len(free)
    A = [[K[free[i]][free[j]] for j in range(n)] for i in range(n)]
    rhs = []
    for g in free:
        v = loads.get(g, Z)
        for c, val in prescribed.items():
            if K[g][c] != 0:
                v -= K[g][c] * val
        rhs.append(v)
    # Gaussian elimination with partial pivoting.
    for col in range(n):
        piv = max(range(col, n), key=lambda r: abs(A[r][col]))
        if A[piv][col] == 0:
            raise ValueError("singular")
        A[col], A[piv] = A[piv], A[col]
        rhs[col], rhs[piv] = rhs[piv], rhs[col]
        pc = A[col][col]
        for r in range(col + 1, n):
            if A[r][col] == 0:
                continue
            f = A[r][col] / pc
            Ar = A[r]
            Ac = A[col]
            for c2 in range(col, n):
                if Ac[c2] != 0:
                    Ar[c2] -= f * Ac[c2]
            rhs[r] -= f * rhs[col]
    x = [Z] * n
    for i in range(n - 1, -1, -1):
        s = rhs[i]
        Ai = A[i]
        for j in range(i + 1, n):
            if Ai[j] != 0:
                s -= Ai[j] * x[j]
        x[i] = s / Ai[i]
    u = [Z] * ndof
    for g, v in prescribed.items():
        u[g] = v
    for g in free:
        u[g] = x[pos[g]]
    return K, u, ops, dblocks


def recover_rows(m, K, u, ops, dblocks, loads, prescribed):
    rows = {}
    ndof = 6 * len(m.nodes)
    for g in range(ndof):
        rows["u.%d.%d" % (g // 6, g % 6)] = u[g]
    for nd in range(len(m.nodes)):
        rows["mag.%d" % nd] = (u[6 * nd] ** 2 + u[6 * nd + 1] ** 2 + u[6 * nd + 2] ** 2).sqrt()
    qs = {}
    acts = {}
    for mem, o in zip(m.members, ops):
        axes = [o["ex"], o["ey"], o["ez"]]
        uu = [u[g] for g in o["dofs"]]
        d = [None] * 12
        for blk in range(4):
            for r in range(3):
                d[3 * blk + r] = dot(axes[r], uu[3 * blk:3 * blk + 3])
        inv = o["inv"]
        e = [d[6] - d[0], d[9] - d[3],
             d[5] + (d[1] - d[7]) * inv, d[11] + (d[1] - d[7]) * inv,
             d[4] + (d[8] - d[2]) * inv, d[10] + (d[8] - d[2]) * inv]
        bz, by = o["bz"], o["by"]
        q = [o["ka"] * e[0], o["kt"] * e[1],
             bz * (4 * e[2] + 2 * e[3]), bz * (2 * e[2] + 4 * e[3]),
             by * (4 * e[4] + 2 * e[5]), by * (2 * e[4] + 4 * e[5])]
        vy = inv * (q[2] + q[3])
        vz = -inv * (q[4] + q[5])
        a = [-q[0], vy, vz, -q[1], q[4], q[2], q[0], -vy, -vz, q[1], q[5], q[3]]
        qs[mem["id"]] = q
        acts[mem["id"]] = a
        for c in range(6):
            rows["end.%d.i.%d" % (mem["id"], c)] = a[c]
            rows["end.%d.j.%d" % (mem["id"], c)] = a[6 + c]
    for s in m.stations:
        q = qs[s["member"]]
        a = acts[s["member"]]
        t = dec(s["t"])
        my = t * q[5] + t * q[4] - q[4]
        mz = t * q[3] + t * q[2] - q[2]
        for c, v in enumerate([a[6], a[7], a[8], a[9], my, mz]):
            rows["st.%d.%d" % (s["id"], c)] = v
    spring_act = {}
    for s in m.springs:
        g = s["node"] * 6 + s["comp"]
        v = -dec(s["k"]) * u[g]
        spring_act[s["id"]] = (s["comp"], v)
        rows["spr.%d.%d" % (s["id"], s["comp"])] = v
    dact = {}
    for s, blk in zip(m.dsprings, dblocks):
        base = s["node"] * 6 + s["off"]
        comps = []
        for a in range(3):
            v = -sum(blk[a][b] * u[base + b] for b in range(3))
            comps.append(v)
            rows["dspr.%d.%d" % (s["id"], s["off"] + a)] = v
        dact[s["id"]] = (s["off"], comps)
    reac = {}
    for g in sorted(prescribed):
        v = sum(K[g][j] * u[j] for j in range(ndof) if K[g][j] != 0) - loads.get(g, Decimal(0))
        reac[g] = v
        rows["R.%d.%d" % (g // 6, g % 6)] = v
    for gp in m.supports:
        comp = []
        for c in range(6):
            g = gp["node"] * 6 + c
            v = Decimal(0)
            if gp["restrained"][c] and g in reac:
                v += reac[g]
            for sid in gp["springs"]:
                cc, val = spring_act[sid]
                if cc == c:
                    v += val
            for sid in gp["dsprings"]:
                off, comps = dact[sid]
                if off <= c < off + 3:
                    v += comps[c - off]
            comp.append(v)
        rows["sf.%d" % gp["id"]] = (comp[0] ** 2 + comp[1] ** 2 + comp[2] ** 2).sqrt()
        rows["sm.%d" % gp["id"]] = (comp[3] ** 2 + comp[4] ** 2 + comp[5] ** 2).sqrt()
    return rows


def round_b64(x):
    """Correctly rounded binary64 of a Decimal (ties to even), as a Fraction-checked float."""
    if x == 0:
        return 0.0
    fr = Fraction(x)
    f = float(fr)  # Fraction.__float__ is correctly rounded
    return f


def exact_of_float(f):
    return Decimal(f)


def main(dump_path, models_paths):
    models = {}
    combos = {}
    for p in models_paths:
        mm, cc = parse_models(open(p).read())
        models.update(mm)
        combos.update(cc)
    dumps = {}
    scales = {}
    combo_ops = {}
    order = []
    for line in open(dump_path):
        f = line.split()
        if not f:
            continue
        if f[0] == "DUMP":
            dumps[f[1]] = dict(sel=int(f[2]), rows=[])
            order.append(f[1])
        elif f[0] == "ROW":
            dumps[f[1]]["rows"].append(dict(key=f[2], kind=f[3], body=int(f[4]), vk=f[5],
                                            bits=int(f[6], 16), cls=f[7], b=int(f[8], 16)))
        elif f[0] == "SCALE":
            scales[(f[1], int(f[2]), f[3])] = struct.unpack(">d", int(f[4], 16).to_bytes(8, "big"))[0]
        elif f[0] == "COMBO":
            combo_ops[f[1]] = [(struct.unpack(">d", int(o.split(":")[0], 16).to_bytes(8, "big"))[0],
                                o.split(":")[1]) for o in f[2:]]
    only = set(sys.argv[3:]) if len(sys.argv) > 3 else None
    worst_all = []
    for name in order:
        if only and name not in only:
            continue
        d = dumps[name]
        if name in combo_ops:
            ops_ = combo_ops[name]
            m = models[ops_[0][1]]
            loads = {}
            presc = {}
            for c, on in ops_:
                om = models[on]
                for g, v in om.loads:
                    loads[g] = loads.get(g, Decimal(0)) + dec(c) * dec(v)
                for g, v in om.constraints:
                    presc[g] = presc.get(g, Decimal(0)) + dec(c) * dec(v)
        else:
            m = models[name]
            loads = {}
            for g, v in m.loads:
                loads[g] = loads.get(g, Decimal(0)) + dec(v)
            presc = {g: dec(v) for g, v in m.constraints}
        try:
            K, u, ops, dblocks = solve_model(m, loads, presc)
        except ValueError as e:
            print("ORACLE %s: %s" % (name, e))
            continue
        truth = recover_rows(m, K, u, ops, dblocks, loads, presc)
        sel = d["sel"]
        tmax = max([abs(v) for v in truth.values()] + [Decimal(1)])
        zero_tol = tmax * Decimal("1e-1200")
        bf = Decimal(1) + p2(-21 if sel == 512 else -22)
        worst = (Decimal(0), "")
        worst_1e9 = (Decimal(0), "")
        bad = []
        for r in d["rows"]:
            key = r["key"]
            if key not in truth:
                bad.append("NOKEY %s" % key)
                continue
            qs = truth[key]
            if r["vk"] in ("N", "S"):
                qpub = struct.unpack(">d", r["bits"].to_bytes(8, "big"))[0]
                qd = Decimal(qpub)
                err = abs(qd - qs)
                if r["cls"] == "abs":
                    b = struct.unpack(">d", r["b"].to_bytes(8, "big"))[0]
                    allow = Decimal(b) * bf
                    if allow == 0:
                        ratio = Decimal(0) if err <= zero_tol else Decimal("Infinity")
                    else:
                        ratio = max(err - zero_tol, Decimal(0)) / allow
                elif r["cls"] == "rel":
                    s = Decimal(scales[(name, r["body"], r["kind"])])
                    allow = p2(-64) * max(abs(qd), s) * (1 + p2(-21)) + p2(-53) * abs(qd)
                    ratio = err / allow
                    r9 = err / (Decimal("1e-9") * abs(qs)) if qs != 0 else Decimal("Infinity")
                    if r9 > worst_1e9[0]:
                        worst_1e9 = (r9, key)
                elif r["cls"] == "inp":
                    ratio = Decimal(0) if round_b64(qs) == qpub else Decimal("Infinity")
                else:
                    ratio = Decimal(0)
                    bad.append("CLASS %s unp with a value" % key)
                if ratio > worst[0]:
                    worst = (ratio, key)
                if ratio > 1:
                    bad.append("FALSE %s cls=%s pub=%r truth=%s ratio=%.3e" % (
                        key, r["cls"], qpub, format(qs, ".6e"), float(ratio)))
            elif r["vk"] == "U":
                if abs(qs) >= p2(-1075):
                    bad.append("UNDERFLOW-WRONG %s truth=%s" % (key, format(qs, ".6e")))
            elif r["vk"] == "O":
                if abs(qs) < Decimal(sys.float_info.max):
                    bad.append("OVERFLOW-WRONG %s truth=%s" % (key, format(qs, ".6e")))
        print("ORACLE %s sel %d rows %d worst %.3e at %s; worst rel vs 1e-9 %.3e at %s%s" % (
            name, sel, len(d["rows"]), float(worst[0]), worst[1], float(worst_1e9[0]), worst_1e9[1],
            "" if not bad else "; ISSUES " + " | ".join(bad[:8])))
        worst_all.append((worst[0], name, worst[1]))
    worst_all.sort(reverse=True)
    print("ORACLE worst five:", [(float(a), b, c) for a, b, c in worst_all[:5]])


if __name__ == "__main__":
    main(sys.argv[1], sys.argv[2].split(","))
