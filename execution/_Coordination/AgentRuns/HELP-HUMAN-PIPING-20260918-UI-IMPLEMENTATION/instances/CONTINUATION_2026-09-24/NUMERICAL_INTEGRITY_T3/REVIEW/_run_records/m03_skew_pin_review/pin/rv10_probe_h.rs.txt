//! RV10 probe H (reviewer's): the margin of RV7's 2^-1050 row on each skew member: the least positive bound over 2^-1022.
use open_pipe_stress_frame_kernel::structural::transform_roundoff;
use open_pipe_stress_frame_kernel::{FrameElement, FrameNode, FrameSection};
use std::f64::consts::PI;
fn p2(k: i32) -> f64 { if k >= -1022 { 2f64.powi(k) } else { f64::from_bits(1u64 << (k + 1074)) } }
#[test]
fn rv10_probe_h() {
    let (od, wall) = (1.0e-11_f64, 1.0e-12_f64);
    let id = od - 2.0 * wall;
    let a = PI * (od.powi(2) - id.powi(2)) / 4.0;
    let i = PI * (od.powi(4) - id.powi(4)) / 64.0;
    let l = p2(-39);
    let s = l / 3f64.sqrt();
    for t in [-1030, -1040, -1045, -1050] {
        let e = (1.0 / (12.0 * i)) * p2(-1000) * p2(t + 1000);
        for (name, nj, yref) in [("skew111", [s, s, s], [0.0, 0.0, 1.0]), ("skew122", [l / 3.0, 2.0 * l / 3.0, 2.0 * l / 3.0], [1.0, 0.0, 0.0])] {
            let el = FrameElement::new(FrameNode::new(0, [0.0; 3]).unwrap(), FrameNode::new(1, nj).unwrap(),
                FrameSection::new(e, 1.0e-100, a, i, i, 2.0 * i).unwrap(), yref).unwrap();
            let r = transform_roundoff(&el.local_stiffness().unwrap(), &el.orientation().unwrap().transformation_matrix()).unwrap();
            let least = r.absolute_roundoff.iter().flatten().copied().filter(|v| *v > 0.0).fold(f64::INFINITY, f64::min);
            println!("RV10H 2^{t} {name}: least bound {least:e} = 2^-1022 x {:.6}; {} ulps above 2^-1022 (relative {:.3e})", least / f64::MIN_POSITIVE, least.to_bits() - f64::MIN_POSITIVE.to_bits(), least / f64::MIN_POSITIVE - 1.0);
        }
    }
}
