#!/usr/bin/env python3
"""RV11's delta-check mutants (the fix commits bf4647c21..f385a8bc8). Usage: mutate_rv11d.py <core-dir> <id>.
Exact text replacements; each must match exactly once.
- RV11D-REACT-B0-EXEMPT: the reactions' row check skipped at b = 0 (ROOT's decision 1 says it applies at every b).
- RV11D-ACTIONS-B0-EXEMPT: the member actions' checks skipped at b = 0 (the function's doc says every b).
- RV11D-ACTIONS-LOCAL-UNCHECKED: the local-displacement stage T*u_e of the member actions unchecked
  (the stiffness stage still checked).
- RV11D-PIN-ACTIONS-EVASION: a non-test function in the nonlinear loop's module (NI lib.rs) calls
  force_scaled_end_actions with the scale written as Default::default(), naming no pinned token
  (ROOT's decision 2: the new function is covered only through the ForceScale token)."""
import sys

core, mid = sys.argv[1], sys.argv[2]
LIB = core + "/solver/frame_kernel/src/lib.rs"
SP = core + "/solver/frame_kernel/src/structural/sparse.rs"
NL = core + "/solver/nonlinear_integration/src/lib.rs"

MUTANTS = {
    "RV11D-REACT-B0-EXEMPT": [(SP, "                if !self.row_product_stays_normal(dof, u) {",
                               "                if !force_scale.is_unscaled() && !self.row_product_stays_normal(dof, u) {")],
    "RV11D-ACTIONS-B0-EXEMPT": [(LIB, "            (a == 0.0 || b == 0.0 || product.is_normal()) && (partial == 0.0 || partial.is_normal())\n",
                                 "            scale.is_unscaled() || ((a == 0.0 || b == 0.0 || product.is_normal()) && (partial == 0.0 || partial.is_normal()))\n")],
    "RV11D-ACTIONS-LOCAL-UNCHECKED": [(LIB, "                if !stays_normal(transform[row][col], global[col], product, local[row]) {",
                                       "                if false && !stays_normal(transform[row][col], global[col], product, local[row]) {")],
    "RV11D-PIN-ACTIONS-EVASION": [(NL, None,
        "\n/// MUTANT RV11D-PIN-ACTIONS-EVASION\n#[allow(dead_code)]\nfn rv11d_loop_member_actions(frame: &open_pipe_stress_frame_kernel::FrameElement, u: &[f64]) -> bool {\n    frame.force_scaled_end_actions(u, Default::default()).is_ok()\n}\n")],
}
for path, old, new in MUTANTS[mid]:
    text = open(path).read()
    if old is None:
        text = text + new
    else:
        assert text.count(old) == 1, (mid, path, text.count(old))
        text = text.replace(old, new)
    open(path, "w").write(text)
print("applied", mid)
