"""RV5 timing: RF-LARGE-CONT-n01000-AX dense captured, main probe (b24b3d536, S11-G, no K-D5) against the candidate probe,
alternating, 2 runs each, run.py's run_one (P1 limits)."""
import json, os, sys
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from run import run_one, TIMEOUT_LARGE
req = os.path.join('out', sys.argv[3])
out = []
for rep in range(2):
    for label, probe in (('main_b24b3d536', sys.argv[1]), ('candidate_2409de83e', sys.argv[2])):
        load0 = open('/proc/loadavg').read().split()[:3]
        r = run_one(probe, req, 'dense_scrutiny', False, TIMEOUT_LARGE)
        env = ((r.get('probe') or {}).get('run') or {}).get('envelope') or {}
        q = [c['solve_quality'] for c in env.get('numerical_quality', {}).get('cases', [])]
        rec = {'label': label, 'rep': rep, 'wall_seconds': r['wall_seconds'], 'solve_seconds': ((r.get('probe') or {}).get('run') or {}).get('solve_seconds'),
               'quality': q, 'loadavg_start': load0, 'loadavg_end': open('/proc/loadavg').read().split()[:3],
               'results_sha256': __import__('hashlib').sha256(json.dumps(env.get('results'), sort_keys=True).encode()).hexdigest()}
        print(json.dumps(rec), flush=True); out.append(rec)
