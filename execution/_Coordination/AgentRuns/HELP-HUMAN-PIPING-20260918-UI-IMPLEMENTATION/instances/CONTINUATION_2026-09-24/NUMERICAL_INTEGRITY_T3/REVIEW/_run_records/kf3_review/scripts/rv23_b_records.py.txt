#!/usr/bin/env python3
"""RV23: KF3's committed B records (T3/IMPLEMENTATION/KF3/_run_records/b/runs)
against V-K's pre-KF3 B (T3/IMPLEMENTATION/VK/_run_records/b/runs): the
per-case `record` line byte for byte, the outcome and the charged work; and
per case the runner's report (rows, pass, absolute-range pass, fail, C9
S_full) with the W1 line's publication.
usage: rv23_b_records.py <T3/IMPLEMENTATION>"""
import glob, json, os, sys
root = sys.argv[1]
kf, vk = os.path.join(root, "KF3/_run_records/b/runs"), os.path.join(root, "VK/_run_records/b/runs")
for f in sorted(glob.glob(os.path.join(kf, "[01][0-9]_*_w1a.jsonl"))):
    name = os.path.basename(f)
    objs = [json.loads(l) for l in open(f)]
    kind = {o["kind"]: o for o in objs}
    rec_k = [l for l in open(f) if '"kind":"record"' in l]
    g = os.path.join(vk, name)
    rec_v = [l for l in open(g) if '"kind":"record"' in l] if os.path.exists(g) else None
    wv = [json.loads(l) for l in open(g) if '"kind":"w1"' in l][0] if rec_v else {}
    w, rep = kind["w1"], kind["report"]
    print("%s | record == V-K's: %s | outcome %s (V-K: %s) | charged %d (V-K: %s) | published_rows %s | R1 rows %d pass %d abs-range %d not-covered %d fail %d | C9 S_full %s | failures %s" % (
        name.replace("_w1a.jsonl", ""), rec_k == rec_v, w["outcome"], wv.get("outcome"), w["invocation_charged"],
        wv.get("invocation_charged"), w["published_rows"], rep["rows"], rep["pass"], rep["pass_absolute_range"],
        rep["not_covered"], rep["fail"], rep["not_covered_rows_s_full"], rep["failures_head"]))
