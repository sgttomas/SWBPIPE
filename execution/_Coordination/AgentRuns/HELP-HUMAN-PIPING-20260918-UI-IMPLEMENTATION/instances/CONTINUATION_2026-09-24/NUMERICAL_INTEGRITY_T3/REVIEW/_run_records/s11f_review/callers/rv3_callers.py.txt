"""RV3 (independent): callers of every function S11-F changed, added or removed.
Written from scratch for RV3; it does not import or copy I4's scan.py/callers.py.
Usage: python3 rv3_callers.py <repo-root> <base-rev> <head-rev>
1. Changed functions: for each .rs file under projects/chirality-piping changed between base and head
   (excluding execution/), map every changed line (unified diff, -U0) of the old and new text to its
   enclosing `fn` item (brace matching on lexed text). Test-only items are recorded but flagged.
2. Callers: every .rs file in the head tree (git ls-tree, excluding execution/, target/), lexed
   (comments removed; string/char literal contents blanked); a call is NAME followed by optional
   turbofish and `(`, not preceded by `fn `, with NAME at an identifier boundary; `.NAME(` method,
   `::NAME(` path and bare `NAME(` are recorded separately with the receiver/path text.
   A call is 'test' if inside a #[cfg(test)] item, a #[test] fn, a tests/ or benches/ directory,
   an examples/ file is 'example'."""
import json, re, subprocess, sys
from collections import defaultdict
root, base, head = sys.argv[1:4]
P = 'projects/chirality-piping/'
def git(*a):
    return subprocess.run(['git', '-C', root] + list(a), capture_output=True, text=True, check=True).stdout
def lex(s):
    out = []; i = 0; n = len(s)
    while i < n:
        c = s[i]
        if s.startswith('//', i):
            j = s.find('\n', i); j = n if j < 0 else j; out.append(' ' * (j - i)); i = j; continue
        if s.startswith('/*', i):
            d = 1; j = i + 2
            while j < n and d:
                if s.startswith('/*', j): d += 1; j += 2
                elif s.startswith('*/', j): d -= 1; j += 2
                else: j += 1
            out.append(''.join('\n' if ch == '\n' else ' ' for ch in s[i:j])); i = j; continue
        m = re.match(r'b?r(#*)"', s[i:]) if (c in 'rb' and (i == 0 or not (s[i-1].isalnum() or s[i-1] == '_'))) else None
        if m:
            hashes = m.group(1); start = i + m.end(); end = s.find('"' + hashes, start)
            end = n if end < 0 else end + 1 + len(hashes)
            out.append(s[i:start] + ''.join('\n' if ch == '\n' else ' ' for ch in s[start:end-1-len(hashes)]) + '"' + hashes); i = end; continue
        if c == '"' or (c == 'b' and i + 1 < n and s[i+1] == '"' and (i == 0 or not (s[i-1].isalnum() or s[i-1] == '_'))):
            j = i + (2 if c == 'b' else 1); buf = s[i:j]
            while j < n and s[j] != '"':
                if s[j] == '\\': j += 2; buf += '  '; continue
                buf += '\n' if s[j] == '\n' else ' '; j += 1
            out.append(buf + '"'); i = j + 1; continue
        if c == "'":
            m2 = re.match(r"'(\\.|\\u\{[0-9a-fA-F]+\}|\\x[0-9a-fA-F]{2}|[^\\'])'", s[i:])
            if m2: out.append("' " + ' ' * (m2.end() - 3) + "'"); i += m2.end(); continue
        out.append(c); i += 1
    return ''.join(out)
def items(code):
    """(name, start, end, is_test) of fn items; test = inside #[cfg(test)] item or #[test] fn."""
    res = []
    for m in re.finditer(r'(?<![A-Za-z0-9_])fn\s+([A-Za-z_][A-Za-z0-9_]*)', code):
        j = m.end(); depth = 0
        # find body open brace at nesting 0 of parens/brackets, stop at ';'
        paren = 0; k = j
        while k < len(code):
            ch = code[k]
            if ch in '([': paren += 1
            elif ch in ')]': paren -= 1
            elif ch == ';' and paren == 0: k = -1; break
            elif ch == '{' and paren == 0: break
            k += 1
        if k < 0 or k >= len(code): continue
        d = 0; e = k
        while e < len(code):
            if code[e] == '{': d += 1
            elif code[e] == '}':
                d -= 1
                if d == 0: break
            e += 1
        res.append([m.group(1), m.start(), e + 1])
    return res
def test_ranges(code):
    out = []
    for m in re.finditer(r'#\[cfg\(test\)\]|#\[test\]', code):
        k = m.end(); d = 0; started = False
        while k < len(code):
            ch = code[k]
            if ch == '{': d += 1; started = True
            elif ch == '}':
                d -= 1
                if started and d == 0: break
            elif ch == ';' and not started: break
            k += 1
        out.append((m.start(), k + 1))
    return out
def in_ranges(pos, rs): return any(a <= pos < b for a, b in rs)
def line_of(code, pos): return code.count('\n', 0, pos) + 1
def enclosing(its, pos):
    best = None
    for name, a, b in its:
        if a <= pos < b and (best is None or a > best[1]): best = (name, a, b)
    return best
def file_is_test(path):
    return '/tests/' in path or '/benches/' in path or path.endswith('_tests.rs') or path.endswith('/tests.rs')
# 1. changed functions
changed = defaultdict(set)   # name -> {(file, side, test?)}
files = [f for f in git('diff', '--name-only', base, head, '--', P + '*.rs').splitlines() if '/execution/' not in f]
for f in files:
    diff = git('diff', '-U0', base, head, '--', f)
    for side, rev, rx in (('old', base, r'^@@ -(\d+)(?:,(\d+))?'), ('new', head, r'^@@ -\d+(?:,\d+)? \+(\d+)(?:,(\d+))?')):
        try: text = git('show', '%s:%s' % (rev, f))
        except subprocess.CalledProcessError: continue
        code = lex(text); its = items(code); tr = test_ranges(code)
        starts = [0]
        for ln in code.split('\n'): starts.append(starts[-1] + len(ln) + 1)
        for m in re.finditer(rx, diff, re.M):
            a = int(m.group(1)); c = int(m.group(2)) if m.group(2) is not None else 1
            for ln in range(a, a + max(c, 1)):
                if ln - 1 >= len(starts) - 1: continue
                pos = starts[ln - 1]
                # use the first non-space char of the line
                line = code[starts[ln-1]:starts[ln]] if ln < len(starts) else ''
                off = len(line) - len(line.lstrip()); pos += off
                e = enclosing(its, pos)
                if e: changed[e[0]].add((f.replace(P, ''), side, file_is_test(f) or in_ranges(pos, tr)))
# 2. callers across the head tree
tree = [f for f in git('ls-tree', '-r', '--name-only', head, P).splitlines() if f.endswith('.rs') and '/execution/' not in f and '/target/' not in f]
names = sorted(changed)
pat = re.compile(r'(?<![A-Za-z0-9_])(' + '|'.join(map(re.escape, names)) + r')\s*(?:::\s*<[^()]*?>\s*)?\(')
calls = defaultdict(list)
for f in tree:
    text = git('show', '%s:%s' % (head, f)); code = lex(text); tr = test_ranges(code); its = items(code)
    for m in pat.finditer(code):
        pre = code[max(0, m.start() - 40):m.start()]
        if re.search(r'fn\s*$', pre): continue
        form = 'method' if pre.rstrip().endswith('.') else ('path' if pre.rstrip().endswith('::') else 'bare')
        recv = re.search(r'([A-Za-z0-9_:]+)\s*(?:\.|::)\s*$', pre)
        kind = 'example' if '/examples/' in f else ('test' if (file_is_test(f) or in_ranges(m.start(), tr)) else 'nontest')
        enc = enclosing(its, m.start())
        calls[m.group(1)].append({'file': f.replace(P, ''), 'line': line_of(code, m.start()), 'form': form,
                                  'receiver': recv.group(1) if recv else '', 'kind': kind, 'in': enc[0] if enc else '<module>'})
out = {'base': base, 'head': head, 'changed_functions': {n: sorted([list(x) for x in changed[n]]) for n in names},
       'callers': {n: calls.get(n, []) for n in names}}
print(json.dumps(out, indent=1))
