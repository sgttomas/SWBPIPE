#!/usr/bin/env python3
"""RV23: compares a vk_scale run with KF3's committed B run, line by line.
The `record` line (the per-case record) must be byte-identical. On the other
lines only the volatile fields (elapsed_ns and the heap figures) may differ.
usage: rv23_bcmp.py <mine.jsonl> <committed.jsonl>"""
import json, sys
VOL = {"elapsed_ns", "heap_current", "heap_peak", "heap_peak_move", "repeats_heap_peak", "repeats_heap_peak_move"}
mine = open(sys.argv[1]).read().splitlines()
ref = open(sys.argv[2]).read().splitlines()
ok = len(mine) == len(ref)
print("lines: mine %d committed %d" % (len(mine), len(ref)))
for a, b in zip(mine, ref):
    ja, jb = json.loads(a), json.loads(b)
    kind = ja.get("kind")
    if kind == "record":
        same = a == b
        print("record line byte-identical: %s (%d bytes)" % (same, len(a)))
        ok &= same
        continue
    da = {k: v for k, v in ja.items() if k not in VOL}
    db = {k: v for k, v in jb.items() if k not in VOL}
    same = da == db
    diff = sorted(k for k in set(ja) | set(jb) if ja.get(k) != jb.get(k))
    print("%-9s equal outside volatile fields: %s (differing: %s)" % (kind, same, ",".join(diff)))
    ok &= same
print("MATCH" if ok else "MISMATCH")
