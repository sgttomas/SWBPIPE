#!/bin/bash
# RV8 delta mutant runner: one clean git archive of b6f1724b4 (piping core) and one
# target per mutant, deleted after the run. Usage: run_mut2.sh <id>; CRATES overrides.
set -u
WT=<wt>
S=$WT/scratch/rv7
REV=b6f1724b4af526e730f78b32d0155101dc1475d2
id=$1
M=$S/mut2/$id
T=$WT/rv7-target/mut2/$id
rm -rf $M $T; mkdir -p $M $T
cd $WT/numerics && git archive $REV projects/chirality-piping/core | tar -x -C $M
CORE=$M/projects/chirality-piping/core
case $id in
  NONE) echo "applied NONE" ;;
  RV8-K2A-*) python3 -B $S/rv8_delta_mutants.py $CORE $id ;;
  RV8-*) python3 -B $S/rv8_mutants.py $CORE $id ;;
  K1-K2A-*) python3 -B $S/mutate_k2a.py $CORE $id ;;
  *) python3 -B $S/mutate_k1.py $CORE $id ;;
esac
source $S/env.sh
export RUST_TEST_THREADS=4
LOG=$S/mutlogs2/$id.log; mkdir -p $S/mutlogs2; : > $LOG
for crate in ${CRATES:-frame_kernel sparse_direct nonlinear_integration}; do
  echo "=== $crate" >> $LOG
  (cd $CORE/solver/$crate && CARGO_TARGET_DIR=$T cargo test --offline --locked -j 4 --no-fail-fast >> $LOG 2>&1); echo "exit $crate $?" >> $LOG
done
rm -rf $T $M
echo "done $id" >> $LOG
