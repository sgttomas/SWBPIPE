#!/bin/bash
# Build RV8's parity probe against a mutated candidate archive and run it.
set -u
WT=<wt>
S=$WT/scratch/rv7
id=$1
M=$S/mutp/$id; T=$WT/rv7-target/mutp/$id
rm -rf $M $T; mkdir -p $M $T
cd $WT/numerics && git archive 43f9e6a78d68feb71f72ff244e3fc1ae27edd4a5 projects/chirality-piping/core | tar -x -C $M
CORE=$M/projects/chirality-piping/core
[ "$id" = NONE ] || python3 $S/rv8_mutants.py $CORE $id
mkdir -p $M/probe/src/kd5
cp $S/probe_parity/src/main.rs $M/probe/src/main.rs
cp $S/probe_parity/src/kd5/kd5_models.rs $M/probe/src/kd5/kd5_models.rs
sed "s|$S/cand/projects/chirality-piping/core|$CORE|" $S/probe_parity/Cargo.toml > $M/probe/Cargo.toml
cp $S/probe_parity/Cargo.lock $M/probe/Cargo.lock
source $S/env.sh
(cd $M/probe && CARGO_TARGET_DIR=$T cargo build --release --offline --locked -j 4 > $S/out/probe_mut_$id.build.log 2>&1) || { echo "build failed $id"; exit 1; }
$T/release/rv7_parity > $S/out/probe_mut_$id.txt 2>&1
echo "$id $(grep -c MISMATCH $S/out/probe_mut_$id.txt) $(grep ^SUMMARY $S/out/probe_mut_$id.txt)"
rm -rf $T $M
