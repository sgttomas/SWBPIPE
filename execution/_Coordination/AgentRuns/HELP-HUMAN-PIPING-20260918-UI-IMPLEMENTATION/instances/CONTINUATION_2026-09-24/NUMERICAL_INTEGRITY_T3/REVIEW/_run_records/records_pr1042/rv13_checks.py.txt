#!/usr/bin/env python3
"""RV13: read-only re-derivation checks for records PR #1042.

Run from the repository root:  python3 rv13_checks.py <candidate> <base>
(RV13 ran it with a8895ee84 and origin/main = e7d930d49). It reads blobs with
`git show` and `git diff`, so the working tree does not matter. It writes
nothing. Standard library only.

Sections:
  1. scope: every changed path is under projects/chirality-piping/execution/
  2. machine-path and model-identifier scans of every changed file (full
     content at the candidate) and of every added line; patterns are built
     from parts so that this file names no path itself
  3. SHA256SUMS: every folder with a changed file and a SHA256SUMS verifies,
     and lists exactly the folder's tracked files
  4. in-place edits, commit by commit on the first-parent chain after PR
     #1035's merge (5a2b9d112): every removed or replaced line of an existing
     file, classified as insert-only (old text kept verbatim) or changed
  5. suites: the three merge records' dec025/suites.log against each other
     and against K1's combined-tree Mac run (the eb52114e9 baseline)
  6. T9: every slice T9 hash list in the candidate against the Mac
     calibration's native main list
"""
import difflib
import hashlib
import os
import re
import subprocess
import sys
from collections import defaultdict
from pathlib import PurePosixPath

CAND = sys.argv[1] if len(sys.argv) > 1 else "a8895ee84"
BASE = sys.argv[2] if len(sys.argv) > 2 else "origin/main"
CHAIN_START = "5a2b9d112"
T3 = ("projects/chirality-piping/execution/_Coordination/AgentRuns/"
      "HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION/instances/"
      "CONTINUATION_2026-09-24/NUMERICAL_INTEGRITY_T3/")


def git(*a):
    return subprocess.run(["git", *a], check=True, capture_output=True).stdout


def gits(*a):
    return git(*a).decode()


def short(p):
    return p.replace(T3, "T3/")


def blob(rev, path):
    return git("show", f"{rev}:{path}")


changed = [c for c in gits("diff", "--name-only", f"{BASE}...{CAND}").split("\n") if c]
status = gits("diff", "--name-status", f"{BASE}...{CAND}").splitlines()
tracked = [t for t in gits("ls-tree", "-r", "--name-only", CAND).split("\n") if t]
tracked_set = set(tracked)

print(f"candidate {gits('rev-parse', CAND).strip()}")
print(f"base {gits('rev-parse', BASE).strip()}; merge-base "
      f"{gits('merge-base', BASE, CAND).strip()}")
print(f"changed paths: {len(changed)} "
      f"(A {sum(1 for s in status if s.startswith('A'))}, "
      f"M {sum(1 for s in status if s.startswith('M'))}, "
      f"other {sum(1 for s in status if not s[:1] in 'AM')})")

# 1. scope
print("\n== 1. scope")
outside = [c for c in changed if not c.startswith("projects/chirality-piping/execution/")]
print(f"paths outside projects/chirality-piping/execution/: {len(outside)}")
for o in outside:
    print("   ", o)
ext = defaultdict(int)
for c in changed:
    name = PurePosixPath(c).name
    if name == "SHA256SUMS":
        ext["SHA256SUMS"] += 1
        continue
    parts = name.split(".")
    ext[".".join(parts[-2:]) if len(parts) > 2 else parts[-1]] += 1
print("by extension:", dict(sorted(ext.items(), key=lambda kv: -kv[1])))

# 2. hygiene scans
print("\n== 2. machine paths and model identifiers")
S = "/"
machine = re.compile(
    "|".join([
        S + "Us" + "ers" + S, S + "ho" + "me" + S, S + "tm" + "p" + S,
        S + "pri" + "vate" + S, S + "var" + S + "fol" + "ders",
        S + "op" + "t" + S, S + "usr" + S + "loc" + "al",
        S + "Lib" + "rary" + S, r"\.rus" + "tup", r"\.car" + "go" + S,
        S + "ni" + "x" + S, S + "Vol" + "umes" + S,
        r"(^|[\s\"'`(=:])~" + S, r"[A-Za-z]:\\\\U" + "sers",
    ]))
model = re.compile(
    "|".join(["cl" + "aude", "anth" + "ropic", "op" + "us", "son" + "net",
              "hai" + "ku", "fa" + "ble", r"gp" + r"t-?\d", "open" + "ai",
              "gem" + "ini", "lla" + "ma", "mis" + "tral"]), re.I)
full_machine = defaultdict(list)
full_model = defaultdict(list)
for c in changed:
    text = blob(CAND, c).decode("utf-8", "replace")
    for i, line in enumerate(text.splitlines(), 1):
        if machine.search(line):
            full_machine[c].append((i, line.strip()[:160]))
        if model.search(line):
            full_model[c].append((i, line.strip()[:160]))
print(f"files with a machine-path pattern hit (full content): {len(full_machine)}")
for f, hits in full_machine.items():
    for i, l in hits:
        # location only: the hit lines are patterns, and are not re-quoted here
        print(f"    {short(f)}:{i}")
print(f"files with a model-identifier hit (full content): {len(full_model)}")
for f, hits in full_model.items():
    for i, l in hits:
        print(f"    {short(f)}:{i}: {l}")
added = [l[1:] for l in gits("diff", "-U0", f"{BASE}...{CAND}").splitlines()
         if l.startswith("+") and not l.startswith("+++")]
roots = [l for l in added if re.search(r"(^|[\s\"'`(])/(bin|usr)/", l)]
kinds = defaultdict(int)
for l in roots:
    m = re.search(r"/(?:usr/)?bin/\S+", l)
    kinds[m.group(0) if m else l[:40]] += 1
print(f"added lines: {len(added)}; root-anchored system-tool strings: "
      f"{dict(kinds)}")

# 3. SHA256SUMS
print("\n== 3. SHA256SUMS")
sums_dirs = {str(PurePosixPath(t).parent) for t in tracked if t.endswith("/SHA256SUMS")}


def covering(path):
    p = PurePosixPath(path).parent
    while str(p).startswith(T3.rstrip("/")):
        if str(p) in sums_dirs:
            return str(p)
        p = p.parent
    return None


cover = defaultdict(list)
uncovered = defaultdict(list)
for c in changed:
    d = covering(c)
    (cover[d] if d else uncovered[str(PurePosixPath(c).parent)]).append(c)
all_ok = True
for d in sorted(cover):
    entries = {}
    for line in blob(CAND, f"{d}/SHA256SUMS").decode().splitlines():
        if not line.strip():
            continue
        h, _, name = line.partition("  ")
        name = name.strip()
        entries[name[2:] if name.startswith("./") else name] = h
    bad = []
    for name, h in entries.items():
        full = f"{d}/{name}"
        if full not in tracked_set:
            bad.append(f"listed, not tracked: {name}")
        elif hashlib.sha256(blob(CAND, full)).hexdigest() != h:
            bad.append(f"mismatch: {name}")
    under = [t[len(d) + 1:] for t in tracked if t.startswith(d + "/")]
    under = [u for u in under if u != "SHA256SUMS"]
    missing = sorted(set(under) - set(entries))
    extra = sorted(set(entries) - set(under))
    ok = not (bad or missing or extra)
    all_ok &= ok
    print(f"{short(d)}: {len(entries)} listed, {len(under)} tracked, "
          f"{len(entries) - len(bad)} verify; {'OK' if ok else 'PROBLEM'}")
    for x in bad + [f"unlisted: {m}" for m in missing] + [f"extra: {e}" for e in extra]:
        print("    ", x)
print("changed files in folders without a SHA256SUMS:")
for d in sorted(uncovered):
    print(f"    {short(d + '/')}: {len(uncovered[d])}")
print("ALL SUMS OK" if all_ok else "SOME SUMS PROBLEM")

# 4. in-place edits
print("\n== 4. in-place edits, commit by commit (first parent after "
      f"{CHAIN_START})")
for c in gits("rev-list", "--reverse", "--first-parent", "--no-merges",
              f"{CHAIN_START}..{CAND}").split():
    subj = gits("log", "-1", "--format=%h %s", c).strip()[:90]
    mods = [l.split("\t", 1)[1] for l in gits("diff", "--name-status", f"{c}^", c).splitlines()
            if l.startswith("M\t") and not l.endswith("SHA256SUMS")]
    for m in mods:
        old = blob(f"{c}^", m).decode().splitlines()
        new = blob(c, m).decode().splitlines()
        ops = [op for op in difflib.SequenceMatcher(None, old, new, autojunk=False).get_opcodes()
               if op[0] in ("replace", "delete")]
        for tag, i1, i2, j1, j2 in ops:
            o, n = "\n".join(old[i1:i2]), "\n".join(new[j1:j2])
            ins = all(t in ("equal", "insert") for t, *_ in
                      difflib.SequenceMatcher(None, o, n, autojunk=False).get_opcodes())
            print(f"{subj}\n    {short(m)} old {i1 + 1}-{i2} -> new {j1 + 1}-{j2}: "
                  f"{'insert-only' if ins else 'OLD TEXT CHANGED'}")

# 5. suites
print("\n== 5. suites (passed counts that differ)")


def suites(path):
    d = {}
    for l in blob(CAND, T3 + path).decode().splitlines():
        m = re.match(r"(\d+) passed=(\d+) failed=(\d+) ignored=(\d+)(?: targets=\d+)? (\S+)", l)
        if m:
            d[m.group(5)] = tuple(int(x) for x in m.groups()[:4])
    return d


runs = [("K1 combined (eb52114e9 tree)", "IMPLEMENTATION/K1/_run_records/combined/suites/suites.log"),
        ("M03 1d105d633", "IMPLEMENTATION/M03_SKEW_PIN_MERGE/dec025/suites.log"),
        ("K3 b7e93650e", "IMPLEMENTATION/K3_MERGE/dec025/suites.log"),
        ("K2b 33e33c723", "IMPLEMENTATION/K2B_MERGE/dec025/suites.log")]
loaded = [(n, suites(p)) for n, p in runs]
for n, d in loaded:
    fails = {k: v[2] for k, v in d.items() if v[2]}
    print(f"{n}: {len(d)} manifests; failing: {fails}")
for (na, a), (nb, b) in zip(loaded, loaded[1:]):
    diff = {k: (a.get(k), b.get(k)) for k in set(a) | set(b) if a.get(k) != b.get(k)}
    print(f"{na} -> {nb}: {diff}")

# 6. T9
print("\n== 6. T9 hash lists against the Mac calibration's native main list")


def hashes(path):
    d = {}
    for l in blob(CAND, T3 + path).decode().splitlines():
        m = re.match(r"([0-9a-f]{64})\s+\*?(.+)$", l.strip())
        if m:
            d[os.path.basename(m.group(2))] = m.group(1)
    return d


ref = hashes("PLATFORM_CALIBRATION_MAC/t9/output_sha256_main_mac_native.txt")
for p in ["IMPLEMENTATION/M03_SKEW_PIN/_run_records/t9/output_sha256_base.txt",
          "IMPLEMENTATION/M03_SKEW_PIN/_run_records/t9/output_sha256_cand.txt",
          "IMPLEMENTATION/K3/_run_records/checkpoint_b/t9/output_sha256_base.txt",
          "IMPLEMENTATION/K3/_run_records/checkpoint_b/t9/output_sha256_cand.txt",
          "IMPLEMENTATION/K2B/_run_records/checkpoint_b/t9/output_sha256_base.txt",
          "IMPLEMENTATION/K2B/_run_records/checkpoint_b/t9/output_sha256_cand.txt",
          "IMPLEMENTATION/K2B/_run_records/rv11_fixes/t9/output_sha256_base.txt",
          "IMPLEMENTATION/K2B/_run_records/rv11_fixes/t9/output_sha256_cand.txt",
          "REVIEW/_run_records/k3_review/t9/output_sha256_head.txt"]:
    d = hashes(p)
    same = sum(1 for k in ref if d.get(k) == ref[k])
    print(f"{p}: {len(d)} outputs, {same} of {len(ref)} equal to calibration")
