"""RV3 mutation driver: apply each group to the scratch worktree, run the product_physics tests, restore by
rewriting the saved original bytes (no git operation). Usage: run_rv3_mutants.py <core dir> <target dir> <out json> [ids...]"""
import json, os, re, subprocess, sys, time
sys.path.insert(0, os.path.dirname(__file__))
from rv3_mutants import MUTANTS
core, target, out_path = sys.argv[1:4]
only = sys.argv[4:]
by_id = {m[0]: m for m in MUTANTS}
GROUPS = [['M1f'], ['M2'], ['M1l'], ['M1n'], ['M3'], ['M8'], ['M9'], ['EV1-fnptr-glv'], ['EV2-reduce-E7'],
          ['EV3-selfassign-E15'], ['EV4-helper', 'EV4-sibling-E16'], ['EV5-no-sensitive-warning'],
          ['EV4-helper', 'EV6-ufcs-helper-E12']]
FULL = {'M8', 'M9', 'EV5-no-sensitive-warning'}
env = dict(os.environ, RUSTUP_TOOLCHAIN='1.97.1', RUSTUP_AUTO_INSTALL='0', CARGO_INCREMENTAL='0', CARGO_TARGET_DIR=target)
results = json.load(open(out_path)) if os.path.exists(out_path) else {}
for group in GROUPS:
    gid = group[-1]
    if only and gid not in only: continue
    originals = {}
    for mid in group:
        _, f, anchor, rep, _ = by_id[mid]
        path = os.path.join(core, f)
        text = originals.get(path) or open(path).read()
        originals.setdefault(path, text)
        cur = open(path).read()
        assert cur.count(anchor) == 1, (mid, cur.count(anchor))
        open(path, 'w').write(cur.replace(anchor, rep))
    try:
        if gid in FULL:
            cmd = ['cargo', 'test', '--offline', '--locked', '--no-fail-fast', '--lib', '--test', 's11f_site_test']
        else:
            cmd = ['cargo', 'test', '--offline', '--locked', '--no-fail-fast', '--lib', '--test', 's11f_site_test',
                   '--', 's11f_tests', 'rule_', 'sparse_direct', 'scanner']
        t0 = time.time()
        p = subprocess.run(cmd, cwd=os.path.join(core, 'product_physics'), env=env, capture_output=True, text=True)
        log = p.stdout + p.stderr
        failed = sorted(set(re.findall(r'^test (\S+) \.\.\. FAILED', log, re.M)))
        passed = len(re.findall(r'^test \S+ \.\.\. ok', log, re.M))
        compile_error = 'error[E' in log or 'error: could not compile' in log
        results[gid] = {'group': group, 'command': ' '.join(cmd), 'exit': p.returncode, 'seconds': round(time.time() - t0),
                        'compile_error': compile_error, 'failed': failed, 'passed': passed,
                        'summaries': re.findall(r'^test result: .*$', log, re.M),
                        'errors': re.findall(r'^error(?:\[E\d+\])?: .*$', log, re.M)[:5]}
        open(os.path.join(os.path.dirname(out_path), gid + '.log'), 'w').write(log)
        print(gid, 'exit', p.returncode, 'failed', len(failed), 'passed', passed, 'compile_error', compile_error, flush=True)
    finally:
        for path, text in originals.items():
            open(path, 'w').write(text)
    json.dump(results, open(out_path, 'w'), indent=1)
