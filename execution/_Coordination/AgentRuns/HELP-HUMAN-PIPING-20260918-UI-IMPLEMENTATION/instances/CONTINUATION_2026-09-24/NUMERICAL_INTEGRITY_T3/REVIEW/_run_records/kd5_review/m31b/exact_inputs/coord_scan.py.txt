"""RV5: exact-input scan of the PP-route chord effect against coordinate magnitude
(one realized R = 0.3 m elbow, PP's binary64 centre, stiff cantilever as in sweep2)."""
import math, sys, json
sys.path.insert(0, '.')
import m31b_attack as A
from ppc import pp_centre
res = []
for X0 in [1e5, 2.5e5, 5e5, 1e6, 2e6, 3.3e6, 5e6, 7.3e6]:
    for phi_deg in [2.0, 5.0, 10.0]:
        for off in [0.0, 0.37, 0.71]:
            R = 0.3; phi = math.radians(phi_deg)
            xi = [X0 + off, X0 * 0.7 + off * 1.3, 0.0]
            xj = [xi[0] + R * math.sin(phi), xi[1] + R * (1 - math.cos(phi)), 0.0]
            c = pp_centre(xi, xj, R, [0.0, 1.0, 0.0])
            ok = A.b64_formula_chord(xi, xj, c)[1]
            if not ok:
                res.append((X0, phi_deg, off, None)); continue
            m = {"name": "s", "section": A.SEC, "nodes": [xi, xj], "members": [{"i": 0, "j": 1, "y_reference": None, "bend": (c, 1.0)}],
                 "rigid": [0, 1, 2], "springs": [(3, 1e6), (4, 1e6), (5, 1e6)], "loads": [(9, 1.0), (10, 1.0), (11, 1.0)]}
            r = A.analyse(m)
            res.append((X0, phi_deg, off, r['max_actual_row']['actual'], r['max_trigger_correct'], r['max_trigger_m31b']))
for x in res:
    print(x)
json.dump(res, open('coord_scan.json', 'w'))
