#!/usr/bin/env python3
"""RV8's own K1 mutants (the K1 review). Usage: rv8_mutants.py <core-dir> <id>
Each is an exact text replacement that must match once."""
import sys

core, mid = sys.argv[1], sys.argv[2]
SP = core + "/solver/frame_kernel/src/structural/sparse.rs"
SD = core + "/solver/sparse_direct/src/structural.rs"
BLOCKS = """    for block in blocks {
        scatter_block(
            &pattern,
            &mut values,
            block.node_i,
            block.node_j,
            &block.stiffness,
        );
    }
"""
SPRINGS = """    for &(dof, stiffness) in springs {
        let index = pattern.find(dof, dof).expect("spring diagonal in pattern");
        values[index] += stiffness;
    }
"""
MUTANTS = {
    # A curved-block-only accumulation-order error: the realized bends (and
    # other explicit blocks) are added in reverse order. It changes bits only
    # where two blocks share a node.
    "RV8-BLOCK-ORDER": [(SP, "    for block in blocks {\n        scatter_block(", "    for block in blocks.iter().rev() {\n        scatter_block(")],
    # Springs added before the blocks: changes bits only where a spring sits
    # on the diagonal of a bend (block) node.
    "RV8-SPRING-FIRST": [(SP, BLOCKS + SPRINGS, SPRINGS + BLOCKS)],
    # A formation-check parity break: the sparse representation's K-D5 check
    # drops the ledger terms (rho from the folded force instead).
    "RV8-FC-TERMS": [(SP, "formation_check::check(&view, source, force_terms, scale_exponents, u, solve)",
                      "formation_check::check(&view, source, force_terms.filter(|_| false), scale_exponents, u, solve)")],
    # A curved-block-only value error: entry [0][0] of each block scaled by
    # (1 + 2^-52) before it is added (a one-ulp formation error in the bends
    # and other explicit blocks only).
    "RV8-BLOCK-VALUE": [(SP, BLOCKS, BLOCKS.replace("            &block.stiffness,\n", "            &{\n                let mut m = block.stiffness;\n                m[0][0] *= 1.0 + f64::EPSILON;\n                m\n            },\n"))],
}

SA = core + "/solver/nonlinear_integration/src/structural_adapter.rs"
NL = core + "/solver/nonlinear_integration/src/lib.rs"
LOOP_CALL = """        assembly.solve_binary64(
            stiffness,
            force,
            &reduced.free_dofs,
            &prescribed,
            linear_solve_mode,
        )?"""
DEAD_LEGACY = """            if false {
                let _ = assembly.solve_binary64(
                    stiffness,
                    force,
                    &reduced.free_dofs,
                    &prescribed,
                    linear_solve_mode,
                );
            }
"""
B64_TAIL = "        solve_prepared(structural::prepare_structural_binary64(&system)?, mode)\n    }"
EVASIONS = {
    # Evasion 1 (alias): `solve_binary64` reaches the exact prepare through a
    # `use ... as` alias; the required legacy text is kept in dead code.
    "RV8-EV-ALIAS": [
        (SA, "use open_pipe_stress_sparse_direct::structural::solve_sparse_prepared;\n",
             "use open_pipe_stress_sparse_direct::structural::solve_sparse_prepared;\nuse open_pipe_stress_frame_kernel::structural::prepare_structural as legacy_prepare;\n"),
        (SA, B64_TAIL, "        if false {\n            let _ = structural::prepare_structural_binary64(&system);\n        }\n        solve_prepared(legacy_prepare(&system)?, mode)\n    }"),
    ],
    # Evasion 2 (string text): the required legacy text only in a string
    # literal, and the exact prepare called.
    "RV8-EV-STRING": [
        (SA, B64_TAIL, "        let _legacy = \"prepare_structural_binary64(\";\n        solve_prepared(structural::prepare_structural(&system)?, mode)\n    }"),
    ],
    # Evasion 3 (new impl): a second `impl AssemblyEvidence` block with a
    # neutral name reaching the exact `solve`; the loop calls it, keeping the
    # legacy call in dead code.
    "RV8-EV-NEWIMPL": [
        (SA, "/// Deterministic storage counts of a `SparseAssemblyEvidence`.\n",
             "impl AssemblyEvidence {\n    pub(crate) fn solve_binary64_v2(\n        &self,\n        k: &[Vec<f64>],\n        f: &[f64],\n        free: &[usize],\n        prescribed: &[(usize, f64)],\n        mode: LinearSolveMode,\n    ) -> Result<StructuralSolution, StructuralError> {\n        self.solve(k, f, free, prescribed, mode)\n    }\n}\n\n/// Deterministic storage counts of a `SparseAssemblyEvidence`.\n"),
        (NL, LOOP_CALL, "        {\n" + DEAD_LEGACY + "            assembly.solve_binary64_v2(\n                stiffness,\n                force,\n                &reduced.free_dofs,\n                &prescribed,\n                linear_solve_mode,\n            )?\n        }"),
    ],
    # Evasion 4 (sibling module): a new non-test module `bridge.rs` runs the
    # exact pattern solve (K1's sparse siblings); the loop calls `bridge::run`,
    # keeping the legacy call in dead code.
    "RV8-EV-SIBLING": [
        (NL, "pub mod product_equilibrium;\n", "pub mod product_equilibrium;\nmod bridge;\n"),
        (NL, LOOP_CALL, "        {\n" + DEAD_LEGACY + "            bridge::run(\n                assembly,\n                force,\n                &reduced.free_dofs,\n                &prescribed,\n            )?\n        }"),
    ],
    # Evasion 5 (UFCS): the loop calls the pattern formation entry in fully
    # qualified form, keeping the legacy call in dead code.
    "RV8-EV-UFCS": [
        (NL, LOOP_CALL, "        {\n" + DEAD_LEGACY + """            let sk = open_pipe_stress_frame_kernel::structural::SparseStiffness::from_contributions(
                stiffness.len(),
                &assembly.contributions,
            )
            .unwrap();
            let sev = structural_adapter::SparseAssemblyEvidence::new(
                sk.pattern(),
                stiffness.len() / 6,
                &[],
                &[],
                &[],
                &[],
            )
            .unwrap();
            let mut ledger = open_pipe_stress_frame_kernel::load_ledger::LoadLedger::new();
            for (dof, &v) in force.iter().enumerate() {
                if v != 0.0 {
                    ledger.push("m", dof, v);
                }
            }
            let typed = ledger.finish(force.len()).unwrap();
            structural_adapter::SparseAssemblyEvidence::solve_assembled_with_formation_check(
                &sev,
                &sk,
                &typed,
                &reduced.free_dofs,
                &prescribed,
                linear_solve_mode,
                &[],
                true,
            )?
        }"""),
    ],
}
BRIDGE = """#![allow(unused_imports)]
use super::*;
use open_pipe_stress_frame_kernel::structural::{
    SparsePattern, SparseStiffness, SparseStructuralSystem, SparseSymmetryEvidence,
};
/// RV8 evasion helper (sibling module): the exact pattern solve.
pub(crate) fn run(
    a: &AssemblyEvidence,
    f: &[f64],
    free: &[usize],
    p: &[(usize, f64)],
) -> Result<open_pipe_stress_frame_kernel::structural::StructuralSolution, StructuralError> {
    let n = f.len();
    let pattern = SparsePattern::from_positions(n, a.contributions.iter().map(|c| (c.row, c.col)))?;
    let sk = SparseStiffness::from_pattern_and_contributions(pattern, &a.contributions)?;
    let pat = sk.pattern();
    let mut r = Vec::new();
    let mut c = Vec::new();
    for row in 0..pat.dimension() {
        for idx in pat.row_range(row) {
            let col = pat.column(idx);
            r.push(a.absolute_roundoff[row][col]);
            c.push(a.operation_counts[row][col]);
        }
    }
    let sys = SparseStructuralSystem::new(
        &sk,
        f,
        free,
        p,
        Some(&a.contributions),
        Some(SparseSymmetryEvidence { absolute_roundoff: &r, operation_counts: &c, basis: "bridge" }),
    );
    open_pipe_stress_sparse_direct::structural::solve_sparse_structural(&sys)
}
"""
MUTANTS.update(EVASIONS)
if mid not in MUTANTS:
    sys.exit("unknown mutant " + mid)
for path, old, new in MUTANTS[mid]:
    text = open(path).read()
    assert text.count(old) == 1, (mid, path, text.count(old))
    open(path, "w").write(text.replace(old, new))
if mid == "RV8-EV-SIBLING":
    open(core + "/solver/nonlinear_integration/src/bridge.rs", "w").write(BRIDGE)
print("applied", mid)
