"""RV5: emit kd5-style ModelData for the M31b counterexample models (exact u_int)."""
import json, math, sys
sys.path.insert(0, '.')
import m31b_attack as A
from ppc import pp_centre
from decimal import Decimal as D

def exact_uint(m):
    n = 6*len(m["nodes"]); free=[d for d in range(n) if d not in m["rigid"]]
    Ka,_ = A.assemble(m, "actual")
    f=[D(0)]*n
    for dof,v in m["loads"]: f[dof]+=D(float(v))
    ua = A.solve([[Ka[a][b] for b in free] for a in free], [f[a] for a in free])
    return [(a, float(ua[r])) for r,a in enumerate(free)]

def rust(name, m):
    s = A.SEC
    mem = ", ".join(
        f"MemberData {{ i: {x['i']}, j: {x['j']}, y_reference: [1.0, 0.0, 0.0], bend: Some(([{', '.join(repr(float(v)) for v in x['bend'][0])}], 1.0)) }}"
        for x in m["members"])
    nodes = ", ".join("[" + ", ".join(repr(float(v)) for v in p) + "]" for p in m["nodes"])
    return (f"pub(super) const {name}: ModelData = ModelData {{\n"
            f"    name: \"{name}\",\n"
            f"    section: SectionData {{ e: {s[0]!r}, g: {s[1]!r}, a: {s[2]!r}, i: {s[3]!r}, j: {s[4]!r} }},\n"
            f"    nodes: &[{nodes}],\n    members: &[{mem}],\n"
            f"    rigid: &{m['rigid']},\n    springs: &[{', '.join(f'({d}, {v!r})' for d,v in m['springs'])}],\n"
            f"    loads: &[{', '.join(f'({d}, {v!r})' for d,v in m['loads'])}],\n"
            f"    u_int: &[{', '.join(f'({d}, {v!r})' for d,v in exact_uint(m))}],\n}};\n")

S=[1e6,1e6,1e6]; LD=[(9,1.0),(10,1.0),(11,1.0)]
out=["// RV5 (review scratch only): M31b counterexample models, generated by gen_models.py.\n"]
specs=[("RV5_CANT90_PLANAR", 90, [0,0,1]), ("RV5_CANT60_PLANAR", 60, [0,0,1]), ("RV5_CANT30_SKEW", 30, [1,2,2]), ("RV5_CANT10_SKEW", 10, [1,2,2])]
for name, deg, nrm in specs:
    phi=math.radians(deg); s=A.max_shift(0.3, phi, nrm)
    m=A.cantilever(name, 0.3, phi, nrm, s, S, LD); out.append(rust(name, m))
# PP-route geometry: centre computed as PP computes it (no designed mismatch)
X0,R,phi=5e6,0.3,math.radians(5)
xi=[X0, X0*0.7, 0.0]; xj=[X0+R*math.sin(phi), X0*0.7+R*(1-math.cos(phi)), 0.0]
m={"name":"RV5_PP_UTM","section":A.SEC,"nodes":[xi,xj],"members":[{"i":0,"j":1,"y_reference":None,"bend":(pp_centre(xi,xj,R,[0.0,1.0,0.0]),1.0)}],
   "rigid":[0,1,2],"springs":[(3,1e6),(4,1e6),(5,1e6)],"loads":LD}
out.append(rust("RV5_PP_UTM", m))
# CSKEW_30 geometry (I3's radius-mismatch elbow) with k_X swept, for the both-sides table
for k in [1000.0, 100.0, 60.0, 45.0, 30.0]:
    mm={"name":f"x","section":A.SEC,"nodes":[[0.0,0.0,0.0],[0.3,0.0,0.3]],
        "members":[{"i":0,"j":1,"y_reference":None,"bend":([0.09999999986211418,0.19999999999999998,0.19999999986211417],1.0)}],
        "rigid":[0,1,2],"springs":[(3,k),(4,1e6),(5,1e6)],"loads":[(9,k*1e-6)]}
    out.append(rust(f"RV5_CSKEW30_K{int(k)}", mm))
for name, dphi, sh in [("RV5_NEARPI_1E6_MAXMIS", 1e-6, True), ("RV5_NEARPI_2E9_MAXMIS", 2e-9, True), ("RV5_NEARPI_2E9_EXACT", 2e-9, False)]:
    phi=math.pi-dphi; s=A.max_shift(0.3, phi, [0,0,1]) if sh else 0.0
    m=A.cantilever(name, 0.3, phi, [0,0,1], s, S, LD)
    ch, ok, ang, ri, rj = A.b64_formula_chord(m["nodes"][0], m["nodes"][1], m["members"][0]["bend"][0])
    print(name, "admissible", ok, "phi64", repr(ang), "pi-phi64", math.pi-ang, "mismatch", (rj-ri)/max(ri,rj))
    out.append(rust(name, m))
open('rv5_models.rs','w').write("".join(out))
