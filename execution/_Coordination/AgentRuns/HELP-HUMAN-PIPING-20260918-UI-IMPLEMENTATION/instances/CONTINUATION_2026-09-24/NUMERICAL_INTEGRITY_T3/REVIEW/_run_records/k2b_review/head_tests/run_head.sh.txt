#!/bin/bash
# RV11: the head (087b3a088) from a git archive: FK, SD and NI in full; PP site and runtime tests. -j 8, RUST_TEST_THREADS=4.
W=$1; REV=$2; LOG=$W/scratch/rv11/logs/head_$REV.log
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=$W/rv11-target RUST_TEST_THREADS=4
CORE=$W/scratch/rv11/src_$REV/projects/chirality-piping/core
: > $LOG
echo "start=$(date -u +%FT%TZ)" >> $LOG
for t in "solver/frame_kernel|" "solver/sparse_direct|" "solver/nonlinear_integration|" "product_physics|--test s11f_site_test --test formation_check_runtime"; do
  dir=${t%%|*}; args=${t#*|}
  echo "=== $dir: cargo test --no-fail-fast $args" >> $LOG
  (cd $CORE/$dir && cargo test --offline --locked --no-fail-fast -j 8 $args >> $LOG 2>&1); echo "rc ${dir##*/}=$?" >> $LOG
done
echo "end=$(date -u +%FT%TZ)" >> $LOG
