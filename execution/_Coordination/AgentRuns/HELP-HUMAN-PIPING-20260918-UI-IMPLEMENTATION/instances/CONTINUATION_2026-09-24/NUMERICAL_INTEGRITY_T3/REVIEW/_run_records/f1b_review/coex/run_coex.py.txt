"""RV17: run the coexistence candidates through the recorded gate probes (P1's probe with the heap cap
and the full-envelope variant): Mac main e7d930d49 (577b10d4...) and the F1b candidate 130445db2
(f4535939...), both entries, both modes. Usage: run_coex.py <main-probe> <cand-probe> <requests-dir> <out-dir>"""
import hashlib, json, os, subprocess, sys
main_bin, cand_bin, req_dir, out_dir = sys.argv[1:5]
os.makedirs(out_dir, exist_ok=True)
rows = []
for name in sorted(os.listdir(req_dir)):
    if not name.endswith('.json'):
        continue
    case = name[:-5]
    for mode in ('sparse_interactive', 'dense_scrutiny'):
        for entry in ('captured', 'typed'):
            shas = {}
            for side, binary in (('main', main_bin), ('cand', cand_bin)):
                full = os.path.join(out_dir, f'{case}__{mode}__{entry}__{side}.json')
                env = dict(os.environ, T3_FULL_ENVELOPE_PATH=full)
                args = [binary, os.path.join(req_dir, name), mode] + (['typed'] if entry == 'typed' else [])
                p = subprocess.run(args, capture_output=True, env=env, timeout=600)
                if not os.path.exists(full):
                    rows.append(dict(case=case, mode=mode, entry=entry, side=side, exit=p.returncode,
                                     note='no full envelope', stdout=p.stdout.decode()[-300:], stderr=p.stderr.decode()[-300:]))
                    continue
                data = open(full, 'rb').read()
                sha = hashlib.sha256(data).hexdigest()
                shas[side] = sha
                e = json.loads(data)
                codes = [d['code'] for d in e['diagnostics']]
                integ = [d for d in e['diagnostics'] if d['id'] == 'diagnostic:numerical-integrity:case']
                srcsel = [d for d in e['diagnostics'] if d['code'] == 'SOURCE_BLOCK_RECOVERY_SELECTED']
                unavail = [d['message'][:260] for d in e['diagnostics'] if d['code'] == 'SOURCE_BLOCK_RECOVERY_UNAVAILABLE']
                disp = [(r['id'], r['value']) for r in e['results'] if r['id'].startswith('result:disp:N1:u')]
                reac = [(r['id'].split(':')[-1], r['value']) for r in e['results'] if r['kind'] == 'support_reaction_component_v2' and r['value'] != 0.0]
                rows.append(dict(case=case, mode=mode, entry=entry, side=side, exit=p.returncode, sha=sha,
                                 mechanics=e['status']['mechanics'], selected=bool(srcsel),
                                 receipt=e.get('source_block_recovery') is not None,
                                 integrity=[(d['code'], d['severity'], d['message'][:230]) for d in integ],
                                 unavailable=unavail, codes=codes, disp=disp, reactions=reac,
                                 range_scaling=any('range_scaling' in d['message'] for d in e['diagnostics'])))
            rows.append(dict(case=case, mode=mode, entry=entry, side='compare',
                             identical=(len(shas) == 2 and shas['main'] == shas['cand'])))
with open(os.path.join(out_dir, 'coex_runs.jsonl'), 'w') as fh:
    for r in rows:
        fh.write(json.dumps(r) + '\n')
for r in rows:
    if r['side'] == 'compare':
        print(f"  {r['case']} {r['mode']} {r['entry']}: main == cand full envelope: {r['identical']}")
    else:
        print(r['case'], r['mode'], r['entry'], r['side'], r.get('mechanics'), 'selected' if r.get('selected') else '-',
              'receipt' if r.get('receipt') else '-', 'W2' if r.get('range_scaling') else '-', r.get('sha', '')[:16])
        for i in r.get('integrity', []):
            print('     integrity:', i)
        for u in r.get('unavailable', []):
            print('     unavailable:', u)
        if r.get('disp'):
            print('     disp:', r['disp'], 'reactions:', r['reactions'])
