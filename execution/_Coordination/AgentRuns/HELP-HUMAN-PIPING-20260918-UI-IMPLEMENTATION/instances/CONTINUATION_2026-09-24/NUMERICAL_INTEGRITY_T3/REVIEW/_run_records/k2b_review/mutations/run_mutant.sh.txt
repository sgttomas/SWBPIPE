#!/bin/bash
# RV11: one mutant on a clean git-archive copy of the head 087b3a088 (execution/ excluded), one clean
# target, both deleted after the run. -j 4, RUST_TEST_THREADS=4. Tests (each --no-fail-fast): FK, SD and
# NI in full; PP --test s11f_site_test --test formation_check_runtime (I10's harness).
W=$1; m=$2; REV=087b3a088
X=$W/scratch/rv11/mut; D=$X/work/$m; T=$W/rv11-target/mut_$m; LOG=$X/logs/$m.log
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=$T CARGO_BUILD_JOBS=4 RUST_TEST_THREADS=4
rm -rf $D $T; mkdir -p $D
git -C $W/numerics archive --format=tar $REV projects/chirality-piping ':(exclude)projects/chirality-piping/execution' | tar -x -m -C $D
CORE=$D/projects/chirality-piping/core
: > $LOG
if [ "$m" != NONE ]; then
  $X/apply.sh $X $CORE $m >> $LOG 2>&1 || { echo "$m PATCH-FAILED" >> $X/MUTANTS.txt; rm -rf $D $T; exit 1; }
else
  echo "control: no patch" >> $LOG
fi
echo "start=$(date -u +%FT%TZ)" >> $LOG
rc_all=""
for t in "solver/frame_kernel|" "solver/sparse_direct|" "solver/nonlinear_integration|" "product_physics|--test s11f_site_test --test formation_check_runtime"; do
  dir=${t%%|*}; args=${t#*|}
  echo "=== $dir: cargo test --no-fail-fast $args" >> $LOG
  (cd $CORE/$dir && cargo test --offline --locked --no-fail-fast -j 4 $args >> $LOG 2>&1); rc_all="$rc_all ${dir##*/}=$?"
done
echo "end=$(date -u +%FT%TZ)" >> $LOG
killers=$(/usr/bin/grep -E '^test .* FAILED$' $LOG | awk '{print $2}' | sort -u | tr '\n' ' ')
comp=$(/usr/bin/grep -c '^error\[\|^error: could not compile' $LOG)
sites=$(/usr/bin/grep -oE "panicked at [^ ]+:[0-9]+:[0-9]+" $LOG | sed 's/panicked at //' | sort | uniq -c | awk '{print $2"x"$1}' | tr '\n' ' ')
passed=$(/usr/bin/grep -E '^test result' $LOG | awk '{s+=$4} END {print s}')
echo "$m exit:$rc_all compile_errors=$comp passed_total=$passed killed_by: ${killers:-NONE} panic_sites: ${sites:-NONE}" >> $X/MUTANTS.txt
rm -rf $D $T
