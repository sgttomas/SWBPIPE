#!/usr/bin/env python3
"""RV2 (K3a review): an independent correct-rounding oracle and spot-check vectors.

Written by the reviewer without reusing gen_wide_vectors.py. Standard library only.
Value tokens follow wide.rs's Debug format: Z+/Z-, or <sign><32 hex>p<e>, value =
sign * m * 2^(e-127) with m normalized (bit 127 set).

  python3 rv2_oracle.py gen  OUT_VECTORS      write probe input lines
  python3 rv2_oracle.py check PROBE_OUTPUT     compare the Rust results with the oracle
"""
import random
import sys
from fractions import Fraction as Fr

TOP = 1 << 127
FULL = (1 << 128) - 1


def tok_of(neg, exp, sig):
    if sig == 0:
        return "Z-" if neg else "Z+"
    return "%s%032xp%d" % ("-" if neg else "+", sig, exp)


def parse(t):
    if t in ("Z+", "Z-"):
        return Fr(0), t == "Z-"
    neg = t[0] == "-"
    h, e = t[1:].split("p")
    v = Fr(int(h, 16)) * Fr(2) ** (int(e) - 127)
    return (-v if neg else v), neg


def from_fr_exact(x):
    """Token of an exactly representable (<= 128 bits) nonzero Fraction."""
    assert x != 0
    neg = x < 0
    a = abs(x)
    assert a.denominator & (a.denominator - 1) == 0
    n, d = a.numerator, a.denominator
    # a = n / d, d = 2^k
    k = d.bit_length() - 1
    while n % 2 == 0:
        n //= 2
        k -= 1
    bl = n.bit_length()
    assert bl <= 128, bl
    exp = bl - 1 - k
    return tok_of(neg, exp, n << (128 - bl))


def ilog2(a):
    """floor(log2(a)) for a positive Fraction, by exact comparison."""
    e = a.numerator.bit_length() - a.denominator.bit_length()
    while Fr(2) ** e > a:
        e -= 1
    while Fr(2) ** (e + 1) <= a:
        e += 1
    return e


def round_p(x, p):
    """Nearest p-bit value, ties to even, as a token; decided by exact comparison
    of x with the two neighbouring p-bit values and their midpoint."""
    if x == 0:
        raise ValueError
    neg = x < 0
    a = abs(x)
    e = ilog2(a)
    ulp = Fr(2) ** (e - p + 1)
    lo_int = int(a / ulp)  # floor, a/ulp in [2^(p-1), 2^p)
    lo = lo_int * ulp
    hi = lo + ulp
    mid = lo + ulp / 2
    if a < mid:
        q = lo_int
    elif a > mid:
        q = lo_int + 1
    else:
        q = lo_int if lo_int % 2 == 0 else lo_int + 1
    v = q * ulp
    return from_fr_exact(-v if neg else v)


def round_sqrt_p(x, p):
    """Nearest p-bit value of sqrt(x), decided by comparing squares exactly."""
    e = ilog2(x) // 2  # sqrt(x) in [2^e, 2^(e+1)) up to one binade; fix below
    # find e with 2^e <= sqrt(x) < 2^(e+1) exactly: 4^e <= x < 4^(e+1)
    while Fr(4) ** e > x:
        e -= 1
    while Fr(4) ** (e + 1) <= x:
        e += 1
    ulp = Fr(2) ** (e - p + 1)
    # bisection for the largest q with (q ulp)^2 <= x, q in [2^(p-1), 2^p)
    lo, hi = 1 << (p - 1), 1 << p
    while hi - lo > 1:
        m = (lo + hi) // 2
        if (m * ulp) ** 2 <= x:
            lo = m
        else:
            hi = m
    q = lo
    mid2 = ((q + Fr(1, 2)) * ulp) ** 2
    if x > mid2:
        q += 1
    elif x == mid2:  # a tie (possible only when 2(p+1) bits fit the operand)
        q += q % 2
    return from_fr_exact(q * ulp)


def oracle(op, p, a_t, b_t):
    a, an = parse(a_t)
    if op == "sqrt":
        if a == 0:
            return a_t
        if a < 0:
            return "E:NegativeSqrt"
        return round_sqrt_p(a, p)
    b, bn = parse(b_t)
    if op == "sub":
        b, bn = -b, not bn
        op = "add"
    if op == "add":
        if a == 0 and b == 0:
            return "Z-" if (an and bn) else "Z+"
        s = a + b
        return "Z+" if s == 0 else round_p(s, p)
    if op == "mul":
        if a == 0 or b == 0:
            return "Z-" if an != bn else "Z+"
        return round_p(a * b, p)
    if op == "div":
        if b == 0:
            return "E:DivisionByZero"
        if a == 0:
            return "Z-" if an != bn else "Z+"
        return round_p(a / b, p)
    raise ValueError(op)


# ---------------------------------------------------------------------------
def W(neg, m, lsb):
    """Token of (-1)^neg * m * 2^lsb, m < 2^128, m > 0."""
    v = Fr(m) * Fr(2) ** lsb
    return from_fr_exact(-v if neg else v)


def gen(path):
    R = random.Random(0x52563220)  # "RV2 "
    lines = []
    PS = (53, 64, 65, 127, 128)

    def odd(bits):
        return (1 << (bits - 1)) | R.getrandbits(bits - 1) | 1

    for p in PS:
        # random operands of assorted lengths
        for _ in range(300):
            for op in ("add", "sub", "mul", "div"):
                la, lb = R.randint(1, 128), R.randint(1, 128)
                a = W(R.random() < .5, odd(la), R.randint(-300, 300))
                b = W(R.random() < .5, odd(lb), R.randint(-300, 300))
                lines.append("%s %d %s %s" % (op, p, a, b))
            lines.append("sqrt %d %s -" % (p, W(False, odd(R.randint(1, 128)), R.randint(-300, 300))))
        # exact ties in add/sub: the sum has p+1 significant bits ending in 1,
        # with both parities of the kept part
        for par in (0, 1):
            for _ in range(20):
                x = (1 << p) | (R.getrandbits(p - 1) << 1) | 1  # p+1 bits, odd
                x = (x & ~2) | (par << 1)
                y = odd(R.randint(1, min(64, p)))
                if y >= x:
                    continue
                s = R.randint(-100, 100)
                neg = R.random() < .5
                if (x - y).bit_length() <= 128:
                    lines.append("add %d %s %s" % (p, W(neg, x - y, s), W(neg, y, s)))
                if (x + y).bit_length() <= 128:
                    lines.append("sub %d %s %s" % (p, W(neg, x + y, s), W(neg, y, s)))
                # the same tie plus a far sticky bit either side (gap > 256)
                if x.bit_length() <= 128:
                    lines.append("add %d %s %s" % (p, W(neg, x, s), W(not neg, 1, s - 300)))
                    lines.append("add %d %s %s" % (p, W(neg, x, s), W(neg, 1, s - 300)))
                    lines.append("add %d %s %s" % (p, W(neg, x, s), W(neg, 1, s - 140)))
                    lines.append("add %d %s %s" % (p, W(neg, x, s), W(not neg, odd(90), s - 200)))
                else:  # p = 128: the tie comes from the sum, plus a far bit
                    lines.append("add %d %s %s" % (p, W(neg, x - y, s), W(neg, y, s)))
        # multiplication ties: odd factors whose product has exactly p+1 bits
        n = 0
        while n < 30:
            l1 = R.randint(max(1, p + 1 - 128), min(128, p))
            f1 = odd(l1)
            l2 = p + 1 - l1 + R.randint(0, 1)
            if not 1 <= l2 <= 128:
                continue
            f2 = odd(l2)
            if (f1 * f2).bit_length() != p + 1:
                continue
            lines.append("mul %d %s %s" % (p, W(R.random() < .5, f1, R.randint(-50, 50)), W(R.random() < .5, f2, R.randint(-50, 50))))
            n += 1
        # division ties and near ties: a = b*q (+-1)
        for _ in range(30):
            q = odd(p + 1)
            lb = R.randint(1, 128 - (p + 1)) if p < 127 else 1
            b = odd(lb)
            a = b * q
            if a.bit_length() <= 128:
                lines.append("div %d %s %s" % (p, W(False, a, 0), W(True, b, R.randint(-9, 9))))
            for d in (-1, 1):
                if (a + d).bit_length() <= 128:
                    lines.append("div %d %s %s" % (p, W(False, a + d, 0), W(False, b, 0)))
            # near-midpoint quotient with 128-bit divisor
            bb = odd(128)
            qq = odd(p + 1)
            num = bb * qq  # exact tie if representable; round the dividend to 128 bits
            sh = max(0, num.bit_length() - 128)
            for dd in (0, 1, -1):
                aa = (num >> sh) + dd
                if 0 < aa <= FULL:
                    lines.append("div %d %s %s" % (p, W(False, aa, sh), W(False, bb, 0)))
        # square root: perfect squares, +-1 lsb, ties (p<=63) and near-midpoints
        for _ in range(30):
            x = odd(R.randint(1, 64))
            sq = x * x
            e2 = 2 * R.randint(-60, 60)
            for d in (0, 1, -1):
                if sq + d > 0 and (sq + d).bit_length() <= 128:
                    lines.append("sqrt %d %s -" % (p, W(False, sq + d, e2)))
            # (m + 1/2)^2 for m of p bits: exact midpoint squared, cut to 128 bits
            m = (1 << (p - 1)) | R.getrandbits(p - 1)
            t2 = (2 * m + 1) ** 2  # = 4 (m+1/2)^2
            sh = max(0, t2.bit_length() - 128)
            for dd in (0, 1, -1):
                v = (t2 >> sh) + dd
                if 0 < v <= FULL:
                    lines.append("sqrt %d %s -" % (p, W(False, v, sh - 2 + 2 * R.randint(-5, 5))))
        # massive cancellation
        for _ in range(30):
            m = TOP | R.getrandbits(127)
            e = R.randint(-200, 200)
            m2 = min(FULL, m + R.randint(1, 3))
            lines.append("sub %d %s %s" % (p, tok_of(False, e, m), tok_of(False, e, m2)))
            lines.append("add %d %s %s" % (p, tok_of(False, e, TOP), tok_of(True, e - 1, FULL)))
            lines.append("sub %d %s %s" % (p, tok_of(True, e, m), tok_of(True, e, m)))
            lines.append("sub %d %s %s" % (p, tok_of(False, e + 1, TOP | 1), tok_of(False, e, FULL)))
        # carries
        lines.append("add %d %s %s" % (p, tok_of(False, 0, FULL), tok_of(False, -128, TOP)))
        lines.append("add %d %s %s" % (p, tok_of(False, 0, FULL), tok_of(False, -127, TOP)))
        lines.append("mul %d %s %s" % (p, tok_of(False, 0, FULL), tok_of(False, 0, FULL)))
        lines.append("sqrt %d %s -" % (p, tok_of(False, 1, FULL)))
        lines.append("sqrt %d %s -" % (p, tok_of(False, 0, FULL)))
        # signed zeros
        for a, b in (("Z+", "Z-"), ("Z-", "Z-"), ("Z-", "Z+")):
            for op in ("add", "sub", "mul", "div"):
                lines.append("%s %d %s %s" % (op, p, a, b))
        lines.append("sqrt %d Z- -" % p)
    # f64 lift: random patterns, subnormals, zeros, extremes, non-finite
    pats = [0, 1 << 63, 1, (1 << 52) - 1, 1 << 52, 0x7FEFFFFFFFFFFFFF, 0x7FF0000000000000,
            0xFFF0000000000000, 0x7FF8000000000001, 0x8000000000000001, 0x3FF0000000000000]
    for _ in range(3000):
        r = R.getrandbits(64)
        if R.random() < .3:
            r &= (1 << 63) | ((1 << 52) - 1)
        pats.append(r)
    for b in pats:
        lines.append("f64 %016x" % b)
    # split: random values across the binary64 range and below 2^-1074
    for _ in range(3000):
        e = R.choice([R.randint(-1250, 1030), R.randint(-1130, -1020), R.randint(-1210, -1070)])
        sig = TOP | R.getrandbits(127)
        if R.random() < .3:
            k = R.randint(1, 128)
            sig = (sig >> (128 - k)) << (128 - k)
        lines.append("split %s" % tok_of(R.random() < .5, e, sig))
    lines = [l for l in lines if l]
    with open(path, "w") as fh:
        fh.write("\n".join(lines) + "\n")
    print(len(lines), "lines")


def f64_value(b):
    neg = b >> 63
    be = (b >> 52) & 0x7FF
    fr = b & ((1 << 52) - 1)
    if be == 0x7FF:
        return None
    v = Fr(fr, 1 << 52) * Fr(2) ** -1022 if be == 0 else (1 + Fr(fr, 1 << 52)) * Fr(2) ** (be - 1023)
    return neg, (-v if neg else v)


def check(path):
    bad = 0
    counts = {}
    for line in open(path):
        lhs, rhs = line.rstrip("\n").split(" => ")
        f = lhs.split()
        op = f[0]
        counts[op] = counts.get(op, 0) + 1
        if op == "f64":
            b = int(f[1], 16)
            fv = f64_value(b)
            if fv is None:
                exp = "E:NonFinite"
            else:
                neg, v = fv
                exp = ("Z-" if neg else "Z+") if v == 0 else from_fr_exact(v)
            if rhs != exp:
                bad += 1
                print("MISMATCH", line.strip(), "expected", exp)
            continue
        if op == "split":
            v, neg = parse(f[1])
            r = rhs.split()
            if r[0].startswith("E:"):
                if not (abs(v) >= Fr(2) ** 1024 and r[0] == "E:SplitOverflow"):
                    bad += 1
                    print("MISMATCH", line.strip())
                continue
            if abs(v) >= Fr(2) ** 1024:
                bad += 1
                print("MISMATCH not refused", line.strip())
                continue
            n = int(r[0])
            terms = [int(x, 16) for x in r[1:1 + n]]
            flag = r[1 + n] == "1"
            tv = [f64_value(t)[1] for t in terms]
            rem = v - sum(tv, Fr(0))
            ok = n <= 3 and all(t != 0 and (t < 0) == (v < 0) for t in tv)
            ok = ok and all(abs(tv[i]) > abs(tv[i + 1]) for i in range(n - 1))
            # exact iff no set bit of v lies below 2^-1074
            scaled = v * Fr(2) ** 1074
            exact_expected = scaled.denominator == 1
            if exact_expected:
                ok = ok and rem == 0 and not flag
            else:
                ok = ok and flag and rem != 0 and abs(rem) < Fr(2) ** -1074 and (rem < 0) == (v < 0)
                # truncation toward zero: the kept part is v with the sub-2^-1074 bits removed
                kept = Fr(int(abs(scaled)), 1) * Fr(2) ** -1074
                ok = ok and abs(sum(tv, Fr(0))) == kept
            if not ok:
                bad += 1
                print("MISMATCH", line.strip())
            continue
        p = int(f[1])
        got = rhs.split()[0]
        exp = oracle(op, p, f[2], f[3] if len(f) > 3 else "-")
        if got != exp:
            bad += 1
            print("MISMATCH", line.strip(), "expected", exp)
    print("checked", counts, "mismatches", bad)
    return bad


if __name__ == "__main__":
    if sys.argv[1] == "gen":
        gen(sys.argv[2])
    else:
        sys.exit(1 if check(sys.argv[2]) else 0)
