#!/usr/bin/env python3
"""RV14: add the delta check's records to REVIEW/_run_records/k5_review/delta/ (machine paths
replaced by placeholders, as build_records.py does), then rewrite k5_review/SHA256SUMS over the
whole folder. Standard library only."""
import hashlib, os, shutil, sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import build_records as b  # noqa: E402  (clean(), OUT, S)

S, OUT = b.S, b.OUT
D = os.path.join(S, "delta")
M = os.path.join(S, "mut")
DOUT = os.path.join(OUT, "delta")


def put(src, rel):
    dst = os.path.join(DOUT, rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    open(dst, "w", encoding="utf-8").write(b.clean(open(src, encoding="utf-8").read()))


def main():
    if os.path.isdir(DOUT):
        shutil.rmtree(DOUT)
    put(os.path.join(D, "README_delta.txt"), "README.txt")
    for f in ["revisions_delta.txt", "records_checks_delta.txt", "compare_old_new.txt",
              "report_all.txt", "report_huge.txt", "report_rv14_4.txt", "report_delta.txt",
              "results_v2_delta.txt", "results_v2_delta2.txt", "pp_runs_head.txt",
              "probes_identity.txt", "ni_build.txt", "memguard_delta.txt"]:
        put(os.path.join(D, f), f)
    for f in ["cases_delta.txt", "cases_delta2.txt"]:
        put(os.path.join(S, "oracle", f), f)
    put(os.path.join(D, "rv14_fk_probe_v2.rs"), "rv14_fk_probe_v2.rs.txt")
    for f in ["rv14_mutate_delta.py", "rv14_run_mutant_delta.sh", "batch_delta.sh"]:
        put(os.path.join(M, f), f"mutations/{f}.txt")
    for f in ["queue_delta.txt", "batch_delta.log", "MUTANTS_DELTA.txt"]:
        put(os.path.join(M, f), f"mutations/{f}")
    for d in sorted(os.listdir(os.path.join(M, "delta_logs"))):
        for f in sorted(os.listdir(os.path.join(M, "delta_logs", d))):
            put(os.path.join(M, "delta_logs", d, f), f"mutations/logs/{d}/{f}")
    put(os.path.abspath(__file__), "build_delta_records.py.txt")
    entries = []
    for root, _, files in os.walk(OUT):
        for f in files:
            if f == "SHA256SUMS":
                continue
            p = os.path.join(root, f)
            entries.append(("./" + os.path.relpath(p, OUT), hashlib.sha256(open(p, "rb").read()).hexdigest()))
    entries.sort()
    with open(os.path.join(OUT, "SHA256SUMS"), "w") as out:
        for rel, h in entries:
            out.write(f"{h}  {rel}\n")
    import re
    bad = re.compile(r"(?<![A-Za-z0-9_.{-])(?:/(?:Users|private|home|tmp|var/folders)/[^\s)\"'`<>]+"
                     r"|[A-Za-z]:\\Users\\[A-Za-z0-9._-]+[^\s`'\"<>)\]]*)")
    hits = [(rel, m.group(0)) for rel, _ in entries
            for m in bad.finditer(open(os.path.join(OUT, rel), encoding="utf-8").read())]
    print(f"files={len(entries)} delta_files={sum(r.startswith('./delta/') for r, _ in entries)} "
          f"machine-path hits={len(hits)}")
    return 1 if hits else 0


if __name__ == "__main__":
    sys.exit(main())
