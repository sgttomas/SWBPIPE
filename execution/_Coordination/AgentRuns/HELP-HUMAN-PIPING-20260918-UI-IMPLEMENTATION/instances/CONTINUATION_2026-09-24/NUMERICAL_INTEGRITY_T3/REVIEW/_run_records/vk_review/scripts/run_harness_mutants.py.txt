#!/usr/bin/env python3
"""RV21: five harness mutants of RV21's own, each an exact edit of a VR source
in the scratch archive, run against VR's committed test set (not RV21's probe
files), then reverted. One cargo job at -j 4. python3 -B only."""
import sys
sys.dont_write_bytecode = True
import os
import re
import subprocess
import time
from pathlib import Path

WT = Path('<wt>')
VR = WT / 'rv21/head/projects/chirality-piping/validation/benchmarks/numerical_robustness'
LOGS = WT / 'rv21/logs/harness_mutants'
TESTS = ['--lib'] + sum((['--test', t] for t in ('lane', 'parity', 'adapter', 'files', 'engine', 'rcm',
                                                  'invariance', 'feature_guard', 'scale')), [])

MUTANTS = [
    ('RV21-H1', 'src/exact.rs', 'the floor comparison strict (>) instead of >=',
     'comparison.cmp(&floor) != Ordering::Less', 'comparison.cmp(&floor) == Ordering::Greater'),
    ('RV21-H2', 'src/exact.rs', 'the tolerance 1e-8 instead of 1e-9',
     'Exact::max(&exp.abs(), scale).scaled(0, -9)', 'Exact::max(&exp.abs(), scale).scaled(0, -8)'),
    ('RV21-H3', 'src/lane.rs', 'N and ext read from the i end',
     'Target::Axial(m) | Target::Extension(m) => vec![end(m, true, Component::Ux)]',
     'Target::Axial(m) | Target::Extension(m) => vec![end(m, false, Component::Ux)]'),
    ('RV21-H4', 'src/lane.rs', 'an overflowed row observed as +0 (plan 5.2: Overflow fails)',
     'Some(Binary64Outcome::Overflow { .. }) => Err(format!("{id:?} overflowed")),',
     'Some(Binary64Outcome::Overflow { .. }) => Ok(0.0),'),
    ('RV21-H5', 'src/compare.rs', 'a magnitude judged on |My| alone',
     'Observed::Magnitude(y, z) => Some(exact::magnitude_predicate(*y, *z, exp, scale)),',
     'Observed::Magnitude(y, _z) => Some(exact::magnitude_predicate(*y, 0.0, exp, scale)),'),
    ('RV21-H6', 'src/compare.rs', 'a sub-range reference passes as absolute-range whatever the observation',
     'Some(true) if exact::outside_binary64(exp) => Verdict::PassAbsoluteRange,',
     'Some(_) if exact::outside_binary64(exp) => Verdict::PassAbsoluteRange,'),
]


def run(mid):
    env = dict(os.environ, RUSTUP_TOOLCHAIN='1.97.1', RUSTUP_AUTO_INSTALL='0', CARGO_INCREMENTAL='0',
               CARGO_TARGET_DIR=str(WT / 'rv21-target'), RUST_TEST_THREADS='2')
    env.pop('FK_SEEDED_FAULT', None)
    t0 = time.time()
    p = subprocess.run(['cargo', 'test', '--offline', '--locked', '-j', '4', '--no-fail-fast'] + TESTS,
                       cwd=VR, env=env, capture_output=True, text=True)
    out = p.stdout + p.stderr
    (LOGS / (mid + '.log')).write_text(out)
    failed = re.findall(r'^test (\S+) \.\.\. FAILED$', out, re.M)
    passed = len(re.findall(r'^test \S+ \.\.\. ok$', out, re.M))
    compiled = 'error[' not in out
    return p.returncode, passed, failed, compiled, time.time() - t0


def main():
    LOGS.mkdir(parents=True, exist_ok=True)
    rc, passed, failed, ok, secs = run('NONE')
    print('NONE rc=%d passed=%d failed=%s compiled=%s %.0fs' % (rc, passed, failed, ok, secs), flush=True)
    for mid, rel, what, old, new in MUTANTS:
        path = VR / rel
        text = path.read_text()
        assert text.count(old) == 1, (mid, text.count(old))
        path.write_text(text.replace(old, new))
        try:
            rc, passed, failed, ok, secs = run(mid)
        finally:
            path.write_text(text)
        verdict = 'killed' if (failed or not ok or rc != 0) else 'SURVIVED'
        print('%s %s (%s): rc=%d passed=%d failed=%s compiled=%s %.0fs'
              % (mid, verdict, what, rc, passed, failed, ok, secs), flush=True)


if __name__ == '__main__':
    main()
