#!/bin/bash
# RV14: one mutant in its own clean `git archive b379e5b27` copy and its own target
# (deleted afterwards). Usage: rv14_run_mutant.sh <id> <sets: fk,ni,pp,probe>
set -u
ID=$1; SETS=$2
W=<wt>
M=$W/scratch/rv14/mut; D=$M/copies/$ID; OUT=$M/logs/$ID; T=$W/rv14-target/mut-$ID
PY=$W/../../projects/chirality-piping/.venv/bin/python
rm -rf $D $OUT $T; mkdir -p $D $OUT
(cd $W/k5 && git archive b379e5b27 projects/chirality-piping/core projects/chirality-piping/fixtures projects/chirality-piping/schemas projects/chirality-piping/validation | tar -x -C $D)
CORE=$D/projects/chirality-piping/core
if [ "${3:-rv14}" = "k1" ]; then $PY -B $M/mutate_k1.py $CORE $ID > $OUT/apply.txt 2>&1; else $PY -B $M/rv14_mutate.py $CORE $ID > $OUT/apply.txt 2>&1; fi; echo "apply exit=$? (script ${3:-rv14})" >> $OUT/apply.txt
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2 CARGO_TARGET_DIR=$T
for s in $(echo $SETS | tr ',' ' '); do
  case $s in
    fk) (cd $CORE/solver/frame_kernel && cargo test --offline --locked -j 4 --no-fail-fast --test k5_constrained_bodies --test k5_scale --lib -- k5_ rigid_body:: > $OUT/fk.txt 2>&1; echo "exit=$?" >> $OUT/fk.txt) ;;
    ni) (cd $CORE/solver/nonlinear_integration && cargo test --offline --locked -j 4 --no-fail-fast --lib > $OUT/ni.txt 2>&1; echo "exit=$?" >> $OUT/ni.txt) ;;
    pp) (cd $CORE/product_physics && cargo test --offline --locked -j 4 --no-fail-fast --test k5_curved_mechanism_runtime > $OUT/pp.txt 2>&1; echo "exit=$?" >> $OUT/pp.txt) ;;
    probe) cp $W/scratch/rv14/probe/rv14_fk_probe.rs $CORE/solver/frame_kernel/tests/ && (cd $CORE/solver/frame_kernel && RV14_IN=$W/scratch/rv14/oracle/cases_all.txt RV14_OUT=$OUT/probe_results.txt cargo test --offline --locked -j 4 --test rv14_fk_probe > $OUT/probe.txt 2>&1; echo "exit=$?" >> $OUT/probe.txt) ;;
    saprobe) mkdir -p $CORE/solver/nonlinear_integration/tests && cp $W/scratch/rv14/probe/rv14_sa_mutprobe.rs $CORE/solver/nonlinear_integration/tests/ && (cd $CORE/solver/nonlinear_integration && RV14_OUT=$OUT/saprobe_results.txt cargo test --offline --locked -j 4 --test rv14_sa_mutprobe > $OUT/saprobe.txt 2>&1; echo "exit=$?" >> $OUT/saprobe.txt) ;;
    saprobe2) mkdir -p $CORE/solver/nonlinear_integration/tests && cp $W/scratch/rv14/probe/rv14_sa_probe.rs $CORE/solver/nonlinear_integration/tests/ && (cd $CORE/solver/nonlinear_integration && RV14_OUT=$OUT/saprobe2_results.txt cargo test --offline --locked -j 4 --test rv14_sa_probe > $OUT/saprobe2.txt 2>&1; echo "exit=$?" >> $OUT/saprobe2.txt) ;;
    ppprobe) cp $W/scratch/rv14/probe/rv14_pp_probe.rs $CORE/product_physics/tests/ && (cd $CORE/product_physics && cargo test --offline --locked -j 4 --test rv14_pp_probe -- --nocapture > $OUT/ppprobe.txt 2>&1; echo "exit=$?" >> $OUT/ppprobe.txt) ;;
  esac
done
rm -rf $T $D
echo "$ID done $(date -u +%FT%TZ)" > $OUT/done.txt
