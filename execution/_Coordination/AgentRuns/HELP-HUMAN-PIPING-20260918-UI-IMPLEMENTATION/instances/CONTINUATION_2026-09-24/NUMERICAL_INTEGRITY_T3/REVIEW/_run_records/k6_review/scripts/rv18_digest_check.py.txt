"""RV18: re-derive the binary's staged-vs-entry and repeat parity from the recorded digests,
without trusting its `equal` flags; and check each summary's heap peaks against its stage lines."""
import json, os, sys
from collections import Counter
d = sys.argv[1]
recs = [json.loads(l) for l in open(os.path.join(d, 'records.jsonl')) if l.strip()]
runs = [r for r in recs if 'order' in r and r.get('classification') not in (None, 'not_run')]
c = Counter(); problems = []
for r in runs:
    objs = [json.loads(l) for l in open(os.path.join(d, r['run_id'] + '.jsonl')) if l.startswith('{')]
    outcome = {o['repeat']: o for o in objs if o.get('kind') == 'outcome'}
    for o in objs:
        if o.get('kind') != 'parity':
            continue
        if o['item'] == 'staged_vs_entry_checked':
            s = outcome[o['repeat']]
            eq = (s['solution_debug_len'], s['solution_debug_fnv64']) == (o['entry_debug_len'], o['entry_debug_fnv64'])
            c[('staged_vs_entry_checked', eq, o['equal'])] += 1
            chk = o
        elif o['item'] == 'entry_plain_vs_entry_checked':
            eq = (o['entry_plain_debug_len'], o['entry_plain_debug_fnv64']) == (chk['entry_debug_len'], chk['entry_debug_fnv64'])
            c[('entry_plain_vs_entry_checked', eq, o['equal'])] += 1
    if r['mode'] in ('sparse', 'dense'):
        digests = {(o['solution_debug_len'], o['solution_debug_fnv64']) for o in outcome.values()}
    else:
        digests = {(o['solution_len'], o['solution_fnv64']) for o in outcome.values()}
    if len(outcome) > 1:
        c[('repeat_digests_single', len(digests) == 1)] += 1
    stages = [o for o in objs if o.get('kind') == 'stage']
    summ = [o for o in objs if o.get('kind') == 'summary']
    if summ:
        s = summ[0]
        mp = max(o['heap_peak'] for o in stages); mm = max(o['heap_peak_move'] for o in stages)
        if (s['repeats_heap_peak'], s['repeats_heap_peak_move']) != (mp, mm):
            problems.append((r['run_id'], 'summary peak != max stage peak'))
        if s['heap_peak'] < s['repeats_heap_peak']:
            problems.append((r['run_id'], 'overall peak below repeats peak'))
        if r['repeats_heap_peak'] != s['repeats_heap_peak']:
            problems.append((r['run_id'], 'record != summary'))
for k in sorted(c, key=str):
    print(k, c[k])
print('summary/stage peak problems:', problems)
