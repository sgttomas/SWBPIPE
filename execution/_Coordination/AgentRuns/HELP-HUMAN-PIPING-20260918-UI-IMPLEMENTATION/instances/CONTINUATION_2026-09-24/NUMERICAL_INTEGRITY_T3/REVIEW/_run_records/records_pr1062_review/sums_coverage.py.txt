#!/usr/bin/env python3
"""RV25: SHA256SUMS coverage and verification for PR #1062's changed paths.

Run from the repository root of a clean Git working tree at <head>:
    python3 sums_coverage.py.txt <base> <head> <prefix>
<prefix> is the folder under which SHA256SUMS files are considered (the
_Coordination folder). Hashes are computed from `git cat-file` blob bytes at
<head>, not from the working tree, so untracked files cannot interfere.

For every SHA256SUMS file at <head> under <prefix> whose folder is an ancestor
of a changed path (or which is itself changed), it reports:
  - entries, entries verified against <head> blobs, entries failing/missing;
  - tracked files in its folder not listed (excluding SHA256SUMS files and
    files under a nested folder that has its own SHA256SUMS);
  - listed entries that resolve outside the folder.
It then lists changed paths that no SHA256SUMS lists at all.
"""
import hashlib
import os
import posixpath
import subprocess
import sys


def git(*args):
    return subprocess.run(["git", *args], check=True, capture_output=True).stdout


def main():
    base, head, prefix = sys.argv[1], sys.argv[2], sys.argv[3].rstrip("/")
    changed = git("diff", "--name-only", base, head).decode().split("\n")
    changed = [c for c in changed if c]
    tree = git("ls-tree", "-r", head, "--", prefix).decode().splitlines()
    blobs = {}
    for line in tree:
        meta, path = line.split("\t", 1)
        blobs[path] = meta.split()[2]
    sums_files = sorted(p for p in blobs if posixpath.basename(p) == "SHA256SUMS")
    sums_dirs = {posixpath.dirname(p) for p in sums_files}

    def sha(path):
        return hashlib.sha256(git("cat-file", "blob", blobs[path])).hexdigest()

    relevant = []
    for s in sums_files:
        d = posixpath.dirname(s)
        if s in changed or any(c.startswith(d + "/") for c in changed):
            relevant.append(s)

    listed_anywhere = set()
    print(f"relevant SHA256SUMS files: {len(relevant)}")
    all_ok = True
    for s in relevant:
        d = posixpath.dirname(s)
        text = git("cat-file", "blob", blobs[s]).decode()
        entries = []
        for line in text.splitlines():
            if not line.strip():
                continue
            h, name = line.split(None, 1)
            name = name.lstrip("*")
            entries.append((h, posixpath.normpath(posixpath.join(d, name))))
        ok = bad = missing = outside = 0
        bad_list = []
        for h, p in entries:
            listed_anywhere.add(p)
            if not p.startswith(d + "/"):
                outside += 1
            if p not in blobs:
                missing += 1
                bad_list.append(("MISSING", p))
            elif sha(p) != h:
                bad += 1
                bad_list.append(("FAILED", p))
            else:
                ok += 1
        listed = {p for _, p in entries}
        # tracked files this SUMS should list: files in d, excluding SUMS files
        # and files under nested folders that have their own SUMS
        should = set()
        for p in blobs:
            if not p.startswith(d + "/") or posixpath.basename(p) == "SHA256SUMS":
                continue
            parent = posixpath.dirname(p)
            nested = False
            while parent != d:
                if parent in sums_dirs:
                    nested = True
                    break
                parent = posixpath.dirname(parent)
            if not nested:
                should.add(p)
        unlisted = sorted(should - listed)
        extra = sorted(p for p in listed - should if p.startswith(d + "/"))
        rel = d[len(prefix) + 1:]
        status = "OK" if not (bad or missing or unlisted) else "CHECK"
        if status != "OK":
            all_ok = False
        print(f"{status} {rel}: entries {len(entries)}, verified {ok}, failed {bad}, "
              f"missing {missing}, outside-folder {outside}, tracked-unlisted {len(unlisted)}, "
              f"listed-but-nested {len(extra)}")
        for kind, p in bad_list:
            print(f"    {kind} {p[len(prefix)+1:]}")
        for p in unlisted:
            print(f"    UNLISTED {p[len(prefix)+1:]}")
        for p in extra:
            print(f"    LISTED-IN-NESTED {p[len(prefix)+1:]}")
    uncovered = sorted(c for c in changed if c not in listed_anywhere
                       and posixpath.basename(c) != "SHA256SUMS")
    # also consider the nested-SUMS delegation: a file listed by a nested SUMS
    print(f"changed paths listed by no relevant SHA256SUMS: {len(uncovered)}")
    for c in uncovered:
        print(f"    {c[len(prefix)+1:] if c.startswith(prefix) else c}")
    print("ALL RELEVANT SUMS VERIFY AND ARE COMPLETE" if all_ok else "SEE CHECK LINES")


if __name__ == "__main__":
    main()
