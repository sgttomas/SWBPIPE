#!/bin/bash
set -u
WT=<wt>
FKD=$WT/rv20/head/projects/chirality-piping/core/solver/frame_kernel
OUT=$WT/rv20/records
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2 CARGO_TARGET_DIR=$WT/rv20-target/head
cd $FKD
{
date; uptime; rustc --version; cargo --version
echo "== non-test lib build"
cargo build --offline --locked -j 4 --lib 2>&1
echo "== full suite"
cargo test --offline --locked -j 4 --no-fail-fast 2>&1
echo "exit: $?"
date; uptime
} > $OUT/fk_full.raw.log 2>&1
{
date; uptime
echo "== kf1_ tests with output"
cargo test --offline --locked -j 4 --lib kf1_ -- --nocapture 2>&1
echo "exit: $?"
date; uptime
} > $OUT/kf1_tests.raw.log 2>&1
