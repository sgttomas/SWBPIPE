"""RV24: lexer scan for the dense witness's reach (reviewer's own; standard library only).
Usage: rv24_reach_scan.py <tree root> <label>
Strips Rust comments (line, nested block, doc), string/char/raw-string/byte-string literals,
then lists every identifier token `negative_pair_witness` (whole token; not the sparse one or the
counted helper), with the enclosing `fn` and whether the site is inside a `#[cfg(test)]` module or
a `tests/` file. Also prints the declared signature of `pub fn negative_pair_witness`."""
import os, re, sys

def strip(src):
    out, i, n = [], 0, len(src)
    while i < n:
        c = src[i]
        if src.startswith("//", i):
            j = src.find("\n", i)
            j = n if j < 0 else j
            out.append(" " * (j - i)); i = j
        elif src.startswith("/*", i):
            depth, j = 1, i + 2
            while j < n and depth:
                if src.startswith("/*", j): depth += 1; j += 2
                elif src.startswith("*/", j): depth -= 1; j += 2
                else: j += 1
            out.append(re.sub(r"[^\n]", " ", src[i:j])); i = j
        elif re.match(r'b?r(#*)"', src[i:i+260]) and (i == 0 or not (src[i-1].isalnum() or src[i-1] == "_")):
            m = re.match(r'b?r(#*)"', src[i:])
            hashes = m.group(1)
            end = src.find('"' + hashes, i + m.end())
            j = n if end < 0 else end + 1 + len(hashes)
            out.append(re.sub(r"[^\n]", " ", src[i:j])); i = j
        elif c == '"' or (c == "b" and src.startswith('b"', i) and (i == 0 or not (src[i-1].isalnum() or src[i-1] == "_"))):
            j = i + (2 if c == "b" else 1)
            while j < n and src[j] != '"':
                j += 2 if src[j] == "\\" else 1
            j += 1
            out.append(re.sub(r"[^\n]", " ", src[i:j])); i = j
        elif c == "'":
            m = re.match(r"'(\\.|\\u\{[0-9a-fA-F]+\}|\\x[0-9a-fA-F]{2}|[^\\'])'", src[i:])
            if m:
                out.append(" " * m.end()); i += m.end()
            else:
                out.append(c); i += 1  # a lifetime
        else:
            out.append(c); i += 1
    return "".join(out)

root, label = sys.argv[1], sys.argv[2]
hits = []
sigs = []
for dp, dn, fn in os.walk(root):
    dn[:] = [d for d in dn if d not in ("target", "node_modules", ".git")]
    for f in fn:
        if not f.endswith(".rs"):
            continue
        p = os.path.join(dp, f)
        src = open(p, encoding="utf-8", errors="replace").read()
        code = strip(src)
        for m in re.finditer(r"(?<![A-Za-z0-9_])negative_pair_witness(?![A-Za-z0-9_])", code):
            line = code.count("\n", 0, m.start()) + 1
            before = code[:m.start()]
            fns = re.findall(r"\bfn\s+([A-Za-z0-9_]+)", before)
            enclosing = fns[-1] if fns else "-"
            test = ("/tests/" in p) or ("#[cfg(test)]" in before and re.search(r"#\[cfg\(test\)\]\s*mod\s", before) is not None) or f.endswith("_tests.rs") or f == "tests.rs"
            text = src.splitlines()[line - 1].strip()
            hits.append((os.path.relpath(p, root), line, enclosing, "test" if test else "product", text))
        for m in re.finditer(r"pub\s+fn\s+negative_pair_witness\s*\(([^)]*)\)\s*->\s*([^{]+)\{", code):
            sigs.append((os.path.relpath(p, root), re.sub(r"\s+", " ", m.group(0))))
print(f"== {label}: token `negative_pair_witness` (comments and literals stripped)")
for h in sorted(hits):
    print("  %s:%d  fn %s  [%s]  %s" % h)
print(f"== {label}: declared signatures")
for s in sigs:
    print("  %s: %s" % s)
