#!/usr/bin/env python3
"""RV14: permute input list order (sub-bodies, nodes within them, ties, tie ends,
grounds) of every case; node indices and coordinates unchanged."""
import random, sys
sys.path.insert(0, '.')
from rv14_oracle import parse_case, format_case
rng = random.Random(1404)
out = open(sys.argv[2], 'w')
for line in open(sys.argv[1]):
    if not line.strip() or line.startswith('#'):
        continue
    c = parse_case(line)
    subs = [list(b) for b in c['subs']]
    for b in subs:
        rng.shuffle(b)
    rng.shuffle(subs)
    ties = [(b, a) if rng.random() < 0.5 else (a, b) for a, b in c['ties']]
    rng.shuffle(ties)
    grounds = list(c['grounds'])
    rng.shuffle(grounds)
    out.write(format_case(c['name'], subs, ties, grounds, c['coords']) + '\n')
