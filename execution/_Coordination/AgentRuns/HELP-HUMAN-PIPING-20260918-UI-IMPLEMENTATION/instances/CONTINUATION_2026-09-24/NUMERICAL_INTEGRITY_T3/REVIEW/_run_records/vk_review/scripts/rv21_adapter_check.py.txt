#!/usr/bin/env python3
"""RV21: an independent check of V-K's committed case files against R1.

Written by RV21 from R1's README section 2 and the committed JSON layout, not
from gen_vk_cases.py's code. Standard library only; run with python3 -B.

For every committed case line it checks:
  - rows, scales and controls against references.json, string for string;
  - the committed kernel model against `references.py --model <id>` (a
    subprocess for every non-RF-LARGE case and for a sample of RF-LARGE; the
    module's model_json(full=True) in process for the rest, checked equal to
    the subprocess output on the sample), with every input rounded once by
    RV21's own integer rounding;
  - for the 12 large models, sha256 of the adapted JSON against
    large_models.sha256 (RV21's own adapter output, serialized as the file
    format states).
"""
import sys
sys.dont_write_bytecode = True
import hashlib
import importlib.util
import json
import math
import struct
import subprocess
from fractions import Fraction
from pathlib import Path

ROOT = Path(sys.argv[1])  # the archive's projects/chirality-piping
VR = ROOT / 'validation/benchmarks/numerical_robustness'
REF = ROOT / ('execution/_Coordination/AgentRuns/HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION/'
              'instances/CONTINUATION_2026-09-24/NUMERICAL_INTEGRITY_T3/REFERENCES')
SAMPLE_LARGE_SUBPROCESS = {'RF-LARGE-CHAIN-n00010-AX', 'RF-LARGE-CONT-n01000-ROT'}
DOFN = ['UX', 'UY', 'UZ', 'RX', 'RY', 'RZ']


def rational(text):
    text = str(text).strip()
    if '*2^' in text:
        a, k = text.split('*2^')
        k = int(k)
        return rational(a) * (Fraction(2) ** k)
    if '/' in text:
        p, q = text.split('/')
        return Fraction(int(p), int(q))
    return Fraction(text)  # Fraction parses decimal and scientific strings exactly


def round_binary64(q):
    """Round-to-nearest-even of an exact rational, by integer arithmetic."""
    q = Fraction(q)
    if q == 0:
        return 0.0
    sign = -1.0 if q < 0 else 1.0
    q = abs(q)
    n, d = q.numerator, q.denominator
    e = n.bit_length() - d.bit_length()
    # adjust e so that 2^e <= q < 2^(e+1)
    if (n << max(0, -e)) < (d << max(0, e)):
        e -= 1
    # quantum: 2^(e-52) for normals, 2^-1074 for subnormals
    qe = max(e - 52, -1074)
    # m = q / 2^qe, rounded to nearest even
    if qe >= 0:
        num, den = n, d << qe
    else:
        num, den = n << (-qe), d
    m, r = divmod(num, den)
    if 2 * r > den or (2 * r == den and (m & 1)):
        m += 1
    try:
        value = math.ldexp(m, qe)
    except OverflowError:
        value = float('inf')
    return sign * value


def self_test_rounding():
    import random
    rng = random.Random(21)
    for _ in range(20000):
        n = rng.getrandbits(rng.randint(1, 300)) or 1
        d = rng.getrandbits(rng.randint(1, 300)) or 1
        q = Fraction(n, d) * Fraction(2) ** rng.randint(-1200, 900)
        if rng.random() < 0.5:
            q = -q
        want = q.numerator / q.denominator if abs(q) < Fraction(2) ** 1023 else None
        if want is not None:
            assert round_binary64(q) == want, q
    # ties and subnormal boundaries
    for q in (Fraction(1, 2 ** 1075), Fraction(3, 2 ** 1076), Fraction(2 ** 53 + 1), Fraction(2 ** 53 + 3)):
        assert round_binary64(q) == q.numerator / q.denominator, q


def bits(x):
    return struct.pack('>d', x).hex()


def exact_section(sec, basis, pi_q):
    if basis == 'represented':
        r = lambda t: Fraction(round_binary64(rational(t)))
    else:
        r = rational
    E, OD, ID = r(sec['E']), r(sec['OD']), r(sec['ID'])
    if 'G' in sec:
        G = r(sec['G'])
    else:
        G = E / (2 * (1 + r(sec['nu'])))
    A = pi_q * (OD ** 2 - ID ** 2) / 4
    I = pi_q * (OD ** 4 - ID ** 4) / 64
    return [round_binary64(v) for v in (E, G, A, I, I, 2 * I)]


def vk_y_reference(xi, xj):
    chord = [abs(Fraction(b) - Fraction(a)) for a, b in zip(xi, xj)]
    k = 0
    for c in (1, 2):
        if chord[c] < chord[k]:
            k = c
    y = [0.0, 0.0, 0.0]
    y[k] = 1.0
    return y


def rv21_adapt(mj, basis, pi_q):
    names = list(mj['nodes_m'].keys())
    pos = {n: i for i, n in enumerate(names)}
    xyz = [[round_binary64(rational(c)) for c in mj['nodes_m'][n]] for n in names]
    members = []
    for idx, (name, a, b, sid) in enumerate(mj['members']):
        props = exact_section(mj['sections'][sid], basis, pi_q)
        y = vk_y_reference(xyz[pos[a]], xyz[pos[b]])
        members.append([name, idx + 1, pos[a], pos[b]] + [bits(v) for v in props] + [bits(v) for v in y])
    springs, omitted, constraints = [], [], []
    ident = 0
    for node, sup in mj.get('supports', {}).items():
        for dof in sup.get('rigid', []):
            constraints.append([pos[node], DOFN.index(dof)])
        for k_pos, s in enumerate(sup.get('springs', [])):
            ident += 1
            key = 'S.%s.%d' % (node, k_pos)
            k = rational(s['k'])
            if k == 0:
                omitted.append(key)
                continue
            d = [rational(c) for c in s['direction']]
            nz = [c for c in range(3) if d[c] != 0]
            base = 0 if s['kind'] == 'translation' else 3
            axis = base + nz[0] if len(nz) == 1 else None
            springs.append([key, ident, pos[node], 't' if base == 0 else 'r', axis,
                            [bits(round_binary64(c)) for c in d], bits(round_binary64(k))])
    loads = []
    count = 0
    for node, ld in mj.get('loads', {}).items():
        for part, off in (('F', 0), ('M', 3)):
            comps = ld.get(part)
            if comps is None:
                continue
            for c in range(3):
                q = rational(comps[c])
                if q != 0:
                    loads.append([pos[node], off + c, bits(round_binary64(q)), 'l%d' % count])
                    count += 1
    for k, (node, dof, v) in enumerate(mj.get('load_contributions_in_authored_order', [])):
        loads.append([pos[node], DOFN.index(dof), bits(round_binary64(rational(v))), 'c%d' % k])
    stations = [[m[1], m[1], bits(0.5)] for m in members]
    return {'nodes': [[n] + [bits(c) for c in p] for n, p in zip(names, xyz)], 'members': members,
            'springs': springs, 'omitted_springs': omitted, 'constraints': constraints,
            'loads': loads, 'stations': stations}


def y_ref_valid(model):
    bad = []
    for m in model['members']:
        xi = [struct.unpack('>d', bytes.fromhex(h))[0] for h in model['nodes'][m[2]][1:]]
        xj = [struct.unpack('>d', bytes.fromhex(h))[0] for h in model['nodes'][m[3]][1:]]
        y = [struct.unpack('>d', bytes.fromhex(h))[0] for h in m[10:13]]
        if sorted(y) != [0.0, 0.0, 1.0]:
            bad.append((m[0], 'not a unit axis'))
            continue
        c = [Fraction(b) - Fraction(a) for a, b in zip(xi, xj)]
        cross = [c[1] * y[2] - c[2] * y[1], c[2] * y[0] - c[0] * y[2], c[0] * y[1] - c[1] * y[0]]
        if all(v == 0 for v in cross):
            bad.append((m[0], 'parallel to the chord'))
    return bad


def dumps(obj):
    return json.dumps(obj, sort_keys=True, separators=(',', ':'), ensure_ascii=True)


def main():
    self_test_rounding()
    ref = json.loads((REF / 'references.json').read_text())
    spec = importlib.util.spec_from_file_location('r1ref', REF / 'references.py')
    r1 = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(r1)
    pi_q = r1.PI_Q
    large_sha = {}
    for line in (VR / 'cases/large_models.sha256').read_text().splitlines():
        msha, ksha, cid = line.split()
        large_sha[cid] = msha
    problems = []
    checked = {'cases': 0, 'rows': 0, 'controls': 0, 'models_subprocess': 0, 'models_inprocess': 0,
               'large_sha': 0}
    large_ids = []
    lines = []
    for f in sorted((VR / 'cases').glob('rf_*.jsonl')):
        for text in f.read_text().splitlines():
            if text:
                lines.append(json.loads(text))
    ids = {c['id'] for c in lines}
    expect_ids = {k for k in ref['cases'] if k not in ('RF-CANCEL-UDL-W1e5', 'RF-CANCEL-UDL-W1e8',
                                                          'RF-CANCEL-UDL-W1e80')}
    if ids != expect_ids:
        problems.append('case ids differ: missing %s extra %s' % (sorted(expect_ids - ids), sorted(ids - expect_ids)))
    built_families = set()
    for c in lines:
        cid = c['id']
        rc = ref['cases'][cid]
        checked['cases'] += 1
        basis = rc.get('basis', 'intended')
        if c['basis'] != basis:
            problems.append('%s basis %s vs %s' % (cid, c['basis'], basis))
        if c['family'] != rc['family']:
            problems.append('%s family' % cid)
        if c['refuse'] != ('expected_outcome' in rc):
            problems.append('%s refuse' % cid)
        if c['needs_directional_spring'] != bool(rc['flags'].get('needs_directional_spring')):
            problems.append('%s needs_directional_spring' % cid)
        # rows
        src_rows = rc.get('expected_represented') if basis == 'represented' else rc.get('expected', [])
        src_rows = src_rows or []
        if len(src_rows) != len(c['rows']):
            problems.append('%s row count %d vs %d' % (cid, len(c['rows']), len(src_rows)))
        for mine, theirs in zip(c['rows'], src_rows):
            checked['rows'] += 1
            want_scale = theirs[3] if rc['family'] == 'RF-CANCEL' else None
            if mine[0] != theirs[0] or mine[1] != theirs[1] or mine[2] != theirs[2] or mine[3] != want_scale:
                problems.append('%s row %s: %s vs %s' % (cid, theirs[0], mine, theirs[:4]))
        if rc['family'] == 'RF-CANCEL':
            cols = rc.get('expected_columns', [])
            if not cols or 'recommended' not in cols[3]:
                problems.append('%s: RF-CANCEL column 3 is %r' % (cid, cols[3] if cols else None))
        # scales
        want_scales = {k: v['value'] for k, v in rc.get('scales', {}).items()}
        if c['scales'] != want_scales:
            problems.append('%s scales differ' % cid)
        # every row class has a scale
        for r in c['rows']:
            if r[3] is None and r[2] not in c['scales']:
                problems.append('%s row %s class %s has no scale' % (cid, r[0], r[2]))
        # controls
        ncs = rc['negative_controls']
        if len(ncs) != len(c['controls']):
            problems.append('%s control count' % cid)
        for mine, nc in zip(c['controls'], ncs):
            checked['controls'] += 1
            if mine[0] != nc['id'] or mine[1] != bool(nc['discriminates']):
                problems.append('%s control %s id/discriminates' % (cid, nc['id']))
            if 'values' in nc:
                if mine[2] != 'value' or mine[3] != nc['values']:
                    problems.append('%s control %s values' % (cid, nc['id']))
            elif mine[2] != 'outcome':
                problems.append('%s control %s kind' % (cid, nc['id']))
        # the model
        is_large = c['model'] is None
        fam = rc['family']
        if fam != 'RF-LARGE' or cid in SAMPLE_LARGE_SUBPROCESS:
            out = subprocess.run([sys.executable, '-B', str(REF / 'references.py'), '--model', cid],
                                 capture_output=True, text=True, check=True, cwd=str(REF))
            mj = json.loads(out.stdout)
            checked['models_subprocess'] += 1
            if fam == 'RF-LARGE':
                if 'RF-LARGE' not in built_families:
                    r1.build_large()
                    built_families.add('RF-LARGE')
                case = next(x for x in r1.CASES if x['id'] == cid)
                if json.loads(json.dumps(r1.model_json(case['defn'], full=True))) != mj:
                    problems.append('%s: in-process model_json differs from --model' % cid)
        else:
            if 'RF-LARGE' not in built_families:
                r1.build_large()
                built_families.add('RF-LARGE')
            case = next(x for x in r1.CASES if x['id'] == cid)
            mj = json.loads(json.dumps(r1.model_json(case['defn'], full=True)))
            checked['models_inprocess'] += 1
        if 'generator' not in rc['model'] and rc['model'] != mj:
            problems.append('%s: references.json model differs from --model' % cid)
        mine = rv21_adapt(mj, basis, pi_q)
        if is_large:
            got = hashlib.sha256((dumps(mine) + '\n').encode()).hexdigest()
            checked['large_sha'] += 1
            if got != large_sha.get(cid) or got != c['model_sha256']:
                problems.append('%s: large model sha %s vs %s' % (cid, got, large_sha.get(cid)))
            large_ids.append(cid)
            continue
        committed = c['model']
        for field in ('nodes', 'constraints', 'loads', 'stations', 'omitted_springs'):
            if committed[field] != mine[field]:
                problems.append('%s: %s differ' % (cid, field))
        if len(committed['members']) != len(mine['members']):
            problems.append('%s: member count' % cid)
        for a, b in zip(committed['members'], mine['members']):
            if a[:10] != b[:10]:
                problems.append('%s: member %s %s vs %s' % (cid, a[0], a[:10], b[:10]))
        for bad in y_ref_valid(committed):
            problems.append('%s: y_reference %s' % (cid, bad))
        if [m[10:13] for m in committed['members']] != [m[10:13] for m in mine['members']]:
            problems.append('%s: y_reference differs from the stated rule' % cid)
        if committed['springs'] != mine['springs']:
            problems.append('%s: springs %s vs %s' % (cid, committed['springs'], mine['springs']))
    print(json.dumps(checked))
    print('large models checked:', len(large_ids))
    print('problems: %d' % len(problems))
    for p in problems[:200]:
        print('PROBLEM', p)
    return 1 if problems else 0


if __name__ == '__main__':
    sys.exit(main())
