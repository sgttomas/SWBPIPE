#!/usr/bin/env python3
"""RV17 delta: copy the delta records into f1b_review/delta/ (machine paths replaced, trailing whitespace stripped,
one final newline), check that the review's original files still match the committed SHA256SUMS, and regenerate
f1b_review/SHA256SUMS over the whole folder."""
import hashlib, os, re, shutil, subprocess
WT = '<wt>'
SRC = WT + '/scratch/rv17'
BASE = WT + ('/numerics/projects/chirality-piping/execution/_Coordination/AgentRuns/'
             'HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION/instances/CONTINUATION_2026-09-24/'
             'NUMERICAL_INTEGRITY_T3/REVIEW/_run_records/f1b_review')
DST = BASE + '/delta'
SUBS = [('<scratch>', '<scratch>'),
        (WT, '<wt>'), ('<repo>', '<repo>'),
        ('<home>', '<home>'), ('<tmp>', '<tmp>')]
FILES = {
    'README.txt': 'delta/README.txt',
    'records_delta.txt': 'delta/records_delta.txt',
    'collect_delta_records.sh.txt': 'delta/collect_delta_records.sh',
    'platform_check.txt': 'delta/platform_check.txt',
    'platform_check.sh.txt': 'delta/platform_check.sh',
    'assemble_delta.py.txt': 'delta/assemble_delta.py',
    'mutants/run_mutant3.sh.txt': 'runs/run_mutant3.sh',
    'mutants/queue4.sh.txt': 'runs/queue4.sh',
    'mutants/MUTANTS.txt': 'runs/delta/MUTANTS.txt',
}
for f in sorted(os.listdir(SRC + '/runs/delta/logs')):
    if f.endswith('.log'):
        FILES[f'mutants/logs/{f}'] = f'runs/delta/logs/{f}'
# The original record set must be unchanged.
old = {}
for line in open(BASE + '/SHA256SUMS'):
    h, p = line.rstrip('\n').split('  ', 1)
    if not p.startswith('./delta/'):
        old[p] = h
for p, h in old.items():
    assert hashlib.sha256(open(os.path.join(BASE, p[2:]), 'rb').read()).hexdigest() == h, p
if os.path.exists(DST):
    shutil.rmtree(DST)
for rec, src in FILES.items():
    text = open(os.path.join(SRC, src), 'rb').read().decode('utf-8', 'replace')
    for s, r in SUBS:
        text = text.replace(s, r)
    text = '\n'.join(l.rstrip(' \t') for l in text.split('\n')).rstrip('\n') + '\n'
    out = os.path.join(DST, rec)
    os.makedirs(os.path.dirname(out), exist_ok=True)
    open(out, 'w').write(text)
bad = [p for r, _, fs in os.walk(DST) for f in fs for p in [os.path.join(r, f)]
       if re.search(r'[/](Users|private|home|tmp)[/][A-Za-z0-9]|[/]var[/]folders[/][A-Za-z0-9]|claude-50[0-9]|Users-[a-z]', open(p).read())]
assert not bad, bad
lines = []
for root, _, files in os.walk(BASE):
    for f in files:
        if f == 'SHA256SUMS' and root == BASE:
            continue
        p = os.path.join(root, f)
        lines.append(f'{hashlib.sha256(open(p, "rb").read()).hexdigest()}  ./{os.path.relpath(p, BASE)}')
open(BASE + '/SHA256SUMS', 'w').write('\n'.join(sorted(lines, key=lambda l: l[66:])) + '\n')
print('original files unchanged:', len(old), '; delta files:', len(FILES), '; SHA256SUMS lines:', len(lines))
