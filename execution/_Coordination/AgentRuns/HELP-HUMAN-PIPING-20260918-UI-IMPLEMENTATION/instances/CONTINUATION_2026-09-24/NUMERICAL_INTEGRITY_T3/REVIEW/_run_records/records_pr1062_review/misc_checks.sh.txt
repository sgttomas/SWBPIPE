#!/bin/bash
# RV25: one-off read-only checks for the records PR #1062 review.
# Usage, from the repository root of the numerics working tree at the head:
#   bash misc_checks.sh.txt <wt>
# <wt> is the T3 worktrees root; it is used only to read uncommitted scratch
# evidence (hashes), and is never printed. Only `git show/diff/log/rev-list/
# patch-id`, `gh pr/run view` and `shasum` are run. No writes.
set -u
WT=$1
T3=projects/chirality-piping/execution/_Coordination/AgentRuns/HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION/instances/CONTINUATION_2026-09-24/NUMERICAL_INTEGRITY_T3
V1=$T3/ROOT_RULINGS_V1.md
WG=projects/chirality-piping/execution/_Coordination/WorkGraphs/HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION/WORK_GRAPH.md
BASE=7ad3a9adf; HEAD=b0e357985

echo "== 1. The head's merge of main (b0e357985)"
git rev-list --parents -n1 $HEAD | awk '{print "parents", substr($2,1,10), substr($3,1,10)}'
echo "merge base of the parents: $(git merge-base $HEAD^1 $HEAD^2 | cut -c1-10)"
SHARED=$T3/IMPLEMENTATION/K6/PLAN_CHECKPOINT0.md
echo "path changed identically on both sides: IMPLEMENTATION/K6/PLAN_CHECKPOINT0.md: numerics $(git show $HEAD^1:$SHARED | shasum -a 256 | cut -c1-12), main $(git show $HEAD^2:$SHARED | shasum -a 256 | cut -c1-12)"
echo "adds main's delta (less that path): $(git diff $HEAD^1 $HEAD -- . ":(exclude)$SHARED" | shasum -a 256 | cut -c1-12) vs $(git diff $(git merge-base $HEAD^1 $HEAD^2) $HEAD^2 -- . ":(exclude)$SHARED" | shasum -a 256 | cut -c1-12)"
echo "keeps numerics' delta (less that path): $(git diff $HEAD^2 $HEAD -- . ":(exclude)$SHARED" | shasum -a 256 | cut -c1-12) vs $(git diff $(git merge-base $HEAD^1 $HEAD^2) $HEAD^1 -- . ":(exclude)$SHARED" | shasum -a 256 | cut -c1-12)"
echo "remerge-diff lines: $(git show --remerge-diff --format= $HEAD | wc -l | tr -d ' ')"

echo "== 2. KF3's merge-order slip"
gh pr view 1061 --repo sgttomas/chirality --json mergeCommit,mergedAt,changedFiles --jq '"#1061 merge \(.mergeCommit.oid[0:10]) at \(.mergedAt), \(.changedFiles) files"'
echo "#1061 paths by top folder:"; git diff --name-only 78f55f927 45ffd91d1 | awk -F/ '{print "   "$1"/"$2}' | sort | uniq -c
echo "KF3 merge dd61120ff parents: $(git rev-list --parents -n1 dd61120ff | cut -d' ' -f2- | awk '{print substr($1,1,10), substr($2,1,10)}')"
echo "45ffd91d1 ancestor of the KF3 head aa83f6796: $(git merge-base --is-ancestor 45ffd91d1 aa83f6796 && echo yes || echo no)"
echo "git diff aa83f6796 dd61120ff -- projects/chirality-piping tools .github: $(git diff aa83f6796 dd61120ff -- projects/chirality-piping tools .github | wc -l | tr -d ' ') lines"
echo "all paths aa83f6796..dd61120ff by top folder:"; git diff --name-only aa83f6796 dd61120ff | awk -F/ '{print "   "$1"/"$2}' | sort | uniq -c
echo "remerge-diff lines of dd61120ff: $(git show --remerge-diff --format= dd61120ff | wc -l | tr -d ' ')"

echo "== 3. Slice-branch merges named in the records (parents)"
for c in c0473301e e114b23c1 b86081221 485320e95 597c81ba4 522167ac6 ba88db495 8f8023a20 1f354c20b 3e90176c6 9aeed9c22 c4879c496 babcf5e65 28517eaaa; do
  echo "   $c: $(git rev-list --parents -n1 $c | cut -d' ' -f2- | awk '{print substr($1,1,9), substr($2,1,9)}')"
done
echo "remerge-diff files of 597c81ba4 (K6b), c0473301e (KF3); line counts of 522167ac6 (KF2) and e114b23c1 (KF3):"
git show --remerge-diff --format= 597c81ba4 | grep '^diff --git' | sed 's|.*/frame_kernel/|   597c81ba4 FK/|; s| b/.*||'
git show --remerge-diff --format= c0473301e | grep '^diff --git' | sed 's|.*/frame_kernel/|   c0473301e FK/|; s| b/.*||'
echo "   522167ac6: $(git show --remerge-diff --format= 522167ac6 | wc -l | tr -d ' '); e114b23c1: $(git show --remerge-diff --format= e114b23c1 | wc -l | tr -d ' ')"
echo "K6b head FK equals main f8400d290's: $(git diff --quiet 597c81ba4 f8400d290 -- projects/chirality-piping/core/solver/frame_kernel && echo yes || echo no)"

echo "== 4. Spawn bases (first-parent first commit's parent)"
for pair in K6b:ab02ee3a6..597c81ba4 VK:ab02ee3a6..5f0d39426 KF1:8cca91701..66adfede4 KF3:0f5d8c7b4..aa83f6796 KF2:78f55f927..522167ac6; do
  n=${pair%%:*}; r=${pair#*:}; first=$(git rev-list --reverse --first-parent $r | head -1)
  echo "   $n: $(git log -1 --format='%h %s' $first | cut -c1-60) <- parent $(git rev-parse --short=9 $first^)"
done
echo "#1055 paths: $(gh pr view 1055 --repo sgttomas/chirality --json files --jq '[.files[].path|split("/")[0:2]|join("/")]|unique|join(",")')"

echo "== 5. A0 export (V1 'K6b A0 accepted')"
for c in bb89e4f8f 3018343c2; do echo "   $c patch-id $(git show $c | git patch-id --stable | cut -c1-12)"; done
echo "   bb89e4f8f: removed pub(crate) lines $(git show bb89e4f8f | grep -c '^-.*pub(crate)'), removed dead_code markers $(git show bb89e4f8f | grep -c '^-.*allow(dead_code)')"

echo "== 6. CI runs cited only in the rulings"
for r in 36584733672 36584733559 36584733590 36584733821 36640221444 36664104717 36669470111 36621651732; do
  gh run view $r --repo sgttomas/chirality --json workflowName,event,headSha,conclusion --jq "\"   $r \(.workflowName) | \(.event) | \(.headSha[0:10]) | \(.conclusion)\""
done
for r in 36640221444 36664104717 36669470111; do
  echo "   $r target_base $(gh run view $r --repo sgttomas/chirality --log 2>/dev/null | grep -oE '"target_base": "[0-9a-f]{40}"' | head -1 | grep -oE '[0-9a-f]{40}' | cut -c1-10)"
done
echo "PR #1062 runs on the head:"
for r in $(gh pr view 1062 --repo sgttomas/chirality --json statusCheckRollup --jq '[.statusCheckRollup[] | .detailsUrl | capture("runs/(?<r>[0-9]+)").r] | unique | .[]'); do
  gh run view $r --repo sgttomas/chirality --json workflowName,event,headSha,conclusion --jq "\"   $r \(.workflowName) | \(.event) | \(.headSha[0:10]) | \(.conclusion)\""
done
gh pr view 1062 --repo sgttomas/chirality --json statusCheckRollup --jq '"   checks: " + ([.statusCheckRollup[] | .conclusion] | group_by(.) | map("\(.[0])=\(length)") | join(" "))'
echo "numerical cargo job minutes (K4 pr/dispatch, K6 dispatch, KF3 pr/dispatch, KF2 pr/dispatch):"
for r in 36593109243 36593106169 36548351414 36669372845 36669370536 36673665663 36673660523; do
  gh api repos/sgttomas/chirality/actions/runs/$r/jobs --jq '.jobs[] | select(.name=="Numerical cargo suite") | "\(.started_at) \(.completed_at)"' \
  | python3 -c "import sys,datetime as d
for l in sys.stdin:
    s,e=l.split(); f=lambda x:d.datetime.fromisoformat(x.replace('Z','+00:00')); print('   $r', round((f(e)-f(s)).total_seconds()/60,2))"
done

echo "== 7. K5's DEC-025 baseline 33e33c723 against main b37331092 (piping paths outside _Coordination)"
echo "   $(git diff --name-only 33e33c723 b37331092 -- projects/chirality-piping | grep -vc '^projects/chirality-piping/execution/_Coordination/') paths; 33e33c723 == e7d930d49 piping: $(git diff --quiet 33e33c723 e7d930d49 -- projects/chirality-piping && echo yes || echo no)"

echo "== 8. In-place edits to ROOT_RULINGS_V1.md inside the PR's own history (commits that remove a line)"
for c in $(git rev-list --reverse --no-merges $BASE..$HEAD -- $V1); do
  n=$(git diff $c^ $c -- $V1 | grep -c '^-[^-]')
  if [ "$n" != "0" ]; then echo "   $(git log -1 --format='%h %cI %s' $c | cut -c1-110): removed $n"; git diff -U0 $c^ $c -- $V1 | grep '^-[^-]' | cut -c1-240 | sed 's/^/      /'; fi
done
echo "   the claim 'relied on it at the margin' was introduced by: $(git log -S'relied on it at the margin' --format='%h %cI' -- $V1 | tr '\n' ' ')"

echo "== 9. Uncommitted evidence cited by the rulings (sha256 prefixes; paths relative to <wt>)"
echo "   scratch/i13/gate2/SHA256SUMS lines: $(wc -l < $WT/scratch/i13/gate2/SHA256SUMS | tr -d ' ')"
echo "   scratch/i13/gate2/part1/runs.jsonl: $(shasum -a 256 $WT/scratch/i13/gate2/part1/runs.jsonl | cut -c1-12)"
echo "   scratch/gate_base_e7d930d49_full/*/runs.jsonl: $(find $WT/scratch/gate_base_e7d930d49_full -maxdepth 2 -name runs.jsonl -exec shasum -a 256 {} \; | cut -c1-12)"
echo "   scratch/i20/b/gate/part1_base/runs.jsonl: $(shasum -a 256 $WT/scratch/i20/b/gate/part1_base/runs.jsonl | cut -c1-12)"
echo "   scratch/i20/b/gate/part1_cand/runs.jsonl: $(shasum -a 256 $WT/scratch/i20/b/gate/part1_cand/runs.jsonl | cut -c1-12)"
for s in KF2:sweep_kf2 KF3:sweep_kf3 K4:sweep_k4 VK:sweep_vk; do
  n=${s%%:*}; d=$WT/scratch/${s#*:}
  echo "   $n dec025: original JSON sha256 (record) $(grep -oE '[0-9a-f]{64}' $T3/IMPLEMENTATION/${n}_MERGE/dec025/sweep_json_original_sha256.txt | head -1 | cut -c1-12), scratch $(shasum -a 256 $d/SWEEP_*.json | cut -c1-12); suites.log identical: $(cmp -s $d/suites.log $T3/IMPLEMENTATION/${n}_MERGE/dec025/suites.log && echo yes || echo no)"
done
echo "   scratch/sweep_kf2/meta.txt head: $(head -1 $WT/scratch/sweep_kf2/meta.txt | grep -oE '[0-9a-f]{40}' | cut -c1-10)"
