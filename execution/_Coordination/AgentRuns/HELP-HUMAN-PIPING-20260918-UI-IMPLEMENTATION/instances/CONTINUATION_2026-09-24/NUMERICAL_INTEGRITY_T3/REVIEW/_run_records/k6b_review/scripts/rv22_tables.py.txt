"""RV22: recompute RETURN §6.1 (and parts of §6.2, §6.3, §7.2, §7.4) from b3's raw JSONL, independently
of k6b_analysis.py and d_tables.py. Usage: rv22_tables.py <b3 records dir> <RETURN.md>"""
import json, os, re, sys, statistics
rec_dir, ret = sys.argv[1], sys.argv[2]
MiB = 1048576.0
recs = [json.loads(l) for l in open(os.path.join(rec_dir, 'records.jsonl'))]
base_fp = {}
for t in ('W1-T1', 'W1-T2', 'W1-T3', 'W1-T4'):
    b = json.load(open(os.path.join(rec_dir, 'baseline_%s.record.json' % t)))
    base_fp[t] = b['rss']['time_peak_footprint_bytes']

def lines(run_id):
    out = {}
    for l in open(os.path.join(rec_dir, run_id + '.jsonl')):
        d = json.loads(l)
        out.setdefault(d['kind'], []).append(d)
    return out

rows = {}
for r in recs:
    if r.get('mode') != 'w1a' or r.get('pass') != 1:
        continue
    L = lines(r['run_id'])
    out0 = [o for o in L['outcome'] if o['repeat'] == 0][0]
    att0 = [a for a in L['attempt'] if a['repeat'] == 0]
    counts = L['counts'][0]
    E = counts['estimate_adm_bytes_w1a']
    solves = [s for s in L['stage'] if s['stage'] == 'w1_solve']
    heap = max(s['heap_peak'] for s in solves)
    heap_move = max(s['heap_peak_move'] for s in solves)
    med = statistics.median([s['elapsed_ns'] for s in solves]) / 1e9
    fp = r['rss']['time_peak_footprint_bytes']
    rss = r['peak_rss_bytes']
    stop = sum(a['stop_rule_work'] for a in att0)
    shifts = [a['verification_shift_factorizations'] for a in att0 if a['verification_shift_factorizations'] is not None]
    cls = out0['class'] + (' %d' % out0['selected_precision'] if out0['class'] == 'Selected' else '(%s)' % out0['reason'])
    rho = max(fp - base_fp[r['tier']], r['summary']['heap_peak_move']) / E
    rows[r['model']] = dict(
        outcome=cls, attempts=', '.join('%d %s' % (a['precision'], a['outcome'].split('(')[0]) for a in att0),
        charged=out0['meter_charged'], stop=stop, stop_pct=100.0 * stop / out0['meter_charged'],
        shift=shifts[0] if shifts else '-', profile=att0[0]['storage_profile_entries'],
        pattern=att0[0]['storage_pattern_entries'], rows=out0.get('rows', '-'),
        heap=heap / MiB, fp=fp / MiB, rss=rss / MiB, med=med, ns=med * 1e9 / out0['meter_charged'],
        load=r['load_before'][0], E=E / MiB, heapE=r['repeats_heap_peak_move'] / E, rho=rho, heap_move=heap_move)

# Parse RETURN §6.1's two tables.
text = open(ret).read()
sec = text[text.index('### 6.1'):text.index('### 6.2')]
cols = None
bad = 0; checked = 0
for line in sec.splitlines():
    if not line.startswith('| '):
        continue
    cells = [c.strip() for c in line.strip('|').split('|')]
    if cells[0] == 'model':
        cols = cells; continue
    if set(cells[0]) <= set('-'):
        continue
    name = cells[0]
    key = ('RF-LARGE-' + name) if not name.startswith('DEC053') else None
    if key is None:
        key = [m for m in rows if m.startswith('DEC053:') and m.endswith(name.split()[1])]
        key = key[0] if key else name
    mine = rows[key]
    def num(s):
        return float(s.split(' ')[0].replace('(', '').replace('%)', ''))
    pairs = [('charged', cells[4], mine['charged'], 0), ('stop', cells[5].split(' ')[0], mine['stop'], 0),
             ('stop_pct', cells[5].split('(')[1].rstrip('%)'), mine['stop_pct'], 0.051),
             ('profile', cells[7], mine['profile'], 0), ('pattern', cells[8], mine['pattern'], 0),
             ('heap', cells[10], mine['heap'], 0.051), ('fp', cells[11], mine['fp'], 0.051), ('rss', cells[12], mine['rss'], 0.051),
             ('med', cells[13], mine['med'], 0.0006), ('ns', cells[14], mine['ns'], 0.0006), ('load', cells[15], mine['load'], 0.006),
             ('E', cells[16], mine['E'], 0.051), ('heapE', cells[17], mine['heapE'], 0.00006), ('rho', cells[18], mine['rho'], 0.00006)]
    if cells[9] != '-':
        pairs.append(('rows', cells[9], mine['rows'], 0))
    if cells[6] != '-':
        pairs.append(('shift', cells[6], mine['shift'], 0))
    for label, s, v, tol in pairs:
        checked += 1
        if abs(float(s) - float(v)) > tol:
            bad += 1
            print('MISMATCH', key, label, 'RETURN', s, 'raw', v)
    if cells[2] != mine['outcome']:
        print('OUTCOME', key, cells[2], mine['outcome'])
    if cells[3] != mine['attempts']:
        print('ATTEMPTS', key, cells[3], '|', mine['attempts'])
print('6.1: %d values checked against raw JSONL, %d mismatches, %d rows' % (checked, bad, len(rows)))
json.dump(rows, open(os.path.join(os.path.dirname(os.path.abspath(__file__)), 'rv22_rows.json'), 'w'), indent=1)
