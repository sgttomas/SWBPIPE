import csv, hashlib, os, sys
tsv, full = sys.argv[1:3]
rows = list(csv.DictReader(open(tsv), delimiter='\t'))
n = ok = missing = 0
bad = []
for i, r in enumerate(rows):
    if i % 7:  # sample every 7th run
        continue
    p = os.path.join(full, f"{r['case']}__{r['mode']}__{r['entry']}.json")
    if not os.path.exists(p):
        missing += 1
        continue
    if os.path.getsize(p) > 50_000_000:
        continue
    n += 1
    h = hashlib.sha256(open(p, 'rb').read()).hexdigest()
    if h == r['full_envelope_sha256']:
        ok += 1
    else:
        bad.append((r['case'], r['mode'], r['entry']))
print('sampled', n, 'match', ok, 'missing', missing, 'bad', bad)
