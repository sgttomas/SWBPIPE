#!/usr/bin/env python3
"""RV2: self-test of rv2_oracle against hardware binary64 (Python floats) at p = 53."""
import math, random, struct, sys
from fractions import Fraction as Fr
sys.path.insert(0, "<scratch>/work")
from rv2_oracle import oracle, from_fr_exact
R = random.Random(7)
def tok(x):
    if x == 0:
        return "Z-" if math.copysign(1, x) < 0 else "Z+"
    return from_fr_exact(Fr(x))
bad = n = 0
for _ in range(20000):
    x = math.ldexp(R.random() + 0.5, R.randint(-300, 300)) * R.choice((1, -1))
    y = math.ldexp(R.random() + 0.5, R.randint(-300, 300)) * R.choice((1, -1))
    if R.random() < 0.2:
        y = -x * (1 + R.choice((1, -1)) * 2 ** -52 * R.randint(0, 3))
    for op, v in (("add", x + y), ("sub", x - y), ("mul", x * y), ("div", x / y)):
        n += 1
        if oracle(op, 53, tok(x), tok(y)) != tok(v):
            bad += 1
    n += 1
    if oracle("sqrt", 53, tok(abs(x)), "-") != tok(math.sqrt(abs(x))):
        bad += 1
print("oracle self-test vs binary64:", n, "ops,", bad, "mismatches")
