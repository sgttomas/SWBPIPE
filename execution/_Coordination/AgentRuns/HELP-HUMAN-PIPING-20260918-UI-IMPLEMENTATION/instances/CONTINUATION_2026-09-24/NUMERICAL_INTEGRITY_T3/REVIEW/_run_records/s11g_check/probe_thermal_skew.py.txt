#!/usr/bin/env python3
"""V1 S11G_CHECK: the load-row guard on a pure-thermal straight skew run (standard library only).
A straight run anchored at both ends, subdivided at user-entered decimal stations, one thermal
axial_load N on every member (same section and material). Each member's direction cosines are the
product's binary64 x = delta * (1/|delta|) (FK frame orientation, as V1's probe_d5_check.product_frame);
the ledger pushes fl(N*x_a) at each end (PP add_thermal_equivalent_loads). Per D1's design: exact defect
E = sum(fl(N x) - N x) per row, intended net n = sum(N x) on held x, free-row S* from free-row intended
nets (DESIGN coupling), fires when |E| > 1e-9 max(|n|, S*)."""
import json, math, sys
from fractions import Fraction as Fr

def frame_x(xi, xj):
    d = [xj[k] - xi[k] for k in range(3)]
    s = 1.0 / math.sqrt(d[0] * d[0] + d[1] * d[1] + d[2] * d[2])
    return [d[0] * s, d[1] * s, d[2] * s]

def run(end, stations, N):
    pts = [[float(Fr(str(t)) * Fr(str(e))) for e in end] for t in stations]  # user-entered decimal coordinates
    pts = [[round(v, 6) for v in p_] for p_ in pts]
    rows = {}
    for a in range(len(pts) - 1):
        x = frame_x(pts[a], pts[a + 1])
        for node, sgn in ((a, -1.0), (a + 1, 1.0)):
            for k in range(3):
                v = sgn * (N * x[k])
                rows.setdefault((node, k), []).append((v, Fr(sgn) * Fr(N) * Fr(x[k])))
    free = {key: val for key, val in rows.items() if 0 < key[0] < len(pts) - 1}
    ext = [max(p_[k] for p_ in pts) - min(p_[k] for p_ in pts) for k in range(3)]
    Lb = math.sqrt(sum(e * e for e in ext))
    nets = {key: sum(i for _, i in val) for key, val in free.items()}
    F = max(abs(v) for v in nets.values())
    Sf = max(F, Fr(0))   # no free moment rows
    worst = 0.0; fires = []
    for key, val in free.items():
        E = sum(Fr(v) - i for v, i in val)
        thr = Fr(1, 10 ** 9) * max(abs(nets[key]), Sf)
        r = float(abs(E) / thr) if thr else (math.inf if E else 0.0)
        worst = max(worst, r)
        if abs(E) > thr:
            fires.append([key, float(E), float(nets[key]), float(thr)])
    return {"end": end, "stations": stations, "N": N, "free_rows": len(free), "S_star_free": float(Sf),
            "max_abs_free_intended_net": float(F), "worst_stat_over_threshold": worst, "fires": len(fires) > 0,
            "firing_rows": fires[:6]}

out = []
for end, st in (((12.0, 5.0, 0.0), (0, 0.13, 0.4, 0.55, 0.81, 1)),
                ((10.0, 3.7, 2.2), (0, 0.3, 0.55, 0.7, 1)),
                ((6.0, 6.0, 0.0), (0, 0.25, 0.5, 0.75, 1)),
                ((9.0, 0.0, 0.0), (0, 0.13, 0.4, 0.55, 0.81, 1))):
    out.append(run(end, st, 1.296e6))
json.dump(out, open(sys.argv[1], "w"), indent=1)
for o in out:
    print(json.dumps({k: o[k] for k in ("end", "stations", "S_star_free", "worst_stat_over_threshold", "fires")}))
