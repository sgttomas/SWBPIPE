"""RV5: I3's PP_UTM product outputs against RV5's own exact-input reference."""
import json, sys, glob, math
sys.path.insert(0, '.')
import m31b_attack as A
from ppc import pp_centre
from decimal import Decimal as D
d = sys.argv[1]
req = json.load(open(sys.argv[2]))["model"]
xi = [req["nodes"][0]["position"][k] for k in "xyz"]; xj = [req["nodes"][1]["position"][k] for k in "xyz"]
c = pp_centre(xi, xj, 0.3, [0.0, 1.0, 0.0])
m = {"name": "PP_UTM_2", "section": [2e11, 8e10, math.pi * (0.2**2 - 0.18**2) / 4, None, None], "nodes": [xi, xj],
     "members": [{"i": 0, "j": 1, "y_reference": None, "bend": (c, 1.0)}],
     "rigid": [0, 1, 2], "springs": [(3, 1e6), (4, 1e6), (5, 1e6)], "loads": [(9, 1.0), (10, 1.0), (11, 1.0)]}
# section as the product derives it from OD/wall (binary64, same expressions as the kd5 models' SEC)
m["section"] = A.SEC
print("centre", c, "admissible", A.b64_formula_chord(xi, xj, c)[1:])
n = 12; free = [k for k in range(n) if k not in m["rigid"]]
Ka, _ = A.assemble(m, "actual")
f = [D(0)] * n
for dof, v in m["loads"]: f[dof] += D(float(v))
ua = A.solve([[Ka[a][b] for b in free] for a in free], [f[a] for a in free])
uact = {a: ua[r] for r, a in enumerate(free)}
comp = {"x": 0, "y": 1, "z": 2}
for path in sorted(glob.glob(f"{d}/out_{sys.argv[3]}_*.json")):
    env = json.load(open(path))["run"]["envelope"]
    codes = [x["code"] for x in env["diagnostics"] if x["code"].startswith("NUMERICAL_INTEGRITY") or "CURVED" in x["code"]]
    kinds = sorted({r["kind"] for r in env["results"]})
    u = {}
    for r in env["results"]:
        k = r["kind"]
        if r["entity"] not in ("N0", "N1"): continue
        node = 0 if r["entity"] == "N0" else 1
        if k.startswith("global_nodal_displacement_"):
            u[6 * node + comp[k[-1]]] = float(D(repr(r["value"])) / 1000) if r["unit"] == "mm" else r["value"]
        if k.startswith("global_nodal_rotation_"):
            u[6 * node + 3 + comp[k[-1]]] = r["value"]
    s_tr = max(abs(u[a]) for a in free if a % 6 < 3); s_ro = max(abs(u[a]) for a in free if a % 6 >= 3)
    lb = math.sqrt(sum((xj[k] - xi[k]) ** 2 for k in range(3)))
    tr, ro = max(s_tr, lb * s_ro), max(s_ro, s_tr / lb)
    worst = max(((float(abs(D(u[a]) - uact[a]) / (D("1e-9") * D(max(abs(u[a]), ro if a % 6 >= 3 else tr))))), a) for a in free)
    print(path.split("/")[-1], codes, "curved rows:", [k for k in kinds if "curved" in k], "actual/criterion=%.4f at DOF %d" % worst)
