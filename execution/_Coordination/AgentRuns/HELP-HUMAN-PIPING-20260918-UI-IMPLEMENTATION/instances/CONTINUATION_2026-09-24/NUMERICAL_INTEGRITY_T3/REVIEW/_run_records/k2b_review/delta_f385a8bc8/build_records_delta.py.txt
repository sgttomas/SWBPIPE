#!/usr/bin/env python3
"""RV11 delta check: copy the delta evidence into REVIEW/_run_records/k2b_review/delta_f385a8bc8/, sanitized as
build_records.py.txt does (placeholders; lines over 600 characters cut with a sha256 marker; trailing whitespace and
trailing blank lines stripped; scripts stored as .txt). The first review's files are not rewritten: their bytes are
checked against the committed SHA256SUMS first, and SHA256SUMS is then regenerated over the whole folder.
Usage: build_records_delta.py <wt> <repo-root> <venv> <session-scratch> <home> <k2b_review-dir>"""
import hashlib, os, sys

wt, repo, venv, pad, home, top = sys.argv[1:7]
PLACES = sorted([(venv, "<VENV>"), (wt, "<wt>"), (repo, "{REPO_ROOT}"), (pad, "<session-scratch>"),
                 (pad.replace("/" + "private" + "/" + "tmp" + "/", "/" + "tmp" + "/"), "<session-scratch>"), (home, "<home>")],
                key=lambda p: -len(p[0]))
LIMIT = 600
R = os.path.join(wt, "scratch", "rv11")
L = os.path.join(R, "logs")
M = os.path.join(R, "mutd")
out = os.path.join(top, "delta_f385a8bc8")

def sha(path):
    with open(path, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()

sums = os.path.join(top, "SHA256SUMS")
old = [line.split("  ", 1) for line in open(sums).read().splitlines()]
for digest, rel in old:
    assert sha(os.path.join(top, rel)) == digest, rel
assert not os.path.exists(out), out

def put(src, rel, cut=False):
    with open(src, encoding="utf-8", errors="replace") as f:
        text = f.read()
    for path, name in PLACES:
        text = text.replace(path, name)
    lines = []
    for line in text.split("\n"):
        line = line.rstrip()
        if cut and len(line) > LIMIT:
            digest = hashlib.sha256(line.encode()).hexdigest()
            line = line[:LIMIT] + f" [... RV11-records cut: {len(line) - LIMIT} more characters; sha256 of the full line {digest}]"
        lines.append(line)
    while lines and lines[-1] == "":
        lines.pop()
    dst = os.path.join(out, rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    with open(dst, "w", encoding="utf-8") as f:
        f.write("\n".join(lines) + "\n")

put(os.path.join(R, "README_delta.txt"), "README.txt")
put(os.path.join(L, "delta_commits_check.txt"), "commits_check.txt")
put(os.path.join(L, "delta_index_check.txt"), "index_check.txt")
put(os.path.join(L, "delta_records_checks.txt"), "records_checks.txt")
put(os.path.join(L, "gen8_delta.txt"), "gen8.txt")
put(os.path.join(L, "head_f385a8bc8.log"), "head_tests/head_f385a8bc8.log", cut=True)
put(os.path.join(R, "lexscan_delta.py"), "callers/lexscan_delta.py.txt")
put(os.path.join(L, "lexscan_f385a8bc8.txt"), "callers/lexscan_f385a8bc8.txt")
put(os.path.join(R, "b0probe", "run_b0probe_delta.sh"), "b0probe/run_b0probe_delta.sh.txt")
for s in ["eb52114e9", "98b1723b1", "f385a8bc8"]:
    put(os.path.join(R, "b0probe", f"sha_{s}.txt"), f"b0probe/sha_{s}.txt")
put(os.path.join(L, "b0probe_delta_compare.txt"), "b0probe/b0probe_compare.txt")
put(os.path.join(R, "probe2", "projects/chirality-piping/core/solver/nonlinear_integration/src/structural_adapter/rv11_probe.rs"),
    "probes/rv11_probe.rs.txt")
put(os.path.join(L, "delta_probe.log"), "probes/delta_probe.log", cut=True)
put(os.path.join(L, "delta_probe_spring.log"), "probes/delta_probe_spring.log", cut=True)
for name in ["mutate_rv11d.py", "run_mutant.sh", "drive.sh", "apply.sh"]:
    put(os.path.join(M, name), f"mutations/{name}.txt")
for name in ["MUTANTS.txt", "drive.log", "mutators.txt"]:
    put(os.path.join(M, name), f"mutations/{name}")
for name in sorted(os.listdir(os.path.join(M, "logs"))):
    put(os.path.join(M, "logs", name), f"mutations/logs/{name}", cut=True)
put(os.path.abspath(__file__), "build_records_delta.py.txt")

rels = []
for d, _, fs in os.walk(top):
    for f in fs:
        rel = os.path.relpath(os.path.join(d, f), top)
        if rel != "SHA256SUMS":
            rels.append(rel)
with open(sums, "w") as f:
    for rel in sorted(rels):
        f.write(f"{sha(os.path.join(top, rel))}  ./{rel}\n")
print(len(rels), "files in SHA256SUMS;", sum(1 for r in rels if r.startswith("delta_f385a8bc8/")), "new")
