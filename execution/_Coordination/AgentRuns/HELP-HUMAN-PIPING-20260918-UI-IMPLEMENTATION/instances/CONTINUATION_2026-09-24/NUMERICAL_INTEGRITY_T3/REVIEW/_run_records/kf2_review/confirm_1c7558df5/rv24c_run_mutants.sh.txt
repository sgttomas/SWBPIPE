#!/bin/bash
# RV24 confirmation at 1c7558df5: per mutant, a clean copy of `core` from the reviewer's own git archive
# of 1c7558df5 (no reviewer probe mounted: the committed tests only), one edit (rv24_mutants.py, the
# review's edit texts), its own target; one at a time; copy and target deleted afterwards. Tests: FK's
# lib filtered to kf2_, krev04 and k1_negative_witness (one binary, 900 s alarm), then FKT/s11_site_table.
set -u
WT=$1; SP=$2; shift 2
OUT=$SP/c/mut
mkdir -p $OUT
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2
HEADFK=$WT/rv24c/head/projects/chirality-piping/core/solver/frame_kernel
for m in "$@"; do
  D=$WT/rv24c/mut/$m
  L=$OUT/$m.log
  rm -rf "$D"; mkdir -p "$D"
  cp -R $WT/rv24c/head/projects/chirality-piping/core "$D/"
  FK=$D/core/solver/frame_kernel
  { echo "== mutant $m $(date -u +%FT%TZ)"; uptime; } > "$L"
  python3 $SP/rv24_mutants.py "$m" "$FK/src/structural.rs" >> "$L" 2>&1
  diff -u "$HEADFK/src/structural.rs" "$FK/src/structural.rs" | sed "s#$WT#<wt>#g" >> "$L"
  diff -rq "$HEADFK/src" "$FK/src" | sed "s#$WT#<wt>#g" >> "$L"
  (cd "$FK" && CARGO_TARGET_DIR="$D/target" cargo test --offline --locked -j 4 --lib --no-run) > "$OUT/$m.build.log" 2>&1
  echo "build_exit $?" >> "$L"
  BIN=$(grep -o 'Executable unittests src/lib.rs ([^)]*)' "$OUT/$m.build.log" | sed 's/.*(\(.*\))/\1/')
  case "$BIN" in /*) ;; *) BIN="$FK/$BIN";; esac
  T0=$(date +%s)
  (cd "$FK" && perl -e 'alarm 900; exec @ARGV' "$BIN" kf2_ krev04 k1_negative_witness --nocapture) >> "$L" 2>&1
  echo "lib_exit $? seconds $(( $(date +%s) - T0 ))" >> "$L"
  (cd "$FK" && CARGO_TARGET_DIR="$D/target" cargo test --offline --locked -j 4 --test s11_site_table) >> "$L" 2>&1
  echo "site_exit $?" >> "$L"
  sed -i '' "s#$WT#<wt>#g; s#$SP#<scratch>/rv24#g; s#$HOME#<home>#g" "$L" "$OUT/$m.build.log"
  rm -rf "$D"
  echo "$m done: $(grep -E '^(build|lib|site)_exit' "$L" | tr '\n' ' ') failed: $(grep -E '^test .* FAILED$' "$L" | sed 's/^test //; s/ \.\.\. FAILED//; s/structural::kf2_witness_tests:://' | tr '\n' ' ')"
done
rmdir $WT/rv24c/mut 2>/dev/null || true
