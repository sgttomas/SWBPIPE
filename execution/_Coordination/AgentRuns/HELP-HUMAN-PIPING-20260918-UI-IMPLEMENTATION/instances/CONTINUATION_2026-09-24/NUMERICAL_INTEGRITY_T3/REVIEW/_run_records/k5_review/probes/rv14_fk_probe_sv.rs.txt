//! RV14 (independent reviewer of K5): FK probe for `assess_constrained_bodies`.
//! Reads cases from $RV14_IN, writes one result line per case to $RV14_OUT.
//! Case line: name|n|subs|ties|grounds|coords (hex bits), grounds dN / tN:x:y:z / rN:x:y:z.
use open_pipe_stress_frame_kernel::rigid_body::{
    assess_constrained_bodies, ConstrainedGround, GroundKind,
};
use std::io::Write;

fn hex(s: &str) -> f64 {
    f64::from_bits(u64::from_str_radix(s, 16).expect("hex"))
}

fn ground(g: &str) -> ConstrainedGround {
    if let Some(rest) = g.strip_prefix('d') {
        return ConstrainedGround::Dof(rest.parse().unwrap());
    }
    let kind = match &g[..1] {
        "t" => GroundKind::Translation,
        "r" => GroundKind::Rotation,
        _ => panic!("ground {g}"),
    };
    let parts: Vec<&str> = g[1..].split(':').collect();
    ConstrainedGround::Directional {
        node: parts[0].parse().unwrap(),
        kind,
        direction: [hex(parts[1]), hex(parts[2]), hex(parts[3])],
    }
}

fn hexs(v: &[f64]) -> String {
    v.iter()
        .map(|x| format!("{:016x}", x.to_bits()))
        .collect::<Vec<_>>()
        .join(",")
}

#[test]
fn rv14_probe_sv() {
    let input = std::env::var("RV14_IN").expect("RV14_IN");
    let output = std::env::var("RV14_OUT").expect("RV14_OUT");
    let text = std::fs::read_to_string(input).unwrap();
    let mut out = std::fs::File::create(output).unwrap();
    for line in text.lines() {
        if line.trim().is_empty() || line.starts_with('#') {
            continue;
        }
        let f: Vec<&str> = line.split('|').collect();
        let name = f[0];
        let n: usize = f[1].parse().unwrap();
        let subs: Vec<Vec<usize>> = f[2]
            .split(';')
            .map(|s| s.split(',').map(|x| x.parse().unwrap()).collect())
            .collect();
        let ties: Vec<[usize; 2]> = if f[3] == "-" {
            vec![]
        } else {
            f[3].split(',')
                .map(|t| {
                    let p: Vec<usize> = t.split('-').map(|x| x.parse().unwrap()).collect();
                    [p[0], p[1]]
                })
                .collect()
        };
        let grounds: Vec<ConstrainedGround> = if f[4] == "-" {
            vec![]
        } else {
            f[4].split(',').map(ground).collect()
        };
        let c: Vec<f64> = f[5].split(',').map(hex).collect();
        let coords: Vec<[f64; 3]> = c.chunks(3).map(|x| [x[0], x[1], x[2]]).collect();
        assert_eq!(coords.len(), n, "{name}");
        match assess_constrained_bodies(&coords, &subs, &ties, &grounds) {
            Ok(a) => {
                let smin = a.singular_values.iter().copied().fold(f64::INFINITY, f64::min);
                let rigid = a.rigid_parameters.map(|r| hexs(&r)).unwrap_or("-".into());
                let motion = a
                    .node_motion
                    .as_ref()
                    .map(|m| hexs(&m.iter().flatten().copied().collect::<Vec<_>>()))
                    .unwrap_or("-".into());
                writeln!(
                    out,
                    "{name}|{:?}|{}|{}|{:016x}|{rigid}|{motion}|{:e}|{:e}",
                    a.status,
                    a.rows,
                    a.cycles,
                    a.characteristic_length.to_bits(),
                    smin,
                    a.rank_screen
                )
                .unwrap();
                if std::env::var("RV14_SV").is_ok() {
                    writeln!(out, "{name}|sv|{:?}|iterations={}", a.singular_values, a.iterations)
                        .unwrap();
                }
            }
            Err(e) => writeln!(out, "{name}|ERR|{e:?}").unwrap(),
        }
    }
}
