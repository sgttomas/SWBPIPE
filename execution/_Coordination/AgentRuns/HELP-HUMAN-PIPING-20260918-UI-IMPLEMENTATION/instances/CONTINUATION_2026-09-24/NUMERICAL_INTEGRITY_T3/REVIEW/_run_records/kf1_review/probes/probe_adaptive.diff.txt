+/// RV20 probe (review only): a log of the stop rule's offers (tracker id,
+/// key, nonzero numerator), per `TrackerSet` (one per `rule` call).
+#[cfg(test)]
+pub(crate) mod rv20_log {
+    use std::cell::RefCell;
+
+    #[derive(Debug, Default, Clone)]
+    pub(crate) struct Call {
+        pub(crate) rows: Vec<(usize, u64, bool)>,
+        pub(crate) finished: bool,
+        pub(crate) cap_events: u64,
+    }
+
+    thread_local! {
+        static LOG: RefCell<Option<Vec<Call>>> = const { RefCell::new(None) };
+    }
+
+    pub(crate) fn start() {
+        LOG.with(|l| *l.borrow_mut() = Some(Vec::new()));
+    }
+
+    pub(crate) fn stop() -> Vec<Call> {
+        LOG.with(|l| l.borrow_mut().take().unwrap_or_default())
+    }
+
+    pub(crate) fn enabled() -> bool {
+        LOG.with(|l| l.borrow().is_some())
+    }
 
+    pub(crate) fn boundary() {
+        LOG.with(|l| {
+            if let Some(v) = l.borrow_mut().as_mut() {
+                v.push(Call::default());
+            }
+        });
+    }
+
+    fn last(f: impl FnOnce(&mut Call)) {
+        LOG.with(|l| {
+            if let Some(v) = l.borrow_mut().as_mut() {
+                if let Some(c) = v.last_mut() {
+                    f(c);
+                }
+            }
+        });
+    }
+
+    pub(crate) fn push(id: usize, key: u64, nonzero: bool) {
+        last(|c| c.rows.push((id, key, nonzero)));
+    }
+
+    pub(crate) fn finished() {
+        last(|c| c.finished = true);
+    }
+
+    pub(crate) fn cap_event() {
+        last(|c| c.cap_events += 1);
+    }
+}
