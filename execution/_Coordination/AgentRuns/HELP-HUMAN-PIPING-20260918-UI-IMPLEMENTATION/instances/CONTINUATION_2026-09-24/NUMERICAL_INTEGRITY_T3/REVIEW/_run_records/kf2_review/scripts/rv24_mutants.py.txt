                (j, j, sign, sign),

            ];"""
LOOP = """            for (a, b, da, db) in cells {
                if prepared.matrix[a][b] == 0.0 {
                    continue;
                }
                let original = radix_scale(
                    system.stiffness[system.free_dofs[a]][system.free_dofs[b]],
                    prepared.scale_exponents[a] + prepared.scale_exponents[b],
                )?;
                let term = checked_product(checked_product(da, original)?, db)?;
                energy = checked_value(energy + term)?;
                magnitude = checked_value(magnitude + term.abs())?;
                terms += 1;
            }"""
LOOP_SWALLOW = """            let mut failed = false;
            for (a, b, da, db) in cells {
                if prepared.matrix[a][b] == 0.0 {
                    continue;
                }
                let step = (|| -> Result<(f64, f64), StructuralError> {
                    let original = radix_scale(
                        system.stiffness[system.free_dofs[a]][system.free_dofs[b]],
                        prepared.scale_exponents[a] + prepared.scale_exponents[b],
                    )?;
                    let term = checked_product(checked_product(da, original)?, db)?;
                    Ok((checked_value(energy + term)?, checked_value(magnitude + term.abs())?))
                })();
                match step {
                    Ok((e, m)) => {
                        energy = e;
                        magnitude = m;
                    }
                    Err(_) => {
                        failed = true;
                        break;
                    }
                }
                terms += 1;
            }
            if failed {
                continue;
            }"""

def edit(src, old, new):
    start = src.index(FN_START)
    end = src.index(FN_END, start)
    body = src[start:end]
    assert body.count(old) == 1, (old[:60], body.count(old))
    return src[:start] + body.replace(old, new) + src[end:]


if __name__ == "__main__":
    if sys.argv[1] == "LIST":
        print(" ".join(MUTANTS))
        sys.exit(0)
    mutant, path = sys.argv[1], sys.argv[2]
    src = open(path).read()
    for old, new in MUTANTS[mutant]:
        src = edit(src, old, new)
    open(path, "w").write(src)
    print(mutant, "applied", len(MUTANTS[mutant]), "edit(s)")
