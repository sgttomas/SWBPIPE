#!/usr/bin/env python3
"""Assembles RV12's delta-check run records (K3 at 2511f5a3c) into
REVIEW/_run_records/k3_review/delta_2511f5a3c/, with its own SHA256SUMS, and
appends the new files to k3_review/SHA256SUMS without changing its existing
lines (standard library only). Machine paths become placeholders; trailing
spaces and trailing blank lines are stripped; nothing else changes."""
import hashlib
import sys
from pathlib import Path

WT = Path(sys.argv[1])
VENV = sys.argv[2]
SCRATCHPAD = sys.argv[3]
HOME = str(Path.home())
S = WT / "scratch" / "rv12" / "delta"
REVIEW = WT / "numerics" / ("projects/chirality-piping/execution/_Coordination/AgentRuns/"
                            "HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION/instances/"
                            "CONTINUATION_2026-09-24/NUMERICAL_INTEGRITY_T3/REVIEW/_run_records/k3_review")
OUT = REVIEW / "delta_2511f5a3c"


def sanitize(text):
    for real, ph in ((VENV, "<VENV>"), (SCRATCHPAD, "<scratch>"), (str(WT), "<wt>"), (HOME, "<home>")):
        text = text.replace(real, ph)
    lines = [l.rstrip() for l in text.splitlines()]
    while lines and not lines[-1]:
        lines.pop()
    return "\n".join(lines) + "\n"


def put(src, rel):
    dst = OUT / rel
    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_text(sanitize(Path(src).read_text(errors="replace")))


assert not OUT.exists() or all(p.is_file() or p.is_dir() for p in OUT.rglob("*"))
if OUT.exists():
    for p in sorted(OUT.rglob("*"), reverse=True):
        p.unlink() if p.is_file() else p.rmdir()
OUT.mkdir(parents=True, exist_ok=True)

for name in ["delta_checks.txt", "vectors_prefix_check.txt", "vector_sha256sums_check.txt",
             "k3_records_sums.txt", "gen8.txt", "ci_status.txt", "taildecides_oracle.txt"]:
    put(S / "logs" / name, "checks/" + name)
put(S / "logs" / "k3_gen_check.log", "checks/k3_gen_wide_k3_vectors_check.log")
put(S / "logs" / "k3a_gen_check_head.txt", "checks/k3a_gen_wide_vectors_check_head.txt")
put(S / "logs" / "fk_head_full_o0.log", "head_tests/fk_head_full_o0.log")
put(S / "logs" / "fk_head_nontest_build.log", "head_tests/fk_head_nontest_build.log")
put(S / "check_taildecides.py", "checks/check_taildecides.py.txt")
put(S / "rv12_delta_mutate.py", "mutations/rv12_delta_mutate.py.txt")
put(S / "mut" / "driver.log", "mutations/driver.log")
for p in sorted((S / "mut" / "out").iterdir()):
    put(p, "mutations/" + p.name)
put(S / "README.txt", "README.txt")
(OUT / "build_delta_records.py.txt").write_text(sanitize(Path(__file__).read_text()))

entries = []
for p in sorted(OUT.rglob("*")):
    if p.is_file() and p.name != "SHA256SUMS":
        entries.append("%s  %s" % (hashlib.sha256(p.read_bytes()).hexdigest(), p.relative_to(OUT).as_posix()))
(OUT / "SHA256SUMS").write_text("\n".join(entries) + "\n")

# Append the new files to the review folder's SHA256SUMS; existing lines kept as they are.
top = REVIEW / "SHA256SUMS"
old = top.read_text().splitlines()
kept = [l for l in old if not l.split("  ", 1)[1].startswith("delta_2511f5a3c/")]
new = []
for p in sorted(OUT.rglob("*")):
    if p.is_file():
        new.append("%s  %s" % (hashlib.sha256(p.read_bytes()).hexdigest(), p.relative_to(REVIEW).as_posix()))
top.write_text("\n".join(kept + new) + "\n")
print("delta files:", len(entries) + 1, "top entries:", len(kept) + len(new), "kept:", len(kept))
