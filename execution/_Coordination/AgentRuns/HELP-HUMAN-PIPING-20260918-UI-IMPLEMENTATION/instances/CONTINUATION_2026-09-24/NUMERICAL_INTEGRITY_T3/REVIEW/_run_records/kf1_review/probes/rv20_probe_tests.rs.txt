//! RV20 (independent review of KF1) probe tests. Not part of the candidate.
//!
//! The candidate's `BoundedExtremeTracker::offer` is given a test-only key
//! override (`rv20_key`), so a row's key can be set independently of its exact
//! value. The keys and the exact directed values are then unrelated, which
//! exercises the invariant argument of KF1 RETURN §3 without the approximation's
//! accuracy (inversions, refusals and ties at every key). The collapse schedule
//! is arbitrary: T is small and `collapse` is also called at random points.
//!
//! After every row the probe checks, against an oracle computed from the
//! definitions (L1, L2) and not from either tracker:
//! - K4's kept set equals L1's set (the reference is K4's code verbatim);
//! - J1: the unevaluated rows are a suffix of L1's set, at most T;
//! - J2: every table entry is (a_e, o(e)) for a row e of L1's set that is not
//!   unevaluated, and the table's keys are distinct;
//! - J3: every row of L1's set that is not unevaluated is dominated by a
//!   table entry with a nearer-or-equal key and an outcome at least as
//!   decisive.
//!
//! At the end, the bounded tracker's `finish`, K4's `finish` and the oracle's
//! F(K_n) are bit-identical, and the bounded tracker's 16-limb work is at most
//! one evaluation per row.
use super::super::wide::multi::{limb_multiply_cost, OpKind};
use super::*;

/// K4's `ExtremeTracker`, verbatim from `adaptive.rs:547-596` at `8cca91701`,
/// renamed, with the same key override after its approximation.
struct K4Tracker {
    direction: Direction,
    best: Option<u64>,
    kept: Vec<(ExactWideSum, ExactWideSum, u64)>,
}

impl K4Tracker {
    fn new(direction: Direction) -> Self {
        Self {
            direction,
            best: None,
            kept: Vec::new(),
        }
    }

    fn offer(
        &mut self,
        ctx64: &mut WideContext<4>,
        num: ExactWideSum,
        den: ExactWideSum,
    ) -> Result<(), AttemptStop> {
        let approx = approximate_ratio(ctx64, &num, &den)?.to_bits();
        let approx = rv20_key::take().unwrap_or(approx);
        let better = match (self.best, self.direction) {
            (None, _) => true,
            (Some(b), Direction::Up) => approx > b,
            (Some(b), Direction::Down) => approx < b,
        };
        if better {
            self.best = Some(approx);
            self.kept.retain(|k| k.2.abs_diff(approx) <= WINDOW_ULPS);
        }
        if self.best.is_some_and(|b| approx.abs_diff(b) <= WINDOW_ULPS) {
            self.kept.push((num, den, approx));
        }
        Ok(())
    }

    fn finish(self, ctx16: &mut WideContext<16>) -> Result<Option<f64>, AttemptStop> {
        let mut best: Option<f64> = None;
        for (num, den, _) in &self.kept {
            let exact = directed_ratio(ctx16, num, den, self.direction)?;
            best = Some(match (best, self.direction) {
                (None, _) => exact,
                (Some(b), Direction::Up) => b.max(exact),
                (Some(b), Direction::Down) => b.min(exact),
            });
        }
        Ok(best)
    }
}

const W: u64 = WINDOW_ULPS;

fn ctx64() -> WideContext<4> {
    WideContext::<4>::new(64).unwrap()
}

fn ctx16() -> WideContext<16> {
    WideContext::<16>::new(1024).unwrap()
}

fn sum_of(terms: &[f64]) -> ExactWideSum {
    let mut s = ExactWideSum::new();
    for &t in terms {
        s.add_binary64(t, false).unwrap();
    }
    s
}

/// The value kinds, independent of the keys.
const VALUES: usize = 9;

fn pair(v: usize) -> (ExactWideSum, ExactWideSum) {
    match v {
        0 => (sum_of(&[1.0]), sum_of(&[1.0])),
        1 => (sum_of(&[2.0]), sum_of(&[1.0])),
        2 => (sum_of(&[3.0]), sum_of(&[7.0])), // inexact: Up and Down differ
        3 => {
            // 2·(1 + 2^-100): Up next_up(2), Down 2.
            let mut n = sum_of(&[2.0]);
            n.add_binary64(2.0 * f64::from_bits(0x39B0_0000_0000_0000), false)
                .unwrap();
            (n, sum_of(&[1.0]))
        }
        4 => (ExactWideSum::new(), sum_of(&[3.0])), // zero
        5 | 6 => {
            // A refusal (Span): num = c, den = 1 + 2^-8100.
            let c = if v == 5 {
                f64::from_bits(0x3FF8_0000_0000_0001)
            } else {
                f64::from_bits(0x4008_0000_0000_0003)
            };
            let mut den = sum_of(&[1.0]);
            den.add_integer(false, &[1], -8100).unwrap();
            (sum_of(&[c]), den)
        }
        7 => (
            // Above the binary64 range: Up +inf, Down f64::MAX.
            sum_of(&[f64::from_bits(0x7E70_0000_0000_0000)]),
            sum_of(&[f64::from_bits(0x39B0_0000_0000_0000)]),
        ),
        _ => (sum_of(&[1.0]), sum_of(&[3.0])), // 1/3, inexact
    }
}

struct Rng(u64);

impl Rng {
    fn next(&mut self) -> u64 {
        self.0 = self.0.wrapping_add(0x9E37_79B9_7F4A_7C15);
        let mut z = self.0;
        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
        z ^ (z >> 31)
    }
    fn below(&mut self, n: u64) -> u64 {
        self.next() % n
    }
    fn chance(&mut self, percent: u64) -> bool {
        self.below(100) < percent
    }
}

fn nearer_or_equal(k: u64, a: u64, direction: Direction) -> bool {
    match direction {
        Direction::Up => k >= a,
        Direction::Down => k <= a,
    }
}

fn better(k: u64, b: u64, direction: Direction) -> bool {
    match direction {
        Direction::Up => k > b,
        Direction::Down => k < b,
    }
}

/// A stream of (key, value kind): keys near the running best's window edges,
/// new bests by small and large steps, values drawn independently.
fn stream(rng: &mut Rng, direction: Direction, refusal_percent: u64) -> Vec<(u64, usize)> {
    let base: u64 = 1 << 50;
    let offsets = [0u64, 1, 2, 3, W - 2, W - 1, W, W + 1, W + 2, 2 * W, 3 * W + 5];
    let n = 1 + rng.below(16) as usize;
    let mut best: Option<u64> = None;
    let mut out = Vec::with_capacity(n);
    for _ in 0..n {
        let key = match best {
            None => base,
            Some(b) => {
                let d = offsets[rng.below(offsets.len() as u64) as usize];
                let toward = rng.chance(25);
                match (direction, toward) {
                    (Direction::Up, true) | (Direction::Down, false) => b + d,
                    (Direction::Up, false) | (Direction::Down, true) => b - d,
                }
            }
        };
        best = Some(match best {
            None => key,
            Some(b) if better(key, b, direction) => key,
            Some(b) => b,
        });
        let v = if rng.chance(refusal_percent) {
            5 + rng.below(2) as usize
        } else {
            let mut v = rng.below(VALUES as u64) as usize;
            if v == 5 || v == 6 {
                v = 8;
            }
            v
        };
        out.push((key, v));
    }
    out
}

#[derive(Default, Debug)]
struct Stats {
    runs: u64,
    rows: u64,
    collapses_explicit: u64,
    err_results: u64,
    ok_results: u64,
    max_table: usize,
    max_value_entries: usize,
    inverted_tables: u64,
}

fn evaluation_cost() -> u64 {
    2 * limb_multiply_cost(OpKind::Round, 16) + limb_multiply_cost(OpKind::Div, 16)
}

#[allow(clippy::too_many_lines)]
fn run(
    s: &[(u64, usize)],
    direction: Direction,
    limit: usize,
    explicit_percent: u64,
    rng: &mut Rng,
    outcomes: &[Result<f64, AttemptStop>],
    stats: &mut Stats,
) {
    let mut r = K4Tracker::new(direction);
    let mut b = BoundedExtremeTracker::with_limit(direction, limit);
    let (mut c64r, mut c64b) = (ctx64(), ctx64());
    let (mut c16r, mut c16b) = (ctx16(), ctx16());
    let keys: Vec<u64> = s.iter().map(|e| e.0).collect();
    let o_of = |j: usize| -> Evaluated {
        match &outcomes[s[j].1] {
            Ok(x) => Evaluated::Ratio(*x),
            Err(stop) => Evaluated::Refused {
                seq: j as u64,
                stop: stop.clone(),
            },
        }
    };
    let mut best: Option<u64> = None;
    for (i, &(key, v)) in s.iter().enumerate() {
        let (num, den) = pair(v);
        rv20_key::set(key);
        r.offer(&mut c64r, num.clone(), den.clone()).unwrap();
        assert_eq!(rv20_key::take(), None);
        rv20_key::set(key);
        b.offer(&mut c64b, &mut c16b, num, den).unwrap();
        assert_eq!(rv20_key::take(), None);
        if explicit_percent > 0 && rng.chance(explicit_percent) {
            b.collapse(&mut c16b);
            stats.collapses_explicit += 1;
        }
        stats.rows += 1;
        best = Some(match best {
            None => key,
            Some(x) if better(key, x, direction) => key,
            Some(x) => x,
        });
        let bb = best.unwrap();
        assert_eq!(b.best, best);
        assert_eq!(r.best, best);
        // L1, the oracle's kept set.
        let kset: Vec<usize> = (0..=i).filter(|&j| in_window(keys[j], bb)).collect();
        let kept_keys: Vec<u64> = r.kept.iter().map(|k| k.2).collect();
        let l1_keys: Vec<u64> = kset.iter().map(|&j| keys[j]).collect();
        assert_eq!(kept_keys, l1_keys, "L1");
        // J1.
        assert!(b.lazy.len() <= limit);
        let lazy_seqs: Vec<usize> = b.lazy.iter().map(|e| e.3 as usize).collect();
        assert!(kset.ends_with(&lazy_seqs), "J1 {kset:?} {lazy_seqs:?}");
        for e in &b.lazy {
            assert_eq!(e.2, keys[e.3 as usize]);
        }
        let evaluated: Vec<usize> = kset
            .iter()
            .copied()
            .filter(|j| !lazy_seqs.contains(j))
            .collect();
        // J2, and distinct keys.
        let mut table_keys: Vec<u64> = b.table.iter().map(|e| e.0).collect();
        table_keys.sort_unstable();
        table_keys.dedup();
        assert_eq!(table_keys.len(), b.table.len(), "J2: distinct keys");
        for (k, o) in &b.table {
            let realized = evaluated.iter().any(|&j| {
                keys[j] == *k
                    && match (o, o_of(j)) {
                        (Evaluated::Ratio(x), Evaluated::Ratio(y)) => x.to_bits() == y.to_bits(),
                        (Evaluated::Refused { seq, stop }, Evaluated::Refused { seq: s2, stop: t2 }) => {
                            *seq == s2 && *stop == t2
                        }
                        _ => false,
                    }
            });
            assert!(realized, "J2: {k} {o:?}");
        }
        // J3.
        for &j in &evaluated {
            let oj = o_of(j);
            let dominated = b
                .table
                .iter()
                .any(|(k, o)| nearer_or_equal(*k, keys[j], direction) && !oj.beats(o, direction));
            assert!(dominated, "J3: row {j}");
        }
        // |Λ| + |Σ| ≤ |K_i|.
        assert!(b.lazy.len() + b.table.len() <= kset.len());
        stats.max_table = stats.max_table.max(b.table.len());
        let values = b
            .table
            .iter()
            .filter(|e| matches!(e.1, Evaluated::Ratio(_)))
            .count();
        stats.max_value_entries = stats.max_value_entries.max(values);
    }
    // L2, the oracle's result.
    let bb = best.unwrap();
    let kset: Vec<usize> = (0..s.len()).filter(|&j| in_window(keys[j], bb)).collect();
    let mut oracle: Result<Option<f64>, AttemptStop> = Ok(None);
    for &j in &kset {
        match &outcomes[s[j].1] {
            Err(stop) => {
                oracle = Err(stop.clone());
                break;
            }
            Ok(x) => {
                let prev = oracle.as_ref().unwrap();
                oracle = Ok(Some(match (prev, direction) {
                    (None, _) => *x,
                    (Some(p), Direction::Up) => p.max(*x),
                    (Some(p), Direction::Down) => p.min(*x),
                }));
            }
        }
    }
    let bits = |o: Result<Option<f64>, AttemptStop>| o.map(|v| v.map(f64::to_bits));
    let values_in_table = b
        .table
        .iter()
        .filter(|e| matches!(e.1, Evaluated::Ratio(_)))
        .count();
    if values_in_table > 1 {
        stats.inverted_tables += 1;
    }
    let reference = bits(r.finish(&mut c16r));
    let bounded = bits(b.finish(&mut c16b));
    let oracle = bits(oracle);
    assert_eq!(reference, oracle, "K4 against the oracle: {s:?}");
    assert_eq!(bounded, oracle, "bounded against the oracle: {direction:?} T = {limit}: {s:?}");
    if oracle.is_err() {
        stats.err_results += 1;
    } else {
        stats.ok_results += 1;
    }
    // Each row is evaluated at most once.
    assert!(lme(&c16b) <= evaluation_cost() * s.len() as u64);
    stats.runs += 1;
}

#[test]
fn rv20_arbitrary_keys_and_collapse_schedules_match_k4_and_the_oracle() {
    let mut rng = Rng(0x5256_3230_5052_4F42); // "RV20PROB"
    let mut stats = Stats::default();
    for direction in [Direction::Up, Direction::Down] {
        let outcomes: Vec<Result<f64, AttemptStop>> = (0..VALUES)
            .map(|v| {
                let (n, d) = pair(v);
                directed_ratio(&mut ctx16(), &n, &d, direction)
            })
            .collect();
        assert!(outcomes[5].is_err() && outcomes[6].is_err());
        for (limit, explicit) in [
            (1usize, 0u64),
            (2, 0),
            (3, 0),
            (4, 0),
            (1, 30),
            (2, 50),
            (3, 20),
            (5, 70),
            (usize::MAX, 40),
            (usize::MAX, 0),
        ] {
            for refusal in [0u64, 8, 30] {
                for _ in 0..600 {
                    let s = stream(&mut rng, direction, refusal);
                    run(&s, direction, limit, explicit, &mut rng, &outcomes, &mut stats);
                }
            }
        }
    }
    println!("RV20 probe: {stats:?}");
    assert!(stats.runs >= 36_000);
    assert!(stats.err_results >= 1_000 && stats.ok_results >= 10_000);
    assert!(stats.inverted_tables >= 100, "keys unrelated to values: {stats:?}");
}

#[test]
fn rv20_tracker_set_with_arbitrary_keys_tiny_caps() {
    struct Hook;
    impl Drop for Hook {
        fn drop(&mut self) {
            tracker_hook::set(None);
        }
    }
    let _hook = Hook;
    let mut rng = Rng(0x5256_3230_5345_5453); // "RV20SETS"
    let mut all_collapses = 0u64;
    for round in 0..400u64 {
        let direction = if round % 2 == 0 {
            Direction::Up
        } else {
            Direction::Down
        };
        let outcomes: Vec<Result<f64, AttemptStop>> = (0..VALUES)
            .map(|v| {
                let (n, d) = pair(v);
                directed_ratio(&mut ctx16(), &n, &d, direction)
            })
            .collect();
        let t = [1usize, 2, 3, 4, 8][rng.below(5) as usize];
        tracker_hook::set(Some(t));
        let g = [1usize, 2, 3, 5, 9][rng.below(5) as usize];
        let mut set: TrackerSet<u32> = TrackerSet::with_limit(g);
        let mut refs: std::collections::BTreeMap<u32, K4Tracker> = std::collections::BTreeMap::new();
        let (mut c64s, mut c64r) = (ctx64(), ctx64());
        let (mut c16s, mut c16r) = (ctx16(), ctx16());
        let trackers = 1 + rng.below(12) as u32;
        let mut rows: Vec<(u32, (u64, usize))> = Vec::new();
        for id in 0..trackers {
            for e in stream(&mut rng, direction, 10) {
                rows.push((id, e));
            }
        }
        for i in (1..rows.len()).rev() {
            rows.swap(i, rng.below(i as u64 + 1) as usize);
        }
        let mut last = 0;
        for (id, (key, v)) in rows {
            let (num, den) = pair(v);
            rv20_key::set(key);
            set.offer(id, direction, &mut c64s, &mut c16s, num.clone(), den.clone())
                .unwrap();
            rv20_key::set(key);
            refs.entry(id)
                .or_insert_with(|| K4Tracker::new(direction))
                .offer(&mut c64r, num, den)
                .unwrap();
            let (held, limit) = set.held();
            assert_eq!(limit, g);
            assert!(held <= g, "{held} > {g}");
            let cap: usize = set.trackers().map(|(_, tr)| tr.held().1).sum();
            assert_eq!(held, cap);
            assert!(set.trackers().all(|(_, tr)| tr.held().0 <= t));
            if held == 0 && last > 0 {
                all_collapses += 1;
            }
            last = held;
        }
        let bits = |o: Result<Option<f64>, AttemptStop>| o.map(|v| v.map(f64::to_bits));
        for (id, tr) in set.into_trackers() {
            assert_eq!(
                bits(tr.finish(&mut c16s)),
                bits(refs.remove(&id).unwrap().finish(&mut c16r)),
                "round {round} tracker {id}"
            );
        }
        let _ = &outcomes;
    }
    println!("RV20 tracker set: {all_collapses} collapses of every holding tracker");
    assert!(all_collapses >= 200);
}

#[test]
fn rv20_rule_test_order_is_k4s_map_order() {
    assert!(RuleTest::Disagreement < RuleTest::Estimate);
    assert!(RuleTest::Estimate < RuleTest::Charge);
    let mut set: TrackerSet<(RuleTest, u32, Kind)> = TrackerSet::with_limit(usize::MAX);
    let mut c64 = ctx64();
    let mut c16 = ctx16();
    for test in [RuleTest::Charge, RuleTest::Disagreement, RuleTest::Estimate] {
        for body in [3u32, 0, 1] {
            for kind in [Kind::Moment, Kind::Force] {
                let (n, d) = pair(0);
                set.offer((test, body, kind), Direction::Up, &mut c64, &mut c16, n, d)
                    .unwrap();
            }
        }
    }
    let order: Vec<(RuleTest, u32, Kind)> = set.into_trackers().map(|(k, _)| k).collect();
    let mut expected = Vec::new();
    for test in [RuleTest::Disagreement, RuleTest::Estimate, RuleTest::Charge] {
        let mut m: std::collections::BTreeMap<(u32, Kind), ()> = std::collections::BTreeMap::new();
        for body in [3u32, 0, 1] {
            for kind in [Kind::Moment, Kind::Force] {
                m.insert((body, kind), ());
            }
        }
        for (body, kind) in m.keys() {
            expected.push((test, *body, *kind));
        }
    }
    assert_eq!(order, expected);
}

#[allow(dead_code)]

const MODELS_5A3: &str = include_str!("models5a3.txt");

/// The extra exact evaluations of one `rule` call under (T, G), from its key
/// stream alone (independent of the tracker code): per tracker, K4's window
/// rule; collapses when T rows are held, and of every holding tracker when
/// the allocated capacities (Vec growth 0, 1, 2, 4, ...; released when
/// emptied) exceed G. Returns (extra evaluations, cap events).
fn replay_call(call: &rv20_log::Call, t: usize, g: usize) -> (i64, u64) {
    #[derive(Default)]
    struct Tr {
        best: Option<u64>,
        all: Vec<(u64, bool)>,
        lazy: Vec<(u64, bool)>,
        cap: usize,
        collapsed: Vec<(u64, bool)>,
    }
    let mut trs: Vec<Tr> = Vec::new();
    let mut cap_events = 0u64;
    for &(id, key, nonzero) in &call.rows {
        while trs.len() <= id {
            trs.push(Tr::default());
        }
        let tr = &mut trs[id];
        tr.all.push((key, nonzero));
        let better = tr.best.is_none_or(|b| key > b);
        if better {
            tr.best = Some(key);
            tr.lazy.retain(|r| r.0.abs_diff(key) <= W);
            if tr.lazy.is_empty() {
                tr.cap = 0;
            }
        }
        if tr.best.is_some_and(|b| key.abs_diff(b) <= W) {
            if tr.lazy.len() >= t {
                let drained: Vec<(u64, bool)> = tr.lazy.drain(..).collect();
                tr.collapsed.extend(drained);
                tr.cap = 0;
            }
            if tr.lazy.len() == tr.cap {
                tr.cap = (2 * tr.cap).max(1);
            }
            tr.lazy.push((key, nonzero));
        }
        let held: usize = trs.iter().map(|x| x.cap).sum();
        if held > g {
            cap_events += 1;
            for x in trs.iter_mut() {
                if x.cap > 0 {
                    let drained: Vec<(u64, bool)> = x.lazy.drain(..).collect();
                    x.collapsed.extend(drained);
                    x.cap = 0;
                }
            }
        }
    }
    let mut extra = 0i64;
    for tr in &trs {
        let bounded = tr.collapsed.iter().filter(|r| r.1).count() as i64;
        if !call.finished {
            extra += bounded;
            continue;
        }
        let b = tr.best.unwrap();
        let k4 = tr.all.iter().filter(|r| r.1 && r.0.abs_diff(b) <= W).count() as i64;
        let lazy = tr.lazy.iter().filter(|r| r.1).count() as i64;
        extra += bounded + lazy - k4;
    }
    (extra, cap_events)
}

struct Setting;

impl Setting {
    fn new(rows: Option<usize>, seeds: &[(usize, f64)]) -> Self {
        tracker_hook::set(rows);
        seed::set(seeds.to_vec());
        Setting
    }
}

impl Drop for Setting {
    fn drop(&mut self) {
        tracker_hook::set(None);
        seed::set(Vec::new());
    }
}

fn stop_rule_total(outcome: &CaseOutcome) -> u64 {
    match outcome {
        CaseOutcome::Selected(s) => s.evidence().attempts.iter().map(|a| a.stop_rule_work).sum(),
        CaseOutcome::Unresolved { attempts, .. } => attempts.iter().map(|a| a.stop_rule_work).sum(),
        CaseOutcome::Refused { .. } => 0,
    }
}

#[test]
fn rv20_replay_derives_the_collapse_pins_from_the_key_streams() {
    let cost = 2 * limb_multiply_cost(OpKind::Round, 16) + limb_multiply_cost(OpKind::Div, 16);
    let mut all = models::models();
    all.extend(models::parse_models(MODELS_5A3));
    let mut frames = 0;
    for m in all.iter().filter(|m| m.name.contains("n00100")) {
        let (base, calls) = {
            let _s = Setting::new(Some(usize::MAX), &m.seeds);
            rv20_log::start();
            let mut meter = InvocationMeter::new(u64::MAX);
            let out = solve_case(m.source(), CaseLimit::new(u64::MAX), &mut meter);
            (out, rv20_log::stop())
        };
        let at64 = {
            let _s = Setting::new(Some(64), &m.seeds);
            let mut meter = InvocationMeter::new(u64::MAX);
            solve_case(m.source(), CaseLimit::new(u64::MAX), &mut meter)
        };
        let measured = (stop_rule_total(&at64) - stop_rule_total(&base)) as i64;
        assert_eq!(measured % cost as i64, 0);
        let mut line = format!("RV20 replay {}: rule calls {}, rows {}:", m.name, calls.len(),
            calls.iter().map(|c| c.rows.len()).sum::<usize>());
        let mut per_t = Vec::new();
        for t in [16usize, 32, 64, 128, 256, 512] {
            let (mut extra, mut events) = (0i64, 0u64);
            for c in &calls {
                let (e, ev) = replay_call(c, t, 8 * t);
                extra += e;
                events += ev;
            }
            line += &format!(" T={t}:+{extra} (cap events {events})");
            per_t.push((t, extra));
        }
        line += &format!("; measured at T=64: +{}", measured / cost as i64);
        println!("{line}");
        assert_eq!(per_t[2].1, measured / cost as i64, "{}: replay against measured at T = 64", m.name);
        assert_eq!(per_t[5].1, 0, "{}: T = 512", m.name);
        if m.name == "RF-LARGE-CHAIN-n00100-AX" {
            assert_eq!(per_t[2].1, 768);
        }
        if m.name == "RF-LARGE-CONT-n00100-AX" {
            assert_eq!(per_t[2].1, 640);
        }
        frames += 1;
    }
    assert_eq!(frames, 6);
}
