#!/bin/bash
# RV7 re-run of the surviving mutants on a new K2a tree: rerun.sh <sha>
set -u
SHA=$1
S=<scratch>/rv7
TGT=<wt>/rv7-target
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0
LOG=$S/logs_rerun; rm -rf $LOG $S/new; mkdir -p $LOG $S/new
(cd <wt>/k2a && git archive $SHA projects/chirality-piping/core/solver/frame_kernel) | tar -x -C $S/new
FK=$S/new/projects/chirality-piping/core/solver/frame_kernel
sha256sum $FK/src/lib.rs $FK/tests/k2a_checked_formation.rs > $LOG/sources_sha256.txt
for m in NONE R1_sub_ok_12EI_L3_quotients R2_sub_ok_EA_GJ R3_sub_ok_L3 R4_sub_ok_12EIz_product R5_sub_ok_12E R5b_check_reciprocal_product R10_refuse_min_positive; do
  free=$(df --output=avail -k <home> | tail -1); if [ "$free" -lt 8912896 ]; then echo "ABORT low disk" >> $LOG/SUMMARY.txt; exit 3; fi
  rm -rf $S/fk_re $TGT/fk_re; cp -r $FK $S/fk_re
  (cd $S && python3 mutants.py $m $FK/src/lib.rs fk_re/src/lib.rs) > $LOG/mut_$m.log 2>&1 || { echo "mutant $m PATCH FAILED" >> $LOG/SUMMARY.txt; continue; }
  CARGO_TARGET_DIR=$TGT/fk_re cargo test --offline --locked --manifest-path $S/fk_re/Cargo.toml --test k2a_checked_formation --no-fail-fast >> $LOG/mut_$m.log 2>&1
  rc=$?
  echo "mutant $m exit $rc | $(grep -h 'test result' $LOG/mut_$m.log | tail -1) | failed: $(grep -o '^    k2a_[a-z0-9_]*' $LOG/mut_$m.log | tr -d ' ' | tr '\n' ' ') | at: $(grep -o 'panicked at tests/[^ ]*' $LOG/mut_$m.log | sed 's/panicked at //' | tr '\n' ' ')" >> $LOG/SUMMARY.txt
done
rm -rf $TGT
echo done >> $LOG/SUMMARY.txt
