#!/usr/bin/env python3
"""RV6 evasion mutants for F1a. Usage: rv6_mutate.py <lib.rs> <NAME>; NONE restores nothing."""
import sys

MUTANTS = {
    # E1: drop the record for sparse mode only (the plumbing constructor).
    "E1_SPARSE_DROP": [(
        "        formation_check: checked.formation_check,\n",
        "        formation_check: checked.formation_check.filter(|_| solver_mode == PreviewSolverMode::DenseScrutiny),\n",
    )],
    # E2: field order changed (scale and trigger_ratio swapped, values stay under their keys).
    "E2_FIELD_ORDER": [(
        "doubled_correction={:?}; scale={:?}; trigger_ratio={:?}\",",
        "doubled_correction={:?}; trigger_ratio={:?}; scale={:?}\",",
    ), (
        "            check.scale,\n            check.ratio,\n",
        "            check.ratio,\n            check.scale,\n",
    )],
    # E3: a line rendered when there is no record.
    "E3_LINE_WITHOUT_RECORD": [(
        "    if let (Some(check), Some(record)) = (formation_check, diagnostics.last_mut()) {\n",
        "    if formation_check.is_none() {\n        if let Some(record) = diagnostics.last_mut() {\n            record.message.push_str(\" formation_check: reason=none\");\n        }\n    }\n    if let (Some(check), Some(record)) = (formation_check, diagnostics.last_mut()) {\n",
    )],
    # E4: Display instead of Debug for one f64.
    "E4_DISPLAY_F64": [(
        "doubled_correction={:?}; scale",
        "doubled_correction={}; scale",
    )],
    # E5: the line is dropped whenever an S11-G finding exists.
    "E5_DROP_WHEN_S11G": [(
        "    if let (Some(check), Some(record)) = (formation_check, diagnostics.last_mut()) {\n",
        "    if let (Some(check), Some(record), None) = (formation_check, diagnostics.last_mut(), formation) {\n",
    )],
    # E6: raw index instead of integrity_dof_label.
    "E6_RAW_ROW": [(
        "|dof| integrity_dof_label(model, dof)),",
        "|dof| format!(\"global_dof={dof}\")),",
    )],
}

path, name = sys.argv[1], sys.argv[2]
text = open(path).read()
for old, new in MUTANTS[name]:
    assert text.count(old) == 1, (name, old)
    text = text.replace(old, new)
open(path, "w").write(text)
print("applied", name)
