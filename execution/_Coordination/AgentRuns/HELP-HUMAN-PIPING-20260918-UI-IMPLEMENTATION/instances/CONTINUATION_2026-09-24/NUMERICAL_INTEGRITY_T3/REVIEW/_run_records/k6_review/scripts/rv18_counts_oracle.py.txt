"""RV18: an independent re-derivation of K6's counts (standard library only).

Usage: python3 rv18_counts_oracle.py <dump dir> <counts.jsonl> <models dir or emit binary>

1. RCM profile and half-bandwidth: RV18's own port of SD's `reverse_cuthill_mckee`
   (SD/lib.rs:505-615: symmetrize, sort, dedup; neighbours sorted by (degree, index);
   components by lowest index; pseudo-peripheral start by level-structure iteration;
   BFS; reverse) and of `from_entries_with_order`'s first-column rule (SD/lib.rs:215-245),
   run on the prepared free block's nonzero lower positions that `k6_observe --counts-only
   --dump-pattern` prints (the positions come from FK's `prepare_sparse_structural`).
2. The identity-order profile and half-bandwidth from the lane's entry positions
   (`from_entries` in identity order).
3. Closed forms from the canonical model alone: pattern entries from the distinct node
   adjacencies, lower entries, free DOFs, free entries (structural, both triangles), free lower
   entries, contributions and dense entries.
It compares each with the counts line of `observations/k6/counts.jsonl`.
"""
import json
import os
import subprocess
import sys
from collections import deque


def rcm(n, adjacency_pairs):
    nb = [set() for _ in range(n)]
    for a, b in adjacency_pairs:
        if a != b:
            nb[a].add(b)
            nb[b].add(a)
    deg = [len(s) for s in nb]
    nbs = [sorted(s, key=lambda x: (deg[x], x)) for s in nb]
    visited = [False] * n
    order = []

    def reach(seed):
        mark = {seed}
        out = [seed]
        q = deque([seed])
        while q:
            v = q.popleft()
            for w in nbs[v]:
                if w not in mark:
                    mark.add(w)
                    out.append(w)
                    q.append(w)
        return out

    def ecc(start):
        mark = {start}
        cur = [start]
        last = [start]
        e = 0
        while True:
            nxt = []
            for v in cur:
                for w in nbs[v]:
                    if w not in mark:
                        mark.add(w)
                        nxt.append(w)
            if not nxt:
                break
            e += 1
            last = nxt
            cur = nxt
        return e, last

    for seed in range(n):
        if visited[seed]:
            continue
        comp = reach(seed)
        cand = min(comp, key=lambda x: (deg[x], x))
        ce, last = ecc(cand)
        while True:
            if not last:
                break
            nxt = min(last, key=lambda x: (deg[x], x))
            ne, nl = ecc(nxt)
            if ne > ce:
                cand, ce, last = nxt, ne, nl
            else:
                break
        visited[cand] = True
        q = deque([cand])
        while q:
            v = q.popleft()
            order.append(v)
            for w in nbs[v]:
                if not visited[w]:
                    visited[w] = True
                    q.append(w)
    order.reverse()
    return order


def skyline(n, pairs, order):
    pos = [0] * n
    for k, orig in enumerate(order):
        pos[orig] = k
    first = list(range(n))
    for a, b in pairs:
        pa, pb = pos[a], pos[b]
        hi, lo = (pa, pb) if pa >= pb else (pb, pa)
        if lo < first[hi]:
            first[hi] = lo
    return sum(r - f + 1 for r, f in enumerate(first)), max((r - f for r, f in enumerate(first)), default=0)


def read_dump(path):
    lines = open(path).read().split('\n')
    k = 0
    assert lines[k] == 'k6-pattern v1'
    mid = lines[1].split(' ', 1)[1]
    nfree = int(lines[2].split()[1])
    m = int(lines[3].split()[1])
    rcm_pairs = [tuple(map(int, lines[4 + i].split())) for i in range(m)]
    j = 4 + m
    lane_dim = int(lines[j].split()[1])
    le = int(lines[j + 1].split()[1])
    lane_pairs = [tuple(map(int, lines[j + 2 + i].split())) for i in range(le)]
    return mid, nfree, rcm_pairs, lane_dim, lane_pairs


def canonical(binary, mid):
    out = subprocess.run([binary, '--emit-model', '--model', mid], capture_output=True, check=True).stdout.decode()
    nodes, members, restraints = 0, [], {}
    for ln in out.splitlines():
        t = ln.split(' ')
        if t[0] == 'node':
            nodes += 1
        elif t[0] == 'member':
            members.append((int(t[3]), int(t[4])))
        elif t[0] == 'restraint':
            restraints[int(t[1])] = t[2]
    return nodes, members, restraints


def closed(nodes, members, restraints):
    pairs = {tuple(sorted(p)) for p in members}
    free = [6 - restraints.get(k, '000000').count('1') for k in range(nodes)]
    nf = sum(free)
    free_entries = sum(f * f for f in free) + 2 * sum(free[a] * free[b] for a, b in pairs)
    return {'nodes': nodes, 'members': len(members), 'dofs': 6 * nodes, 'free_dofs': nf,
            'restrained_dofs': 6 * nodes - nf, 'pattern_entries': 36 * (nodes + 2 * len(pairs)),
            'lower_entries': (36 * (nodes + 2 * len(pairs)) + 6 * nodes) // 2,
            'free_entries': free_entries, 'free_lower_entries': (free_entries + nf) // 2,
            'contributions': 144 * len(members), 'dense_entries': (6 * nodes) ** 2}


def main():
    dump_dir, counts_path, binary = sys.argv[1:4]
    counts = {}
    for line in open(counts_path):
        o = json.loads(line)
        counts[o['model']] = o
    bad = 0
    for name in sorted(os.listdir(dump_dir)):
        if not name.endswith('.pattern'):
            continue
        mid, nfree, rp, ld, lp = read_dump(os.path.join(dump_dir, name))
        c = counts[mid]
        order = rcm(nfree, rp)
        prof, bw = skyline(nfree, rp, order)
        iprof, ibw = skyline(ld, lp, list(range(ld)))
        derived = {'rcm_profile_entries': prof, 'rcm_half_bandwidth': bw, 'identity_profile_entries': iprof,
                   'identity_half_bandwidth': ibw, 'free_lower_nonzero': len(rp), 'lane_entries': len(lp),
                   'lane_off_diagonal': sum(1 for a, b in lp if a != b)}
        derived.update(closed(*canonical(binary, mid)))
        diffs = {k: (v, c[k]) for k, v in derived.items() if c[k] != v}
        bad += bool(diffs)
        print('%s %s rcm=%d/%d id=%d/%d %s' % ('EQUAL' if not diffs else 'DIFFER', mid, prof, bw, iprof, ibw,
                                                diffs if diffs else ''))
    print('summary: %d differ' % bad)
    return 1 if bad else 0


if __name__ == '__main__':
    sys.exit(main())
