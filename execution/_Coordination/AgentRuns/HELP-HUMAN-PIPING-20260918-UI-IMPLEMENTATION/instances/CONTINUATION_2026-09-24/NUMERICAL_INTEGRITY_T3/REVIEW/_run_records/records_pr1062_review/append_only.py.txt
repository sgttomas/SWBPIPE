#!/usr/bin/env python3
"""RV25: append-only check of every file that PR #1062 modifies.

Run from the repository root of a Git working tree that has both commits:
    python3 append_only.py.txt <base> <head>
For each modified file it reads base and head bytes by `git show`, and reports
difflib opcodes on lines. For each 'replace' it checks whether each old line is
a character subsequence of its new line (an insert-only in-place edit), and
prints the inserted text. SHA256SUMS files are reported entry by entry.
"""
import difflib
import subprocess
import sys


def show(rev, path):
    return subprocess.run(["git", "show", f"{rev}:{path}"], check=True,
                          capture_output=True).stdout.decode("utf-8")


def is_subseq(a, b):
    it = iter(b)
    return all(ch in it for ch in a)


def inserted(a, b):
    sm = difflib.SequenceMatcher(None, a, b, autojunk=False)
    out = []
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == "insert":
            out.append(("+", j1, b[j1:j2]))
        elif tag != "equal":
            out.append((tag, i1, a[i1:i2], b[j1:j2]))
    return out


def main():
    base, head = sys.argv[1], sys.argv[2]
    names = subprocess.run(["git", "diff", "--name-status", base, head],
                           check=True, capture_output=True, text=True).stdout
    mods = [l.split("\t", 1)[1] for l in names.splitlines() if l.startswith("M\t")]
    for path in mods:
        old = show(base, path)
        new = show(head, path)
        a = old.splitlines(keepends=True)
        b = new.splitlines(keepends=True)
        print(f"== {path}")
        print(f"   base lines {len(a)}, head lines {len(b)}; "
              f"head starts with base bytes: {new.startswith(old)}")
        if path.endswith("SHA256SUMS"):
            oa = {l.split(None, 1)[1].strip(): l.split(None, 1)[0] for l in a if l.strip()}
            ob = {l.split(None, 1)[1].strip(): l.split(None, 1)[0] for l in b if l.strip()}
            removed = sorted(set(oa) - set(ob))
            added = sorted(set(ob) - set(oa))
            changed = sorted(k for k in oa if k in ob and oa[k] != ob[k])
            print(f"   entries base {len(oa)}, head {len(ob)}; removed {len(removed)}, "
                  f"added {len(added)}, hash changed {len(changed)}")
            for k in removed:
                print(f"   REMOVED {k}")
            for k in changed:
                print(f"   CHANGED {k}: {oa[k][:12]} -> {ob[k][:12]}")
            continue
        sm = difflib.SequenceMatcher(None, a, b, autojunk=False)
        for tag, i1, i2, j1, j2 in sm.get_opcodes():
            if tag == "equal":
                continue
            if tag == "insert":
                print(f"   insert at head {j1+1}-{j2} ({j2-j1} lines) after base {i1}")
                continue
            if tag == "delete":
                print(f"   DELETE base {i1+1}-{i2} ({i2-i1} lines)")
                for l in a[i1:i2]:
                    print(f"      - {l.rstrip()[:200]}")
                continue
            # replace
            print(f"   replace base {i1+1}-{i2} -> head {j1+1}-{j2}")
            if i2 - i1 == j2 - j1:
                for k in range(i2 - i1):
                    ol, nl = a[i1 + k].rstrip("\n"), b[j1 + k].rstrip("\n")
                    sub = is_subseq(ol, nl)
                    print(f"      base {i1+k+1} / head {j1+k+1}: insert-only={sub}")
                    for op in inserted(ol, nl):
                        if op[0] == "+":
                            print(f"         + at col {op[1]}: {op[2]!r}")
                        else:
                            print(f"         {op[0]} at col {op[1]}: {op[2]!r} -> {op[3]!r}")
            else:
                # multi-line replace: check old block is a subsequence of new block
                ob = "".join(a[i1:i2])
                nb = "".join(b[j1:j2])
                print(f"      block insert-only={is_subseq(ob, nb)}")
                for op in inserted(ob, nb):
                    if op[0] == "+":
                        print(f"         + at char {op[1]}: {op[2][:300]!r}")
                    else:
                        print(f"         {op[0]} at char {op[1]}: {op[2][:200]!r} -> {op[3][:200]!r}")


if __name__ == "__main__":
    main()
