"""Checks the appended `taildecides` targeted vectors (K3's generator) against
RV12's own oracle, and classifies each one: does the (1 - f) tail decide it?"""
import sys
sys.path.insert(0, sys.argv[1])
import rv12_oracle as o

def parse(tok, L):
    if tok in ("Z+", "Z-"):
        return (tok == "Z-", 0, 0)
    neg = tok[0] == "-"
    h, e = tok[1:].split("p")
    n = int(h.ljust(16 * L, "0"), 16)
    return o.norm((neg, n, int(e) - (64 * L - 1)))

total = bad = decided = 0
for L in (4, 8, 16):
    lines = [l for l in open("%s/new_l%d.txt" % (sys.argv[2], L)) if l.startswith("taildecides")]
    for line in lines:
        cls, op, p, a, b, want = line.split()
        p = int(p)
        A, B = parse(a, L), parse(b, L)
        if op == "sub":
            B = o.neg_of(B)
        got = o.op_add(A, B, p, L)
        exp = parse(want, L)
        ok = o.norm(got) == o.norm(exp)
        total += 1
        bad += not ok
        # RV5's semantics: (big - floor(small at the window bottom) - 1) rounded
        # with no sticky. Does it differ from the correct result?
        big, small = (A, B) if o.lead(A) >= o.lead(B) else (B, A)
        u = o.lead(big) - (64 * L - 1) - 64 * L
        if big[0] != small[0] and o.lead(big) - o.lead(small) > 64 * L:
            bw = big[1] << (big[2] - u)
            sw = small[1] >> (u - small[2]) if small[2] < u else small[1] << (small[2] - u)
            mut = o.rnd(big[0], bw - sw - 1, u, False, p)
            if o.norm(mut) != o.norm(got):
                decided += 1
    print("L=%d vectors %d" % (L, len(lines)))
print("taildecides vectors", total, "oracle mismatches", bad, "where RV5 (tail dropped) gives a different result", decided)
