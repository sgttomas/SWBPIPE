#!/bin/bash
# RV7 cargo slot: targeted tests from git archive 80290ce98 only.
set -u
S=<scratch>/rv7
TGT=<wt>/rv7-target
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0
LOG=$S/logs; mkdir -p $LOG
C=$S/src/projects/chirality-piping/core
guard() { free=$(df --output=avail -k <home> | tail -1); echo "disk before $1: ${free} kB" >> $LOG/SUMMARY.txt; if [ "$free" -lt 8912896 ]; then echo "ABORT low disk before $1" >> $LOG/SUMMARY.txt; exit 3; fi; }
df -h <home> | tail -1 > $LOG/disk_start.txt
# 1. candidate FK k2a tests + RV7's rotated-member probe (baseline copy)
guard fk_base; CARGO_TARGET_DIR=$TGT/fk_base cargo test --offline --locked --manifest-path $S/fk_base/Cargo.toml --test k2a_checked_formation --test rv7_rotated_m03 -- --nocapture > $LOG/fk_base.log 2>&1; echo "fk_base exit $?" >> $LOG/SUMMARY.txt
# 2. diagnostics k2a unit test
guard diag; CARGO_TARGET_DIR=$TGT/diag cargo test --offline --locked --manifest-path $C/solver/diagnostics/Cargo.toml k2a > $LOG/diag.log 2>&1; echo "diag exit $?" >> $LOG/SUMMARY.txt
# 3. mutants on the FK k2a kernel tests
for m in $(cd $S && python3 -c "import mutants;print(' '.join(mutants.M))"); do
  rm -rf $S/fk_mut; cp -r $S/fk_base $S/fk_mut; rm -f $S/fk_mut/tests/rv7_rotated_m03.rs
  (cd $S && python3 mutants.py $m fk_base/src/lib.rs fk_mut/src/lib.rs) > $LOG/mut_$m.log 2>&1
  CARGO_TARGET_DIR=$TGT/fk_mut cargo test --offline --locked --manifest-path $S/fk_mut/Cargo.toml --test k2a_checked_formation --no-fail-fast >> $LOG/mut_$m.log 2>&1
  echo "mutant $m exit $? | $(grep -h 'test result' $LOG/mut_$m.log | tail -1) | failed: $(grep -o '^    k2a_[a-z0-9_]*' $LOG/mut_$m.log | tr -d ' ' | tr '\n' ' ')" >> $LOG/SUMMARY.txt
done
# 4. PP product-reach tests (candidate)
guard pp; CARGO_TARGET_DIR=$TGT/pp cargo test --offline --locked --manifest-path $C/product_physics/Cargo.toml --test k2a_formation_range_runtime --test rv7_skew_runtime -- --nocapture > $LOG/pp.log 2>&1; echo "pp exit $?" >> $LOG/SUMMARY.txt
df -h <home> | tail -1 >> $LOG/disk_mid.txt; du -sh $TGT >> $LOG/disk_mid.txt
rm -rf $TGT/pp
# 5. main 5ae22926e (git archive), RV7's skew product probe only, captured entry, both modes
CM=$S/src_main/projects/chirality-piping/core
guard pp_main; CARGO_TARGET_DIR=$TGT/pp_main cargo test --offline --locked --manifest-path $CM/product_physics/Cargo.toml --test rv7_skew_runtime -- --nocapture > $LOG/pp_main_skew.log 2>&1; echo "pp_main_skew exit $?" >> $LOG/SUMMARY.txt
df -h <home> | tail -1 > $LOG/disk_end.txt
du -sh $TGT >> $LOG/disk_end.txt
