#!/usr/bin/env python3
"""RV2: a tighter first-order bound for included_angle, by the reviewer.

First-order relative error of phi_hat for exact (s, c), with every operation
|delta| <= u:
  eps = kappa(T) eta0 + sum_j kappa(t_j) eta_j + sigma(t_k)
  kappa(t)  = t / ((1 + t^2) atan t)            (sensitivity of atan, <= 1)
  eta0     <= 2u                                  (1 + c, then s/(1 + c))
  eta_j    <= (2 + w(t)(1.5 + 0.5 t^2/(1+t^2))) u, t = t_{j-1},
              w(t) = sqrt(1+t^2)/(1+sqrt(1+t^2))  (t^2, 1+a, sqrt, 1+r, t/d)
  sigma    <= (1 + 0.5 t/atan t + 42 t^2/(3(1-t^2)) t/atan t) u, t = t_k
The paper proof bounds kappa by 1 and eta_j by 4u; this keeps both exact.
Scans theta0 = atan T over (0, pi/2) on a fine grid plus the step boundaries.
"""
import math


def kappa(t):
    return t / ((1 + t * t) * math.atan(t)) if t > 0 else 1.0


def w(t):
    r = math.sqrt(1 + t * t)
    return r / (1 + r)


def bound(theta0):
    T = math.tan(theta0)
    b = 2 * kappa(T)
    t = T
    k = 0
    while t >= 0.05:
        eta = 2 + w(t) * (1.5 + 0.5 * t * t / (1 + t * t))
        t = math.tan(theta0 / 2 ** (k + 1))
        b += kappa(t) * eta
        k += 1
    s = 1 + 0.5 * t / math.atan(t) + 42 * t * t / (3 * (1 - t * t)) * t / math.atan(t)
    return b + s, k


best = (0, None, None)
N = 400000
pts = [math.pi / 2 * (i + 0.5) / N for i in range(N)]
# step boundaries: theta0 = 2^k atan(1/20) (just above)
for k in range(0, 6):
    th = math.atan(0.05) * 2 ** k
    if th < math.pi / 2:
        pts += [th * (1 + 1e-12), th * (1 - 1e-12)]
pts.append(math.pi / 2 * (1 - 1e-15))
for th in pts:
    b, k = bound(th)
    if b > best[0]:
        best = (b, th, k)
print("first-order bound max %.4f u at theta0 = %.9f (phi = %.9f), k = %d" % (best[0], best[1], 2 * best[1], best[2]))
# the paper proof's first-order value for comparison
print("paper first-order: 2 + 4*5 + 1.5357 = %.4f u" % (2 + 20 + 1 + 0.5 / 0.99916 + 42 * 8.354e-4 / 0.99916))
# per-k maxima
for kk in range(6):
    m = max((bound(th)[0], th) for th in pts if bound(th)[1] == kk)
    print("k = %d: max %.4f u at phi = %.6f" % (kk, m[0], 2 * m[1]))
# in ulp of phi_hat: |err| <= eps u phi and ulp = 2^(e-p+1) with 2^e <= phi < 2^(e+1),
# so err/ulp <= eps * phi / 2^(e+1)
bu = (0, None)
for th in pts:
    phi = 2 * th
    e = math.floor(math.log2(phi))
    v = bound(th)[0] * phi / 2 ** (e + 1)
    if v > bu[0]:
        bu = (v, phi)
print("first-order bound in ulp: max %.4f ulp at phi = %.9f" % bu)
