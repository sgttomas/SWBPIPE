import re, subprocess, sys
REV = sys.argv[1]
root = '<wt>/k2a'
files = subprocess.run(['git','-C',root,'ls-tree','-r','--name-only',REV,'projects/chirality-piping/core','projects/chirality-piping/validation','projects/chirality-piping/apps'],capture_output=True,text=True).stdout.split()
files = [f for f in files if f.endswith('.rs')]
def strip(src):
    out=[];i=0;n=len(src)
    while i<n:
        c=src[i]
        if src.startswith('//',i):
            j=src.find('\n',i); j=n if j<0 else j; out.append(' '*(j-i)); i=j; continue
        if src.startswith('/*',i):
            depth=1; j=i+2
            while j<n and depth:
                if src.startswith('/*',j): depth+=1; j+=2
                elif src.startswith('*/',j): depth-=1; j+=2
                else: j+=1
            out.append(''.join('\n' if ch=='\n' else ' ' for ch in src[i:j])); i=j; continue
        m=re.match(r'b?r(#*)"',src[i:i+10])
        if m and (i==0 or not (src[i-1].isalnum() or src[i-1]=='_')):
            hashes=m.group(1); start=i+len(m.group(0)); end=src.find('"'+hashes,start)
            end = n if end<0 else end+1+len(hashes)
            out.append(''.join('\n' if ch=='\n' else ' ' for ch in src[i:end])); i=end; continue
        if c=='"' :
            j=i+1
            while j<n and src[j]!='"':
                j+= 2 if src[j]=='\\' else 1
            j+=1; out.append(''.join('\n' if ch=='\n' else ' ' for ch in src[i:j])); i=j; continue
        if c=="'":
            m=re.match(r"'(\\.|[^\\'])'|'\\u\{[0-9a-fA-F]+\}'|'\\x..'",src[i:i+12])
            if m: out.append(' '*len(m.group(0))); i+=len(m.group(0)); continue
        out.append(c); i+=1
    return ''.join(out)
tot=0
for f in files:
    src=subprocess.run(['git','-C',root,'show',f'{REV}:{f}'],capture_output=True,text=True).stdout
    if 'FrameKernelError' not in src and 'Self::' not in src: continue
    s=strip(src)
    lines=s.split('\n')
    for k,l in enumerate(lines,1):
        # pattern-position occurrences: FrameKernelError::Variant followed by optional {..}/(..) then => or |
        for m in re.finditer(r'FrameKernelError::(\w+)',l):
            rest=l[m.end():]
            ctx=' '.join(lines[k-1:k+3])
            after=ctx[ctx.find(m.group(0))+len(m.group(0)):]
            # remove a balanced {...} or (...) head
            a=after.lstrip()
            if a[:1] in '{(':
                o=a[0]; c={'{':'}','(':')'}[o]; d=0
                for idx,ch in enumerate(a):
                    if ch==o: d+=1
                    elif ch==c:
                        d-=1
                        if d==0: a=a[idx+1:]; break
            a=a.lstrip()
            pat = a.startswith('=>') or a.startswith('|') or a.startswith('if ')
            if pat or 'matches!' in l or re.search(r'\bif let\b|\blet\b.*=\s*$',l):
                tot+=1
                print(f'{f}:{k}: {"PATTERN" if pat else "other"}: {l.strip()[:140]}')
print('total',tot)
