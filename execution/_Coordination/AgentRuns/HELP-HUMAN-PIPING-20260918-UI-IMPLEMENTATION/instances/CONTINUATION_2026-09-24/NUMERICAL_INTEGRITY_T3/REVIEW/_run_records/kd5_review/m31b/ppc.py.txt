import math
def pp_centre(xi, xj, radius, yref):
    ch = [xj[k]-xi[k] for k in range(3)]
    cl = math.sqrt(ch[0]*ch[0]+ch[1]*ch[1]+ch[2]*ch[2])
    h = 0.5*cl
    cu = [ch[k]/cl for k in range(3)]
    ax = yref[0]*cu[0]+yref[1]*cu[1]+yref[2]*cu[2]
    pn = [yref[k]-ax*cu[k] for k in range(3)]
    pm = math.sqrt(pn[0]*pn[0]+pn[1]*pn[1]+pn[2]*pn[2])
    s = math.sqrt(radius*radius-h*h)
    return [0.5*(xi[k]+xj[k]) - s*pn[k]/pm for k in range(3)]
