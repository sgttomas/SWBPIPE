#!/bin/bash
# RV20-1's proposed assertion: in the shared-cap test, every round charges the set's 16-limb
# context at least as much as K4's references (every row K4's finish evaluates, up to its first
# refusal, is evaluated once by the bounded tracker, at a collapse or at finish). NONE must pass; RV20-M4 (the shared cap's collapse uncharged) must fail.
set -u
WT=<wt>
OUT=$WT/rv20/records/fixcheck.log
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2
: > $OUT
for variant in NONE RV20-M4; do
  D=$WT/rv20/fix-$variant
  rm -rf $D; mkdir -p $D
  git -C $WT/kf1 archive 1854911d16d73a4c66b3e158e26de60febe71dc6 projects/chirality-piping/core/solver/frame_kernel | tar -x -C $D
  FK=$D/projects/chirality-piping/core/solver/frame_kernel
  python3 -B - "$FK" "$variant" <<'PY'
import sys
fk, variant = sys.argv[1], sys.argv[2]
t = fk + "/tests/retained_k4/kf1_tracker_tests.rs"
s = open(t).read()
old = "        // Interleave the trackers' rows.\n"
assert s.count(old) == 1
s = s.replace(old, "        let refusing_round = rows.iter().any(|(_, (_, den))| is_refusing(den));\n" + old)
old = "        assert!(references.is_empty());\n"
assert s.count(old) == 1
s = s.replace(old, old + "        let _ = refusing_round;\n        assert!(lme(&c16s) >= lme(&c16r), \"round {round}: the set's work {} < K4's {}\", lme(&c16s), lme(&c16r));\n")
open(t, "w").write(s)
if variant == "RV20-M4":
    a = fk + "/src/structural/retained/adaptive.rs"
    s = open(a).read()
    old = "                    t.collapse(ctx16);"
    assert s.count(old) == 1
    s = s.replace(old, '                    t.collapse(&mut WideContext::<16>::new(1024).expect("p"));')
    open(a, "w").write(s)
print("edited", variant)
PY
  echo "== $variant" >> $OUT
  cd $FK
  CARGO_TARGET_DIR=$WT/rv20-target/fix-$variant cargo test --offline --locked -j 4 --lib kf1_the_shared_cap > $WT/rv20/fix.tmp 2>&1
  rc=$?
  cd $WT
  grep -E "^test |test result|panicked|round [0-9]+:" $WT/rv20/fix.tmp | sed "s|$WT|<wt>|g" >> $OUT
  echo "exit: $rc" >> $OUT
  rm -rf $D $WT/rv20-target/fix-$variant
done
