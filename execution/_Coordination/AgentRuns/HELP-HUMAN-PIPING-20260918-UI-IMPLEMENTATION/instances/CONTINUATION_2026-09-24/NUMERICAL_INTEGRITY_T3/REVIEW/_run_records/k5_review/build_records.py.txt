#!/usr/bin/env python3
"""RV14: assemble REVIEW/_run_records/k5_review/ from <wt>/scratch/rv14, replacing
machine paths with placeholders, and write its SHA256SUMS. Standard library only."""
import hashlib, os, re, shutil, sys

WT = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", ".."))
HOME = os.path.expanduser("~")
S = os.path.join(WT, "scratch", "rv14")
T3 = ("projects/chirality-piping/execution/_Coordination/AgentRuns/HELP-HUMAN-PIPING-20260918-"
      "UI-IMPLEMENTATION/instances/CONTINUATION_2026-09-24/NUMERICAL_INTEGRITY_T3")
OUT = os.path.join(WT, "numerics", T3, "REVIEW", "_run_records", "k5_review")
VENV = os.path.normpath(os.path.join(WT, "..", "..", "projects", "chirality-piping", ".venv"))
REPO = os.path.normpath(os.path.join(WT, "..", ".."))

SUBS = [
    (VENV, "<VENV>"),
    (WT, "<wt>"),
    (REPO, "<repo>"),
    (HOME, "<home>"),
]
TMP_RE = re.compile(r"<tmp>'\"`)<>]*|<tmp>'\"`)<>]*|<tmp>'\"`)<>]*")


def clean(text):
    for old, new in SUBS:
        text = text.replace(old, new)
    text = TMP_RE.sub("<tmp>", text)
    # the records-check line quoted its own scan pattern; describe it instead
    text = re.sub(r"\(/usr/bin/grep -E '[^\n]*registry'\):",
                  "(home, private, tmp and var/folders roots, the worktree marker, a Windows user root, "
                  "Homebrew, rustup and the cargo registry):", text)
    return text


def put(src, rel, transform=clean):
    dst = os.path.join(OUT, rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    data = open(src, encoding="utf-8").read()
    open(dst, "w", encoding="utf-8").write(transform(data))


def main():
    if os.path.isdir(OUT):
        shutil.rmtree(OUT)
    os.makedirs(OUT)
    # checks
    for f in ["revisions.txt", "records_checks.txt", "gen8.txt", "libm_scan.txt",
              "product_runs_reclass.txt", "memguard.txt", "toolchain.txt"]:
        put(os.path.join(S, "checks", f), f)
    # oracle
    for f in ["rv14_oracle.py", "rv14_permute.py", "rv14_scale.py", "rv14_huge.py"]:
        put(os.path.join(S, "oracle", f), f"oracle/{f}.txt")
    for f in ["cases_all.txt", "results_all.txt", "report_all.txt", "cases_huge.txt",
              "results_huge.txt", "report_huge.txt", "cases_rv14_4.txt", "results_rv14_4.txt",
              "report_rv14_4.txt", "comparisons.txt", "case_m0438.txt", "m0438_sv.txt"]:
        put(os.path.join(S, "oracle", f), f"oracle/{f}")
    # probes
    for f in ["rv14_fk_probe.rs", "rv14_fk_probe_sv.rs", "rv14_sa_probe.rs",
              "rv14_sa_mutprobe.rs", "rv14_pp_probe.rs", "run_pp.sh"]:
        put(os.path.join(S, "probe", f), f"probes/{f}.txt")
    for f in ["sa_probe_base.txt", "sa_probe_cand.txt", "sa_mutprobe_cand.txt", "pp_runs_base.txt",
              "pp_runs_cand.txt", "pp_run.log", "probe_comparisons.txt"]:
        put(os.path.join(S, "probe", f), f"probes/{f}")
    # mutations
    M = os.path.join(S, "mut")
    for f in ["rv14_mutate.py", "rv14_run_mutant.sh", "batch.sh"]:
        put(os.path.join(M, f), f"mutations/{f}.txt")
    put(os.path.join(M, "mutate_k1.py"), "mutations/mutate_k1.py.txt")
    for f in ["queue.txt", "batch.log", "MUTANTS.txt"]:
        put(os.path.join(M, f), f"mutations/{f}")
    for d in sorted(os.listdir(os.path.join(M, "logs"))):
        for f in sorted(os.listdir(os.path.join(M, "logs", d))):
            src = os.path.join(M, "logs", d, f)
            if f.endswith("_results.txt"):
                # large probe outputs: keep a digest and the lines the review cites
                data = open(src, encoding="utf-8").read()
                digest = hashlib.sha256(data.encode()).hexdigest()
                keep = [l[:400] for l in data.split("\n")
                        if l.startswith(("T1_cycle_band_", "M0809|"))
                        or (l.startswith(("P1_", "P2_")) and "b=0|selected=true" in l)]
                text = (f"excerpt of {f} (full file sha256 {digest}, {data.count(chr(10))} lines; "
                        f"the full output is reproducible from the probe and corpus)\n" + "\n".join(keep) + "\n")
                dst = os.path.join(OUT, "mutations", "logs", d, f.replace(".txt", "_excerpt.txt"))
                os.makedirs(os.path.dirname(dst), exist_ok=True)
                open(dst, "w", encoding="utf-8").write(clean(text))
            else:
                put(src, f"mutations/logs/{d}/{f}")
    put(os.path.join(S, "README_records.txt"), "README.txt")
    put(__file__, "build_records.py.txt")
    # SHA256SUMS
    entries = []
    for root, _, files in os.walk(OUT):
        for f in files:
            if f == "SHA256SUMS":
                continue
            p = os.path.join(root, f)
            rel = "./" + os.path.relpath(p, OUT)
            entries.append((rel, hashlib.sha256(open(p, "rb").read()).hexdigest()))
    entries.sort()
    with open(os.path.join(OUT, "SHA256SUMS"), "w") as out:
        for rel, h in entries:
            out.write(f"{h}  {rel}\n")
    # self-check: no machine path survives
    bad = re.compile(r"(?<![A-Za-z0-9_.{-])(?:/(?:Users|private|home|tmp|var/folders)/[^\s)\"'`<>]+"
                     r"|[A-Za-z]:\\Users\\[A-Za-z0-9._-]+[^\s`'\"<>)\]]*)")
    hits = []
    for rel, _ in entries:
        text = open(os.path.join(OUT, rel), encoding="utf-8").read()
        for m in bad.finditer(text):
            hits.append((rel, m.group(0)))
    print(f"files={len(entries)} machine-path hits={len(hits)}")
    for h in hits[:20]:
        print("  HIT", h)
    return 1 if hits else 0


if __name__ == "__main__":
    sys.exit(main())
