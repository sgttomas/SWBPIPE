#!/usr/bin/env python3
"""RV11's own mutants (independent review of K2b). Usage: python3 mutate_rv11.py <core-dir> <id>.
Exact text replacements; each must match the stated number of times (once unless noted).
- RV11-REACT-TERMS-UNSCALED: `force_scaled_reactions` sums K'u at 2^b with the ledger terms
  left UNSCALED (a restrained DOF's load term then enters at the wrong scale).
- RV11-CENSUS-PRODUCT-X: the census records a load product x*y at e(x) only, not e(x)+e(y).
- RV11-CENSUS-NO-CURVED: the orchestrator's census omits the realized curved-slot matrices.
- RV11-FIDELITY-AT-SCALE: `unscale_structural_solution` leaves the load-fidelity rows'
  exact_net_bits and actual_bits at 2^b (no recompute from the unscaled ledger)."""
import sys

core, mid = sys.argv[1], sys.argv[2]
LIB = core + "/solver/frame_kernel/src/lib.rs"
ST = core + "/solver/frame_kernel/src/structural.rs"
SP = core + "/solver/frame_kernel/src/structural/sparse.rs"
SA = core + "/solver/nonlinear_integration/src/structural_adapter.rs"

MUTANTS = {
    "RV11-REACT-TERMS-UNSCALED": [(SP, "        let scaled = if force_scale.is_unscaled() {\n            force\n",
                                   "        let scaled = if true {\n            force\n")],
    "RV11-CENSUS-PRODUCT-X": [(LIB, "                self.exponent(structural::binary_exponent(x) + structural::binary_exponent(y));",
                               "                self.exponent(structural::binary_exponent(x));")],
    "RV11-CENSUS-NO-CURVED": [(SA, "        census.matrix(&slot.global_stiffness);", "        let _ = slot;")],
    "RV11-FIDELITY-AT-SCALE": [(ST, "        if let Some(fidelity) = solution.load_fidelity.as_mut() {",
                                "        if let Some(fidelity) = solution.load_fidelity.as_mut().filter(|_| false) {")],
}
for path, old, new, *rest in MUTANTS[mid]:
    count = rest[0] if rest else 1
    text = open(path).read()
    assert text.count(old) == count, (mid, path, text.count(old))
    open(path, "w").write(text.replace(old, new))
print("applied", mid)
