"""RV17: NI kill sites of the NI pins' original mutants (and F1b's pin mutants) on c4879c496,
against I13's record at e215c6007 (kill_sites.txt, kill_sites_f1b.txt; NI crate only)."""
import re
def i13(path):
    sites, cur = {}, None
    for line in open(path):
        m = re.match(r'## (\S+)', line)
        if m:
            cur = m.group(1); sites.setdefault(cur, set()); continue
        m = re.match(r'\s+solver/nonlinear_integration: (\S+)', line)
        if m and cur:
            sites[cur].add(m.group(1))
    return sites
old = i13('i13_kill_sites.txt'); old.update(i13('i13_kill_sites_f1b.txt'))
mine = {}
for line in open('MUTANTS.txt'):
    m = re.match(r'H-(\S+) exit:.* failed: (.*?)  panic_sites', line)
    if m:
        mine[m.group(1)] = set(m.group(2).split())
ids = ["K2B-PIN-LOOP", "K2B-PIN-PLUMBING", "K2B-PIN-THIRD", "KD5-M32a", "KD5-M32b", "KD5-E4",
       "S11K-RV-OPT1", "S11K-RV-OPT3", "S11K-RV-OPT4", "S11K-RV-PUB", "K1-PIN-BINARY64",
       "K1-PIN-BINARY64-B", "K1-PIN-THIRD", "K1-PIN-LOOP", "K1-PIN-LOOP-B",
       "F1B-PIN-SECOND", "F1B-PIN-HELPER", "F1B-PIN-NEVER"]
for i in ids:
    a, b = old.get(i, set()), {t for t in mine.get(i, set()) if not t.startswith('s11g_tests::t13')}
    lost, added = sorted(a - b), sorted(b - a)
    print(f"{i}: I13 {len(a)} NI sites, RV17 {len(b)}, lost {len(lost)}, added {len(added)}, {'KILLED' if b else 'SURVIVED'}")
    for t in lost: print('    lost', t)
    for t in added: print('    added', t)
