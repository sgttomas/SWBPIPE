#!/bin/bash
# RV5 delta: product_physics integration test under a variant. Usage: run_pp_variant.sh <variant|base> <test-target>
source "$(dirname "$0")/env.sh"
V=$1; T=$2
C=$S/ppcopy; rm -rf $C; mkdir -p $C/projects/chirality-piping
tar -C $S/cand2/projects/chirality-piping --exclude=target -cf - core fixtures schemas | tar -C $C/projects/chirality-piping -xf - --touch
if [ "$V" != base ]; then python3 $S/rv5_mutate.py $C/projects/chirality-piping/core $V > $S/logs/pp_$V.$T.log 2>&1 || { echo PATCH FAILED; exit 9; }; else : > $S/logs/pp_$V.$T.log; fi
waitfree
(cd $C/projects/chirality-piping/core/product_physics && cargo test --offline --locked --test $T -- --test-threads=1 >> $S/logs/pp_$V.$T.log 2>&1); rc=$?
echo "pp $V test=$T exit=$rc failed: $(grep -E '^    [a-z0-9_:]+$' $S/logs/pp_$V.$T.log | tr '\n' ' ') result: $(grep '^test result' $S/logs/pp_$V.$T.log | tr '\n' ' ')" | tee -a $S/logs/SUMMARY2.txt
