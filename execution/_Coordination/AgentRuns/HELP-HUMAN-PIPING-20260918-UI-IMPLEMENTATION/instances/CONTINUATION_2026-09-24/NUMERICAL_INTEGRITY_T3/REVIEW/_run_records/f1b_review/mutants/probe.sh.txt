#!/bin/bash
# RV17: run the scratch probe tests in a kept tree (after its suite ran). Usage: probe.sh <run-id> <test-name> [filter]
W=<wt>; R=$W/scratch/rv17
m=$1; t=$2; f=${3:-}
CORE=$W/rv17/$m/tree/projects/chirality-piping/core
cp $R/probe/$t.rs $CORE/product_physics/tests/$t.rs
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=$W/rv17-target/$m RUST_TEST_THREADS=1
export RV17_OUT=$R/probe/out/$m
cd $CORE/product_physics && cargo test --offline --locked -j 6 --test $t $f -- --nocapture > $R/runs/logs/probe_${m}_$t.log 2>&1
echo "probe $m $t rc=$?" >> $R/runs/logs/probe_${m}_$t.log
