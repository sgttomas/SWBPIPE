static mut MISMATCHES: usize = 0;
use std::collections::BTreeMap;
use std::sync::Mutex;
static STATS: Mutex<BTreeMap<String, usize>> = Mutex::new(BTreeMap::new());
fn stat(key: impl Into<String>) {
    *STATS.lock().unwrap().entry(key.into()).or_insert(0) += 1;
}
fn class<T: std::fmt::Debug>(x: &T) -> String {
    let t = format!("{x:?}");
    if t.starts_with("Ok(StructuralSolution") {
        let q = t.find("quality: ").map(|i| t[i + 9..].split(|c: char| !c.is_alphanumeric()).next().unwrap().to_string()).unwrap_or_default();
        let fc = if t.contains("formation_check: Some") { "+fc" } else { "" };
        let lf = if t.contains("load_fidelity: Some") { "+lf" } else { "" };
        format!("Ok {q}{fc}{lf}")
    } else {
        t.split(|c: char| c == '{' || c == '(' ).take(2).collect::<Vec<_>>().join("(")
    }
}
static mut CHECKS: usize = 0;

fn check(case: &str, what: &str, a: &str, b: &str) {
    unsafe {
        CHECKS += 1;
    }
    if a != b {
        unsafe {
            MISMATCHES += 1;
        }
        println!("MISMATCH {case} {what}\n  dense : {}\n  sparse: {}", &a[..a.len().min(3000)], &b[..b.len().min(3000)]);
    }
}

struct Lcg(u64);
impl Lcg {
    fn next(&mut self) -> u64 {
        self.0 = self.0.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        self.0 >> 11
    }
    fn unit(&mut self) -> f64 {
        (self.next() as f64) / ((1u64 << 53) as f64)
    }
    fn range(&mut self, a: f64, b: f64) -> f64 {
        a + (b - a) * self.unit()
    }
    fn below(&mut self, n: usize) -> usize {
        (self.next() % n as u64) as usize
    }
}

#[derive(Clone)]
struct Model {
    name: String,
    nodes: Vec<[f64; 3]>,
    frames: Vec<FrameElement>,
    users: Vec<UserStiffnessElement>,
    macros: Vec<CurvedBendMacroElement>,
    slots: Vec<CurvedBendStiffnessElement>,
    springs: Vec<(usize, f64)>,
    prescribed: Vec<(usize, f64)>,
    loads: Vec<(usize, f64)>,
}

fn fnode(i: usize, p: [f64; 3]) -> FrameNode {
    FrameNode::new(i, p).unwrap()
}

impl Model {
    fn n(&self) -> usize {
        6 * self.nodes.len()
    }
    fn free(&self) -> Vec<usize> {
        (0..self.n()).filter(|d| !self.prescribed.iter().any(|p| p.0 == *d)).collect()
    }
    fn blocks(&self) -> Vec<StiffnessBlock> {
        self.slots.iter().map(|s| StiffnessBlock { node_i: s.node_i, node_j: s.node_j, stiffness: s.global_stiffness }).collect()
    }
    /// The product's dense order.
    fn dense(&self) -> Result<Vec<Vec<f64>>, String> {
        let mut k = assemble_global_stiffness_with_user_elements(self.nodes.len(), &self.frames, &self.users).map_err(|e| format!("{e:?}"))?;
        for s in &self.slots {
            let map = element_dof_map(s.node_i, s.node_j);
            for (r, &row) in map.iter().enumerate() {
                for (c, &col) in map.iter().enumerate() {
                    k[row][col] += s.global_stiffness[r][c];
                }
            }
        }
        for &(dof, v) in &self.springs {
            k[dof][dof] += v;
        }
        Ok(k)
    }
    fn sparse(&self) -> Result<SparseStiffness, String> {
        assemble_sparse_stiffness(self.nodes.len(), &self.frames, &self.users, &self.blocks(), &self.springs, &SparseAssemblyOptions::new()).map_err(|e| format!("{e:?}"))
    }
    fn ledger(&self, cancel: Option<f64>) -> AssembledForce {
        let mut l = LoadLedger::new();
        for (i, &(d, v)) in self.loads.iter().enumerate() {
            match cancel {
                None => l.push(format!("load:{i}"), d, v),
                Some(g) => {
                    l.push(format!("big:{i}"), d, g);
                    l.push(format!("load:{i}"), d, v);
                    l.push(format!("back:{i}"), d, -g);
                }
            }
        }
        l.finish(self.n()).unwrap()
    }
}

fn bits2(k: &[Vec<f64>]) -> String {
    format!("{:?}", k.iter().map(|r| r.iter().map(|v| v.to_bits()).collect::<Vec<_>>()).collect::<Vec<_>>())
}

fn section(r: &mut Lcg, e: f64) -> FrameSection {
    let od = r.range(0.05, 0.6);
    let wall = od * r.range(0.02, 0.2);
    let id = od - 2.0 * wall;
    let a = std::f64::consts::PI * (od * od - id * id) / 4.0;
    let i = std::f64::consts::PI * (od.powi(4) - id.powi(4)) / 64.0;
    FrameSection::new(e, e / r.range(2.4, 2.8), a, i, i * r.range(0.9, 1.1), 2.0 * i).unwrap()
}

fn yref_for(a: [f64; 3], b: [f64; 3]) -> [f64; 3] {
    let d = [b[0] - a[0], b[1] - a[1], b[2] - a[2]];
    let l = (d[0] * d[0] + d[1] * d[1] + d[2] * d[2]).sqrt();
    if (d[2] / l).abs() < 0.9 { [0.0, 0.0, 1.0] } else { [1.0, 0.0, 0.0] }
}

/// A random product-shaped model: skew frames with many members per node,
/// duplicate connectivity, users, explicit blocks (from frames and negated
/// frames), traced curved bends, springs on free and prescribed DOFs (repeated,
/// zero, -0.0), prescribed motion (zero, -0.0 and nonzero), and loads.
fn random_model(seed: u64, e: f64) -> Model {
    let mut r = Lcg(seed * 7919 + 17);
    let nn = 3 + r.below(9);
    let axis = seed % 3 == 0;
    let nodes: Vec<[f64; 3]> = (0..nn)
        .map(|k| {
            if axis {
                // axis-aligned grid (explicit zeros)
                [(k % 3) as f64 * 1.5, ((k / 3) % 3) as f64 * 2.0, (k / 9) as f64 * 1.25]
            } else {
                [r.range(-4.0, 4.0), r.range(-4.0, 4.0), r.range(-4.0, 4.0)]
            }
        })
        .collect();
    let mut m = Model { name: format!("rand{seed}-E{e}"), nodes: nodes.clone(), frames: vec![], users: vec![], macros: vec![], slots: vec![], springs: vec![], prescribed: vec![], loads: vec![] };
    let s = section(&mut r, e);
    // spanning chain + extra members (star/duplicates)
    let mut pairs: Vec<(usize, usize)> = (1..nn).map(|k| (r.below(k), k)).collect();
    let extra = r.below(2 * nn + 1);
    for _ in 0..extra {
        let a = r.below(nn);
        let b = r.below(nn);
        if a != b {
            pairs.push((a, b));
        }
    }
    if nn > 3 {
        // a star: node 0 joined to 4+ others
        for b in 1..nn.min(6) {
            pairs.push((0, b));
        }
        // exact duplicate and reversed duplicate
        pairs.push(pairs[0]);
        pairs.push((pairs[1].1, pairs[1].0));
    }
    for &(a, b) in &pairs {
        let kind = r.below(10);
        let (pa, pb) = (nodes[a], nodes[b]);
        let y = yref_for(pa, pb);
        if kind < 6 {
            let sec = if r.below(2) == 0 { s } else { section(&mut r, e) };
            if let Ok(f) = FrameElement::new(fnode(a, pa), fnode(b, pb), sec, y) {
                m.frames.push(f);
            }
        } else if kind < 8 {
            if let Ok(u) = UserStiffnessElement::new(fnode(a, pa), fnode(b, pb), y, r.range(1e6, 1e9), r.range(1e4, 1e8), r.range(1e3, 1e7), r.range(1e3, 1e7)) {
                m.users.push(u);
            }
        } else {
            // explicit block from a frame; sometimes its exact negation (cancels
            // a duplicate frame to exact zeros) or with -0.0 entries
            if let Ok(f) = FrameElement::new(fnode(a, pa), fnode(b, pb), s, y) {
                let mut g: Matrix12 = f.global_stiffness().unwrap();
                match r.below(3) {
                    0 => {
                        m.frames.push(f);
                        for row in g.iter_mut() {
                            for v in row.iter_mut() {
                                *v = -*v;
                            }
                        }
                    }
                    1 => {
                        for row in g.iter_mut() {
                            for v in row.iter_mut() {
                                if *v == 0.0 {
                                    *v = -0.0;
                                }
                            }
                        }
                    }
                    _ => {}
                }
                if let Ok(slot) = CurvedBendStiffnessElement::new(format!("blk{a}-{b}"), a, b, g) {
                    m.slots.push(slot);
                }
            }
        }
    }
    // A traced planar bend (90 degrees, invented), when there is room: a new
    // pair of nodes.
    if r.below(2) == 0 {
        let base = m.nodes.len();
        let rad = r.range(0.3, 1.5);
        let c = [r.range(-2.0, 2.0), r.range(-2.0, 2.0), r.range(-2.0, 2.0)];
        let pi = [c[0] + rad, c[1], c[2]];
        let pj = [c[0], c[1] + rad, c[2]];
        m.nodes.push(pi);
        m.nodes.push(pj);
        if let Ok(bend) = CurvedBendMacroElement::new(fnode(base, pi), fnode(base + 1, pj), c, e, e / 2.6, s.area, s.second_moment_y, s.torsion_constant, r.range(1.0, 3.0), r.range(1.0, 3.0)) {
            if let Ok(slot) = CurvedBendStiffnessElement::from_macro_element(format!("bend{base}"), &bend) {
                m.slots.push(slot);
                m.macros.push(bend);
                // connect the bend to the rest
                let a = r.below(base);
                if let Ok(f) = FrameElement::new(fnode(a, m.nodes[a]), fnode(base, pi), s, yref_for(m.nodes[a], pi)) {
                    m.frames.push(f);
                }
            }
        }
    }
    let n = 6 * m.nodes.len();
    // boundary: node 0 fixed with a settlement or -0.0; one extra prescribed DOF
    for d in 0..6 {
        let v = match (seed + d as u64) % 4 {
            0 => 0.0,
            1 => -0.0,
            2 => r.range(-0.01, 0.01),
            _ => 0.0,
        };
        m.prescribed.push((d, v));
    }
    let extra_p = 6 + r.below(n - 6);
    m.prescribed.push((extra_p, if seed % 2 == 0 { r.range(-0.002, 0.002) } else { 0.0 }));
    if seed % 5 == 0 {
        m.prescribed.reverse();
    }
    // springs: on free DOFs, repeated, zero, -0.0, and on prescribed DOFs;
    // every DOF of a node with no element gets a spring so the free block is
    // not trivially singular
    let mut has_elem = vec![false; m.nodes.len()];
    for f in &m.frames {
        has_elem[f.node_i.index] = true;
        has_elem[f.node_j.index] = true;
    }
    for u in &m.users {
        has_elem[u.node_i.index] = true;
        has_elem[u.node_j.index] = true;
    }
    for s in &m.slots {
        has_elem[s.node_i] = true;
        has_elem[s.node_j] = true;
    }
    for (k, &h) in has_elem.iter().enumerate() {
        if !h {
            for d in 0..6 {
                m.springs.push((6 * k + d, r.range(1e3, 1e6)));
            }
        }
    }
    for _ in 0..(3 + r.below(8)) {
        let d = r.below(n);
        let v = match r.below(6) {
            0 => 0.0,
            1 => -0.0,
            _ => r.range(1.0, 1e7),
        };
        m.springs.push((d, v));
        if r.below(3) == 0 {
            m.springs.push((d, r.range(1.0, 1e5)));
        }
    }
    m.springs.push((m.prescribed[0].0, 4.5e4));
    // stabilizing springs on every free rotational DOF of weakly held nodes
    for k in 1..m.nodes.len() {
        for d in 0..6 {
            if r.below(4) == 0 {
                m.springs.push((6 * k + d, r.range(10.0, 1e4)));
            }
        }
    }
    for _ in 0..(2 + r.below(5)) {
        m.loads.push((r.below(n), r.range(-1e4, 1e4)));
    }
    m
}

/// Today's dense-derived sparse sequence (`solve_structural_sparse`): factor,
/// or the witness on a refused factor, then the completion.
fn today(p: &PreparedSystem<'_>) -> Result<StructuralSolution, StructuralError> {
    match sds::factor_structural_ldlt(p) {
        Ok(f) => st::finish_structural(&f),
        Err(error) => Err(st::negative_pair_witness(p)?.unwrap_or(error)),
    }
}

/// A K-D5 model as a product-shaped case (the adapter tests' construction).
fn kd5_model(m: &kd5::ModelData, e_scale: f64) -> Model {
    let node = |i: usize| FrameNode::new(i, m.nodes[i]).unwrap();
    let s = &m.section;
    let section = FrameSection::new(s.e * e_scale, s.g * e_scale, s.a, s.i, s.i, s.j).unwrap();
    let mut c = Model { name: format!("{}-E{e_scale}", m.name), nodes: m.nodes.to_vec(), frames: vec![], users: vec![], macros: vec![], slots: vec![], springs: m.springs.to_vec(), prescribed: m.rigid.iter().map(|&d| (d, 0.0)).collect(), loads: m.loads.iter().copied().filter(|l| l.1 != 0.0).collect() };
    for (index, member) in m.members.iter().enumerate() {
        match member.bend {
            None => c.frames.push(FrameElement::new(node(member.i), node(member.j), section, member.y_reference).unwrap()),
            Some((center, factor)) => {
                let e = CurvedBendMacroElement::new(node(member.i), node(member.j), center, s.e * e_scale, s.g * e_scale, s.a, s.i, s.j, factor, factor).unwrap();
                c.slots.push(CurvedBendStiffnessElement::from_macro_element(format!("bend-{index}"), &e).unwrap());
                c.macros.push(e);
            }
        }
    }
    c
}

/// Targeted cases: K-D5's models with each load split into three ledger
/// terms whose exact sum is not the rounded net (so the formation check's rho
/// from the terms differs from the one from the folded force), and an S-bend
/// (two realized bends sharing a node) with springs on the shared node.
fn targeted() {
    for m in kd5::all() {
        for e_scale in [1.0, 0.9] {
            let model = kd5_model(m, e_scale);
            let (d, s) = match assembly_parity(&model) {
                Some(x) => x,
                None => continue,
            };
            let ev = AssemblyEvidence::new(model.nodes.len(), &model.frames, &model.users, &model.slots, &model.springs).unwrap();
            let sev = SparseAssemblyEvidence::new(s.pattern(), model.nodes.len(), &model.frames, &model.users, &model.slots, &model.springs).unwrap();
            let mut l = LoadLedger::new();
            for (i, &(dof, v)) in model.loads.iter().enumerate() {
                let a = v * 0.1;
                let b = v * 0.7;
                let c = v - a - b;
                l.push(format!("a{i}"), dof, a);
                l.push(format!("b{i}"), dof, b);
                l.push(format!("c{i}"), dof, c);
            }
            let force = l.finish(model.n()).unwrap();
            let free = model.free();
            for mode in [LinearSolveMode::DenseScrutiny, LinearSolveMode::SparseInteractive] {
                let dres = ev.solve_assembled_with_formation_check(&d, &force, &free, &model.prescribed, mode, &model.macros, true);
                stat(format!("targeted kd5 split {mode:?}: {}", class(&dres)));
                check(&model.name, &format!("split-ledger fc {mode:?}"), &render(&dres), &render(&sev.solve_assembled_with_formation_check(&s, &force, &free, &model.prescribed, mode, &model.macros, true)));
            }
        }
    }
    // S-bend (invented): R = 0.6 m, bends (0 -> 1) about c1 and (1 -> 2)
    // about c2 in the XY plane, frames to anchored nodes 3 and 4, springs on
    // the shared node 1 (and two on one DOF), a load at node 1.
    let r = 0.6;
    let p = [[r, 0.0, 0.0], [0.0, r, 0.0], [-r, 2.0 * r, 0.0], [r, -2.0, 0.0], [-r, 2.0 * r + 2.0, 0.0]];
    let (e, g, a, i, j) = (2.0e11, 8.0e10, 0.005969026041820614, 2.700984283923829e-05, 5.401968567847658e-05);
    let sec = FrameSection::new(e, g, a, i, i, j).unwrap();
    for (tag, flex) in [("f1", 1.0), ("f2.3", 2.3)] {
        let b1 = CurvedBendMacroElement::new(fnode(0, p[0]), fnode(1, p[1]), [0.0, 0.0, 0.0], e, g, a, i, j, flex, flex).unwrap();
        let b2 = CurvedBendMacroElement::new(fnode(1, p[1]), fnode(2, p[2]), [0.0, 2.0 * r, 0.0], e, g, a, i, j, flex, flex).unwrap();
        let mut m = Model { name: format!("sbend-{tag}"), nodes: p.to_vec(), frames: vec![], users: vec![], macros: vec![b1, b2], slots: vec![], springs: vec![], prescribed: vec![], loads: vec![] };
        m.slots.push(CurvedBendStiffnessElement::from_macro_element("b1", &b1).unwrap());
        m.slots.push(CurvedBendStiffnessElement::from_macro_element("b2", &b2).unwrap());
        m.frames.push(FrameElement::new(fnode(3, p[3]), fnode(0, p[0]), sec, [0.0, 0.0, 1.0]).unwrap());
        m.frames.push(FrameElement::new(fnode(2, p[2]), fnode(4, p[4]), sec, [0.0, 0.0, 1.0]).unwrap());
        for d in 0..6 {
            m.springs.push((6 + d, 1.7e5 * (d as f64 + 1.3)));
        }
        m.springs.push((6 + 2, 3.1e4));
        m.springs.push((12 + 5, 2.2e3));
        m.prescribed = (18..30).map(|d| (d, if d == 19 { 0.0007 } else { 0.0 })).collect();
        m.loads = vec![(7, -2500.0), (8, 400.0), (11, 55.0), (13, 900.0)];
        if let Some((d, s)) = assembly_parity(&m) {
            let free = m.free();
            gate_parity(&m.name, &m, &d, &s, &free, &m.prescribed);
        }
    }
    // Product-shaped junctions (invented): (1) a realized bend between two
    // straight runs with a support spring at each bend end; (2) a tee at a
    // bend-bend junction (two realized bends and a straight branch meeting at
    // one node). Each is assembled in the product's order and compared.
    for (tag, flex) in [("f1", 1.0), ("f1.7", 1.7)] {
        let b1 = CurvedBendMacroElement::new(fnode(0, p[0]), fnode(1, p[1]), [0.0, 0.0, 0.0], e, g, a, i, j, flex, flex).unwrap();
        let b2 = CurvedBendMacroElement::new(fnode(1, p[1]), fnode(2, p[2]), [0.0, 2.0 * r, 0.0], e, g, a, i, j, flex, flex).unwrap();
        // (1) bend with supports at both ends
        let mut m = Model { name: format!("bend-supports-{tag}"), nodes: p.to_vec(), frames: vec![], users: vec![], macros: vec![b1], slots: vec![], springs: vec![], prescribed: vec![], loads: vec![] };
        m.slots.push(CurvedBendStiffnessElement::from_macro_element("b1", &b1).unwrap());
        m.frames.push(FrameElement::new(fnode(3, p[3]), fnode(0, p[0]), sec, [0.0, 0.0, 1.0]).unwrap());
        m.frames.push(FrameElement::new(fnode(1, p[1]), fnode(4, [-2.4, r, 0.3]), sec, [0.0, 0.0, 1.0]).unwrap());
        m.nodes[4] = [-2.4, r, 0.3];
        for (node, k) in [(0usize, 3.3e5), (1usize, 7.7e5)] {
            for d in [1usize, 2, 3] {
                m.springs.push((6 * node + d, k * (1.0 + 0.1 * d as f64)));
            }
        }
        // Variants: support springs on every DOF of both bend ends, over a
        // range of magnitudes (the order of a spring and a bend on one
        // diagonal can change its last bit).
        for (vi, kk) in [1.1e3, 3.7e4, 2.9e5, 1.3e6, 7.9e6, 4.4e7, 2.1e8, 9.7e8].iter().enumerate() {
            let mut v = m.clone();
            v.name = format!("bend-supports-{tag}-k{vi}");
            v.springs = (0..6).flat_map(|d| [(d, kk * (1.0 + 0.07 * d as f64)), (6 + d, kk * (1.3 - 0.05 * d as f64))]).collect();
            if let Some((d, s)) = assembly_parity(&v) {
                let free = v.free();
                gate_parity(&v.name, &v, &d, &s, &free, &v.prescribed);
            }
        }
        m.prescribed = (12..24).map(|d| (d, 0.0)).chain((24..30).map(|d| (d, if d == 25 { -0.0004 } else { 0.0 }))).collect();
        m.loads = vec![(1, -1800.0), (8, 350.0), (9, 42.0), (10, 700.0)];
        m.springs.push((12 + 1, 5.0e4));
        if let Some((d, s)) = assembly_parity(&m) {
            let free = m.free();
            gate_parity(&m.name, &m, &d, &s, &free, &m.prescribed);
        }
        // (2) tee at a bend-bend junction
        let mut t = Model { name: format!("bendbend-tee-{tag}"), nodes: p.to_vec(), frames: vec![], users: vec![], macros: vec![b1, b2], slots: vec![], springs: vec![], prescribed: vec![], loads: vec![] };
        t.nodes.push([0.0, r, 2.2]);
        t.slots.push(CurvedBendStiffnessElement::from_macro_element("b1", &b1).unwrap());
        t.slots.push(CurvedBendStiffnessElement::from_macro_element("b2", &b2).unwrap());
        t.frames.push(FrameElement::new(fnode(3, p[3]), fnode(0, p[0]), sec, [0.0, 0.0, 1.0]).unwrap());
        t.frames.push(FrameElement::new(fnode(2, p[2]), fnode(4, p[4]), sec, [0.0, 0.0, 1.0]).unwrap());
        t.frames.push(FrameElement::new(fnode(1, p[1]), fnode(5, [0.0, r, 2.2]), sec, [1.0, 0.0, 0.0]).unwrap());
        t.prescribed = (18..30).map(|d| (d, 0.0)).chain((30..36).map(|d| (d, 0.0))).collect();
        t.loads = vec![(7, -2500.0), (8, 400.0), (11, 55.0), (13, 900.0)];
        if let Some((d, s)) = assembly_parity(&t) {
            let free = t.free();
            gate_parity(&t.name, &t, &d, &s, &free, &t.prescribed);
        }
    }
}

fn render<T: std::fmt::Debug>(x: &T) -> String {
    format!("{x:?}")
}

/// Assembly, coalescing, reduction, multiply and reactions.
fn assembly_parity(m: &Model) -> Option<(Vec<Vec<f64>>, SparseStiffness)> {
    let case = &m.name;
    let d = m.dense();
    let s = m.sparse();
    match (&d, &s) {
        (Ok(d), Ok(s)) => {
            check(case, "K bits (to_dense)", &bits2(d), &bits2(&s.to_dense()));
            let zeros = s.values().iter().filter(|v| **v == 0.0).count();
            if zeros > 0 { stat("model: stored explicit zeros"); }
            if s.values().iter().any(|v| v.to_bits() == (-0.0f64).to_bits()) { stat("model: stored -0.0 (unexpected)"); }
            let mut deg = vec![0usize; m.nodes.len()];
            for f in &m.frames { deg[f.node_i.index] += 1; deg[f.node_j.index] += 1; }
            for u in &m.users { deg[u.node_i.index] += 1; deg[u.node_j.index] += 1; }
            for b in &m.slots { deg[b.node_i] += 1; deg[b.node_j] += 1; }
            if deg.iter().any(|&x| x >= 4) { stat("model: node with >= 4 elements"); }
            let mut pairs: Vec<(usize,usize)> = m.frames.iter().map(|f| (f.node_i.index.min(f.node_j.index), f.node_i.index.max(f.node_j.index))).chain(m.users.iter().map(|f| (f.node_i.index.min(f.node_j.index), f.node_i.index.max(f.node_j.index)))).chain(m.slots.iter().map(|b| (b.node_i.min(b.node_j), b.node_i.max(b.node_j)))).collect();
            let np = pairs.len(); pairs.sort(); pairs.dedup(); if pairs.len() < np { stat("model: duplicate connectivity"); }
            if !m.users.is_empty() && !m.frames.is_empty() && !m.slots.is_empty() { stat("model: frames+users+blocks"); }
            if !m.macros.is_empty() { stat("model: traced curved bend"); }
            if m.slots.iter().any(|b| b.global_stiffness.iter().flatten().any(|v| v.to_bits() == (-0.0f64).to_bits())) { stat("model: block with -0.0 entries"); }
            if m.springs.iter().any(|sp| m.prescribed.iter().any(|p| p.0 == sp.0)) { stat("model: spring on prescribed DOF"); }
            if m.springs.iter().any(|sp| sp.1.to_bits() == (-0.0f64).to_bits()) { stat("model: -0.0 spring"); }
            if m.prescribed.iter().any(|p| p.1 != 0.0) { stat("model: nonzero prescribed"); }
            if m.prescribed.iter().any(|p| p.1.to_bits() == (-0.0f64).to_bits()) { stat("model: -0.0 prescribed"); }
            // entries that are exactly zero here but nonzero in some contribution (cancellation)
            if let Ok(ev) = AssemblyEvidence::new(m.nodes.len(), &m.frames, &m.users, &m.slots, &m.springs) {
                let mut nz = std::collections::HashSet::new();
                for c in &ev.contributions { if c.value != 0.0 { nz.insert((c.row, c.col)); } }
                if (0..d.len()).any(|i| (0..d.len()).any(|j| d[i][j] == 0.0 && nz.contains(&(i, j)))) { stat("model: exact cancellation to 0"); }
            }
            // every entry outside the pattern is +0.0 in the dense matrix
            let mut outside = String::new();
            for i in 0..d.len() {
                for j in 0..d.len() {
                    if s.pattern().find(i, j).is_none() && d[i][j].to_bits() != 0 {
                        outside.push_str(&format!("({i},{j})={:?} ", d[i][j]));
                    }
                }
            }
            check(case, "outside pattern is +0.0", "", &outside);
            // the adapter's contribution list coalesces to the same values
            if let Ok(ev) = AssemblyEvidence::new(m.nodes.len(), &m.frames, &m.users, &m.slots, &m.springs) {
                let c = SparseStiffness::from_contributions(m.n(), &ev.contributions).unwrap();
                let from_pattern = SparseStiffness::from_pattern_and_contributions(s.pattern().clone(), &ev.contributions).unwrap();
                check(case, "from_contributions bits", &bits2(d), &bits2(&c.to_dense()));
                check(case, "from_pattern_and_contributions", &render(&s.values().iter().map(|v| v.to_bits()).collect::<Vec<_>>()), &render(&from_pattern.values().iter().map(|v| v.to_bits()).collect::<Vec<_>>()));
            }
            // from_dense round trip
            let fd = SparseStiffness::from_dense(d).unwrap();
            check(case, "from_dense bits", &bits2(d), &bits2(&fd.to_dense()));
            // reduction (both kinds)
            let force = m.ledger(Some(1e9));
            let dofs: Vec<usize> = m.prescribed.iter().map(|p| p.0).collect();
            let vals: Vec<f64> = m.prescribed.iter().map(|p| p.1).collect();
            let dr = reduce_assembled_system_with_prescribed_displacements(d, &force, &dofs, &vals);
            let sr = reduce_assembled_sparse_system(s, &force, &dofs, Some(&vals));
            check(case, "reduce prescribed", &render(&dr.as_ref().map(|x| (x.free_dofs.clone(), x.force.values().iter().map(|v| v.to_bits()).collect::<Vec<_>>()))), &render(&sr.as_ref().map(|x| (x.free_dofs.clone(), x.force.values().iter().map(|v| v.to_bits()).collect::<Vec<_>>()))));
            let dr0 = reduce_assembled_system(d, &force, &dofs);
            let sr0 = reduce_assembled_sparse_system(s, &force, &dofs, None);
            check(case, "reduce restrained", &render(&dr0.as_ref().map(|x| (x.free_dofs.clone(), x.force.values().iter().map(|v| v.to_bits()).collect::<Vec<_>>()))), &render(&sr0.as_ref().map(|x| (x.free_dofs.clone(), x.force.values().iter().map(|v| v.to_bits()).collect::<Vec<_>>()))));
            // multiply and reactions, with signed zeros in u
            let mut r = Lcg(m.n() as u64 * 31 + 7);
            for trial in 0..4 {
                let u: Vec<f64> = (0..m.n())
                    .map(|_| match (r.below(5), trial) {
                        (0, _) => 0.0,
                        (1, _) => -0.0,
                        (_, 3) => -0.0,
                        _ => r.range(-1e-3, 1e-3),
                    })
                    .collect();
                let dm: Vec<f64> = d.iter().map(|row| row.iter().zip(&u).map(|(a, b)| a * b).sum()).collect();
                let smul = s.multiply(&u).unwrap();
                check(case, &format!("multiply {trial}"), &render(&dm.iter().map(|v| v.to_bits()).collect::<Vec<_>>()), &render(&smul.iter().map(|v| v.to_bits()).collect::<Vec<_>>()));
                let dreact: Vec<f64> = dm
                    .into_iter()
                    .enumerate()
                    .map(|(dof, internal)| {
                        let mut acc = open_pipe_stress_frame_kernel::exact_sum::ExactAccumulator::new();
                        acc.add(internal).and_then(|()| force.accumulate_dof(dof, &mut acc, true)).and_then(|()| acc.round()).unwrap_or(f64::NAN)
                    })
                    .collect();
                let sreact = s.reactions(&u, &force).unwrap();
                check(case, &format!("reactions {trial}"), &render(&dreact.iter().map(|v| v.to_bits()).collect::<Vec<_>>()), &render(&sreact.iter().map(|v| v.to_bits()).collect::<Vec<_>>()));
            }
        }
        (Err(a), Err(b)) => check(case, "assembly error", a, b),
        _ => check(case, "assembly outcome", &render(&d.as_ref().map(|_| ())), &render(&s.as_ref().map(|_| ()))),
    }
    match (d, s) {
        (Ok(d), Ok(s)) => Some((d, s)),
        _ => None,
    }
}

/// The dense-derived order of a dense prepared matrix (today's
/// `factor_structural_ldlt` steps).
fn dense_order(p: &PreparedSystem<'_>) -> (Vec<usize>, usize) {
    let adj = sd::adjacency_from_dense(p.matrix()).unwrap();
    let order = sd::reverse_cuthill_mckee(&adj).unwrap();
    let prof = sd::SymmetricProfileMatrix::from_dense_with_order(p.matrix(), &order).unwrap();
    (order, prof.profile_entry_count())
}

/// The gate: every sparse entry against its dense sibling on the same
/// equations, `Debug`-identical, with the free and prescribed lists in the
/// given order.
fn gate_parity(case: &str, m: &Model, d: &[Vec<f64>], s: &SparseStiffness, free: &[usize], prescribed: &[(usize, f64)]) {
    let n = m.n();
    let ev = match AssemblyEvidence::new(m.nodes.len(), &m.frames, &m.users, &m.slots, &m.springs) {
        Ok(ev) => ev,
        Err(e) => {
            let sev = SparseAssemblyEvidence::new(s.pattern(), m.nodes.len(), &m.frames, &m.users, &m.slots, &m.springs);
            check(case, "evidence error", &render(&Err::<(), _>(e)), &render(&sev.map(|_| ())));
            return;
        }
    };
    let sev = SparseAssemblyEvidence::new(s.pattern(), m.nodes.len(), &m.frames, &m.users, &m.slots, &m.springs).unwrap();
    // allowances and counts per pattern entry equal the dense evidence
    let (vr, vc) = sev.dense_symmetry_view();
    check(case, "evidence roundoff", &bits2(&ev.absolute_roundoff), &bits2(&vr));
    check(case, "evidence counts", &render(&ev.operation_counts), &render(&vc));
    check(case, "evidence contributions", &render(&ev.contributions), &render(&sev.contributions()));
    let mut fv = vec![0.0; n];
    for &(dof, v) in &m.loads {
        fv[dof] += v;
    }
    for (tag, cancel) in [("plain", None), ("cancel", Some(1e12))] {
        let force = m.ledger(cancel);
        for mode in [LinearSolveMode::DenseScrutiny, LinearSolveMode::SparseInteractive] {
            let t = format!("{tag} {mode:?}");
            let dres = ev.solve_assembled(d, &force, free, prescribed, mode);
            stat(format!("outcome {t}: {}", class(&dres)));
            check(case, &format!("{t} solve_assembled"), &render(&dres), &render(&sev.solve_assembled(s, &force, free, prescribed, mode)));
            let fres = ev.solve_assembled_with_formation_check(d, &force, free, prescribed, mode, &m.macros, true);
            stat(format!("outcome {t} fc: {}", class(&fres)));
            check(case, &format!("{t} fc"), &render(&fres), &render(&sev.solve_assembled_with_formation_check(s, &force, free, prescribed, mode, &m.macros, true)));
            check(case, &format!("{t} fc unmatched"), &render(&ev.solve_assembled_with_formation_check(d, &force, free, prescribed, mode, &[], true)), &render(&sev.solve_assembled_with_formation_check(s, &force, free, prescribed, mode, &[], true)));
        }
        // kernel-level: legacy, C3-detect, typed, bare, with the evidence
        let basis = "rv7 basis";
        let dsys = StructuralSystem { stiffness: d, force: force.values(), free_dofs: free, prescribed, contributions: Some(&ev.contributions), symmetry: Some(SymmetryEvidence { absolute_roundoff: &ev.absolute_roundoff, operation_counts: &ev.operation_counts, basis }) };
        let ssys = SparseStructuralSystem::new(s, force.values(), free, prescribed, Some(sev.contributions()), Some(SparseSymmetryEvidence { absolute_roundoff: sev.absolute_roundoff(), operation_counts: sev.operation_counts(), basis }));
        check(case, &format!("{tag} legacy sparse path"), &render(&sds::solve_structural_sparse(&dsys)), &render(&sds::solve_sparse_structural(&ssys)));
        check(case, &format!("{tag} c3"), &render(&st::prepare_structural_with_force_terms(&dsys, force.terms()).and_then(|p| today(&p))), &render(&st::prepare_sparse_structural_with_force_terms(&ssys, force.terms()).and_then(|p| sds::solve_sparse_prepared(p))));
        let dt = StructuralSystem::assembled(d, &force, free, prescribed, Some(&ev.contributions), Some(SymmetryEvidence { absolute_roundoff: &ev.absolute_roundoff, operation_counts: &ev.operation_counts, basis }));
        let stt = SparseStructuralSystem::assembled(s, &force, free, prescribed, Some(sev.contributions()), Some(SparseSymmetryEvidence { absolute_roundoff: sev.absolute_roundoff(), operation_counts: sev.operation_counts(), basis }));
        check(case, &format!("{tag} typed"), &render(&st::prepare_assembled_structural(&dt).and_then(|p| today(&p))), &render(&sds::solve_assembled_sparse_structural(&stt)));
        let bare_d = StructuralSystem { stiffness: d, force: force.values(), free_dofs: free, prescribed, contributions: None, symmetry: None };
        let bare_s = SparseStructuralSystem::new(s, force.values(), free, prescribed, None, None);
        check(case, &format!("{tag} bare"), &render(&sds::solve_structural_sparse(&bare_d)), &render(&sds::solve_sparse_structural(&bare_s)));
        // prepared order and profile; witness; verify
        let pd = st::prepare_structural(&dsys);
        let ps = st::prepare_sparse_structural(&ssys);
        match (&pd, &ps) {
            (Ok(pd), Ok(ps)) => {
                let (order, profile) = dense_order(pd);
                let so = sds::order_sparse_structural(ps).unwrap();
                check(case, &format!("{tag} order"), &render(&order), &render(&so.order));
                check(case, &format!("{tag} profile"), &render(&profile), &render(&so.profile_entry_count));
                if free.len() > 200 {
                    // the dense search is O(n^4); only the order and profile here
                    continue;
                }
                let w = st::negative_pair_witness(pd);
                stat(format!("witness {tag}: {}", class(&w).chars().take(24).collect::<String>()));
                if so.order != (0..free.len()).collect::<Vec<_>>() { stat("order: non-identity RCM"); }
                check(case, &format!("{tag} witness"), &render(&w), &render(&st::sparse_negative_pair_witness(ps)));
                let nf = free.len();
                let mut r = Lcg(nf as u64 + 99);
                for k in 0..3 {
                    let v: Vec<f64> = (0..nf).map(|_| if r.below(3) == 0 { 0.0 } else { r.range(-1.0, 1.0) }).collect();
                    check(case, &format!("{tag} verify {k}"), &render(&st::verify_negative_direction(pd, &v)), &render(&st::verify_sparse_negative_direction(ps, &v)));
                }
                let dres = st::evaluate_assembled_original_residual(&dt, &vec![0.0; n]);
                let _ = dres;
            }
            _ => check(case, &format!("{tag} prepare"), &render(&pd.as_ref().map(|_| ())), &render(&ps.as_ref().map(|_| ()))),
        }
    }
}

/// A 2x2-per-node indefinite family for the witness (invented).
fn witness_parity() {
    let mut r = Lcg(4242);
    for t in 0..300 {
        let n = 3 + r.below(10);
        let mut k = vec![vec![0.0; n]; n];
        for i in 0..n {
            k[i][i] = r.range(0.5, 4.0);
        }
        for _ in 0..(n + r.below(2 * n)) {
            let i = r.below(n);
            let j = r.below(n);
            if i != j {
                let v = if r.below(7) == 0 { 0.0 } else { r.range(-5.0, 5.0) };
                k[i][j] = v;
                k[j][i] = v;
            }
        }
        let f: Vec<f64> = (0..n).map(|_| r.range(-1.0, 1.0)).collect();
        let free: Vec<usize> = if t % 3 == 0 { (0..n).rev().collect() } else { (0..n).collect() };
        let s = SparseStiffness::from_dense(&k).unwrap();
        let dsys = StructuralSystem { stiffness: &k, force: &f, free_dofs: &free, prescribed: &[], contributions: None, symmetry: None };
        let ssys = SparseStructuralSystem::new(&s, &f, &free, &[], None, None);
        let case = format!("witness{t}");
        let r0 = sds::solve_structural_sparse(&dsys);
        stat(format!("witness-family solve: {}", class(&r0).chars().take(30).collect::<String>()));
        check(&case, "solve", &render(&r0), &render(&sds::solve_sparse_structural(&ssys)));
        if let (Ok(pd), Ok(ps)) = (st::prepare_structural(&dsys), st::prepare_sparse_structural(&ssys)) {
            check(&case, "witness", &render(&st::negative_pair_witness(&pd)), &render(&st::sparse_negative_pair_witness(&ps)));
        }
    }
}

fn main() {
    let mut models = Vec::new();
    for seed in 0..400u64 {
        for e in [2.0e11, 1.93e11] {
            models.push(random_model(seed, e));
        }
    }
    let mut assembled = 0;
    for m in &models {
        if let Some((d, s)) = assembly_parity(m) {
            assembled += 1;
            let free = m.free();
            gate_parity(&format!("{} asc", m.name), m, &d, &s, &free, &m.prescribed);
            if m.name.starts_with("rand1") {
                let rev: Vec<usize> = free.iter().rev().copied().collect();
                let mut pre = m.prescribed.clone();
                pre.rotate_left(1);
                gate_parity(&format!("{} rev", m.name), m, &d, &s, &rev, &pre);
            }
        }
    }
    // Zero on one basis only: a fixed block equal to minus a member's matrix
    // at E = 2e11 (invented), with the frames re-formed per basis.
    for e in [2.0e11, 1.93e11] {
        let mut r = Lcg(77);
        let s0 = section(&mut r, 2.0e11);
        let s1 = FrameSection { elastic_modulus: e, shear_modulus: e / 2.5, ..s0 };
        let p = [[0.0, 0.0, 0.0], [1.3, 0.4, 0.2], [2.1, -0.7, 0.9], [3.3, 0.2, 1.4]];
        let fr = |a: usize, b: usize, sec: FrameSection| FrameElement::new(fnode(a, p[a]), fnode(b, p[b]), sec, [0.0, 0.0, 1.0]).unwrap();
        let fixed = fr(1, 2, s0).global_stiffness().unwrap();
        let mut neg = fixed;
        for row in neg.iter_mut() { for v in row.iter_mut() { *v = -*v; } }
        let mut m = Model { name: format!("onebasis-E{e}"), nodes: p.to_vec(), frames: vec![fr(0, 1, s1), fr(1, 2, s1), fr(2, 3, s1)], users: vec![], macros: vec![], slots: vec![CurvedBendStiffnessElement::new("neg", 1, 2, neg).unwrap()], springs: vec![(9, 1e3), (13, 2e3)], prescribed: (0..6).map(|d| (d, if d == 2 { 0.001 } else { 0.0 })).collect(), loads: vec![(19, -500.0), (22, 30.0)] };
        m.springs.push((0, 5e4));
        if let Some((d, s)) = assembly_parity(&m) {
            let zeros = s.values().iter().filter(|v| **v == 0.0).count();
            println!("INFO {} stored zeros {zeros}", m.name);
            let free = m.free();
            gate_parity(&m.name, &m, &d, &s, &free, &m.prescribed);
        }
    }
    // Scale: 300-member skew chain and comb with springs (invented; dense
    // 1806^2 is well under the 10,000-member limit).
    for comb in [false, true] {
        let mut r = Lcg(if comb { 5 } else { 6 });
        let s0 = section(&mut r, 2.0e11);
        let members = 300;
        let mut nodes = vec![[0.0, 0.0, 0.0]];
        let mut frames = vec![];
        for k in 1..=members {
            let parent = if comb && k > 1 && k % 2 == 0 { k - 2 } else { k - 1 };
            let pp: [f64; 3] = nodes[parent];
            let q = [pp[0] + 1.1 + 0.1 * (k as f64 * 0.37).sin(), pp[1] + 0.37 + (if comb { 0.8 } else { 0.0 }), pp[2] + 0.21 * (k as f64 * 0.11).cos()];
            nodes.push(q);
            frames.push(FrameElement::new(fnode(parent, pp), fnode(k, q), s0, [0.0, 0.0, 1.0]).unwrap());
        }
        let n = 6 * nodes.len();
        let m = Model { name: format!("scale300-comb{comb}"), nodes, frames, users: vec![], macros: vec![], slots: vec![], springs: (0..20).map(|k| (6 * (15 * k + 3) % n, 1e3 * (k as f64 + 1.0))).collect(), prescribed: (0..6).map(|d| (d, 0.0)).collect(), loads: vec![(n - 5, -1000.0), (n - 1, 20.0)] };
        if let Some((d, s)) = assembly_parity(&m) {
            let free = m.free();
            gate_parity(&m.name, &m, &d, &s, &free, &m.prescribed);
        }
    }
    witness_parity();
    targeted();
    for (k, v) in STATS.lock().unwrap().iter() {
        println!("STAT {v:6} {k}");
    }
    unsafe {
        println!("SUMMARY models={} assembled={} checks={} mismatches={}", models.len(), assembled, CHECKS, MISMATCHES);
    }
}
