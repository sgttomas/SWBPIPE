#!/usr/bin/env python3
"""RV25: machine-path, host-name and model-identifier scan of PR #1062.

Run from the repository root of a Git working tree that has both commits:
    RV25_EXTRA_PATTERNS_FILE=<file> python3 leak_scan.py.txt <base> <head>
It reads every added or modified file at <head> by `git show` (for a modified
file, only the lines the PR adds), and reports hits per pattern class:
  gen8     GEN-8's MACHINE_ABS_PATH_RE (tools/practitioner_harness/surface_roles.py)
  broad    /Users/, /private/, /home/, /tmp/, /var/folders, /Volumes/, ~/ , $HOME
  host     patterns read from RV25_EXTRA_PATTERNS_FILE, one regex per line: the
           host's computer name, local host name and user name. They are supplied
           at run time and never printed: a hit prints path:line and the class only.
  model    model identifiers (claude-*, Opus, Sonnet, Haiku, GPT-*, gpt-*)
Lines are printed only for classes gen8, broad and model, truncated to 160 chars.
"""
import os
import re
import subprocess
import sys

GEN8 = re.compile(
    r"(?<![A-Za-z0-9_.{-])"
    r"(?:/(?:Users|private|home|tmp|var/folders)/[^\s)\"'`<>]+"
    r"|[A-Za-z]:\\Users\\[A-Za-z0-9._-]+[^\s`'\"<>)\]]*)"
)
BROAD = re.compile(r"/Users/|/private/|/home/|/tmp/|/var/folders|/Volumes/|(?<![\w.])~/|\$HOME\b")
MODEL = re.compile(r"claude-[a-z0-9]|\bOpus\b|\bSonnet\b|\bHaiku\b|\bGPT-\d|\bgpt-\d", re.I)


def git(*args):
    return subprocess.run(["git", *args], check=True, capture_output=True).stdout


def main():
    base, head = sys.argv[1], sys.argv[2]
    extra = []
    pf = os.environ.get("RV25_EXTRA_PATTERNS_FILE")
    if pf:
        with open(pf, encoding="utf-8") as fh:
            extra = [re.compile(l.strip(), re.I) for l in fh if l.strip()]
    rows = git("diff", "--name-status", base, head).decode().splitlines()
    counts = {"gen8": 0, "broad": 0, "host": 0, "model": 0}
    files = 0
    for row in rows:
        status, path = row.split("\t", 1)
        if status == "A":
            text = git("show", f"{head}:{path}").decode("utf-8", "replace")
            lines = list(enumerate(text.splitlines(), 1))
        elif status == "M":
            diff = git("diff", "-U0", base, head, "--", path).decode("utf-8", "replace")
            lines = []
            ln = 0
            for d in diff.splitlines():
                m = re.match(r"@@ -\S+ \+(\d+)", d)
                if m:
                    ln = int(m.group(1))
                    continue
                if d.startswith("+") and not d.startswith("+++"):
                    lines.append((ln, d[1:]))
                    ln += 1
        else:
            continue
        files += 1
        for n, line in lines:
            for cls, rx in (("gen8", GEN8), ("broad", BROAD), ("model", MODEL)):
                if rx.search(line):
                    counts[cls] += 1
                    print(f"{cls}\t{path}:{n}\t{line.strip()[:160]}")
            for rx in extra:
                if rx.search(line):
                    counts["host"] += 1
                    print(f"host\t{path}:{n}\t<redacted>")
                    break
    print(f"files scanned: {files}; hits: {counts}")


if __name__ == "__main__":
    main()
