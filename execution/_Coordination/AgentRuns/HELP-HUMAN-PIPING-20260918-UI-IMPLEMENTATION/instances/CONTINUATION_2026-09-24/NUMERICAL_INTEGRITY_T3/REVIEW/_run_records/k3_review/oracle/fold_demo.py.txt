"""Why K4 needs a single correct rounding of an exact expansion (S1): a
three-term exact sum at p = 128, rounded once, versus folded at p in either
order, using the oracle's correctly rounded addition (which K3's add matches)."""
import rv12_oracle as o
L, p = 4, 128
e1 = o.norm((False, (1 << 127) + 1, -127))   # 1 + 2^-127: 128 bits, odd last bit
e2 = (False, 1, -128)                         # 2^-128: half an ulp of e1
e3 = (True, 1, -400)                          # -2^-400
exact = o.exact_add(o.exact_add(e1, e2), e3)
once = o.rnd(exact[0], exact[1], exact[2], False, p)
big_first = o.op_add(o.op_add(e1, e2, p, L), e3, p, L)
small_first = o.op_add(e1, o.op_add(e2, e3, p, L), p, L)
tok = lambda v: o.to_tok(o.norm(v), L)[:48] + "..."
print("rounded once       ", tok(once))
print("folded, big first  ", tok(big_first), "wrong" if o.norm(big_first) != o.norm(once) else "ok")
print("folded, small first", tok(small_first), "wrong" if o.norm(small_first) != o.norm(once) else "ok")
