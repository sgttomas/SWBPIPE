"""RV24: the second search of rv24_built_order_search.py, at the verdict's crossing. For built-like
2x2 pairs (diagonals a, b in [1, 4), exponents 0, positive couplings k (lower) and k + 1..4 ulps
(upper) under symmetry evidence), find the coupling k where the verifier-order verdict changes, scan
80 ulps around it, and count pairs whose verdict differs when the two coupling cells are swapped."""
import math, random
u = 2.0 ** -53
G = 64.0 * (14*u/(1-14*u))
def verdict(terms):
    e = 0.0; m = 0.0
    for t in terms:
        e = e + t; m = m + abs(t)
    return e < -(G * m)
random.seed(0x52563235)
found = []; scanned = 0
for trial in range(20000):
    a = 1.0 + 3.0 * random.random()
    b = 1.0 + 3.0 * random.random()
    skew = random.randrange(1, 5)
    def f(k):
        kt = k
        for _ in range(skew): kt = math.nextafter(kt, math.inf)
        return verdict([a, -kt, -k, b])
    lo, hi = (a + b) / 2.0 * (1 - 1e-9), (a + b) / 2.0 * (1 + 1e-9)
    if f(lo) == f(hi): continue
    for _ in range(200):
        mid = lo / 2 + hi / 2
        if mid in (lo, hi): break
        if f(mid) == f(lo): lo = mid
        else: hi = mid
    k = lo
    for _ in range(40): k = math.nextafter(k, -math.inf)
    for step in range(80):
        kt = k
        for _ in range(skew): kt = math.nextafter(kt, math.inf)
        scanned += 1
        fw = verdict([a, -kt, -k, b]); sw = verdict([a, -k, -kt, b])
        if fw != sw:
            found.append((a, b, k, kt, skew, fw, sw))
        k = math.nextafter(k, math.inf)
    if len(found) >= 3: break
print("built-like pairs scanned at the crossing:", scanned, "order flips found:", len(found))
