#!/bin/bash
set -u
WT=<wt>
FKD=$WT/rv20/probe/frame_kernel
OUT=$WT/rv20/records
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2 CARGO_TARGET_DIR=$WT/rv20-target/probe
cd $FKD
{
date; uptime
echo "== RV20 probe tests (probe copy: head + key override + offer log + rv20_probe_tests.rs)"
cargo test --offline --locked -j 4 --lib rv20_ -- --nocapture 2>&1
echo "exit: $?"
date; uptime
} > $OUT/probe_tests.raw.log 2>&1
