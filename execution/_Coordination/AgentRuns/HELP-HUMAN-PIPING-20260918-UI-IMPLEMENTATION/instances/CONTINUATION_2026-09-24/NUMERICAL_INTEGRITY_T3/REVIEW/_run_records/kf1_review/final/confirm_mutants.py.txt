#!/usr/bin/env python3
"""RV20 confirmation at 66adfede4: NONE, RV20-M4 and RV20-M5 (RV20's exact edits), each from a clean
`git archive` of the head's FK tree with a fresh target, serialized (-j 4, RUST_TEST_THREADS=2).
Kill = a failing test of `cargo test --lib kf1_`. Copies and targets are deleted after each run."""
import json, os, shutil, subprocess, sys, time
WT, HEAD, OUT = sys.argv[1], sys.argv[2], sys.argv[3]
FK = "projects/chirality-piping/core/solver/frame_kernel"
REL = "src/structural/retained/adaptive.rs"
M = [
    ("NONE", "no edit", []),
    ("RV20-M4", "the shared cap's collapse work is not charged (throwaway ctx16)",
     [("                    t.collapse(ctx16);",
       '                    t.collapse(&mut WideContext::<16>::new(1024).expect("p"));', 1)]),
    ("RV20-M5", "RuleTest ordered (d), (a), (b): trackers finished in another order",
     [("enum RuleTest {\n    /// (a): (|Δ| + V)/M.\n    Disagreement,\n    /// (b): Ŵ/V.\n    Estimate,\n"
       "    /// (d): C over its allowance.\n    Charge,\n}",
       "enum RuleTest {\n    /// (d): C over its allowance.\n    Charge,\n    /// (a): (|Δ| + V)/M.\n"
       "    Disagreement,\n    /// (b): Ŵ/V.\n    Estimate,\n}", 1)]),
]
env = dict(os.environ, RUSTUP_TOOLCHAIN="1.97.1", RUSTUP_AUTO_INSTALL="0",
           CARGO_INCREMENTAL="0", RUST_TEST_THREADS="2")
repo = os.path.join(WT, "kf1")
os.makedirs(os.path.join(WT, "rv20c", "logs"), exist_ok=True)
for name, what, edits in M:
    root = os.path.join(WT, "rv20c", "mut", name)
    target = os.path.join(WT, "rv20c-target", name)
    shutil.rmtree(root, ignore_errors=True); shutil.rmtree(target, ignore_errors=True)
    os.makedirs(root)
    archive = subprocess.run(["git", "-C", repo, "archive", HEAD, FK], capture_output=True, check=True).stdout
    subprocess.run(["tar", "-x", "-C", root], input=archive, check=True)
    fk = os.path.join(root, "frame_kernel")
    shutil.move(os.path.join(root, FK), fk)
    shutil.rmtree(os.path.join(root, "projects"))
    path = os.path.join(fk, REL)
    text = open(path, encoding="utf-8").read()
    for old, new, count in edits:
        n = text.count(old)
        assert n == count, (name, old[:70], n)
        text = text.replace(old, new)
    open(path, "w", encoding="utf-8").write(text)
    t0 = time.time()
    p = subprocess.run(["cargo", "test", "--offline", "--locked", "-j", "4", "--lib", "kf1_", "--", "--nocapture"],
                       cwd=fk, env=dict(env, CARGO_TARGET_DIR=target), capture_output=True, text=True)
    log = (p.stdout + p.stderr).replace(WT, "<wt>")
    open(os.path.join(WT, "rv20c", "logs", name + ".log"), "w").write(log)
    failed = [l.split()[1].rsplit("::", 1)[-1] for l in log.splitlines() if l.startswith("test ") and l.endswith("FAILED")]
    passed = [l for l in log.splitlines() if l.startswith("test ") and l.endswith("ok")]
    built = "could not compile" not in log and "error[" not in log
    rec = {"mutant": name, "edit": what, "exit": p.returncode, "built": built, "passed": len(passed),
           "failed": failed, "killed": built and p.returncode != 0, "seconds": round(time.time() - t0, 1)}
    print(json.dumps(rec), flush=True)
    with open(OUT, "a") as f:
        f.write(json.dumps(rec) + "\n")
    shutil.rmtree(target, ignore_errors=True); shutil.rmtree(root, ignore_errors=True)
