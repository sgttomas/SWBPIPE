import json, os, sys
from fractions import Fraction
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import compare as C
gen = sys.argv[1]
cases = {c['id']: c for c in json.load(open(os.path.join(gen, 'cases.json')))}
for path in sys.argv[2:]:
    name = os.path.basename(path)
    case = name.split('__')[0]
    c = cases[case]
    E = C.Envelope(json.load(open(path)))
    n = bad = 0
    worst = (0.0, None)
    keys = set()
    for (k, exp_s, cls, scale_s, gov) in c['rows']:
        obs, note = E.observe(k, c['meta'])
        if obs is None:
            continue
        f = C.case_unit_factor(cls, c['meta']['units'])
        exp, scale = C.dfr(exp_s), C.dfr(scale_s)
        for x in obs:
            n += 1
            ok = C.compare_value(x / f, exp, scale)[1]
            den = max(abs(exp), scale)
            r = abs(Fraction(x / f) - exp) / den if den else Fraction(0)
            if not ok:
                bad += 1
                keys.add(k)
            if r > worst[0]:
                worst = (float(r), k)
    print(f"{name}: observations {n}, outside 1e-9: {bad} ({len(keys)} keys), worst ratio {worst[0]:.3e} at {worst[1]}")
