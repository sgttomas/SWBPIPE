#!/usr/bin/env python3
"""RV23: checks the synthetic multi-block probe's output (`RV23 SYN` lines).

usage: rv23_syn.py <probe stdout> <path to gen_k4_vectors.py>
"""
import importlib.util
import sys
from fractions import Fraction as Fr

import math

from rv23_oracle import (floor_log2, bounds_from, full_from_profile, gamma, inv_norm1_dense,
                         inv_norm1_nilpotent, opt, parse_tok, passes)

P = 256
lines = [l[l.index("RV23 SYN"):].split() for l in open(sys.argv[1]) if "RV23 SYN" in l]
first, rows, blk = {}, {}, {}
Lf, Lrows = {}, {}
bl, shf, cert = {}, {}, {}
misc = {}
for f in lines:
    k = f[2]
    if k == "K":
        i = int(f[3]); blk[i] = int(f[4]); first[i] = int(f[5]); rows[i] = [parse_tok(t) for t in f[6:]]
    elif k == "L":
        i = int(f[3]); Lf[i] = int(f[5]); Lrows[i] = [parse_tok(t) for t in f[6:]]
    elif k == "blk":
        bl[int(f[3])] = dict(ref=f[4], U=f[6], NL=f[8], t=f[10], Uc=f[12])
    elif k == "shf":
        shf[int(f[3])] = dict(zip(f[4::2], f[5::2]))
    elif k == "cert":
        cert[int(f[3])] = dict(data=f[5], B=f[7])
    else:
        misc[k] = f[3:]
n = len(first)
nb = max(blk.values()) + 1
first_l = [first[i] for i in range(n)]
rows_l = [rows[i] for i in range(n)]
blk_l = [blk[i] for i in range(n)]
L = [Lrows[i] for i in range(n)]
assert [Lf[i] for i in range(n)] == first_l
out = []

# 0. γ and the factor: L·D·Lᵀ = K̃ exactly (the factor the passes read).
gam = gamma(n, P)
out.append("gamma equal: %s" % (gam == parse_tok(misc["gamma"][0])))
bad = 0
bad_by = {}
for i in range(n):
    for j in range(first_l[i], i + 1):
        s = Fr(0)
        for k in range(max(first_l[i], first_l[j]), j + 1):
            lik = L[i][k - first_l[i]] if k < i else Fr(1)
            ljk = L[j][k - first_l[j]] if k < j else Fr(1)
            s += lik * L[k][-1] * ljk
        if s != rows_l[i][j - first_l[i]]:
            bad += 1
            bad_by[blk_l[i]] = bad_by.get(blk_l[i], 0) + 1
out.append("L·D·Lᵀ == K̃ exactly: %s (%d mismatches, by block %s; the nilpotent chains 1 and 3 are exact,"
           " the others are rounded factors as expected)" % (bad == 0, bad, bad_by))

# 1. The reviewer's own passes (A2) against the Rust's tokens.
Lget = [[L[i][j - first_l[i]] if j < i else L[i][-1] for j in range(first_l[i], i + 1)] for i in range(n)]
Ldense = [dict() for _ in range(n)]
for i in range(n):
    for j in range(first_l[i], i + 1):
        Ldense[i][j] = L[i][j - first_l[i]]
Lacc = [[Ldense[i].get(j, Fr(0)) for j in range(0, i + 1)] for i in range(n)]
c, ct, refused = passes(Lacc, first_l, blk_l, P)
for b in range(nb):
    rb = bl[b]["ref"]
    rows_b = [i for i in range(n) if blk_l[i] == b]
    if b in refused:
        mine = "refused:span:%s:%d" % refused[b]
        out.append("block %d: mine %s rust %s equal %s" % (b, mine, rb, mine == rb))
    else:
        U = max(c[i] for i in rows_b)
        NL = max(ct[i] for i in rows_b)
        t, uc = bounds_from(U, NL, gam, P)
        eq = (U == parse_tok(bl[b]["U"]) and NL == parse_tok(bl[b]["NL"]) and t == parse_tok(bl[b]["t"])
              and uc == opt(bl[b]["Uc"]) and rb == "-")
        out.append("block %d: mine U,NL,t,Uc equal rust: %s (Uc %s)" % (b, eq, "none" if uc is None else "2^%.2f" % (uc.numerator.bit_length() - uc.denominator.bit_length())))

# 2. GEN's A2 passes on the same factor.
spec = importlib.util.spec_from_file_location("gen", sys.argv[2])
gen = importlib.util.module_from_spec(spec)
spec.loader.exec_module(gen)
get = lambda i, j: Ldense[i].get(j, Fr(0))
gref = {}
gc = gen.u_pass_em(get, first_l, n, P, blk_l, gref)
gct = gen.nl_pass_em(get, first_l, n, P, blk_l, gref)
for b in range(nb):
    rows_b = [i for i in range(n) if blk_l[i] == b]
    if b in gref:
        tokg = gen.refusal_token(gref[b])
        out.append("GEN block %d: %s rust %s equal %s" % (b, tokg, bl[b]["ref"], tokg == bl[b]["ref"]))
    else:
        U = max(gc[i] for i in rows_b); NL = max(gct[i] for i in rows_b)
        t, uc = gen.uc_from_em_a2(U, NL, gen.gamma_em(n, P), P)
        eq = U == parse_tok(bl[b]["U"]) and NL == parse_tok(bl[b]["NL"]) and uc == opt(bl[b]["Uc"])
        out.append("GEN block %d: U,NL,Uc equal rust: %s" % (b, eq))

# 3. Exact norms and 7d.
names = {0: "A tridiagonal", 1: "F nilpotent chain (data)", 2: "C pentadiagonal (no data)",
         3: "F2 nilpotent chain (S fails)", 4: "M near-singular"}
norms = {}
for b in range(nb):
    idx = [i for i in range(n) if blk_l[i] == b]
    if len(idx) <= 40:
        norms[b], _ = inv_norm1_dense(full_from_profile(first_l, rows_l, idx))
    else:
        norms[b] = inv_norm1_nilpotent(first_l, rows_l, L, idx)
for b in range(nb):
    B = opt(cert[b]["B"])
    S = opt(shf.get(b, {}).get("S", "-"))
    Uc = opt(bl[b]["Uc"])
    nrm = norms[b]
    lg = lambda x: "%.3f" % (floor_log2(x) + math.log2(float(x / Fr(2) ** floor_log2(x)))) if x else "-"
    line = "7d block %d (%s): data %s refusedUc %s Uc 2^%s S 2^%s B 2^%s norm 2^%s" % (
        b, names[b], cert[b]["data"], bl[b]["ref"], lg(Uc), lg(S), lg(B), lg(nrm))
    if B is not None:
        cands = [x for x in (Uc, S) if x is not None]
        line += " | B>=norm %s, B==min(formed) %s" % (B >= nrm, B == min(cands))
    else:
        line += " | no B"
    if S is not None:
        line += ", S>=norm %s" % (S >= nrm)
    if Uc is not None:
        line += ", Uc>=norm %s" % (Uc >= nrm)
    out.append(line)
out.append("7d: ucmissing/stop = %s" % " ".join(misc["ucmissing"]))
out.append("7d precedence [M, F2]: %s" % " ".join(misc["precedence_MF2"]))
out.append("iso_uc: %s" % [(int(f[3]), f[4]) for f in lines if f[2] == "iso_uc"])
out.append("iso_shift: %s" % [(int(f[3]), f[4]) for f in lines if f[2] == "iso_shift"])
out.append("shiftcount %s; shf %s" % (misc["shiftcount"], {b: (v["tries"], v["S"][:12], v["refused"]) for b, v in shf.items()}))
print("\n".join(out))
