"""RV22: RETURN §6.2, §6.3, §6.5, §7.1, §7.2 and §7.4 against b3's raw JSONL (independent code)."""
import json, math, os, re, statistics, sys
rec_dir, ret = sys.argv[1], sys.argv[2]
MiB = 1048576.0
recs = [json.loads(l) for l in open(os.path.join(rec_dir, 'records.jsonl'))]
text = open(ret).read()
def lines(run_id):
    out = {}
    for l in open(os.path.join(rec_dir, run_id + '.jsonl')):
        d = json.loads(l); out.setdefault(d['kind'], []).append(d)
    return out
def table(section_start, section_end):
    a = text.index(section_start); sec = text[a:text.index(section_end, a)]
    rows = []
    for line in sec.splitlines():
        if line.startswith('| ') and not line.startswith('| model') and not line.startswith('| mode') and not line.startswith('| family'):
            rows.append([c.strip() for c in line.strip('|').split('|')])
    return rows
bad = 0; checked = 0
def check(label, a, b, tol=0):
    global bad, checked
    checked += 1
    if abs(float(a) - float(b)) > tol:
        bad += 1; print('MISMATCH', label, 'RETURN', a, 'raw', b)
w1 = {r['model']: r for r in recs if r.get('mode') == 'w1a' and r.get('pass') == 1}
# 6.2
for c in table('### 6.2', '### 6.3'):
    model = 'RF-LARGE-' + c[0]; p = int(c[1])
    L = lines(w1[model]['run_id'])
    a = [x for x in L['attempt'] if x['repeat'] == 0 and x['precision'] == p][0]
    solve = a['own_rhs'] + a['own_solve'] + a['own_refinement'] + a['own_recovery']
    build = sum(a['shared_' + k] for k in ('formation', 'assembly', 'residual_formation', 'factor', 'condition'))
    pas = sum(a['own_' + k] for k in ('scale', 'estimate', 'charge', 'bound', 'shift'))
    vb = [a['shared_bounded_formation'], a['shared_wide_formation'], a['shared_uc']]
    check(model + ' %d role' % p, 0, 0)
    if c[2] != a['role'] or c[3] != a['outcome']:
        bad += 1; print('ROLE/OUTCOME', model, p, c[2], c[3], a['role'], a['outcome'])
    check(model + ' %d solve' % p, c[4], solve); check(model + ' %d stop' % p, c[5], a['own_stop_rule'])
    check(model + ' %d pass' % p, c[6], pas); check(model + ' %d build' % p, c[7], build)
    for lab, s, v in zip(('bounded', 'wide', 'uc'), c[8].split(' / '), vb):
        check(model + ' %d vb-%s' % (p, lab), s, v)
    check(model + ' %d unstaged' % p, c[9], a['own_unstaged'] + a['shared_unstaged'])
    check(model + ' %d charged_by' % p, c[10], a['charged_by'])
    # my own charged_by, from the fields (not the binary's)
    own = a['own_total']; cb = own + (a['shared_work'] if a['shared_built_here'] else 0) + (a['verification_shared_work'] if a['verification_shared_built_here'] else 0)
    check(model + ' %d charged_by (recomputed)' % p, c[10], cb)
    if a['own_unstaged'] != 0:
        print('NOTE own_unstaged nonzero', model, p, a['own_unstaged'])
print('6.2 checked', checked)
# 6.3 prefixes
for c in table('### 6.3', '### 6.4'):
    if not c[0].startswith(('CHAIN', 'TREE', 'CONT')):
        continue
    model = 'RF-LARGE-' + c[0]
    L = lines(w1[model]['run_id'])
    pre = sorted([s for s in L['stage'] if s['stage'].startswith('w1_prefix_')], key=lambda s: s['stage'])
    calls = [s for s in L['stage'] if s['stage'] == 'w1_solve']
    call0 = [s for s in calls if s['repeat'] == 0][0]
    peaks = [(s['heap_peak'] - s['heap_current_begin']) / MiB for s in pre]
    for j, cell in enumerate(c[1:4]):
        if cell == '-':
            if j < len(pre): bad += 1; print('PREFIX present but RETURN says -', model, j)
            continue
        v = float(cell.split(' ')[0]); tsec = float(cell.split(', ')[1].rstrip(' s'))
        check(model + ' prefix %d heap' % (j + 1), v, peaks[j], 0.051)
        check(model + ' prefix %d time' % (j + 1), tsec, pre[j]['elapsed_ns'] / 1e9, 0.0051)
    callpeak = max((s['heap_peak'] - s['heap_current_begin']) for s in calls) / MiB
    check(model + ' call heap', c[4].split(',')[0], callpeak, 0.051)
    check(model + ' last segment', c[5], callpeak - max(peaks), 0.051)
    for pl in L.get('prefix', []):
        pass
    par = [p for p in L['parity'] if p['item'] == 'w1_prefix_segments']
    if not all(p['equal'] for p in par): bad += 1; print('prefix parity false', model)
print('6.3 checked', checked)
# 6.5 parity lines and 6.4 comparisons
nparity = 0; nfalse = 0
for r in recs:
    L = lines(r['run_id'])
    for p in L.get('parity', []):
        nparity += 1; nfalse += (not p['equal'])
print('parity lines', nparity, 'false', nfalse)
# 7.2 rho and heap/E by size (both passes? RETURN says n=6 per size: pass 1)
base_fp = {t: json.load(open(os.path.join(rec_dir, 'baseline_%s.record.json' % t)))['rss']['time_peak_footprint_bytes'] for t in ('W1-T1','W1-T2','W1-T3','W1-T4')}
groups = {}
for r in recs:
    if r.get('pass') != 1: continue
    L = lines(r['run_id']); E = r['estimate_adm_bytes']
    rho = max(r['rss']['time_peak_footprint_bytes'] - base_fp[r['tier']], r['summary']['heap_peak_move']) / E
    size = 'DEC053' if r['model'].startswith('DEC053') else str(r['members'])
    groups.setdefault((r['mode'], size), []).append((rho, r['repeats_heap_peak_move'] / E, r['model']))
for c in table('### 7.2', '- **W1'):
    key = (c[0], c[1]); g = groups[key]
    check('7.2 n %s' % (key,), c[2], len(g))
    lo, hi = [float(x) for x in c[3].split('–')]; check('7.2 rho lo %s' % (key,), lo, min(x[0] for x in g), 0.0006); check('7.2 rho hi %s' % (key,), hi, max(x[0] for x in g), 0.0006)
    lo, hi = [float(x) for x in c[4].split('–')]; check('7.2 hE lo %s' % (key,), lo, min(x[1] for x in g), 0.0006); check('7.2 hE hi %s' % (key,), hi, max(x[1] for x in g), 0.0006)
    top = max(g)[2]
    if not top.endswith(c[5].split()[-1]): bad += 1; print('7.2 largest', key, c[5], top)
print('7.2 checked', checked)
# 7.1 fits: log-log least squares of heap move against members, both passes
fam_pts = {}
for r in recs:
    fam = 'DEC053' if r['model'].startswith('DEC053') else r['model'].split('-')[2]
    fam_pts.setdefault((fam, r['mode']), []).append((r['members'], r['summary']['heap_peak_move'], r['peak_rss_bytes']))
def slope(pts):
    xs = [math.log(p[0]) for p in pts]; ys = [math.log(p[1]) for p in pts]
    mx, my = sum(xs)/len(xs), sum(ys)/len(ys)
    return sum((x-mx)*(y-my) for x, y in zip(xs, ys)) / sum((x-mx)**2 for x in xs)
for c in table('### 7.1', '- **W1'):
    fam, mode = c[0].split('/')
    pts = fam_pts[(fam, mode)]
    check('7.1 n %s' % c[0], c[2], len(pts))
    check('7.1 heap slope %s' % c[0], c[1], slope([(m, h) for m, h, _ in pts]), 0.0006)
print(checked, '7.1 fits: heap-move slope checked; net-RSS slope not recomputed (baseline definition)')
print('TOTAL: %d checked, %d mismatches' % (checked, bad))
