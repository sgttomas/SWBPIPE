#!/bin/sh
# RV25: run `shasum -a 256 -c` for each SHA256SUMS that covers a path PR #1062
# changes, from the folder that holds it. Run from the T3 folder of a clean
# working tree at the head. The list is sums_coverage.out.txt's 26 folders.
# REVIEW/_run_records/SHA256SUMS (unchanged by the PR) lists paths relative to
# REVIEW/, so it is run from there as well as from its own folder.
set -u
for d in \
  DESIGN_NUMERICS/REV_5A3_CANDIDATE \
  DESIGN_NUMERICS/REV_5A3_CANDIDATE/_v4_records \
  DESIGN_NUMERICS/REV_5A3_CANDIDATE/_v4_records/r4 \
  DESIGN_NUMERICS/REV_5A3_CANDIDATE/_v4_records/r5 \
  DESIGN_NUMERICS/REV_5A3_CANDIDATE/_v4_records/r6 \
  DESIGN_NUMERICS/REV_5A3_CANDIDATE/_v4_records/r7 \
  HANDOFF_2026-09-30_TOOLS \
  IMPLEMENTATION/F1B_MERGE \
  IMPLEMENTATION/K4_MERGE \
  IMPLEMENTATION/K5_MERGE \
  IMPLEMENTATION/K6B_MERGE \
  IMPLEMENTATION/K6_MERGE \
  IMPLEMENTATION/KF1_MERGE \
  IMPLEMENTATION/KF2_MERGE \
  IMPLEMENTATION/KF3_MERGE \
  IMPLEMENTATION/VK_MERGE \
  REVIEW/_run_records \
  REVIEW/_run_records/f1b_review \
  REVIEW/_run_records/k4_review \
  REVIEW/_run_records/k5_review \
  REVIEW/_run_records/k6_review \
  REVIEW/_run_records/k6b_review \
  REVIEW/_run_records/kf1_review \
  REVIEW/_run_records/kf2_review \
  REVIEW/_run_records/kf3_review \
  REVIEW/_run_records/vk_review
do
  n=$(grep -c . "$d/SHA256SUMS")
  res=$(cd "$d" && shasum -a 256 -c SHA256SUMS 2>&1)
  ok=$(printf '%s\n' "$res" | grep -c ': OK$')
  bad=$(printf '%s\n' "$res" | grep -vc ': OK$')
  echo "$d/SHA256SUMS (run from $d): entries $n, OK $ok, not-OK $bad"
done
res=$(cd REVIEW && shasum -a 256 -c _run_records/SHA256SUMS 2>&1)
ok=$(printf '%s\n' "$res" | grep -c ': OK$')
bad=$(printf '%s\n' "$res" | grep -vc ': OK$')
echo "REVIEW/_run_records/SHA256SUMS (run from REVIEW): OK $ok, not-OK $bad"
