#!/bin/bash
# RV26: scope, hashes and record facts for records PR #1063 (head 21e2285e3, base main 490b75bd9).
# Usage, from the repository root of the numerics working tree at the head:
#   bash checks.sh.txt
# Read-only: git show/diff/log/rev-list/merge-base/cat-file, gh pr/run view, shasum. No writes.
# Other outputs in this folder:
#   append_only.out.txt    RV25's committed append_only.py.txt 490b75bd9 21e2285e3
#   sums_coverage.out.txt  RV25's committed sums_coverage.py.txt 490b75bd9 21e2285e3 <_Coordination>
#   leak_scan.out.txt      RV25's committed leak_scan.py.txt 490b75bd9 21e2285e3, with host patterns
#                          (computer name, local host name, user name) supplied at run time, never printed
#   ci_job_times.*, test_counts.*, gen8.out.txt
set -u
BASE=490b75bd992d056dbf9566ac1bdc85af2f74966d
HEAD=21e2285e30d47113a4d4dceae71c60696f1832ab
T3=projects/chirality-piping/execution/_Coordination/AgentRuns/HELP-HUMAN-PIPING-20260918-UI-IMPLEMENTATION/instances/CONTINUATION_2026-09-24/NUMERICAL_INTEGRITY_T3
R=$T3/REVIEW/_run_records/records_pr1062_review
V1=$T3/ROOT_RULINGS_V1.md
REPO=sgttomas/chirality
ts() { git log -1 --format='%ad' --date=format-local:'%Y-%m-%dT%H:%M:%S' "$1"; }
first_add() { git log --reverse --format='%h' -S"$1" -- "$2" | head -1; }

echo "== 1. The PR, its commits and scope"
gh pr view 1063 --repo $REPO --json headRefOid,baseRefName,state,mergeable --jq '"   PR head \(.headRefOid[0:10]) base \(.baseRefName) \(.state) \(.mergeable)"'
echo "   origin/main: $(git rev-parse --short=10 origin/main); merge-base(HEAD, main): $(git merge-base $HEAD origin/main | cut -c1-10)"
for c in $(git rev-list --reverse $BASE..$HEAD); do echo "   commit ${c:0:10} parents $(git rev-list --parents -n1 $c | cut -d' ' -f2- | cut -c1-10) | $(git log -1 --format=%s $c | cut -c1-110)"; done
echo "   #1062 merge 490b75bd9 parents: $(git rev-list --parents -n1 $BASE | cut -d' ' -f2- | tr ' ' '\n' | cut -c1-10 | tr '\n' ' ')"
gh pr view 1062 --repo $REPO --json headRefOid,mergedAt,mergeCommit --jq '"   #1062 head \(.headRefOid[0:10]) merged \(.mergedAt) as \(.mergeCommit.oid[0:10])"'
git diff --name-status $BASE $HEAD | awk '{print $1}' | sort | uniq -c | sed 's/^/   /'
echo "   paths outside _Coordination: $(git diff --name-only $BASE $HEAD | grep -vc '^projects/chirality-piping/execution/_Coordination/')"
echo "   binaries: $(git diff --numstat $BASE $HEAD | awk '$1=="-"' | wc -l | tr -d ' '); modes: $(git ls-tree -r $HEAD -- $(git diff --name-only $BASE $HEAD) | awk '{print $1}' | sort | uniq -c | tr -s ' ')"
echo "   git diff --check lines: $(git diff --check $BASE $HEAD | grep -c 'trailing whitespace')"
git diff --check $BASE $HEAD | grep 'trailing' | sed 's#^.*records_pr1062_review/#   #'

echo "== 2. shasum -c from each folder (working tree at the head)"
( cd $R && shasum -a 256 -c SHA256SUMS 2>&1 | awk '{print $NF}' | sort | uniq -c | sed 's/^/   records_pr1062_review: /' )
( cd $T3/REVIEW && shasum -a 256 -c _run_records/SHA256SUMS 2>&1 | awk '{print $NF}' | sort | uniq -c | sed 's/^/   REVIEW\/_run_records (run from REVIEW\/): /' )
echo "   RECORDS_PR1062_REVIEW.md at head: $(git show $HEAD:$T3/REVIEW/RECORDS_PR1062_REVIEW.md | shasum -a 256 | cut -c1-16); records_pr1062_review/SHA256SUMS: $(git show $HEAD:$R/SHA256SUMS | shasum -a 256 | cut -c1-16)"
echo "   OPERATING_NOTES_2026-09-30.md at head: $(git show $HEAD:$T3/OPERATING_NOTES_2026-09-30.md | shasum -a 256 | cut -c1-16), $(git show $HEAD:$T3/OPERATING_NOTES_2026-09-30.md | wc -l | tr -d ' ') lines"

echo "== 3. DEC-025 durations (meta.txt start -> ALL-DONE)"
for s in K5 F1B K6 K4 KF1 VK K6B KF3 KF2; do
  m=$T3/IMPLEMENTATION/${s}_MERGE/dec025/meta.txt
  python3 - "$s" "$m" <<'PY'
import re,sys,datetime as d
s,m=sys.argv[1],sys.argv[2]; t=open(m).read()
a=re.search(r'start (\S+)',t).group(1); b=re.search(r'ALL-DONE (\S+)',t).group(1)
f=lambda x:d.datetime.strptime(x,'%Y-%m-%dT%H:%M:%SZ')
print(f"   {s}: {a} -> {b} = {(f(b)-f(a)).total_seconds()/60:.1f} min")
PY
done

echo "== 4. KF2's B: the grant, its records' time span, and the acceptance"
echo "   grant ruling commit:  $(first_add '## KF2: checkpoint A accepted; B granted' $V1) $(ts $(first_add '## KF2: checkpoint A accepted; B granted' $V1)) (local -0600)"
echo "   accept ruling commit: $(first_add '## KF2: checkpoint B accepted; D now' $V1) $(ts $(first_add '## KF2: checkpoint B accepted; D now' $V1)) (local -0600)"
B=$T3/IMPLEMENTATION/KF2/_run_records/b
echo "   timestamps in KF2/_run_records/b (UTC): first $(grep -rhoE '20[0-9]{2}-[0-9]{2}-[0-9]{2}T[0-9:]{8}Z' $B | sort | head -1), last $(grep -rhoE '20[0-9]{2}-[0-9]{2}-[0-9]{2}T[0-9:]{8}Z' $B | sort | tail -1)"
grep -hE 'done 884 runs|part 2 (start|done)|Finished' $B/gate/part1/*_driver.log $B/gate/part2/driver.log $B/t9/build_*.log $B/gate/build_probe_*.log | sed 's/^/   /'

echo "== 5. Review windows: 'to review' ruling commit -> 'rulings on the review' commit (local -0600)"
while IFS='|' read -r who a b; do
  ca=$(first_add "$a" $V1); cb=$(first_add "$b" $V1)
  python3 -c "import datetime as d;f=lambda x:d.datetime.strptime(x,'%Y-%m-%dT%H:%M:%S');print(f'   $who: {(f(\"$(ts $cb)\")-f(\"$(ts $ca)\")).total_seconds()/60:.0f} min ($(ts $ca) -> $(ts $cb))')"
done <<'LIST'
RV19 (K4)|## K4: D accepted; to review|## K4: rulings on RV19's review
RV20 (KF1)|## KF1: to review|## KF1: rulings on RV20's review
RV21 (V-K)|## K6b: slot K6B-S4 (b3) accepted; D now, with W1-T4 pre-KF3; V-K's PR to review|## V-K: rulings on RV21's review
RV22 (K6b)|## K6b: D accepted; PR to review|## K6b: rulings on RV22's review
RV23 (KF3)|## KF3: D accepted; KF3-B1 and KF3-B2 routed; PR to review|## KF3: rulings on RV23's review
RV24 (KF2)|## KF2: D accepted; PR to review|## KF2: rulings on RV24's review
RV23 confirmation|## KF3 and KF2: review fixes committed|## KF3: RV23 confirms aa83f6796
LIST
gh pr view 1062 --repo $REPO --json createdAt --jq '"   RV25 (records #1062): PR opened \(.createdAt)"'
echo "   RV25 ruling commit $(first_add '## Records PR #1062: rulings on RV25' $V1) at $(git log -1 --format=%aI $(first_add '## Records PR #1062: rulings on RV25' $V1))"
echo "   KF3 DEC-025 start 2026-09-30T04:52:24Z; 'KF3: RV23 confirms' commit at $(git log -1 --format=%aI $(first_add '## KF3: RV23 confirms aa83f6796' $V1)) (the sweep ran during the confirmation)"

echo "== 6. Rulings behind named findings and events (line numbers at the head)"
for p in "RV24-N3's count" "Correction: ROOT's first message asked both implementers" "recorded late, RV13-S2" "recorded late, RV16-S3" "the C ruling was given in session" "Disclosed: ROOT merged without first checking" "Whichever PR merges first carries the export" "whichever of KF3 and K6b merges second" "KF3 merges second, so KF3 updates" "RV23-N1: add the test" "a separate dense-screen slice" "RV25-D2" "Fix before merge, as with earlier slices"; do
  echo "   V1:$(git show $HEAD:$V1 | grep -nF "$p" | head -1 | cut -d: -f1) \"$p\""
done
echo "   KF3_REVIEW.md:$(git show $HEAD:$T3/REVIEW/KF3_REVIEW.md | grep -nF 'five of I19' | cut -d: -f1) \"five of I19's (re-implemented ... since their diffs are not recorded)\""
echo "   KF2_REVIEW.md:$(git show $HEAD:$T3/REVIEW/KF2_REVIEW.md | grep -nF 'my "eight" was wrong' | cut -d: -f1) \"I20's correction is right, and my eight was wrong\""
echo "   I21 brief asks for mutant diffs: $(git show $HEAD:$T3/TASK_BRIEFS/I21_K6C_IMPLEMENTATION.md | grep -ciE '\.diff|mutant diff|patch text')"
echo "   RV briefs committed in TASK_BRIEFS/: $(git ls-tree --name-only $HEAD $T3/TASK_BRIEFS/ | grep -oE 'RV[0-9]+' | sort -V | tr '\n' ' ')"
