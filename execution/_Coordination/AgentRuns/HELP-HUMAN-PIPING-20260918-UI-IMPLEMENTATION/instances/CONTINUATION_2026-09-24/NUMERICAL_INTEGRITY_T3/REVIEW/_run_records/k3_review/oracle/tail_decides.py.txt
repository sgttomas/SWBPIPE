"""Cases where the (1 - f) tail of a far subtraction decides a tie at p = 64L:
big odd with a full 64L-bit significand, minus an all-ones small at gap 64L + 1
(its exact value is just above the midpoint below big). Also the mirror cases."""
import random, sys
import rv12_oracle as o
rng = random.Random(int(sys.argv[1]))
lines = []
for L in (4, 8, 16):
    W = 64 * L
    for _ in range(50):
        for p in (W, W - 1, W - 2):
            n = rng.getrandbits(W) | (1 << (W - 1)) | 1  # odd, full width
            E = rng.randrange(-100, 100)
            s = rng.random() < 0.5
            big = (s, n, E - W + 1)
            for g in (W + 1, W + 2):
                small = (not s, (1 << W) - 1, E - g - W + 1)
                lines.append("x add %d %d %s %s" % (L, p, o.to_tok(o.norm(big), L), o.to_tok(o.norm(small), L)))
                small2 = (not s, ((1 << W) - 1) ^ rng.choice([1, 2, 1 << 63]), E - g - W + 1)
                lines.append("x add %d %d %s %s" % (L, p, o.to_tok(o.norm(big), L), o.to_tok(o.norm(small2), L)))
            nn = n & ~1  # even variant
            big2 = (s, nn | (1 << (W - 1)), E - W + 1)
            lines.append("x add %d %d %s %s" % (L, p, o.to_tok(o.norm(big2), L), o.to_tok(o.norm((not s, (1 << W) - 1, E - W - 1 - W + 1)), L)))
out = ["%d %s" % (i + 1, l.split(" ", 1)[1]) for i, l in enumerate(lines)]
open(sys.argv[2], "w").write("\n".join(out) + "\n")
print(len(out))
