"""RV10 (reviewer's) mutants. Usage: mutate_rv10.py <tree root> <mutant id>.
I9's mutants are applied from I9's own committed patch file (mutate_i9.py.txt), unchanged, by
running it; RV10's own are defined here. Each anchor must match exactly once."""
import os, runpy, sys

root, mid = sys.argv[1], sys.argv[2]
FK = root + "/projects/chirality-piping/core/solver/frame_kernel/src/structural.rs"
SPARSE = root + "/projects/chirality-piping/core/solver/frame_kernel/src/structural/sparse.rs"
I9 = ("M-AXIS-BOUND", "M-FLOOR-UP", "M-NO-COUPLING", "M-CLOSE-GAP", "NONE")

BOUND_SUMS = """                inherited = checked_value(
                    inherited + checked_product(transform[k][i].abs(), magnitudes[k][j])?,
                )?;
                second = checked_value(
                    second + checked_product(transform[k][i].abs(), temp[k][j].abs())?,
                )?;
"""
FN_HEAD = """) -> Result<TransformationRoundoff, StructuralError> {
    let mut temp = [[0.0; 12]; 12];
"""


def deep_refuse(exponent):
    return [(FK, FN_HEAD, f""") -> Result<TransformationRoundoff, StructuralError> {{
    for row in local {{
        for &c in row {{
            if c != 0.0 && c.abs() < 2f64.powi({exponent}) {{
                return Err(StructuralError::Range("arithmetic outside normal range"));
            }}
        }}
    }}
    let mut temp = [[0.0; 12]; 12];
""")]


MUTANTS = {
    # The bound's sums replaced by maxima: the axis-aligned bound is unchanged (one term per
    # entry); on a skew member each entry's bound is its largest single term.
    "RV10-MAX-NOT-SUM": [(FK, BOUND_SUMS, """                inherited = checked_value(
                    f64::max(inherited, checked_product(transform[k][i].abs(), magnitudes[k][j])?),
                )?;
                second = checked_value(
                    f64::max(second, checked_product(transform[k][i].abs(), temp[k][j].abs())?),
                )?;
""")],
    # An index slip: |T_ik| for |T_ki| in the second stage (T^T read as T). The same on an
    # axis-aligned member (T = I); a different mixing on a skew one.
    "RV10-TRANSPOSE": [(FK, BOUND_SUMS, BOUND_SUMS.replace("transform[k][i]", "transform[i][k]"))],
    # The torsion rows (local RX of either node) dropped from the second stage: no GJ/L lift.
    "RV10-NO-TORSION-MIX": [(FK, BOUND_SUMS, "                if k % 6 == 3 {\n                    continue;\n                }\n" + BOUND_SUMS)],
    # A partial "fix" that refuses every nonzero local coefficient below 2^-1016 (below every
    # accepted row's 4EI/L and 2EI/L, above the 2^-1055 row's).
    "RV10-DEEP-REFUSE-1016": deep_refuse(-1016),
    # The same below 2^-1019 (below every coefficient any pin test forms).
    "RV10-DEEP-REFUSE-1019": deep_refuse(-1019),
    # K1's pattern path only (FK sparse.rs prepare_sparse_bound): the column's scale exponent
    # replaced by the row's when the prepared values are formed. Dense is untouched.
    "RV10-K1-PATTERN-SCALE": [(SPARSE, """            values.push(radix_scale(
                stiffness.values[index],
                exponents[r] + exponents[c],
            )?);
""", """            values.push(radix_scale(
                stiffness.values[index],
                exponents[r] + exponents[r],
            )?);
""")],
}

if mid in I9:
    sys.argv = ["mutate_i9.py", root, mid]
    runpy.run_path(os.path.join(os.path.dirname(os.path.abspath(__file__)), "mutate_i9.py"), run_name="__main__")
else:
    for path, old, new in MUTANTS[mid]:
        text = open(path).read()
        assert text.count(old) == 1, (mid, path, text.count(old))
        open(path, "w").write(text.replace(old, new))
    print(f"{mid}: {len(MUTANTS[mid])} hunk(s) applied")
