"""Named binary64-conversion boundary cases through the Rust probe results."""
import sys
import rv12_oracle as o
LIMIT = 1 << 62
def named(L):
    W = 64 * L
    out = []
    for neg in (False, True):
        s = "-" if neg else "+"
        out += [
            (s + "zero", (neg, 0, 0)),
            (s + "2^-1074 (least subnormal)", (neg, 1, -1074)),
            (s + "2^-1075 exactly (tie to 0)", (neg, 1, -1075)),
            (s + "2^-1075 + smallest tail", (neg, (1 << (W - 1)) + 1, -1075 - (W - 1))),
            (s + "2^-1075 - smallest tail", (neg, (1 << W) - 1, -1075 - W)),
            (s + "largest subnormal", (neg, (1 << 52) - 1, -1074)),
            (s + "2^-1022 - 2^-1075 (tie, up to normal)", (neg, (1 << 53) - 1, -1075)),
            (s + "2^-1022 - 2^-1075 - tail (largest subnormal)", (neg, ((1 << 53) - 1) * (1 << (W - 53)) - 1, -1075 - (W - 53))),
            (s + "smallest normal", (neg, 1, -1022)),
            (s + "subnormal tie k=2 (even) + far sticky", (neg, (5 << (W - 3)) | 1, -1075 - (W - 3))),
            (s + "subnormal tie k=3 (odd), exact", (neg, 7 << (W - 3), -1075 - (W - 3))),
            (s + "double-rounding trap 2^-1075 + 2^-(1075+W-1)", (neg, (1 << (W - 1)) | 1, -1075 - (W - 1))),
            (s + "MAX", (neg, (1 << 53) - 1, 971)),
            (s + "midpoint 2^1024 - 2^970 - smallest tail", (neg, ((1 << 54) - 1) * (1 << (W - 54)) - 1, 970 - (W - 54))),
            (s + "midpoint 2^1024 - 2^970", (neg, (1 << 54) - 1, 970)),
            (s + "midpoint 2^1024 - 2^970 + smallest tail", (neg, ((1 << 54) - 1) * (1 << (W - 54)) + 1, 970 - (W - 54))),
            (s + "2^1024", (neg, 1, 1024)),
            (s + "2^(2^62)", (neg, 1, LIMIT)),
            (s + "2^(-2^62)", (neg, 1, -LIMIT)),
        ]
    return out
ops, names = [], []
for L in (2, 4, 8, 16):
    for name, v in named(L):
        ops.append("%d conv %d - %s" % (len(ops) + 1, L, o.to_tok(o.norm(v), L)))
        names.append((L, name))
if sys.argv[1] == "ops":
    open(sys.argv[2], "w").write("\n".join(ops) + "\n")
else:
    got = dict(l.split(None, 1) for l in open(sys.argv[2]))
    bad = 0
    for i, line in enumerate(ops, 1):
        f = line.split()
        exp = o.expected(f)
        g = got[str(i)].strip()
        ok = "ok" if exp == g else "MISMATCH"
        bad += exp != g
        L, name = names[i - 1]
        print("L=%-2d %-48s %-40s %s" % (L, name, g, ok))
    print("mismatches", bad)
