#!/usr/bin/env python3
"""RV19 delta check: GEN's 128-bit expectation tokens (x:, err:, underflow and
overflow markers) against RV19's independent 1600-digit solve of the same
intended model (rv19_oracle.py's formation, solve and recovery). For every
`expect` key of every model and combination that has tokens:
  |x - q*| <= 2^-127·|x| + 2^err   (the slack compare_honest uses)
and the marker agrees with q*'s binary64 range. Also reports how the slack
compares with the tightest claim a row of that kind can carry. Standard
library only."""
import sys
import importlib.util
from decimal import Decimal
from fractions import Fraction

spec = importlib.util.spec_from_file_location("o", sys.argv[1])
o = importlib.util.module_from_spec(spec)
spec.loader.exec_module(o)
TWO = Decimal(2)


def parse_expect(text):
    """name -> {key: (x Decimal or None, err or None, marker or None)}"""
    out = {}
    cur = None
    for line in text.splitlines():
        f = line.split()
        if not f:
            continue
        if f[0] in ("model", "combo"):
            cur = out.setdefault(f[1], {})
            continue
        if f[0] == "end":
            cur = None
            continue
        if f[0] == "expect" and cur is not None:
            x = err = mk = None
            for t in f[3:]:
                if t.startswith("x:"):
                    v = t[2:]
                    if v == "0":
                        x = Decimal(0)
                    else:
                        m, e = v[1:].split("p")
                        x = Decimal(int(m, 16)) * TWO ** int(e)
                        if v[0] == "-":
                            x = -x
                elif t.startswith("err:"):
                    err = int(t[4:])
                elif t in ("underflow", "overflow"):
                    mk = t
            cur[f[1]] = (x, err, mk)
    return out


def with_derived(m, rows):
    rows = dict(rows)
    for mem in m.members:
        i = mem["id"]
        rows["N.%d" % i] = rows["end.%d.j.0" % i]
        rows["T.%d" % i] = rows["end.%d.j.3" % i]
        for end in ("i", "j"):
            y, z = rows["end.%d.%s.4" % (i, end)], rows["end.%d.%s.5" % (i, end)]
            rows["Mb.%d.%s" % (i, end)] = (y * y + z * z).sqrt()
    for s in m.stations:
        y, z = rows["st.%d.4" % s["id"]], rows["st.%d.5" % s["id"]]
        rows["Mbs.%d" % s["id"]] = (y * y + z * z).sqrt()
    return rows


def main():
    models, combos = {}, {}
    exp = {}
    for p in sys.argv[2].split(","):
        text = open(p).read()
        mm, cc = o.parse_models(text)
        models.update(mm)
        combos.update(cc)
        exp.update(parse_expect(text))
    only = set(sys.argv[3:]) if len(sys.argv) > 3 else None
    n_keys = n_bad = n_models = 0
    tight = []
    for name, ex in exp.items():
        if not ex or (only and name not in only):
            continue
        if name in combos:
            ops = combos[name]
            m = models[ops[0][1]]
            loads, presc = {}, {}
            for c, on in ops:
                om = models[on]
                for g, v in om.loads:
                    loads[g] = loads.get(g, Decimal(0)) + Decimal(c) * Decimal(v)
                for g, v in om.constraints:
                    presc[g] = presc.get(g, Decimal(0)) + Decimal(c) * Decimal(v)
        else:
            m = models[name]
            loads = {}
            for g, v in m.loads:
                loads[g] = loads.get(g, Decimal(0)) + Decimal(v)
            presc = {g: Decimal(v) for g, v in m.constraints}
        try:
            K, u, ops_, db = o.solve_model(m, loads, presc)
        except ValueError as e:
            print("XCHECK %s: oracle %s" % (name, e))
            continue
        truth = with_derived(m, o.recover_rows(m, K, u, ops_, db, loads, presc))
        tmax = max([abs(v) for v in truth.values()] + [Decimal(1)])
        noise = tmax * Decimal("1e-1200")
        n_models += 1
        bad = []
        worst_err_rel = None
        for key, (x, err, mk) in ex.items():
            if x is None:
                continue
            n_keys += 1
            if key not in truth:
                bad.append("NOKEY %s" % key)
                continue
            q = truth[key]
            slack = TWO ** -127 * abs(x) + (TWO ** err if err is not None else 0)
            d = abs(x - q)
            if d > slack + noise:
                bad.append("%s x=%s q*=%s |x-q*|=%s slack=%s" % (
                    key, format(x, ".6e"), format(q, ".6e"), format(d, ".3e"), format(slack, ".3e")))
            if err is not None:
                r = TWO ** err / tmax
                if worst_err_rel is None or r > worst_err_rel:
                    worst_err_rel = r
            # marker against q*'s binary64 range
            if abs(q) <= noise:
                want = None
            else:
                aq = abs(q)
                if aq <= TWO ** -1075:
                    want = "underflow"
                elif aq >= TWO ** 1024 - TWO ** 970:
                    want = "overflow"
                else:
                    want = None
            if want != mk:
                bad.append("MARKER %s q*=%s marker=%s want=%s" % (key, format(q, ".6e"), mk, want))
        n_bad += len(bad)
        rel = "" if worst_err_rel is None else " err<=2^%.1f of max" % (float(worst_err_rel.ln() / TWO.ln()))
        print("XCHECK %s keys %d%s%s" % (name, len(ex), rel, "" if not bad else " ISSUES " + " | ".join(bad[:6])))
    print("XCHECK totals: models %d, keys %d, disagreements %d" % (n_models, n_keys, n_bad))


main()
