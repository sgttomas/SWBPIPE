"""RV24: re-run a sample of KF2 B's gate part 1 with P1's full-envelope probe (main.rs cd1052f7...),
built by the reviewer from its own git-archive copies of base 78f55f927 and head f2b8c85a2.
Usage: rv24_gate_sample.py <probe_base> <probe_head> <requests dir> <index_base.tsv> <index_cand.tsv>
                           <runs_base.jsonl> <runs_cand.jsonl> <out dir> [<probe_instrumented>]
Each (request, mode, entry) runs once per probe, in a fresh process, one at a time, with
T3_FULL_ENVELOPE_PATH set. Compared: base vs head (full-envelope bytes, the probe's summary object),
and each against I20's recorded index (full_envelope_sha256, outcome) and runs.jsonl summary object.
The optional instrumented probe (the head with one stderr line in `negative_pair_witness`) shows
whether the dense witness ran in each run; its envelope is also compared."""
import csv, hashlib, json, os, subprocess, sys, time

pb, ph, reqdir, ib, ic, rb, rc, out = sys.argv[1:9]
pi = sys.argv[9] if len(sys.argv) > 9 else None
os.makedirs(out, exist_ok=True)
cases = sorted(f[:-5] for f in os.listdir(reqdir) if f.endswith(".json"))
MODES = ("dense_scrutiny", "sparse_interactive")
ENTRIES = ("captured", "typed")

def index(path):
    return {(r["case"], r["mode"], r["entry"]): r for r in csv.DictReader(open(path), delimiter="\t")}
IB, IC = index(ib), index(ic)

def recorded(path):
    want = {(c, m, e) for c in cases for m in MODES for e in ENTRIES}
    got = {}
    with open(path) as fh:
        for line in fh:
            if not any(('"case": "%s"' % c) in line for c in cases):
                continue
            r = json.loads(line)
            k = (r["case"], r["mode"], r["entry"])
            if k in want:
                got[k] = r["probe"]["run"].get("envelope")
    return got
RB, RC = recorded(rb), recorded(rc)

def run(binary, req, mode, entry, tag):
    full = os.path.join(out, "full_%s_%s_%s_%s.json" % (tag, os.path.basename(req)[:-5], mode, entry))
    env = dict(os.environ, T3_FULL_ENVELOPE_PATH=full)
    args = [binary, req, mode] + (["typed"] if entry == "typed" else [])
    t0 = time.monotonic()
    p = subprocess.run(args, capture_output=True, env=env, timeout=600)
    wall = time.monotonic() - t0
    rec = json.loads(p.stdout.decode().strip().splitlines()[-1])
    fsha = hashlib.sha256(open(full, "rb").read()).hexdigest() if os.path.exists(full) else None
    return rec, fsha, p.returncode, p.stderr.decode(errors="replace"), wall

rows = []
for c in cases:
    req = os.path.join(reqdir, c + ".json")
    for m in MODES:
        for e in ENTRIES:
            k = (c, m, e)
            rb_, fb, xb, _, wb = run(pb, req, m, e, "base")
            rh, fh, xh, _, wh = run(ph, req, m, e, "head")
            row = {
                "case": c, "mode": m, "entry": e,
                "exit_base": xb, "exit_head": xh,
                "full_sha_base": fb, "full_sha_head": fh,
                "probe_sha_base": rb_["run"].get("envelope_sha256"), "probe_sha_head": rh["run"].get("envelope_sha256"),
                "i20_full_base": IB[k]["full_envelope_sha256"], "i20_full_cand": IC[k]["full_envelope_sha256"],
                "i20_outcome_cand": IC[k]["outcome"],
                "summary_base_eq_head": rb_["run"].get("envelope") == rh["run"].get("envelope"),
                "summary_base_eq_i20_base": rb_["run"].get("envelope") == RB.get(k),
                "summary_head_eq_i20_cand": rh["run"].get("envelope") == RC.get(k),
            }
            row["full_base_eq_head"] = fb == fh and fb is not None
            row["full_base_eq_i20"] = fb == row["i20_full_base"] == row["probe_sha_base"]
            row["full_head_eq_i20"] = fh == row["i20_full_cand"] == row["probe_sha_head"]
            if pi:
                ri, fi, xi, erri, _ = run(pi, req, m, e, "instr")
                row["full_instr_eq_head"] = fi == fh
                row["witness_lines"] = [l for l in erri.splitlines() if l.startswith("RV24_WITNESS")]
            rows.append(row)
            print(json.dumps(row), flush=True)
            for tag in ("base", "head", "instr"):
                f = os.path.join(out, "full_%s_%s_%s_%s.json" % (tag, c, m, e))
                if os.path.exists(f):
                    os.remove(f)
checks = ["full_base_eq_head", "full_base_eq_i20", "full_head_eq_i20", "summary_base_eq_head",
          "summary_base_eq_i20_base", "summary_head_eq_i20_cand"] + (["full_instr_eq_head"] if pi else [])
for ck in checks:
    print("SUMMARY %s: %d of %d" % (ck, sum(1 for r in rows if r[ck]), len(rows)))
if pi:
    print("SUMMARY runs where the dense witness ran: %d of %d" % (sum(1 for r in rows if r["witness_lines"]), len(rows)))
