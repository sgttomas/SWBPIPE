#!/usr/bin/env python3
"""RV14: huge-coordinate corpus (random bodies at 1e16..1e200 and near 2^1000, and
constructed mechanisms shifted by 2^1000), seed 1405, for the exactness of
`rigid_parameters` (= r/L) and every other check at large L."""
import math, random, sys
sys.path.insert(0, '.')
import rv14_oracle as o
rng = random.Random(1405)
lines = []
orig = o.coord_style
def huge_style(r, style):
    if style == 'huge':
        return r.choice([1.0, -1.0]) * r.choice([1e16, 3e100, 1e200]) * r.randint(1, 5)
    if style == 'p1000':
        return math.ldexp(r.randint(-64, 64) / 8.0, 1000)
    return orig(r, style)
o.coord_style = huge_style
for i in range(400):
    c = o.random_case(rng, f"G{i:04d}")
    # re-draw coordinates in a huge style
    f = c.split('|')
    n = int(f[1])
    style = rng.choice(['huge', 'p1000'])
    coords = [[huge_style(rng, style) for _ in range(3)] for _ in range(n)]
    case = o.parse_case(c)
    lines.append(o.format_case(case['name'], case['subs'], case['ties'], case['grounds'], coords))
k = 0
while k < 300:
    c = o.constructed_mechanism(rng, f"GM{k:04d}", shift=rng.choice([math.ldexp(1.0, 1000), 1e200, 2.0 ** 1020]))
    if c is not None:
        lines.append(c)
        k += 1
open(sys.argv[1], 'w').write("# RV14 huge-coordinate corpus, seed 1405\n" + "\n".join(lines) + "\n")
print(len(lines))
