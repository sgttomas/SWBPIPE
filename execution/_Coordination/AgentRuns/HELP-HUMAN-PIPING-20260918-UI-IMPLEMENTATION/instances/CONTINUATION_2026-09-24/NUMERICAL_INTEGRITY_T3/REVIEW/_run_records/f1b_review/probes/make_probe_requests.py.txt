                  {"id": "N1", "position": {"x": length, "y": 0.0, "z": 0.0}, "provenance": PROV}],
        "pipe_segments": [{"id": "M1", "from": "N0", "to": "N1", "material": "mat:P",
                           "y_reference": {"x": 0.0, "y": 1.0, "z": 0.0},

        "load_cases": [{"id": "case", "label": id, "kind": "primitive_user_load", "primitive_loads": loads, "provenance": PROV}],

def anchor(node, r):
    return {"id": f"anchor:{node}", "node": node, "family": "anchor", "restraints": r, "provenance": PROV}
def spring(node, dof, k):
    unit = "N*m/rad" if dof.startswith("R") else "N/m"
    return {"id": f"spring:{node}:{dof}", "node": node, "family": "spring", "restraints": [dof],
            "stiffness": {"dof": dof, "value": {"value": k, "unit": unit}}, "provenance": PROV}
def force(id, dof, value):
    return {"id": id, "category": "concentrated_force", "target": {"type": "node", "node": "N1"},
            "direction": f"global_{dof[1:].lower()}", "magnitude": {"value": value, "unit": "N"},
            "dimension": "force", "provenance": PROV}
req = {}

     {"id": "support:open-gap", "node": "N1", "family": "nonlinear", "restraints": [],
