"""Counts the rounding classes the RV12 oracle operations exercise."""
import sys
import rv12_oracle as o
from collections import Counter
stats = Counter()
cur = {"op": None, "L": None, "p": None}
orig = o.rnd
def rnd(neg, n, k, sticky, p):
    b = n.bit_length()
    key = (cur["op"], cur["L"])
    if n and b > p:
        shift = b - p
        r = n & ((1 << shift) - 1)
        half = 1 << (shift - 1)
        q = n >> shift
        if r == half and not sticky:
            stats[key + ("tie_even_down" if q % 2 == 0 else "tie_odd_up",)] += 1
        if r == half and sticky:
            stats[key + ("half_plus_sticky",)] += 1
        if r < half and sticky and r == half - 0:
            pass
        if (r > half or (r == half and (sticky or q & 1))) and q + 1 == 1 << p:
            stats[key + ("carry_to_next_binade",)] += 1
        if sticky:
            stats[key + ("sticky",)] += 1
        if shift > 0 and r != 0 and r < (1 << max(0, shift - 64)) :
            stats[key + ("far_sticky_bit",)] += 1
        if cur["p"] is not None and cur["p"] % 64 in (0, 1, 63):
            stats[key + ("limb_boundary_p",)] += 1
    try:
        return orig(neg, n, k, sticky, p)
    except o.RangeErr:
        stats[key + ("exponent_range",)] += 1
        raise
o.rnd = rnd
for line in open(sys.argv[1]):
    f = line.split()
    cur["op"], cur["L"] = f[1], f[2]
    cur["p"] = int(f[3]) if f[3].isdigit() else None
    e = o.expected(f)
    if f[1] == "conv":
        stats[(f[1], f[2], e[0])] += 1
    if f[1] in ("add", "sub"):
        a = o.from_tok(f[4], int(f[2])); b = o.from_tok(f[5], int(f[2]))
        if a[1] and b[1] and not e.startswith("E"):
            r = o.from_tok(e, int(f[2]))
            if r[1] and o.lead(r) < max(o.lead(a), o.lead(b)) - 60:
                stats[(f[1], f[2], "massive_cancellation")] += 1
            if r[1] == 0:
                stats[(f[1], f[2], "exact_zero")] += 1
for k in sorted(stats):
    print(*k, stats[k])
