fn out(case: &str, entry: &str, value: impl std::fmt::Debug) {
    println!("{case}|{entry}|{value:?}");
}

fn ledger_of(f: &[f64], cancel: Option<f64>) -> AssembledForce {
    let mut l = LoadLedger::new();
    for (i, &v) in f.iter().enumerate() {
        if v != 0.0 {
            match cancel {
                None => l.push(format!("load:{i}"), i, v),
                Some(g) => {
                    l.push(format!("big:{i}"), i, g);
                    l.push(format!("load:{i}"), i, v);
                    l.push(format!("back:{i}"), i, -g);
                }
            }
        }
    }
    l.finish(f.len()).unwrap()
}

/// Every base entry on one kernel system (legacy vector `f`).
#[allow(clippy::too_many_arguments)]
fn kernel_case(
    case: &str,
    k: &[Vec<f64>],
    f: &[f64],
    free: &[usize],
    prescribed: &[(usize, f64)],
    contributions: Option<&[StiffnessContribution]>,
    symmetry: Option<(&[Vec<f64>], &[Vec<usize>])>,
    source: Option<&FormationSource>,
) {
    let sym = || {
        symmetry.map(|(r, c)| SymmetryEvidence {
            absolute_roundoff: r,
            operation_counts: c,
            basis: "rv7 basis",
        })
    };
    let sys = StructuralSystem {
        stiffness: k,
        force: f,
        free_dofs: free,
        prescribed,
        contributions,
        symmetry: sym(),
    };
    out(case, "dense", st::solve_structural_dense(&sys));
    out(case, "dense_b64", st::solve_structural_dense_binary64(&sys));
    out(case, "sparse_today", sds::solve_structural_sparse(&sys));
    out(case, "nl_sparse_b64", solve_structural_sparse_binary64(&sys));
    for (tag, g) in [("c3", None), ("c3cancel", Some(1e12))] {
        let force = ledger_of(f, g);
        out(case, &format!("{tag}_dense"), st::solve_structural_dense_with_force_terms(&sys, force.terms()));
        let prepared = st::prepare_structural_with_force_terms(&sys, force.terms());
        if let Ok(p) = &prepared {
            out(case, &format!("{tag}_ldlt"), sds::factor_structural_ldlt(p).and_then(|x| st::finish_structural(&x)));
        } else {
            out(case, &format!("{tag}_prep"), &prepared);
        }
        let typed = StructuralSystem::assembled(k, &force, free, prescribed, contributions, sym());
        out(case, &format!("{tag}_typed_dense"), st::solve_assembled_structural_dense(&typed));
        let tp = st::prepare_assembled_structural(&typed);
        out(case, &format!("{tag}_typed_prepared"), &tp);
        if let Ok(p) = &tp {
            out(case, &format!("{tag}_typed_ldlt"), sds::factor_structural_ldlt(p).and_then(|x| st::finish_structural(&x)));
            out(case, &format!("{tag}_typed_witness"), st::negative_pair_witness(p));
        }
        if let Some(src) = source {
            let typed = StructuralSystem::assembled(k, &force, free, prescribed, contributions, sym());
            let checked = typed.with_formation_source(src);
            out(case, &format!("{tag}_typed_fc_dense"), st::solve_formation_checked_structural_dense(&checked));
            let fp = st::prepare_formation_checked_structural(&checked);
            if let Ok(p) = &fp {
                out(case, &format!("{tag}_typed_fc_ldlt"), sds::factor_structural_ldlt(p).and_then(|x| st::finish_structural(&x)));
            } else {
                out(case, &format!("{tag}_typed_fc_prep"), &fp);
            }
        }
        // Residual and audits at the dense solution (or a compatible guess).
        let u = match st::solve_structural_dense(&sys) {
            Ok(s) => s.displacements,
            Err(_) => {
                let mut u = vec![1.0e-3; f.len()];
                for &(d, v) in prescribed {
                    if d < u.len() {
                        u[d] = v;
                    }
                }
                u
            }
        };
        out(case, &format!("{tag}_residual"), st::evaluate_original_residual(&sys, &u));
        out(case, &format!("{tag}_residual_typed"), st::evaluate_assembled_original_residual(&typed, &u));
        out(case, &format!("{tag}_load_audit"), st::audit_load_fidelity(&sys, &u, force.terms()));
    }
    if let Some(src) = source {
        let checked = StructuralSystem {
            stiffness: k,
            force: f,
            free_dofs: free,
            prescribed,
            contributions,
            symmetry: sym(),
        }
        .with_formation_source(src);
        out(case, "fc_legacy_dense", st::solve_formation_checked_structural_dense(&checked));
    }
    let prepared = st::prepare_structural(&sys);
    out(case, "prepared", &prepared);
    let b64 = st::prepare_structural_binary64(&sys);
    out(case, "prepared_b64", &b64);
    if let Ok(p) = &prepared {
        out(case, "witness", st::negative_pair_witness(p));
        let n = p.matrix().len();
        let mut dirs: Vec<Vec<f64>> = Vec::new();
        for i in 0..n.min(6) {
            for j in 0..i {
                let mut v = vec![0.0; n];
                v[i] = 1.0;
                v[j] = -1.0;
                dirs.push(v.clone());
                v[j] = 0.5;
                dirs.push(v);
            }
        }
        dirs.push((0..n).map(|i| if i % 2 == 0 { 1.0 } else { -0.75 }).collect());
        for (d, v) in dirs.iter().enumerate() {
            out(case, &format!("verify{d}"), st::verify_negative_direction(p, v));
        }
        out(case, "rcond_identity", st::estimate_rcond(p.matrix(), &|x: &[f64]| Ok(x.to_vec())));
        out(case, "cholesky", st::factor_structural_cholesky(p).and_then(|x| st::finish_structural(&x)));
        out(case, "ldlt", sds::factor_structural_ldlt(p).and_then(|x| st::finish_structural(&x)));
        let order: Vec<usize> = (0..n).rev().collect();
        let first = vec![0; n];
        out(case, "profile_rev_full", st::factor_structural_profile(p, &order, &first).and_then(|x| st::finish_structural(&x)));
    }
}

/// A product-shaped case for the adapter (the product's dense order).
struct NiCase {
    name: String,
    node_count: usize,
    frames: Vec<FrameElement>,
    users: Vec<UserStiffnessElement>,
    macros: Vec<CurvedBendMacroElement>,
    slots: Vec<CurvedBendStiffnessElement>,
    springs: Vec<(usize, f64)>,
    prescribed: Vec<(usize, f64)>,
    loads: Vec<(usize, f64)>,
}

impl NiCase {
    fn from_model(m: &ModelData, suffix: &str, e_scale: f64) -> Self {
        let node = |i: usize| FrameNode::new(i, m.nodes[i]).unwrap();
        let s = &m.section;
        let section = FrameSection::new(s.e * e_scale, s.g * e_scale, s.a, s.i, s.i, s.j).unwrap();
        let mut c = NiCase {
            name: format!("{}{}", m.name, suffix),
            node_count: m.nodes.len(),
            frames: vec![],
            users: vec![],
            macros: vec![],
            slots: vec![],
            springs: m.springs.to_vec(),
            prescribed: m.rigid.iter().map(|&d| (d, 0.0)).collect(),
            loads: m.loads.iter().copied().filter(|l| l.1 != 0.0).collect(),
        };
        for (index, member) in m.members.iter().enumerate() {
            match member.bend {
                None => c.frames.push(FrameElement::new(node(member.i), node(member.j), section, member.y_reference).unwrap()),
                Some((center, factor)) => {
                    let e = CurvedBendMacroElement::new(node(member.i), node(member.j), center, s.e * e_scale, s.g * e_scale, s.a, s.i, s.j, factor, factor).unwrap();
                    c.slots.push(CurvedBendStiffnessElement::from_macro_element(format!("bend-{index}"), &e).unwrap());
                    c.macros.push(e);
                }
            }
        }
        c
    }
    fn free(&self) -> Vec<usize> {
        (0..6 * self.node_count).filter(|d| !self.prescribed.iter().any(|p| p.0 == *d)).collect()
    }
    fn dense(&self) -> Result<Vec<Vec<f64>>, String> {
        let mut k = assemble_global_stiffness_with_user_elements(self.node_count, &self.frames, &self.users).map_err(|e| format!("{e:?}"))?;
        for s in &self.slots {
            let map = element_dof_map(s.node_i, s.node_j);
            for (r, &row) in map.iter().enumerate() {
                for (cc, &col) in map.iter().enumerate() {
                    k[row][col] += s.global_stiffness[r][cc];
                }
            }
        }
        for &(dof, v) in &self.springs {
            k[dof][dof] += v;
        }
        Ok(k)
    }
    fn run(&self) {
        let case = &self.name;
        let k = match self.dense() {
            Ok(k) => k,
            Err(e) => {
                out(case, "dense_assembly", e);
                return;
            }
        };
        out(case, "k_bits", k.iter().map(|r| r.iter().map(|v| v.to_bits()).collect::<Vec<_>>()).collect::<Vec<_>>());
        let ev = AssemblyEvidence::new(self.node_count, &self.frames, &self.users, &self.slots, &self.springs);
        out(case, "evidence", &ev);
        let ev = match ev {
            Ok(ev) => ev,
            Err(_) => return,
        };
        let n = 6 * self.node_count;
        let mut fv = vec![0.0; n];
        for &(d, v) in &self.loads {
            fv[d] += v;
        }
        let free = self.free();
        for (tag, cancel) in [("plain", None), ("cancel", Some(1e12))] {
            let force = ledger_of(&fv, cancel);
            for mode in [LinearSolveMode::DenseScrutiny, LinearSolveMode::SparseInteractive] {
                let m = format!("{tag}_{mode:?}");
                out(case, &format!("{m}_solve_assembled"), ev.solve_assembled(&k, &force, &free, &self.prescribed, mode));
                out(case, &format!("{m}_fc_selected"), ev.solve_assembled_with_formation_check(&k, &force, &free, &self.prescribed, mode, &self.macros, true));
                out(case, &format!("{m}_fc_unmatched"), ev.solve_assembled_with_formation_check(&k, &force, &free, &self.prescribed, mode, &[], true));
                out(case, &format!("{m}_fc_not_selected"), ev.solve_assembled_with_formation_check(&k, &force, &free, &self.prescribed, mode, &self.macros, false));
                out(case, &format!("{m}_solve_binary64"), ev.solve_binary64(&k, force.values(), &free, &self.prescribed, mode));
            }
            let basis = "rv7 external basis";
            let sys = StructuralSystem {
                stiffness: &k,
                force: force.values(),
                free_dofs: &free,
                prescribed: &self.prescribed,
                contributions: Some(&ev.contributions),
                symmetry: Some(SymmetryEvidence { absolute_roundoff: &ev.absolute_roundoff, operation_counts: &ev.operation_counts, basis }),
            };
            out(case, &format!("{tag}_nl_sparse_b64"), solve_structural_sparse_binary64(&sys));
            out(case, &format!("{tag}_sparse_today"), sds::solve_structural_sparse(&sys));
            out(case, &format!("{tag}_c3_dense"), st::solve_structural_dense_with_force_terms(&sys, force.terms()));
            if let Ok(p) = st::prepare_structural(&sys) {
                out(case, &format!("{tag}_witness"), st::negative_pair_witness(&p));
            }
        }
        out(case, "qualified", ev.qualified_passive_family());
        out(case, "geometry", ev.geometry(&self.prescribed));
    }
}

fn node(i: usize, p: [f64; 3]) -> FrameNode {
    FrameNode::new(i, p).unwrap()
}

fn product_cases() -> Vec<NiCase> {
    let all = kd5::all();
    let mut cases = Vec::new();
    for m in all {
        cases.push(NiCase::from_model(m, "", 1.0));
        // Second modulus basis: E and G scaled by 0.9 (invented).
        cases.push(NiCase::from_model(m, "-E0.9", 0.9));
        // Settlement on the first rigid DOF, a spring on a prescribed DOF,
        // two springs on one free DOF and a zero spring (invented values).
        let mut c = NiCase::from_model(m, "-settle-springs", 1.0);
        if let Some(p) = c.prescribed.first_mut() {
            p.1 = 0.0015;
        }
        let pd = c.prescribed.last().map(|p| p.0).unwrap_or(0);
        let fd = c.free()[0];
        c.springs.push((pd, 3.5e4));
        c.springs.push((fd, 1.25e3));
        c.springs.push((fd, 7.0e2));
        c.springs.push((fd, 0.0));
        cases.push(c);
    }
    // A mixed model on E6 (bends): a star node joined by four skew members, a
    // duplicate frame, two user elements (one duplicating a member's
    // connectivity), an explicit (untraced) slot, and springs (invented).
    let base = NiCase::from_model(kd5::e6(), "", 1.0);
    let mut c = NiCase::from_model(kd5::e6(), "-mixed", 1.0);
    let s = FrameSection::new(2.0e11, 8.0e10, 0.005969026041820614, 2.700984283923829e-05, 2.700984283923829e-05, 5.401968567847658e-05).unwrap();
    let star = c.node_count;
    c.node_count += 1;
    let sp = [3.1, -1.7, 2.3];
    let nodes: Vec<[f64; 3]> = kd5::e6().nodes.to_vec();
    for &other in &[0usize, 2, 5, 7] {
        if other < nodes.len() {
            c.frames.push(FrameElement::new(node(other, nodes[other]), node(star, sp), s, [0.0, 0.0, 1.0]).unwrap());
        }
    }
    if let Some(f0) = base.frames.first() {
        c.frames.push(*f0);
    }
    c.users.push(UserStiffnessElement::new(node(1, nodes[1]), node(star, sp), [0.0, 0.0, 1.0], 3.0e7, 4.0e6, 2.5e5, 1.5e5).unwrap());
    if let Some(f0) = base.frames.first() {
        c.users.push(UserStiffnessElement::new(f0.node_i, f0.node_j, [0.0, 0.0, 1.0], 2.0e7, 1.0e6, 3.5e5, 2.5e5).unwrap());
    }
    let stand_in = FrameElement::new(node(3, nodes[3]), node(star, sp), s, [0.0, 0.0, 1.0]).unwrap();
    let gk: Matrix12 = stand_in.global_stiffness().unwrap();
    c.slots.push(CurvedBendStiffnessElement::new("explicit-slot", 3, star, gk).unwrap());
    c.springs.push((6 * star + 2, 2.0e5));
    c.springs.push((6 * star + 2, 2.5e4));
    c.springs.push((c.prescribed[0].0, 9.0e3));
    c.loads.push((6 * star + 1, -850.0));
    c.loads.push((6 * star + 3, 14.0));
    c.prescribed[1].1 = -0.0025;
    cases.push(c);
    cases
}

fn main() {
    references();
    for c in product_cases() {
        c.run();
    }
    // Frame-only formation-checked FK entry on the settled chain (invented).
    let s = FrameSection::new(2.0e11, 8.0e10, 0.005969026041820614, 2.700984283923829e-05, 2.700984283923829e-05, 5.401968567847658e-05).unwrap();
    let pts: Vec<[f64; 3]> = (0..7).map(|k| { let t = k as f64; [1.1 * t, 0.37 * t + 0.05 * (t * 0.7).sin(), 0.21 * t] }).collect();
    let frames: Vec<FrameElement> = (0..6).map(|k| FrameElement::new(node(k, pts[k]), node(k + 1, pts[k + 1]), s, [0.0, 0.0, 1.0]).unwrap()).collect();
    let springs = vec![(6 * 6 + 3, 1.0e2)];
    let mut k = assemble_global_stiffness_with_user_elements(7, &frames, &[]).unwrap();
    for &(d, v) in &springs {
        k[d][d] += v;
    }
    let ev = AssemblyEvidence::new(7, &frames, &[], &[], &springs).unwrap();
    let mut f = vec![0.0; 42];
    f[6 * 6 + 1] = -1200.0;
    f[6 * 6 + 5] = 35.0;
    let prescribed: Vec<(usize, f64)> = (0..6).map(|d| (d, if d == 1 { 0.003 } else { 0.0 })).collect();
    let free: Vec<usize> = (6..42).collect();
    let src = FormationSource { node_count: 7, frames: frames.clone(), users: vec![], curved: vec![], springs: springs.clone(), unavailable: vec![] };
    kernel_case("chain6-settle", &k, &f, &free, &prescribed, Some(&ev.contributions), Some((&ev.absolute_roundoff, &ev.operation_counts)), Some(&src));
    let src_un = FormationSource { unavailable: vec!["rv7-seed".into()], ..src.clone() };
    kernel_case("chain6-settle-unavail", &k, &f, &free, &prescribed, Some(&ev.contributions), Some((&ev.absolute_roundoff, &ev.operation_counts)), Some(&src_un));
    let rev: Vec<usize> = free.iter().rev().copied().collect();
    kernel_case("chain6-settle-revfree", &k, &f, &rev, &prescribed, Some(&ev.contributions), Some((&ev.absolute_roundoff, &ev.operation_counts)), Some(&src));
    kernel_case("chain6-bare", &k, &f, &free, &prescribed, None, None, None);
}
