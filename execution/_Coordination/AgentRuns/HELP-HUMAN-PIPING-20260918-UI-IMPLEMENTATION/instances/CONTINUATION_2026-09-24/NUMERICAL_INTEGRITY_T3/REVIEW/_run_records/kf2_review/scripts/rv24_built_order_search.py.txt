"""RV24: search for a *built* 2x2 system (all DOFs free, diagonals in [1, 4) so every exponent is 0,
positive couplings k_ij = k and k_ji = k + m ulps, allowed by symmetry evidence of m ulps) on which
the verifier's cell order and the order with the two coupling cells swapped give different verdicts.
binary64 in Python follows IEEE 754 round-to-nearest, as the Rust code does."""
import math, random
u = 2.0 ** -53
def gamma(k):
    x = k * u
    return x / (1 - x)
G = 64.0 * gamma(14)
def verdict(terms):
    e = 0.0; m = 0.0
    for t in terms:
        e = e + t; m = m + abs(t)
    return e < -(G * m), e
random.seed(0x52563234)
found = []
for trial in range(400000):
    a = 1.0 + 3.0 * random.random()
    b = 1.0 + 3.0 * random.random()
    k = (a + b) / 2.0
    for _ in range(random.randrange(0, 200)):
        k = math.nextafter(k, math.inf)
    kt = k
    for _ in range(random.randrange(1, 5)):
        kt = math.nextafter(kt, math.inf)
    # source: K[1][0] = k (lower, cell (i,j)), K[0][1] = kt (upper, cell (j,i)); s = -1 (coupling >= 0)
    fwd = verdict([a, -kt, -k, b])      # verifier order: (j,j), (j,i), (i,j), (i,i)
    swp = verdict([a, -k, -kt, b])      # coupling cells swapped
    if fwd[0] != swp[0] and 1.0 <= b < 4.0 and 1.0 <= a < 4.0:
        found.append((a, b, k, kt, fwd, swp))
        if len(found) >= 3:
            break
print("trials", trial + 1, "found", len(found))
for a, b, k, kt, f, s in found:
    print("a=%r b=%r k_lower=%r k_upper=%r skew_ulps=%d verifier_order=%s swapped=%s" %
          (a, b, k, kt, round((kt - k) / math.ulp(k)), f, s))
