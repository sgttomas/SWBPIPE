"""RV24: the instrumented candidate over gate part 1's requests (the reviewer's copy of head f2b8c85a2
with one stderr line in `negative_pair_witness`; P1's full-envelope probe cd1052f7...).
Usage: rv24_gate_instr.py <probe> <gen_out> <index_cand.tsv> <out.jsonl> <workers> [--only CASE ...]
Runs every part-1 (case, mode, entry) except the six 10,000-member cases (host rule: no dense matrix at
10,000 members; their dense runs are refused by F1b's guard before any n^2 allocation, and their sparse
runs take the pattern route), in fresh processes, `workers` at a time. For each run it records whether
the dense witness ran (its stderr lines), and compares the full envelope's sha256 with I20's recorded
candidate index."""
import concurrent.futures as cf, csv, hashlib, json, os, subprocess, sys, tempfile, time

probe, gen, idx, outp, workers = sys.argv[1:6]
only = sys.argv[7:] if len(sys.argv) > 6 and sys.argv[6] == "--only" else None
IDX = {(r["case"], r["mode"], r["entry"]): r for r in csv.DictReader(open(idx), delimiter="\t")}
cases = json.load(open(os.path.join(gen, "cases.json")))
jobs = []
for c in cases:
    if "request_file" not in c or c.get("members", 0) >= 10000:
        continue
    if only and c["id"] not in only:
        continue
    for m in ("dense_scrutiny", "sparse_interactive"):
        for e in ("captured", "typed"):
            k = (c["id"], m, e)
            if k in IDX or only:
                jobs.append((k, os.path.join(gen, c["request_file"]), c.get("members", 0)))
tmp = tempfile.mkdtemp(prefix="rv24_full_", dir=os.path.dirname(os.path.abspath(outp)))

def run(job):
    (case, mode, entry), req, members = job
    full = os.path.join(tmp, "%s_%s_%s.json" % (case, mode, entry))
    env = dict(os.environ, T3_FULL_ENVELOPE_PATH=full)
    args = [probe, req, mode] + (["typed"] if entry == "typed" else [])
    t0 = time.monotonic()
    try:
        p = subprocess.run(args, capture_output=True, env=env, timeout=1800)
        rc, so, se, to = p.returncode, p.stdout, p.stderr, False
    except subprocess.TimeoutExpired as x:
        rc, so, se, to = None, x.stdout or b"", x.stderr or b"", True
    wall = time.monotonic() - t0
    fsha = None
    if os.path.exists(full):
        fsha = hashlib.sha256(open(full, "rb").read()).hexdigest()
        os.remove(full)
    try:
        rec = json.loads(so.decode().strip().splitlines()[-1])["run"]
    except Exception:
        rec = {}
    lines = [l for l in se.decode(errors="replace").splitlines() if l.startswith("RV24_WITNESS")]
    k = (case, mode, entry)
    i20 = IDX.get(k, {}).get("full_envelope_sha256")
    i20 = None if i20 in (None, "-", "") else i20
    return {"case": case, "mode": mode, "entry": entry, "members": members, "exit": rc, "timed_out": to,
            "wall": round(wall, 3), "full_sha": fsha, "probe_sha": rec.get("envelope_sha256"),
            "i20_full_sha": i20, "full_eq_i20": fsha == i20 and fsha == rec.get("envelope_sha256"),
            "witness_calls": lines}

rows = []
with cf.ThreadPoolExecutor(int(workers)) as ex, open(outp, "w") as fh:
    for r in ex.map(run, jobs):
        rows.append(r)
        fh.write(json.dumps(r) + "\n"); fh.flush()
os.rmdir(tmp)
n = len(rows)
print("runs", n)
print("full envelope equal to I20's candidate index:", sum(r["full_eq_i20"] for r in rows), "of", n)
print("timeouts", sum(r["timed_out"] for r in rows), "nonzero exits", sum(1 for r in rows if r["exit"] not in (0,)))
w = [r for r in rows if r["witness_calls"]]
print("runs where the dense witness ran:", len(w))
for r in w:
    print("  %s %s %s: %s" % (r["case"], r["mode"], r["entry"], " | ".join(r["witness_calls"])))
