#!/bin/bash
# RV18: one runner mutant from a clean archive; the runner's 35 tests (unittest, as the wrapper collects them).
set -u
W=<wt>
M=$1
ROOT=$W/rv18/pymut/$M
rm -rf $ROOT; mkdir -p $ROOT $W/rv18/mut-logs
(cd $W/k6 && git archive ae3320b5a9 projects/chirality-piping/core/solver/performance_harness) | tar -x -C $ROOT
if [ "$M" != NONE-py ]; then python3 -B $W/rv18/scripts/rv18_mutants.py make $M $ROOT || exit 9; fi
cd $ROOT/projects/chirality-piping/core/solver/performance_harness/runner
start=$(date +%s)
PYTHONDONTWRITEBYTECODE=1 python3 -m unittest test_k6_runner > $W/rv18/mut-logs/py-$M.log 2>&1
code=$?
echo "=== $M exit $code elapsed $(( $(date +%s) - start )) s" >> $W/rv18/mut-logs/py-$M.log
echo "$M exit $code"
