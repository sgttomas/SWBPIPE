+/// or a forced non-refusal stop (`Arithmetic`) of one block's operation in one
+/// pass.
+pub(crate) mod rv23_force {
+    use super::super::wide::WideError;
+    use super::AttemptStop;
+    use std::cell::Cell;
+    thread_local! {
+        static AT: Cell<Option<(usize, u8)>> = const { Cell::new(None) };
+        static FAIL: Cell<Option<(usize, u8)>> = const { Cell::new(None) };
+    }
+    pub(crate) fn set(v: Option<(usize, u8)>) {
+        AT.with(|c| c.set(v));
+    }
+    pub(crate) fn set_fail(v: Option<(usize, u8)>) {
+        FAIL.with(|c| c.set(v));
+    }
+    pub(crate) fn wrap<T>(block: usize, pass: u8, r: Result<T, AttemptStop>) -> Result<T, AttemptStop> {
+        if FAIL.with(|c| c.get()) == Some((block, pass)) {
+            return Err(AttemptStop::Arithmetic(WideError::NonFinite));
+        }
+        if AT.with(|c| c.get()) == Some((block, pass)) {
+            match r {
+                Ok(_) => Err(AttemptStop::Span),
+                Err(e) => Err(e),
+            }
+        } else {
+            r
+        }
+    }
