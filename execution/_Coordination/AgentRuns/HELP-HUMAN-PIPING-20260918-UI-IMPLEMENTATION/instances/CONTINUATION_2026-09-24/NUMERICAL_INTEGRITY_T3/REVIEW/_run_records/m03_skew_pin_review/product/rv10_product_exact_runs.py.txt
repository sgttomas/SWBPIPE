"""RV10: an independent exact (Fraction) solve of the free node for the sensitive runs of I9's product probe,
re-run by RV10 from a git archive of 134eefc24. Inputs taken exactly from the probe's printed bits (E, G, moment;
A, I, J, L; FK's axes). The free node's 6x6 global block is T^T K_l T with K_l main's local (j, j) block at the
intended coefficients k*E*I/L^n, E*A/L, G*J/L; the load is a global RZ moment at N1."""
import json, struct, sys, collections
from fractions import Fraction as F
fb = lambda b: struct.unpack('<d', struct.pack('<Q', b))[0]
fx = lambda b: F(fb(b))
lines = open(sys.argv[1]).read().splitlines()
sec = json.loads(next(l for l in lines if l.startswith('I9SECTION '))[10:])
geo = {g['member']: g for g in (json.loads(l[7:]) for l in lines if l.startswith('I9GEOM '))}
A, I, J, L = (fx(sec[k]) for k in ('A', 'I', 'J', 'L'))
def solve(M, b):
    n = len(b); M = [row[:] + [b[i]] for i, row in enumerate(M)]
    for c in range(n):
        p = next(r for r in range(c, n) if M[r][c] != 0); M[c], M[p] = M[p], M[c]
        for r in range(n):
            if r != c and M[r][c] != 0:
                f = M[r][c] / M[c][c]; M[r] = [x - f * y for x, y in zip(M[r], M[c])]
    return [M[i][n] / M[i][i] for i in range(n)]
cache = {}
def exact(member, E, G, m):
    key = (member, E, G, m)
    if key in cache: return cache[key]
    g = geo[member]; assert fx(g['length']) == L
    R = [[fx(v) for v in row] for row in g['axes']]
    ea, gj = E * A / L, G * J / L
    k12, k6, k4 = 12 * E * I / L**3, 6 * E * I / L**2, 4 * E * I / L
    Kl = [[0] * 6 for _ in range(6)]
    Kl[0][0] = ea; Kl[3][3] = gj
    Kl[1][1] = k12; Kl[1][5] = Kl[5][1] = -k6; Kl[5][5] = k4      # z-bending, node j block
    Kl[2][2] = k12; Kl[2][4] = Kl[4][2] = k6; Kl[4][4] = k4       # y-bending, node j block
    T = [[F(0)] * 6 for _ in range(6)]
    for a in range(3):
        for b in range(3):
            T[a][b] = R[a][b]; T[a + 3][b + 3] = R[a][b]
    KT = [[sum(Kl[r][k] * T[k][c] for k in range(6)) for c in range(6)] for r in range(6)]
    Kg = [[sum(T[k][r] * KT[k][c] for k in range(6)) for c in range(6)] for r in range(6)]
    u = solve(Kg, [F(0)] * 5 + [m])
    cache[key] = u
    return u
names = ['ux', 'uy', 'uz', 'rx', 'ry', 'rz']
out = collections.Counter(); worst = {}; runs = []
for l in lines:
    if not l.startswith('I9PROBE '): continue
    r = json.loads(l[8:]); out[r['status']] += 1
    assert r['status'] != 'checks_passed' and not any(c[1] == 'checks_passed' for c in r['cases'])
    if not r['N1']:
        continue
    u = exact(r['member'], fx(r['E']), fx(r['G']), fx(r['moment']))
    exp = [float(u[i] * (1000 if i < 3 else 1)) for i in range(6)]
    pub = {x[0].rsplit(':', 1)[1]: x[1] for x in r['N1']}
    st = max(abs(e) for e in exp[:3]); sr = max(abs(e) for e in exp[3:])
    err = max(abs(pub[n] - exp[i]) / max(abs(exp[i]), st if i < 3 else sr) for i, n in enumerate(names))
    cell = (r['t'], r['g_label'], r['member'])
    worst[cell] = max(worst.get(cell, 0), err); runs.append((err, cell, r["mode"], r["entry"]))
    out[('published', r['status'], r['entry'], r['cases'][0][1])] += 1
print(dict(out))
print(f"cells with published values: {len(worst)}; runs over 1e-9: see per cell")
for cell, e in sorted(worst.items()):
    print(f"  t={cell[0]} G={cell[1]:7} {cell[2]}: worst criterion error {e:.3e}")
print(f"range {min(worst.values()):.2e} .. {max(worst.values()):.2e}; cells > 1e-9: {sum(e > 1e-9 for e in worst.values())}")
print(f"per run: {len(runs)} published runs; min {min(runs)[0]:.2e}; max {max(runs)[0]:.2e}; > 1e-9: {sum(e > 1e-9 for e, *_ in runs)}")
for e, cell, mode, entry in sorted(runs)[:12]:
    print(f"   {e:.3e} {cell} {mode} {entry}")
