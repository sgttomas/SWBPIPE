#!/bin/bash
WT=<wt>
cd $WT/rv22c/head/projects/chirality-piping/core/solver/performance_harness || exit 1
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=$WT/rv22c-target RUST_TEST_THREADS=2
start=$(date +%s)
cargo test --offline --locked -j 4 --all-targets > $WT/rv22c/logs/h_suite.log 2>&1
rc=$?
echo "rc=$rc wall=$(( $(date +%s) - start ))s" >> $WT/rv22c/logs/h_suite.log
cargo build --offline --locked -j 4 --release --bin k6_observe > $WT/rv22c/logs/release_build.log 2>&1
echo "release rc=$?" >> $WT/rv22c/logs/release_build.log
