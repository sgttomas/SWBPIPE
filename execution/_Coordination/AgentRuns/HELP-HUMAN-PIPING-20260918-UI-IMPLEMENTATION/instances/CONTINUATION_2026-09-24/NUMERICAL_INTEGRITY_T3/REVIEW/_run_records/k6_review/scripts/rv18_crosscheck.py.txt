"""RV18: independent cross-check of K6's 24 RF-LARGE kernel models against R1.

Usage: python3 rv18_crosscheck.py <references.py> <k6_observe binary> <models_sha256.txt>

Written by RV18 from R1's references.py (lg_chain_defn, lg_tree_defn, lg_cont_defn, rot_defn,
model_json) and P1's gen.py y_reference rule; it reuses nothing of K6's checker. For every case
it compares `model_json(defn, full=True)` (what `references.py --model` prints) with the canonical
bytes the Rust binary emits (`--emit-model`), each exact input rounded once to binary64:
node order, labels and coordinate bits; member order, labels and end nodes; y_reference (P1's
rule, recomputed from the exact coordinates); E and G bits; A, I and J against the stated
formula; the restraint sets; every load component and no extra load. It also checks the
sha256 of the emitted bytes against models_sha256.txt. Standard library only.
"""
import hashlib
import importlib.util
import json
import math
import struct
import subprocess
import sys
from fractions import Fraction as Fr

DOFS = ('UX', 'UY', 'UZ', 'RX', 'RY', 'RZ')


def hexbits(x):
    return struct.pack('>d', float(x)).hex()


def main():
    ref_path, binary, hashes_path = sys.argv[1:4]
    spec = importlib.util.spec_from_file_location('references', ref_path)
    R = importlib.util.module_from_spec(spec)
    sys.modules['references'] = R
    spec.loader.exec_module(R)
    R.build_large()
    listed = {}
    for line in open(hashes_path):
        if line.strip() and not line.startswith('#'):
            d, i = line.split()[:2]
            listed[i] = d
    cases = [c for c in R.CASES if c['id'].startswith('RF-LARGE-')]
    ok_all = True
    for case in cases:
        cid = case['id']
        mj = R.model_json(case['defn'], full=True)
        raw = subprocess.run([binary, '--emit-model', '--model', cid], capture_output=True, check=True).stdout
        problems = []
        if hashlib.sha256(raw).hexdigest() != listed.get(cid):
            problems.append('sha256 differs from models_sha256.txt')
        lines = raw.decode('ascii').splitlines()
        nodes, members, restraints, loads, section = [], [], {}, {}, None
        for ln in lines:
            t = ln.split(' ')
            if t[0] == 'node':
                nodes.append((t[2], t[3:6]))
            elif t[0] == 'member':
                members.append((t[2], int(t[3]), int(t[4]), t[5:8]))
            elif t[0] == 'restraint':
                restraints[int(t[1])] = t[2]
            elif t[0] == 'load':
                loads[int(t[1])] = t[2]
            elif t[0] == 'section':
                section = dict(zip(t[1::2], t[2::2]))
        # nodes: order, labels, bits
        ref_nodes = list(mj['nodes_m'].items())
        index = {label: k for k, (label, _) in enumerate(ref_nodes)}
        exact = {label: [R.parse_input(c) for c in v] for label, v in ref_nodes}
        if [l for l, _ in ref_nodes] != [l for l, _ in nodes]:
            problems.append('node order or labels differ')
        else:
            for (label, bits) in nodes:
                if [hexbits(c) for c in exact[label]] != bits:
                    problems.append('node %s bits' % label)
        # members: order, labels, ends, y_reference
        ref_members = mj['members']
        if len(ref_members) != len(members):
            problems.append('member count')
        for (rm, km) in zip(ref_members, members):
            label, i, j, sec = rm
            if (label, index[i], index[j]) != km[:3]:
                problems.append('member %s' % label)
                continue
            e = [abs(exact[j][k] - exact[i][k]) for k in range(3)]
            best = min((1, 2, 0), key=lambda k: e[k])
            y = [1.0 if k == best else 0.0 for k in range(3)]
            if [hexbits(c) for c in y] != km[3]:
                problems.append('member %s y_reference' % label)
        # section
        (sname, s), = mj['sections'].items()
        E, G, OD, ID = (R.parse_input(s[k]) for k in ('E', 'G', 'OD', 'ID'))
        od, id_ = float(OD), float(ID)
        od2, id2 = od * od, id_ * id_
        A = math.pi * (od2 - id2) / 4.0
        I = math.pi * (od2 * od2 - id2 * id2) / 64.0
        want = {'E': hexbits(E), 'G': hexbits(G), 'A': hexbits(A), 'Iy': hexbits(I), 'Iz': hexbits(I),
                'J': hexbits(2.0 * I)}
        if section != want:
            problems.append('section %s != %s' % (section, want))
        # restraints
        ref_r = {}
        for label, sup in mj['supports'].items():
            if sup.get('springs'):
                problems.append('springs at %s' % label)
            ref_r[index[label]] = ''.join('1' if d in sup['rigid'] else '0' for d in DOFS)
        if ref_r != restraints:
            problems.append('restraints differ')
        # loads
        ref_l = {}
        for label, ld in mj['loads'].items():
            for kind, vec in ld.items():
                off = {'F': 0, 'M': 3}[kind]
                for a, c in enumerate(vec):
                    q = R.parse_input(c)
                    if q != 0:
                        f = float(q)
                        if Fr(f) != q:
                            problems.append('load %s %s not exact in binary64' % (label, kind))
                        ref_l[index[label] * 6 + off + a] = hexbits(f)
        if ref_l != loads:
            problems.append('loads differ (%d ref, %d k6)' % (len(ref_l), len(loads)))
        ok = not problems
        ok_all &= ok
        print('%s %s nodes=%d members=%d loads=%d %s' % ('EQUAL' if ok else 'DIFFER', cid, len(nodes), len(members),
                                                        len(loads), '; '.join(problems[:5])))
    print('summary: %d cases, all equal: %s' % (len(cases), ok_all))
    exact_A = Fr(R.PI_Q) * (Fr('0.2') ** 2 - Fr('0.18') ** 2) / 4 if hasattr(R, 'PI_Q') else None
    if exact_A is not None:
        A = math.pi * (0.2 * 0.2 - 0.18 * 0.18) / 4.0
        print('section A vs exact (PI_Q): rel %.3e' % float((Fr(A) - exact_A) / exact_A))
    return 0 if ok_all else 1


if __name__ == '__main__':
    sys.exit(main())
