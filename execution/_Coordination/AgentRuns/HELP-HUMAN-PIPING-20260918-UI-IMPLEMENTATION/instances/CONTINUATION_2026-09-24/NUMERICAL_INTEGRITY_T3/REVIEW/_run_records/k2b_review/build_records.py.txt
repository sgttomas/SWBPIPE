#!/usr/bin/env python3
"""RV11 (K2b review): copy the review's run evidence into REVIEW/_run_records/k2b_review/, sanitized.
- Machine paths become placeholders, longest first: <VENV>, <wt> (the T3 worktrees root; the reviewer's
  scratch is <wt>/scratch/rv11 and its target <wt>/rv11-target), {REPO_ROOT} (the outer worktree),
  <session-scratch> (the host session's scratchpad) and <home>.
- A log line longer than 600 characters is cut to 600, followed by a marker with the number of characters
  removed and the sha256 of the full line. Summary files are never cut.
- Trailing whitespace and trailing blank lines are stripped; every file ends with one newline.
- Scripts and sources are stored with a .txt suffix. SHA256SUMS covers every file but itself.
Usage: build_records.py <wt> <repo-root> <venv> <session-scratch> <home> <out-dir>"""
import hashlib, os, sys

wt, repo, venv, pad, home, out = sys.argv[1:7]
PLACES = sorted([(venv, "<VENV>"), (wt, "<wt>"), (repo, "{REPO_ROOT}"), (pad, "<session-scratch>"),
                 (pad.replace("/" + "private" + "/" + "tmp" + "/", "/" + "tmp" + "/"), "<session-scratch>"), (home, "<home>")],
                key=lambda p: -len(p[0]))
LIMIT = 600
R = os.path.join(wt, "scratch", "rv11")
L = os.path.join(R, "logs")
M = os.path.join(R, "mut")
files = []

def put(src, rel, cut=False):
    with open(src, encoding="utf-8", errors="replace") as f:
        text = f.read()
    for path, name in PLACES:
        text = text.replace(path, name)
    lines = []
    for line in text.split("\n"):
        line = line.rstrip()
        if cut and len(line) > LIMIT:
            digest = hashlib.sha256(line.encode()).hexdigest()
            line = line[:LIMIT] + f" [... RV11-records cut: {len(line) - LIMIT} more characters; sha256 of the full line {digest}]"
        lines.append(line)
    while lines and lines[-1] == "":
        lines.pop()
    dst = os.path.join(out, rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    with open(dst, "w", encoding="utf-8") as f:
        f.write("\n".join(lines) + "\n")
    files.append(rel)

put(os.path.join(R, "README.txt"), "README.txt")
put(os.path.join(L, "toolchain_raw.txt"), "toolchain.txt")
put(os.path.join(L, "merge_check.txt"), "merge_check.txt")
put(os.path.join(L, "records_checks.txt"), "records_checks.txt")
put(os.path.join(L, "gen8.txt"), "gen8.txt")
put(os.path.join(R, "run_head.sh"), "head_tests/run_head.sh.txt")
put(os.path.join(L, "head_087b3a088.log"), "head_tests/head_087b3a088.log", cut=True)
put(os.path.join(R, "lexscan.py"), "callers/lexscan.py.txt")
put(os.path.join(L, "lexscan_087b3a088.txt"), "callers/lexscan_087b3a088.txt")
put(os.path.join(L, "lexscan_eb52114e9.txt"), "callers/lexscan_eb52114e9.txt")
put(os.path.join(R, "b0probe", "run_b0probe.sh"), "b0probe/run_b0probe.sh.txt")
put(os.path.join(R, "b0probe", "sha_eb52114e9.txt"), "b0probe/sha_eb52114e9.txt")
put(os.path.join(R, "b0probe", "sha_087b3a088.txt"), "b0probe/sha_087b3a088.txt")
put(os.path.join(L, "b0probe_compare.txt"), "b0probe/b0probe_compare.txt")
put(os.path.join(R, "probe", "projects/chirality-piping/core/solver/nonlinear_integration/src/structural_adapter/rv11_probe.rs"),
    "probes/rv11_probe.rs.txt")
put(os.path.join(L, "probe_final.log"), "probes/probe_final.log", cut=True)
put(os.path.join(L, "float_boundary.txt"), "probes/float_boundary.txt")
for name in ["mutate_rv11.py", "run_mutant.sh", "drive.sh", "apply.sh"]:
    put(os.path.join(M, name), f"mutations/{name}.txt")
put(os.path.join(M, "MUTANTS.txt"), "mutations/MUTANTS.txt")
put(os.path.join(M, "drive.log"), "mutations/drive.log")
put(os.path.join(R, "mutators.txt"), "mutations/mutators.txt")
for name in sorted(os.listdir(os.path.join(M, "logs"))):
    put(os.path.join(M, "logs", name), f"mutations/logs/{name}", cut=True)
put(os.path.abspath(__file__), "build_records.py.txt")
with open(os.path.join(out, "SHA256SUMS"), "w") as f:
    for rel in sorted(files):
        with open(os.path.join(out, rel), "rb") as g:
            f.write(f"{hashlib.sha256(g.read()).hexdigest()}  ./{rel}\n")
print(len(files), "files")
