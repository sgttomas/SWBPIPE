"""RV24: arithmetic check of KF2 plan section 7.2 (the dense screen at N10's refusing rows).
Standard library only; the model data are R1's SEC_N as the plan states them (E 200 GPa, OD 0.2 m,
ID 0.18 m, 3 m members), and the counts follow from the DOF numbering (6 DOFs per node, N0 restrained)."""
import math
u = 2.0 ** -53
def g(m): return m * u / (1 - m * u)
E = 200e9; od, idd = 0.2, 0.18
A = math.pi * (od**2 - idd**2) / 4; I = math.pi * (od**4 - idd**4) / 64
Le = 3.0
print("EA/L_e = %.4e  12EI/L_e^3 = %.4e" % (E*A/Le, 12*E*I/Le**3))
for name, dof, first_global in [("CHAIN-n01000-ROT N1000.UY", 6001, 5994), ("TREE-n01000-AX B500.UZ", 6002, 5996)]:
    i = dof - 6; f = first_global - 6
    dense, prof = 2*i + 2, 2*(i - f) + 2
    print("%s: ordered i=%d dense count=%d 64g=%.4e | first f_i=%d profile count=%d 64g=%.4e" % (name, i, dense, 64*g(dense), f, prof, 64*g(prof)))
# CHAIN-ROT: axis (1,2,-2)/3, tip of a 3,000 m cantilever, rotations clamped (guided), UZ clamped, UX condensed.
Ltot = 3000.0
ka, kt = E*A/Ltot, 12*E*I/Ltot**3
ax, ay = 1/3, 2/3
Kxx = ka*ax*ax + kt*(1 - ax*ax); Kxy = (ka - kt)*ax*ay; Kyy = ka*ay*ay + kt*(1 - ay*ay)
piv = Kyy - Kxy*Kxy/Kxx
aii = E*A/Le*ay*ay + 12*E*I/Le**3*(1 - ay*ay)
print("CHAIN-ROT: guided-tip k_t=%.4e N/m, Schur pivot=%.4e (=%.3f k_t), a_ii=%.4e, pivot/(2 a_ii)=%.3e" % (kt, piv, piv/kt, aii, piv/(2*aii)))
# TREE-AX: spine 1,500 m, P500 rotation effectively clamped by the branch (EI/L_e >> EI/L_spine); branch axial in series.
Ls = 1500.0
ks = 12*E*I/Ls**3; kb = E*A/Le
piv2 = 1/(1/ks + 1/kb)
print("TREE-AX: spine guided 12EI/L^3=%.4e, with branch axial in series=%.4e, a_ii=%.4e, pivot/(2 a_ii)=%.3e; rotational restraint ratio (EI/L_e)/(EI/L_spine)=%.0f" % (ks, piv2, kb, piv2/(2*kb), Ls/Le))
for name, r in [("CHAIN-ROT", piv/(2*aii)), ("TREE-AX", piv2/(2*kb))]:
    print("%s: dense screen 64g(count)=%.3e -> %s; profile screen -> %s (margin x%.0f)" % (name, 64*g(11992), "refuses" if r <= 64*g(11992) else "passes", "passes" if r > 64*g(16) else "refuses", r/(64*g(16 if name=="CHAIN-ROT" else 14))))
