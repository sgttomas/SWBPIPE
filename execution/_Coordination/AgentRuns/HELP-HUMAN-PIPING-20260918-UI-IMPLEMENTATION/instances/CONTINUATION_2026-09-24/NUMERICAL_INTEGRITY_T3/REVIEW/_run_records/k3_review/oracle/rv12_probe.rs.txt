//! RV12 review probe (scratch copy only; never committed). Reads operations
//! from the file named by RV12_IN, applies K3's public (crate) API, and writes
//! one result line per operation to RV12_OUT. The expected values are computed
//! separately by the reviewer's own Python oracle.
use super::super::{Wide, WideError};
use super::*;
use std::fmt::Write as _;

fn parse_limbs<const L: usize>(hex: &str) -> [u64; L] {
    assert_eq!(hex.len(), 16 * L, "limb hex length {hex}");
    let mut s = [0u64; L];
    for k in 0..L {
        s[L - 1 - k] = u64::from_str_radix(&hex[16 * k..16 * k + 16], 16).unwrap();
    }
    s
}

/// token: <sign 0|1>:<exponent>:<16L hex>
fn val<const L: usize>(tok: &str) -> Wide<L>
where
    Wide<L>: SupportedWidth,
{
    let mut it = tok.split(':');
    let neg = it.next().unwrap() == "1";
    let e: i64 = it.next().unwrap().parse().unwrap();
    let s = parse_limbs::<L>(it.next().unwrap());
    Wide::<L>::from_parts(neg, e, s).unwrap()
}

fn val2(tok: &str) -> Wide<2> {
    let mut it = tok.split(':');
    let neg = it.next().unwrap() == "1";
    let e: i64 = it.next().unwrap().parse().unwrap();
    let s = parse_limbs::<2>(it.next().unwrap());
    Wide::<2>::from_parts(neg, e, s).unwrap()
}

fn show_parts(neg: bool, e: i64, s: &[u64]) -> String {
    let mut out = String::new();
    write!(out, "{}:{}:", u8::from(neg), e).unwrap();
    for limb in s.iter().rev() {
        write!(out, "{limb:016x}").unwrap();
    }
    out
}

fn show<const L: usize>(w: &Wide<L>) -> String
where
    Wide<L>: SupportedWidth,
{
    let (n, e, s) = w.parts();
    show_parts(n, e, &s)
}

fn show_res<const L: usize>(r: Result<Wide<L>, WideError>) -> String
where
    Wide<L>: SupportedWidth,
{
    match r {
        Ok(w) => show(&w),
        Err(e) => format!("E:{e:?}"),
    }
}

fn show_outcome(o: Binary64Outcome) -> String {
    match o {
        Binary64Outcome::Normal(v) => format!("N:{:016x}", v.to_bits()),
        Binary64Outcome::Subnormal {
            value,
            relative_precision,
        } => format!("S:{:016x}:{:016x}", value.to_bits(), relative_precision.to_bits()),
        Binary64Outcome::Underflow { negative } => format!("U:{}", u8::from(negative)),
        Binary64Outcome::Overflow { negative } => format!("O:{}", u8::from(negative)),
    }
}

fn run_l<const L: usize>(f: &[&str]) -> String
where
    Wide<L>: SupportedWidth,
{
    let op = f[1];
    match op {
        "add" | "sub" | "mul" | "div" | "sqrt" | "twosum" | "twoprod" => {
            let p: u32 = f[3].parse().unwrap();
            let mut c = WideContext::<L>::new(p).unwrap();
            let a = val::<L>(f[4]);
            let r = match op {
                "add" => show_res(c.add(&a, &val::<L>(f[5]))),
                "sub" => show_res(c.sub(&a, &val::<L>(f[5]))),
                "mul" => show_res(c.mul(&a, &val::<L>(f[5]))),
                "div" => show_res(c.div(&a, &val::<L>(f[5]))),
                "sqrt" => show_res(c.sqrt(&a)),
                "twosum" | "twoprod" => {
                    let b = val::<L>(f[5]);
                    let r = if op == "twosum" {
                        c.two_sum(&a, &b)
                    } else {
                        c.two_product(&a, &b)
                    };
                    match r {
                        Ok((s, e)) => format!("{};{}", show(&s), show(&e)),
                        Err(e) => format!("E:{e:?}"),
                    }
                }
                _ => unreachable!(),
            };
            r
        }
        "conv" => show_outcome(val::<L>(f[4]).to_binary64()),
        "round" => {
            // round <L> <p> <M> <value@M>
            let p: u32 = f[3].parse().unwrap();
            let m: usize = f[4].parse().unwrap();
            let mut c = WideContext::<L>::new(p).unwrap();
            match m {
                2 => show_res(c.round(&val2(f[5]))),
                4 => show_res(c.round(&val::<4>(f[5]))),
                8 => show_res(c.round(&val::<8>(f[5]))),
                16 => show_res(c.round(&val::<16>(f[5]))),
                _ => panic!("width"),
            }
        }
        "widen" => widen_dispatch(f),
        "fromint" => {
            // fromint <L> <p> <neg> <exponent> <magnitude hex, big-endian, 16k digits>
            let p: u32 = f[3].parse().unwrap();
            let neg = f[4] == "1";
            let e: i64 = f[5].parse().unwrap();
            let hex = f[6];
            let k = hex.len() / 16;
            let mut mag = vec![0u64; k];
            for i in 0..k {
                mag[k - 1 - i] = u64::from_str_radix(&hex[16 * i..16 * i + 16], 16).unwrap();
            }
            let mut c = WideContext::<L>::new(p).unwrap();
            show_res(c.from_integer(neg, &mag, e))
        }
        "ctx" => {
            let p: u32 = f[3].parse().unwrap();
            match WideContext::<L>::new(p) {
                Ok(_) => "OK".into(),
                Err(e) => format!("E:{e:?}"),
            }
        }
        "mulpow2" => {
            // mulpow2 <L> - <k> <value>
            let k: i64 = f[4].parse().unwrap();
            show_res(val::<L>(f[5]).mul_pow2(k))
        }
        "cmp" => {
            let o = val::<L>(f[4]).cmp_value(&val::<L>(f[5]));
            format!("{o:?}")
        }
        "fits" => {
            let p: u32 = f[3].parse().unwrap();
            format!("{}", val::<L>(f[4]).fits_precision(p))
        }
        "fromf64" => {
            let bits = u64::from_str_radix(f[4], 16).unwrap();
            show_res(Wide::<L>::from_f64(f64::from_bits(bits)))
        }
        other => panic!("op {other}"),
    }
}

fn widen_dispatch(f: &[&str]) -> String {
    let l: usize = f[2].parse().unwrap();
    let m: usize = f[4].parse().unwrap();
    match (l, m) {
        (4, 4) => show::<4>(&val::<4>(f[5]).widen::<4>()),
        (4, 8) => show::<8>(&val::<4>(f[5]).widen::<8>()),
        (4, 16) => show::<16>(&val::<4>(f[5]).widen::<16>()),
        (8, 8) => show::<8>(&val::<8>(f[5]).widen::<8>()),
        (8, 16) => show::<16>(&val::<8>(f[5]).widen::<16>()),
        (16, 16) => show::<16>(&val::<16>(f[5]).widen::<16>()),
        _ => panic!("widen {l} {m}"),
    }
}

#[test]
fn rv12_probe_run() {
    let Ok(input) = std::env::var("RV12_IN") else {
        return;
    };
    let output = std::env::var("RV12_OUT").unwrap();
    let text = std::fs::read_to_string(input).unwrap();
    let mut out = String::new();
    for line in text.lines() {
        let f: Vec<&str> = line.split_whitespace().collect();
        if f.is_empty() {
            continue;
        }
        let l: usize = f[2].parse().unwrap();
        let r = if l == 2 {
            match f[1] {
                "conv" => show_outcome(val2(f[4]).to_binary64()),
                "widen" => {
                    let m: usize = f[4].parse().unwrap();
                    let a = val2(f[5]);
                    match m {
                        4 => show::<4>(&a.widen::<4>()),
                        8 => show::<8>(&a.widen::<8>()),
                        16 => show::<16>(&a.widen::<16>()),
                        _ => panic!("width"),
                    }
                }
                other => panic!("L=2 op {other}"),
            }
        } else if l == 4 {
            run_l::<4>(&f)
        } else if l == 8 {
            run_l::<8>(&f)
        } else if l == 16 {
            run_l::<16>(&f)
        } else {
            panic!("L {l}")
        };
        writeln!(out, "{} {}", f[0], r).unwrap();
    }
    std::fs::write(output, out).unwrap();
}
