#!/usr/bin/env python3
"""RV20 (KF1 review) mutants: one clean `git archive` copy of the KF1 head's FK
tree and one fresh target per mutant, exact string edits with their match
counts checked, serialized (one cargo job at -j 4, RUST_TEST_THREADS=2); NONE
first. Kill = a failing test of `cargo test --lib kf1_` (the candidate's own
KF1 tests; the edit must build). Each copy and target is deleted afterwards.

usage: rv20_mutants.py <wt> <head-sha> <out.jsonl> [only,comma,list]"""
import json, os, shutil, subprocess, sys, time

WT, HEAD, OUT = sys.argv[1], sys.argv[2], sys.argv[3]
ONLY = sys.argv[4].split(",") if len(sys.argv) > 4 else None
FK = "projects/chirality-piping/core/solver/frame_kernel"
REL = "src/structural/retained/adaptive.rs"
THROWAWAY = '&mut WideContext::<16>::new(1024).expect("p")'

M = [
    ("NONE", "no edit", []),
    ("RV20-M1", "finish returns the latest surviving table refusal, not the earliest",
     [("if refused.as_ref().is_none_or(|r| seq < r.0) {",
       "if refused.as_ref().is_none_or(|r| seq > r.0) {", 1)]),
    ("RV20-M2", "finish evaluates the unevaluated rows before a surviving table refusal",
     [("        // A refusal among the evaluated rows came before every unevaluated row.\n"
       "        if let Some((_, stop)) = refused {\n"
       "            return Err(stop);\n"
       "        }\n"
       "        for (num, den, _, _) in &self.lazy {\n"
       "            let exact = directed_ratio(ctx16, num, den, self.direction)?;\n"
       "            best = Some(extreme(best, exact, self.direction));\n"
       "        }\n",
       "        for (num, den, _, _) in &self.lazy {\n"
       "            let exact = directed_ratio(ctx16, num, den, self.direction)?;\n"
       "            best = Some(extreme(best, exact, self.direction));\n"
       "        }\n"
       "        if let Some((_, stop)) = refused {\n"
       "            return Err(stop);\n"
       "        }\n", 1)]),
    ("RV20-M3", "the residual gate's collapse work is not charged (throwaway ctx16)",
     [("        tracker.offer(ctx64, ctx16, num, den)?;\n        rows.push((passes, ratio, rounded));",
       "        let _ = &ctx16;\n        tracker.offer(ctx64, " + THROWAWAY + ", num, den)?;\n"
       "        rows.push((passes, ratio, rounded));", 1)]),
    ("RV20-M4", "the shared cap's collapse work is not charged (throwaway ctx16)",
     [("                    t.collapse(ctx16);", "                    t.collapse(" + THROWAWAY + ");", 1)]),
    ("RV20-M5", "RuleTest ordered (d), (a), (b): trackers finished in another order",
     [("enum RuleTest {\n    /// (a): (|Δ| + V)/M.\n    Disagreement,\n    /// (b): Ŵ/V.\n    Estimate,\n"
       "    /// (d): C over its allowance.\n    Charge,\n}",
       "enum RuleTest {\n    /// (d): C over its allowance.\n    Charge,\n    /// (a): (|Δ| + V)/M.\n"
       "    Disagreement,\n    /// (b): Ŵ/V.\n    Estimate,\n}", 1)]),
    ("RV20-M6", "a tracker collapses only above T (holds T + 1 rows)",
     [("            if self.lazy.len() >= self.limit {", "            if self.lazy.len() > self.limit {", 1)]),
    ("RV20-M7", "the set counts rows instead of allocated capacity",
     [("    fn capacity(&self) -> usize {\n        self.lazy.capacity()\n    }",
       "    fn capacity(&self) -> usize {\n        self.lazy.len()\n    }", 1)]),
    ("RV20-M8", "the shared cap collapses only the offered tracker",
     [("            for k in std::mem::take(&mut self.holding) {\n"
       "                if let Some(t) = self.trackers.get_mut(&k) {\n"
       "                    t.collapse(ctx16);\n"
       "                }\n"
       "            }\n",
       "            if let Some(t) = self.trackers.get_mut(&key) {\n"
       "                t.collapse(ctx16);\n"
       "            }\n"
       "            self.holding.remove(&key);\n", 1)]),
    ("RV20-M9", "the unevaluated rows' window is one ulp narrower (< instead of <=)",
     [("            self.lazy.retain(|row| in_window(row.2, key));",
       "            self.lazy.retain(|row| row.2.abs_diff(key) < WINDOW_ULPS);", 1)]),
    ("RV20-M10", "the Down extreme takes the maximum",
     [("        (Some(b), Direction::Down) => b.min(value),", "        (Some(b), Direction::Down) => b.max(value),", 1)]),
    ("RV20-M11", "the pivot margin's collapse work is not charged (throwaway ctx16)",
     [("            tracker.offer(&mut ctx64, &mut ctx16, num, den)?;",
       "            tracker.offer(&mut ctx64, " + THROWAWAY + ", num, den)?;", 1)]),
    ("RV20-M12", "the fallback's collapse work is not charged (throwaway ctx16)",
     [("            tracker.offer(ctx64, ctx16, num.clone(), den.clone())?;",
       "            tracker.offer(ctx64, " + THROWAWAY + ", num.clone(), den.clone())?;", 1)]),
]

env = dict(os.environ, RUSTUP_TOOLCHAIN="1.97.1", RUSTUP_AUTO_INSTALL="0",
           CARGO_INCREMENTAL="0", RUST_TEST_THREADS="2")
repo = os.path.join(WT, "kf1")
for name, what, edits in M:
    if ONLY and name not in ONLY:
        continue
    root = os.path.join(WT, "rv20", "mut", name)
    target = os.path.join(WT, "rv20-target", "mut-" + name)
    shutil.rmtree(root, ignore_errors=True)
    shutil.rmtree(target, ignore_errors=True)
    os.makedirs(root)
    archive = subprocess.run(["git", "-C", repo, "archive", HEAD, FK],
                             capture_output=True, check=True).stdout
    subprocess.run(["tar", "-x", "-C", root], input=archive, check=True)
    fk = os.path.join(root, "frame_kernel")
    shutil.move(os.path.join(root, FK), fk)
    shutil.rmtree(os.path.join(root, "projects"))
    path = os.path.join(fk, REL)
    text = open(path, encoding="utf-8").read()
    for old, new, count in edits:
        n = text.count(old)
        assert n == count, (name, old[:70], n)
        text = text.replace(old, new)
    open(path, "w", encoding="utf-8").write(text)
    t0 = time.time()
    p = subprocess.run(["cargo", "test", "--offline", "--locked", "-j", "4", "--lib", "kf1_"],
                       cwd=fk, env=dict(env, CARGO_TARGET_DIR=target),
                       capture_output=True, text=True)
    log = p.stdout + p.stderr
    os.makedirs(os.path.join(WT, "rv20", "records", "mutant_logs"), exist_ok=True)
    open(os.path.join(WT, "rv20", "records", "mutant_logs", name + ".log"), "w").write(log)
    failed = [l.split()[1] for l in log.splitlines() if l.startswith("test ") and l.endswith("FAILED")]
    passed = [l.split()[1] for l in log.splitlines() if l.startswith("test ") and l.endswith("ok")]
    built = "could not compile" not in log and "error[" not in log
    rec = {"mutant": name, "edit": what, "exit": p.returncode, "built": built,
           "passed": len(passed), "failed": [f.rsplit("::", 1)[-1] for f in failed],
           "killed": built and p.returncode != 0, "seconds": round(time.time() - t0, 1)}
    print(json.dumps(rec), flush=True)
    with open(OUT, "a") as f:
        f.write(json.dumps(rec) + "\n")
    shutil.rmtree(target, ignore_errors=True)
    shutil.rmtree(root, ignore_errors=True)
