#!/bin/bash
# RV24: extract today's (base 78f55f927) verify_negative_direction and negative_pair_witness
# from the base's own bytes, and rename them for the reviewer's differential harness.
# Usage: make_old_fns.sh <repo worktree> <out file>
set -eu
REPO=$1; OUT=$2
SRC=projects/chirality-piping/core/solver/frame_kernel/src/structural.rs
git -C "$REPO" show 78f55f927db47c7f44299fd32793d4b64e8b3572:$SRC > "$OUT.base_structural.rs"
# The two functions are contiguous at base lines 2152-2215 (checked below by their first and last lines).
sed -n 2152,2215p "$OUT.base_structural.rs" > "$OUT.raw"
head -1 "$OUT.raw" | grep -qx 'pub fn verify_negative_direction(' 
sed -n 44p "$OUT.raw" | grep -qx 'pub fn negative_pair_witness('
tail -1 "$OUT.raw" | grep -qx '}'
sed -n 2216p "$OUT.base_structural.rs" | grep -qx ''
{
  echo "// RV24: verbatim base 78f55f927 FK/structural.rs:2152-2215, renamed only (generated by make_old_fns.sh)."
  sed -e 's/^pub fn verify_negative_direction(/pub(super) fn rv24_old_verify(/' \
      -e 's/^pub fn negative_pair_witness(/pub(super) fn rv24_old_witness(/' \
      -e 's/verify_negative_direction(prepared, &v)/rv24_old_verify(prepared, \&v)/' "$OUT.raw"
} > "$OUT"
shasum -a 256 "$OUT.raw" "$OUT"
diff "$OUT.raw" <(tail -n +2 "$OUT") || true
rm "$OUT.base_structural.rs"
