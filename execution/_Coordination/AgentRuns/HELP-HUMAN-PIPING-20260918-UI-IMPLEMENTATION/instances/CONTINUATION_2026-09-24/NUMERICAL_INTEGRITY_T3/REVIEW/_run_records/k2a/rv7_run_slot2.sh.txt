#!/bin/bash
set -u
S=<scratch>/rv7
TGT=<wt>/rv7-target
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0
LOG=$S/logs
guard() { free=$(df --output=avail -k <home> | tail -1); echo "disk before $1: ${free} kB" >> $LOG/SUMMARY3.txt; if [ "$free" -lt 8912896 ]; then echo "ABORT low disk before $1" >> $LOG/SUMMARY3.txt; exit 3; fi; }
guard fk_base; CARGO_TARGET_DIR=$TGT/fk_base cargo test --offline --locked --manifest-path $S/fk_base/Cargo.toml --test rv7_rotated_m03 -- --nocapture > $LOG/fk_base3.log 2>&1; echo "fk_base exit $?" >> $LOG/SUMMARY3.txt
C=$S/src/projects/chirality-piping/core
guard pp; CARGO_TARGET_DIR=$TGT/pp cargo test --offline --locked --manifest-path $C/product_physics/Cargo.toml --test rv7_skew_runtime -- --nocapture > $LOG/pp3.log 2>&1; echo "pp exit $?" >> $LOG/SUMMARY3.txt
du -sh $TGT >> $LOG/SUMMARY3.txt; rm -rf $TGT/pp
CM=$S/src_main/projects/chirality-piping/core
guard pp_main; CARGO_TARGET_DIR=$TGT/pp_main cargo test --offline --locked --manifest-path $CM/product_physics/Cargo.toml --test rv7_skew_runtime -- --nocapture > $LOG/pp_main_skew3.log 2>&1; echo "pp_main_skew exit $?" >> $LOG/SUMMARY3.txt
du -sh $TGT >> $LOG/SUMMARY3.txt
