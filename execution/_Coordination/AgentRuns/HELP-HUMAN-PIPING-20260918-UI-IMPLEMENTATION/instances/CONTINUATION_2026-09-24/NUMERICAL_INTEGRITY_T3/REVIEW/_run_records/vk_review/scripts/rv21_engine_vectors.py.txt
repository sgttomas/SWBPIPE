#!/usr/bin/env python3
"""RV21: random exact-engine vectors, decided by Python Fractions (RV21's own
generator, independent of gen_vk_cases.py's samples). python3 -B only.

Kinds:
  p  predicate:      |obs - exp| <= 1e-9 * max(|exp|, scale)
  m  magnitude:      |sqrt(y^2 + z^2) - exp| <= 1e-9 * max(|exp|, scale),
                     decided by exact integer square-root bracketing
  c  covered:        max(|exp|, scale) >= 2^-34 * s_star
  o  outside range:  exp != 0 and fl(exp) is +-0 or +-inf
"""
import sys
sys.dont_write_bytecode = True
import json
import math
import random
import struct
from fractions import Fraction as F

rng = random.Random(0x52563231)
TOL = F(1, 10 ** 9)


def bits(x):
    return struct.pack('>d', x).hex()


def dec(q_exp10):
    """A random decimal string with up to 40 significant digits."""
    digits = rng.randint(1, 40)
    mant = rng.randint(10 ** (digits - 1), 10 ** digits - 1)
    e = q_exp10
    sign = '-' if rng.random() < 0.3 else ''
    style = rng.randint(0, 2)
    if style == 0:
        text = '%s%de%d' % (sign, mant, e)
    elif style == 1:
        s = str(mant)
        text = '%s%s.%se%+d' % (sign, s[0], s[1:] or '0', e + len(s) - 1)
    else:
        if -30 < e < 30:
            v = F(mant) * F(10) ** e
            if v.denominator == 1:
                text = '%s%d' % (sign, v.numerator)
            else:
                # plain decimal
                s = str(mant)
                if e < 0:
                    k = -e
                    s = s.rjust(k + 1, '0')
                    text = '%s%s.%s' % (sign, s[:-k], s[-k:])
                else:
                    text = '%s%s' % (sign, s + '0' * e)
        else:
            text = '%s%de%d' % (sign, mant, e)
    return text


def val(text):
    return F(text)


def fl(q):
    return q.numerator / q.denominator


def near(x, k):
    for _ in range(k):
        x = math.nextafter(x, math.inf if rng.random() < 0.5 else -math.inf)
    return x


def magnitude_holds(y, z, exp, scale):
    s = F(y) ** 2 + F(z) ** 2
    t = TOL * max(abs(exp), scale)
    hi = exp + t
    lo = exp - t
    if hi < 0 or s > hi * hi:
        return False
    if lo <= 0:
        return True
    return s >= lo * lo


out = []
for _ in range(6000):
    e10 = rng.choice([rng.randint(-40, 40), rng.randint(-330, 310), rng.randint(-3000, -330)])
    exp_t = dec(e10) if rng.random() > 0.05 else '0'
    scale_t = dec(e10 + rng.randint(-20, 20)).lstrip('-') if rng.random() > 0.2 else '0'
    exp, scale = val(exp_t), val(scale_t)
    t = TOL * max(abs(exp), scale)
    try:
        base = [fl(exp), fl(exp + t), fl(exp - t), fl(exp + 2 * t), fl(-exp), 0.0, -0.0, 5e-324,
                rng.uniform(-1, 1) * 10.0 ** rng.randint(-300, 300)]
    except OverflowError:
        continue
    for b in base:
        if not math.isfinite(b):
            continue
        for k in (0, 1, 2):
            o = near(b, k)
            if not math.isfinite(o):
                continue
            out.append({'k': 'p', 'o': bits(o), 'e': exp_t, 's': scale_t,
                        'v': abs(F(o) - exp) <= t})
    if exp >= 0 and exp != 0 and abs(exp) < F(10) ** 300:
        for target in (exp, exp + t, exp - t, exp + 2 * t):
            if target <= 0:
                continue
            try:
                w = fl(target)
            except OverflowError:
                continue
            for k in (0, 1, 3):
                ww = near(w, k)
                y, z = fl(F(ww) * F(3, 5)), fl(F(ww) * F(4, 5))
                if not (math.isfinite(y) and math.isfinite(z)):
                    continue
                y2, z2 = near(y, rng.randint(0, 2)), near(z, rng.randint(0, 2))
                if rng.random() < 0.5:
                    y2 = -y2
                out.append({'k': 'm', 'y': bits(y2), 'z': bits(z2), 'e': exp_t, 's': scale_t,
                            'v': magnitude_holds(y2, z2, exp, scale)})
    # covered: s_star a binary64 near 2^34 * comparison
    comp = max(abs(exp), scale)
    if comp != 0:
        try:
            s0 = fl(comp * 2 ** 34)
        except OverflowError:
            s0 = None
        if s0 is not None and math.isfinite(s0):
            for k in (0, 1, 2):
                s = near(s0, k)
                if s < 0 or not math.isfinite(s):
                    continue
                out.append({'k': 'c', 'star': bits(s), 'e': exp_t, 's': scale_t,
                            'v': comp >= F(s) / 2 ** 34})
    out.append({'k': 'o', 'e': exp_t, 'v': exp != 0 and (abs(exp) <= F(1, 2 ** 1075) or
                                                        abs(exp) >= F(2) ** 1024 - F(2) ** 970)})
# the range boundaries exactly
for q, v in ((F(1, 2 ** 1075), True), (F(1, 2 ** 1075) + F(1, 10 ** 400), False),
             (F(2) ** 1024 - F(2) ** 970, True), (F(2) ** 1024 - F(2) ** 970 - 1, False)):
    n, d = q.numerator, q.denominator
    # an exact decimal spelling of q: q = n/d with d a power of two -> n*5^k / 10^k
    k = 0
    while d % 2 == 0:
        d //= 2
        k += 1
    if d == 1:
        text = '%de-%d' % (n * 5 ** k, k) if k else str(n)
    else:
        # d = 10^400 case
        text = None
    if text is None:
        num = q.numerator
        den = q.denominator
        # q = num/den with den = 2^1075 * 10^400 / gcd...: write as decimal with 1475 digits
        scale10 = 1475
        m = num * 10 ** scale10 // den
        assert F(m, 10 ** scale10) == q
        text = '%de-%d' % (m, scale10)
    out.append({'k': 'o', 'e': text, 'v': v})
    out.append({'k': 'o', 'e': '-' + text, 'v': v})

with open(sys.argv[1], 'w') as fh:
    for v in out:
        fh.write(json.dumps(v, sort_keys=True) + '\n')
from collections import Counter
print(len(out), Counter((v['k'], v['v']) for v in out))
