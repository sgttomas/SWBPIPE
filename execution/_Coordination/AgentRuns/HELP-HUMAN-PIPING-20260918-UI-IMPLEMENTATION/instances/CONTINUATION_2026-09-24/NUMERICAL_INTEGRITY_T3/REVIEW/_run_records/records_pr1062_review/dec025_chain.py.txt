#!/usr/bin/env python3
"""RV25: independent re-derivation of the DEC-025 suite comparisons in the nine
new merge records, from the committed dec025/ files only.

Run from the repository root of a Git working tree at the PR head:
    python3 dec025_chain.py.txt <T3 folder>
For each slice it reads its dec025/suites.log and its stated baseline's
dec025/suites.log (the previous merge record's, or K2B_MERGE's for K5), keys
both by manifest path, and prints every manifest whose (passed, failed,
ignored) differs, plus the manifest counts. It prints the last summary lines of
python_pytest.log and desktop_vitest.log (and K5's vitest_rerun). It also checks
each "piping source equals main X's" claim: `git diff --quiet <baseline head>
<main> -- projects/chirality-piping` (plus tools/ and .github/, reported apart).
"""
import re
import subprocess
import sys

CHAIN = [  # slice, baseline dec025 folder, baseline head, main the record names
    ("K5", "K2B_MERGE/dec025", "33e33c723", "b37331092"),
    ("F1B", "K5_MERGE/dec025", "babcf5e65", "1cdeae2c1"),
    ("K6", "F1B_MERGE/dec025", "6fa422979", "59cb20073"),
    ("K4", "K6_MERGE/dec025", "cd325c1fe", "7ac7b1c37"),
    ("KF1", "K4_MERGE/dec025", "5a46a6278", "8cca91701"),
    ("VK", "KF1_MERGE/dec025", "66adfede4", "0f5d8c7b4"),
    ("K6B", "VK_MERGE/dec025", "5f0d39426", "f8400d290"),
    ("KF3", "K6B_MERGE/dec025", "597c81ba4", "78f55f927"),
    ("KF2", "KF3_MERGE/dec025", "aa83f6796", "dd61120ff"),
]
HEADS = {"K5": "babcf5e65", "F1B": "6fa422979", "K6": "cd325c1fe", "K4": "5a46a6278",
         "KF1": "66adfede4", "VK": "5f0d39426", "K6B": "597c81ba4", "KF3": "aa83f6796",
         "KF2": "522167ac6"}


def suites(path):
    out = {}
    for line in open(path, encoding="utf-8", errors="replace"):
        m = re.match(r"\s*(-?\d+) passed=(\d+) failed=(\d+) ignored=(\d+) (\S+)", line)
        if m:
            out[m.group(5)] = (int(m.group(2)), int(m.group(3)), int(m.group(4)), int(m.group(1)))
    return out


def tail_match(path, rx):
    try:
        txt = open(path, encoding="utf-8", errors="replace").read()
    except FileNotFoundError:
        return "(no file)"
    txt = re.sub(r"\x1b\[[0-9;]*m", "", txt)
    hits = re.findall(rx, txt, re.M)
    return hits[-3:] if hits else "(no match)"


def same_tree(a, b, paths):
    return subprocess.run(["git", "diff", "--quiet", a, b, "--", *paths]).returncode == 0


def main():
    t3 = sys.argv[1].rstrip("/") + "/IMPLEMENTATION"
    for s, base_dir, base_head, main_ref in CHAIN:
        cand = suites(f"{t3}/{s}_MERGE/dec025/suites.log")
        base = suites(f"{t3}/{base_dir}/suites.log")
        print(f"== {s} (candidate {HEADS[s]}) vs {base_dir} (baseline {base_head})")
        print(f"   manifests: baseline {len(base)}, candidate {len(cand)}")
        for k in sorted(set(base) | set(cand)):
            b, c = base.get(k), cand.get(k)
            if (b and b[:3]) != (c and c[:3]):
                print(f"   CHANGED {k}: {b[:3] if b else None} -> {c[:3] if c else None}")
        fails = {k: v[1] for k, v in cand.items() if v[1]}
        print(f"   candidate manifests with failures: {fails}")
        print(f"   pytest: {tail_match(f'{t3}/{s}_MERGE/dec025/python_pytest.log', r'^(\d+ passed.*)$')}")
        print(f"   vitest: {tail_match(f'{t3}/{s}_MERGE/dec025/desktop_vitest.log', r'^\s*(Test Files .*|Tests .*)$')}")
        if s == "K5":
            print(f"   vitest rerun: {tail_match(f'{t3}/K5_MERGE/dec025/vitest_rerun/desktop_vitest.log', r'^\s*(Test Files .*|Tests .*)$')}")
        piping = same_tree(base_head, main_ref, ["projects/chirality-piping"])
        other = same_tree(base_head, main_ref, ["tools", ".github"])
        print(f"   baseline {base_head} piping tree == main {main_ref}: {piping}; tools/.github equal: {other}")
    # the handoff's latest baseline
    print(f"== handoff: KF2 head 522167ac6 piping tree == main 7ad3a9adf: "
          f"{same_tree('522167ac6', '7ad3a9adf', ['projects/chirality-piping'])}")


if __name__ == "__main__":
    main()
