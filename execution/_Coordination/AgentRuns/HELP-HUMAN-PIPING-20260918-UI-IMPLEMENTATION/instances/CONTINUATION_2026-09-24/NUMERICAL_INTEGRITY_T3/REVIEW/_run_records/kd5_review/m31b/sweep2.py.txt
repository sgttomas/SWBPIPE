import json, math, sys
sys.path.insert(0, '.')
import m31b_attack as A

def report(m, extra=None):
    r = A.analyse(m)
    d = r['max_delta_row']; a = r['max_actual_row']
    adm = r['admissible']
    print(f"{m['name']:40s} ok={all(x['ok'] for x in adm)} phi={adm[0]['phi']:.6g} mis={max(abs(x['rel_mismatch']) for x in adm):.2e} "
          f"Tcorr={r['max_trigger_correct']:.4f} Tm31b={r['max_trigger_m31b']:.3e} |dEF|/crit={d['delta']:.4f}@{d['dof']} actual={a['actual']:.4f}@{a['dof']}", flush=True)
    r['model'] = m
    if extra: r.update(extra)
    return r

def pp_centre(xi, xj, radius, yref):
    ch = [xj[k]-xi[k] for k in range(3)]
    cl = math.sqrt(ch[0]*ch[0]+ch[1]*ch[1]+ch[2]*ch[2])
    h = 0.5*cl
    cu = [ch[k]/cl for k in range(3)]
    ax = yref[0]*cu[0]+yref[1]*cu[1]+yref[2]*cu[2]
    pn = [yref[k]-ax*cu[k] for k in range(3)]
    pm = math.sqrt(pn[0]*pn[0]+pn[1]*pn[1]+pn[2]*pn[2])
    s = math.sqrt(radius*radius-h*h)
    return [0.5*(xi[k]+xj[k]) - s*pn[k]/pm for k in range(3)]

out = []
S = [1e6,1e6,1e6]; LD = [(9,1.0),(10,1.0),(11,1.0)]
# near pi (max admissible mismatch) and small angle at zero designed mismatch
for phi in [math.pi-1e-3, math.pi-1e-6, math.pi-1e-8]:
    for sh in ["max", 0.0]:
        s = A.max_shift(0.3, phi, [0,0,1]) if sh == "max" else 0.0
        m = A.cantilever(f"NEARPI-{math.pi-phi:.0e}-shift{sh}", 0.3, phi, [0,0,1], s, S, LD)
        out.append(report(m))
# extreme R/L at max mismatch: large R small angle, tiny R
for R, phi in [(30.0, 1e-3), (30.0, 0.05), (0.01, math.radians(30)), (3.0, math.radians(10))]:
    s = A.max_shift(R, phi, [0,0,1])
    m = A.cantilever(f"RL-R{R}-phi{phi:.3g}", R, phi, [0,0,1], s, S, LD)
    out.append(report(m))
# stiff-X / soft-Y combos on the phi=60 skew elbow
for springs in [[1e8,1e2,1e6],[1e2,1e8,1e6],[1e6,1e6,1e2],[3e1,1e6,1e6]]:
    phi = math.radians(60)
    s = A.max_shift(0.3, phi, [1,2,2])
    m = A.cantilever(f"COMBO-phi60-k{springs}", 0.3, phi, [1,2,2], s, springs, LD)
    out.append(report(m))
# multi-bend chain: zigzag of 4 elbows (60 deg each, alternating), max mismatch each
def chain(n, phi, R, sign_alt):
    nodes=[[0.0,0.0,0.0]]; members=[]
    heading = 0.0; pos=[0.0,0.0,0.0]
    for k in range(n):
        turn = phi if (not sign_alt or k%2==0) else -phi
        # radial at start points from centre to node; centre at pos + R*left normal
        nl = [-math.sin(heading), math.cos(heading), 0.0]
        sgn = 1.0 if turn>0 else -1.0
        c = [pos[0]+sgn*R*nl[0], pos[1]+sgn*R*nl[1], 0.0]
        a0 = math.atan2(pos[1]-c[1], pos[0]-c[0])
        a1 = a0 + turn
        p1 = [c[0]+R*math.cos(a1), c[1]+R*math.sin(a1), 0.0]
        chd = [p1[0]-pos[0], p1[1]-pos[1], 0.0]; lc=math.hypot(chd[0],chd[1])
        # shift centre along chord by the max admissible fraction
        lo, hi = 0.0, 5e-9*R/max(math.sin(abs(turn)/2),1e-12)
        for _ in range(80):
            mid=0.5*(lo+hi); cc=[c[0]-mid*chd[0]/lc, c[1]-mid*chd[1]/lc, 0.0]
            ok = A.b64_formula_chord(pos, p1, cc)[1]
            lo,hi = (mid,hi) if ok else (lo,mid)
        cc=[c[0]-lo*chd[0]/lc, c[1]-lo*chd[1]/lc, 0.0]
        nodes.append(p1); members.append({"i":k,"j":k+1,"y_reference":None,"bend":(cc,1.0)})
        heading += turn; pos = p1
    return nodes, members
for n, alt in [(4, False), (4, True)]:
    nodes, members = chain(n, math.radians(60), 0.3, alt)
    m = {"name": f"CHAIN{n}-60deg-{'zigzag' if alt else 'spiral'}", "section": A.SEC, "nodes": nodes, "members": members,
         "rigid":[0,1,2], "springs":[(3,1e6),(4,1e6),(5,1e6)], "loads":[(6*n+3,1.0),(6*n+4,1.0),(6*n+5,1.0)]}
    out.append(report(m))
# PP-route geometry: centre computed as PP computes it, nodes at large coordinates
for X0, R, phi in [(0.0,0.3,math.radians(10)), (1e3,0.3,math.radians(10)), (1e5,0.3,math.radians(10)), (5e6,0.3,math.radians(5)), (5e6,1.5,math.radians(10)), (5e6,0.3,math.radians(1)), (5e5, 0.3, math.radians(2))]:
    xi=[X0, X0*0.7, 0.0]
    xj=[X0 + R*math.sin(phi), X0*0.7 + R*(1-math.cos(phi)), 0.0]
    xj=[float(v) for v in xj]
    c = pp_centre(xi, xj, R, [0.0,1.0,0.0])
    ok = A.b64_formula_chord(xi, xj, c)[1]
    m = {"name": f"PP-X0{X0:.0e}-R{R}-phi{math.degrees(phi):.0f}", "section": A.SEC, "nodes":[xi,xj],
         "members":[{"i":0,"j":1,"y_reference":None,"bend":(c,1.0)}],
         "rigid":[0,1,2], "springs":[(3,1e6),(4,1e6),(5,1e6)], "loads":[(9,1.0),(10,1.0),(11,1.0)]}
    if not ok:
        print(m["name"], "not admissible (product refuses)"); continue
    out.append(report(m))
