#!/usr/bin/env python3
"""RV19 delta check: how the slack compare_honest adds (delta = 2^-127|x| +
2^err) compares with each published row's allowance a (the claim it
checks), over every selected publication. Reports the largest delta/a, and
every row where delta/a exceeds 2^-10. Standard library only."""
import sys, struct
from fractions import Fraction as Fr
def hf(b): return struct.unpack('>d', int(b,16).to_bytes(8,'big'))[0]
tok = {}
cur = None
for p in sys.argv[2].split(','):
    for line in open(p):
        f = line.split()
        if not f: continue
        if f[0] in ('model','combo'): cur = tok.setdefault(f[1], {}); continue
        if f[0] == 'end': cur = None; continue
        if f[0] == 'expect' and cur is not None:
            x = err = None
            for t in f[3:]:
                if t.startswith('x:'):
                    v = t[2:]
                    if v == '0': x = Fr(0)
                    else:
                        m, e = v[1:].split('p'); x = Fr(int(m,16)) * Fr(2)**int(e)
                        if v[0] == '-': x = -x
                elif t.startswith('err:'): err = int(t[4:])
            cur[f[1]] = (x, err)
sel = {}; rows = {}; scales = {}
for line in open(sys.argv[1]):
    f = line.split()
    if not f: continue
    if f[0] == 'DUMP': sel[f[1]] = int(f[2]); rows[f[1]] = []
    elif f[0] == 'ROW': rows[f[1]].append(f[2:])
    elif f[0] == 'SCALE': scales[(f[1], int(f[2]), f[3])] = hf(f[4])
worst = (Fr(0), '')
flag = []
n = 0
for name, rs in rows.items():
    bf = Fr(1) + (Fr(1, 2**21) if sel[name] == 512 else Fr(1, 2**22))
    for key, kind, body, vk, bits, cls, b in rs:
        if vk not in ('N', 'S') or key not in tok.get(name, {}): continue
        x, err = tok[name][key]
        if x is None: continue
        q = Fr(hf(bits))
        if cls == 'abs': a = Fr(hf(b)) * bf
        elif cls == 'rel': a = Fr(1, 2**64) * max(abs(q), Fr(scales[(name, int(body), kind)])) * (1 + Fr(1, 2**21)) + abs(q) / 2**53 + Fr(1, 2**1074)
        elif cls == 'inp': a = abs(q) / 2**53 + Fr(1, 2**1074)
        else: continue
        d = abs(x) / 2**127 + (Fr(2)**err if err is not None else 0)
        n += 1
        if a == 0:
            if d != 0: flag.append('%s %s a=0 delta=2^%d' % (name, key, err if err is not None else -99999))
            continue
        r = d / a
        if r > worst[0]: worst = (r, '%s %s' % (name, key))
        if r > Fr(1, 2**10): flag.append('%s %s %s delta/a=%.3e' % (name, key, cls, float(r)))
import math
print('rows %d; largest delta/a %.3e (2^%.1f) at %s' % (n, float(worst[0]), math.log2(float(worst[0])) if worst[0] > 0 else -9999, worst[1]))
print('rows with delta/a > 2^-10 (or a = 0 with delta > 0): %d' % len(flag))
for s in flag[:20]: print('  ' + s)
