"""RV18 probe: does an in-flight observation process group survive a SIGTERM to the runner?

The runner process calls k6_runner.launch() on a self-limiting sleeper (a Python child that
sleeps 20 s, tagged RV18_PROBE_TAG) under /usr/bin/time, with a 60 s timeout; the probe sends
SIGTERM to the runner after 2 s and then lists the processes carrying the tag.
"""
import os, signal, subprocess, sys, time
runner_dir, rec_dir = sys.argv[1:3]
code = ("import sys; sys.path.insert(0, %r); import k6_runner as r; "
        "r.launch([sys.executable, '-c', 'import time; RV18_PROBE_TAG=1; time.sleep(20)'], rss_cap_bytes=128*2**20, "
        "timeout_s=60, record_dir=%r, run_id='probe', k6_protocol=False)") % (runner_dir, rec_dir)
runner = subprocess.Popen([sys.executable, '-c', code])
time.sleep(2)
before = subprocess.run(['pgrep', '-f', 'RV18_PROBE_TAG'], capture_output=True, text=True).stdout.split()
os.kill(runner.pid, signal.SIGTERM)
runner.wait()
time.sleep(1)
after = subprocess.run(['pgrep', '-f', 'RV18_PROBE_TAG'], capture_output=True, text=True).stdout.split()
after = [p for p in after if int(p) != os.getpid()]
print('runner exit', runner.returncode)
print('tagged processes before SIGTERM:', len(before), 'after:', len(after))
for p in after:
    print(subprocess.run(['ps', '-o', 'pid=,ppid=,pgid=,command=', '-p', p], capture_output=True, text=True).stdout.strip()[:120])
    os.kill(int(p), signal.SIGKILL)
print('cleanup: killed', len(after))
