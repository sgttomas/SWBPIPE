#!/usr/bin/env python3
"""V1 BACKCHECK_R4 check W: the rows a W1-selected case would withhold under DESIGN r4 s4.1.6.1,
recounted independently on committed selected cases, with every NON-ZERO withheld row listed.

Usage: python3 probe_r4_withheld.py <dir> [<dir> ...]   (standard library only)
Reads every *.raw.json with a source_block_recovery object and its sibling request. Classification
follows DESIGN r4: closed (kind, unit) table; input_derived for pressure-only stresses, echoed inputs
and displacement/rotation rows at a rigidly restrained DOF; non_quantity list; S* per case (one body)
with the coupling of s4.1.6.1 item 6; stress S* = fo/A + k*mo/Z with section terms from the request
(nominal wall; D1's withheld_rows.py does the same). Two k sets are evaluated:
  r4    D1 revision 4: circular maximum sqrt2, open-formula summary 2;
  span  V1's R4-1: rows formed from span statics (exact_straight_summary_extrema), circular maximum
        2*sqrt2, open-formula summary 4.
The committed selected values stand in for W1's values, as in D1's probe.
"""
import json
import math
import os
import sys
from collections import Counter, defaultdict

R = 2.0 ** -34
UNIT = {"m": 1.0, "mm": 1e-3, "rad": 1.0, "N": 1.0, "kN": 1e3, "N*m": 1.0, "kN*m": 1e3, "Pa": 1.0, "MPa": 1e6}
BASE = {
    "global_nodal_displacement_x": "translation", "global_nodal_displacement_y": "translation",
    "global_nodal_displacement_z": "translation", "displacement_magnitude": "translation",
    "global_nodal_rotation_x": "rotation", "global_nodal_rotation_y": "rotation", "global_nodal_rotation_z": "rotation",
    "element_local_axial_force": "force", "element_local_shear_force_y": "force", "element_local_shear_force_z": "force",
    "pipe_wall_axial_force_v2": "force", "pipe_effective_axial_force_v2": "force",
    "support_reaction_force_magnitude_v2": "force", "reaction_resultant": "force",
    "element_local_torsional_moment": "moment", "element_local_bending_moment_y": "moment",
    "element_local_bending_moment_z": "moment", "support_reaction_moment_magnitude_v2": "moment",
    "support_reaction_component_v2": "by_unit", "pipe_wall_endpoint_action_v2": "by_unit",
}
STRESS = {"element_local_axial_normal_stress": "k1", "element_local_bending_normal_stress_y": "k1",
          "element_local_bending_normal_stress_z": "k1", "element_local_torsional_shear_stress": "k1",
          "pipe_axial_membrane_stress_v2": "k1", "pipe_elastic_normal_stress_maximum_v2": "kmax",
          "open_formula_stress_summary": "ksum", "component_equal_factor_intensified_bending_stress_v1": "ki"}
KSETS = {"r4": {"k1": 1.0, "kmax": math.sqrt(2.0), "ksum": 2.0},
         "span": {"k1": 1.0, "kmax": 2.0 * math.sqrt(2.0), "ksum": 4.0}}
INPUT_DERIVED = {"pipe_lame_hoop_stress_v2", "pipe_lame_radial_stress_v2", "pipe_section_pressure_hoop_stress",
                 "pipe_section_pressure_longitudinal_stress", "constant_effort_support_applied_load",
                 "expansion_joint_pressure_thrust_load_review", "component_user_stress_multiplier_review",
                 "component_user_stiffness_macro_element_review", "constant_effort_user_input_review",
                 "spring_hanger_user_input_review"}
NON_QUANTITY = {"sparse_live_path_dense_parity_relative_delta", "linear_solver_mode_basis", "modulus_basis_record",
                "combination_modulus_basis_record"}
RESTRAINT = {"global_nodal_displacement_x": "UX", "global_nodal_displacement_y": "UY", "global_nodal_displacement_z": "UZ",
             "global_nodal_rotation_x": "RX", "global_nodal_rotation_y": "RY", "global_nodal_rotation_z": "RZ"}


def request_for(path):
    d, f = os.path.split(path)
    base = f.replace(".raw.json", "")
    for cand in (base + ".request.json", base.rsplit("-", 1)[0] + ".request.json"):
        p = os.path.join(d, cand)
        if os.path.exists(p):
            return json.load(open(p))
    return None


def classify(rows, model, kset):
    m = model.get("model", model)
    pos = [(n["position"]["x"], n["position"]["y"], n["position"]["z"]) for n in m["nodes"]]
    Lb = math.sqrt(sum((max(p[a] for p in pos) - min(p[a] for p in pos)) ** 2 for a in range(3)))
    sec = {}
    for p in m["pipe_segments"]:
        od = p["section"]["outside_diameter"]["value"]
        t = p["section"]["wall_thickness"]["value"]
        idd = od - 2 * t
        A = math.pi * (od * od - idd * idd) / 4.0
        I = math.pi * (od ** 4 - idd ** 4) / 64.0
        sec[p["id"]] = (A, I / (od / 2.0))
    restrained = defaultdict(set)
    for s in m["supports"]:
        if s.get("family") != "spring":
            for r in s.get("restraints", []):
                restrained[s["node"]].add(r)
    S = defaultdict(float)
    parsed = []
    for r in rows:
        k = r["kind"]
        v = abs(r["value"]) * UNIT.get(r["unit"], float("nan")) if isinstance(r["value"], (int, float)) else None
        cls = BASE.get(k)
        if cls == "by_unit":
            cls = "force" if r["unit"] in ("N", "kN") else "moment"
        parsed.append((r, cls, v))
        if cls and v is not None and v == v:
            S[cls] = max(S[cls], v)
    star = {"translation": max(S["translation"], Lb * S["rotation"]),
            "rotation": max(S["rotation"], S["translation"] / Lb if Lb else 0.0),
            "force": max(S["force"], S["moment"] / Lb if Lb else 0.0),
            "moment": max(S["moment"], Lb * S["force"])}
    count = Counter()
    nonzero = []
    for r, cls, v in parsed:
        k = r["kind"]
        if k in NON_QUANTITY:
            count["non_quantity"] += 1
            continue
        if k in INPUT_DERIVED or (k in RESTRAINT and RESTRAINT[k] in restrained.get(r["entity_ref"], set())):
            count["input_derived"] += 1
            continue
        if v is None or v != v:
            count["not_covered"] += 1
            continue
        if k in STRESS:
            A, Z = sec.get(r["entity_ref"], (None, None))
            kk = KSETS[kset].get(STRESS[k])
            if A is None or kk is None:
                count["not_covered"] += 1
                continue
            s = star["force"] / A + kk * star["moment"] / Z
            label = "stress"
        elif cls:
            s = star[cls]
            label = cls
        else:
            count["not_covered"] += 1
            continue
        t = R * s
        if v < t:
            count["absolute_verified"] += 1
            if v == 0.0:
                count["absolute_verified_exact_zero"] += 1
            else:
                nonzero.append({"kind": k, "unit": r["unit"], "entity": r["entity_ref"], "value": r["value"],
                                "value_over_threshold": v / t, "value_over_scale": v / s, "scale_kind": label})
        else:
            count["relative_verified"] += 1
    count["rows"] = len(rows)
    count["withheld"] = count["absolute_verified"] + count["not_covered"]
    return dict(count), nonzero


def main():
    out = {}
    for root in sys.argv[1:]:
        for dp, dn, fn in os.walk(root):
            for f in sorted(fn):
                if not f.endswith(".raw.json"):
                    continue
                p = os.path.join(dp, f)
                env = json.load(open(p))
                sbr = env.get("source_block_recovery")
                if not isinstance(sbr, dict):
                    continue
                model = request_for(p)
                if model is None:
                    continue
                cases = sorted({c["basis_ref"]["ref_id"] for c in sbr.get("body", {}).get("cases", [])
                                if c.get("status") in (None, "selected", "qualified") and c.get("basis_ref")})
                for case in cases:
                    rows = [r for r in env.get("results", []) if (r.get("basis_ref") or {}).get("ref_id") == case]
                    if not rows:
                        continue
                    rec = {}
                    for kset in ("r4", "span"):
                        c, nz = classify(rows, model, kset)
                        rec[kset] = c
                        if kset == "r4":
                            rec["nonzero_withheld_r4"] = nz
                    out[f"{os.path.basename(dp)}/{f} :: {case}"] = rec
    print(json.dumps(out, indent=1, sort_keys=True))


if __name__ == "__main__":
    main()
