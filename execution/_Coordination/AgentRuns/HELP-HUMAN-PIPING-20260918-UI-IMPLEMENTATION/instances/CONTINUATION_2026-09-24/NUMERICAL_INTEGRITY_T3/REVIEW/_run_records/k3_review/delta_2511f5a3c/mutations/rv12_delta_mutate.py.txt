#!/usr/bin/env python3
"""RV12 mutation driver for the K3 delta check at 2511f5a3c (standard library only).

The same RV1, RV2, RV4 and RV5 edits as the review; filters aimed at the new tests
(RV5: only the L = 4, 8 and 16 targeted vectors, so the L = 2 cross-check cannot be the kill).

Each mutant: a clean `git archive` of the piping `core/` tree at the candidate head
(FK's s11_site_table test reads sibling crates) into
<wt>/scratch/rv12/mut/<id>/, one exact-string edit asserted to apply once, its
own target <wt>/rv12-target/mut-<id> (deleted afterwards), and `cargo test
--offline --locked -j 4 --no-fail-fast` with RUST_TEST_THREADS=4, either the
whole FK suite or named test filters. A failing test counts as a kill only
when its last panic is at an `assert` line of a test file; a panic in the code
under test, an unwrap/expect/panic! in a test, or a compile failure does not.

Usage: rv12_mutate.py <wt> <commit> <out_dir> <id> [<id> ...]
"""
import concurrent.futures
import json
import os
import re
import shutil
import subprocess
import sys
import time
from pathlib import Path

WT = Path(sys.argv[1])
COMMIT = sys.argv[2]
OUT = Path(sys.argv[3])
IDS = sys.argv[4:]
FK = "projects/chirality-piping/core/solver/frame_kernel"
MULTI = FK + "/src/structural/retained/wide/multi.rs"
WIDE = FK + "/src/structural/retained/wide.rs"
TOML = FK + "/Cargo.toml"

K3 = "structural::retained::wide::multi::tests::"
K3A = "structural::retained::wide::tests::"
FULL = None  # the whole FK suite

MUTANTS = {
    "NONE_D": ("control for the delta filtered runs", None, None, None,
               [K3 + "cmp_value_and_mul_pow2", K3 + "two_sum", K3 + "signs_of_zero", K3 + "targeted_hard"]),
    'RV1': ('cmp_value orders two negative values as if positive', MULTI,
            '                (true, true) => cmp_mag(other, self),\n',
            '                (true, true) => cmp_mag(self, other),\n',
            [K3 + "cmp_value_and_mul_pow2", K3 + "signs_of_zero"]),
    'RV2': ('mul_pow2 scales by 2^(2k) instead of 2^k', MULTI,
            '        let exponent = checked_exponent(i128::from(self.exponent) + i128::from(k))?;\n',
            '        let exponent = checked_exponent(i128::from(self.exponent) + 2 * i128::from(k))?;\n',
            [K3 + "cmp_value_and_mul_pow2", K3 + "signs_of_zero"]),
    'RV4': ("TwoSum checks only its first operand's precision", MULTI,
            '    if !fits(a, p) || !fits(b, p) {\n        return Err(WideError::OperandPrecision);\n    }\n    let s = add_rounded(a, b, p)?;\n',
            '    if !fits(a, p) {\n        return Err(WideError::OperandPrecision);\n    }\n    let s = add_rounded(a, b, p)?;\n',
            [K3 + "two_sum"]),
    'RV5': ('far subtraction drops the (1 - f) tail after the borrow', MULTI,
            '        round_limbs::<L>(big.negative, window.as_ref(), sticky, scale, p)\n    }\n}\n\nfn mul_rounded',
            '        round_limbs::<L>(big.negative, window.as_ref(), false, scale, p)\n    }\n}\n\nfn mul_rounded',
            [K3 + "targeted_hard"]),
}

ENV = dict(os.environ, RUSTUP_TOOLCHAIN="1.97.1", RUSTUP_AUTO_INSTALL="0",
           CARGO_INCREMENTAL="0", RUST_TEST_THREADS="4")


def sanitize(text):
    return text.replace(str(WT), "<wt>").replace(str(Path.home()), "<home>")


def run(mid):
    desc, path, old, new, filters = MUTANTS[mid]
    root = WT / "scratch" / "rv12" / "delta" / "mut" / mid
    if root.exists():
        shutil.rmtree(root)
    root.mkdir(parents=True)
    arch = subprocess.run(
        "git -C %s archive %s projects/chirality-piping/core | tar -x -C %s" % (WT / "k3", COMMIT, root),
        shell=True, capture_output=True, text=True)
    assert arch.returncode == 0, arch.stderr
    rec = {"id": mid, "description": desc, "filters": filters}
    if path is not None:
        f = root / path
        text = f.read_text()
        assert text.count(old) == 1, (mid, text.count(old))
        f.write_text(text.replace(old, new))
        rec["file"] = path.split("/")[-1]
        rec["line"] = text[: text.index(old)].count("\n") + 1
        rec["old"], rec["new"] = old, new
    target = WT / "rv12-target" / ("dmut-" + mid)
    cmd = ["cargo", "test", "--offline", "--locked", "-j", "4", "--no-fail-fast",
           "--manifest-path", str(root / FK / "Cargo.toml")]
    if filters:
        cmd += ["--lib", "--"] + filters
    start = time.time()
    proc = subprocess.run(cmd, env=dict(ENV, CARGO_TARGET_DIR=str(target)), capture_output=True, text=True)
    log = proc.stdout + proc.stderr
    rec["wall_s"] = round(time.time() - start, 1)
    rec["rc"] = proc.returncode
    (OUT / (mid + ".log")).write_text(sanitize(log))
    rec["compiled"] = "could not compile" not in log and "error[" not in log
    res = re.findall(r"^test result: \w+\. (\d+) passed; (\d+) failed", log, re.M)
    rec["passed"] = sum(int(a) for a, _ in res)
    rec["failed"] = sum(int(b) for _, b in res)
    kills, other = [], []
    for m in re.finditer(r"^---- (\S+) stdout ----\n(.*?)(?=^---- |^failures:$|\Z)", log, re.M | re.S):
        name, body = m.group(1), m.group(2)
        found = re.findall(r"panicked at ([^:\n]+):(\d+):\d+:", body)
        if not found:
            other.append({"test": name, "why": "no panic location"})
            continue
        file, line = found[-1][0], int(found[-1][1])
        try:
            src = (root / FK / file).read_text().splitlines()[line - 1].strip()
        except Exception:
            src = "<unreadable>"
        norm = os.path.normpath(file)
        entry = {"test": name, "at": "%s:%d" % (norm, line), "source": src[:160]}
        if norm.startswith("tests/") and "assert" in src:
            kills.append(entry)
        else:
            other.append(entry)
    rec["behavioural_kills"] = kills
    rec["other_failures"] = other
    rec["killed"] = rec["compiled"] and bool(kills)
    shutil.rmtree(target, ignore_errors=True)
    shutil.rmtree(root, ignore_errors=True)
    (OUT / (mid + ".json")).write_text(json.dumps(rec, indent=1))
    return rec


def main():
    OUT.mkdir(parents=True, exist_ok=True)
    ids = list(IDS)
    results = {}
    if ids and ids[0].startswith("NONE"):
        r = run(ids.pop(0))
        results[r["id"]] = r
        print(r["id"], "rc=%d passed=%d failed=%d" % (r["rc"], r["passed"], r["failed"]), flush=True)
        assert r["rc"] == 0 and r["failed"] == 0 and r["passed"] > 0, "control failed"
    with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
        futs = {pool.submit(run, i): i for i in ids}
        for fut in concurrent.futures.as_completed(futs):
            r = fut.result()
            results[r["id"]] = r
            print(r["id"], "KILLED" if r["killed"] else "SURVIVED", "compiled=%s passed=%d failed=%d kills=%d other=%d wall=%s"
                  % (r["compiled"], r["passed"], r["failed"], len(r["behavioural_kills"]), len(r["other_failures"]), r["wall_s"]),
                  flush=True)
    prev = {}
    summary = OUT / "summary.json"
    if summary.exists():
        prev = json.loads(summary.read_text())
    prev.update(results)
    summary.write_text(json.dumps(prev, indent=1))
    print("DONE", flush=True)


if __name__ == "__main__":
    main()
