"""RV10: independent Fraction re-derivation of the formed coefficients' exact relative errors.
Inputs: the bits printed by the reviewer's probe on 134eefc24 (RV10C lines), taken exactly."""
import json, struct, sys
from fractions import Fraction as F
def fb(b): return struct.unpack('<d', struct.pack('<Q', b))[0]
def fx(b): return F(fb(b))
floor = None
rows = [json.loads(l.split(' ', 1)[1]) for l in open(sys.argv[1]) if l.startswith('RV10C ')]
print("label | L==2^-39 | (12E)I exact / 2^t - 1 | 2EI/L rel | 4EI/L rel | 6EI/L2 rel | 12EI/L3 rel | y==z")
tv = {"2^-1030": -1030, "2^-1040": -1040, "2^-1045": -1045, "2^-1050": -1050, "2^-1055": -1055, "S6a": None}
out = {}
for r in rows:
    E, I, L = fx(r["E"]), fx(r["I"]), fx(r["L"])
    assert L == F(1, 2**39), r
    ex = {k: F(k) * E * I / L**n for k, n in [(2, 1), (4, 1), (6, 2), (12, 3)]}
    formed = {2: fx(r["z2"]), 4: fx(r["z4"]), 6: fx(r["z6"]), 12: fx(r["z12"])}
    rel = {k: abs(formed[k] - ex[k]) / ex[k] for k in ex}
    t = tv[r["label"]]
    p = 12 * E * I
    dev = (p / (F(2) ** t) - 1) if t is not None else (p / (F(5, 2) * F(1, 2**1075)) - 1)
    yz = (r["y2"], r["y4"], r["y6"]) == (r["z2"], r["z4"], r["z6"])
    out[r["label"]] = {k: float(v) for k, v in rel.items()}
    print(f'{r["label"]:8} | {L == F(1, 2**39)} | {float(dev):+.3e} | {float(rel[2]):.6e} | {float(rel[4]):.6e} | {float(rel[6]):.3e} | {float(rel[12]):.3e} | {yz}')
# The candidate's hard-coded figures (m03_skew_scope.rs cases()): exact_errors (2EI/L, 4EI/L) to five digits, RV7's as stated.
hard = {"2^-1030": ((1.1378e-13, 5.6750e-14), (1.1e-13, 5.7e-14)), "2^-1040": ((1.1642e-10, 5.8208e-11), (1.16e-10, 5.8e-11)),
        "2^-1045": ((3.7253e-9, 1.8626e-9), (3.73e-9, 1.86e-9)), "2^-1050": ((1.1921e-7, 5.9605e-8), (1.19e-7, 5.96e-8))}
for lab, ((x2, x4), (r2, r4)) in hard.items():
    e2, e4 = out[lab][2], out[lab][4]
    print(f"{lab}: 2EI/L {e2:.6e} vs test {x2:.4e} (5 sig: {float(f'{e2:.4e}') == x2}), RV7 {r2}; 4EI/L {e4:.6e} vs test {x4:.4e} (5 sig: {float(f'{e4:.4e}') == x4}), RV7 {r4}")
