import re,struct,math
def f(h): return struct.unpack('<d',struct.pack('<Q',int(h,16)))[0]
def kind(x):
    if x==0: return 'zero'
    if not math.isfinite(x): return 'inf'
    if abs(x)<2.0**-1022: return 'SUB'
    return 'normal'
rows=[]
for l in open('<scratch>/rv7/sites.txt'):
    m=re.search(r'\("([^"]+)", \[([^\]]+)\]\)',l)
    if m: rows.append((m.group(1),[f(x.strip()) for x in m.group(2).split(',')]))
PARTIAL=[f(x) for x in '0x1a70000000000000, 0x1a60000000000000, 0x3ff0000000000000, 0x2220000000000000, 0x3ff0000000000000, 0x3ff0000000000000, 0x3e10000000000000'.split(', ')]
SUBI=[f(x) for x in '0x1a70000000000000, 0x1a60000000000000, 0x3ff0000000000000, 0x23e999999999999a, 0x3ff0000000000000, 0x3ff0000000000000, 0x3e10000000000000'.split(', ')]
def dv(a,b):
    try: return a/b
    except ZeroDivisionError: return math.copysign(math.inf,a) if a!=0 else math.nan
def seq(v):
    e,g,a,iy,iz,j,L=v
    out=[]
    L2=L*L; out.append(('L2',L2)); L3=L2*L; out.append(('L3',L3))
    p=e*a; out.append(('EA',p)); out.append(('EA/L',dv(p,L)))
    p=g*j; out.append(('GJ',p)); out.append(('GJ/L',dv(p,L)))
    k12=12.0*e; out.append(('12E',k12))
    p=k12*iy; out.append(('12EIy',p)); out.append(('12EIy/L3',dv(p,L3)))
    k6=6.0*e; out.append(('6E',k6)); p=k6*iy; out.append(('6EIy',p)); out.append(('6EIy/L2',dv(p,L2)))
    k4=4.0*e; out.append(('4E',k4)); p=k4*iy; out.append(('4EIy',p)); out.append(('4EIy/L',dv(p,L)))
    k2=2.0*e; out.append(('2E',k2)); p=k2*iy; out.append(('2EIy',p)); out.append(('2EIy/L',dv(p,L)))
    for k,LL in ((k12,L3),(k6,L2),(k4,L),(k2,L)):
        p=k*iz; out.append(('kEIz',p)); out.append(('/',dv(p,LL)))
    return out
for name,v in rows:
    s=seq(v)
    first=next(((i,n,x) for i,(n,x) in enumerate(s) if kind(x)!='normal'),None)
    subs=[kind(x) for n,x in s]
    print(f"{name:34s} first#{first[0]:2d} {kind(first[2]):6s} inputs_subnormal={[i for i,x in enumerate(v) if kind(x)=='SUB']} later_kinds={''.join(k[0] for k in subs)}")
for nm,v in (('PARTIAL',PARTIAL),('SUBI',SUBI)):
    s=seq(v); print(nm,[ (n,kind(x)) for n,x in s if kind(x)!='normal'][:6])
