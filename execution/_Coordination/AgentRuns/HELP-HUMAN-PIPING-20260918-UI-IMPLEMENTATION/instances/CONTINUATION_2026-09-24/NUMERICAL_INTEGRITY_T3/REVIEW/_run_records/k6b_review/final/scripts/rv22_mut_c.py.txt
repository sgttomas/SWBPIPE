"""RV22 mutation driver: each mutant from a clean `git archive` of the PR head, its own target, deleted
afterwards; one cargo job at -j 4, RUST_TEST_THREADS=2. Usage: rv22_mut.py <name>... (or 'all')."""
import json, os, shutil, subprocess, sys, time
WT = '<wt>'
HEAD = '011911e4e9ab8b73b502066089276ab356a03bf9'
H = 'projects/chirality-piping/core/solver/performance_harness'
RUST_TESTS = ['k6b_w1', 'k6b_bin', 'k6b_adapter', 'k6b_export']
MUTANTS = {
    'NONE': ('rust', []),
    'NONE-PY': ('py', []),
    'RV22-M6': ('rust', [(H + '/src/k6/w1/counts.rs', 'pub const SOLVE_TRACKER_PEAK_ROWS: u128 = 2816;', 'pub const SOLVE_TRACKER_PEAK_ROWS: u128 = 2048;')]),
    'RV22-M7': ('py', [(H + '/runner/k6_runner.py', "    if estimate > run['heap_cap_bytes'] // 2:", "    if estimate > run['rss_cap_bytes'] // 2:")]),
    'RV22-M5B': ('rust', [(H + '/src/bin/k6_observe/w1.rs', '        prefix_matches(j, full, &solve.outcome),', '        { let _ = (full, prefix_matches); true },')]),
    'RV22-C1': ('rust', [(H + '/src/k6/w1/counts.rs', '    c + c / 2', '    c')]),
    'RV22-C2': ('rust', [(H + '/src/k6/w1/counts.rs', '            + nf * PROFILE_ROW_EXTRA\n            + p_entries * w(l)\n', '            + nf * PROFILE_ROW_EXTRA\n')]),
    'RV22-C3': ('rust', [(H + '/src/k6/w1/staged.rs', '|| a.stop_rule_work > 0', '|| a.verification_work > 0')]),
    'RV22-C4': ('rust', [(H + '/src/k6/w1/staged.rs', 'own <= own_t && shared <= shared_t && (own == own_t || shared == shared_t)', 'own == own_t && shared <= shared_t')]),
}

def sh(cmd, cwd, env=None, log=None):
    t = time.time()
    with open(log, 'w') as fh:
        rc = subprocess.run(cmd, cwd=cwd, env=env, stdout=fh, stderr=subprocess.STDOUT).returncode
    return rc, time.time() - t

def run(name):
    kind, edits = MUTANTS[name]
    root = os.path.join(WT, 'rv22c', 'mut', name)
    target = os.path.join(WT, 'rv22c-target', 'mut', name)
    shutil.rmtree(root, ignore_errors=True); os.makedirs(root)
    arch = subprocess.run(['git', '-C', os.path.join(WT, 'k6b'), 'archive', HEAD, 'projects/chirality-piping/core'], capture_output=True, check=True).stdout
    subprocess.run(['tar', '-x', '-C', root], input=arch, check=True)
    for path, old, new in edits:
        p = os.path.join(root, path); s = open(p).read()
        assert s.count(old) == 1, (name, path, s.count(old))
        open(p, 'w').write(s.replace(old, new))
    log = os.path.join(WT, 'rv22c', 'logs', 'mut_%s.log' % name)
    if kind == 'rust':
        env = dict(os.environ, RUSTUP_TOOLCHAIN='1.97.1', RUSTUP_AUTO_INSTALL='0', CARGO_INCREMENTAL='0',
                   CARGO_TARGET_DIR=target, RUST_TEST_THREADS='2')
        cmd = ['cargo', 'test', '--offline', '--locked', '-j', '4', '--no-fail-fast'] + sum([['--test', t] for t in RUST_TESTS], [])
        rc, dt = sh(cmd, os.path.join(root, H), env, log)
    else:
        env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1')
        rc, dt = sh([sys.executable, '-m', 'unittest', 'test_k6_runner'], os.path.join(root, H, 'runner'), env, log)
    text = open(log).read()
    failed = sorted(set(l.split()[1] for l in text.splitlines() if l.startswith('test ') and l.rstrip().endswith('FAILED')))
    failed += sorted(set(l.split('(')[0].replace('FAIL: ', '').replace('ERROR: ', '').strip() for l in text.splitlines() if l.startswith(('FAIL: ', 'ERROR: '))))
    compile_error = 'error[E' in text or 'error: could not compile' in text
    result = {'mutant': name, 'kind': kind, 'rc': rc, 'seconds': round(dt, 1), 'compile_error': compile_error,
              'failed_tests': failed, 'result': 'passes' if rc == 0 else 'killed'}
    shutil.rmtree(target, ignore_errors=True); shutil.rmtree(root, ignore_errors=True)
    with open(os.path.join(WT, 'rv22c', 'logs', 'mut_results.jsonl'), 'a') as fh:
        fh.write(json.dumps(result) + '\n')
    print(json.dumps(result), flush=True)

names = list(MUTANTS) if sys.argv[1:] == ['all'] else sys.argv[1:]
for n in names:
    run(n)
