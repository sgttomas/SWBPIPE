#!/bin/bash
# RV21: re-run a subset of V-K's seeded faults and two of RV21's own, one process set per fault.
WT=<wt>
cd $WT/rv21/head/projects/chirality-piping/validation/benchmarks/numerical_robustness || exit 1
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=$WT/rv21-mut-target RUST_TEST_THREADS=2
mkdir -p $WT/rv21/logs/faults
for id in NONE RV21-A RV21-B VK-F13 VK-S2 VK-F17 VK-F06 VK-F04 VK-R02 VK-F99; do
  start=$(date +%s)
  FK_SEEDED_FAULT=$id cargo test --offline --locked -j 4 --features seeded-faults --lib \
    --test lane --test parity --test adapter --test files --test engine --test rcm \
    --test invariance --test feature_guard --test scale --no-fail-fast > $WT/rv21/logs/faults/$id.log 2>&1
  rc=$?
  end=$(date +%s)
  failed=$(grep -E "^test .* FAILED$" $WT/rv21/logs/faults/$id.log | sed 's/^test //; s/ \.\.\. FAILED$//' | tr '\n' ' ')
  passed=$(grep -cE "^test .* ok$" $WT/rv21/logs/faults/$id.log)
  panics=$(grep -c "unknown fault id" $WT/rv21/logs/faults/$id.log)
  echo "$id rc=$rc secs=$((end-start)) passed=$passed unknown_id_panics=$panics failed=[$failed]"
done
