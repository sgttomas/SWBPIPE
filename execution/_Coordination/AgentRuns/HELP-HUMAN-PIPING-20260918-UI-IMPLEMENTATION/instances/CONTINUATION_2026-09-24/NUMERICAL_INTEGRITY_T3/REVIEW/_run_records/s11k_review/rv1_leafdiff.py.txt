import json, subprocess, sys
BASE="6bb3ee490"; HEAD="4912dc636"
files=subprocess.run(["git","diff","--name-only",f"{BASE}..{HEAD}","--","projects/chirality-piping/fixtures","projects/chirality-piping/core/reporting/result_export/tests/fixtures"],capture_output=True,text=True).stdout.split()
def leaves(o,p=""):
    if isinstance(o,dict):
        for k,v in o.items(): yield from leaves(v,p+"/"+k)
    elif isinstance(o,list):
        for i,v in enumerate(o): yield from leaves(v,p+f"[{i}]")
    else: yield p,o
def load(rev,f):
    b=subprocess.run(["git","show",f"{rev}:{f}"],capture_output=True).stdout
    return b, json.loads(b)
for f in files:
    bb,a=load(BASE,f); hb,b=load(HEAD,f)
    la=dict(leaves(a)); lb=dict(leaves(b))
    keys=set(la)|set(lb)
    ch=[k for k in keys if la.get(k,object())!=lb.get(k,object()) or (type(la.get(k)) is not type(lb.get(k)))]
    # also compare repr to catch -0.0 vs 0.0
    ch2=[k for k in keys if repr(la.get(k))!=repr(lb.get(k))]
    print(f"{len(bb):>8} -> {len(hb):>8}  leaves {len(ch2):>4}  keys_equal={set(la)==set(lb)}  {f.split('chirality-piping/')[1]}")
