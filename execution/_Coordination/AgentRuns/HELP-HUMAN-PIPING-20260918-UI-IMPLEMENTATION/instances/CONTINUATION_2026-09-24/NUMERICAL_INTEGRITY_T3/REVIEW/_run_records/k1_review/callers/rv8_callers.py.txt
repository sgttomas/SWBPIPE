#!/usr/bin/env python3
"""RV7's independent lexer scan of callers (standard library only).

Usage: rv7_callers.py <tree root (projects/chirality-piping)> <out file>

Lexes every .rs file under the tree (comments and string/char literals
blanked, line structure kept), tracks brace depth to find the enclosing
`fn` of each token, and marks test context:
  - files whose module is declared under #[cfg(test)] (found by walking
    `mod name;` declarations from every lib.rs/main.rs, honouring #[path]),
    files under a `tests/` or `benches/` directory, and `build.rs` excluded;
  - inline `#[cfg(test)] mod x { ... }` blocks and `#[test]` fns.
Every occurrence of each NAME on an identifier boundary is listed.
"""
import os
import re
import sys

NAMES = [
    # existing named entries (brief item 1)
    "solve_structural_dense", "solve_structural_dense_binary64",
    "solve_structural_dense_with_force_terms", "solve_assembled_structural_dense",
    "solve_formation_checked_structural_dense",
    "solve_structural_sparse", "solve_structural_sparse_binary64",
    "factor_structural_ldlt", "negative_pair_witness", "verify_negative_direction",
    # refactored FK stages
    "finish_checked_factor", "evaluate_original_residual_bound", "audit_intended_action",
    "audit_load_row", "unaudited_row", "exact_scaled_rhs", "estimate_rcond",
    "factor_structural_profile", "audit_load_fidelity", "finish_structural",
    "prepare_structural", "prepare_assembled_structural", "prepare_formation_checked_structural",
    "prepare_structural_with_force_terms", "prepare_structural_binary64",
    # new FK sparse entries
    "SparsePattern", "SparseStiffness", "assemble_sparse_stiffness", "SparseAssemblyOptions",
    "StiffnessBlock", "reduce_assembled_sparse_system", "SparseReducedSystem",
    "SparseStructuralSystem", "AssembledSparseStructuralSystem", "FormationCheckedSparseSystem",
    "SparseSymmetryEvidence", "SparsePreparedSystem", "SparsePositiveFactor", "SparseStorageCounts",
    "prepare_sparse_structural", "prepare_assembled_sparse_structural",
    "prepare_formation_checked_sparse_structural", "prepare_sparse_structural_with_force_terms",
    "factor_sparse_structural_profile", "finish_sparse_structural", "audit_sparse_load_fidelity",
    "verify_sparse_negative_direction", "sparse_negative_pair_witness",
    "sparse_negative_pair_witness_counted",
    # new sparse_direct entries
    "order_sparse_structural", "factor_sparse_structural_ldlt", "solve_sparse_prepared",
    "solve_sparse_structural", "solve_assembled_sparse_structural",
    "solve_formation_checked_sparse_structural", "SparseStructuralOrdering",
    # SA
    "SparseAssemblyEvidence", "SparseEvidenceCounts", "EvidenceParts", "BodyEvidence",
    "AssemblyEvidence", "solve_binary64", "solve_assembled_with_formation_check",
    "dense_symmetry_view",
]

IDENT = re.compile(r"[A-Za-z0-9_]")


def lex(src):
    out = []
    i, n = 0, len(src)
    while i < n:
        c = src[i]
        if src.startswith("//", i):
            while i < n and src[i] != "\n":
                i += 1
            continue
        if src.startswith("/*", i):
            depth, i = 1, i + 2
            while i < n and depth:
                if src.startswith("/*", i):
                    depth, i = depth + 1, i + 2
                elif src.startswith("*/", i):
                    depth, i = depth - 1, i + 2
                else:
                    if src[i] == "\n":
                        out.append("\n")
                    i += 1
            continue
        prev_ident = i > 0 and IDENT.match(src[i - 1])
        m = None
        if not prev_ident:
            m = re.match(r'b?r(#*)"', src[i:])
        if m:
            hashes = m.group(1)
            j = i + m.end()
            end = src.find('"' + hashes, j)
            end = n if end < 0 else end
            out.append('""' + "\n" * src[i:end].count("\n"))
            i = end + 1 + len(hashes)
            continue
        if c == '"' or (not prev_ident and src.startswith('b"', i)):
            j = i + (2 if c == "b" else 1)
            while j < n and src[j] != '"':
                if src[j] == "\\":
                    j += 1
                j += 1
            out.append('""' + "\n" * src[i:j].count("\n"))
            i = j + 1
            continue
        if c == "'":
            m2 = re.match(r"'(\\.[^']*|[^'\\])'", src[i:])
            if m2:
                out.append("' '")
                i += m2.end()
                continue
        out.append(c)
        i += 1
    return "".join(out)


def mod_decls(code):
    """(name, offset, cfg_test) for `mod name;` declarations."""
    res = []
    for m in re.finditer(r"(?<![A-Za-z0-9_])mod\s+([A-Za-z0-9_]+)\s*;", code):
        head = code[: m.start()]
        k = max(head.rfind(";"), head.rfind("}"), head.rfind("{"))
        attrs = head[k + 1:]
        res.append((m.group(1), m.start(), "cfg(test)" in attrs.replace(" ", ""), attrs))
    return res


def walk_modules(root):
    """Map file -> is_test, by walking module declarations from crate roots."""
    status = {}
    roots = []
    for dp, dn, fn in os.walk(root):
        dn[:] = [d for d in dn if d not in ("target", "node_modules", ".git")]
        if "Cargo.toml" in fn:
            for r in ("src/lib.rs", "src/main.rs"):
                p = os.path.join(dp, r)
                if os.path.isfile(p):
                    roots.append(p)
            bins = os.path.join(dp, "src", "bin")
            if os.path.isdir(bins):
                for f in os.listdir(bins):
                    if f.endswith(".rs"):
                        roots.append(os.path.join(bins, f))
    stack = [(p, False) for p in roots]
    while stack:
        path, test = stack.pop()
        if path in status and (status[path] is False or status[path] == test):
            continue
        status[path] = test if path not in status else (status[path] and test)
        raw = open(path, encoding="utf-8").read()
        code = lex(raw)
        base = os.path.basename(path)
        if base in ("lib.rs", "main.rs", "mod.rs"):
            child_dir = os.path.dirname(path)
        else:
            child_dir = path[:-3]
        raw_lines = raw.split("\n")
        for name, at, cfg_test, attrs in mod_decls(code):
            if "#[path" in attrs:
                line = code[:at].count("\n")
                target = None
                for l in range(line, -1, -1):
                    if "#[path" in raw_lines[l]:
                        target = os.path.join(os.path.dirname(path), raw_lines[l].split('"')[1])
                        break
            else:
                flat = os.path.join(child_dir, name + ".rs")
                target = flat if os.path.isfile(flat) else os.path.join(child_dir, name, "mod.rs")
            if os.path.isfile(target):
                stack.append((os.path.normpath(target), test or cfg_test))
    return status


def test_regions(code):
    """Byte ranges of inline #[cfg(test)] items and #[test] fns."""
    regions = []
    for m in re.finditer(r"#\[\s*(cfg\s*\(\s*test\s*\)|test)\s*\]", code):
        j = m.end()
        # find the item's body
        k = j
        depth = 0
        started = False
        while k < len(code):
            ch = code[k]
            if ch == "{":
                depth += 1
                started = True
            elif ch == "}":
                depth -= 1
                if started and depth == 0:
                    break
            elif ch == ";" and not started:
                break
            k += 1
        regions.append((m.start(), k + 1))
    return regions


def enclosing_fns(code):
    """For each offset, the innermost enclosing fn name (list of (start,end,name))."""
    spans = []
    for m in re.finditer(r"(?<![A-Za-z0-9_])fn\s+([A-Za-z0-9_]+)", code):
        k = m.end()
        depth = 0
        started = False
        while k < len(code):
            ch = code[k]
            if ch == "{":
                depth += 1
                started = True
            elif ch == "}":
                depth -= 1
                if started and depth == 0:
                    break
            elif ch == ";" and not started:
                break
            k += 1
        spans.append((m.start(), k + 1, m.group(1)))
    return spans


def main():
    root = sys.argv[1]
    out = open(sys.argv[2], "w")
    status = walk_modules(root)
    files = []
    for dp, dn, fn in os.walk(root):
        dn[:] = [d for d in dn if d not in ("target", "node_modules", ".git", "execution")]
        for f in fn:
            if f.endswith(".rs"):
                files.append(os.path.normpath(os.path.join(dp, f)))
    files.sort()
    unreached = [f for f in files if f not in status]
    for path in files:
        rel = os.path.relpath(path, root)
        in_tests_dir = "/tests/" in "/" + rel or "/benches/" in "/" + rel
        file_test = status.get(path, None)
        code = lex(open(path, encoding="utf-8").read())
        regions = test_regions(code)
        fns = enclosing_fns(code)
        for name in NAMES:
            for m in re.finditer(r"(?<![A-Za-z0-9_])" + re.escape(name) + r"(?![A-Za-z0-9_])", code):
                at = m.start()
                line = code[:at].count("\n") + 1
                enclosing = [s for s in fns if s[0] <= at < s[1]]
                fn_name = min(enclosing, key=lambda s: s[1] - s[0])[2] if enclosing else "-"
                is_def = bool(re.search(r"(fn|struct|trait|enum|impl(<[^>]*>)?|for|type)\s*$", code[max(0, at - 40):at]))
                test = (in_tests_dir or file_test is True
                        or any(a <= at < b for a, b in regions))
                ctx = "test" if test else ("unreached" if file_test is None and not in_tests_dir else "non-test")
                kind = "def" if is_def else "use"
                src_line = code.split("\n")[line - 1].strip()
                out.write(f"{rel}:{line}\t{ctx}\t{kind}\t{name}\t[{fn_name}]\t{src_line[:160]}\n")
    out.write("# unreached .rs files (not in any crate module tree; tests/ dirs expected):\n")
    for f in unreached:
        out.write("#   " + os.path.relpath(f, root) + "\n")


if __name__ == "__main__":
    main()
