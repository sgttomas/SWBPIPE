#!/usr/bin/env python3
"""RV11: lexer scan of Rust sources for K2b's new names, outside test code.

Strips comments, string/char/raw-string literals; removes `#[cfg(test)]` items
(the attribute's following item, brace-matched) and skips test files (tests/
directories, *_tests.rs / tests.rs / test_*.rs files and files included only
under #[cfg(test)] via #[path]). Prints every non-test hit per name.
"""
import os, re, sys

NAMES = ["ForceScale", "force_scaled_value", "force_scaled_matrix", "force_scaled", "ForceScaleCensus",
         "with_force_scale", "force_scaled_reactions", "unscale_for_publication", "unscale_descriptive",
         "unscale_structural_solution", "unscale_structural_error", "ForceScaleReason", "RangeTrigger",
         "ForceScalingRefusal", "ForceScaledError", "Representability", "PublishedValue",
         "RecordRepresentability", "RecordOutcome", "ForceScaledSolution", "new_force_scaled",
         "solve_force_scaled", "solve_force_scaled_with_formation_check", "EvidenceRepresentation",
         "ForceScalingCase", "ForceScalingOutcome", "solve_with_force_scaling", "exact_normal_scaling",
         "force_scale", "force_scale_census", "force_scaled_primitives", "force_scaled_formation_source",
         "force_scaled_inputs", "unscaled_evidence", "force_scaled_outcome", "evaluate_force_scaled"]

def strip(src):
    out = []; i = 0; n = len(src)
    while i < n:
        c = src[i]
        if src.startswith("//", i):
            j = src.find("\n", i); j = n if j < 0 else j
            out.append(" " * (j - i)); i = j; continue
        if src.startswith("/*", i):
            depth = 1; j = i + 2
            while j < n and depth:
                if src.startswith("/*", j): depth += 1; j += 2
                elif src.startswith("*/", j): depth -= 1; j += 2
                else: j += 1
            out.append(re.sub(r"[^\n]", " ", src[i:j])); i = j; continue
        m = re.match(r'b?r(#*)"', src[i:i+20])
        if m and (i == 0 or not (src[i-1].isalnum() or src[i-1] == "_")):
            hashes = m.group(1); start = i + len(m.group(0)); end = src.find('"' + hashes, start)
            end = n if end < 0 else end + 1 + len(hashes)
            out.append(re.sub(r"[^\n]", " ", src[i:end])); i = end; continue
        if c == '"' or (c == "b" and i + 1 < n and src[i+1] == '"' and (i == 0 or not (src[i-1].isalnum() or src[i-1] == "_"))):
            j = i + (2 if c == "b" else 1)
            while j < n and src[j] != '"':
                j += 2 if src[j] == "\\" else 1
            out.append(re.sub(r"[^\n]", " ", src[i:j+1])); i = j + 1; continue
        if c == "'":
            m = re.match(r"'(\\.|\\u\{[0-9a-fA-F]+\}|\\x[0-9a-fA-F]{2}|[^\\'])'", src[i:i+12])
            if m:
                out.append(" " * len(m.group(0))); i += len(m.group(0)); continue
        out.append(c); i += 1
    return "".join(out)

def remove_cfg_test_items(code):
    # remove `#[cfg(test)]` + following item up to its matching brace or semicolon
    res = []; i = 0
    pat = re.compile(r"#\s*\[\s*cfg\s*\(\s*test\s*\)\s*\]")
    while True:
        m = pat.search(code, i)
        if not m:
            res.append(code[i:]); break
        res.append(code[i:m.start()])
        j = m.end()
        # find first '{' or ';' after the item header
        k = j
        while k < len(code) and code[k] not in "{;":
            k += 1
        if k < len(code) and code[k] == ";":
            end = k + 1
        else:
            depth = 0; end = k
            while end < len(code):
                if code[end] == "{": depth += 1
                elif code[end] == "}":
                    depth -= 1
                    if depth == 0: end += 1; break
                end += 1
        res.append(re.sub(r"[^\n]", " ", code[m.start():end])); i = end
    return "".join(res)

def is_test_file(path):
    parts = path.split(os.sep)
    base = os.path.basename(path)
    return ("tests" in parts or "benches" in parts or base == "tests.rs" or base.endswith("_tests.rs")
            or base.startswith("test_") or base.endswith("_test.rs"))

root = sys.argv[1]
test_included = set()
files = []
for d, _, fs in os.walk(root):
    if "/target" in d or "/execution" in d or "/node_modules" in d:
        continue
    for f in fs:
        if f.endswith(".rs"):
            files.append(os.path.join(d, f))
# files included via #[path] under #[cfg(test)]
for p in files:
    src = open(p, encoding="utf-8", errors="replace").read()
    for m in re.finditer(r"#\s*\[\s*cfg\s*\(\s*test\s*\)\s*\]\s*#\s*\[\s*path\s*=\s*\"([^\"]+)\"\s*\]", src):
        test_included.add(os.path.normpath(os.path.join(os.path.dirname(p), m.group(1))))
    for m in re.finditer(r"#\s*\[\s*path\s*=\s*\"([^\"]+)\"\s*\]\s*#\s*\[\s*cfg\s*\(\s*test\s*\)\s*\]", src):
        test_included.add(os.path.normpath(os.path.join(os.path.dirname(p), m.group(1))))
    for m in re.finditer(r"#\s*\[\s*cfg\s*\(\s*test\s*\)\s*\]\s*(?:pub(?:\([^)]*\))?\s+)?mod\s+(\w+)\s*;", src):
        base = os.path.dirname(p); stem = os.path.splitext(os.path.basename(p))[0]
        for cand in [os.path.join(base, m.group(1) + ".rs"), os.path.join(base, stem, m.group(1) + ".rs"), os.path.join(base, m.group(1), "mod.rs")]:
            test_included.add(os.path.normpath(cand))
hits = {}
for p in sorted(files):
    if is_test_file(p) or os.path.normpath(p) in test_included:
        continue
    code = remove_cfg_test_items(strip(open(p, encoding="utf-8", errors="replace").read()))
    for name in NAMES:
        for m in re.finditer(r"(?<![A-Za-z0-9_])" + re.escape(name) + r"(?![A-Za-z0-9_])", code):
            line = code.count("\n", 0, m.start()) + 1
            hits.setdefault(os.path.relpath(p, root), []).append((line, name))
for f, hs in sorted(hits.items()):
    names = sorted(set(n for _, n in hs))
    print(f"{f}: {len(hs)} hits: {', '.join(names)}")
print("test-only files excluded (by #[path]/mod under cfg(test)):", len(test_included))
