#!/usr/bin/env python3
"""RV25: check each new *_MERGE/RECORD.md's merge facts against GitHub and Git.

Run from the repository root of a Git working tree whose object store has the
merged commits (read-only: `gh pr view`, `gh run view`, `gh run view --log`,
`git cat-file`, `git merge-base`, `git rev-parse`). No writes.
    python3 merge_facts.py.txt <T3 folder>
For each slice it reads RECORD.md's claimed PR, merge SHA, merge time, head,
"Main was" and run IDs, and prints what GitHub and Git say, with a verdict per
fact. For each run it prints workflow, event, head SHA and conclusion; for the
workflow_dispatch run it reads "target_base" from the run log.
"""
import json
import re
import subprocess
import sys

SLICES = ["F1B", "K5", "K6", "K4", "KF1", "VK", "K6B", "KF3", "KF2"]


def run(*args, check=True):
    return subprocess.run(list(args), check=check, capture_output=True, text=True).stdout


def full(sha):
    return run("git", "rev-parse", sha + "^{commit}").strip()


def main():
    t3 = sys.argv[1].rstrip("/")
    for s in SLICES:
        rec = open(f"{t3}/IMPLEMENTATION/{s}_MERGE/RECORD.md", encoding="utf-8").read()
        pr = re.search(r"pull/(\d+)", rec).group(1)
        m = re.search(r"merged it on (\S+) at (\S+?)Z as `([0-9a-f]{40})`: .*?--match-head-commit ([0-9a-f]+)", rec)
        date, time, merge_sha, match_head = m.groups()
        head = re.search(r"\*\*Candidate head:\*\* `([0-9a-f]{40})`", rec).group(1)
        mainwas = re.search(r"Main was `([0-9a-f]+)`", rec)
        runs = sorted(set(re.findall(r"\b(3\d{10})\b", rec)))
        pj = json.loads(run("gh", "pr", "view", pr, "--repo", "sgttomas/chirality", "--json",
                            "state,mergeCommit,mergedAt,headRefOid,mergedBy,headRefName"))
        print(f"== {s}: PR #{pr} ({pj['headRefName']})")
        gh_merge = pj["mergeCommit"]["oid"]
        print(f"   merge sha: record {merge_sha[:12]} github {gh_merge[:12]} "
              f"{'OK' if gh_merge == merge_sha else 'MISMATCH'}")
        rec_time = f"{date}T{time}Z"
        print(f"   merged at: record {rec_time} github {pj['mergedAt']} "
              f"{'OK' if pj['mergedAt'] == rec_time else 'MISMATCH'}; by {pj['mergedBy']['login']}; state {pj['state']}")
        print(f"   head: record {head[:12]} github {pj['headRefOid'][:12]} "
              f"{'OK' if pj['headRefOid'] == head else 'MISMATCH'}; match-head {match_head} "
              f"{'OK' if head.startswith(match_head) else 'MISMATCH'}")
        parents = run("git", "rev-list", "--parents", "-n", "1", gh_merge).split()[1:]
        print(f"   merge parents: {[p[:12] for p in parents]}; second parent is head: "
              f"{parents[1] == head if len(parents) > 1 else False}")
        if mainwas:
            mw = full(mainwas.group(1))
            anc = subprocess.run(["git", "merge-base", "--is-ancestor", mw, head]).returncode == 0
            print(f"   'Main was' {mainwas.group(1)}: equals first parent: {parents[0] == mw}; "
                  f"ancestor of head: {anc}")
        else:
            anc = subprocess.run(["git", "merge-base", "--is-ancestor", parents[0], head]).returncode == 0
            print(f"   no 'Main was' line; first parent {parents[0][:12]} ancestor of head: {anc}")
        for r in runs:
            rj = json.loads(run("gh", "run", "view", r, "--repo", "sgttomas/chirality", "--json",
                                "workflowName,event,headSha,conclusion,status,createdAt"))
            line = (f"   run {r}: {rj['workflowName']} / {rj['event']} / head {rj['headSha'][:12]} "
                    f"{'(=head)' if rj['headSha'] == head else '(NOT this head)'} / {rj['conclusion']} / {rj['createdAt']}")
            if rj["event"] == "workflow_dispatch":
                log = run("gh", "run", "view", r, "--repo", "sgttomas/chirality", "--log", check=False)
                tb = re.search(r'"target_base": "([0-9a-f]{40})"', log)
                if tb:
                    tbv = tb.group(1)
                    claimed = re.search(rf"\*\*{r}\*\* \(target_base `([0-9a-f]{{40}})`\)", rec)
                    anc = subprocess.run(["git", "merge-base", "--is-ancestor", tbv, rj["headSha"]]).returncode == 0
                    line += (f"\n      target_base in log {tbv[:12]}; record "
                             f"{claimed.group(1)[:12] if claimed else '(not stated with this run)'} "
                             f"{'OK' if claimed and claimed.group(1) == tbv else ''}; ancestor of run head: {anc}; "
                             f"equals merge first parent: {tbv == parents[0]}")
                else:
                    line += "\n      target_base not found in log"
            print(line)


if __name__ == "__main__":
    main()
