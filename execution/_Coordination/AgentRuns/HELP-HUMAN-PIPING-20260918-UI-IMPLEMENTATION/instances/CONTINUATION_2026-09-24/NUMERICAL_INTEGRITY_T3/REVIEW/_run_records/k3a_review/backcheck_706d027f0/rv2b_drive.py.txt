#!/usr/bin/env python3
"""RV2 backcheck driver (706d027f0): frame_kernel full suite, then R2, R1, M5, M6
mutants in a separate scratch copy with its own target. Waits for cargo and the sweep."""
import json, os, re, subprocess, time
from pathlib import Path
FKREL = "projects/chirality-piping/core/solver/frame_kernel"
SRC_FK = Path("<scratch>/cand") / FKREL
MUT_FK = Path("<scratch>/mut") / FKREL
W = Path("<scratch>/work")
BASE = dict(os.environ, RUSTUP_TOOLCHAIN="1.97.1", CARGO_INCREMENTAL="0", RUSTUP_AUTO_INSTALL="0")
MUTANTS = [
    ("R2_atan_tail_summed_largest_first", "        for term in terms[1..count].iter().rev() {\n",
     "        for term in terms[1..count].iter() {\n"),
    ("R1_atan_stop_threshold_off_by_one",
     "        let threshold = i128::from(t.exponent) - i128::from(self.precision) - 1;\n",
     "        let threshold = i128::from(t.exponent) - i128::from(self.precision);\n"),
    ("M5_atan_one_reduction_fewer",
     "            if reductions == ATAN_MAX_REDUCTIONS {\n                return Err(WideError::ArctangentLimit);\n            }\n",
     "            if reductions == ATAN_MAX_REDUCTIONS - 1 {\n                break;\n            }\n"),
    ("M6_atan_truncated_series", "        for n in 1..=ATAN_MAX_TERMS {\n", "        for n in 1..=8 {\n"),
]
def wait_host():
    n = 0
    while (subprocess.run(["pgrep", "-x", "cargo"], capture_output=True).returncode == 0 or
           subprocess.run(["pgrep", "-f", "python[0-9.]* .*run_evidence_sweep"], capture_output=True).returncode == 0):
        time.sleep(60); n += 1
        if n > 360: raise SystemExit("host busy 6 h")
    return n
def run(cmd, cwd, target, log):
    waited = wait_host()
    p = subprocess.run(cmd, cwd=cwd, env=dict(BASE, CARGO_TARGET_DIR=target), capture_output=True, text=True)
    (W / "logs" / log).write_text(p.stdout + p.stderr)
    return p, waited
p, wt = run(["cargo", "--version"], SRC_FK, "<rv2-target>/cand", "toolchain.txt")
p, wt = run(["cargo", "test", "--offline", "--locked"], SRC_FK, "<rv2-target>/cand", "cand_full_debug.log")
print("cand_full exit", p.returncode, "waited", wt, "min", flush=True)
src = MUT_FK / "src/structural/retained/wide.rs"
orig = src.read_text()
res = []
try:
    for name, old, new in MUTANTS:
        assert orig.count(old) == 1, name
        src.write_text(orig.replace(old, new))
        p, wt = run(["cargo", "test", "--offline", "--locked", "--lib", "retained", "--", "--test-threads=4"],
                    MUT_FK, "<rv2-target>/mut", "../mut/%s.log" % name)
        src.write_text(orig)
        text = p.stdout + p.stderr
        failed = sorted(set(re.findall(r"^test (\S+) \.\.\. FAILED", text, re.M)))
        compiled = "Compiling open_pipe_stress_frame_kernel" in text
        first = re.findall(r"^(angle|atanpos) (\S+) ", text, re.M)
        res.append(dict(mutant=name, exit=p.returncode, compiled=compiled, killed=p.returncode != 0 and bool(failed),
                        killing_tests=failed, first_vector=first[0][1] if first else None,
                        summary=[l for l in text.splitlines() if l.startswith("test result:")]))
        print(name, "KILLED" if res[-1]["killed"] else "SURVIVED", "compiled" if compiled else "NOT-COMPILED",
              res[-1]["first_vector"], flush=True)
finally:
    src.write_text(orig)
(W / "mut" / "mutation_results.json").write_text(json.dumps(res, indent=1) + "\n")
print("DONE", flush=True)
