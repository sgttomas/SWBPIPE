#!/bin/bash
. <scratch>/runs/env.sh
T=$S/runs/t9; mkdir -p $T
for tree in base cand; do
  P=$S/$tree/projects/chirality-piping
  (cd $P/core/product_physics && nice cargo build --offline --locked --release --example fixdiff_s11g) > $T/$tree.build.log 2>&1 || { echo "$tree build failed" >> $T/status; exit 1; }
  cp $CARGO_TARGET_DIR/release/examples/fixdiff_s11g $T/fixdiff_$tree
  for root in fixtures core validation; do
    mkdir -p $T/$tree/$root
    (cd $P && $T/fixdiff_$tree $root $T/$tree/$root) > $T/$tree.$root.log 2>&1
    echo "$tree $root exit=$? $(tail -1 $T/$tree.$root.log)" >> $T/status
  done
  rm -rf $CARGO_TARGET_DIR/release
  df -h / | tail -1 >> $T/status
done
echo DONE3 >> $T/status
