#!/usr/bin/env python3
"""V1 VERIFY_R5: the K3a arctangent as specified (R5_4_CURVED.md §2 step 3, D1 §6 K3a row),
emulated with every operation correctly rounded to p = 128 bits (round to nearest even),
against a 120-digit Decimal reference. Standard library only.
Measures the error in ulps of p of phi = 2^(k+1) * atan_series(t_k), t_0 = s/(1+c)."""
import json, math, sys
from fractions import Fraction as Fr
from decimal import Decimal as D, getcontext
getcontext().prec = 120
P = 128

def rnd(x):
    x = Fr(x)
    if x == 0:
        return x
    s = -1 if x < 0 else 1
    a = abs(x)
    e = a.numerator.bit_length() - a.denominator.bit_length()
    if Fr(2) ** e > a:
        e -= 1
    sc = Fr(2) ** (P - 1 - e)
    m = a * sc
    q, r = divmod(m.numerator, m.denominator)
    if 2 * r > m.denominator or (2 * r == m.denominator and q % 2):
        q += 1
    return s * Fr(q) / sc

def rsqrt(x):
    # correctly rounded sqrt at p: integer sqrt at high precision then round
    x = Fr(x)
    sh = 2 * (P + 64)
    n = math.isqrt(x.numerator * 4 ** sh // x.denominator)
    # n/2^sh is sqrt truncated; the extra 64 bits make the rounding decision exact except in ties
    return rnd(Fr(n, 2 ** sh))

def ulp(x):
    a = abs(Fr(x))
    e = a.numerator.bit_length() - a.denominator.bit_length()
    if Fr(2) ** e > a:
        e -= 1
    return Fr(2) ** (e - P + 1)

def atan_spec(s, c):
    t = rnd(Fr(s) / rnd(1 + Fr(c)))
    k = 0
    while t >= Fr(1, 20):
        t = rnd(t / rnd(1 + rsqrt(rnd(1 + rnd(t * t)))))
        k += 1
    t2 = rnd(t * t)
    term, n, acc, nterms = t, 1, Fr(0), 0
    while True:
        add = rnd(term / n)
        if acc != 0 and abs(add) < ulp(acc) / 4:
            break
        acc = rnd(acc + add)
        term = rnd(-term * t2)
        n += 2
        nterms += 1
    return acc * 2 ** (k + 1), k, nterms

def dref(phi_d):
    return phi_d

out = []
for phi in [1e-12, 1e-6, 1e-3, 0.1, 0.5, 1.0, math.pi / 2, 2.0, 3.0, 3.1, math.pi - 1e-6, math.pi - 1e-7, 1e-300]:
    # exact inputs: s, c are binary64 sin/cos of phi (any consistent pair); the reference is the
    # 120-digit atan2 of those exact inputs, so the test isolates the arctangent itself
    s, c = math.sin(phi), math.cos(phi)
    Ds, Dc = D(Fr(s).numerator) / D(Fr(s).denominator), D(Fr(c).numerator) / D(Fr(c).denominator)
    # reference: 2*atan(s/(1+c)) by Decimal with many reductions
    t = Ds / (1 + Dc)
    kk = 0
    while t > D("1e-10"):
        t = t / (1 + (1 + t * t).sqrt()); kk += 1
    ss, term, n = D(0), t, 1
    while ss == 0 or abs(term / n) > abs(ss) * D(10) ** -118:
        ss += term / n; term = -term * t * t; n += 2
    ref = ss * 2 ** (kk + 1)
    got, k, nt = atan_spec(s, c)
    gotD = D(got.numerator) / D(got.denominator)
    u = ulp(got)
    err_ulps = float(abs(gotD - ref) / (D(u.numerator) / D(u.denominator)))
    out.append(dict(phi=phi, reductions=k, series_terms=nt, err_ulps_p128=err_ulps))
    print(json.dumps(out[-1]))
print(json.dumps({"max_err_ulps": max(o["err_ulps_p128"] for o in out), "max_reductions": max(o["reductions"] for o in out),
                  "max_terms": max(o["series_terms"] for o in out)}))
