#!/usr/bin/env python3
"""RV8's own K1 x K2a interaction mutants (the delta check).
Usage: rv8_delta_mutants.py <core-dir> <id>. Each is an exact text replacement in
FK/structural/sparse.rs that must match once."""
import sys

core, mid = sys.argv[1], sys.argv[2]
SP = core + "/solver/frame_kernel/src/structural/sparse.rs"
FRAMES = """    for element in frames {
        check_node(element.node_i.index, node_count)?;
        check_node(element.node_j.index, node_count)?;
        formed.push((
            element.node_i.index,
            element.node_j.index,
            element.global_stiffness()?,
        ));
    }
"""
MUTANTS = {
    # Every frame's and user's node indices are checked before any element is
    # formed. A model whose element k is refused at formation (K2a) and whose
    # later element has a node outside the model then reports InvalidNodeIndex,
    # where the dense assembly reports element k's NumericalRange.
    "RV8-K2A-NODESFIRST": [(SP, FRAMES, """    for element in frames {
        check_node(element.node_i.index, node_count)?;
        check_node(element.node_j.index, node_count)?;
    }
    for element in users {
        check_node(element.node_i.index, node_count)?;
        check_node(element.node_j.index, node_count)?;
    }
""" + FRAMES)],
    # A label-dependent formation order: frames formed (and refused) in
    # ascending (node_i, node_j) order; accumulation is unchanged.
    "RV8-K2A-SORTED": [(SP, FRAMES, """    let mut by_label: Vec<&FrameElement> = frames.iter().collect();
    by_label.sort_by_key(|e| (e.node_i.index, e.node_j.index));
    for element in by_label {
        element.global_stiffness()?;
    }
""" + FRAMES)],
    # The refusal's name is replaced by the first failing coefficient's
    # family only ("GJ/L: G*J" for every refusal): the same variant, another name.
    "RV8-K2A-NAME": [(SP, "            element.global_stiffness()?,\n        ));\n    }\n    for element in users {",
                      "            element.global_stiffness().map_err(|e| match e {\n                FrameKernelError::NumericalRange { .. } => FrameKernelError::NumericalRange { name: \"GJ/L: G*J\" },\n                other => other,\n            })?,\n        ));\n    }\n    for element in users {")],
}
if mid not in MUTANTS:
    sys.exit("unknown mutant " + mid)
for path, old, new in MUTANTS[mid]:
    text = open(path).read()
    assert text.count(old) == 1, (mid, path, text.count(old))
    open(path, "w").write(text.replace(old, new))
print("applied", mid)
