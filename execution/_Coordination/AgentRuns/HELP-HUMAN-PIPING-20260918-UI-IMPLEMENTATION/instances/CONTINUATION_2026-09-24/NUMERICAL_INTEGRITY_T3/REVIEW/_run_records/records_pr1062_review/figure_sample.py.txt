def main():
    for line, fig, src, rx in CASES:
        try:
            txt = open(src, encoding="utf-8", errors="replace").read().splitlines()
        except FileNotFoundError:
            print(f"V1:{line} [{fig}] {src}: FILE NOT FOUND")
            continue
        hits = [(i + 1, l) for i, l in enumerate(txt) if re.search(rx, l)]
        if hits:
            h = "; ".join(f":{n} {l.strip()[:150]}" for n, l in hits[:2])
            print(f"V1:{line} [{fig}] {src}: {len(hits)} hit(s) {h}")
        else:
            print(f"V1:{line} [{fig}] {src}: NOT FOUND")


if __name__ == "__main__":
    main()
