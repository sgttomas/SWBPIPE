#!/bin/bash
# RV18 delta: one runner mutant from a clean archive of cd325c1fe8e56e536ef2a7503a3ab3f97b1ed01a; the runner's tests (unittest, the classes the wrapper collects).
set -u
W=<wt>
M=$1; SCRIPT=${2:-rv18_mutants.py}
ROOT=$W/rv18/pymut/$M
rm -rf $ROOT; mkdir -p $ROOT $W/rv18/mut-logs
(cd $W/k6 && git archive cd325c1fe8e56e536ef2a7503a3ab3f97b1ed01a projects/chirality-piping/core/solver/performance_harness) | tar -x -C $ROOT
if [ "$M" != NONE-py ]; then python3 -B $W/rv18/scripts/$SCRIPT make $M $ROOT || exit 9; fi
cd $ROOT/projects/chirality-piping/core/solver/performance_harness/runner
start=$(date +%s)
PYTHONDONTWRITEBYTECODE=1 python3 -m unittest test_k6_runner > $W/rv18/mut-logs/py-$M.log 2>&1
code=$?
echo "=== $M exit $code elapsed $(( $(date +%s) - start )) s" >> $W/rv18/mut-logs/py-$M.log
rm -rf $ROOT
echo "$M exit $code"
