#!/bin/bash
# RV17 delta check: NONE and the three mutants on the PR head 6fa422979, strictly one cargo job at a time,
# -j 4, RUST_TEST_THREADS=4 (K6's timed runs share the host).
cd <wt>/scratch/rv17/runs
export REV=6fa422979
: > delta/MUTANTS.txt
while read id patcher crates; do
  [ -z "$id" ] && continue
  KEEP=0 JOBS=4 ./run_mutant3.sh $id $patcher $crates > delta/logs/$id.driver 2>&1
done <<LIST
D-NONE NONE solver/nonlinear_integration product_physics
D-F1B-M2 mutate_f1b.py solver/nonlinear_integration product_physics
D-RV17-M1 mutate_rv17.py solver/nonlinear_integration product_physics
D-RV17-M2 mutate_rv17.py solver/nonlinear_integration product_physics
LIST
echo QUEUE4-DONE >> delta/MUTANTS.txt
