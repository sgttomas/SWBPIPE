"""K6 D: sparse bytes per pattern and RCM profile entry, and stage medians by size (from d_tables.json).

Usage: python3 sparse_and_timing.py <d_tables.json> <counts.jsonl>
"""
import json
import sys

d = json.load(open(sys.argv[1]))
counts = {}
for line in open(sys.argv[2]):
    o = json.loads(line)
    counts[o['model']] = o
print('# Sparse path: peak requested heap (move model) per pattern entry and per RCM profile entry, and against E_adm.')
print('model | members | nnz | P_rcm | heap_move MiB | B/nnz | B/P_rcm | heap_move/E_adm | outcome')
for r in sorted([x for x in d['results'] if x['mode'] == 'sparse'], key=lambda x: (x['family'], x['members'], x['model'])):
    c = counts[r['model']]
    print('%s | %d | %d | %d | %.1f | %.1f | %.1f | %.3f | %s' % (
        r['model'], r['members'], c['pattern_entries'], c['rcm_profile_entries'], r['heap_move'] / 2 ** 20,
        r['heap_move'] / c['pattern_entries'], r['heap_move'] / c['rcm_profile_entries'],
        r['heap_move'] / r['estimate_adm_bytes'], r['outcome']))
print()
print('# Stage medians (s): geometry, factor (or lane_solve), finish, the staged total (sum of stage medians), entry_checked.')
print('model | mode | members | geometry | factor | finish | staged total | entry_checked | wall | load start')
for r in sorted([x for x in d['results'] if x['model'].startswith(('RF-LARGE', 'K6-CEIL'))],
                key=lambda x: (x['mode'], x['members'], x['model'])):
    st = r['stages']

    def g(k):
        return '%.4f' % (st[k]['median_ns'] / 1e9) if k in st else '-'
    tot = sum(v['median_ns'] for k, v in st.items() if k not in ('entry_checked', 'entry_plain')) / 1e9
    print('%s | %s | %d | %s | %s | %s | %.3f | %s | %.1f | %s' % (
        r['model'], r['mode'], r['members'], g('geometry'), g('factor') if 'factor' in st else g('lane_solve'),
        g('finish'), tot, g('entry_checked'), r['wall_s'], '%.2f' % r['load_before'] if r['load_before'] else '-'))
