#!/bin/bash
# K6 checkpoint C, one Rust run: a clean git archive copy of the K6 head, the mutant applied
# (none for NONE), a fresh target folder, and H's full cargo test suite.
W=<wt>
C=$W/scratch/i15/c; HEAD_COMMIT=ada18de702b47e8bcca194754815ec53e4ecd161
MID=$1; D=$W/k6-mut/rs-$MID; T=$W/k6-mut-target/rs-$MID; LOG=$C/logs/rs-$MID.log
rm -rf "$D" "$T"; mkdir -p "$D"
{
  echo "=== rs $MID: git archive $HEAD_COMMIT (core/solver, core/loads) into <wt>/k6-mut/rs-$MID; target <wt>/k6-mut-target/rs-$MID"
  git -C $W/k6 archive $HEAD_COMMIT projects/chirality-piping/core/solver projects/chirality-piping/core/loads | tar -x -C "$D"
  [ "$MID" != NONE ] && python3 $C/mutants.py apply "$MID" "$D"
  echo "=== cargo test --offline --locked -j 8 --no-fail-fast (RUSTUP_TOOLCHAIN=1.97.1 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=4)"
} > "$LOG" 2>&1
cd "$D/projects/chirality-piping/core/solver/performance_harness" || exit 9
start=$(date +%s)
RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=4 CARGO_TARGET_DIR="$T" \
  cargo test --offline --locked -j 8 --no-fail-fast >> "$LOG" 2>&1
rc=$?
echo "=== exit $rc elapsed $(( $(date +%s) - start )) s" >> "$LOG"
rm -rf "$T"
sed -i '' -e "s#$W#<wt>#g" -e 's/[[:space:]]*$//' "$LOG"
exit 0
