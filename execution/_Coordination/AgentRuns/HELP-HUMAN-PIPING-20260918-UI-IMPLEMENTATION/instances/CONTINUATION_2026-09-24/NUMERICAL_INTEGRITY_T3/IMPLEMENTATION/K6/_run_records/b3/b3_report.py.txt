"""I15's B3 report figures (read-only over the records; no runs).

Usage: python3 b3_report.py <runner dir> <records dir> <counts>
"""
import json
import os
import sys

runner_dir, record_dir, counts_path = sys.argv[1:4]
sys.path.insert(0, runner_dir)
import k6_runner as r  # noqa: E402

MIB, GIB = r.MIB, r.GIB
counts = r.read_counts(counts_path)
records = r.read_records(record_dir)
runs = [x for x in r.run_entries(records) if x.get('tier') == 'T6' and x.get('classification') != 'not_run']
if not runs:
    sys.exit('no measured T6 record')
x = runs[-1]
line = counts[x['model']]
f1b, e_adm = line['estimate_f1b_bytes_dense'], line['estimate_adm_bytes_dense']
a = x['admission']
s = x.get('summary') or {}
print('%s: %s (%s), wall %.1f s, exit %s, runner %s' % (x['run_id'], x['classification'],
      x.get('classification_detail'), x['wall_s'], x.get('exit_code'), x.get('runner_commit')))
print('counts: dofs %d, free %d; F1b 96n^2 %d B (%.3f GiB); E_adm %d B' % (
    line['dofs'], line['free_dofs'], f1b, f1b / GIB, e_adm))
print('admission: %s' % json.dumps(a, sort_keys=True))
print('host: quiet %s; load before %s after %s; memorystatus before %s after %s' % (
    x.get('quiet_host'), x.get('load_before'), x.get('load_after'), x.get('memorystatus_before'),
    x.get('memorystatus_after')))
print('summary: %s' % json.dumps(s, sort_keys=True))


def fig(label, b):
    if b is None:
        print('%-22s -' % label)
        return
    print('%-22s %14d B  %9.1f MiB  /F1b %.4f  /E_adm %.4f' % (label, b, b / MIB, b / f1b, b / e_adm))


objects = r.jsonl(open(os.path.join(record_dir, x['run_id'] + '.jsonl')))
stage_heap = max([o['heap_peak'] for o in objects if o.get('kind') == 'stage'] or [0]) or None
stage_move = max([o['heap_peak_move'] for o in objects if o.get('kind') == 'stage'] or [0]) or None
fig('heap peak (in place)', s.get('repeats_heap_peak') or stage_heap)
fig('heap peak (move)', s.get('repeats_heap_peak_move') or stage_move)
fig('footprint', r.peak_footprint(x))
fig('RSS', x.get('peak_rss_bytes'))
print('RSS source: %s; RSS/footprint %s' % (x.get('peak_rss_source'),
      '%.3f' % r.rss_to_footprint(x) if r.rss_to_footprint(x) else '-'))
if a.get('projected_rss_bytes'):
    print('RSS / projected RSS: %.3f (projected %.2f GiB, limit 0.8C %.2f GiB); footprint / footprint estimate %s'
          % (x['peak_rss_bytes'] / a['projected_rss_bytes'], a['projected_rss_bytes'] / GIB,
             a['projected_rss_limit_bytes'] / GIB,
             '%.3f' % (r.peak_footprint(x) / a['footprint_estimate_bytes']) if r.peak_footprint(x) else '-'))
base = [b for b in (os.path.join(record_dir, 'baseline_T6.record.json'),) if os.path.exists(b)]
if base:
    bb = json.load(open(base[0]))
    bfp = bb.get('rss', {}).get('time_peak_footprint_bytes')
    print('rho_fp %.3f, rho_rss %.3f (net of the T6 baseline: RSS %s B, footprint %s B)' % (
        r.measured_footprint_ratio(x, bfp) or 0, r.measured_ratio(x, bb.get('peak_rss_bytes')) or 0,
        bb.get('peak_rss_bytes'), bfp))

print('\nstage lines (in order):')
for o in objects:
    if o.get('kind') == 'stage':
        print('  r%d %-14s ok=%-5s %10.3f s  heap_peak %8.1f MiB  move %8.1f MiB  begin %8.1f  end %8.1f%s' % (
            o['repeat'], o['stage'], o['ok'], o['elapsed_ns'] / 1e9, o['heap_peak'] / MIB,
            o['heap_peak_move'] / MIB, o['heap_current_begin'] / MIB, o['heap_current_end'] / MIB,
            ('  error: ' + o['error']) if o.get('error') else ''))
    elif o.get('kind') == 'stage_begin':
        pass
    elif o.get('kind') in ('outcome', 'parity'):
        print('  %s' % json.dumps(o, sort_keys=True)[:300])
begun = [(o['repeat'], o['stage']) for o in objects if o.get('kind') == 'stage_begin']
done = [(o['repeat'], o['stage']) for o in objects if o.get('kind') == 'stage']
total = sum(o['elapsed_ns'] for o in objects if o.get('kind') == 'stage') / 1e9
for b in begun:
    if b not in done:
        print('  r%d %-14s began, in flight at the end: about %.1f s (wall - completed stages %.1f s)' % (
            b[0], b[1], x['wall_s'] - total, total))
print('observed class: %s' % r.observed_class(objects))
print('\nstages aggregated: %s' % json.dumps(x.get('stages'), sort_keys=True))
print('stderr tail: %r' % (x.get('stderr_tail'),))
