#!/usr/bin/env python3
"""RV11 fixes (I10): per mutant, the killing tests now against checkpoint C's (kill_sites.txt,
kill_sites_batch2.txt), batch 3's (kill_sites_batch3.txt) and, for RV11's own mutants, RV11's
review (its MUTANTS.txt 'killed_by' field). Compared by test name, since the checkpoint-D and RV11
fixes moved lines in k2b_tests.rs and s11_site_table.rs. A mutant that was killed and is no longer
killed, or that loses a killing test, is listed. Usage: compare_kill_tests.py <new-kill-sites>
<old-kill-sites>... --rv11 <rv11-MUTANTS.txt>"""
import re, sys
args = sys.argv[1:]
rv11 = args[args.index("--rv11") + 1]
files = [a for a in args[: args.index("--rv11")]]
def parse(path):
    out, cur = {}, None
    for line in open(path):
        m = re.match(r"## (\S+)", line)
        if m:
            cur = m.group(1); out.setdefault(cur, set()); continue
        m = re.match(r"   (?:solver/)?(\S+): (\S+)$", line)
        if m and cur and "BUILD ERROR" not in line:
            out[cur].add(m.group(2))
    return out
new = parse(files[0])
old = {}
for f in files[1:]:
    for k, v in parse(f).items():
        old.setdefault(k, set()).update(v)
for line in open(rv11):
    m = re.match(r"(RV11-\S+) exit:.* killed_by: (.*?)\s+panic_sites", line)
    if m:
        old[m.group(1)] = set() if m.group(2).strip() == "NONE" else set(m.group(2).split())
for mid in new:
    if mid == "NONE":
        continue
    n, o = new[mid], old.get(mid)
    if o is None:
        print(f"{mid}: new mutant; killed by {len(n)} test(s)" + ("" if n else " -- NOT KILLED"))
        continue
    lost, added = sorted(o - n), sorted(n - o)
    state = "KILLED" if n else "NOT KILLED"
    note = " (survived before)" if not o else ""
    print(f"{mid}: before {len(o)} test(s){note}, now {len(n)}; lost {len(lost)}, added {len(added)}; {state}")
    for t in lost:
        print(f"    lost: {t}")
    for t in added:
        print(f"    added: {t}")
