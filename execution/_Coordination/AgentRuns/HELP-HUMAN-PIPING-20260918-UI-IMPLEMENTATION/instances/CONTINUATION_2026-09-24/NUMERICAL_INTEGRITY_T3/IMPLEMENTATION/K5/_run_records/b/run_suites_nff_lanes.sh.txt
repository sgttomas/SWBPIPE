#!/bin/bash
# I14 (K5, checkpoint B): CI's numerical cargo profile, as <wt>/scratch/calib/run_suites_nff.sh
# (every discovered manifest; cargo test --offline --locked --no-fail-fast; -j 8; RUST_TEST_THREADS=4),
# split into two lanes (two cargo jobs), each with its own target. Same per-manifest log and summary line.
# Usage: run_suites_nff_lanes.sh <tree> <out> <target-prefix>
M=$1; OUT=$2; TP=$3; mkdir -p $OUT
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_BUILD_JOBS=8 RUST_TEST_THREADS=4
cd $M/projects/chirality-piping
python3 - <<'PY' > $OUT/manifests.txt
import importlib.util, sys
from pathlib import Path
spec = importlib.util.spec_from_file_location('r', 'tools/release/check_release_readiness.py'); m = importlib.util.module_from_spec(spec); sys.modules['r'] = m; spec.loader.exec_module(m)
for p in m.discover_cargo_manifests(Path('.')): print(p)
PY
for mf in $(cat $OUT/manifests.txt); do cargo fetch --locked --manifest-path $mf > /dev/null 2>&1 || echo "FETCH-FAIL $mf"; done
lane() {
  local L=$1; local i=0
  export CARGO_TARGET_DIR=$TP-$L
  for mf in $(cat $OUT/manifests.txt); do
    if [ $((i % 2)) -eq $L ]; then
      log=$OUT/$(printf '%03d' $i)_$(echo $mf | tr '/' '_').log
      cargo test --offline --locked --no-fail-fast --manifest-path $mf > $log 2>&1; rc=$?
      pass=$(/usr/bin/grep -E '^test result:' $log | awk '{s+=$4} END {print s+0}'); fail=$(/usr/bin/grep -E '^test result:' $log | awk '{s+=$6} END {print s+0}'); ign=$(/usr/bin/grep -E '^test result:' $log | awk '{s+=$8} END {print s+0}')
      echo "$(printf '%03d' $i) $rc passed=$pass failed=$fail ignored=$ign $mf" >> $OUT/lane$L.txt
    fi
    i=$((i+1))
  done
}
: > $OUT/lane0.txt; : > $OUT/lane1.txt
lane 0 & lane 1 & wait
cat $OUT/lane0.txt $OUT/lane1.txt | sort | cut -d' ' -f2- > $OUT/suites.log
echo SUITES-DONE >> $OUT/suites.log
echo SUITES-DONE
