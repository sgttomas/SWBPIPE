// Scratch (not committed): time the binary64 gate's RF-MECH parity per case and mode.
use piping_numerical_robustness::cases::load_family;
use piping_numerical_robustness::parity::parity;
use std::time::Instant;
fn main() {
    for c in load_family("RF-MECH") {
        let m = c.model.as_ref().unwrap();
        let t = Instant::now();
        let p = parity(m, false).unwrap();
        let ts = t.elapsed().as_secs_f64();
        let (td, dense) = if m.members.len() <= 105 {
            let t = Instant::now();
            let p = parity(m, true).unwrap();
            (t.elapsed().as_secs_f64() - ts, p.dense)
        } else {
            (f64::NAN, None)
        };
        println!("{} members {}: sparse-only {:.2}s ({}), with dense extra {:.2}s ({:?})", c.id, m.members.len(), ts, p.sparse, td, dense);
    }
}
