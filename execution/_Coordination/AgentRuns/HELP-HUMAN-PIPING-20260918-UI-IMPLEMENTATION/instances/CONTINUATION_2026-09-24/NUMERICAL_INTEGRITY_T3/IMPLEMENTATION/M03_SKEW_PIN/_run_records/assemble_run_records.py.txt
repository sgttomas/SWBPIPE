"""I9: build T3/IMPLEMENTATION/M03_SKEW_PIN/_run_records/ from I9's scratch directory, with machine
paths replaced by placeholders. Logs: trailing spaces and trailing blank lines stripped; a line over
600 characters is cut, with a marker giving the characters removed and the sha256 of the full line.
The probe logs and product_runs.jsonl are data and are not cut. As recorded, this file's own path constants
appear sanitized. Usage: python3 -B assemble_run_records.py <dest>"""
import hashlib, os, re, shutil, sys
WT = "<wt>"
SRC = WT + "/scratch/i9"
DEST = sys.argv[1]
SUBS = [
    (WT + "/scratch/i9", "<wt>/scratch/i9"),
    (WT, "<wt>"),
    ("<VENV>", "<VENV>"),
    ("<home>", "<home>"),
]
# The session scratchpad (built from parts so that this file names no machine path itself).
SCRATCHPAD = re.compile("/" + "priv" + r"ate/tmp/claude-\d+/[^\s\"']*")
def sanitize(text):
    for a, b in SUBS:
        text = text.replace(a, b)
    return SCRATCHPAD.sub("<scratch>", text)
def cut(line):
    if len(line) <= 600:
        return line
    digest = hashlib.sha256(line.encode()).hexdigest()
    return line[:600] + f" [... {len(line) - 600} characters cut; sha256 of the full line {digest}]"
def put(src, dst, log=False):
    text = sanitize(open(os.path.join(SRC, src), errors="strict").read())
    lines = [l.rstrip() for l in text.split("\n")]
    if log:
        lines = [cut(l) for l in lines]
    while lines and lines[-1] == "":
        lines.pop()
    path = os.path.join(DEST, dst)
    os.makedirs(os.path.dirname(path), exist_ok=True)
    open(path, "w").write("\n".join(lines) + "\n")
if os.path.exists(DEST):
    shutil.rmtree(DEST)
put("assemble_run_records.py", "assemble_run_records.py.txt")
put("logs/toolchain.txt", "toolchain.txt")
# kernel pin
put("logs/fk_m03_final.log", "fk_m03_skew_scope.log", log=True)
put("logs/ni_i9_final.log", "ni_i9_adapter.log", log=True)
put("exact_reference.py", "exact_reference.py.txt")
put("exact_reference.stdout.txt", "exact_reference.stdout.txt")
put("counterfactual_diagnosis.py", "development/counterfactual_diagnosis.py.txt")
put("counterfactual_diagnosis.stdout.txt", "development/counterfactual_diagnosis.stdout.txt")
put("logs/fk_m03_1.log", "development/fk_m03_run1_counterfactual_failed.log", log=True)
put("logs/fk_m03_2.log", "development/fk_m03_run2.log", log=True)
put("logs/ni_i9_1.log", "development/ni_i9_run1.log", log=True)
put("logs/fk_m03_worktree_final.log", "development/fk_m03_worktree_final.log", log=True)
put("logs/ni_i9_worktree_final.log", "development/ni_i9_worktree_final.log", log=True)
# product evidence
put("i9_skew_product_probe.rs", "product/i9_skew_product_probe.rs.txt")
put("run_probe.sh", "product/run_probe.sh.txt")
for t in ("pe134", "pe_eb5"):
    put(f"logs/probe_{t}.log", f"product/probe_{t}.log")
    put(f"logs/probe_{t}.time", f"product/probe_{t}.time.txt")
put("product_reference.py", "product/product_reference.py.txt")
put("product_reference.stdout.txt", "product/product_reference.stdout.txt")
put("product_runs.jsonl", "product/product_runs.jsonl")
put("product_table.txt", "product/product_table.txt")
# mutations
put("mut/mutate_i9.py", "mutations/mutate_i9.py.txt")
put("mut/run_mutant.sh", "mutations/run_mutant.sh.txt")
put("mut/kill_sites.py", "mutations/kill_sites.py.txt")
put("mut/kill_sites.txt", "mutations/kill_sites.txt")
for f in sorted(os.listdir(SRC + "/mut/logs")):
    put("mut/logs/" + f, "mutations/logs/" + f, log=True)
# suites
put("suites/run_suites_i9.sh", "suites/run_suites_i9.sh.txt")
put("suites/compare_suites_i9.py", "suites/compare_suites_i9.py.txt")
put("suites/suites_compare.txt", "suites/suites_compare.txt")
for t in ("base", "cand"):
    for f in ("frame_kernel.log", "sparse_direct.log", "nonlinear_integration.log", "summary.txt"):
        put(f"suites/{t}/{f}", f"suites/{t}/{f}", log=f.endswith(".log"))
# T9
put("t9/run_t9.sh", "t9/run_t9.sh.txt")
for t in ("base", "cand"):
    put(f"t9/h_{t}/Cargo.toml", f"t9/harness_{t}_Cargo.toml.txt")
    for f in (f"build_{t}.log", f"run_{t}.log"):
        put("t9/" + f, "t9/" + f, log=True)
    put(f"t9/output_sha256_{t}.txt", f"t9/output_sha256_{t}.txt")
put("t9/t9_summary.txt", "t9/t9_summary.txt")
put("trees.txt", "trees.txt")
put("README.txt", "README.txt")
print("assembled", sum(len(fs) for _, _, fs in os.walk(DEST)), "files")
