#!/usr/bin/env python3
"""K2b checkpoint B: the failure blocks ('---- <test> stdout ----' to the next
block or 'failures:') of the listed manifests, with thread ids removed, and
their sha256 in the baseline and candidate logs."""
import hashlib, os, re, sys
base_dir, cand_dir = sys.argv[1], sys.argv[2]
def blocks(path):
    text = open(path, errors='replace').read()
    out = {}
    for m in re.finditer(r'^---- (.+?) stdout ----\n(.*?)(?=^---- |^failures:\s*$)', text, re.S | re.M):
        body = re.sub(r"thread '([^']*)' \(\d+\)", r"thread '\1'", m.group(2))
        out[m.group(1)] = body
    return out
def log(d, mf):
    manifests = [l.strip() for l in open(os.path.join(d, 'manifests.txt')) if l.strip()]
    i = manifests.index(mf)
    return os.path.join(d, f"{i:03d}_{mf.replace('/', '_')}.log")
for mf in ['core/product_physics/Cargo.toml', 'core/runner/headless/Cargo.toml']:
    b, c = blocks(log(base_dir, mf)), blocks(log(cand_dir, mf))
    for name in sorted(set(b) | set(c)):
        hb = hashlib.sha256(b.get(name, '').encode()).hexdigest()
        hc = hashlib.sha256(c.get(name, '').encode()).hexdigest()
        print(f"{mf} {name} bytes={len(c.get(name, ''))} base={hb[:16]} cand={hc[:16]} {'IDENTICAL' if hb == hc else 'DIFFER'}")
