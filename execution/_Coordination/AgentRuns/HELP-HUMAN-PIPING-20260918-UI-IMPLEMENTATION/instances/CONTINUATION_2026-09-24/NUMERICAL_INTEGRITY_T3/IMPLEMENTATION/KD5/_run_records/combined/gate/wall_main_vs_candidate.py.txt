"""K-D5 (I3R): wall time per run, main b24b3d536 probe against the combined candidate probe, over the
836 runs under 1000 members that both runs files hold (fresh process per run; includes process start
and JSON I/O). Split by whether the candidate published checks_passed (the check runs only there).
Usage: python3 -B wall_main_vs_candidate.py <compare_dir> <main.jsonl> <cand.jsonl>"""
import json, sys, statistics as st
sys.path.insert(0, sys.argv[1])
import compare as C

def load(path):
    out = {}
    for line in open(path):
        r = json.loads(line)
        run = r.get('probe', {}).get('run', {})
        q = C.identity(run['envelope']).get('numerical_quality_status') if run.get('ok') else None
        out[(r['case'], r['mode'], r['entry'])] = (r['wall_seconds'], q)
    return out

m, c = load(sys.argv[2]), load(sys.argv[3])
keys = sorted(set(m) & set(c))
for label, sel in (('candidate checks_passed (check ran)', lambda k: c[k][1] == 'checks_passed'),
                   ('all other runs', lambda k: c[k][1] != 'checks_passed')):
    ks = [k for k in keys if sel(k)]
    d = [c[k][0] - m[k][0] for k in ks]
    print(f"{label}: n={len(ks)} main total {sum(m[k][0] for k in ks):.2f} s, candidate total {sum(c[k][0] for k in ks):.2f} s, "
          f"per-run difference median {st.median(d):+.4f} s, max {max(d):+.4f} s, min {min(d):+.4f} s")
top = sorted(keys, key=lambda k: c[k][0] - m[k][0], reverse=True)[:5]
print('largest per-run increases:', [(k, round(m[k][0], 4), round(c[k][0], 4)) for k in top])
