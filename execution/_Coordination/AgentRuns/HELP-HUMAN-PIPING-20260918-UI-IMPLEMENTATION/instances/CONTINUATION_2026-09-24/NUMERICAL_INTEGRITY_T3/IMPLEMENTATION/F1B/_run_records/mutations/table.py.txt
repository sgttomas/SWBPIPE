#!/usr/bin/env python3
"""F1b checkpoint C (I13): the mutation table from MUTANTS.txt and logs/. A mutant is killed when a
test fails that does not fail in the NONE control (a behavioural or pin assertion), with no compile
error; an allocator abort or stack overflow is never counted. For each F1b mutant the intended killer
is checked. Usage: table.py <c-dir>"""

    "F1B-FOLD": ["rule_8_every_accumulation_in_the_listed_files_is_in_the_table"],
}
rows = {}
for line in open(os.path.join(C, "MUTANTS.txt")):
    m = re.match(r"(\S+) exit:(.*?) compile_errors=(\d+) aborts=(\d+) failed: (.*?) panic_sites: (.*)$", line.strip())
    if m:
        rows[m.group(1)] = dict(exit=m.group(2).strip(), comp=int(m.group(3)), aborts=int(m.group(4)),
                                failed=set(m.group(5).split()) - {"NONE"}, sites=m.group(6).split())
    elif "PATCH-FAILED" in line:
        rows[line.split()[0]] = dict(patch_failed=True)
base = rows.get("NONE", {}).get("failed", set())
print(f"NONE: failed {sorted(base)} exit {rows.get('NONE', {}).get('exit')}")
survivors = []
for mid, r in rows.items():
    if mid == "NONE":
        continue
    if r.get("patch_failed"):
        print(f"{mid}: PATCH-FAILED")
        survivors.append(mid)
        continue
    kills = sorted(r["failed"] - base)
    short = [k.split("::")[-1] for k in kills]
    status = "KILLED" if kills and r["comp"] == 0 else ("COMPILE-ERROR" if r["comp"] else "SURVIVED")
    intended = INTENDED.get(mid)
    hit = [t for t in (intended or []) if t in short]
    note = ""
    if intended is not None:
        note = f" intended {'hit: ' + ', '.join(hit) if hit else 'NOT among the failures: ' + ', '.join(intended)}"
    if r["aborts"]:
        note += f" aborts={r['aborts']} (not counted)"
    print(f"{mid}: {status} by {len(kills)} test(s){note}\n    {' '.join(short)}")
    if status != "KILLED":
        survivors.append(mid)
print("survivors:", survivors or "none")
