"""F1b records (I13, checkpoint D): copy the run evidence into _run_records/, sanitized, and generate
the indexes of the large gate files that stay uncommitted.

- Machine paths become placeholders, longest first: <VENV> (the DEC-025 venv), <wt> (the T3 worktrees
  root: scratch <wt>/scratch/i13, gate targets <wt>/gate-*), {REPO_ROOT} (the outer worktree), <scratch>
  (the session scratchpad) and <home>. That substitution is the only change made to a copied file: raw
  logs keep their bytes otherwise, including any trailing whitespace (listed in trailing_whitespace.txt).
- Scripts are stored with a .txt suffix. Nothing under a build target, tree or full-envelope folder is
  copied; the large files are indexed (envelope_sha256.tsv, summaries) and hashed in
  uncommitted_sha256.txt.
Usage: assemble.py <wt> <outer-worktree> <venv> <session-scratchpad> <out-dir> <G1-full-dir>"""

PLACES = sorted([(venv, "<VENV>"), (wt, "<wt>"), (outer, "{REPO_ROOT}"), (pad, "<scratch>"),
                 (alt_pad, "<scratch>"), (home, "<home>")], key=lambda p: -len(p[0]))
MACHINE = re.compile(r"(?<![A-Za-z0-9_.{-])(?:/(?:Users|private|home|tmp|var/folders)/[^\s)\"'`<>]+)")
S = os.path.join(wt, "scratch", "i13")
count = 0
trailing = []


def clean(text):
    for path, name in PLACES:
        text = text.replace(path, name)
    return text


def put_text(text, rel):
    global count
    dst = os.path.join(out, rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    text = clean(text)
    with open(dst, "w", encoding="utf-8") as f:
        f.write(text)
    n = sum(1 for line in text.split("\n") if line != line.rstrip())
    if n:
        trailing.append((rel, n))
    count += 1


def put(src, rel):
    with open(src, encoding="utf-8", errors="replace") as f:
        put_text(f.read(), rel)


def sha_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for block in iter(lambda: f.read(1 << 20), b""):
            h.update(block)
    return h.hexdigest()


def tree(src_dir, rel_dir, skip=()):
    for name in sorted(os.listdir(src_dir)):
        p = os.path.join(src_dir, name)
        if name in skip or name == "__pycache__":
            continue
        if os.path.isdir(p):
            tree(p, os.path.join(rel_dir, name), skip)
        else:
            suffix = ".txt" if name.endswith((".sh", ".py", ".rs")) else ""
            put(p, os.path.join(rel_dir, name + suffix))

def envelope_index(d, rel):
    rows = ["case\tmode\tentry\toutcome\tok\texit_code\tsummary_envelope_sha256\tfull_envelope_sha256\t"
            "full_envelope_bytes\terror_text_sha256\tstdout_sha256"]
    for line in open(os.path.join(d, "runs.jsonl")):
        rec = json.loads(line)
        key = "%s__%s__%s" % (rec["case"], rec["mode"], rec["entry"])
        outcome, _ = CMP.classify(rec)
        run = (rec.get("probe") or {}).get("run", {})
        env = os.path.join(d, "envelopes", key + ".json")
        full = os.path.join(d, "full", key + ".json")
        std = os.path.join(d, "stdout", key + ".out")
        rows.append("\t".join([
            rec["case"], rec["mode"], rec["entry"], outcome, "ok" if run.get("ok") else "ERR",
            str(rec.get("exit_code")), sha_file(env) if os.path.exists(env) else "-",
            run.get("envelope_sha256", "-"), str(os.path.getsize(full)) if os.path.exists(full) else "-",
            hashlib.sha256(run["error"].encode()).hexdigest() if "error" in run else "-",
            sha_file(std) if os.path.exists(std) else "-"]))
    put_text("\n".join(rows) + "\n", rel)


def rv11d_summary(d, rel, note):
    lines = [note, "case\tmode\tentry\texit_code\toutcome\tfull_envelope_sha256\t"
                   "rv11d_n2 rows by (kind, status)\tother probe rows"]
    for line in open(os.path.join(d, "runs.jsonl")):
        rec = json.loads(line)
        key = "%s__%s__%s" % (rec["case"], rec["mode"], rec["entry"])
        outcome, _ = CMP.classify(rec)
        obs = os.path.join(d, "full", key + ".json.obs.jsonl")
        counts, other = {}, {}
        if os.path.exists(obs):
            for o in open(obs):
                r = json.loads(o)
                if r.get("probe") == "rv11d_n2":
                    k = "%s:%s" % (r["kind"], r["status"])
                    counts[k] = counts.get(k, 0) + 1
                else:
                    other[r.get("probe")] = other.get(r.get("probe"), 0) + 1
        run = (rec.get("probe") or {}).get("run", {})
        lines.append("\t".join([rec["case"], rec["mode"], rec["entry"], str(rec.get("exit_code")), outcome,
                                run.get("envelope_sha256", "-"),
                                json.dumps(counts, sort_keys=True) if os.path.exists(obs) else "no observation log",
                                json.dumps(other, sort_keys=True)]))
    put_text("\n".join(lines) + "\n", rel)


def c_tables(compare_json, rel_dir):
    t = json.load(open(compare_json))
    cols = ["case", "mode", "entry", "base.outcome", "base.exit", "base.blocking", "base.standing",
            "cand.outcome", "cand.exit", "cand.blocking", "cand.standing", "cand.full16", "cand.guard",
            "cand.result_count", "cand.mechanics", "cand.nq_status", "cand.solve_quality", "cand.lane_not_run",
            "cand.peak_rss_bytes"]
    for name in ["c1_table", "c2_table"]:
        rows = ["\t".join(cols)]
        for r in t[name]:
            b, c = r["base"], r["cand"]
            rows.append("\t".join(str(x) for x in [
                r["case"], r["mode"], r["entry"], b["outcome"], b["exit"], ",".join(b["blocking"]) or "-",
                b["standing"], c["outcome"], c["exit"], ",".join(c["blocking"]) or "-", c["standing"], c["full"],
                c.get("guard"), c.get("result_count"), c.get("mechanics", "-"), c.get("nq_status", "-"),
                ",".join(c.get("solve_quality") or []) or "-", c.get("lane_not_run", "-"), c.get("peak_rss_kib")]))
        put_text("\n".join(rows) + "\n", os.path.join(rel_dir, name + ".tsv"))
