"""I13 (F1b) gate part 1: candidate c4879c496 (F1b after K5's merge) against G1's full-envelope Mac base (e7d930d49).
A copy of the 130445db2 gate's script, with this line changed; nothing else.
Copy of the 948e0bb99 gate's compare_gate_f1b.py; the classes, checks and verdict are unchanged. Added
for C2's table only: per candidate run, whether the envelope carries SPARSE_OBSERVATION_LANE_NOT_RUN
(lane_not_run), the numerical_quality status (nq_status), the per-case solve_quality values and the
mechanics status.

Derived from I14's (K5) compare_gate_part1.py. index(dir): one row per run key (case, mode, entry)
from <dir>/runs.jsonl: outcome (compare.classify, unchanged), ok, exit code, run.envelope_sha256
(the full envelope), the summary span's sha256, the error text's sha256, the stderr tail's sha256.
The index is first rebuilt from G1's runs.jsonl and checked against G1's envelope_sha256.tsv.

Classes (the brief's Gates, ROOT's rulings on F1b A2):
- C1: the ruled 28-run list (RF-RANGE); a C1 run may differ.
- C2: the 24 runs at 10,000 members; they may differ. A sparse C2 run that aborts at the heap cap
  is a finding and fails PASS; a dense C2 run must be the guard's refusal (no row published).
- C3: every other run: outcome, ok, exit code, full-envelope sha256, summary sha256, error text,
  stderr tail of a heap-cap abort, and the gate_check row (outcome, quality, standing, trusted,
  breaches) identical to the base.
PASS: 0 trusted breaches (gate_check), C3 identical, the runs that differ outside C2 are within C1,
no sparse C2 heap-cap abort, and every dense C2 run is the guard's refusal.
Usage: python3 -B compare_gate_f1b.py <G1-dir> <cand-dir> <G1-scripts> <G1-result.json> <cand-result.json> <out.json>
"""
import hashlib
import json
import os
import sys

g1, cand, scripts, rb_path, rc_path, out_path = sys.argv[1:7]
sys.path.insert(0, scripts)
import compare as C  # noqa: E402  (G1's copy, unchanged)

C1 = {(c, m, e) for c in ('RF-RANGE-CHAIN-E-1000', 'RF-RANGE-CONT-E-1000', 'RF-RANGE-SKEW-E-1000',
                          'RF-RANGE-SKEW-F-960', 'RF-RANGE-THIN-B')
      for m in ('sparse_interactive', 'dense_scrutiny') for e in ('captured', 'typed')}
C1 |= {('RF-RANGE-SKEW-E+960', m, 'typed') for m in ('sparse_interactive', 'dense_scrutiny')}
C1 |= {('RF-RANGE-%s-LEF-large' % s, m, 'typed') for s in ('CHAIN', 'SKEW', 'CONT')
       for m in ('sparse_interactive', 'dense_scrutiny')}
assert len(C1) == 28


def sha(b):
    return hashlib.sha256(b).hexdigest()


def index(d):
    out = {}
    for line in open(os.path.join(d, 'runs.jsonl')):
        rec = json.loads(line)
        key = (rec['case'], rec['mode'], rec['entry'])
        outcome, detail = C.classify(rec)
        run = (rec.get('probe') or {}).get('run', {})
        env_path = os.path.join(d, 'envelopes', '%s__%s__%s.json' % key)
        env = run.get('envelope') or {}
        out[key] = dict(outcome=outcome, ok='ok' if run.get('ok') else 'ERR', exit=str(rec.get('exit_code')),
                        envelope=sha(open(env_path, 'rb').read()) if os.path.exists(env_path) else '-',
                        error=sha(run['error'].encode()) if 'error' in run else '-',
                        full=run.get('envelope_sha256', '-'),
                        stderr_tail=sha(rec.get('stderr_tail', '').encode()),
                        detail=detail[:200], timed_out=rec.get('timed_out'),
                        mechanics=(env.get('status') or {}).get('mechanics'),
                        blocking=sorted({x['code'] for x in env.get('diagnostics', []) if x['severity'] == 'blocking'}),
                        guard=any(x['id'] == 'diagnostic:physics:dense-scrutiny-resource-guard' for x in env.get('diagnostics', [])),
                        result_count=env.get('result_count'), wall=rec.get('wall_seconds'),
                        lane_not_run=any(x['code'] == 'SPARSE_OBSERVATION_LANE_NOT_RUN' for x in env.get('diagnostics', [])),
                        nq_status=(env.get('numerical_quality') or {}).get('status'),
                        solve_quality=sorted({x.get('solve_quality') for x in (env.get('numerical_quality') or {}).get('cases', [])} - {None}),
                        peak_rss_kib=rec.get('peak_rss_kib'), stderr=rec.get('stderr_tail', '')[-200:])
    return out


def tsv(d):
    out = {}
    lines = open(os.path.join(d, 'envelope_sha256.tsv')).read().splitlines()
    head = lines[0].split('\t')
    names = dict(outcome=['outcome'], ok=['ok'], exit=['exit_code'],
                 envelope=['envelope_sha256', 'summary_envelope_sha256'], error=['error_text_sha256'],
                 full=['full_envelope_sha256'])
    for line in lines[1:]:
        f = dict(zip(head, line.split('\t')))
        out[(f['case'], f['mode'], f['entry'])] = {k: f[v] for k, vs in names.items() for v in vs if v in f}
    return out


base, listed = index(g1), tsv(g1)
assert base.keys() == listed.keys()
shared = [c for c in ('outcome', 'ok', 'exit', 'envelope', 'error', 'full') if c in next(iter(listed.values()))]
for k in base:
    for col in shared:
        assert base[k][col] == listed[k][col], ('formula check', k, col)
new = index(cand)
cases = {c['id']: c for c in json.load(open(os.path.join(os.path.dirname(scripts), 'gen_out', 'cases.json')))}
C2 = {k for k in base if cases[k[0]].get('members', 0) >= 10000}
assert len(C2) == 24
rows = lambda p: {(r['case'], r['mode'], r['entry']): r for r in json.load(open(p))['runs']}
rb, rc = rows(rb_path), rows(rc_path)
result_c = json.load(open(rc_path))
COLS = ('outcome', 'ok', 'exit', 'full', 'envelope', 'error')
changed, c3_diffs = [], []
for k in sorted(set(base) | set(new)):
    if k not in base or k not in new:
        c3_diffs.append((k, 'missing'))
        continue
    b, c = base[k], new[k]
    d = [col for col in COLS if b[col] != c[col]]
    if b['outcome'] == 'memory_refused' and c['outcome'] == 'memory_refused' and b['stderr_tail'] != c['stderr_tail']:
        d.append('stderr_tail')
    g = [col for col in ('outcome', 'quality', 'standing', 'trusted', 'breaches')
         if (k in rb or k in rc) and (rb.get(k) or {}).get(col) != (rc.get(k) or {}).get(col)]
    if d or g:
        changed.append((k, d, g))
        if k not in C1 and k not in C2:
            c3_diffs.append((k, d + ['gate_check.' + x for x in g]))


def row(k):
    b, c = base[k], new[k]
    return dict(case=k[0], mode=k[1], entry=k[2],
                base=dict(outcome=b['outcome'], exit=b['exit'], blocking=b['blocking'], standing=(rb.get(k) or {}).get('standing'),
                          full=b['full'][:16], detail=b['detail'][:120]),
                cand=dict(outcome=c['outcome'], exit=c['exit'], blocking=c['blocking'], standing=(rc.get(k) or {}).get('standing'),
                          full=c['full'][:16], detail=c['detail'][:120], guard=c['guard'], result_count=c['result_count'],
                          wall=c['wall'], peak_rss_kib=c['peak_rss_kib'], stderr=c['stderr'][-120:] if c['outcome'] == 'memory_refused' else '',
                          mechanics=c['mechanics'], lane_not_run=c['lane_not_run'], nq_status=c['nq_status'],
                          solve_quality=c['solve_quality']))


c1_table = [row(k) for k in sorted(C1)]
c2_table = [row(k) for k in sorted(C2)]
c1_changed = sorted(k for k, _, _ in changed if k in C1)
outside = sorted(k for k, _, _ in changed if k not in C1 and k not in C2)
sparse_aborts = sorted(k for k in C2 if k[1] == 'sparse_interactive' and new[k]['outcome'] == 'memory_refused')
dense_not_guard = sorted(k for k in C2 if k[1] == 'dense_scrutiny'
                         and not (new[k]['guard'] and new[k]['outcome'] == 'refused_blocked' and not new[k]['result_count']))
c3_keys = [k for k in base if k not in C1 and k not in C2]
verdict = dict(
    trusted_breach_triples=result_c['trusted_breach_triples'], runs_evaluated=result_c['runs_evaluated'],
    trusted_runs=result_c['trusted_runs'], violations=result_c.get('violations_outside_pinned_exceptions'),
    c3_runs=len(c3_keys), c3_differences=len(c3_diffs), c1_changed=len(c1_changed), changed_outside_c1_c2=len(outside),
    c2_sparse_heap_cap_aborts=len(sparse_aborts), c2_dense_not_guard=len(dense_not_guard))
passed = (result_c['trusted_breach_triples'] == 0 and not c3_diffs and not outside and not sparse_aborts
          and not dense_not_guard)
verdict['RESULT'] = 'PASS' if passed else 'FAIL'
json.dump(dict(verdict=verdict, c3_differences=[list(map(str, x)) for x in c3_diffs], c1_changed=c1_changed,
               changed_outside=outside, c2_sparse_aborts=sparse_aborts, c2_dense_not_guard=dense_not_guard,
               c1_table=c1_table, c2_table=c2_table), open(out_path, 'w'), indent=1)
print(json.dumps(verdict, indent=1))
