#!/bin/bash
# I9 mutant run: a fresh copy of the candidate archive tree (git archive of eb52114e9 without execution/,
# plus the two I9 test files), the patch, its own target; FK m03_skew_scope and NI i9_ tests, --no-fail-fast.
# Caps: -j 4, RUST_TEST_THREADS=4. The copy and the target are deleted afterwards.
set -u
ID=$1
WT=<wt>
S=$WT/scratch/i9
M=$WT/skewpin-mut/$ID
LOG=$S/mut/logs/$ID.log
mkdir -p $S/mut/logs
rm -rf $M && mkdir -p $M
cp -R $S/trees/cand $M/tree
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=4 CARGO_TARGET_DIR=$M/target
{
  echo "== mutant $ID"; python3 $S/mut/mutate_i9.py $M/tree $ID
  C=$M/tree/projects/chirality-piping/core/solver
  echo "== FK --test m03_skew_scope"
  cargo test --offline --locked -j 4 --manifest-path $C/frame_kernel/Cargo.toml --no-fail-fast --test m03_skew_scope 2>&1
  echo "== FK exit $?"
  echo "== NI --lib i9_"
  cargo test --offline --locked -j 4 --manifest-path $C/nonlinear_integration/Cargo.toml --no-fail-fast --lib i9_ 2>&1
  echo "== NI exit $?"
} > $LOG 2>&1
rm -rf $M
echo "$ID done"
