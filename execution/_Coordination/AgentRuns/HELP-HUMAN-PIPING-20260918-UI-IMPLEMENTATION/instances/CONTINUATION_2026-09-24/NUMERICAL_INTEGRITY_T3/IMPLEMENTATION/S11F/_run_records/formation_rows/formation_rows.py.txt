"""T3 S11-F (I4): the 14 formation rows (GATE/FORMATION_EXCEPTIONS.json), base vs candidate.
Usage: formation_rows.py <references.json> <FORMATION_EXCEPTIONS.json> <probe_out dir>
obs is the published binary64 value decoded exactly (Fraction), converted exactly from its unit;
exp and scale are R1's decimal strings; ratio = |obs - exp| / (1e-9 max(|exp|, scale)), exact."""
import json, sys
from decimal import Decimal, localcontext
from fractions import Fraction as Fr
refs = json.load(open(sys.argv[1]))['cases']
rows = json.load(open(sys.argv[2]))['rows']
out_dir = sys.argv[3]
UNIT = {'mm': Fr(1, 1000), 'm': Fr(1), 'rad': Fr(1), 'N': Fr(1), 'N*m': Fr(1)}
def sqrt_fr(q):
    with localcontext() as c:
        c.prec = 60
        return Fr((Decimal(q.numerator) / Decimal(q.denominator)).sqrt())
def observe(env, key):
    p = key.split('.')
    res = [r for r in env['results'] if r.get('basis_ref', {}).get('ref_id') == 'case']
    if p[0] in ('u', 'th'):
        r = next(r for r in res if r['id'] == 'result:disp:%s:%s' % (p[1], p[2].lower()))
        return Fr(r['value']) * UNIT[r['unit']], repr(r['value'])
    if p[0] == 'Mb':
        loc = {'i': 'end_i', 'j': 'end_j'}[p[2]]
        comp = lambda c: next(r for r in res if r['kind'].startswith('element_local_') and r['entity_ref'] == p[1] and r['metadata']['component'] == c and r['metadata']['location'] == loc)
        y, z = comp('bending_moment_y')['value'], comp('bending_moment_z')['value']
        return sqrt_fr(Fr(y) ** 2 + Fr(z) ** 2), 'hypot(%r, %r)' % (y, z)
    raise ValueError(key)
table = []
for entry, case, key, mode in rows:
    row = next(r for r in refs[case]['expected'] if r[0] == key)
    exp, scale = Fr(Decimal(row[1])), Fr(Decimal(row[3]))
    rec = {'entry': entry, 'case': case, 'key': key, 'mode': mode, 'exp': row[1]}
    for side in ('base', 'cand'):
        env = json.load(open('%s/%s.%s.%s.%s.json' % (out_dir, side, case, mode, entry)))
        obs, text = observe(env, key)
        ratio = abs(obs - exp) / (Fr(1, 10 ** 9) * max(abs(exp), scale))
        rec[side] = {'obs': text, 'ratio': '%.6g' % float(ratio)}
    rec['bit_identical'] = rec['base']['obs'] == rec['cand']['obs']
    rec['no_worse'] = float(rec['cand']['ratio']) <= float(rec['base']['ratio'])
    table.append(rec)
print(json.dumps(table, indent=1))
