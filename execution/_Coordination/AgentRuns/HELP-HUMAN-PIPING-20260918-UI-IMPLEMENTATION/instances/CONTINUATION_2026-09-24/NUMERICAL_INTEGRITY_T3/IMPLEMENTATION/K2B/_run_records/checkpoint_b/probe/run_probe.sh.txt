#!/bin/bash
set -e
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0
for side in base cand; do
  P=<wt>/scratch/i10/b/probe/p_$side; mkdir -p $P/src
  T=<wt>/scratch/i10/b/trees/$side/projects/chirality-piping/core/solver
  sed -e "s#KD5_MODELS#<wt>/scratch/i10/b/trees/cand/projects/chirality-piping/core/solver/nonlinear_integration/src/structural_adapter/kd5_models.rs#" -e "s#K2B_MODELS#<wt>/scratch/i10/b/trees/cand/projects/chirality-piping/core/solver/nonlinear_integration/src/structural_adapter/k2b_models.rs#" <wt>/scratch/i10/b/probe/src/main.rs > $P/src/main.rs
  cat > $P/Cargo.toml <<TOML
[package]
name = "k2b_b0_probe"
version = "0.0.0"
edition = "2021"
publish = false
[dependencies]
open_pipe_stress_frame_kernel = { path = "$T/frame_kernel" }
open_pipe_stress_nonlinear_integration = { path = "$T/nonlinear_integration" }
open_pipe_stress_curved_bend = { path = "$T/curved_bend" }
open_pipe_stress_sparse_direct = { path = "$T/sparse_direct" }
open_pipe_stress_nonlinear_supports = { path = "$T/nonlinear_supports" }
[workspace]
TOML
  ( cd $P && cargo generate-lockfile --offline && CARGO_TARGET_DIR=<wt>/k2b-target/probe_$side cargo build --release --offline --locked -j 8 ) > <wt>/scratch/i10/b/probe/build_$side.log 2>&1
  rm -rf <wt>/scratch/i10/b/probe/out_$side; <wt>/k2b-target/probe_$side/release/k2b_b0_probe <wt>/scratch/i10/b/probe/out_$side > <wt>/scratch/i10/b/probe/run_$side.log 2>&1
  ( cd <wt>/scratch/i10/b/probe/out_$side && ls | sort | xargs shasum -a 256 ) > <wt>/scratch/i10/b/probe/sha_$side.txt
done
echo PROBE-DONE
