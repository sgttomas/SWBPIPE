#!/bin/bash
# I9 T9 (Mac-only): S11-K's fixdiff harness built --release against a tree's product_physics, run over every
# committed JSON request or model under P/core, P/fixtures and P/validation of that tree, both modes.
set -u
WT=<wt>
S=$WT/scratch/i9
T=$1
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=$WT/skewpin-target/t9_$T
cd $S/t9/h_$T && cargo build --release --offline --locked -j 8 > $S/t9/build_$T.log 2>&1; echo "build rc=$?" >> $S/t9/build_$T.log
BIN=$WT/skewpin-target/t9_$T/release/calib_fixdiff_native
rm -rf $S/t9/out_$T; : > $S/t9/run_$T.log
for root in core fixtures validation; do
  mkdir -p $S/t9/out_$T/$root
  $BIN $S/trees/$T/projects/chirality-piping/$root $S/t9/out_$T/$root >> $S/t9/run_$T.log 2>&1; echo "$root rc=$?" >> $S/t9/run_$T.log
done
(cd $S/t9/out_$T && find . -type f | sort | xargs shasum -a 256) > $S/t9/output_sha256_$T.txt
