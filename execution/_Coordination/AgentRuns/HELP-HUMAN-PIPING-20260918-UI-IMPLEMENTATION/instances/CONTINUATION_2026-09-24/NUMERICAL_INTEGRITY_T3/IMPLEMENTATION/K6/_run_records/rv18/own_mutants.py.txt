"""I15's own mutants of the RV18 fixes (same interface as RV18's rv18_mutants.py): one exact edit each.

Usage: python3 own_mutants.py make <mutant> <copy root>
"""
import sys

H = 'projects/chirality-piping/core/solver/performance_harness/'
RUNNER = H + 'runner/k6_runner.py'
MODELS = H + 'src/k6/models.rs'
MUTANTS = {
    'K6-M27': (RUNNER, '        if proc is not None and watched is None:\n            stop_group(proc.pid, proc.pid)',
               '        if False:\n            stop_group(proc.pid, proc.pid)',
               "RV18-3: launch's finally does not kill the group"),
    'K6-M28': (RUNNER, '    signal.signal(signal.SIGTERM, handler)\n    return lambda: signal.signal(signal.SIGTERM, previous)',
               '    return lambda: None', 'RV18-3: SIGTERM is not mapped to SystemExit'),
    'K6-M29': (MODELS, '        self.member_count() >= CONT_LANE_REFUSAL_MEMBERS\n'
                       '            && (self.family == Family::Cont || self.has_cont_restraints())',
               '        self.id.starts_with("RF-LARGE-CONT-n10000-")', 'RV18-N6: the lane refusal keyed on the id again'),
    'K6-M30': (RUNNER, "    if clean != text:\n        with open(path, 'w') as fh:\n            fh.write(clean)",
               "    if False:\n        with open(path, 'w') as fh:\n            fh.write(clean)",
               "RV18-N2: time -v's output file is not scrubbed"),
}


def main():
    cmd, name, root = sys.argv[1:4]
    path, old, new, what = MUTANTS[name]
    full = root + '/' + path
    text = open(full).read()
    assert text.count(old) == 1, (name, text.count(old))
    open(full, 'w').write(text.replace(old, new))
    print('applied', name, '-', what)


if __name__ == '__main__':
    main()
