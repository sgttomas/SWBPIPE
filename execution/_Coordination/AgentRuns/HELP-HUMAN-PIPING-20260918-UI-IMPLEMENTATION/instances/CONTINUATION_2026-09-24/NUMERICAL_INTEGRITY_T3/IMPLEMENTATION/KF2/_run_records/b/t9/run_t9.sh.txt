#!/bin/bash
# KF2 B: T9 (S11-K's fixdiff harness ec089c1d..., release) on git-archive trees of base 78f55f927 and
# candidate 1b10121fa, plus F1b's extra corpus (8 requests, 16 outputs). Mac-only; base vs candidate.
set -u
WT=<wt>
B=$WT/scratch/i20/b
EXTRA=$WT/scratch/i13/a1/extra_corpus
for n in base cand; do
  BIN=$WT/kf2-target/b-t9-$n/release/calib_fixdiff_native
  TREE=$B/tree_$n
  rm -rf $B/t9/out_$n $B/t9/extra_$n; : > $B/t9/run_$n.log
  for root in core fixtures validation; do
    mkdir -p $B/t9/out_$n/$root
    $BIN $TREE/projects/chirality-piping/$root $B/t9/out_$n/$root >> $B/t9/run_$n.log 2>&1; echo "$root rc=$?" >> $B/t9/run_$n.log
  done
  ( cd $B/t9/out_$n && find . -type f -name '*.out' | sort | xargs shasum -a 256 ) > $B/t9/sha_$n.txt
  mkdir -p $B/t9/extra_$n; $BIN $EXTRA $B/t9/extra_$n >> $B/t9/run_$n.log 2>&1; echo "extra rc=$?" >> $B/t9/run_$n.log
  ( cd $B/t9/extra_$n && ls | sort | xargs shasum -a 256 ) > $B/t9/extra_sha_$n.txt
done
echo T9-DONE
