//! T3 S11-G (I5) scratch harness: S11-F (I4) fixture-diff harness, unchanged in behaviour
//! (IMPLEMENTATION/S11K/_run_records/fixture_diff/fixdiff_main.rs.txt).
//! - `fixdiff <root> <out>`: every committed JSON request or model under <root> through the
//!   captured entry in both modes; one pretty envelope (or ERR/PANIC) per run.
//! S11-G addition: each run also goes through the typed entry when the request parses as
//!   `LinearStaticPreviewRequest` (file suffix `.typed.out`); captured files are unchanged.
//! - `fixdiff probe <request.json> <mode> <captured|typed>`: one run, the envelope on stdout.
use open_pipe_stress_product_physics::{
    run_linear_static_preview_value_with_mode, run_linear_static_preview_with_mode,
    LinearStaticPreviewRequest, PreviewSolverMode,
};
use std::path::{Path, PathBuf};
fn walk(dir: &Path, out: &mut Vec<PathBuf>) {
    let mut entries: Vec<_> = std::fs::read_dir(dir).unwrap().map(|e| e.unwrap().path()).collect();
    entries.sort();
    for p in entries {
        if p.is_dir() { walk(&p, out); } else if p.extension().is_some_and(|e| e == "json") { out.push(p); }
    }
}
fn main() {
    let args: Vec<String> = std::env::args().collect();
    if args[1] == "probe" {
        let request: serde_json::Value = serde_json::from_str(&std::fs::read_to_string(&args[2]).unwrap()).unwrap();
        let mode = PreviewSolverMode::from_wire(&args[3]).unwrap();
        let out = if args[4] == "typed" {
            let typed: LinearStaticPreviewRequest = serde_json::from_value(request).unwrap();
            serde_json::to_string(&run_linear_static_preview_with_mode(typed, mode)).unwrap()
        } else {
            match run_linear_static_preview_value_with_mode(request, mode) {
                Ok(e) => serde_json::to_string(&e).unwrap(),
                Err(e) => format!("ERR {e}"),
            }
        };
        println!("{out}");
        return;
    }
    let root = PathBuf::from(&args[1]);
    let out = PathBuf::from(&args[2]);
    let mut files = Vec::new();
    walk(&root, &mut files);
    let mut count = 0;
    for file in files {
        let text = std::fs::read_to_string(&file).unwrap();
        let Ok(value) = serde_json::from_str::<serde_json::Value>(&text) else { continue };
        let request = if value.get("model").is_some_and(|m| m.is_object()) {
            value
        } else if value.get("document_kind").and_then(|k| k.as_str()) == Some("openpipestress.product_preview.model") {
            serde_json::json!({"model": value, "materials": []})
        } else { continue };
        let rel = file.strip_prefix(&root).unwrap().to_string_lossy().replace('/', "__");
        for (name, mode) in [("sparse_interactive", PreviewSolverMode::SparseInteractive), ("dense_scrutiny", PreviewSolverMode::DenseScrutiny)] {
            let req = request.clone();
            let result = std::panic::catch_unwind(move || run_linear_static_preview_value_with_mode(req, mode));
            let text = match result {
                Ok(Ok(envelope)) => serde_json::to_string_pretty(&envelope).unwrap(),
                Ok(Err(error)) => format!("ERR {error}"),
                Err(_) => "PANIC".to_string(),
            };
            std::fs::write(out.join(format!("{rel}.{name}.out")), text).unwrap();
            count += 1;
            if let Ok(typed) = serde_json::from_value::<LinearStaticPreviewRequest>(request.clone()) {
                let result = std::panic::catch_unwind(move || run_linear_static_preview_with_mode(typed, mode));
                let text = match result {
                    Ok(envelope) => serde_json::to_string_pretty(&envelope).unwrap(),
                    Err(_) => "PANIC".to_string(),
                };
                std::fs::write(out.join(format!("{rel}.{name}.typed.out")), text).unwrap();
                count += 1;
            }
        }
    }
    eprintln!("runs: {count}");
}
