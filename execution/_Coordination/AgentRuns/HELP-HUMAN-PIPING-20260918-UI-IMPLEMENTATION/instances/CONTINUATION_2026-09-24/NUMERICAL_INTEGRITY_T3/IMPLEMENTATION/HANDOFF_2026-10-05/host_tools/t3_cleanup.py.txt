#!/usr/bin/env python3
"""T3 host cleanup: remove only what is regenerable, after a reviewed plan.

Usage:
  python3 t3_cleanup.py plan  --out plan.json          # read-only: lists candidates with sizes
  python3 t3_cleanup.py apply --plan plan.json --log manifest.jsonl

Removes three classes under the T3 root (this file's parent's parent), never anything else:
  cache     a Cargo target directory: its root holds CACHEDIR.TAG or .rustc_info.json (topmost only)
  treecopy  a copied source tree: a directory named `projects` holding chirality-piping/core,
            that is not inside a Git worktree's own checkout (git archive copies, mutant copies)
  worktree  a Git worktree under the root whose HEAD is contained in origin/main or the
            integration branch on origin, whose branch (if any) equals its origin copy, and which
            `git worktree remove` accepts as clean (ignored files only). Never `numerics` (NUM)
            or `sweep-skewpin` (the DEC-025 tree), or any name passed with --keep.

Never touched: any path containing a file tracked in the Git worktree it sits in, logs and other files, `guard/`, `preserved-evidence/`, `tools/`, anything outside
the root. `apply` refuses while any cargo, rustc, pytest or vitest process runs, re-checks every
path's class immediately before removing it, skips a path modified in the last --min-age-hours
(default 2), and appends one JSON line per action to the log.
"""
import argparse, json, os, shutil, subprocess, sys, time

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
NUM = os.path.join(ROOT, "numerics")
NEVER = {"guard", "preserved-evidence", "tools"}
KEEP_WORKTREES = {"numerics", "sweep-skewpin"}
INTEGRATION = "origin/codex/piping-numerical-integrity-20260926"


def git(*a, cwd=NUM):
    return subprocess.run(["git", *a], cwd=cwd, capture_output=True, text=True,
                          env=dict(os.environ, GIT_OPTIONAL_LOCKS="0"))


def size(p):
    s = 0
    for a, _, fs in os.walk(p):
        for f in fs:
            try:
                s += os.lstat(os.path.join(a, f)).st_size
            except OSError:
                pass
    return s


def newest_mtime(p, limit=2000):
    """Newest mtime among p and up to `limit` entries below it (a cheap recency probe)."""
    m, n = os.lstat(p).st_mtime, 0
    for a, ds, fs in os.walk(p):
        for e in ds + fs:
            try:
                m = max(m, os.lstat(os.path.join(a, e)).st_mtime)
            except OSError:
                pass
            n += 1
            if n >= limit:
                return m
    return m


def is_cache(p):
    return os.path.isdir(p) and not os.path.islink(p) and (
        os.path.isfile(os.path.join(p, "CACHEDIR.TAG")) or os.path.isfile(os.path.join(p, ".rustc_info.json")))


def is_treecopy(p):
    return (os.path.basename(p) == "projects" and os.path.isdir(os.path.join(p, "chirality-piping", "core"))
            and not os.path.islink(p))


def tracked(p, wt_paths):
    """True when p lies inside a Git worktree and contains (or is) a tracked path there."""
    owners = [w for w in wt_paths if p == w or p.startswith(w + os.sep)]
    if not owners:
        return False
    w = max(owners, key=len)
    r = git("ls-files", "--", os.path.relpath(p, w), cwd=w)
    return r.returncode != 0 or bool(r.stdout.strip())


def worktrees():
    out, cur = {}, {}
    for line in git("worktree", "list", "--porcelain").stdout.splitlines() + [""]:
        if not line:
            if cur.get("worktree", "").startswith(ROOT + os.sep):
                out[cur["worktree"]] = cur
            cur = {}
        else:
            k, _, v = line.partition(" ")
            cur[k] = v
    return out


def worktree_ok(path, info):
    name = os.path.relpath(path, ROOT)
    if name in KEEP_WORKTREES or os.sep in name:
        return False, "kept"
    head = info.get("HEAD", "")
    contained = any(git("merge-base", "--is-ancestor", head, ref).returncode == 0
                    for ref in ("origin/main", INTEGRATION))
    if not contained:
        return False, "HEAD not contained in origin/main or the integration branch"
    br = info.get("branch", "").replace("refs/heads/", "")
    if br:
        r = git("rev-parse", "-q", "--verify", "origin/" + br)
        if r.returncode != 0 or r.stdout.strip() != head:
            return False, "branch not pushed as-is"
    st = git("status", "--porcelain", "--untracked-files=all", cwd=path)
    if st.returncode != 0 or st.stdout.strip():
        return False, "not clean"
    return True, "clean, contained and pushed"


def plan(args):
    wts = worktrees()
    wt_paths = set(wts)
    cands = []
    for a, ds, fs in os.walk(ROOT):
        rel = os.path.relpath(a, ROOT)
        if rel == ".":
            ds[:] = [d for d in ds if d not in NEVER]
        keep = []
        for d in ds:
            p = os.path.join(a, d)
            if os.path.islink(p):
                continue
            if is_cache(p) and not tracked(p, wt_paths):
                cands.append({"class": "cache", "path": p})
            elif is_treecopy(p) and not any(p == w + "/projects" for w in wt_paths) and not tracked(p, wt_paths):
                cands.append({"class": "treecopy", "path": p})
            elif os.path.isfile(os.path.join(p, ".git")) and p in wt_paths:
                keep.append(d)  # descend into worktrees for their caches only
            else:
                keep.append(d)
        ds[:] = keep
    for p, info in sorted(wts.items()):
        ok, why = worktree_ok(p, info)
        if ok:
            cands.append({"class": "worktree", "path": p, "head": info.get("HEAD"), "branch": info.get("branch", "")})
        else:
            print(f"keep worktree {os.path.relpath(p, ROOT)}: {why}", file=sys.stderr)
    # a worktree that will be removed whole needs no separate cache entries inside it
    whole = [c["path"] for c in cands if c["class"] == "worktree"]
    cands = [c for c in cands if c["class"] == "worktree" or not any(c["path"].startswith(w + "/") for w in whole)]
    for c in cands:
        c["bytes"] = size(c["path"])
    tot = {}
    for c in cands:
        tot[c["class"]] = tot.get(c["class"], 0) + c["bytes"]
    json.dump({"root": ROOT, "created": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()), "totals": tot,
               "candidates": cands}, open(args.out, "w"), indent=1)
    for k, v in sorted(tot.items()):
        print(f"{k}: {sum(1 for c in cands if c['class'] == k)} paths, {v / 1e9:.1f} GB")


def remove_tree(p):
    """rmtree that restores owner write permission inside p (read-only copies) and retries once."""
    import stat
    def onexc(func, path, exc):
        for d in {os.path.dirname(path), path}:
            if d.startswith(p) and os.path.exists(d) and not os.path.islink(d):
                os.chmod(d, os.lstat(d).st_mode | stat.S_IWUSR | (stat.S_IXUSR if os.path.isdir(d) else 0))
        func(path)
    parent = os.path.dirname(p)
    mode = os.lstat(parent).st_mode
    try:
        shutil.rmtree(p, onexc=onexc)
    except PermissionError:
        # the candidate's own entry sits in a read-only parent: allow it for this one removal
        os.chmod(parent, mode | stat.S_IWUSR)
        try:
            shutil.rmtree(p, onexc=onexc)
        finally:
            os.chmod(parent, mode)


def busy():
    ps = subprocess.run(["ps", "-axo", "command"], capture_output=True, text=True).stdout.splitlines()
    return [l for l in ps if any(t in l for t in ("cargo ", "rustc ", "pytest", "vitest")) and "t3_cleanup" not in l]


def attempt(p):
    try:
        remove_tree(p)
        return "removed"
    except OSError as e:
        return f"error: {type(e).__name__}: {e}"[:300]


def apply(args):
    b = busy()
    if b:
        sys.exit("refusing: build or test processes are running:\n  " + "\n  ".join(b[:5]))
    data = json.load(open(args.plan))
    if data["root"] != ROOT:
        sys.exit("refusing: the plan is for another root")
    wts = worktrees()
    log = open(args.log, "a")
    freed = 0
    for c in data["candidates"]:
        p, cls = c["path"], c["class"]
        rec = {"class": cls, "path": os.path.relpath(p, ROOT), "planned_bytes": c["bytes"]}
        real = os.path.realpath(p)
        if not real.startswith(ROOT + os.sep) or os.path.relpath(real, ROOT).split(os.sep)[0] in NEVER:
            rec["action"] = "skipped: outside the root or in a protected directory"
        elif not os.path.exists(p):
            rec["action"] = "skipped: already gone"
        elif time.time() - newest_mtime(p) < args.min_age_hours * 3600:
            rec["action"] = f"skipped: modified within {args.min_age_hours} h"
        elif cls != "worktree" and tracked(p, set(wts)):
            rec["action"] = "skipped: contains tracked files"
        elif cls == "cache" and is_cache(p):
            rec["action"] = attempt(p)
        elif cls == "treecopy" and is_treecopy(p) and not any(p == w + "/projects" for w in wts):
            rec["action"] = attempt(p)
        elif cls == "worktree" and p in wts and worktree_ok(p, wts[p])[0]:
            r = git("worktree", "remove", p)
            rec["action"] = "removed" if r.returncode == 0 else "skipped: " + r.stderr.strip()[:200]
        else:
            rec["action"] = "skipped: no longer matches its class"
        if rec["action"] == "removed":
            freed += c["bytes"]
        rec["at"] = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
        log.write(json.dumps(rec) + "\n")
    log.close()
    print(f"freed about {freed / 1e9:.1f} GB; log {args.log}")


ap = argparse.ArgumentParser()
sub = ap.add_subparsers(dest="cmd", required=True)
p1 = sub.add_parser("plan"); p1.add_argument("--out", required=True)
p1.add_argument("--keep", action="append", default=[])
p2 = sub.add_parser("apply"); p2.add_argument("--plan", required=True); p2.add_argument("--log", required=True)
p2.add_argument("--min-age-hours", type=float, default=2.0)
a = ap.parse_args()
if a.cmd == "plan":
    KEEP_WORKTREES.update(a.keep)
    plan(a)
else:
    apply(a)
