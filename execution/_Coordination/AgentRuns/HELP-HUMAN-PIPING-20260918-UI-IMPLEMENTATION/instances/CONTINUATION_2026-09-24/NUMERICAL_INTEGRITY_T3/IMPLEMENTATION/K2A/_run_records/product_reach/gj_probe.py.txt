"""K2a (I6): 1/L-lift zero probe (reach_gj/): G*J rounds to 0 (G = 1e-300 Pa, OD 1e-6 m, L = 2^-39 m), so
main's GJ/L = 0 while its true value is ~2^-1036; N1's only free DOF, RX, carries a ground spring k = 2^-p and
a moment 0.25 k. Main solves on k alone; relative error = GJ/L_true / (k + GJ/L_true), with and without an
open gap. Usage: python3 -B gj_probe.py <reach_requests.py> <outdir>"""
import math, json, sys
from fractions import Fraction as Fr
ns = {}; exec(open(sys.argv[1]).read().split('out = sys.argv[1]')[0], ns)
out = sys.argv[2]
L = 2.0 ** -39; od = 1e-6; t = 1e-7; E = 2e11; G = 1e-300
idd = od - 2 * t; i = math.pi * ((od * od) * (od * od) - (idd * idd) * (idd * idd)) / 64; j = 2 * i
gj = Fr(G) * Fr(j) / Fr(L)
print('G*J', G * j, 'true GJ/L 2^%.1f' % math.log2(float(gj)))
for p in (1000, 1005, 1008, 1010, 1015):
    k = 2.0 ** -p; M = float('%.3g' % (k * 0.25))
    rel = float(gj / (Fr(k) + gj))
    print(p, 'k', k, 'M', M, 'main theta error / criterion %.3g' % (rel * 1e9))
    for gap in (False, True):
        name = 'GJ-%d%s' % (p, '-gap' if gap else '')
        json.dump(ns['request'](name, L, od, t, E, G, 'RX', k, M, gap), open('%s/%s.json' % (out, name), 'w'))
