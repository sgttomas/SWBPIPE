#!/usr/bin/env python3
"""K2a (I6): the all-normal-coefficient partial-underflow probes (reach_probe/) and the M03 threshold
scan (reach_thr/), built with reach_requests.py's `request`. Invented inputs; L = 2^-39 m (the
smallest dyadic length above the kernel's 1e-12 m axis tolerance), G = 1e-200 Pa, P = 0.25 m times
main's formed 12EIz/L^3, rounded to 3 digits.
  reach_probe: A = (OD 1e-6 m, E 1.5e-294 Pa): main's 12EI/L^3 4.3e-6 wrong, smallest coefficient 2^-1021.x;
               B = (OD 5e-7 m, E 1.3e-290 Pa): 8.0e-9 wrong, smallest coefficient 2^-1012 (each with and
               without an open gap on N1 UY). Found by a search over the checked-formation emulation
               (k2a_models.py): every final coefficient normal, first refused intermediate (12*E)*Iy.
  reach_thr:   B with E scaled by 2^k, k = 0..100: the intermediate becomes normal at k = 26.
Usage: python3 -B gen_probes.py <reach_requests.py> <outdir>"""
import json, math, os, sys
ns = {}
exec(open(sys.argv[1]).read().split('out = sys.argv[1]')[0], ns)
out = sys.argv[2]
L = 2.0 ** -39
def write(sub, name, od, E, gap):
    t = od / 10; idd = od - 2 * t
    i = math.pi * ((od * od) * (od * od) - (idd * idd) * (idd * idd)) / 64
    K = 12.0 * E * i / (L * L * L)
    P = float('%.3g' % (K * 0.25))
    os.makedirs(os.path.join(out, sub), exist_ok=True)
    json.dump(ns['request'](name, L, od, t, E, 1e-200, 'UY', None, P, gap), open(os.path.join(out, sub, name + '.json'), 'w'))
for tag, od, E in (('A', 1e-6, 1.5e-294), ('B', 5e-7, 1.3e-290)):
    write('reach_probe', tag, od, E, False); write('reach_probe', tag + '-gap', od, E, True)
for k in (0, 20, 26, 30, 40, 50, 60, 70, 80, 90, 100):
    write('reach_thr', 'thr-%03d' % k, 5e-7, 1.3e-290 * 2.0 ** k, False)
