#!/bin/bash
# Dispatch one mutant id: K2B-* -> mutate_k2b.py (I10's, with the RV11-fix mutants);
# RV11-* -> RV11's own mutate_rv11.py (copied unchanged from its review records);
# everything else -> K1's mutate_k1.py (unchanged).
X=$1; CORE=$2; m=$3
case $m in
  K2B-*) python3 -B $X/mutate_k2b.py $CORE $m ;;
  RV11-*) python3 -B $X/mutate_rv11.py $CORE $m ;;
  *) python3 -B $X/mutate_k1.py $CORE $m ;;
esac
