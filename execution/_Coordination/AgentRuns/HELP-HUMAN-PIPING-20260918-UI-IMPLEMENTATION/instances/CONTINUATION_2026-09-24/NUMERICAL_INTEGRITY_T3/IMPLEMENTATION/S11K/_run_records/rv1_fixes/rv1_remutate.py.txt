"""RV1 fixes: re-run RV1's mutants RV-OPT1, RV-OPT3, RV-OPT4 and RV-PUB (patch
texts taken from RV1's rv1_mutate.py) against the fixed tests, on a scratch
copy of core/, and record every failing test."""
import json, os, re, shutil, subprocess, sys
WT = "<s11k-pr-worktree>/projects/chirality-piping/core"
SCR = sys.argv[1] + "/rv1/core"
ENV = dict(os.environ, RUSTUP_TOOLCHAIN="1.97.1", CARGO_INCREMENTAL="0",
           CARGO_TARGET_DIR="<t3-target>/s11k-rv1")
NI = "solver/nonlinear_integration/src/lib.rs"
FK = "solver/frame_kernel/src"
MUTANTS = [
 ("RV-OPT1", "option (c): loop reduction switched to the exact KS2",
  [(NI, "    let reduced = reduce_system_with_prescribed_displacements_binary64(\n        stiffness,",
        "    let reduced = open_pipe_stress_frame_kernel::reduce_system_with_prescribed_displacements(\n        stiffness,")],
  "solver/nonlinear_integration"),
 ("RV-OPT3", "option (c): loop dense and sparse solves (no assembly) switched to the exact solves",
  [(NI, "open_pipe_stress_frame_kernel::structural::solve_structural_dense_binary64(\n                    &original,\n                )?",
        "open_pipe_stress_frame_kernel::structural::solve_structural_dense(\n                    &original,\n                )?"),
   (NI, "structural_adapter::solve_structural_sparse_binary64(&original)?",
        "open_pipe_stress_sparse_direct::structural::solve_structural_sparse(&original)?")],
  "solver/nonlinear_integration"),
 ("RV-OPT4", "option (c) textual workaround: exact assembly solve through a helper, required text kept in a comment",
  [(NI, "        assembly.solve_binary64(\n", "        // assembly.solve_binary64(\n        exact_solve(assembly,\n"),
   (NI, "fn solve_linearized_system_evidence(\n",
        "fn exact_solve(a: &AssemblyEvidence, k: &[Vec<f64>], f: &[f64], free: &[usize], p: &[(usize, f64)], m: LinearSolveMode) -> Result<open_pipe_stress_frame_kernel::structural::StructuralSolution, StructuralError> { a.solve(k, f, free, p, m) }\nfn solve_linearized_system_evidence(\n")],
  "solver/nonlinear_integration"),
 ("RV-PUB", "public evaluate_original_residual switched to the exact KS3 numerator",
  [(FK + "/structural.rs", "    evaluate_original_residual_bound(system, ForceBinding::Binary64, u)\n",
        "    evaluate_original_residual_bound(system, ForceBinding::Legacy, u)\n")],
  "solver/frame_kernel"),
]
def reset():
    if os.path.exists(SCR): shutil.rmtree(SCR)
    shutil.copytree(WT, SCR, ignore=shutil.ignore_patterns("target"))
def patch(edits):
    for path, old, new in edits:
        full = os.path.join(SCR, path)
        text = open(full).read()
        assert text.count(old) == 1, (path, old[:60], text.count(old))
        open(full, "w").write(text.replace(old, new))
results = []
for label, what, edits, crate in MUTANTS:
    reset(); patch(edits)
    cmd = ["cargo", "test", "--offline", "--locked", "--lib", "--no-fail-fast"]
    p = subprocess.run(cmd, cwd=os.path.join(SCR, crate), env=ENV, capture_output=True, text=True)
    out = p.stdout + p.stderr
    failed = sorted(set(re.findall(r"^test (\S+) \.\.\. FAILED", out, re.M)))
    results.append({"mutant": label, "mutation": what, "crate": crate, "command": " ".join(cmd),
                    "exit": p.returncode, "failed_tests": failed,
                    "patch": [{"file": e[0], "old": e[1], "new": e[2]} for e in edits]})
    print(label, p.returncode, failed, flush=True)
reset()
json.dump(results, open(sys.argv[1] + "/rv1/rv1_remutate_results.json", "w"), indent=1)
