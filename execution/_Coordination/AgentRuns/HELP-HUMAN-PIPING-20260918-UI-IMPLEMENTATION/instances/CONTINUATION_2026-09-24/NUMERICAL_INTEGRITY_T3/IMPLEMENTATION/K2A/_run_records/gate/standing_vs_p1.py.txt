"""K-D5 (I3): numerical_quality status and outcome of every gate run against P1's main baseline
(same case, mode and entry: captured vs P1 captured; typed vs P1 typed_on_captured/typed_secondary).
Usage: python3 -B standing_vs_p1.py <runs.jsonl> <DETECTION/results.json>  (committed as .txt)"""
import json, sys
sys.path.insert(0, __file__.rsplit('/', 1)[0])
import compare as C
runs, res = sys.argv[1:3]
cap, typ = {}, {}
for c in json.load(open(res))['cases']:
    for m, v in c.get('modes', {}).items():
        cap[(c['id'], m)] = (v.get('outcome'), (v.get('identity') or {}).get('numerical_quality_status'))
        for tk in ('typed_on_captured', 'typed_secondary'):
            t = v.get(tk)
            if t:
                typ[(c['id'], m)] = (t.get('outcome'), (t.get('identity') or {}).get('numerical_quality_status'))
n, chg = 0, []
for line in open(runs):
    r = json.loads(line)
    k = (r['case'], r['mode'])
    base = cap.get(k) if r['entry'] == 'captured' else typ.get(k)
    if base is None:
        continue
    out, _ = C.classify(r)
    run = r.get('probe', {}).get('run', {})
    q = C.identity(run['envelope']).get('numerical_quality_status') if run.get('ok') else None
    n += 1
    if (out, q) != base:
        chg.append((r['case'], r['mode'], r['entry'], base, (out, q)))
print('runs compared with a same-entry P1 record:', n)
print('changes (P1 main -> K-D5 candidate):')
for x in chg:
    print(' ', x)
