"""K6b: the mutants of the RV22 fixes (ROOT's rulings on RV22's review; NONE first). Standard library only.

Usage: python3 mut.py <worktree k6b> <mutants root> <results.jsonl> [ids...]

Each mutant: a clean `git archive <base>` of the core tree, with the candidate's changed files
copied over it (the candidate is the base plus the working tree's uncommitted changes), then
the mutant's exact edits (each old string must occur exactly the stated number of times). Rust
mutants run the K6b test binaries in debug with the mutant's own target, deleted afterwards;
Python mutants run the runner's unittest suite. One cargo job at -j 4, RUST_TEST_THREADS=2.
"""
import json
import os
import shutil
import subprocess
import sys
import time

sys.dont_write_bytecode = True
BASE = '1123d19b9'
SOLVER = 'projects/chirality-piping/core/solver'
H = SOLVER + '/performance_harness'
RUST_TESTS = ['--test', 'k6b_adapter', '--test', 'k6b_w1', '--test', 'k6b_bin', '--test', 'k6b_export']

MUTANTS = [
    ('NONE', 'the candidate', 'both', []),
    ('M21', 'RV22-1 reverted: a stopped build may be short on both sides', 'rust', [(
        H + '/src/k6/w1/staged.rs', '            own <= own_t && shared <= shared_t && (own == own_t || shared == shared_t)\n',
        '            own <= own_t && shared <= shared_t\n', 1)]),
    ('M22', 'a stop in the stop rule is relaxed like a stopped build', 'rust', [(
        H + '/src/k6/w1/staged.rs', ' || a.stop_rule_work > 0\n', '\n', 1)]),
    ('M23', 'stages_complete is the outcome test, not unstaged == (0, 0)', 'rust', [(
        H + '/src/k6/w1/staged.rs', '    unstaged(a) == (0, 0)\n',
        '    !matches!(a.outcome, AttemptOutcome::Failed(AttemptReason::Stop(_)))\n', 1)]),
    ('RV22-M6', 'SOLVE_TRACKER_PEAK_ROWS 2816 -> 2048', 'rust', [(
        H + '/src/k6/w1/counts.rs', 'pub const SOLVE_TRACKER_PEAK_ROWS: u128 = 2816;\n',
        'pub const SOLVE_TRACKER_PEAK_ROWS: u128 = 2048;\n', 1)]),
    ('E1', 'the estimate omits the fallback', 'rust', [(
        H + '/src/k6/w1/counts.rs', '        SOLVE_VECTORS * n * w(r) + nf * (RESIDUAL_ROW_EXTRA + w(l)) + solve_trackers + fallback\n',
        '        let _ = fallback;\n        SOLVE_VECTORS * n * w(r) + nf * (RESIDUAL_ROW_EXTRA + w(l)) + solve_trackers\n', 1)]),
    ('E2', 'the pass omits its three full-length vectors (w_abs, delta_full, w_s)', 'rust', [(
        H + '/src/k6/w1/counts.rs', '        let live = PASS_FULL_VECTORS * n * w(l)\n',
        '        let live = 0 * PASS_FULL_VECTORS * n * w(l)\n', 1)]),
    ('E3', 'the row list at 2 n_f (in place), not its move-model peak', 'rust', [(
        H + '/src/k6/w1/counts.rs', '                .max(n * w(l) + fallback_row_list_rows(nf) * TRACKER_ENTRY_BYTES);\n',
        '                .max(n * w(l) + 2 * nf * TRACKER_ENTRY_BYTES);\n', 1)]),
    ('E4', 'the tables without the growing one under the move model', 'rust', [(
        H + '/src/k6/w1/counts.rs', 'pub const TABLE_MOVE_EXTRA: u128 = 1;\n', 'pub const TABLE_MOVE_EXTRA: u128 = 0;\n', 1)]),
    ('RV22-M5L', "the prefix predicate always true (the library's, which the binary calls)", 'rust', [(
        H + '/src/k6/w1/staged.rs', '    budget_stop && own.len() >= j && full_segments.len() > j && own[..j] == full_segments[..j]\n',
        '    let _ = (budget_stop, own, full_segments, j);\n    true\n', 1)]),
    ('RV22-M5B', "the binary's call site forced true (RV22-M5 as written)", 'rust', [(
        H + '/src/bin/k6_observe/w1.rs', '        prefix_matches(j, full, &solve.outcome),\n',
        '        { let _ = prefix_matches(j, full, &solve.outcome); true },\n', 1)]),
    ('RV22-M7', "the runner's backstop against the RSS cap's half", 'python', [(
        H + '/runner/k6_runner.py', "    if estimate > run['heap_cap_bytes'] // 2:\n",
        "    if estimate > run['rss_cap_bytes'] // 2:\n", 1)]),
]


def run(cmd, cwd, env, log):
    t0 = time.time()
    p = subprocess.run(cmd, cwd=cwd, env=env, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    log.write('$ %s\n%s\n' % (' '.join(cmd), p.stdout))
    return p.returncode, time.time() - t0, p.stdout


def main():
    wt, root, results_path = sys.argv[1], sys.argv[2], sys.argv[3]
    only = set(sys.argv[4:])
    changed = subprocess.run(['git', 'diff', '--name-only', BASE, '--', 'projects/chirality-piping/core'],
                             cwd=wt, stdout=subprocess.PIPE, text=True, check=True).stdout.split()
    changed += subprocess.run(['git', 'ls-files', '--others', '--exclude-standard', '--',
                               'projects/chirality-piping/core'],
                              cwd=wt, stdout=subprocess.PIPE, text=True, check=True).stdout.split()
    os.makedirs(root, exist_ok=True)
    for mid, what, kind, edits in MUTANTS:
        if only and mid not in only:
            continue
        mdir = os.path.join(root, mid)
        shutil.rmtree(mdir, ignore_errors=True)
        src = os.path.join(mdir, 'src')
        os.makedirs(src)
        archive = subprocess.run(['git', 'archive', BASE, 'projects/chirality-piping/core'], cwd=wt,
                                 stdout=subprocess.PIPE, check=True).stdout
        subprocess.run(['tar', '-x', '-C', src], input=archive, check=True)
        for rel in changed:
            dest = os.path.join(src, rel)
            os.makedirs(os.path.dirname(dest), exist_ok=True)
            shutil.copy2(os.path.join(wt, rel), dest)
        for rel, old, new, count in edits:
            path = os.path.join(src, rel)
            with open(path) as fh:
                text = fh.read()
            found = text.count(old)
            assert found == count, (mid, rel, found, count)
            with open(path, 'w') as fh:
                fh.write(text.replace(old, new))
        log_path = os.path.join(root, mid + '.log')
        env = dict(os.environ, RUSTUP_TOOLCHAIN='1.97.1', RUSTUP_AUTO_INSTALL='0', CARGO_INCREMENTAL='0',
                   CARGO_TARGET_DIR=os.path.join(mdir, 'target'), RUST_TEST_THREADS='2',
                   PYTHONDONTWRITEBYTECODE='1')
        record = {'mutant': mid, 'what': what, 'kind': kind, 'edits': len(edits)}
        with open(log_path, 'w') as log:
            if kind in ('rust', 'both'):
                rc, secs, out = run(['cargo', 'test', '--offline', '--locked', '-j', '4'] + RUST_TESTS,
                                    os.path.join(src, H), env, log)
                record.update(rust_rc=rc, rust_s=round(secs, 1),
                              rust_failed=[l.split()[1] for l in out.splitlines()
                                           if l.startswith('test ') and l.endswith('FAILED')],
                              rust_compile_error='error[' in out or 'could not compile' in out)
            if kind in ('python', 'both'):
                rc, secs, out = run([sys.executable, '-m', 'unittest', 'test_k6_runner'],
                                    os.path.join(src, H, 'runner'), env, log)
                record.update(py_rc=rc, py_s=round(secs, 1),
                              py_failed=[l.split()[1] for l in out.splitlines()
                                         if l.startswith(('FAIL:', 'ERROR:'))])
        failed = record.get('rust_rc', 0) != 0 or record.get('py_rc', 0) != 0
        record['result'] = ('passes (control)' if not failed else 'CONTROL FAILED') if mid == 'NONE' else \
            ('killed' if failed else 'SURVIVED')
        shutil.rmtree(mdir, ignore_errors=True)  # the clean copy and its target
        with open(results_path, 'a') as fh:
            fh.write(json.dumps(record, sort_keys=True) + '\n')
        print(json.dumps(record, sort_keys=True), flush=True)


if __name__ == '__main__':
    main()
