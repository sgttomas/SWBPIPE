#!/bin/bash
# F1b records (I13, checkpoint D): every reference to every product item F1b changed, added or removed,
# and to the kernel entries F1b newly calls from the product, in a git archive of 130445db2 (the candidate
# head; no working-tree or index state is read). /usr/bin/grep -rnw only (a word-bounded lexer scan); Rust,
# TypeScript and Python sources under projects/ and tools/, target/, node_modules/ and .venv/ excluded;
# execution/ (records) excluded. Each occurrence is listed as file:line with the line's text.
# Usage: scan_callers.sh <archive-root>
cd "$1" || exit 1
echo "F1b callers scan of git archive 130445db2 (projects/ and tools/, *.rs *.ts *.tsx *.py; target/, node_modules/, .venv/ and projects/chirality-piping/execution/ excluded)"
scan() {
  echo
  echo "== $1"
  /usr/bin/grep -rnw --include='*.rs' --include='*.ts' --include='*.tsx' --include='*.py' \
    --exclude-dir=target --exclude-dir=node_modules --exclude-dir=.venv --exclude-dir=execution "$1" projects tools 2>/dev/null \
    | sed -E 's/^([^:]+:[0-9]+):[[:space:]]*/\1: /' | cut -c1-220 | sort -t: -k1,1 -k2,2n
  echo "   ($(/usr/bin/grep -rnw --include='*.rs' --include='*.ts' --include='*.tsx' --include='*.py' --exclude-dir=target --exclude-dir=node_modules --exclude-dir=.venv --exclude-dir=execution "$1" projects tools 2>/dev/null | wc -l | tr -d ' ') occurrences)"
}
echo
echo "#### Existing product functions whose signature or body F1b changed"
for n in solve_load_case solve_preview_reduced_system run_linear_static_preview_captured_once append_integrity_report append_integrity_failure restrained_reactions legacy_observation_force legacy_dense_observation prepare_sources; do scan $n; done
echo
echo "#### Removed from PP lib.rs (expected: no occurrence in product_physics; other crates have their own functions of this name)"
scan multiply_matrix_vector
echo
echo "#### New product items (PP lib.rs, source_recovery.rs) and the visibility change (formation_guard.rs)"
for n in BasisStiffness assemble_basis_stiffness form_basis_stiffness DENSE_SCRUTINY_BYTES_PER_ENTRY DENSE_SCRUTINY_CEILING_BYTES DenseScrutinyRefusal dense_scrutiny_estimate_bytes dense_scrutiny_guard dense_scrutiny_ceiling_bytes dense_scrutiny_refusal_diagnostic SPARSE_OBSERVATION_BYTES_PER_PROFILE_ENTRY observation_lane_profile ObservationLaneRefusal observation_lane_guard observation_lane_refusal_diagnostic OrdinaryFailure ordinary_range_trigger ForceScaledPublication ForceScalingFailure force_scaling_attempt force_scaling_admission force_scaled_publication range_scaling_evidence_line append_force_scaling_refusal integrity_failure_code DENSE_SOURCE_DOF_LIMIT range_formation_decline_without_attempt solve_ordinary NAMED; do scan $n; done
echo
echo "#### Kernel entries F1b newly calls from the product (the product occurrences are the new callers)"
for n in assemble_sparse_stiffness reduce_assembled_sparse_system SparseAssemblyEvidence SparseAssemblyOptions solve_with_force_scaling ForceScalingCase force_scaled_reactions force_scaled_end_actions force_scaled_spring_action; do scan $n; done
