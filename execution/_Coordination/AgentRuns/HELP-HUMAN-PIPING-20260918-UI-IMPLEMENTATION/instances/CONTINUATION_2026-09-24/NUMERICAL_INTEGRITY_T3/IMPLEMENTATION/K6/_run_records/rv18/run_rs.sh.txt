#!/bin/bash
# RV18 fixes: one Rust run from a clean `git archive ae3320b5a` copy (core/solver, core/loads), with I15's changed
# files overlaid, the mutant applied (RV18's script, or I15's own for K6-M27+), a fresh target, H's full cargo test.
W=<wt>; R=$W/scratch/i15/rv18
M=$1; D=$W/k6-mut/rv18-rs-$M; T=$W/k6-mut-target/rv18-rs-$M; LOG=$R/logs/rs-$M.log
rm -rf "$D" "$T"; mkdir -p "$D"
{
  echo "=== rs $M: git archive ae3320b5a93aa0e65a71483b4db6f40eeb9d8f1c (core/solver, core/loads) + overlay (I15's RV18 fixes)"
  git -C $W/k6 archive ae3320b5a93aa0e65a71483b4db6f40eeb9d8f1c projects/chirality-piping/core/solver projects/chirality-piping/core/loads | tar -x -C "$D"
  (cd $R/overlay && find . -type f | sort | while read f; do cp "$f" "$D/$f"; echo "overlay $f $(shasum -a 256 "$f" | cut -c1-16)"; done)
  case "$M" in NONE) ;; RV18-*|K6-M4r|K6-M12r|K6-M16r) python3 $R/rv18_mutants.py make "$M" "$D" ;; *) python3 $R/own_mutants.py make "$M" "$D" ;; esac
  echo "=== cargo test --offline --locked -j 4 --no-fail-fast (RUSTUP_TOOLCHAIN=1.97.1 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2)"
} > "$LOG" 2>&1
cd "$D/projects/chirality-piping/core/solver/performance_harness" || exit 9
s=$(date +%s)
RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2 CARGO_TARGET_DIR="$T" \
  cargo test --offline --locked -j 4 --no-fail-fast >> "$LOG" 2>&1
echo "=== exit $? elapsed $(( $(date +%s) - s )) s" >> "$LOG"
rm -rf "$T"
sed -i '' -e "s#$W#<wt>#g" -e 's/[[:space:]]*$//' "$LOG"
