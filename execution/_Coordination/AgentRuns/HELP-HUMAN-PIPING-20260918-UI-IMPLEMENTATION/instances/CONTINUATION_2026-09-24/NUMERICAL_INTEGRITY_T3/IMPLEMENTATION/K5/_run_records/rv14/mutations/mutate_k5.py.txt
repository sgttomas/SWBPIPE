#!/usr/bin/env python3
"""K5 (I14, checkpoint C) mutation patches. Usage: python3 mutate_k5.py <core-dir> <id>
Applies one mutant to a clean scratch copy of projects/chirality-piping/core. Every patch is an
exact text replacement that must match exactly once (asserted). The original pins' mutants are
applied with K1's `mutate_k1.py` (unchanged), not here; KD5-SELECT is added here."""

core, mid = sys.argv[1], sys.argv[2]
RB = core + "/solver/frame_kernel/src/rigid_body.rs"
SA = core + "/solver/nonlinear_integration/src/structural_adapter.rs"

                    .add(coordinates[near][axis])

                offset[axis]
                    .add(-coordinates[far][axis])
                    .map_err(|_| RANGE)?;"""

            let d = difference3(&translations[a], &translations[b])?;"""

                    let mut action = Expansion::default();
                    for axis in 0..3 {
                        for &term in &u[axis].terms {
                            action.add_product(ground.direction[axis], term, 0)?;
                        }
                    }
                    action.is_zero()
                }"""
SKIP = """            if !qualified {
                if let Some(w4) = &self.w4 {"""

            let d = difference3(&translations[a], &translations[b])?;""")],
    # K5-M2: the virtual shift's sign flipped (x_b - x_a).
    "K5-M2": [(RB, OFFSET, OFFSET.replace("coordinates[near]", "coordinates[@]").replace(

    "K5-M3": [(RB, "        if !(value.is_finite() && value > 0.0) {", "        if false && !(value.is_finite() && value > 0.0) {")],

    "K5-M4a": [(SA, "                if slot.explicit {\n                    return W4Body::Unqualified(W4Unqualified::CurvedExplicit { element_id });",

    # K5-M4b: an unmatched curved slot qualified (as a link, with the recorded coordinates).
    "K5-M4b": [(SA, "                else {\n                    return W4Body::Unqualified(W4Unqualified::CurvedUnmatched { element_id });\n                };",
                "                else {\n                    links.push([la, lb]);\n                    continue;\n                };")],
    # K5-M6: a witness published from the rounded candidate without the exact check.
    "K5-M6": [(RB, TIES, "        return Ok(Some(translations));\n" + TIES)],
    # K5-M7: the τ_B band dropped: Restrained whenever no exact witness exists.
    "K5-M7": [(RB, "    Ok(result)\n}\n\n/// Validation, canonical order and the exact tie reduction.",
               "    result.status = RigidBodyStatus::Restrained;\n    Ok(result)\n}\n\n/// Validation, canonical order and the exact tie reduction.")],
    # K5-M8a: W4 reached from `solve_binary64` (the loop's solve), sources empty.
    "K5-M8a": "binary64",
    # K5-M8b: W4 reached before the `if !selected` return (unselected dense branch).
    "K5-M8b": [(SA, """        if !selected {
            return self.solve_assembled(k, f, free, prescribed, mode);
        }
        unscaled_evidence(self.force_scale)?;
        self.constrained_geometry(prescribed, curved_sources)?;""",
                """        self.constrained_geometry(prescribed, curved_sources)?;
        if !selected {
            return self.solve_assembled(k, f, free, prescribed, mode);
        }
        unscaled_evidence(self.force_scale)?;""")],
    # K5-M9: frame-only bodies routed through W4 in the selected branches.
    "K5-M9": [(SA, SKIP, "            if true {\n                if let Some(w4) = &self.w4 {")],
    # K5-M9b: the order changed: every mixed body screened after all frame-only bodies.
    "K5-M9b": [(SA, "        let mut seen = vec![false; n];\n        for seed in 0..n {",

    "K5-M10": [(SA, "                &element.global_stiffness,\n                bounds,\n                operations,\n                false,",

if mid in ("NONE", "D-NONE", "D2-NONE"):
    pass
elif MUTANTS[mid] == "binary64":
    text = open(SA).read()
    at = text.index("    pub fn solve_binary64(")
    old = "        self.geometry(prescribed)?;\n"
    k = text.index(old, at)
    assert text.count("    pub fn solve_binary64(") == 1
    text = text[:k] + "        self.constrained_geometry(prescribed, &[])?;\n" + text[k + len(old):]
    open(SA, "w").write(text)
else:
    for path, old, new in MUTANTS[mid]:
        text = open(path).read()
        assert text.count(old) == 1, (mid, path, text.count(old))
        open(path, "w").write(text.replace(old, new))
print("applied", mid)
