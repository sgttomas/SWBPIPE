#!/usr/bin/env python3
"""K2b records (I10, checkpoint D): copy the run evidence into _run_records/, sanitized.
- Machine paths become placeholders, longest first: <VENV> (the DEC-025 venv), <wt> (the T3
  worktrees root, so the scratch directory reads <wt>/scratch/i10 and the target <wt>/k2b-target),
  {REPO_ROOT} (the outer worktree), <scratch> (the session scratchpad) and <home>.
- A line longer than 600 characters in a log (Debug renderings in failure output) is cut to 600
  characters, followed by a marker naming the number of characters removed and the sha256 of the
  full line. Summary files, which CHANGE_RECORD and RETURN quote, are never cut.
- Trailing whitespace is stripped. Scripts are stored with a .txt suffix.
Usage: assemble.py <wt> <outer-worktree> <venv> <session-scratchpad> <out-dir>"""
import hashlib, os, sys

wt, outer, venv, pad, out = sys.argv[1:6]
home = os.path.expanduser("~")
LIMIT = 600
PLACES = sorted([(venv, "<VENV>"), (wt, "<wt>"), (outer, "{REPO_ROOT}"), (pad, "<scratch>"),
                 ("/private" + pad if not pad.startswith("/private") else pad[len("/private"):], "<scratch>"),
                 (home, "<home>")], key=lambda p: -len(p[0]))
UNCUT = ("MUTANTS.txt", "kill_sites.txt", "kill_sites_batch2.txt", "kill_site_comparison.txt",
         "k1_combined_kill_sites.txt", "suites_compare.txt", "suites_compare.json", "callers.txt",
         "a_k2b_summary.log", "b_k2b_summary.log", "failure_blocks_sha256.txt", "retry_probe.log",
         "retry_probe2.log", "kill_sites_batch3.txt")
count = 0

def clean(text, cut):
    for path, name in PLACES:
        text = text.replace(path, name)
    lines = []
    for line in text.split("\n"):
        line = line.rstrip()
        if cut and len(line) > LIMIT:
            digest = hashlib.sha256(line.encode()).hexdigest()
            line = line[:LIMIT] + f" [... K2b-records cut: {len(line) - LIMIT} more characters; sha256 of the full line {digest}]"
        lines.append(line)
    return "\n".join(lines)

def put(src, rel):
    global count
    dst = os.path.join(out, rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    with open(src, encoding="utf-8", errors="replace") as f:
        text = f.read()
    with open(dst, "w", encoding="utf-8") as f:
        f.write(clean(text, not rel.endswith(UNCUT)))
    count += 1

S = os.path.join(wt, "scratch", "i10")
L = os.path.join(S, "logs")
# checkpoint A: compile and targeted tests (the tree committed as 6ce4d694b), in run order
put(os.path.join(S, "run_a.sh"), "checkpoint_a/run_a.sh.txt")
for name in ["ni_first.log", "a_fk.log", "a_sd.log", "a_ni.log", "ni_k2b_1.log", "ni_k2b_2.log",
             "pp_targeted_1.log", "a_k2b_summary.log"]:
    put(os.path.join(L, name), f"checkpoint_a/{name}")
# checkpoint B: the reach and LEF summary, the suites, the b = 0 probe and T9
B = os.path.join(S, "b")
put(os.path.join(L, "b_k2b_summary.log"), "checkpoint_b/b_k2b_summary.log")
put(os.path.join(B, "overlay.txt"), "checkpoint_b/overlay.txt")
put(os.path.join(B, "run_suites_nff.sh"), "checkpoint_b/suites/run_suites_nff.sh.txt")
put(os.path.join(B, "suites.log"), "checkpoint_b/suites/suites.log")
put(os.path.join(B, "compare_suites.py"), "checkpoint_b/suites/compare_suites.py.txt")
put(os.path.join(B, "suites_compare.txt"), "checkpoint_b/suites/suites_compare.txt")
put(os.path.join(B, "suites_compare.json"), "checkpoint_b/suites/suites_compare.json")
put(os.path.join(B, "failure_blocks.py"), "checkpoint_b/suites/failure_blocks.py.txt")
put(os.path.join(B, "failure_blocks_sha256.txt"), "checkpoint_b/suites/failure_blocks_sha256.txt")
for name in sorted(os.listdir(os.path.join(B, "suites"))):
    put(os.path.join(B, "suites", name), f"checkpoint_b/suites/logs/{name}")
P = os.path.join(B, "probe")
put(os.path.join(P, "src", "main.rs"), "checkpoint_b/probe/main.rs.txt")
put(os.path.join(P, "run_probe.sh"), "checkpoint_b/probe/run_probe.sh.txt")
for side in ["base", "cand"]:
    put(os.path.join(P, f"p_{side}", "Cargo.toml"), f"checkpoint_b/probe/probe_{side}_Cargo.toml.txt")
    put(os.path.join(P, f"build_{side}.log"), f"checkpoint_b/probe/build_{side}.log")
    put(os.path.join(P, f"run_{side}.log"), f"checkpoint_b/probe/run_{side}.log")
    put(os.path.join(P, f"sha_{side}.txt"), f"checkpoint_b/probe/output_sha256_{side}.txt")
T = os.path.join(B, "t9")
put(os.path.join(T, "run_t9.sh"), "checkpoint_b/t9/run_t9.sh.txt")
put(os.path.join(T, "t9.log"), "checkpoint_b/t9/t9.log")
for side in ["base", "cand"]:
    put(os.path.join(T, f"h_{side}", "Cargo.toml"), f"checkpoint_b/t9/harness_{side}_Cargo.toml.txt")
    put(os.path.join(T, f"build_{side}.log"), f"checkpoint_b/t9/build_{side}.log")
    put(os.path.join(T, f"run_{side}.log"), f"checkpoint_b/t9/run_{side}.log")
    put(os.path.join(T, f"sha_{side}.txt"), f"checkpoint_b/t9/output_sha256_{side}.txt")
put(os.path.join(wt, "scratch", "calib", "fixdiff", "sha_native.txt"), "checkpoint_b/t9/calibration_sha_native.txt")
# checkpoint C: mutations
C = os.path.join(S, "c")
for name in ["mutate_k2b.py", "mutate_k1.py", "apply.sh", "run_mutant.sh", "drive.sh", "drive2.sh",
             "kill_sites.py", "compare_kill_sites.py"]:
    put(os.path.join(C, name), f"mutations/{name}.txt")
for name in ["ids.txt", "ids2.txt", "overlay.txt", "MUTANTS.txt", "kill_sites.txt",
             "kill_sites_batch2.txt", "kill_site_comparison.txt", "k1_combined_kill_sites.txt",
             "drive.log", "drive2.log"]:
    put(os.path.join(C, name), f"mutations/{name}")
for name in sorted(os.listdir(os.path.join(C, "logs"))):
    put(os.path.join(C, "logs", name), f"mutations/logs/{name}")
# the generator of k2b_models.rs
put(os.path.join(S, "gen", "k2b_models.py"), "generator/k2b_models.py.txt")
put(os.path.join(S, "gen", "k2b_models.json"), "generator/k2b_models.json")
# checkpoint D: the "third attempt" probe (scratch tests, not committed), the callers scan, the toolchain
D = os.path.join(S, "d")
put(os.path.join(D, "retry_probe_test.rs"), "retry_probe/retry_probe_test.rs.txt")
put(os.path.join(D, "retry_probe_test2.rs"), "retry_probe/retry_probe_test2.rs.txt")
put(os.path.join(D, "retry_probe.log"), "retry_probe/retry_probe.log")
put(os.path.join(D, "retry_probe2.log"), "retry_probe/retry_probe2.log")
# after checkpoint C (ROOT, 97000ab9f): the pin test's targeted run, and batch 3 of the mutations
put(os.path.join(D, "pin_targeted.log"), "retry_probe/pin_targeted.log")
C3 = os.path.join(S, "c3")
for name in ["mutate_k2b.py", "drive3.sh", "drive3b.sh"]:
    put(os.path.join(C3, name), f"mutations/batch3/{name}.txt")
for name in ["ids3.txt", "overlay3.txt", "MUTANTS.txt", "kill_sites_batch3.txt", "drive3.log", "drive3b.log"]:
    put(os.path.join(C3, name), f"mutations/batch3/{name}")
for name in sorted(os.listdir(os.path.join(C3, "logs"))):
    put(os.path.join(C3, "logs", name), f"mutations/batch3/logs/{name}")
put(os.path.join(D, "callers.txt"), "callers.txt")
put(os.path.join(D, "scan_callers.sh"), "scan_callers.sh.txt")
put(os.path.join(D, "toolchain.txt"), "toolchain.txt")
put(os.path.join(D, "assemble.py"), "assemble_run_records.py.txt")
print(f"done: {count} files")
