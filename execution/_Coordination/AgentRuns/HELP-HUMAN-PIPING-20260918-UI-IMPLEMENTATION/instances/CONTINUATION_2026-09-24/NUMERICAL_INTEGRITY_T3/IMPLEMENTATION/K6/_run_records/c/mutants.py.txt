def apply(mid, root):
    kind, what, rel, old, new = MUTANTS[mid]
    path = root.rstrip('/') + '/' + rel
    with open(path) as fh:
        text = fh.read()
    count = text.count(old)
    if count != 1:
        sys.exit('%s: the target text occurs %d times in %s' % (mid, count, rel))
    mutated = text.replace(old, new)
    with open(path, 'w') as fh:
        fh.write(mutated)
    print('%s applied to %s: %s' % (mid, rel, what))
    import difflib
    sys.stdout.writelines(difflib.unified_diff(text.splitlines(True), mutated.splitlines(True),
                                               'a/' + rel, 'b/' + rel, n=1))


if __name__ == '__main__':
    if sys.argv[1] == 'list':
        for mid, (kind, what, rel, old, new) in MUTANTS.items():
            print(json.dumps({'id': mid, 'kind': kind, 'what': what, 'file': rel.split('/')[-1]}))
    elif sys.argv[1] == 'apply':
        apply(sys.argv[2], sys.argv[3])
