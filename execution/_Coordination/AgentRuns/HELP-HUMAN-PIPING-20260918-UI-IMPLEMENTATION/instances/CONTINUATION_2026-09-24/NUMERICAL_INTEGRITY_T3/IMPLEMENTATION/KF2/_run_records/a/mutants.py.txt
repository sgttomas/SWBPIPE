"""KF2 mutants (plan §9). Each applies one edit to the candidate's
negative_pair_witness_counted in a clean copy. Usage: mutants.py <id> <structural.rs>."""
import sys

FN_START = "fn negative_pair_witness_counted("
FN_END = "\n// ------------------------------------------------------------------ K2b"


def edit(src, old, new, count=1):
    start = src.index(FN_START)
    end = src.index(FN_END, start)
    body = src[start:end]
    assert body.count(old) == count, (old, body.count(old))
    return src[:start] + body.replace(old, new) + src[end:]


MUTANTS = {
    "NONE": [],
    # brief 1: skip the first pair
    "M1": [("            visited += 1;\n",
            "            if (i, j) == (1, 0) {\n                continue;\n            }\n            visited += 1;\n")],
    # brief 2: the four cells in reverse order
    "M2": [("""            let cells = [
                (j, j, sign, sign),
                (j, i, sign, 1.0),
                (i, j, 1.0, sign),
                (i, i, 1.0, 1.0),
            ];""", """            let cells = [
                (i, i, 1.0, 1.0),
                (i, j, 1.0, sign),
                (j, i, sign, 1.0),
                (j, j, sign, sign),
            ];""")],
    # brief 3: terms + 1 in the allowance (too large)
    "M3": [("gamma(3 * terms + 2)", "gamma(3 * (terms + 1) + 2)")],
    # 3·terms + 1 (too small; result-equivalent by design)
    "M3p": [("gamma(3 * terms + 2)", "gamma(3 * terms + 1)")],
    # brief 4: return the last witness
    "M4": [("    let mut verified = 0;\n", "    let mut verified = 0;\n    let mut last = None;\n"),
           ("                    return Ok((Some(witness), visited, evaluated, verified));\n",
            "                    last = Some(witness);\n"),
           ("    Ok((None, visited, evaluated, verified))\n", "    Ok((last, visited, evaluated, verified))\n")],
    # brief 5: the sparse witness's skip of zero-coupling pairs
    "M5": [("            visited += 1;\n",
            "            if prepared.matrix[i][j] == 0.0 {\n                continue;\n            }\n            visited += 1;\n")],
    # sign rule >= 0.0 -> > 0.0
    "M6": [("if prepared.matrix[i][j] >= 0.0 {", "if prepared.matrix[i][j] > 0.0 {")],
    # verdict < -> <=
    "M7": [("if energy < -allowance {", "if energy <= -allowance {")],
    # drop the pre-loop validation
    "M8": [("    validate_prepared(prepared)?;\n", "")],
    # guard always true: today's O(n^4) search restored
    "M9": [("if energy < -allowance {", "if true || energy < -allowance {")],
    # the cell zero test removed
    "M10": [("""                if prepared.matrix[a][b] == 0.0 {
                    continue;
                }
""", "")],
}

if __name__ == "__main__":
    mutant, path = sys.argv[1], sys.argv[2]
    src = open(path).read()
    for old, new in MUTANTS[mutant]:
        src = edit(src, old, new)
    open(path, "w").write(src)
    print(mutant, "applied", len(MUTANTS[mutant]), "edit(s)")
