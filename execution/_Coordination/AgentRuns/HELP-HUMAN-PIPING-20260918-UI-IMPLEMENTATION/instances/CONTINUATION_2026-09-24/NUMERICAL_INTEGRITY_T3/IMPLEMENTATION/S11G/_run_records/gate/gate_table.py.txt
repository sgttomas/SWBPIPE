"""T3 S11-G (I5) gate table: the 14 formation rows through base and candidate probes
(the fixture-diff harness's `probe` mode), with the case's integrity code and guard, and
the leaves that differ between base and candidate. Standard library only.
Usage: gate_table.py <rf_cancel_cases.json> <fixdiff_base> <fixdiff_cand> <workdir>"""
import json, subprocess, sys, os
data = json.load(open(sys.argv[1])); base_bin, cand_bin, work = sys.argv[2:5]
cases = {c['id']: c for c in data['cases']}
def probe(binary, request_path, mode, entry):
    return subprocess.run([binary, 'probe', request_path, mode, entry], capture_output=True, text=True, check=True).stdout
def leaves(a, b, path=''):
    if isinstance(a, dict) and isinstance(b, dict):
        for k in sorted(set(a) | set(b)):
            if k not in a or k not in b: yield path + '/' + k
            else: yield from leaves(a[k], b[k], path + '/' + k)
    elif isinstance(a, list) and isinstance(b, list):
        if len(a) != len(b): yield path + '[len]'
        for i, (x, y) in enumerate(zip(a, b)): yield from leaves(x, y, '%s/%d' % (path, i))
    elif a != b: yield path
def integrity(env):
    for d in env['diagnostics']:
        if d['code'].startswith('NUMERICAL_INTEGRITY') and d.get('id', '').endswith(':case'):
            return d
    raise SystemExit('no integrity record')
print('| entry | case | row | mode | base code | candidate code | guard | differing leaves (base vs candidate) |')
print('|---|---|---|---|---|---|---|---|')
for entry, case, key, mode in data['formation_exceptions']:
    path = os.path.join(work, case + '.request.json')
    json.dump(cases[case]['request'], open(path, 'w'))
    b = json.loads(probe(base_bin, path, mode, entry)); c = json.loads(probe(cand_bin, path, mode, entry))
    ib, ic = integrity(b), integrity(c)
    guard = 'load-row' if 'S11-G formation-noise guard' in ic['message'] else ("R-b'" if "S11-G recovery guard (R-b')" in ic['message'] else 'none')
    diff = sorted(set(leaves(b, c)))
    print(f"| {entry} | {case} | {key} | {mode} | {ib['code']} | {ic['code']} | {guard} | {', '.join(diff)} |")
