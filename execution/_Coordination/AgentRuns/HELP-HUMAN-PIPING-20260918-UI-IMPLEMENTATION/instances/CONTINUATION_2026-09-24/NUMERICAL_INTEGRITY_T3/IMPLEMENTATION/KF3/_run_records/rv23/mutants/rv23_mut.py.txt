# RV23-1 and RV23-N1 mutants (T3 KF3, after RV23's review): each is one
# substitution on the patched clean copy of FK (<wt>/kf3-mut), restored after
# its run; one cargo job, debug, RUST_TEST_THREADS=2, filter `kf3`.
import json, os, re, shutil, subprocess, sys, time
W = sys.argv[1]
OUT = sys.argv[2]
FK = W + '/kf3-mut/projects/chirality-piping/core/solver/frame_kernel'
R = 'src/structural/retained/'
MUT = {
    'NONE': [],
    'RV23-1-M1-uc-refusals-dropped-in-verify-precision': [(R + 'adaptive.rs',
        '            if vs.is_err() {\n                record.bound_refusals = vs_refusals;\n            }\n',
        '            let _ = &vs_refusals;\n')],
    'RV23-1-M2-uc-refusals-dropped-in-build-verify-shared': [(R + 'verify.rs',
        '    let refusals = if result.is_err() {\n        uc_refusals(&uc_refused)\n    } else {',
        '    let refusals = if result.is_err() {\n        let _ = (&uc_refused, uc_refusals);\n        Vec::new()\n    } else {')],
    'RV23-1-M3-s-refusals-dropped-in-shift-schedule': [(R + 'bound.rs',
        '    if steps.is_err() {\n        for (b, r) in &results {',
        '    if false && steps.is_err() {\n        for (b, r) in &results {')],
    'RV23-1-M3a-kept-results-refusals-dropped': [(R + 'bound.rs',
        '        for (b, r) in &results {\n            if let Some(x) = r.refused {\n                record(&mut s_refused[*b], x.kind, x.pass, x.row);\n            }\n        }\n',
        '')],
    'RV23-1-M3b-in-flight-refusals-dropped': [(R + 'bound.rs',
        '        for &b in &in_flight {\n            if let Some(x) = refused[b] {\n                record(&mut s_refused[b], x.kind, x.pass, x.row);\n            }\n        }\n',
        '        let _ = &in_flight;\n')],
    'RV23-M4b-precedence-swapped-in-verify-state': [(R + 'verify.rs',
        '        if let Some((_, refusal)) = stop {',
        '        if let (Some((_, refusal)), None) = (stop, uc_missing) {')],
}
env = dict(os.environ, RUST_TEST_THREADS='2', RUSTUP_TOOLCHAIN='1.97.1',
           RUSTUP_AUTO_INSTALL='0', CARGO_INCREMENTAL='0')
results = []
for name, subs in MUT.items():
    saved = {}
    for f, a, b in subs:
        p = FK + '/' + f
        text = open(p, encoding='utf-8').read()
        assert text.count(a) == 1, (name, f)
        saved[p] = text
        open(p, 'w', encoding='utf-8').write(text.replace(a, b))
    t = time.time()
    r = subprocess.run(['cargo', 'test', '--offline', '--locked', '-j', '4', '--lib',
                        '--target-dir', W + '/kf3-mut-target', '--', 'kf3'],
                       cwd=FK, env=env, capture_output=True, text=True)
    for p, text in saved.items():
        open(p, 'w', encoding='utf-8').write(text)
    log = r.stdout + r.stderr
    open(OUT + '/' + name + '.log', 'w').write(log.replace(W, '<wt>'))
    m = re.search(r'test result: \w+\. (\d+) passed; (\d+) failed', log)
    failed = re.findall(r'^test (\S+) \.\.\. FAILED', log, re.M)
    rec = {'mutant': name, 'passed': int(m.group(1)) if m else None,
           'failed': int(m.group(2)) if m else None,
           'killed_by': [f.split('::')[-1] for f in failed],
           'compile_error': m is None, 'seconds': round(time.time() - t, 1)}
    results.append(rec)
    print(json.dumps(rec), flush=True)
open(OUT + '/mutants.jsonl', 'w').write(''.join(json.dumps(r) + '\n' for r in results))
