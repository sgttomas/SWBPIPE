#!/bin/bash
# K-D5 mutation runner, clean-build form (I3, completed by I3R; committed as run_mutants_clean.sh.txt).
# Usage: run_mutants_clean.sh <kd5-worktree> <scratch> <mutant>...   (mutate.py must be in <scratch>)
#
# RV5 found that the earlier runner (run_mutants.sh) could reuse a stale frame_kernel build: tar
# preserved the worktree's mtimes, so a mutant that did not touch frame_kernel (M32a, M32b) was
# tested against the previous mutant's frame_kernel artifact (M31b0's). This runner removes that:
#   - each mutant gets a fresh copy of P/core/{solver,loads,serialization,units,product_physics}
#     (no target dirs) extracted with `tar -m`, so every source file is newer than any artifact,
#     plus P/fixtures, P/schemas and the numerical_integrity fixtures.json that the crates
#     include_str! from outside core (I3R: the first clean attempt failed to compile PP without them);
#   - the shared target dir is removed before every mutant (and once more at the end), so nothing
#     can be reused across mutants;
#   - a control run (id NONE, no patch) must pass everything.
# Tests per mutant (each failing test is a killer; its panic site is listed):
#   frame_kernel          cargo test --offline --locked kd5
#   nonlinear_integration cargo test --offline --locked kd5
#   product_physics       cargo test --offline --locked --test formation_check_runtime
# Before each mutant it waits for idle cargo, DEC-025 sweeps and timed probe comparisons (the wait
# patterns anchor on the interpreter, so they never match this shell).
W=$1; X=$2; shift 2
export RUSTUP_TOOLCHAIN=1.97.1 CARGO_INCREMENTAL=0 RUSTUP_AUTO_INSTALL=0 CARGO_TARGET_DIR=$X/mut-target
for m in "$@"; do
  while pgrep -x cargo >/dev/null || pgrep -f 'python[0-9.]* .*(run_evidence_sweep|timing_compare)' >/dev/null; do sleep 60; done
  rm -rf $X/copy "$CARGO_TARGET_DIR"; mkdir -p $X/copy
  tar -C $W/projects/chirality-piping --exclude=target -cf - core/solver core/loads core/serialization core/units \
      core/product_physics fixtures schemas validation/benchmarks/numerical_integrity/fixtures.json \
    | tar -C $X/copy -xmf -
  if [ "$m" != NONE ]; then
    python3 -B $X/mutate.py $X/copy/core $m > $X/$m.log 2>&1 || { echo "$m PATCH-FAILED" >> $X/MUTANTS.txt; continue; }
  else
    echo "control: no patch" > $X/$m.log
  fi
  echo "disk_free_kb=$(df -k --output=avail $X | tail -1) start=$(date -u +%FT%TZ)" >> $X/$m.log
  rc_all=""
  for t in "solver/frame_kernel|kd5" "solver/nonlinear_integration|kd5" "product_physics|--test formation_check_runtime"; do
    dir=${t%%|*}; args=${t#*|}
    echo "=== $dir: cargo test $args" >> $X/$m.log
    (cd $X/copy/core/$dir && cargo test --offline --locked $args -- --test-threads=1 >> $X/$m.log 2>&1); rc_all="$rc_all $dir=$?"
  done
  echo "end=$(date -u +%FT%TZ)" >> $X/$m.log
  killers=$(grep -E '^test .* FAILED$' $X/$m.log | awk '{print $2}' | tr '\n' ' ')
  comp=$(grep -c '^error\[' $X/$m.log)
  sites=$(grep -oE "panicked at [^ ]+:[0-9]+:[0-9]+" $X/$m.log | sed 's/panicked at //' | sort | uniq -c | awk '{print $2"x"$1}' | tr '\n' ' ')
  echo "$m exit:$rc_all compile_errors=$comp killed_by: ${killers:-NONE} panic_sites: ${sites:-NONE}" >> $X/MUTANTS.txt
  sleep 30
done
rm -rf $X/copy "$CARGO_TARGET_DIR"
echo DONE >> $X/MUTANTS.txt
