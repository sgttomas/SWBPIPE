"""I14 (K5, checkpoint D): Q10's observation. Instruments a scratch copy of SA (never committed):
each curved slot entering `EvidenceParts::new` prints its count of exactly zero and of subnormal
global-matrix entries, and its smallest nonzero |entry|. Usage: python3 q10_instrument.py <core-dir>"""
import sys
p = sys.argv[1] + "/solver/nonlinear_integration/src/structural_adapter.rs"
s = open(p).read()
old = """            // Explicit slots have no silently inferred objective/nullspace contract.
            result.element(
                element.node_i,
                element.node_j,
                &element.global_stiffness,"""
new = """            {
                let (mut zeros, mut subnormal, mut least) = (0, 0, f64::INFINITY);
                for v in element.global_stiffness.iter().flatten() {
                    if *v == 0.0 {
                        zeros += 1;
                    } else {
                        if !v.is_normal() {
                            subnormal += 1;
                        }
                        least = least.min(v.abs());
                    }
                }
                eprintln!(
                    "K5Q10 test={} slot={} explicit={} zeros={} subnormal={} least_nonzero={:e}",
                    std::thread::current().name().unwrap_or("-"),
                    element.element_id,
                    element.symmetry_formation.is_none(),
                    zeros,
                    subnormal,
                    least
                );
            }
""" + old
assert s.count(old) == 1
open(p, "w").write(s.replace(old, new))
print("instrumented")
