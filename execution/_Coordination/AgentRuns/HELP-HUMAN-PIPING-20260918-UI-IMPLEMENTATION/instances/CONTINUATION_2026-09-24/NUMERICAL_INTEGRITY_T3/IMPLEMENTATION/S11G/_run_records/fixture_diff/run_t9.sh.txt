#!/bin/sh
# T3 S11-G (I5) T9: every committed request/model under fixtures, core and validation,
# through base and candidate, both modes, captured and typed entries. Sequential.
S=<scratch>
W=<wt>/s11g/projects/chirality-piping
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=<s11g-target>
A=$W/core/serialization/canonical_json/target/checked-json/release
export OPENPIPESTRESS_CHECKED_JSON_BIN=$A/openpipestress_jcs_ijson OPENPIPESTRESS_BINARY64_JSON_BIN=$A/openpipestress_jcs_binary64
export OPENPIPESTRESS_UNITS_BIN=$W/core/units/target/units-authority/release/openpipestress_units
for tree in base cand; do
  df -h / | tail -1 > $S/t9/$tree.disk_before
  P=$S/$tree/projects/chirality-piping
  (cd $P/core/product_physics && nice cargo build --offline --locked --release --example fixdiff_s11g) > $S/t9/$tree.build.log 2>&1 || { echo "$tree build failed" >> $S/t9/status; exit 1; }
  cp <s11g-target>/release/examples/fixdiff_s11g $S/t9/fixdiff_$tree
  for root in fixtures core validation; do
    mkdir -p $S/t9/$tree/$root
    (cd $P && $S/t9/fixdiff_$tree $root $S/t9/$tree/$root) > $S/t9/$tree.$root.log 2>&1
    echo "$tree $root exit=$? $(tail -1 $S/t9/$tree.$root.log)" >> $S/t9/status
  done
  df -h / | tail -1 > $S/t9/$tree.disk_after
  rm -rf <s11g-target>/release
done
echo DONE >> $S/t9/status
