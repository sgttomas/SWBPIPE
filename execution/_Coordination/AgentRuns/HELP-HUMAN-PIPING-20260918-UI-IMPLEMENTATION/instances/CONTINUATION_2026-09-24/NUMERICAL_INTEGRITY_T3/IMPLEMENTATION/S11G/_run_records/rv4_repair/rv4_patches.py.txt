"""T3 S11-G (I5) repair: apply each of RV4's review patches (GNU patch -p1 -F3, no Git), run
the repair's killing tests, restore the file bytes and verify the sha256.
Usage (from the worktree root, cargo environment exported): rv4_patches.py <patch_dir> <out_dir>"""
import hashlib, json, os, re, subprocess, sys
PP = 'projects/chirality-piping/core/product_physics'
KILLERS = {
    'RV-M1': ['d21_1_second_test_adds_the_bound_exactly'],
    'RV-M2': ['exact_pressure_operand_bound_is_gamma_20', 'exact_pressure_operand_rows_carry_their_bound'],
    'RV-M3': ['curved_thermal_records_name_the_pushed_products'],
    'RV-M6': ['e1_ordinary_err_guard_fired_case_keeps_mains_attempt'],
    'RV-M10': ['restrained_rows_take_the_all_rows_scale'],
}
patch_dir, out = sys.argv[1], sys.argv[2]
os.makedirs(out, exist_ok=True)
results = {}
sha = lambda p: hashlib.sha256(open(p, 'rb').read()).hexdigest()
for name, tests in KILLERS.items():
    patch = os.path.join(patch_dir, name + '.patch')
    target = re.search(r'^\+\+\+ b/(\S+)', open(patch).read(), re.M).group(1)
    saved, digest = open(target, 'rb').read(), sha(target)
    try:
        applied = subprocess.run(['patch', '-p1', '-F3', '-i', patch], capture_output=True, text=True)
        assert applied.returncode == 0, applied.stdout + applied.stderr
        command = ['cargo', 'test', '--offline', '--locked', '--manifest-path', PP + '/Cargo.toml', '--lib',
                   '--', '--exact'] + ['s11g_tests::' + t for t in tests]
        log = os.path.join(out, name + '.log')
        with open(log, 'w') as f:
            f.write('$ patch -p1 -F3 -i ' + os.path.basename(patch) + '\n' + applied.stdout + '$ ' + ' '.join(command) + '\n')
            f.flush()
            code = subprocess.run(command, stdout=f, stderr=subprocess.STDOUT).returncode
        text = open(log).read()
        failed = sorted(set(re.findall(r'^test (\S+) \.\.\. FAILED', text, re.M)))
        verdict = 'COMPILE_ERROR' if 'could not compile' in text else ('KILLED' if code else 'SURVIVED')
        results[name] = {'patch_sha256': sha(patch), 'file': target, 'exit': code, 'failing_tests': failed, 'verdict': verdict}
    finally:
        open(target, 'wb').write(saved)
        assert sha(target) == digest, target + ' not restored'
    print(name, results[name]['verdict'], results[name]['failing_tests'], flush=True)
json.dump(results, open(os.path.join(out, 'results.json'), 'w'), indent=1, sort_keys=True)
