"""I9 (T3): exact references and the standing table for the product evidence
(the skew probe `i9_skew_product_probe.rs` on git archives of main 134eefc24 and
eb52114e9). Standard-library Python only.

The exact reference takes the model's represented inputs as exact: E, G and the
moment (the probe's binary64 values), A, I and J as PP derive_pipe_section forms
them, the member length and local axes as FK forms them (all printed by the probe
as bits); the global stiffness T^T K T and the free-node solve are exact rational
arithmetic (fractions.Fraction). Published translations are in mm, rotations in rad.
The criterion is the protected relative form |obs - exp| <= 1e-9 * max(|exp|, scale),
with scale = the largest |exp| of the same kind (translations, or rotations) in the run.

Usage: python3 -B product_reference.py <probe log> [<probe log> ...]
"""
import json, struct, sys, collections
from fractions import Fraction as F

def f64(b):
    return struct.unpack("<d", struct.pack("<Q", b))[0]

DOFS = ["ux", "uy", "uz", "rx", "ry", "rz"]

def local_node_j_block(c):
    ax, to, y12, y6, y4, y2, z12, z6, z4, z2 = c
    K = [[F(0)] * 6 for _ in range(6)]
    K[0][0] = ax
    K[3][3] = to
    K[1][1] = z12; K[1][5] = -z6; K[5][1] = -z6; K[5][5] = z4
    K[2][2] = y12; K[2][4] = y6; K[4][2] = y6; K[4][4] = y4
    return K

def solve(A, b):
    n = len(A)
    M = [row[:] + [b[i]] for i, row in enumerate(A)]
    for col in range(n):
        piv = next(r for r in range(col, n) if M[r][col] != 0)
        M[col], M[piv] = M[piv], M[col]
        for r in range(n):
            if r != col and M[r][col] != 0:
                fac = M[r][col] / M[col][col]
                M[r] = [M[r][k] - fac * M[col][k] for k in range(n + 1)]
    return [M[i][n] / M[i][i] for i in range(n)]

def main_binary64_coefficients(section, geom, E, G):
    """Main's pre-K2a coefficients, operation for operation, in binary64."""
    a, i, j = (f64(section[k]) for k in ("A", "I", "J"))
    L = f64(geom["length"]); e, g = f64(E), f64(G)
    L2 = L * L; L3 = L2 * L
    return [F(x) for x in (e * a / L, g * j / L, 12.0 * e * i / L3, 6.0 * e * i / L2, 4.0 * e * i / L, 2.0 * e * i / L,
                           12.0 * e * i / L3, 6.0 * e * i / L2, 4.0 * e * i / L, 2.0 * e * i / L)]

def exact_n1(section, geom, E, G, moment, formed=False):
    A, I, J = (F(f64(section[k])) for k in ("A", "I", "J"))
    Lm = F(f64(geom["length"]))
    R = [[F(f64(v)) for v in row] for row in geom["axes"]]
    if formed:
        c = main_binary64_coefficients(section, geom, E, G)
    else:
        E, G = F(f64(E)), F(f64(G))
        c = [E * A / Lm, G * J / Lm,
             12 * E * I / Lm**3, 6 * E * I / Lm**2, 4 * E * I / Lm, 2 * E * I / Lm,
             12 * E * I / Lm**3, 6 * E * I / Lm**2, 4 * E * I / Lm, 2 * E * I / Lm]
    if any(x == 0 for x in c):
        return None
    K = local_node_j_block(c)
    # T = blockdiag(R, R) for node j; global block = T^T K T
    T = [[F(0)] * 6 for _ in range(6)]
    for b in range(2):
        for r in range(3):
            for q in range(3):
                T[3 * b + r][3 * b + q] = R[r][q]
    KT = [[sum(K[i][k] * T[k][j] for k in range(6)) for j in range(6)] for i in range(6)]
    Kg = [[sum(T[k][i] * KT[k][j] for k in range(6)) for j in range(6)] for i in range(6)]
    f = [F(0)] * 6
    f[5] = F(f64(moment))
    return solve(Kg, f)

def criterion(obs, exp):
    ratios = []
    for kind in (DOFS[:3], DOFS[3:]):
        scale = max(abs(exp[d]) for d in kind)
        for d in kind:
            denom = max(abs(exp[d]), scale)
            ratios.append(abs(obs[d] - exp[d]) / denom if denom else F(0))
    return float(max(ratios))

def main(paths):
    worst_trusted = None
    out = open("product_runs.jsonl", "w")
    for path in paths:
        section, geoms, runs = None, {}, []
        for line in open(path):
            if line.startswith("I9SECTION "):
                section = json.loads(line[10:])
            elif line.startswith("I9GEOM "):
                g = json.loads(line[7:]); geoms[g["member"]] = g
            elif line.startswith("I9PROBE "):
                runs.append(json.loads(line[8:]))
        print(f"== {path.split('/')[-1]}: {len(runs)} runs")
        cache = {}
        table = collections.OrderedDict()
        for r in runs:
            key = (r["t"], r["g_label"], r["member"])
            if key not in cache:
                cache[key] = exact_n1(section, geoms[r["member"]], r["E"], r["G"], r["moment"])
            u = cache[key]
            err = formation_err = None
            record = {"tree": path.split("/")[-1], "t": r["t"], "G": r["g_label"], "E_bits": r["E"], "G_bits": r["G"],
                      "moment_bits": r["moment"], "member": r["member"], "mode": r["mode"], "entry": r["entry"],
                      "status": r["status"], "cases": r["cases"], "codes": r["codes"],
                      "messages": [m[:240] for m in r["messages"]]}
            if r["N1"]:
                obs = {row[0].split(":")[-1]: F(row[1]) for row in r["N1"]}
                units = {row[0].split(":")[-1]: row[2] for row in r["N1"]}
                scale = {d: (1000 if units[d] == "mm" else 1) for d in DOFS}
                for k, d in enumerate(DOFS):
                    assert units[d] == ("mm" if k < 3 else "rad"), (d, units[d])
                exp = {d: u[k] * scale[d] for k, d in enumerate(DOFS)}
                err = criterion(obs, exp)
                formed_key = key + ("formed",)
                if formed_key not in cache:
                    cache[formed_key] = exact_n1(section, geoms[r["member"]], r["E"], r["G"], r["moment"], formed=True)
                uf = cache[formed_key]
                formation_err = criterion({d: uf[k] * scale[d] for k, d in enumerate(DOFS)}, exp)
                record.update({"published": {d: [str(float(obs[d])), units[d]] for d in DOFS},
                               "exact": {d: [repr(float(exp[d])), units[d]] for d in DOFS},
                               "criterion_error": err, "formation_only_error": formation_err})
            out.write(json.dumps(record, sort_keys=True) + "\n")
            standing = r["status"]
            first = r["messages"][0].split(":", 1)[0] if r["messages"] else "-"
            reason = ""
            for m in r["messages"]:
                if "Range(" in m or "NumericallyUnresolved" in m or "stiffness formation" in m:
                    reason = m.split("structural integrity: ")[-1][:95] if "structural integrity" in m else m.split(": ", 1)[1][:110]
                    break
            if standing == "sensitive":
                reason = "Sensitive (published)"
            cell = table.setdefault(key, [])
            cell.append((r["mode"], r["entry"], standing, reason, err, formation_err))
            if standing == "checks_passed" and err is not None and err > 1e-9:
                worst_trusted = max(worst_trusted or 0, err)
                print(f"   STOP-CANDIDATE trusted and wrong: {key} {r['mode']} {r['entry']} err {err:.3e}")
        # the standing table: one line per (t, G, member), with the four runs' agreement
        for key, cell in table.items():
            standings = sorted({c[2] for c in cell})
            reasons = sorted({c[3] for c in cell})
            errs = [c[4] for c in cell if c[4] is not None]
            ferrs = [c[5] for c in cell if c[5] is not None]
            e = (f"max err {max(errs):.2e} (>1e-9: {sum(x > 1e-9 for x in errs)}/{len(errs)}); exact solve of main's formed coefficients: {max(ferrs):.2e}"
                 if errs else "no values published")
            modes = ";".join(f"{c[0][:6]}/{c[1][:4]}={c[2]}" for c in cell) if len(standings) > 1 or len(reasons) > 1 else f"all 4 runs {standings[0]}"
            print(f"   t={key[0]} G={key[1]:<8} {key[2]:<8} {modes}; {' | '.join(reasons)}; {e}")
    print("\nSTOP RULE (a trusted value wrong by more than 1e-9 on either tree):", "TRIGGERED" if worst_trusted else "not triggered")

if __name__ == "__main__":
    main(sys.argv[1:])
