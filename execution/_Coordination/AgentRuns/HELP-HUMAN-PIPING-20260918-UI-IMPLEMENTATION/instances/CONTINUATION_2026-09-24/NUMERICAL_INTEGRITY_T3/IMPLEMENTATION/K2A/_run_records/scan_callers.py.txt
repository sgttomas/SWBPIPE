#!/usr/bin/env python3
"""K2a caller enumeration (I6): lexer scan of every .rs file under projects/chirality-piping
(outside execution/, target/ and node_modules/), with comments and string literals removed and
#[cfg(test)] items separated. Standard library only.

It lists:
  A. every call of a function that forms, or reaches the formation of, binary64 frame stiffness
     (`local_stiffness(`, `global_stiffness(`, `assemble_global_stiffness(`,
     `assemble_global_stiffness_with_user_elements(`), with its receiver text so that the
     frame-kernel function can be told apart from the same-named methods of other types;
  B. every `match` expression one of whose arm patterns names `FrameKernelError::`, with its arm
     count and whether it has a catch-all arm (`_ =>` or a bare binding), i.e. whether a new
     variant breaks it;
  C. every other mention of `FrameKernelError::<Variant>` (constructions, `matches!`, `if let`,
     test patterns), and every wrapper variant that carries a FrameKernelError.

The lexer functions `lex`, `test_mask` and `enclosing` are copied unchanged from K-D5's scanner
(`T3/IMPLEMENTATION/KD5/_run_records/repair/callers/scan_callers.py.txt`, sha256 7d9ecc3d...).

Usage: python3 -B scan_callers.py <repo_root>
Each line: <section> <path>:<line> [<enclosing fn>] <test|non-test> <pattern> :: <source line>
(A file named *tests.rs is a #[cfg(test)] module file in every crate here, or is under tests/.)
"""
import os
import re
import sys

CALLS = ["local_stiffness(", "global_stiffness(", "assemble_global_stiffness(",
         "assemble_global_stiffness_with_user_elements("]
WRAPPERS = re.compile(r"\b(Kernel|FrameKernel)\s*\(\s*(FrameKernelError|error|e|err|source)\b")


def lex(src):
    out, i, n = [], 0, len(src)
    while i < n:
        c = src[i]
        if src.startswith("//", i):
            j = src.find("\n", i)
            i = n if j < 0 else j
            continue
        if src.startswith("/*", i):
            depth, i = 1, i + 2
            while i < n and depth:
                if src.startswith("/*", i):
                    depth, i = depth + 1, i + 2
                elif src.startswith("*/", i):
                    depth, i = depth - 1, i + 2
                else:
                    if src[i] == "\n":
                        out.append("\n")
                    i += 1
            continue
        m = re.match(r'b?r(#*)"', src[i:]) if (c in "rb" and (i == 0 or not (src[i - 1].isalnum() or src[i - 1] == "_"))) else None
        if m:
            end = src.find('"' + m.group(1), i + m.end())
            end = n if end < 0 else end + 1 + len(m.group(1))
            out.append('""' + "\n" * src[i:end].count("\n"))
            i = end
            continue
        if c == '"':
            j = i + 1
            while j < n and src[j] != '"':
                j += 2 if src[j] == "\\" else 1
            out.append('""' + "\n" * src[i:j].count("\n"))
            i = j + 1
            continue
        if c == "'" and i + 2 < n and src[i + 1] != "\\" and src[i + 2] == "'":
            out.append("' '")
            i += 3
            continue
        out.append(c)
        i += 1
    return "".join(out)


def test_mask(code):
    """True per character inside a #[cfg(test)] item."""
    mask = [False] * len(code)
    for m in re.finditer(r"#\[cfg\(test\)\]", code):
        depth, started, k = 0, False, m.end()
        while k < len(code):
            ch = code[k]
            if ch == "{":
                depth, started = depth + 1, True
            elif ch == "}":
                depth -= 1
                if started and depth == 0:
                    break
            elif ch == ";" and not started:
                break
            k += 1
        for q in range(m.start(), min(k + 1, len(code))):
            mask[q] = True
    return mask


def enclosing(code, pos):
    best = None
    for m in re.finditer(r"\bfn\s+([A-Za-z_][A-Za-z0-9_]*)", code[:pos]):
        best = m.group(1)
    return best or "<module>"



def receiver(code, k):
    """The receiver expression text before a method call (`x.local_stiffness(`), or `<free>`."""
    j = k - 1
    while j >= 0 and code[j] in " \t\n":
        j -= 1
    if j < 0 or code[j] != ".":
        if code[max(0, k - 2):k] == "::":
            m = re.search(r"([A-Za-z_][A-Za-z0-9_:]*)::$", code[:k])
            return (m.group(1) + "::") if m else "::"
        return "<free>"
    end, depth, i = j, 0, j - 1
    while i >= 0:
        ch = code[i]
        if ch in ")]":
            depth += 1
        elif ch in "([":
            if depth == 0:
                break
            depth -= 1
        elif depth == 0 and not (ch.isalnum() or ch in "_.?:"):
            break
        i -= 1
    return re.sub(r"\s+", "", code[i + 1:end]) or "?"


def self_impl_spans(code):
    """Spans of `impl <Trait> for FrameKernelError { ... }` blocks, where arms say `Self::`."""
    spans = []
    for m in re.finditer(r"\bimpl\s+[A-Za-z_][A-Za-z0-9_:<>]*\s+for\s+FrameKernelError\s*\{", code):
        depth, e = 0, m.end() - 1
        while e < len(code):
            if code[e] == "{":
                depth += 1
            elif code[e] == "}":
                depth -= 1
                if depth == 0:
                    break
            e += 1
        spans.append((m.start(), e))
    return spans


def match_blocks(code):
    """Yield (start, arms, has_catch_all, variants) for every `match` whose arms name FrameKernelError::
    (or `Self::` inside an `impl ... for FrameKernelError` block)."""
    spans = self_impl_spans(code)
    for m in re.finditer(r"\bmatch\b", code):
        k = m.end()
        depth = 0
        while k < len(code):
            if code[k] in "([":
                depth += 1
            elif code[k] in ")]":
                depth -= 1
            elif code[k] == "{" and depth == 0:
                break
            k += 1
        if k >= len(code):
            continue
        body_start, depth, e = k + 1, 1, k + 1
        while e < len(code) and depth:
            if code[e] == "{":
                depth += 1
            elif code[e] == "}":
                depth -= 1
            e += 1
        body = code[body_start:e - 1]
        # split arms at top level: an arm head is text up to a top-level "=>"
        heads, depth, i, cur = [], 0, 0, 0
        seg_start = 0
        while i < len(body):
            ch = body[i]
            if ch in "([{":
                depth += 1
            elif ch in ")]}":
                depth -= 1
                if depth == 0 and ch == "}":
                    # an arm whose body is a block ends here; next arm starts after optional comma
                    pass
            elif body.startswith("=>", i) and depth == 0:
                heads.append(body[seg_start:i].strip())
                # skip the arm body: until a top-level comma, or a closing block at depth 0
                j = i + 2
                d = 0
                while j < len(body):
                    c = body[j]
                    if c in "([{":
                        d += 1
                    elif c in ")]}":
                        d -= 1
                        if d == 0 and c == "}":
                            k2 = j + 1
                            while k2 < len(body) and body[k2] in " \t\n":
                                k2 += 1
                            if k2 < len(body) and body[k2] == ",":
                                k2 += 1
                            # a block arm body ends here only if the block started the arm body
                            if body[i + 2:j + 1].strip().startswith("{"):
                                j = k2
                                break
                    elif c == "," and d == 0:
                        j += 1
                        break
                    j += 1
                seg_start = j
                i = j
                continue
            i += 1
        in_self = any(a <= m.start() <= b for a, b in spans)
        prefix = "Self::" if in_self else "FrameKernelError::"
        if not any(prefix in h for h in heads):
            continue
        catch_all = any(re.fullmatch(r"(_|[a-z_][a-z0-9_]*)(\s+if\b.*)?", h, re.S) for h in heads)
        variants = sorted({v for h in heads for v in re.findall(re.escape(prefix) + r"([A-Za-z]+)", h)})
        yield m.start(), len(heads), catch_all, variants


def main():
    root = sys.argv[1]
    base = os.path.join(root, "projects", "chirality-piping")
    rows = {"A": [], "B": [], "C": []}
    for dp, dn, fn in os.walk(base):
        dn[:] = sorted(d for d in dn if d not in ("execution", "target", "node_modules", ".git"))
        for f in sorted(fn):
            if not f.endswith(".rs"):
                continue
            path = os.path.join(dp, f)
            src = open(path, encoding="utf-8").read()
            code = lex(src)
            mask = test_mask(code)
            src_lines = src.split("\n")
            rel = os.path.relpath(path, base)
            test_file = f.endswith("tests.rs") or "/tests/" in "/" + rel + "/"

            def row(sec, k, pat):
                line = code.count("\n", 0, k) + 1
                kind = "test" if (mask[k] or test_file) else "non-test"
                rows[sec].append(f"{sec} {rel}:{line} [{enclosing(code, k)}] {kind} {pat} :: {src_lines[line - 1].strip()}")

            for pat in CALLS:
                start = 0
                while True:
                    k = code.find(pat, start)
                    if k < 0:
                        break
                    start = k + 1
                    prev = code[k - 1] if k else " "
                    if prev.isalnum() or prev == "_":
                        continue
                    if re.search(r"\bfn\s*$", code[max(0, k - 20):k]):
                        row("A", k, "definition " + pat)
                        continue
                    row("A", k, pat + " receiver=" + receiver(code, k))
            in_match = set()
            for start, arms, catch_all, variants in match_blocks(code):
                row("B", start, f"match arms={arms} catch_all={'yes' if catch_all else 'NO'} variants={','.join(variants)}")
                # mark the FrameKernelError mentions inside this match so section C skips them
                depth, k = 0, code.find("{", start)
                e = k
                while e < len(code):
                    if code[e] == "{":
                        depth += 1
                    elif code[e] == "}":
                        depth -= 1
                        if depth == 0:
                            break
                    e += 1
                for mm in re.finditer(r"FrameKernelError::", code[k:e]):
                    in_match.add(k + mm.start())
            for mm in re.finditer(r"FrameKernelError::([A-Za-z]+)", code):
                if mm.start() in in_match:
                    continue
                row("C", mm.start(), "FrameKernelError::" + mm.group(1))
            for mm in WRAPPERS.finditer(code):
                row("C", mm.start(), "wrapper " + mm.group(0))
    for sec in "ABC":
        print("\n".join(rows[sec]))


main()
