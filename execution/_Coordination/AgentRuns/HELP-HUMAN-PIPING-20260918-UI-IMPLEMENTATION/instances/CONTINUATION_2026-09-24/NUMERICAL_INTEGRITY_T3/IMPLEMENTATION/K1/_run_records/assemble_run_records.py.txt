#!/usr/bin/env python3
"""K1 records (I8R, checkpoint D): copy the run evidence into _run_records/, sanitized.
- Machine paths become placeholders: <scratch> (the TASK's scratch directory), <wt> (the T3
  worktrees root), <home>.
- In a log, a line longer than 600 characters (Debug renderings of whole documents in failure output) is cut
  to 600 characters, followed by a marker naming the number of characters removed and the sha256
  of the full line, so the cut is visible and the original line is identifiable.
Usage: assemble.py <wt> <scratch> <out-dir>"""
import hashlib, os, re, shutil, sys
wt, scratch, out = sys.argv[1], sys.argv[2], sys.argv[3]
home = os.path.expanduser("~")
LIMIT = 600
def clean(text, cut):
    text = text.replace(scratch, "<scratch>").replace(wt, "<wt>").replace(home, "<home>")
    lines = []
    for line in text.split("\n"):
        if cut and len(line) > LIMIT:
            digest = hashlib.sha256(line.encode()).hexdigest()
            line = line[:LIMIT] + f" [... K1-records cut: {len(line) - LIMIT} more characters; sha256 of the full line {digest}]"
        lines.append(line)
    return "\n".join(lines)
# Summary files, which the table and RETURN quote, are never cut.
UNCUT = ("MUTANTS.txt", "kill_sites.txt", "suites_compare.txt", "suites_compare.json", "callers.txt")
def put(src, rel):
    dst = os.path.join(out, rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    with open(src, encoding="utf-8", errors="replace") as f:
        text = f.read()
    with open(dst, "w", encoding="utf-8") as f:
        f.write(clean(text, not rel.endswith(UNCUT)))
L = os.path.join(scratch, "logs")
# checkpoint A: the compile check and the targeted tests, in run order
for name in ["check_fk_1.log", "check_fk_2.log", "check_sparse_direct_1.log", "check_nonlinear_integration_1.log",
             "test_fk_k1_1.log", "test_sd_k1_1.log", "test_ni_k1_1.log", "test_fk_k1_2.log", "test_fk_k1_3.log",
             "test_fk_k1_4.log", "test_fk_k1_5.log", "test_sd_k1_2.log", "test_ni_pins_1.log",
             "test_fk_sitetable_1.log", "test_pp_s11f_1.log", "test_fk_full_1.log",
             "test_fk_k1_final.log", "test_sd_k1_final.log", "test_ni_k1_pins_final.log"]:
    put(os.path.join(L, name), f"checkpoint_a/{name}")
# checkpoint B: suites
put(os.path.join(scratch, "run_suites_k1.sh"), "suites/run_suites_k1.sh.txt")
put(os.path.join(scratch, "suites_k1.log"), "suites/suites_k1.log")
put(os.path.join(scratch, "compare_suites.py"), "suites/compare_suites.py.txt")
put(os.path.join(scratch, "suites_compare.txt"), "suites/suites_compare.txt")
put(os.path.join(scratch, "suites_compare.json"), "suites/suites_compare.json")
for name in sorted(os.listdir(os.path.join(scratch, "suites_k1"))):
    put(os.path.join(scratch, "suites_k1", name), f"suites/logs/{name}")
with open(os.path.join(out, "suites/failure_blocks_sha256.txt"), "w") as f:
    f.write("Failure output of the two crates with Mac platform failures, from the first '---- ' line to\n"
            "'failures:', thread ids removed; base = ROOT's Mac main no-fail-fast logs, k1 = this run.\n")
    for n in ["product_physics", "runner_headless"]:
        for side in ["base", "k1"]:
            p = os.path.join(scratch, f"fail_{side}_{n}.txt")
            data = open(p, "rb").read()
            f.write(f"{hashlib.sha256(data).hexdigest()}  {len(data)} bytes  {side} {n}\n")
# checkpoint B: T9 (Mac-only)
T = os.path.join(scratch, "t9")
for v in ["base", "cand"]:
    put(os.path.join(T, f"h_{v}", "Cargo.toml"), f"t9/harness_{v}_Cargo.toml.txt")
    put(os.path.join(T, f"build_{v}.log"), f"t9/build_{v}.log")
    put(os.path.join(T, f"run_{v}.log"), f"t9/run_{v}.log")
    put(os.path.join(T, f"sha_{v}.txt"), f"t9/output_sha256_{v}.txt")
put(os.path.join(T, "t9_summary.txt"), "t9/t9_summary.txt")
# checkpoint C: mutations
M = os.path.join(scratch, "mut")
put(os.path.join(M, "run_mutant.sh"), "mutations/run_mutant.sh.txt")
put(os.path.join(M, "mutate_k1.py"), "mutations/mutate_k1.py.txt")
put(os.path.join(M, "kill_sites.py"), "mutations/kill_sites.py.txt")
put(os.path.join(M, "MUTANTS.txt"), "mutations/MUTANTS.txt")
put(os.path.join(M, "kill_sites.txt"), "mutations/kill_sites.txt")
for name in sorted(os.listdir(os.path.join(M, "logs"))):
    put(os.path.join(M, "logs", name), f"mutations/logs/{name}")
# callers
R = os.path.join(scratch, "records")
put(os.path.join(R, "callers.txt"), "callers.txt")
put(os.path.join(R, "scan_callers_k1.py"), "scan_callers_k1.py.txt")
put(os.path.join(R, "storage_counts.txt"), "storage_counts.txt")
put(os.path.join(R, "toolchain.txt"), "toolchain.txt")
put(os.path.join(R, "assemble.py"), "assemble_run_records.py.txt")
print("done")
