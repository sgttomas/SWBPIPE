"""K2a (I6): 6EI/L^2 probe (reach_six/). The only way 6EI/L^2 alone can matter: N1 UY restrained (so
12EI/L^3 moves no displacement), N1 RZ free with a ground spring k and a moment M. (6*E)*I rounds to 0, so
main's 6EI/L^2 = 0 and the restrained-UY reactions it carries (6EI/L^2 * theta) are published as 0; theta
itself uses 4EI/L (also 0, true <= 2^-1035) and k. Usage: python3 -B six_probe.py <reach_requests.py> <outdir>"""
import math, json, sys
from fractions import Fraction as Fr
ns = {}; exec(open(sys.argv[1]).read().split('out = sys.argv[1]')[0], ns)
out = sys.argv[2]
L = 2.0 ** -39; od = 1e-11; t = 1e-12; G = 1e-100
idd = od - 2 * t
i = math.pi * ((od * od) * (od * od) - (idd * idd) * (idd * idd)) / 64
E = 6.4e-280   # (12E)I = 2.23e-324 -> 0, (6E)I -> 0
six_true = 6 * Fr(E) * Fr(i) / Fr(L) ** 2
four_true = 4 * Fr(E) * Fr(i) / Fr(L)
for tag, k in (('k-floor', 2.0 ** -970), ('k-small', 2.0 ** -1000), ('k-tiny', 2.0 ** -1015)):
    M = float('%.3g' % (k * 0.25))
    theta_true = Fr(M) / (Fr(k) + four_true)
    R_true = six_true * theta_true
    print(tag, 'k', k, 'M', M, '6EI/L2 main', 6.0 * E * i / (L * L), 'true 2^%.1f' % math.log2(float(six_true)),
          'theta', float(theta_true), 'R_true', float(R_true), 'R_true*L/M 2^%.1f' % math.log2(float(R_true * Fr(L) / Fr(M))))
    json.dump(ns['request']('SIX-' + tag, L, od, t, E, G, 'RZ', k, M, False), open('%s/SIX-%s.json' % (out, tag), 'w'))
