"""K6b A0: the visibility edits, marker removals and the facade (exact, line-addressed, verified)."""
import re, sys, pathlib

R = FK / 'structural/retained'
FIELD = re.compile(r'^\s*pub\(crate\) [a-z_][a-z0-9_]*:')
# (file, [item lines], [method lines], [(field range lo, hi)], [marker lines])
PLAN = {

 'source.rs': (

   [39,48,52,69,73,353,570,573,576,579,582,585,588,591,594,597,601,605,610,613,617,622,628,663],
   [(64,65),(83,92),(98,100),(123,127),(133,134),(140,142),(148,150),(157,161),(167,174)],
   [352]),
 'recover.rs': ([30,46,53,80,101], [], [(81,85)], []),
 'factor.rs': ([60,290], [], [], []),
 'ledger.rs': ([72], [], [], []),
 'combine.rs': ([45,68,77], [82], [], [66,81]),
 'verify.rs': ([51,321,335,360], [], [], []),
 'wide.rs': ([150], [], [], []),
 'wide/multi.rs': ([666,1002,1074], [681,1107], [], [680,1106]),
 'wide_sum.rs': ([77], [92], [], []),
}
totals = {'items': 0, 'methods': 0, 'fields': 0, 'markers': 0}
log = []
for name, (items, methods, ranges, markers) in PLAN.items():
    path = R / name
    lines = path.read_text().split('\n')
    def raise_(n, kind):
        s = lines[n - 1]
        assert s.count('pub(crate) ') == 1, (name, n, s)
        lines[n - 1] = s.replace('pub(crate) ', 'pub ', 1)
        totals[kind] += 1
        log.append('%s:%d %s %s' % (name, n, kind, lines[n - 1].strip()[:90]))
    for n in items:
        s = lines[n - 1].lstrip()
        assert re.match(r'pub\(crate\) (const|fn|struct|enum) ', s), (name, n, s)
        raise_(n, 'items')
    for n in methods:
        s = lines[n - 1].lstrip()
        assert re.match(r'pub\(crate\) (const|fn) ', s), (name, n, s)
        raise_(n, 'methods')
    for lo, hi in ranges:
        for n in range(lo, hi + 1):
            if FIELD.match(lines[n - 1]):
                raise_(n, 'fields')
    for n in sorted(markers, reverse=True):
        s = lines[n - 1].strip()
        assert s.startswith('#[allow(dead_code)]'), (name, n, s)
        log.append('%s:%d marker-removed %s' % (name, n, s))
        del lines[n - 1]
        totals['markers'] += 1
    path.write_text('\n'.join(lines))
