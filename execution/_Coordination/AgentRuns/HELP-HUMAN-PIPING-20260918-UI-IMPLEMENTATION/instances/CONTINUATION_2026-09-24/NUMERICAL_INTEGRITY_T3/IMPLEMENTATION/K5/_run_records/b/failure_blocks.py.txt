"""I14 (K5, checkpoint B): the failure blocks of a suites run, for comparison with the baseline.
A block is cargo test's `---- <test> stdout ----` section, up to the next section or the
`failures:` list. Normalized only for what differs by run: thread ids (`'name' (12345)`) and the
tree's absolute path prefix (everything before `projects/chirality-piping/`).
Usage: python3 -B failure_blocks.py <suites-dir> <out-file>"""
import os
import re
import sys

d, out = sys.argv[1], sys.argv[2]
blocks = []
for name in sorted(os.listdir(d)):
    if not name.endswith('.log') or name[:3].isdigit() is False:
        continue
    text = open(os.path.join(d, name), errors='replace').read()
    for m in re.finditer(r'^---- (\S+) stdout ----\n(.*?)(?=^---- |^failures:$)', text, re.S | re.M):
        body = re.sub(r"(thread '[^']*') \(\d+\)", r'\1 (N)', m.group(2))
        body = re.sub(r'\S*/(projects/chirality-piping/)', r'<tree>/\1', body)
        blocks.append('== %s :: %s\n%s' % (name[4:], m.group(1), body.rstrip()))
with open(out, 'w') as fh:
    fh.write('\n'.join(blocks) + '\n')
print(len(blocks), 'failure blocks')
