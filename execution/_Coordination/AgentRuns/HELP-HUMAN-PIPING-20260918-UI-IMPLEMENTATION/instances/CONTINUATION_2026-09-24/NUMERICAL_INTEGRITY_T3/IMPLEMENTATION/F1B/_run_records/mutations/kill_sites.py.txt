#!/usr/bin/env python3
"""K1 checkpoint C (I8R): per mutant, each failing test with its panic site and the first line of
its message, the crates whose build failed, and the crates' exits. Usage: kill_sites.py <logs-dir> <id>..."""
import re, sys, os
logs = sys.argv[1]
for mid in sys.argv[2:]:
    text = open(os.path.join(logs, f"{mid}.log"), errors="replace").read()
    print(f"## {mid}")
    crate = None; out = []
    blocks = re.split(r"^(=== .*)$", text, flags=re.M)
    for part in blocks:
        if part.startswith("=== "):
            crate = part.split(":")[0][4:]; continue
        if crate is None: continue
        if re.search(r"^error(\[E\d+\])?: ", part, re.M):
            errs = sorted(set(re.findall(r"^error(?:\[E\d+\])?: (.*)$", part, re.M)))
            print(f"   BUILD ERROR in {crate}: {errs[:3]}")
        for m in re.finditer(r"^---- (\S+) stdout ----\n(.*?)(?=^---- |\Z|^failures:)", part, re.M | re.S):
            name, body = m.group(1), m.group(2)
            site = re.search(r"panicked at ([^\s]+):\n(.*)", body)
            msg = site.group(2).strip()[:220] if site else body.strip()[:220]
            print(f"   {crate}: {name}\n      at {site.group(1) if site else '?'}: {msg}")
