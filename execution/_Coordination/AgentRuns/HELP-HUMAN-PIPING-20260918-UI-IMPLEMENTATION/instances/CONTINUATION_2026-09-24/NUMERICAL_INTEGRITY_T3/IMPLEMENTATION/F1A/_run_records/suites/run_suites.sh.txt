#!/bin/bash
# F1a suites (I7): one crate at a time, sequential, own target. Logs to $S/suites/<crate>.log
set -u
S=<scratch>/i7
P=<wt>/f1a/projects/chirality-piping
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=<wt>/f1a-target
mkdir -p $S/suites
for spec in "$@"; do
  name=${spec%%:*}; dir=${spec#*:}
  echo "== $name ($dir) $(date -u +%FT%TZ)" | tee $S/suites/$name.log
  ( cd $P/$dir && cargo test --offline --locked --no-fail-fast 2>&1 ) >> $S/suites/$name.log
  echo "exit=$? $(date -u +%FT%TZ)" | tee -a $S/suites/$name.log
  grep -E '^test result:' $S/suites/$name.log | awk '{p+=$4; f+=$6; i+=$8} END {print "passed="p" failed="f" ignored="i}' | tee -a $S/suites/$name.log
  df -h <home> | tail -1
done
