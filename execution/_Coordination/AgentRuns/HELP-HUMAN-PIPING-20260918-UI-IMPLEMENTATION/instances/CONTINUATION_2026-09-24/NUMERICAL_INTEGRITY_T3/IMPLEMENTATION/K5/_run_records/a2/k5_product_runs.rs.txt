//! I14 (K5, A2) scratch product-run probe, never committed. Built into PP's
//! tests of `git archive` copies of Mac main and of the candidate; the two
//! outputs are compared run by run. Invented inputs; every model is stated
//! here, read from a committed fixture, or read from T3's K-D5 review records.
//! One record line per (input, entry, mode):
//! K5RUN <label> <entry> <mode> mechanics=<..> nq=<..> blocking=<codes> sha256=<envelope> integrity=<first 200 chars>
use open_pipe_stress_product_physics::{
    run_linear_static_preview_value_with_mode, run_linear_static_preview_with_mode,
    LinearStaticPreviewRequest, MechanicsEnvelope, PreviewSolverMode,
};
use serde_json::{json, Value};
use sha2::{Digest, Sha256};

const MODES: [PreviewSolverMode; 2] = [PreviewSolverMode::SparseInteractive, PreviewSolverMode::DenseScrutiny];
const SQRT1_2: f64 = std::f64::consts::FRAC_1_SQRT_2;

fn base(nodes: Value, pipes: Value, supports: Value, components: Value, cases: Value) -> Value {
    json!({"model":{"schema_version":"0.2.0","document_kind":"openpipestress.product_preview.model",
        "project":{"id":"project:k5runs","units":{"length":"m","force":"N","angle":"rad","pressure":"Pa","stress":"Pa","temperature":"degC"}},
        "analysis_status":{"mechanics":"ready_for_preview_diagnostics","rule_check":"not_performed_user_rule_inputs_missing","professional_acceptance":"not_provided"},
        "nodes":nodes,"pipe_segments":pipes,"supports":supports,"components":components,
        "materials":[{"id":"material:m","elastic_modulus":{"value":200e9,"unit":"Pa"},"shear_modulus":{"value":76.923076923e9,"unit":"Pa"},
            "thermal_expansion_coefficient":{"value":1.2e-5,"unit":"1/degC"},
            "temperature_points":[{"id":"point:hot","temperature":{"value":300.0,"unit":"degC"},"elastic_modulus":{"value":180e9,"unit":"Pa"},"shear_modulus":{"value":69.2e9,"unit":"Pa"}}],
            "provenance":"k5_invented_test_values"}],
        "load_cases":cases,"combinations":[]},"materials":[]})
}
fn node(id: &str, x: f64, y: f64, z: f64) -> Value {
    json!({"id":id,"position":{"x":x,"y":y,"z":z},"provenance":"k5"})
}
fn pipe(id: &str, from: &str, to: &str, yref: [f64; 3]) -> Value {
    json!({"id":id,"from":from,"to":to,"section":{"outside_diameter":{"value":0.12,"unit":"m"},"wall_thickness":{"value":0.01,"unit":"m"}},"material":"material:m","y_reference":{"x":yref[0],"y":yref[1],"z":yref[2]},"provenance":"k5"})
}
fn force(id: &str, node: &str, dir: &str, v: f64) -> Value {
    json!({"id":id,"category":"concentrated_force","target":{"type":"node","node":node},"direction":dir,"magnitude":{"value":v,"unit":"N"},"dimension":"force","provenance":"k5"})
}
fn support(id: &str, node: &str, restraints: &[&str]) -> Value {
    json!({"id":id,"node":node,"family":"anchor","restraints":restraints,"provenance":"k5"})
}
fn case(id: &str, loads: Value) -> Value {
    json!({"id":id,"primitive_loads":loads,"provenance":"k5"})
}
fn bend(pipe_ref: &str, at: &str, r: f64, k: f64, sif: Option<f64>) -> Value {
    let mut modifiers = json!({"flexibility_factor_user_value":{"value":k,"unit":"none"},"source_reference":"k5_invented"});
    if let Some(s) = sif {
        modifiers["sif_user_value"] = json!({"value":s,"unit":"none"});
    }
    json!({"id":"component:bend","label":"k5 arc","kind":"bend","node":at,
        "geometry":{"bend_pipe_ref":pipe_ref,"bend_radius":{"value":r,"unit":"m"},"bend_angle":{"value":std::f64::consts::FRAC_PI_2,"unit":"rad"},"bend_plane_orientation":"global_xy_preview","bend_geometry_source_reference":"k5"},
        "modifiers":modifiers,
        "mechanics_interface":{"solver_consumption":"curved_bend_macro_element","rule_check_consumption":"user_rule_pack_inputs_only"},"provenance":"k5"})
}
fn gap_support() -> Value {
    json!({"id":"support:gap","node":"node:b","family":"nonlinear","restraints":[],
        "nonlinear":{"behavior":"gap","dof":"UZ","initial_state":"inactive","closes_when":"positive_displacement","gap":{"value":1000.0,"unit":"mm"}},
        "provenance":"k5 invented: an open gap support that never closes"})
}

/// The constructed curved mechanism: a–b straight, b–c a realized 90° bend of
/// radius r, c–d straight; translation pins at a and d; `rx` adds RX at a;
/// `gap` adds an open gap support at b.
fn mechanism(o: f64, r: f64, rx: bool, gap: bool) -> Value {
    let pin_a: &[&str] = if rx { &["UX", "UY", "UZ", "RX"] } else { &["UX", "UY", "UZ"] };
    let mut supports = vec![support("support:pa", "node:a", pin_a), support("support:pd", "node:d", &["UX", "UY", "UZ"])];
    if gap {
        supports.push(gap_support());
    }
    base(
        json!([node("node:a", o, o, 0.0), node("node:b", o + 1.0, o, 0.0), node("node:c", o + 1.0 + r, o + r, 0.0), node("node:d", o + 1.0 + r, o + 1.0 + r, 0.0)]),
        json!([pipe("pipe:a-b", "node:a", "node:b", [0.0, 1.0, 0.0]), pipe("pipe:b-c", "node:b", "node:c", [SQRT1_2, -SQRT1_2, 0.0]), pipe("pipe:c-d", "node:c", "node:d", [1.0, 0.0, 0.0])]),
        json!(supports),
        json!([bend("pipe:b-c", "node:c", r, 2.0, None)]),
        json!([case("case:z", json!([force("z", "node:b", "global_z", 1000.0)])), case("case:x", json!([force("x", "node:b", "global_x", 1000.0)]))]),
    )
}

/// PP's `arc_model` (preview_physics_runtime.rs): anchor at a, R = 0.2 m.
fn arc_model(k: f64, yref_bc: [f64; 3], sif: Option<f64>) -> Value {
    base(
        json!([node("node:a", 0.0, 0.0, 0.0), node("node:b", 1.0, 0.0, 0.0), node("node:c", 1.2, 0.2, 0.0), node("node:d", 1.2, 1.2, 0.0)]),
        json!([pipe("pipe:a-b", "node:a", "node:b", [0.0, 1.0, 0.0]), pipe("pipe:b-c", "node:b", "node:c", yref_bc), pipe("pipe:c-d", "node:c", "node:d", [1.0, 0.0, 0.0])]),
        json!([support("support:a", "node:a", &["UX", "UY", "UZ", "RX", "RY", "RZ"])]),
        json!([bend("pipe:b-c", "node:c", 0.2, k, sif)]),
        json!([case("case:z", json!([force("z", "node:d", "global_z", 1000.0)]))]),
    )
}

/// PP's REF-B2 lone arc: anchor at b, a roller UX at c.
fn lone_arc(k: f64) -> Value {
    base(
        json!([node("node:b", 1.0, 0.0, 0.0), node("node:c", 1.2, 0.2, 0.0)]),
        json!([pipe("pipe:b-c", "node:b", "node:c", [SQRT1_2, -SQRT1_2, 0.0])]),
        json!([support("support:a", "node:b", &["UX", "UY", "UZ", "RX", "RY", "RZ"]), support("support:roller", "node:c", &["UX"])]),
        json!([bend("pipe:b-c", "node:c", 0.2, k, None)]),
        json!([case("case:y", json!([force("y", "node:c", "global_y", 1000.0)]))]),
    )
}

/// The headless producer cases realize a curved bend; as the headless test
/// does, legacy pressure is set to zero before solving.
fn unpressurized(mut model: Value) -> Value {
    for case in model["load_cases"].as_array_mut().unwrap() {
        for load in case["primitive_loads"].as_array_mut().unwrap() {
            if load["category"] == "pressure" || load["dimension"] == "pressure" {
                load["magnitude"]["value"] = json!(0.0);
            }
        }
    }
    json!({"model": model, "materials": []})
}

fn record(label: &str, entry: &str, mode: PreviewSolverMode, envelope: &MechanicsEnvelope) {
    let text = serde_json::to_string(envelope).unwrap();
    let sha: String = Sha256::digest(text.as_bytes()).iter().map(|b| format!("{b:02x}")).collect();
    let blocking: Vec<&str> = envelope.diagnostics.iter().filter(|d| d.severity == "blocking").map(|d| d.code.as_str()).collect();
    let integrity: String = envelope
        .diagnostics
        .iter()
        .find(|d| d.code.starts_with("NUMERICAL_INTEGRITY"))
        .map(|d| d.message.chars().take(200).collect::<String>().replace(' ', "_"))
        .unwrap_or_else(|| "-".into());
    let nq = serde_json::to_value(&envelope.numerical_quality).unwrap();
    println!(
        "K5RUN {label} {entry} {mode:?} mechanics={} nq={} blocking={} sha256={sha} integrity={integrity}",
        envelope.status.mechanics,
        nq["status"].as_str().unwrap_or("-"),
        if blocking.is_empty() { "-".to_string() } else { blocking.join(",") },
    );
}

fn run(label: &str, value: &Value) {
    for mode in MODES {
        match run_linear_static_preview_value_with_mode(value.clone(), mode) {
            Ok(envelope) => record(label, "captured", mode, &envelope),
            Err(error) => println!("K5RUN {label} captured {mode:?} error={}", error.replace(' ', "_")),
        }
        match serde_json::from_value::<LinearStaticPreviewRequest>(value.clone()) {
            Ok(request) => record(label, "typed", mode, &run_linear_static_preview_with_mode(request, mode)),
            Err(error) => println!("K5RUN {label} typed {mode:?} dto_error={}", error.to_string().replace(' ', "_")),
        }
    }
}

#[test]
fn k5_product_runs() {
    // Constructed curved mechanisms (dyadic R = 0.25 at 1 m, 1 km, 5e6 m; the
    // non-dyadic R = 0.2 and 0.3 of the checkpoint-0 run), each with its
    // companion and its nonlinear-support variant.
    for (r, offsets) in [(0.25, &[0.0, 1024.0, 5.0e6][..]), (0.2, &[0.0, 5.0e6][..]), (0.3, &[0.0, 1000.0, 5.0e6][..])] {
        for &o in offsets {
            for (variant, rx, gap) in [("mechanism", false, false), ("companion", true, false), ("gap", false, true)] {
                run(&format!("constructed_{variant}_r{r}_o{o}"), &mechanism(o, r, rx, gap));
            }
        }
    }
    // PP's curved-arc product tests.
    for k in [2.0, 4.0] {
        run(&format!("arc_model_k{k}"), &arc_model(k, [SQRT1_2, -SQRT1_2, 0.0], Some(1.3)));
    }
    run("arc_model_kinked", &arc_model(2.0, [0.0, 0.0, 1.0], None));
    for k in [1.0, 2.0, 4.0] {
        run(&format!("lone_arc_k{k}"), &lone_arc(k));
    }
    // K-D5's PP-route elbows (formation_check_runtime.rs).
    run("pp_route_utm_5e5", &pp_route_elbow("PP-UTM-5E5-PHI2", [500000.0, 350000.0], [500000.010469849, 350000.0001827519]));
    run("pp_route_utm_5e6", &pp_route_elbow("PP-UTM-5E6-PHI5", [5000000.0, 3500000.0], [5000000.026146723, 3500000.0011415905]));
    // K-D5's review UTM elbows (T3 REVIEW/_run_records/kd5_review/m31b).
    let review = std::env::var("K5_REVIEW_DIR").expect("K5_REVIEW_DIR");
    for name in ["PP-UTM-5e6-phi2", "PP-UTM-5e6-phi5", "PP-UTM-7.3e6-phi10"] {
        let value: Value = serde_json::from_str(&std::fs::read_to_string(format!("{review}/{name}.json")).unwrap()).unwrap();
        run(&format!("review_{name}"), &value);
    }
    // The headless producer cases that realize a curved bend.
    let root = std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
    let export: Value = serde_json::from_str(&std::fs::read_to_string(root.join("fixtures/results/invented/result_export_v0_2.json")).unwrap()).unwrap();
    for case in export["producer_cases"].as_array().unwrap() {
        let id = case["case_id"].as_str().unwrap();
        if id.starts_with("curved") {
            run(&format!("producer_{id}"), &unpressurized(case["model"].clone()));
        }
    }
    // The committed models with a realized joint flag (refused upstream).
    let preview: Value = serde_json::from_str(&std::fs::read_to_string(root.join("fixtures/product_preview/invented_preview_model.json")).unwrap()).unwrap();
    run("committed_invented_preview_model", &json!({"model": preview, "materials": []}));
}
