#!/usr/bin/env python3
"""K2b models generator (T3 IMPLEMENTATION/K2B/_run_records/k2b_models.py.txt).

Standard library only. Usage:
    python3 -B k2b_models.py <references.json> <out.rs> <out.json>

Reads R1's frozen references (T3/REFERENCES/references.json, c0f14201c,
sha256 7b176dbb...), unchanged, and writes the generated test models of K2b's
adapter tests (`structural_adapter/k2b_models.rs`):

1. The three RF-RANGE LEF-large cases (CHAIN, SKEW, CONT) at kernel level:
   nodes, the section inputs (E, G, OD, wall) as the product receives them,
   members, rigid supports, axis-aligned rotation springs, nodal loads, and
   the expected values u, th, R, S, N, T, Mb.i and Mb.j with their class
   scales (R1's 40-digit decimals, rounded once to binary64). The first
   LEF-small member is carried for the geometry assertion. Each case's base
   case (the RF-CHAIN, RF-SKEW or RF-LARGE case it scales exactly by lengths
   2^pl, moduli 2^pm and loads 2^pf) is carried in the same form, with pl, pm
   and pf, for the exact-equivariance comparison.
2. Four K2a product-reach cases (the constants of PP
   tests/k2a_formation_range_runtime.rs) and a synthetic PHYS-R4 element at
   kernel level, each with its exact reference: the exact rational solution of
   the intended system formed from the binary64 inputs (the section as the
   product derives it in binary64, then exact coefficients), rounded once.
3. For every case, the force scale b that the census rule gives (D1 §4.7
   steps 2-3 with ROOT's K2b rulings 1 and 2), computed here independently of
   the kernel: E and G, the 24 predicted exponents per frame, the exact
   exponents of springs and load terms.

Python floats are IEEE binary64 with round-to-nearest-even; math.pi equals
Rust's std::f64::consts::PI; math.sqrt is correctly rounded. Every binary64
operation below is written in the kernel's evaluation order.
"""
import json
import math
import sys
from decimal import Decimal as D, getcontext
from fractions import Fraction as Fr

getcontext().prec = 80
PI = math.pi
DOFS = ['UX', 'UY', 'UZ', 'RX', 'RY', 'RZ']


def parse_input(t):
    """R1's parse_input (T3/REFERENCES/references.py), in effect."""
    if '*2^' in t:
        a, k = t.split('*2^')
        return parse_input(a) * Fr(2) ** int(k)
    if '/' in t:
        p, q = t.split('/')
        return Fr(int(p), int(q))
    return Fr(D(t))


def f64(t):
    return float(parse_input(t))


def bits(x):
    import struct
    return '0x%016x' % struct.unpack('<Q', struct.pack('<d', x))[0]


def bexp(x):
    """The kernel's binary_exponent for a normal x: floor(log2 |x|)."""
    assert x != 0.0 and math.isfinite(x) and abs(x) >= 2.0 ** -1022, x
    return math.frexp(x)[1] - 1


def section(od, wall):
    """PP derive_pipe_section, in binary64 and in its order."""
    idd = od - 2.0 * wall
    area = PI * ((od * od) - (idd * idd)) / 4.0
    i = PI * (((od * od) * (od * od)) - ((idd * idd) * (idd * idd))) / 64.0
    return area, i, 2.0 * i


def length(a, b):
    d = [b[0] - a[0], b[1] - a[1], b[2] - a[2]]
    return math.sqrt(d[0] * d[0] + d[1] * d[1] + d[2] * d[2])


class Census:
    """D1 §4.7 steps 2-3 (ROOT's K2b rulings 1 and 2), independently."""

    def __init__(self):
        self.values = []
        self.subnormal = False

    def exact(self, x):
        if x == 0.0:
            return
        if abs(x) < 2.0 ** -1022:
            self.subnormal = True
            return
        self.values.append(bexp(x))

    def frame(self, e, g, a, i, j, l):
        ee, eg, ea, ei, ej, el = map(bexp, (e, g, a, i, j, l))
        self.values += [ee, eg, ee + ea, ee + ea - el, eg + ej, eg + ej - el]
        for k, p in [(3, 3), (2, 2), (2, 1), (1, 1)]:
            self.values.append(k + ee)
            for _ in range(2):  # Iy and Iz
                self.values += [k + ee + ei, k + ee + ei - p * el]

    def b(self):
        assert not self.subnormal
        e_min, e_max = min(self.values), max(self.values)
        low, high = -1022 + 64 - e_min, 1023 - 8 - e_max
        assert low <= high, (e_min, e_max)
        m = (low + high) // 2
        if m % 2 == 0:
            b = m
        elif m - 1 >= low:
            b = m - 1
        else:
            assert m + 1 <= high
            b = m + 1
        return b, e_min, e_max


def rs_f64(x):
    return 'f64::from_bits(%s)' % bits(x)


def lef_case(cid, c):
    m = c['model']
    names = list(m['nodes_m'])
    index = {n: k for k, n in enumerate(names)}
    nodes = [[f64(v) for v in m['nodes_m'][n]] for n in names]
    (sid, sec), = m['sections'].items()
    E, G = f64(sec['E']), f64(sec['G'])
    OD, ID = parse_input(sec['OD']), parse_input(sec['ID'])
    od, wall = float(OD), float((OD - ID) / 2)
    area, i, j = section(od, wall)
    members = []
    census = Census()
    for (mid, a, b, _s) in m['members']:
        members.append((mid, index[a], index[b]))
        census.frame(E, G, area, i, j, length(nodes[index[a]], nodes[index[b]]))
    rigid, springs = [], []
    for n, s in m['supports'].items():
        for d in s['rigid']:
            rigid.append(6 * index[n] + DOFS.index(d))
        for sp in s['springs']:
            direction = [parse_input(v) for v in sp['direction']]
            axis = [k for k, v in enumerate(direction) if v != 0]
            assert len(axis) == 1 and direction[axis[0]] == 1, (cid, sp)
            offset = 3 if sp['kind'] == 'rotation' else 0
            k = f64(sp['k'])
            springs.append((6 * index[n] + offset + axis[0], k))
            census.exact(k)
    loads = []
    for n, l in m['loads'].items():
        for key, offset in (('F', 0), ('M', 3)):
            for axis, v in enumerate(l.get(key, [])):
                x = f64(v)
                if x != 0.0:
                    loads.append((6 * index[n] + offset + axis, x))
                    census.exact(x)
    b, e_min, e_max = census.b()
    rng = c.get('range') or {}
    pl, pm, pf = rng.get('pl', 0), rng.get('pm', 0), rng.get('pf', 0)
    scales = {k: float(D(v['value'])) for k, v in c['scales'].items()}
    expected = []
    spring_index = {}
    for n, s in m['supports'].items():
        for k, sp in enumerate(s['springs']):
            spring_index[(n, k)] = sp
    for (name, value, cls) in c['expected']:
        parts = name.split('.')
        kind = parts[0]
        x = float(D(value))
        scale = scales[cls.split('@')[0]]
        if kind in ('u', 'th', 'R'):
            target = ('dof', 6 * index[parts[1]] + DOFS.index(parts[2]))
        elif kind == 'S':
            sp = spring_index[(parts[1], int(parts[2]))]
            direction = [parse_input(v) for v in sp['direction']]
            axis = [k for k, v in enumerate(direction) if v != 0][0]
            offset = 3 if sp['kind'] == 'rotation' else 0
            component = 'FM'.index(parts[3][0]) * 3 + 'XYZ'.index(parts[3][1])
            spring_dof = 6 * index[parts[1]] + offset + axis
            target = ('spring', spring_dof, 6 * index[parts[1]] + component)
        elif kind in ('N', 'T'):
            target = ('member', [mm[0] for mm in members].index(parts[1]), kind)
        elif kind == 'Mb' and parts[2] in ('i', 'j'):
            target = ('member', [mm[0] for mm in members].index(parts[1]), 'Mb' + parts[2])
        else:
            continue
        expected.append((name, target, x, scale))
    return {
        'id': cid, 'nodes': nodes, 'e': E, 'g': G, 'od': od, 'wall': wall,
        'members': members, 'rigid': sorted(rigid), 'springs': springs, 'loads': loads,
        'expected': expected, 'b': b, 'e_min': e_min, 'e_max': e_max,
        'pl': pl, 'pm': pm, 'pf': pf,
    }


# K2a product-reach cases (PP tests/k2a_formation_range_runtime.rs, invented) and
# a synthetic PHYS-R4 element (E = 1 Pa, OD 4e-77 m, wall 1e-77 m, L = 1 m, as
# D1 §2.4 states; G = 0.4 Pa and the loads are invented here).
REACH = [
    # id, L, OD, wall, E, G, free DOF at N1, spring, [(dof at N1, load)]
    ('spring-carried', 2.0, 1.0e-6, 1.0e-7, 2.0e11, 1.0e-300, 'RX', 1.0, [('RX', 1.0)]),
    ('partial-underflow', 9.5367431640625e-07, 3.0e-8, 3.0e-9, 1.3e-292, 1.0e-200, 'UY', None,
     [('UY', 1.0e-307)]),
    ('reach-zero', 1.8189894035458565e-12, 1.0e-11, 1.0e-12, 6.4e-280, 1.0e-100, 'UY', 3.7e-289,
     [('UY', 9.25e-290)]),
    ('reach-lef', 1.8189894035458565e-12, 1.0e-11, 1.0e-12, 9.6e-280, 1.0e-100, 'UY', None,
     [('UY', 2.05e-289)]),
    ('phys-r4', 1.0, 4.0e-77, 1.0e-77, 1.0, 0.4, None, None,
     [('UX', 1.0e-156), ('UY', 1.0e-307), ('RX', 1.0e-307)]),
]


def reach_case(rid, L, od, wall, E, G, free, spring, loads):
    area, i, j = section(od, wall)
    census = Census()
    census.frame(E, G, area, i, j, L)
    if spring is not None:
        census.exact(spring)
    for _, x in loads:
        census.exact(x)
    b, e_min, e_max = census.b()
    Fe, Fg, Fa, Fi, Fj, Fl = map(Fr, (E, G, area, i, j, L))
    load = {d: Fr(x) for d, x in loads}
    exact = {}
    if free == 'UY':
        k = 12 * Fe * Fi / Fl ** 3 + (Fr(spring) if spring is not None else 0)
        exact['UY'] = load['UY'] / k
    elif free == 'RX':
        k = Fg * Fj / Fl + Fr(spring)
        exact['RX'] = load['RX'] / k
    else:  # the PHYS-R4 cantilever: N1 free in all six DOFs
        exact['UX'] = load['UX'] * Fl / (Fe * Fa)
        exact['UY'] = load['UY'] * Fl ** 3 / (3 * Fe * Fi)
        exact['RZ'] = load['UY'] * Fl ** 2 / (2 * Fe * Fi)
        exact['RX'] = load['RX'] * Fl / (Fg * Fj)
        exact['UZ'] = Fr(0)
        exact['RY'] = Fr(0)
    return {
        'id': rid, 'length': L, 'od': od, 'wall': wall, 'e': E, 'g': G, 'free': free,
        'spring': spring, 'loads': loads, 'area': area, 'i': i, 'j': j,
        'exact': {d: float(v) for d, v in exact.items()},
        'exact_rational': {d: str(v) for d, v in exact.items()},
        'b': b, 'e_min': e_min, 'e_max': e_max,
    }


def main():
    refs, out_rs, out_json = sys.argv[1:4]
    cases = json.load(open(refs))['cases']
    lef = [lef_case(cid, cases[cid]) for cid in
           ('RF-RANGE-CHAIN-LEF-large', 'RF-RANGE-SKEW-LEF-large', 'RF-RANGE-CONT-LEF-large')]
    base = [lef_case(cases[c['id']]['range']['base'], cases[cases[c['id']]['range']['base']]) for c in lef]
    small = cases['RF-RANGE-CHAIN-LEF-small']['model']
    small_nodes = [[f64(v) for v in small['nodes_m'][n]] for n in ('N0', 'N1')]
    reach = [reach_case(*r) for r in REACH]
    with open(out_rs, 'w') as f:
        w = f.write
        w('//! Generated by the K2b run record `k2b_models.py` (T3 IMPLEMENTATION/K2B/_run_records);\n')
        w('//! do not edit by hand. Invented inputs only: R1\'s frozen RF-RANGE references\n')
        w('//! (T3/REFERENCES/references.json, c0f14201c), each input and expected value rounded\n')
        w('//! once to binary64; the K2a product-reach constants; a synthetic PHYS-R4 element.\n')
        w('//! `b` is the census rule\'s force scale (D1 §4.7 steps 2-3, ROOT\'s K2b rulings 1-2),\n')
        w('//! computed by the generator independently of the kernel. Exact references are the\n')
        w('//! exact rational solutions of the intended systems formed from the binary64 inputs.\n')
        w('#![cfg_attr(rustfmt, rustfmt::skip)]\n#![allow(dead_code)]\n\n')
        w('pub enum Target { Dof(usize), Spring(usize, usize), Member(usize, &\'static str) }\n\n')
        w('pub struct LefCase {\n    pub id: &\'static str,\n    pub nodes: &\'static [[f64; 3]],\n'
          '    pub e: f64, pub g: f64, pub od: f64, pub wall: f64,\n'
          '    pub members: &\'static [(&\'static str, usize, usize)],\n'
          '    pub rigid: &\'static [usize],\n    pub springs: &\'static [(usize, f64)],\n'
          '    pub loads: &\'static [(usize, f64)],\n'
          '    pub expected: &\'static [(&\'static str, Target, f64, f64)],\n'
          '    pub b: i32, pub e_min: i32, pub e_max: i32,\n'
          '    /// RF-RANGE\'s exact scaling of the base case: lengths 2^pl, moduli 2^pm, loads 2^pf.\n'
          '    pub pl: i32, pub pm: i32, pub pf: i32,\n}\n\n')
        for label, group in (('LEF_LARGE', lef), ('LEF_BASE', base)):
          w('pub const %s: &[LefCase] = &[\n' % label)
          for c in group:
              w('    LefCase {\n        id: "%s",\n' % c['id'])
              w('        nodes: &[%s],\n' % ', '.join('[%s]' % ', '.join(rs_f64(v) for v in n) for n in c['nodes']))
              w('        e: %s, g: %s, od: %s, wall: %s,\n' % tuple(rs_f64(c[k]) for k in ('e', 'g', 'od', 'wall')))
              w('        members: &[%s],\n' % ', '.join('("%s", %d, %d)' % m for m in c['members']))
              w('        rigid: &[%s],\n' % ', '.join(str(d) for d in c['rigid']))
              w('        springs: &[%s],\n' % ', '.join('(%d, %s)' % (d, rs_f64(k)) for d, k in c['springs']))
              w('        loads: &[%s],\n' % ', '.join('(%d, %s)' % (d, rs_f64(x)) for d, x in c['loads']))
              w('        expected: &[\n')
              for name, t, x, s in c['expected']:
                  if t[0] == 'dof':
                      tt = 'Target::Dof(%d)' % t[1]
                  elif t[0] == 'spring':
                      tt = 'Target::Spring(%d, %d)' % (t[1], t[2])
                  else:
                      tt = 'Target::Member(%d, "%s")' % (t[1], t[2])
                  w('            ("%s", %s, %s, %s),\n' % (name, tt, rs_f64(x), rs_f64(s)))
              w('        ],\n        b: %d, e_min: %d, e_max: %d,\n        pl: %d, pm: %d, pf: %d,\n    },\n' % (c['b'], c['e_min'], c['e_max'], c['pl'], c['pm'], c['pf']))
          w('];\n\n')
        w('/// RF-RANGE-CHAIN-LEF-small, member M1 (N0 -> N1): the geometry refusal.\n')
        w('pub const LEF_SMALL_M1: [[f64; 3]; 2] = [%s];\n\n' % ', '.join(
            '[%s]' % ', '.join(rs_f64(v) for v in n) for n in small_nodes))
        w('pub struct ReachCase {\n    pub id: &\'static str,\n    pub length: f64, pub od: f64, pub wall: f64, pub e: f64, pub g: f64,\n'
          '    /// The one free DOF at N1 (None: all six free).\n    pub free: Option<usize>,\n'
          '    pub spring: Option<f64>,\n    /// (local DOF at N1, load).\n    pub loads: &\'static [(usize, f64)],\n'
          '    /// The section as the product derives it (A, I, J).\n    pub area: f64, pub i: f64, pub j: f64,\n'
          '    /// (local DOF at N1, exact reference rounded once).\n    pub exact: &\'static [(usize, f64)],\n'
          '    pub b: i32, pub e_min: i32, pub e_max: i32,\n}\n\n')
        w('pub const REACH: &[ReachCase] = &[\n')
        for r in reach:
            free = 'Some(%d)' % DOFS.index(r['free']) if r['free'] else 'None'
            spring = 'Some(%s)' % rs_f64(r['spring']) if r['spring'] is not None else 'None'
            w('    ReachCase {\n        id: "%s",\n' % r['id'])
            w('        length: %s, od: %s, wall: %s, e: %s, g: %s,\n' % tuple(rs_f64(r[k]) for k in ('length', 'od', 'wall', 'e', 'g')))
            w('        free: %s, spring: %s,\n' % (free, spring))
            w('        loads: &[%s],\n' % ', '.join('(%d, %s)' % (DOFS.index(d), rs_f64(x)) for d, x in r['loads']))
            w('        area: %s, i: %s, j: %s,\n' % tuple(rs_f64(r[k]) for k in ('area', 'i', 'j')))
            w('        exact: &[%s],\n' % ', '.join('(%d, %s)' % (DOFS.index(d), rs_f64(x)) for d, x in r['exact'].items()))
            w('        b: %d, e_min: %d, e_max: %d,\n    },\n' % (r['b'], r['e_min'], r['e_max']))
        w('];\n')
    json.dump({'references_sha256_expected': '7b176dbbf2296be02d8bca19c698ee56d0d5753751150a9175e0d3ad4cf89cc9',
               'python': sys.version.split()[0],
               'lef_large': [{k: v for k, v in c.items() if k != 'expected'} | {'expected_count': len(c['expected'])} for c in lef],
               'reach': reach}, open(out_json, 'w'), indent=1)


if __name__ == '__main__':
    main()
