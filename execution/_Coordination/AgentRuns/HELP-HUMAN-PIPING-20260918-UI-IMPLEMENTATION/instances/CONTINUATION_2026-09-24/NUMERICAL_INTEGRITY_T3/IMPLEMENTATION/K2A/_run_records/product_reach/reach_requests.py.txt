#!/usr/bin/env python3
"""K2a (I6): the product-reach requests of PP tests/k2a_formation_range_runtime.rs, written as JSON
for the fixture-diff harness (captured entry) on main and on the candidate. Mirrors the test's
`request` builder field for field. Usage: python3 -B reach_requests.py <outdir>"""
import json, os, sys
PROV = "invented_t3_k2a_product_reach_input_no_library_data"
ALL = ["UX", "UY", "UZ", "RX", "RY", "RZ"]
def request(id, length, od, wall, e, g, free, spring, load, open_gap):
    sup = [{"id": "rigid:N0", "node": "N0", "restraints": ALL, "family": "anchor", "provenance": PROV},
           {"id": "rigid:N1", "node": "N1", "restraints": [d for d in ALL if d != free], "family": "anchor", "provenance": PROV}]
    rot = free.startswith("R")
    if spring is not None:
        sup.append({"id": "spring:N1:0", "node": "N1", "family": "spring", "restraints": [free],
                    "stiffness": {"dof": free, "value": {"value": spring, "unit": "N*m/rad" if rot else "N/m"}}, "provenance": PROV})
    if open_gap:
        sup.append({"id": "support:k2a-open-gap", "node": "N1", "family": "nonlinear", "restraints": [],
                    "nonlinear": {"behavior": "gap", "dof": free, "initial_state": "inactive",
                                  "closes_when": "positive_displacement", "gap": {"value": 1000.0, "unit": "mm"}}, "provenance": PROV})
    if rot:
        ld = {"id": "load:0", "category": "concentrated_moment", "target": {"type": "node", "node": "N1"}, "direction": free,
              "magnitude": {"value": load, "unit": "N*m"}, "dimension": "moment", "provenance": PROV}
    else:
        ld = {"id": "load:0", "category": "concentrated_force", "target": {"type": "node", "node": "N1"},
              "direction": "global_" + free[1:].lower(), "magnitude": {"value": load, "unit": "N"}, "dimension": "force", "provenance": PROV}
    return {"model": {"schema_version": "0.1.0", "document_kind": "openpipestress.product_preview.model",
            "analysis_status": {"mechanics": "ready_for_preview_diagnostics", "rule_check": "not_performed_user_rule_inputs_missing", "professional_acceptance": "not_provided"},
            "project": {"id": "invented:t3-k2a:" + id, "units": {"length": "m", "force": "N", "angle": "rad", "pressure": "Pa", "temperature": "degC", "stress": "Pa"}},
            "nodes": [{"id": "N0", "position": {"x": 0.0, "y": 0.0, "z": 0.0}, "provenance": PROV},
                      {"id": "N1", "position": {"x": length, "y": 0.0, "z": 0.0}, "provenance": PROV}],
            "pipe_segments": [{"id": "M1", "from": "N0", "to": "N1", "material": "mat:K2A", "y_reference": {"x": 0, "y": 1, "z": 0},
                               "section": {"outside_diameter": {"value": od, "unit": "m"}, "wall_thickness": {"value": wall, "unit": "m"}}, "provenance": PROV}],
            "materials": [{"id": "mat:K2A", "elastic_modulus": {"value": e, "unit": "Pa"}, "shear_modulus": {"value": g, "unit": "Pa"}, "provenance": PROV}],
            "supports": sup,
            "load_cases": [{"id": "case", "label": id, "kind": "primitive_user_load", "primitive_loads": [ld], "provenance": PROV}],
            "combinations": []}, "materials": []}
CASES = {
    "product-reach-torsion-underflow-spring-carried": dict(length=2.0, od=1e-6, wall=1e-7, e=2e11, g=1e-300, free="RX", spring=1.0, load=1.0, open_gap=False),
    "product-reach-partial-underflow": dict(length=9.5367431640625e-07, od=3e-8, wall=3e-9, e=1.3e-292, g=1e-200, free="UY", spring=None, load=1e-307, open_gap=False),
    "product-reach-partial-underflow-open-gap": dict(length=9.5367431640625e-07, od=3e-8, wall=3e-9, e=1.3e-292, g=1e-200, free="UY", spring=None, load=1e-307, open_gap=True),
}
out = sys.argv[1]
os.makedirs(out, exist_ok=True)
for k, v in CASES.items():
    json.dump(request(k, **v), open(os.path.join(out, k + ".json"), "w"), indent=1)
