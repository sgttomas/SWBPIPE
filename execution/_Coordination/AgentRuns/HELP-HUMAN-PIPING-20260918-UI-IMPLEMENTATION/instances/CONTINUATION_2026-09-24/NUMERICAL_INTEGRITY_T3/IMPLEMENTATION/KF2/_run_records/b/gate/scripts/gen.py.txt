"""T3 P1 detection: request and expectation generator (scratch; committed as gen.py.txt).

Usage: python3 -B gen.py <references.py> <references.json> <fixtures.json> <outdir>

Every model input is taken from references.py itself: the script is imported as a
module, its family builders run, and each case's exact rational definition (`defn`)
is converted to a product request by rounding each intended input once to binary64
(float(Fraction), correctly rounded).  The printed JSON is used only for the
published expected values, class scales, RF-CANCEL recommended scales and negative
controls; never for model inputs.

The existing N/R/NP fixture cases are authored from fixtures.json and from the
geometry stated in P/validation/benchmarks/numerical_integrity/README.md, as the
in-repository observer (src/main.rs) authors them.  V1's check L, the S11 probe-A
member and a support-settlement probe are supplementary cases defined here.
"""
from __future__ import annotations

import importlib.util
import json
import os
import sys
from fractions import Fraction as Fr

PROV = 'invented_t3_p1_detection_input_no_library_data'
DOFS = ('UX', 'UY', 'UZ', 'RX', 'RY', 'RZ')
AXIS_OF = {0: 'X', 1: 'Y', 2: 'Z'}


def f64(x):
    return float(Fr(x))


def load_refs(path):
    spec = importlib.util.spec_from_file_location('references', path)
    m = importlib.util.module_from_spec(spec)
    sys.modules['references'] = m
    spec.loader.exec_module(m)
    for b in m.BUILD_ORDER:
        b()
    return m


def y_reference(pi, pj):
    e = [Fr(pj[k]) - Fr(pi[k]) for k in range(3)]
    mags = [abs(c) for c in e]
    # least-aligned global axis, preference Y, Z, X on ties
    best = min((1, 2, 0), key=lambda k: mags[k])
    v = [0, 0, 0]
    v[best] = 1
    return {'x': v[0], 'y': v[1], 'z': v[2]}


UNITS_SI = {'length': 'm', 'force': 'N', 'moment': 'N*m', 'rotation': 'rad', 'modulus': 'Pa'}


def base_model(pid, length_unit='m'):
    return {
        'schema_version': '0.1.0', 'document_kind': 'openpipestress.product_preview.model',
        'analysis_status': {'mechanics': 'ready_for_preview_diagnostics',
                            'rule_check': 'not_performed_user_rule_inputs_missing',
                            'professional_acceptance': 'not_provided'},
        'project': {'id': 'invented:t3-p1:' + pid,
                    'units': {'length': length_unit, 'force': 'N', 'angle': 'rad', 'pressure': 'Pa',
                              'temperature': 'degC', 'stress': 'Pa'}},
        'nodes': [], 'pipe_segments': [], 'materials': [], 'supports': [],
        'load_cases': [{'id': 'case', 'label': pid, 'kind': 'primitive_user_load', 'primitive_loads': [],
                        'provenance': PROV}],
        'combinations': [],
    }


def nodal_load(lid, node, comp, value, units):
    """comp in 0..5 (UX..RZ).  Forces as concentrated_force global_x/y/z; moments as RX/RY/RZ."""
    if comp < 3:
        return {'id': lid, 'category': 'concentrated_force', 'target': {'type': 'node', 'node': node},
                'direction': 'global_' + 'xyz'[comp], 'magnitude': {'value': value, 'unit': units['force']},
                'dimension': 'force', 'provenance': PROV}
    return {'id': lid, 'category': 'concentrated_moment', 'target': {'type': 'node', 'node': node},
            'direction': DOFS[comp], 'magnitude': {'value': value, 'unit': units['moment']},
            'dimension': 'moment', 'provenance': PROV}


def author_defn(cid, defn, units=None):
    """Return (request, None, meta) or (None, reason, meta)."""
    units = dict(units or UNITS_SI)
    meta = {'rigid_support_of': {}, 'springs_of': {}, 'members': [m[0] for m in defn['members']],
            'nodes': list(defn['nodes'].keys()), 'units': units}
    if units['moment'] != 'N*m':
        uses_moment = any(any(Fr(c) != 0 for c in ld.get('M', (0, 0, 0))) for ld in defn.get('loads', {}).values())
        uses_moment = uses_moment or any(d[0] == 'R' for (_, d, _) in defn.get('load_contributions', []))
        uses_rot_spring = any(kind == 'r' for s in defn.get('supports', {}).values() for (kind, _, _) in s.get('springs', []))
        if uses_moment or uses_rot_spring:
            return None, ('the product has no %s moment or %s/rad rotational-stiffness input unit '
                          '(units registry at c61a540ea); converting it would approximate the case'
                          % (units['moment'], units['moment'])), meta
    length_unit = units['length']
    force_per_len = {'m': 'N/m', 'mm': 'N/mm'}[length_unit]
    modulus_unit = units['modulus']
    model = base_model(cid, length_unit)
    for nid, p in defn['nodes'].items():
        model['nodes'].append({'id': nid, 'position': {'x': f64(p[0]), 'y': f64(p[1]), 'z': f64(p[2])},
                               'provenance': PROV})
    for sid, s in defn['sections'].items():
        E = Fr(s['E'])
        G = Fr(s['G']) if 'G' in s else E / (2 * (1 + Fr(s['nu'])))
        model['materials'].append({'id': 'mat:' + sid,
                                   'elastic_modulus': {'value': f64(E), 'unit': modulus_unit},
                                   'shear_modulus': {'value': f64(G), 'unit': modulus_unit},
                                   'provenance': PROV})
    for (mid, i, j, sid) in defn['members']:
        s = defn['sections'][sid]
        OD, ID = Fr(s['OD']), Fr(s['ID'])
        model['pipe_segments'].append({
            'id': mid, 'from': i, 'to': j, 'material': 'mat:' + sid,
            'y_reference': y_reference(defn['nodes'][i], defn['nodes'][j]),
            'section': {'outside_diameter': {'value': f64(OD), 'unit': length_unit},
                        'wall_thickness': {'value': f64((OD - ID) / 2), 'unit': length_unit}},
            'provenance': PROV})
    for nid, s in defn.get('supports', {}).items():
        rigid = list(s.get('rigid', []))
        if rigid:
            sup_id = 'rigid:' + nid
            sup = {'id': sup_id, 'node': nid, 'restraints': rigid, 'provenance': PROV}
            if any(d[0] == 'R' for d in rigid):
                # an omitted family with translations only is inferred as a guide, which may not restrain
                # rotations; the committed N05 fixture authors partial rigid restraint sets as 'anchor'
                sup['family'] = 'anchor'
            model['supports'].append(sup)
            meta['rigid_support_of'][nid] = sup_id
        for idx, (kind, d, k) in enumerate(s.get('springs', [])):
            d = [Fr(c) for c in d]
            nz = [a for a in range(3) if d[a] != 0]
            if len(nz) != 1:
                return None, ('spring %d at %s acts along the non-global direction (%s); the product authors '
                              'springs by global DOF only (SupportStiffnessInput.dof)'
                              % (idx, nid, ','.join(str(c) for c in d))), meta
            dof = ('U' if kind == 't' else 'R') + AXIS_OF[nz[0]]
            unit = force_per_len if kind == 't' else units['moment'] + '/rad'
            sup_id = 'spring:%s:%d' % (nid, idx)
            model['supports'].append({'id': sup_id, 'node': nid, 'family': 'spring', 'restraints': [dof],
                                      'stiffness': {'dof': dof, 'value': {'value': f64(k), 'unit': unit}},
                                      'provenance': PROV})
            meta['springs_of'].setdefault(nid, {})[str(idx)] = sup_id
    loads = model['load_cases'][0]['primitive_loads']
    n = 0
    for nid, ld in defn.get('loads', {}).items():
        for comp in range(6):
            vec = ld.get('F' if comp < 3 else 'M', (0, 0, 0))
            v = Fr(vec[comp % 3])
            if v != 0:
                loads.append(nodal_load('load:%d' % n, nid, comp, f64(v), units))
                n += 1
    for (nid, dof, v) in defn.get('load_contributions', []):
        loads.append(nodal_load('contribution:%d' % n, nid, DOFS.index(dof), f64(v), units))
        n += 1
    for mid, q in defn.get('member_loads', {}).items():
        for a in range(3):
            if Fr(q[a]) != 0:
                loads.append({'id': 'udl:%d' % n, 'category': 'distributed_force',
                              'target': {'type': 'element', 'pipe': mid}, 'direction': 'global_' + 'xyz'[a],
                              'magnitude': {'value': f64(q[a]), 'unit': force_per_len},
                              'dimension': 'force_per_length', 'provenance': PROV})
                n += 1
    return {'model': model, 'materials': []}, None, meta


def ref_rows(rec):
    """Expected rows as [key, expected_str, class, scale_str, governed_by|None, absolute_below_range(bool)]."""
    scales = {c: v['value'] for c, v in rec.get('scales', {}).items()}
    out = []
    if rec['family'] == 'RF-CANCEL':
        for (k, e, cl, rec_scale, gross, gov) in rec['expected']:
            out.append([k, e, cl, rec_scale, gov])
        return out, 'recommended (net-governed) per-value scale, ROOT_RULINGS_V2 1'
    rows = rec['expected_represented'] if rec.get('basis') == 'represented' else rec['expected']
    for (k, e, cl) in rows:
        out.append([k, e, cl, scales[cl], None])
    return out, 'published class scale'

# fixture (N/R/NP) cases, authored as the in-repository observer does

def fixture_model(cid, L, torque=0.0, force=None):
    m = base_model(cid)
    m['nodes'] = [{'id': 'root', 'position': {'x': 0.0, 'y': 0.0, 'z': 0.0}, 'provenance': PROV},
                  {'id': 'tip', 'position': {'x': float(L), 'y': 0.0, 'z': 0.0}, 'provenance': PROV}]
    m['pipe_segments'] = [{'id': 'pipe', 'from': 'root', 'to': 'tip', 'material': 'material',
                           'y_reference': {'x': 0, 'y': 1, 'z': 0},
                           'section': {'outside_diameter': {'value': 0.2, 'unit': 'm'},
                                       'wall_thickness': {'value': 0.01, 'unit': 'm'}}, 'provenance': PROV}]
    m['materials'] = [{'id': 'material', 'elastic_modulus': {'value': 200e9, 'unit': 'Pa'},
                       'shear_modulus': {'value': 80e9, 'unit': 'Pa'}, 'provenance': PROV}]
    m['supports'] = [{'id': 'anchor', 'node': 'root', 'family': 'anchor',
                      'restraints': list(DOFS), 'provenance': PROV}]
    loads = m['load_cases'][0]['primitive_loads']
    if force is not None:
        loads.append(nodal_load('force', 'tip', 1, float(force), UNITS_SI))
    elif torque is not None:
        loads.append(nodal_load('torque', 'tip', 3, float(torque), UNITS_SI))
    return m


def fixture_meta(model):
    meta = {'rigid_support_of': {}, 'springs_of': {}, 'units': UNITS_SI,
            'members': [p['id'] for p in model['pipe_segments']], 'nodes': [n['id'] for n in model['nodes']]}
    for sup in model['supports']:
        if 'stiffness' in sup:
            meta['springs_of'].setdefault(sup['node'], {})[sup['id']] = sup['id']
        else:
            assert sup['node'] not in meta['rigid_support_of'], sup
            meta['rigid_support_of'][sup['node']] = sup['id']
    return meta


def fixture_cases(fx, PI):
    N, R, NP = fx['N'], fx['R'], fx['NP']
    cases = []

    def add(cid, family, model, rows, expect, note, refusal_expected=False):
        cases.append({'id': cid, 'family': family, 'request': {'model': model, 'materials': []} if model else None,
                      'meta': fixture_meta(model) if model else None,
                      'rows': rows, 'expectation': expect, 'note': note, 'refusal_expected': refusal_expected,
                      'scale_rule': 'analytical 1e-9 relative (scale 0: max(|exp|, 0))'})

    # N01, N09 bending, N08, N09 torsion
    m = fixture_model('N01', 2, force=1000)
    add('FX-N01', 'FIXTURE-N', m, [['u.tip.UY', N['N01']['uy_m'], 'translation', '0', None],
                                   ['th.tip.RZ', N['N01']['rz_rad'], 'rotation', '0', None],
                                   ['R.root.UY', str(N['N01']['root_FY_N']), 'force', '0', None],
                                   ['R.root.RZ', str(N['N01']['root_MZ_Nm']), 'moment', '0', None]],
        'solve', 'N01 tip force')
    b = N['N09']['bending']
    m = fixture_model('N09-bending', 10, force=b['F_N'])
    add('FX-N09-bending', 'FIXTURE-N', m, [['u.tip.UY', b['uy_m'], 'translation', '0', None],
                                           ['th.tip.RZ', b['rz_rad'], 'rotation', '0', None],
                                           ['R.root.UY', str(b['root_FY_N']), 'force', '0', None],
                                           ['R.root.RZ', str(b['root_MZ_Nm']), 'moment', '0', None]],
        'solve', 'N09 bending, L/D = 50')
    t = N['N09']['torsion']
    m = fixture_model('N09-torsion', 10, torque=float(t['T_Nm']))
    add('FX-N09-torsion', 'FIXTURE-N', m, [['th.tip.RX', t['theta_x_rad'], 'rotation', '0', None],
                                           ['R.root.RX', t['root_MX_Nm'], 'moment', '0', None],
                                           ['T.pipe', t['T_Nm'], 'moment', '0', None]], 'solve', 'N09 torsion')
    for tc in N['N08']['torque_cases']:
        T = tc['T_Nm']
        m = fixture_model('N08-%s' % T, 2, torque=float(T))
        add('FX-N08-T%s' % T, 'FIXTURE-N', m, [['th.tip.RX', tc['theta_x_rad'], 'rotation', '0', None],
                                               ['R.root.RX', tc['root_MX_Nm'], 'moment', '0', None],
                                               ['T.pipe', T, 'moment', '0', None]], 'solve', 'N08 torque sign/magnitude')
    # N05, N06 and the NP-A ulp series (N05 geometry, spring k, tip torque T)
    for row in NP['A']:
        rid = row['id']
        k = float(Fr(row['k_intended']))
        T = float(Fr(row['T_intended']))
        m = fixture_model('NP-A-' + rid, 2, torque=T)
        m['supports'][0]['restraints'] = ['UX', 'UY', 'UZ', 'RY', 'RZ']
        m['supports'].append({'id': 'soft', 'node': 'root', 'family': 'spring', 'restraints': ['RX'],
                              'stiffness': {'dof': 'RX', 'value': {'value': k, 'unit': 'N*m/rad'}}, 'provenance': PROV})
        rows = [['th.root.RX', row['intended_root'], 'rotation', '0', None],
                ['th.tip.RX', row['intended_tip'], 'rotation', '0', None],
                ['T.pipe', row['T_intended'], 'moment', '0', None],
                ['S.root.soft.MX', '-' + row['T_intended'], 'moment', '0', None]]
        name = {'N05': 'FX-N05', 'N06': 'FX-N06'}.get(rid, 'FX-NP-A-' + rid)
        add(name, 'FIXTURE-N' if rid in ('N05', 'N06') else 'FIXTURE-NP-A', m, rows, 'solve',
            'intended basis; the NP-A stored-exact root (%s) is recorded for information'
            % row.get('stored_exact_root', row.get('stored_status')))
    # N02 / N03 / N04 / R07 (refusal and stabilisation checks)
    for (cid, t) in (('FX-N02-loaded', 1.0), ('FX-N02-unloaded-R07', 0.0)):
        m = fixture_model(cid, 2, torque=None)
        m['nodes'][1]['position'] = {'x': 1.2, 'y': 1.6, 'z': 0.0}
        m['supports'] = [{'id': 'root-translations', 'node': 'root', 'restraints': ['UX', 'UY', 'UZ'], 'provenance': PROV},
                         {'id': 'tip-translations', 'node': 'tip', 'restraints': ['UX', 'UY', 'UZ'], 'provenance': PROV}]
        if t:
            m['load_cases'][0]['primitive_loads'] = [nodal_load('torque-x', 'tip', 3, 0.6 * t, UNITS_SI),
                                                     nodal_load('torque-y', 'tip', 4, 0.8 * t, UNITS_SI)]
        add(cid, 'FIXTURE-N', m, [], 'refuse', 'N02 null rotation (3/5,4/5,0); R07 is the zero-load non-unique case',
            refusal_expected=True)
        if t:
            for (cid2, dof, exp) in (('FX-N03-RX-stabilized', 'RX', 'solve'), ('FX-N03-RZ-not-stabilized', 'RZ', 'refuse')):
                m2 = json.loads(json.dumps(m))
                m2['supports'][0]['family'] = 'anchor'
                m2['supports'][0]['restraints'].append(dof)
                add(cid2, 'FIXTURE-N', m2, [], exp, 'N03: an extra root %s %s the null rotation'
                    % (dof, 'removes' if dof == 'RX' else 'does not remove'), refusal_expected=(exp == 'refuse'))
    m = fixture_model('N04', 2, force=1000)
    m['nodes'] += [{'id': 'free-root', 'position': {'x': 0.0, 'y': 3.0, 'z': 0.0}, 'provenance': PROV},
                   {'id': 'free-tip', 'position': {'x': 2.0, 'y': 3.0, 'z': 0.0}, 'provenance': PROV}]
    p = json.loads(json.dumps(m['pipe_segments'][0]))
    p.update({'id': 'free-pipe', 'from': 'free-root', 'to': 'free-tip'})
    m['pipe_segments'].append(p)
    add('FX-N04', 'FIXTURE-N', m, [], 'refuse', 'N04 free body (six null modes)', refusal_expected=True)
    # R01: 2 m axial bar, only tip UX free, tip force 1000 N
    m = fixture_model('R01', 2, force=None, torque=None)
    m['supports'].append({'id': 'tip-lateral', 'node': 'tip', 'family': 'anchor', 'restraints': ['UY', 'UZ', 'RX', 'RY', 'RZ'], 'provenance': PROV})
    m['load_cases'][0]['primitive_loads'] = [nodal_load('force', 'tip', 0, 1000.0, UNITS_SI)]
    add('FX-R01', 'FIXTURE-R', m, [['R.root.UX', '-1000', 'force', '0', None],
                                   ['N.pipe', '1000', 'force', '0', None]], 'solve',
        'R01 K = EA/2 one free DOF. R01 publishes a residual-test CANDIDATE (r = 10), not the solution, so only the '
        'statically determined reaction and axial force are compared')
    # R05: 1 m bar, axial spring k = EA/4 at the tip, tip force 1000 N
    m = fixture_model('R05', 1, force=None, torque=None)
    m['supports'].append({'id': 'tip-lateral', 'node': 'tip', 'family': 'anchor', 'restraints': ['UY', 'UZ', 'RX', 'RY', 'RZ'], 'provenance': PROV})
    kspring = f64(95000000 * PI)
    m['supports'].append({'id': 'tip-spring', 'node': 'tip', 'family': 'spring', 'restraints': ['UX'],
                          'stiffness': {'dof': 'UX', 'value': {'value': kspring, 'unit': 'N/m'}}, 'provenance': PROV})
    m['load_cases'][0]['primitive_loads'] = [nodal_load('force', 'tip', 0, 1000.0, UNITS_SI)]
    add('FX-R05', 'FIXTURE-R', m, [['u.tip.UX', R['R05']['valid_u'][0], 'translation', '0', None]], 'solve',
        'R05 bar plus spring EA/4 (k authored as float(95e6*PI_Q)); the omitted-spring K_used is a matrix injection, not authored')
    # R06: fully prescribed (zero) node carrying loads 1..6; authored as a member fixed at both ends
    m = fixture_model('R06', 2, force=None, torque=None)
    m['supports'].append({'id': 'tip-anchor', 'node': 'tip', 'family': 'anchor', 'restraints': list(DOFS), 'provenance': PROV})
    m['load_cases'][0]['primitive_loads'] = [nodal_load('l%d' % c, 'tip', c, float(c + 1), UNITS_SI) for c in range(6)]
    rows = [['R.tip.%s' % DOFS[c], str(-(c + 1)), 'force' if c < 3 else 'moment', '0', None] for c in range(6)]
    add('FX-R06', 'FIXTURE-R', m, rows, 'solve', 'R06 no free DOF; loads on restrained DOFs go straight to reactions')
    # not authorable (matrix-level or prescribed)
    for cid, why in (('FX-N07', 'generic indefinite 2x2 matrix; no frame model realizes it'),
                     ('FX-R02', 'residual test of an injected wrong matrix K_used; no model authoring'),
                     ('FX-R03', 'residual test of an injected wrong load f_used; no model authoring'),
                     ('FX-NP-B', 'banded K = CC^T matrix family; matrix-level'),
                     ('FX-NP-C', 'synthetic connector generalized-coordinate seam; matrix-level'),
                     ('FX-NP-D', 'inadmissible skew/non-finite injection matrices and operand-range cases; matrix-level')):
        cases.append({'id': cid, 'family': 'FIXTURE-NP' if 'NP' in cid else 'FIXTURE-' + cid[3], 'request': None, 'rows': [],
                      'expectation': 'not_authorable', 'note': why, 'refusal_expected': False, 'not_authorable': why})
    # R04 and a realistic support settlement: prescribed motion (refused on main unless 0.4.0)
    m = fixture_model('R04', 2, force=None, torque=None)
    m['supports'].append({'id': 'tip-lateral', 'node': 'tip', 'family': 'anchor', 'restraints': ['UY', 'UZ', 'RX', 'RY', 'RZ'], 'provenance': PROV})
    m['load_cases'][0]['primitive_loads'] = [{'id': 'imp', 'category': 'imposed_displacement',
                                              'target': {'type': 'support', 'support': 'anchor', 'dof': 'UX'},
                                              'direction': 'UX', 'magnitude': {'value': 0.0001, 'unit': 'm'},
                                              'dimension': 'displacement', 'provenance': PROV}]
    add('FX-R04', 'FIXTURE-R', m, [], 'not_authorable_if_refused', 'R04 prescribed displacement 1e-4 m')
    return cases


def check_l_cases(m):
    """V1 check L (REVIEW/_run_records/v1_stop_rule_probe.py.txt check_L, D1 pin_case): one N-section member
    (0,0,0)->(3,4,0); N0 translations rigid, N0 rotations held by three global rotational springs k = 1e-4 N*m/rad;
    tip moment contributions RX {1e80, 1e-8, -1e80} (exact sum 1e-8), with and without MY = 2e-8.  Expected values,
    scales and negative controls come from references.py's own tree-integration routine (solve/flatten/compute_scales)
    applied to this model: supplementary, not a frozen reference."""
    out = []
    k = Fr(1, 10 ** 4)
    for tag, extra in (('MIXED', {'N1': {'M': (Fr(0), Fr(2, 10 ** 8), Fr(0))}}), ('ONLY', {})):
        defn = {'nodes': {'N0': (Fr(0), Fr(0), Fr(0)), 'N1': (Fr(3), Fr(4), Fr(0))}, 'sections': {'N': m.SEC_N},
                'members': [('M1', 'N0', 'N1', 'N')],
                'supports': {'N0': {'rigid': ['UX', 'UY', 'UZ'],
                                    'springs': [('r', (1, 0, 0), k), ('r', (0, 1, 0), k), ('r', (0, 0, 1), k)]}},
                'loads': extra,
                'load_contributions': [('N1', 'RX', Fr(10) ** 80), ('N1', 'RX', Fr(1, 10 ** 8)), ('N1', 'RX', -Fr(10) ** 80)]}
        method = ('tree', 'N0')
        sol = m.solve(defn, method)
        flat = m.flatten(sol, defn)
        scales = m.compute_scales(sol, defn, m.Model(defn, m.Exact()))
        rows = [[kk, m.fmt(v), c, m.fmt(scales[c][0], 25), None] for (kk, v, c) in flat]
        # negative control: the binary64 left-to-right fold of the contributions (net 0) -> response to MY only
        fold = 0.0
        for (_, _, v) in defn['load_contributions']:
            fold += float(v)
        d2 = dict(defn, load_contributions=[('N1', 'RX', Fr(fold))])
        vals = {kk: v for (kk, v, c) in m.flatten(m.solve(d2, method), d2)}
        exp = {kk: v for (kk, v, c) in flat}
        smap = {kk: scales[c][0] for (kk, v, c) in flat}
        viol = m.compare_scaled(vals, exp, smap)
        ncs = [{'id': 'NC-FLOAT-SUM-GnG', 'description': 'binary64 left-to-right fold of (1e80, 1e-8, -1e80) = %r' % fold,
                'discriminates': bool(viol), 'violating_values': len(viol),
                'values': {kk: m.fmt(v, 25) for (_, kk, v) in viol[:12]}}]
        req, reason, meta = author_defn('V1-CHECK-L-' + tag, defn)
        out.append({'id': 'V1-CHECK-L-' + tag, 'family': 'V1-CHECK-L', 'request': req, 'meta': meta, 'rows': rows,
                    'expectation': 'solve', 'negative_controls': ncs, 'scale_rule': 'R1 class scales (compute_scales)',
                    'note': 'supplementary: V1 check L as a product model (tip loads %s)' % tag, 'refusal_expected': False})
    return out


def supplementary_cases(PI):
    cases = []
    EI = 1719500 * PI
    # S11 probe A: one 2 m cantilever element, uniform global-y loads (G, 0.3, -G) N/m
    for G in ('1e5', '1e6', '1e7', '1e8'):
        m = fixture_model('S11A-' + G, 2, force=None, torque=None)
        m['load_cases'][0]['primitive_loads'] = [
            {'id': 'udl:%d' % i, 'category': 'distributed_force', 'target': {'type': 'element', 'pipe': 'pipe'},
             'direction': 'global_y', 'magnitude': {'value': v, 'unit': 'N/m'}, 'dimension': 'force_per_length',
             'provenance': PROV} for i, v in enumerate((float(G), 0.3, -float(G)))]
        w, L = Fr(3, 10), Fr(2)
        rows = [['u.tip.UY', w * L ** 4 / (8 * EI), 'translation', None, None],
                ['th.tip.RZ', w * L ** 3 / (6 * EI), 'rotation', None, None],
                ['R.root.UY', -w * L, 'force', None, None],
                ['R.root.RZ', -w * L * L / 2, 'moment', None, None],
                ['Mb.pipe.i', w * L * L / 2, 'moment', None, None],
                ['Vb.pipe.i', w * L, 'force', None, None]]
        rows = [[k, format_frac(v), c, '0', 'net'] for (k, v, c, _, _) in rows]
        cases.append({'id': 'S11-PROBE-A-G' + G, 'family': 'S11-PROBE-A', 'request': {'model': m, 'materials': []},
                      'meta': fixture_meta(m),
                      'rows': rows, 'expectation': 'solve',
                      'note': 'supplementary (not a frozen reference): textbook cantilever under the exact net w = 0.3 N/m, '
                              'EI = 1719500*PI_Q; scale = |exp| (net-governed); V1 S11_CHECK probe A as a product model',
                      'refusal_expected': False, 'scale_rule': 'net-governed: max(|exp|, 0)'})
    return cases


def format_frac(q, sig=40):
    from decimal import Decimal, localcontext
    q = Fr(q)
    if q == 0:
        return '0'
    with localcontext() as c:
        c.prec = sig + 10
        d = Decimal(q.numerator) / Decimal(q.denominator)
    return format(d, '.%de' % (sig - 1))


def main():
    ref_py, ref_json, fixtures_json, outdir = sys.argv[1:5]
    os.makedirs(os.path.join(outdir, 'requests'), exist_ok=True)
    m = load_refs(ref_py)
    refs = json.load(open(ref_json))['cases']
    fx = json.load(open(fixtures_json))
    out = []
    for case in m.CASES:
        cid = case['id']
        rec = refs[cid]
        req, reason, meta = author_defn(cid, case['defn'], case.get('units'))
        entry = {'id': cid, 'family': case['family'], 'meta': meta, 'basis': rec.get('basis'),
                 'mechanism': bool(case.get('mechanism')),
                 'expected_outcome': rec.get('expected_outcome'),
                 'negative_controls': rec['negative_controls'],
                 'flags': rec.get('flags')}
        if case.get('mechanism'):
            entry['rows'], entry['scale_rule'] = [], 'refusal required'
            entry['expectation'] = 'refuse'
        else:
            entry['rows'], entry['scale_rule'] = ref_rows(rec)
            entry['expectation'] = 'solve'
        if req is None:
            entry['not_authorable'] = reason
        else:
            path = os.path.join(outdir, 'requests', cid + '.json')
            with open(path, 'w') as fh:
                json.dump(req, fh, indent=None, separators=(',', ':'))
            entry['request_file'] = 'requests/' + cid + '.json'
            entry['members'] = len(case['defn']['members'])
        out.append(entry)
    # V1 check L: N05-class skew is not re-authored (no model in the V1 record beyond its emulation);
    # the product-level equivalent of its load pattern is RF-CANCEL-*-G1e80 (1e80, 1e-8, -1e80).
    for c in fixture_cases(fx, Fr(m.PI_Q)) + check_l_cases(m) + supplementary_cases(Fr(m.PI_Q)):
        entry = {'id': c['id'], 'family': c['family'], 'rows': c['rows'], 'expectation': c['expectation'],
                 'note': c['note'], 'scale_rule': c.get('scale_rule', ''),
                 'negative_controls': c.get('negative_controls', []), 'meta': c.get('meta'),
                 'mechanism': c.get('refusal_expected', False)}
        if c.get('not_authorable'):
            entry['not_authorable'] = c['not_authorable']
        else:
            path = os.path.join(outdir, 'requests', c['id'] + '.json')
            with open(path, 'w') as fh:
                json.dump(c['request'], fh, separators=(',', ':'))
            entry['request_file'] = 'requests/' + c['id'] + '.json'
            entry['members'] = len(c['request']['model']['pipe_segments'])
        out.append(entry)
    with open(os.path.join(outdir, 'cases.json'), 'w') as fh:
        json.dump(out, fh, indent=1)
    print('cases', len(out), 'authorable', sum(1 for e in out if 'request_file' in e),
          'not_authorable', sum(1 for e in out if 'not_authorable' in e))


if __name__ == '__main__':
    main()
