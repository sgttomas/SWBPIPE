"""K6 B2 driver (ROOT's grant "K6: B1 accepted; B2 granted").

It runs the committed runner (k6_runner.py at the K6 branch commit given below)
unchanged, and adds only two things, both from the grant:
- every record it writes carries `runner_commit`;
- grid 128x128, admitted by ROOT, is no longer deferred as conditional (N14);
  every other rule, including admission and the ascent, is unchanged.
Order: T3b (dense and lane-lu at 1,000 members), then T5's grid 128x128.

Usage: python3 b2_driver.py <runner dir> <runner commit> <binary> <counts> <records> <source commit> <source tree>
"""
import json
import sys

runner_dir, runner_commit, binary, counts, records, source_commit, source_tree = sys.argv[1:8]
sys.path.insert(0, runner_dir)
import k6_runner as r  # noqa: E402

original_append = r.append_record


def append_with_runner_commit(record_dir, record):
    original_append(record_dir, dict(record, runner_commit=runner_commit))


r.append_record = append_with_runner_commit
r.CONDITIONAL_MODELS = ()
outcomes = {}
for tier in ('T3b', 'T5'):
    print('=== tier %s' % tier, flush=True)
    outcomes[tier] = r.run_tier(tier, binary, counts, records, source_commit, source_tree,
                                log=lambda m: print(m, flush=True))
    print(json.dumps({tier: outcomes[tier]}), flush=True)
    if outcomes[tier]['stop']:
        break
print('B2 done %s' % json.dumps(outcomes), flush=True)
