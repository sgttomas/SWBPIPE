#!/bin/bash
# RV18 fixes: one runner run from a clean `git archive ae3320b5a` copy (H and the DEC-025 wrapper), with I15's
# changed files overlaid, the mutant applied, and the wrapper's pytest (the N15 invocation with --noconftest, as at C).
W=<wt>; R=$W/scratch/i15/rv18
VENV=<VENV>
M=$1; D=$W/k6-mut/rv18-py-$M; LOG=$R/logs/py-$M.log
rm -rf "$D"; mkdir -p "$D"
{
  echo "=== py $M: git archive ae3320b5a93aa0e65a71483b4db6f40eeb9d8f1c (performance_harness, the wrapper) + overlay (I15's RV18 fixes)"
  git -C $W/k6 archive ae3320b5a93aa0e65a71483b4db6f40eeb9d8f1c projects/chirality-piping/core/solver/performance_harness \
      projects/chirality-piping/tests/test_performance_harness_runner.py | tar -x -C "$D"
  (cd $R/overlay && find . -type f | sort | while read f; do cp "$f" "$D/$f"; echo "overlay $f $(shasum -a 256 "$f" | cut -c1-16)"; done)
  case "$M" in NONE) ;; RV18-*|K6-M4r) python3 $R/rv18_mutants.py make "$M" "$D" ;; *) python3 $R/own_mutants.py make "$M" "$D" ;; esac
  echo "=== <VENV>/bin/python -m pytest -q -p no:cacheprovider --noconftest tests/test_performance_harness_runner.py"
} > "$LOG" 2>&1
cd "$D/projects/chirality-piping" || exit 9
s=$(date +%s)
"$VENV/bin/python" -m pytest -q -p no:cacheprovider --noconftest tests/test_performance_harness_runner.py >> "$LOG" 2>&1
echo "=== exit $? elapsed $(( $(date +%s) - s )) s" >> "$LOG"
sed -i '' -e "s#$VENV#<VENV>#g" -e "s#$W#<wt>#g" -e 's/[[:space:]]*$//' "$LOG"
