def classify_rs(text):
    if re.search(r'^error(\[E\d+\])?: could not compile|^error\[E\d+\]', text, re.M):
        return 'not_viable', ['compile error']
    failed = re.findall(r'^test (\S+) \.\.\. FAILED', text, re.M)
    targets = re.findall(r'^\s+`(--(?:test|lib|bin)[^`]*)`', text, re.M)
    if 'k6_alloc: F1 accounting and F2 cap boundary passed' not in text and 'Running tests/k6_alloc.rs' in text:
        failed.append('k6_alloc (harness=false: %s)' % (
            (re.search(r"panicked at [^\n]*\n([^\n]*)", text) or [None, 'aborted'])[1].strip()[:120]))
    elif 'Running tests/k6_alloc.rs' not in text:
        failed.append('k6_alloc not run')
    exit_ok = re.search(r'^=== exit 0 ', text, re.M) is not None
    if failed or targets or not exit_ok:
        return 'killed', failed + (['failed targets: ' + ', '.join(targets)] if targets else [])
    return 'survived', []


def classify_py(text):
    failed = re.findall(r'^FAILED \S+::(\S+::\S+)', text, re.M)
    summary = re.search(r'^(\d+ failed.*|\d+ passed.*)$', text, re.M)
    if re.search(r'^E\s+.*Error|^ERROR ', text, re.M) and not failed:
        return 'killed', ['collection or setup error']
    if failed:
        return 'killed', failed
    return 'survived', [summary.group(1) if summary else 'no summary']


def main(logs):
    rows = []
    for kind in ('py', 'rs'):
        path = os.path.join(logs, '%s-NONE.log' % kind)
        text = open(path).read()
        status, detail = (classify_py if kind == 'py' else classify_rs)(text)
        rows.append({'id': 'NONE (%s)' % kind, 'kind': kind, 'what': 'control: unmutated archive copy',
                     'status': 'passes' if status == 'survived' else 'CONTROL FAILED', 'tests': detail,
                     'intended': '-', 'elapsed': (re.search(r'^=== exit (\d+) elapsed (\d+) s', text, re.M)
                                                  or [None, None, None])[2]})
    for mid, (kind, what, rel, old, new) in mutants.MUTANTS.items():
        path = os.path.join(logs, '%s-%s.log' % (kind, mid))
        if not os.path.exists(path):
            rows.append({'id': mid, 'kind': kind, 'what': what, 'status': 'not run', 'tests': [],
                         'intended': INTENDED[mid], 'elapsed': None})
            continue
        text = open(path).read()
        status, detail = (classify_py if kind == 'py' else classify_rs)(text)
        m = re.search(r'^=== exit (\d+) elapsed (\d+) s', text, re.M)
        rows.append({'id': mid, 'kind': kind, 'what': what, 'status': status, 'tests': detail,
                     'intended': INTENDED[mid], 'elapsed': m.group(2) if m else None})
    with open(os.path.join(logs, '..', 'mutation_table.json'), 'w') as fh:
        json.dump(rows, fh, indent=1)
        fh.write('\n')
    print('| Mutant | Side | Mutation | Result | Killing tests | Intended kill |')
    print('|---|---|---|---|---|---|')
    for row in rows:
        tests = '; '.join(row['tests'])
        print('| %s | %s | %s | %s | %s | %s |' % (row['id'], row['kind'], row['what'], row['status'],
                                                  tests[:400], row['intended']))
    counts = {}
    for row in rows:
        counts[row['status']] = counts.get(row['status'], 0) + 1
    print('\n%s' % json.dumps(counts, sort_keys=True))
    print('\nNotes:')
    for note in NOTES:
        print('- ' + note)


if __name__ == '__main__':
    main(sys.argv[1])
