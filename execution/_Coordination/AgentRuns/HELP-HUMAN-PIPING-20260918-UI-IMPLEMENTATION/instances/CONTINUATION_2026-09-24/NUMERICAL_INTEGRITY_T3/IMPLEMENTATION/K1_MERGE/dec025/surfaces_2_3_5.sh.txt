#!/bin/bash
# The sweep's surfaces 2, 3 and 5, run with its exact commands, because run_evidence_sweep.py stops at
# surface 1, whose cargo failures on this Mac are the three committed-Linux-byte platform tests.
SW=$1; L=$2; V=$3; cd $SW
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_BUILD_JOBS=8 RUST_TEST_THREADS=4 PATH=$(dirname $V):$PATH
run() { name=$1; shift; echo "== $name: $* ($(date -u +%T)Z)" >> $L/surfaces.txt; "$@" > $L/$name.log 2>&1; rc=$?; echo "   exit $rc ($(date -u +%T)Z)" >> $L/surfaces.txt; }
run python_pytest $V -m pytest -q tests
run desktop_vitest_build_wasm npm run build:wasm:desktop
run desktop_vitest npm run test:desktop
run desktop_production_build npm run build:desktop
echo SURFACES-DONE >> $L/surfaces.txt
