def e4(core):
    """K-D5's RV5 evasion E4, reproduced from KD5/_run_records/repair/mutations/mutate.py.txt."""
    helper = """
    /// RV5 evasion helper: the typed formation-checked entry behind a neutral name.
    pub(crate) fn solve_binary64_audited(
        &self,
        k: &[Vec<f64>],
        f: &[f64],
        free: &[usize],
        prescribed: &[(usize, f64)],
        mode: LinearSolveMode,
    ) -> Result<StructuralSolution, StructuralError> {
        let mut ledger = open_pipe_stress_frame_kernel::load_ledger::LoadLedger::new();
        for (dof, &v) in f.iter().enumerate() {
            if v != 0.0 {
                ledger.push("rv5", dof, v);
            }
        }
        let typed = ledger.finish(f.len()).unwrap();
        self.solve_assembled_with_formation_check(k, &typed, free, prescribed, mode, &[], true)
    }
"""
    anchor = "    /// Named, unchanged binary64 variant of `solve` for the nonlinear active-set\n"
    sa = open(SA).read()
    assert sa.count(anchor) == 1
    open(SA, "w").write(sa.replace(anchor, helper + anchor))
    nl = open(NL).read()
    m = re.search(r"\nfn solve_linearized_system_evidence\(.*?\n}\n", nl, re.S)
    body = m.group(0).replace("fn solve_linearized_system_evidence(", "pub(crate) fn solve_linearized_system_evidence_unit(")
    assert body.count(LOOP_CALL) == 1
    body = body.replace(LOOP_CALL, LOOP_CALL.replace("assembly.solve_binary64(", "assembly.solve_binary64_audited("))
    open(core + "/solver/nonlinear_integration/src/unit_bridge.rs", "w").write("#![allow(unused_imports)]\nuse super::*;\n" + body)
    for old, new in (("pub mod product_equilibrium;\n", "pub mod product_equilibrium;\nmod unit_bridge;\n"),
                     ("            let unit = solve_linearized_system_evidence(\n                Some(assembly),",
                      "            let unit = unit_bridge::solve_linearized_system_evidence_unit(\n                Some(assembly),")):
        assert nl.count(old) == 1, (mid, old)
        nl = nl.replace(old, new)
    open(NL, "w").write(nl)


if mid == "KD5-E4":
    e4(core)
elif mid != "NONE":
    for path, old, new in MUTANTS[mid]:
        text = open(path).read()
        assert text.count(old) == 1, (mid, path, text.count(old))
        open(path, "w").write(text.replace(old, new))
print("applied", mid)
