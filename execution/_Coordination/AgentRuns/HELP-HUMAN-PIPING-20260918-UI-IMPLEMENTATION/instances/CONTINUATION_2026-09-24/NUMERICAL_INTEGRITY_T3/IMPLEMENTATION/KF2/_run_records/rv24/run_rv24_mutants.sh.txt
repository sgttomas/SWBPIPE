#!/bin/bash
# KF2 RV24-1: RV24's mutants (their rv24_mutants.py edit texts, unchanged) against the committed tests
# plus the adopted RV24 cases. One clean `git archive f2b8c85a2` copy of P/core per mutant, the new
# kf2_witness_tests.rs overlaid, one edit, its own target; one at a time; copy and target deleted after.
set -u
WT=<wt>
KF2=$WT/kf2
S=$WT/scratch/i20/rv24
OUT=$S/mut
mkdir -p $OUT
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 RUST_TEST_THREADS=2
FKREL=projects/chirality-piping/core/solver/frame_kernel
for m in "$@"; do
  D=$WT/kf2-mut/$m
  L=$OUT/$m.log
  rm -rf "$D"; mkdir -p "$D"
  git -C "$KF2" archive f2b8c85a2 projects/chirality-piping/core | tar -x -C "$D"
  cp "$KF2/$FKREL/src/structural/kf2_witness_tests.rs" "$D/$FKREL/src/structural/kf2_witness_tests.rs"
  { echo "== mutant $m $(date -u +%FT%TZ)"; uptime; } > "$L"
  python3 $S/rv24_mutants.py "$m" "$D/$FKREL/src/structural.rs" >> "$L" 2>&1
  diff -u "$KF2/$FKREL/src/structural.rs" "$D/$FKREL/src/structural.rs" | sed "s#$WT#<wt>#g" >> "$L"
  (cd "$D/$FKREL" && CARGO_TARGET_DIR="$D/target" cargo test --offline --locked -j 4 --lib --no-run) > "$OUT/$m.build.log" 2>&1
  echo "build_exit $?" >> "$L"
  BIN=$(grep -o 'Executable unittests src/lib.rs ([^)]*)' "$OUT/$m.build.log" | sed 's/.*(\(.*\))/\1/')
  (cd "$D/$FKREL" && perl -e 'alarm 600; exec @ARGV' "$BIN" kf2_ krev04 k1_negative_witness) >> "$L" 2>&1
  echo "lib_exit $?" >> "$L"
  (cd "$D/$FKREL" && CARGO_TARGET_DIR="$D/target" cargo test --offline --locked -j 4 --test s11_site_table) >> "$L" 2>&1
  echo "site_exit $?" >> "$L"
  rm -rf "$D"
  echo "$m done: $(grep -E '^(build|lib|site)_exit' "$L" | tr '\n' ' ') failed: $(grep -E '^test .* FAILED$' "$L" | sed 's/^test //; s/ \.\.\. FAILED//; s/structural::kf2_witness_tests:://' | tr '\n' ' ')"
done
rmdir $WT/kf2-mut 2>/dev/null || true
