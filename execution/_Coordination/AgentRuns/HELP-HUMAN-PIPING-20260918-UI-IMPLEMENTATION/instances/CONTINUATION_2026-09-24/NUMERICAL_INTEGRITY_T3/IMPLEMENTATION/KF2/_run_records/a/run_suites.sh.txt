#!/bin/bash
# KF2 checkpoint A: the callers' suites, as CI runs them (cargo test --offline per manifest), one at a time.
WT=<wt>
C=$WT/kf2/projects/chirality-piping/core
R=$WT/scratch/i20/a
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=$WT/kf2-target RUST_TEST_THREADS=2
for crate in "$@"; do
  case $crate in
    SD) m=$C/solver/sparse_direct ;;
    NI) m=$C/solver/nonlinear_integration ;;
    H) m=$C/solver/performance_harness ;;
    PP) m=$C/product_physics ;;
  esac
  { echo "== $crate $(date -u +%FT%TZ)"; uptime; } > $R/suite_$crate.log
  ( time cargo test --offline --locked -j 4 --manifest-path $m/Cargo.toml ) >> $R/suite_$crate.log 2>&1
  echo "exit $?" >> $R/suite_$crate.log
  uptime >> $R/suite_$crate.log
  echo "$crate done: $(grep -c 'test result: ok' $R/suite_$crate.log) ok-lines, $(grep -c 'FAILED' $R/suite_$crate.log) FAILED, $(tail -2 $R/suite_$crate.log | head -1)"
done
