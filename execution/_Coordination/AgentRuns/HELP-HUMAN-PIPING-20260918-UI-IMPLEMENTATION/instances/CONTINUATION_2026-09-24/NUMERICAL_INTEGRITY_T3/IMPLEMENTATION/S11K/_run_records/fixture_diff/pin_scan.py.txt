"""List files that pin the named fixtures by name or by sha256 (stdlib only)."""
import hashlib, os, sys, json, re
repo = sys.argv[1]
targets = [
 "fixtures/results/preview_physics_invented_dense.json",
 "fixtures/results/preview_physics_invented_sparse.json",
 "fixtures/product_preview/load_reference/connected-dense_scrutiny.raw.json",
 "fixtures/product_preview/load_reference/connected-sparse_interactive.raw.json",
 "fixtures/product_preview/load_reference_source/eigen_motion-dense_scrutiny.raw.json",
 "fixtures/product_preview/load_reference_source/eigen_motion-sparse_interactive.raw.json",
 "core/reporting/result_export/tests/fixtures/load_reference_fallback_uz-dense_scrutiny.raw.json",
 "core/reporting/result_export/tests/fixtures/load_reference_fallback_uz-sparse_interactive.raw.json",
]
# derived documents of A and B1
for f in sorted(os.listdir(os.path.join(repo, "fixtures/results"))):
    if re.match(r"load_reference_(connected|source_eigen_motion)_", f):
        targets.append("fixtures/results/" + f)
info = {}
for t in targets:
    data = open(os.path.join(repo, t), "rb").read()
    info[t] = {"sha256": hashlib.sha256(data).hexdigest(), "basename": os.path.basename(t), "pins": []}
exts = (".rs", ".py", ".ts", ".tsx", ".mjs", ".js", ".json", ".toml", ".yaml", ".yml", ".md", ".txt", ".sh")
for dp, dn, fn in os.walk(repo):
    dn[:] = [d for d in dn if d not in ("target", "node_modules", ".git", "execution")]
    for f in fn:
        if not f.endswith(exts): continue
        p = os.path.join(dp, f)
        rel = os.path.relpath(p, repo)
        try: text = open(p, encoding="utf-8", errors="ignore").read()
        except Exception: continue
        for t, v in info.items():
            if rel == t: continue
            stem = v["basename"].rsplit(".json", 1)[0]
            hits = []
            if v["basename"] in text: hits.append("name")
            elif stem in text: hits.append("stem")
            if v["sha256"] in text: hits.append("sha256")
            if hits: v["pins"].append({"file": rel, "by": hits})
print(json.dumps(info, indent=1))
