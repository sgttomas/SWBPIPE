#!/bin/bash
W=<wt>; B=$W/scratch/i10/b
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0
for side in base cand; do
  ( cd $B/t9/h_$side && CARGO_TARGET_DIR=$W/k2b-target/t9_$side cargo build --release --offline --locked -j 8 > $B/t9/build_$side.log 2>&1; echo "build rc=$?" >> $B/t9/build_$side.log )
  BIN=$W/k2b-target/t9_$side/release/calib_fixdiff_native
  : > $B/t9/run_$side.log
  for root in core fixtures validation; do
    mkdir -p $B/t9/out_$side/$root
    $BIN $B/trees/$side/projects/chirality-piping/$root $B/t9/out_$side/$root >> $B/t9/run_$side.log 2>&1; echo "$root rc=$?" >> $B/t9/run_$side.log
  done
  ( cd $B/t9/out_$side && find . -type f -name '*.out' | sort | xargs shasum -a 256 ) > $B/t9/sha_$side.txt
done
echo T9-DONE
