#!/usr/bin/env python3
"""K1 records (I8R, addenda): copy evidence files into a new _run_records/ subfolder, sanitized as
assemble_run_records.py does (placeholders <scratch>, <wt>, <home>; in logs, lines over 600 characters
cut with a marker naming the characters removed and the sha256 of the full line), and, as ROOT did at
the first records commit, with trailing spaces and trailing blank lines removed so that
`git diff --check` is clean. Usage: sanitize_copy.py <wt> <scratch> <out-dir> <src>=<rel> ..."""
import hashlib, os, sys
wt, scratch, out = sys.argv[1], sys.argv[2], sys.argv[3]
home = os.path.expanduser("~")
LIMIT = 600
UNCUT = ("MUTANTS.txt", "kill_sites.txt", ".py.txt", ".sh.txt", "overlay.txt")
for arg in sys.argv[4:]:
    src, rel = arg.split("=", 1)
    text = open(src, encoding="utf-8", errors="replace").read()
    text = text.replace(scratch, "<scratch>").replace(wt, "<wt>").replace(home, "<home>")
    cut = not rel.endswith(UNCUT)
    lines = []
    for line in text.split("\n"):
        line = line.rstrip(" \t")
        if cut and len(line) > LIMIT:
            digest = hashlib.sha256(line.encode()).hexdigest()
            line = line[:LIMIT] + f" [... K1-records cut: {len(line) - LIMIT} more characters; sha256 of the full line {digest}]"
        lines.append(line)
    while lines and lines[-1] == "":
        lines.pop()
    dst = os.path.join(out, rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    with open(dst, "w", encoding="utf-8") as f:
        f.write("\n".join(lines) + "\n")
print("done")
