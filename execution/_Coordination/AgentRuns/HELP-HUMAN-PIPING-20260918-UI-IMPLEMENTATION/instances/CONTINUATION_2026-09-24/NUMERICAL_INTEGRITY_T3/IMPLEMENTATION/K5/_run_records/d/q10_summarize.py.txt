"""I14 (K5, D): Q10's observation from the instrumented logs. Usage: q10_summarize.py <ni-log> <pp-log>
Per test (NI) or per product-run input (PP; the K5Q10 lines before each K5RUN line), each distinct
curved slot record: explicit, zero entries, subnormal entries, smallest nonzero |entry| (of 144)."""
import re, sys
def rec(line):
    m = re.search(r"slot=(\S+) explicit=(\S+) zeros=(\d+) subnormal=(\d+) least_nonzero=(\S+)", line)
    return m.groups()
out = []
ni = {}
for line in open(sys.argv[1], errors="replace"):
    if line.startswith("K5Q10 "):
        test = re.search(r"test=(\S+)", line).group(1).split("::")[-1]
        ni.setdefault(test, set()).add(rec(line))
pp, pending = {}, []
for line in open(sys.argv[2], errors="replace"):
    if line.startswith("K5Q10 "):
        pending.append(rec(line))
    elif line.startswith("K5RUN "):
        label = line.split()[1]
        pp.setdefault(label, set()).update(pending)
        pending = []
sub_total = 0
out.append("# NI k5_tests (the SA corpus): test | slot explicit zeros subnormal least_nonzero")
for test in sorted(ni):
    for r in sorted(ni[test]):
        out.append(f"{test} | {' '.join(r)}"); sub_total += int(r[3])
out.append("# PP product-run inputs (a2 table's corpus): input | slot explicit zeros subnormal least_nonzero")
for label in sorted(pp):
    for r in sorted(pp[label]):
        out.append(f"{label} | {' '.join(r)}"); sub_total += int(r[3])
if pending:
    out.append(f"# {len(pending)} trailing records without a K5RUN line")
out.append(f"# distinct records: NI {sum(len(v) for v in ni.values())}, PP {sum(len(v) for v in pp.values())}; subnormal entries in all records: {sub_total}")
print("\n".join(out))
