"""I13 F1b pin (ROOT's approval of the pressure_thrust_load product pin): the exact requests that
tests/f1b_w2_runtime.rs's `cantilever_family("pressure_thrust_load", s)` builds (the T0R shape), for
s = 2^-1000 (the range case) and s = 1 (its b = 0 twin), replicated field for field from the Rust
builders `t0r`, `tnode`, `tpipe`, `tforce` and `anchor`, to record main's envelope hash for the twin."""
import json
PROV = "invented_t3_f1b_w2_runtime_input_no_library_data"
ALL = ["UX", "UY", "UZ", "RX", "RY", "RZ"]
def tnode(i, x, y): return {"id": i, "position": {"x": x, "y": y, "z": 0.0}, "provenance": PROV}
def tpipe(i, a, b, y): return {"id": i, "from": a, "to": b, "section": {"outside_diameter": {"value": 0.12, "unit": "m"},
    "wall_thickness": {"value": 0.01, "unit": "m"}}, "material": "material:m",
    "y_reference": {"x": y[0], "y": y[1], "z": y[2]}, "provenance": PROV}
def tforce(node, d, v): return {"id": "f", "category": "concentrated_force", "target": {"type": "node", "node": node},
    "direction": d, "magnitude": {"value": v, "unit": "N"}, "dimension": "force", "provenance": PROV}
def anchor(node, r): return {"id": "anchor:" + node, "node": node, "family": "anchor", "restraints": r, "provenance": PROV}
def t0r(nodes, pipes, supports, components, loads, s):
    return {"model": {"schema_version": "0.2.0", "document_kind": "openpipestress.product_preview.model",
        "project": {"id": "project:t3-f1b-w2", "units": {"length": "m", "force": "N", "angle": "rad",
            "pressure": "Pa", "stress": "Pa", "temperature": "degC"}},
        "analysis_status": {"mechanics": "ready_for_preview_diagnostics",
            "rule_check": "not_performed_user_rule_inputs_missing", "professional_acceptance": "not_provided"},
        "nodes": nodes, "pipe_segments": pipes, "supports": supports, "components": components,
        "materials": [{"id": "material:m", "elastic_modulus": {"value": 200e9 * s, "unit": "Pa"},
            "shear_modulus": {"value": 76.923076923e9 * s, "unit": "Pa"},
            "thermal_expansion_coefficient": {"value": 1.2e-5, "unit": "1/degC"}, "provenance": PROV}],
        "load_cases": [{"id": "case:f", "primitive_loads": loads, "provenance": PROV}],
        "combinations": []}, "materials": []}
def pressure_family(s):
    loads = [tforce("node:b", "global_y", 100.0),
             {"id": "p", "category": "pressure", "target": {"type": "element", "pipe": "pipe:a-b"},
              "direction": "global_x", "magnitude": {"value": 0.0, "unit": "Pa"}, "dimension": "pressure",
              "provenance": PROV}]
    return t0r([tnode("node:a", 0.0, 0.0), tnode("node:b", 1.0, 0.0)],
               [tpipe("pipe:a-b", "node:a", "node:b", [0.0, 1.0, 0.0])],
               [anchor("node:a", ALL)], [], loads, s)
json.dump(pressure_family(2.0 ** -1000), open("pressure_thrust_range.json", "w"), indent=1)
json.dump(pressure_family(1.0), open("pressure_thrust_b0.json", "w"), indent=1)
