// ---------------------------------------------------------------- SCRATCH ONLY
// I13 F1b lane guard: the estimate against the lane's own profile, corpus-wide
// (every JSON request under F1B_CORPUS with at most 1,001 nodes); above that,
// the estimate alone (the lane's profile is never formed here).
fn scratch_walk(dir: &std::path::Path, out: &mut Vec<std::path::PathBuf>) {
    let mut entries: Vec<_> = std::fs::read_dir(dir).unwrap().map(|e| e.unwrap().path()).collect();
    entries.sort();
    for p in entries {
        if p.is_dir() {
            if p.file_name().is_some_and(|n| n == "target" || n == "node_modules") { continue; }
            scratch_walk(&p, out);
        } else if p.extension().is_some_and(|e| e == "json") {
            out.push(p);
        }
    }
}

#[test]
fn scratch_corpus_lane_estimate() {
    use std::io::Write;
    let dirs = std::env::var("F1B_CORPUS").unwrap();
    let mut out = std::fs::File::create(std::env::var("F1B_CORPUS_OUT").unwrap()).unwrap();
    let mut files = Vec::new();
    for d in dirs.split(':') { scratch_walk(std::path::Path::new(d), &mut files); }
    let (mut equal, mut mismatch, mut estimate_only) = (0, 0, 0);
    for file in files {
        let Ok(text) = std::fs::read_to_string(&file) else { continue };
        let Ok(value) = serde_json::from_str::<Value>(&text) else { continue };
        let request = if value.get("model").is_some_and(Value::is_object) { value }
            else if value.get("document_kind").and_then(|k| k.as_str()) == Some("openpipestress.product_preview.model") { json!({"model": value, "materials": []}) }
            else { continue };
        let nodes = request["model"]["nodes"].as_array().map_or(0, |n| n.len());
        let name = file.display().to_string();
        let r = request.clone();
        let Ok(bases) = std::panic::catch_unwind(move || product_bases(&r)) else {
            writeln!(out, "{}", json!({"file": name, "result": "front_end_unwrap"})).unwrap();
            continue;
        };
        for basis in &bases {
            let Some(system) = observation_lane_system(basis) else { continue };
            let (entries, bandwidth) = observation_lane_profile(&system);
            if nodes > 1001 {
                estimate_only += 1;
                writeln!(out, "{}", json!({"file": name, "basis": basis.label, "nodes": nodes, "dimension": system.dimension, "estimate_entries": entries.to_string(), "estimate_bandwidth": bandwidth, "estimate_bytes": (24 * entries).to_string(), "result": "estimate_only"})).unwrap();
                continue;
            }
            let profile = open_pipe_stress_sparse_direct::SymmetricProfileMatrix::from_entries(system.dimension, &system.entries).unwrap();
            let same = entries == profile.profile_entry_count() as u128 && bandwidth == profile.max_half_bandwidth();
            if same { equal += 1 } else { mismatch += 1 }
            writeln!(out, "{}", json!({"file": name, "basis": basis.label, "nodes": nodes, "dimension": system.dimension, "estimate_entries": entries.to_string(), "lane_entries": profile.profile_entry_count(), "estimate_bandwidth": bandwidth, "lane_bandwidth": profile.max_half_bandwidth(), "result": if same {"equal"} else {"MISMATCH"}})).unwrap();
        }
    }
    writeln!(out, "{}", json!({"summary": {"equal": equal, "mismatch": mismatch, "estimate_only": estimate_only}})).unwrap();
    eprintln!("SCRATCH-LANE equal={equal} mismatch={mismatch} estimate_only={estimate_only}");
    assert_eq!(mismatch, 0);
}
