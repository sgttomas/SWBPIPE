#!/usr/bin/env python3
"""Standard-library port of frame_kernel/tests/s11_site_table.rs's scanner (lex,
strip_cfg_test, scan, self_assignment_at). Usage: site_scan.py <file.rs> [...]"""
import sys

def ident(c): return c.isalnum() or c == '_'

def lex(src):
    b = src; n = len(b); out = []; i = 0
    while i < n:
        c = b[i]
        if c == '/' and i+1 < n and b[i+1] == '/':
            while i < n and b[i] != '\n': i += 1
            continue
        if c == '/' and i+1 < n and b[i+1] == '*':
            depth = 1; i += 2
            while i < n and depth > 0:
                if b[i] == '/' and i+1 < n and b[i+1] == '*': depth += 1; i += 2
                elif b[i] == '*' and i+1 < n and b[i+1] == '/': depth -= 1; i += 2
                else:
                    if b[i] == '\n': out.append('\n')
                    i += 1
            continue
        prev_ident = i > 0 and ident(b[i-1])
        raw_start = None
        if not prev_ident and c == 'r': raw_start = i+1
        elif not prev_ident and c == 'b' and i+1 < n and b[i+1] == 'r': raw_start = i+2
        if raw_start is not None:
            j = raw_start; hashes = 0
            while j < n and b[j] == '#': hashes += 1; j += 1
            if j < n and b[j] == '"':
                j += 1
                while True:
                    if j >= n: break
                    if b[j] == '"' and all(j+1+h < n and b[j+1+h] == '#' for h in range(hashes)):
                        j += 1 + hashes; break
                    if b[j] == '\n': out.append('\n')
                    j += 1
                out.append('""'); i = j; continue
        if c == '"' or (not prev_ident and c == 'b' and i+1 < n and b[i+1] == '"'):
            j = i+2 if c == 'b' else i+1
            while j < n and b[j] != '"':
                if b[j] == '\\': j += 1
                if j < n and b[j] == '\n': out.append('\n')
                j += 1
            out.append('""'); i = j+1; continue
        if c == "'":
            if i+2 < n and b[i+1] != '\\' and b[i+2] == "'":
                out.append("' '"); i += 3; continue
            if i+1 < n and b[i+1] == '\\':
                j = i+2
                while j < n and b[j] != "'": j += 1
                out.append("' '"); i = j+1; continue
        out.append(c); i += 1
    return ''.join(out)

def strip_cfg_test(code):
    ATTR = "#[cfg(test)]"; out = []; rest = code
    while True:
        at = rest.find(ATTR)
        if at < 0: break
        out.append(rest[:at]); item = rest[at:]
        depth = 0; started = False; end = len(item)
        for k in range(len(ATTR), len(item)):
            ch = item[k]
            if ch == '{': depth += 1; started = True
            elif ch == '}':
                depth -= 1
                if started and depth == 0: end = k+1; break
            elif ch == ';' and not started: end = k+1; break
        out.append(''.join(ch for ch in item[:end] if ch == '\n'))
        rest = item[end:]
    out.append(rest)
    return ''.join(out)

def self_assignment_at(code, i):
    prev = code[i-1] if i > 0 else ' '
    nxt = code[i+1] if i+1 < len(code) else ' '
    if prev in "=!<>+-*/%&|^" or nxt == '=' or nxt == '>': return False
    starts = [code.rfind(c, 0, i) for c in ';{}\n']
    start = max(starts) + 1 if max(starts) >= 0 else 0
    lhs = code[start:i].strip()
    if lhs.startswith("let "):
        lhs = lhs[4:].lstrip()
        if lhs.startswith("mut "): lhs = lhs[4:].lstrip()
        if ':' in lhs: lhs = lhs[:lhs.index(':')].rstrip()
    if not lhs or not all(c.isalnum() or c in "_.[]* " for c in lhs): return False
    rhs = code[i+1:].lstrip()
    if rhs.startswith("checked_value("): rhs = rhs[len("checked_value("):].lstrip()
    if not rhs.startswith(lhs): return False
    after = rhs[len(lhs):]
    if after and (after[0].isalnum() or after[0] in '_[.'): return False
    after = after.lstrip()
    return after.startswith('+') or (after.startswith('-') and not after.startswith('->'))

def scan(src):
    code = strip_cfg_test(lex(src)); b = code; counts = {}; stack = []; pending = None
    depth = 0; nest = 0; i = 0
    while i < len(b):
        rest = code[i:]
        boundary = i == 0 or not ident(b[i-1])
        if boundary and rest.startswith("fn") and len(rest) > 2 and b[i+2].isspace():
            name = ''
            for ch in rest[2:].lstrip():
                if ch.isalnum() or ch == '_': name += ch
                else: break
            if name: pending = name
            i += 2; continue
        m = None
        for p in ["+=", "-=", ".sum(", ".sum::<"]:
            if rest.startswith(p): m = len(p); break
        if m is None and boundary and rest.startswith("fold("): m = 5
        if m is not None:
            name = stack[-1][0] if stack else "<module>"
            counts[name] = counts.get(name, 0) + 1
            if rest[:m].endswith('('): nest += 1
            i += m; continue
        if b[i] == '=' and self_assignment_at(code, i):
            name = stack[-1][0] if stack else "<module>"
            counts[name] = counts.get(name, 0) + 1
        ch = b[i]
        if ch in '([': nest += 1
        elif ch in ')]': nest = max(0, nest-1)
        elif ch == ';' and nest == 0: pending = None
        elif ch == '{':
            depth += 1
            if pending is not None: stack.append((pending, depth)); pending = None
        elif ch == '}':
            if stack and stack[-1][1] == depth: stack.pop()
            depth = max(0, depth-1)
        i += 1
    return counts

if __name__ == '__main__':
    for path in sys.argv[1:]:
        src = open(path).read() if path != '-' else sys.stdin.read()
        for k, v in sorted(scan(src).items()): print(f"{path}\t{k}\t{v}")
