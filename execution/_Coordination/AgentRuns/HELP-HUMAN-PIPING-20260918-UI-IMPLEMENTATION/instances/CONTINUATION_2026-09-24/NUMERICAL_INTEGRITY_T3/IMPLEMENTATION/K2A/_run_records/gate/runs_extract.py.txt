"""K2a (I6): a committed, sanitized extract of the uncommitted gate runs.jsonl (888 runs, 246 MB; sha256 in
runs_jsonl_sha256.txt). One JSON line per run: case, mode, entry, P1's outcome class (compare.classify),
numerical_quality status, wall seconds, and each blocking or integrity diagnostic's code and message (messages
cut at 400 characters), or the probe's error text. No results or envelopes.
Usage: python3 -B runs_extract.py <runs.jsonl> <out.jsonl>   (run beside P1's compare.py)"""
import json, sys, os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import compare as C
with open(sys.argv[2], 'w') as out:
    for line in open(sys.argv[1]):
        r = json.loads(line)
        run = r.get('probe', {}).get('run', {})
        outcome, _ = C.classify(r)
        row = {'case': r['case'], 'mode': r['mode'], 'entry': r['entry'], 'outcome': outcome,
               'wall_seconds': r.get('wall_seconds')}
        if run.get('ok'):
            env = run['envelope']
            row['quality'] = C.identity(env).get('numerical_quality_status')
            row['diagnostics'] = [[d.get('code'), d.get('message', '')[:400]] for d in env.get('diagnostics', [])
                                  if d.get('severity') == 'blocking' or 'INTEGRITY' in (d.get('code') or '')]
        else:
            row['error'] = str(run.get('error', r.get('probe', {}).get('error', '')))[:400]
        out.write(json.dumps(row, sort_keys=True) + '\n')
