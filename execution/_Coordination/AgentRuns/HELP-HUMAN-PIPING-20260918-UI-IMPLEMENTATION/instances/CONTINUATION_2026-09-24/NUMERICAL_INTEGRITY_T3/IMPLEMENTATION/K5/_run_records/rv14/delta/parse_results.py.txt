def is_test_path(path):
    return path.startswith('tests/') or '_tests.rs' in path or '/tests/' in path


def parse(id_):
    d = os.path.join(logs, id_)
    apply = open(os.path.join(d, 'apply.txt')).read()
    rows = []
    compile_error = False
    aborts = []
    for name in ('fk', 'ni', 'ni5', 'pp'):
        p = os.path.join(d, name + '.txt')
        if not os.path.exists(p):
            continue
        text = open(p, errors='replace').read()
        if re.search(r'^error(\[E\d+\])?: ', text, re.M) and 'could not compile' in text:
            compile_error = True
        for m in re.finditer(r"thread '([^']+)' \(\d+\) panicked at ([^:\n]+):(\d+):\d+:\n(.*)", text):
            test, path, line, msg = m.group(1), m.group(2), m.group(3), m.group(4).strip()
            rows.append((name, test, path, int(line), msg[:110]))
        for m in re.finditer(r'process didn\'t exit successfully: .*\((signal: \d+[^)]*)\)', text):
            aborts.append((name, m.group(1)))
        if 'memory allocation of' in text:
            aborts.append((name, 'allocation failure'))
    kills = [r for r in rows if is_test_path(r[2]) or (r[2].endswith('rigid_body.rs') and 'k5_unit_tests' in r[1])]
    product_panics = [r for r in rows if r not in kills]
    return apply, rows, kills, product_panics, compile_error, aborts


print('| Mutant | Site | Result | Killing tests (assertion site) |')
print('|---|---|---|---|')
for id_ in ids:
    apply, rows, kills, pp, ce, ab = parse(id_)
    if 'apply exit=0' not in apply:
        result = 'NOT APPLIED'
    elif id_ in ('NONE', 'D-NONE', 'D2-NONE', 'D3-NONE'):
        result = 'control passes' if not rows and not ce and not ab else 'CONTROL FAILS'
    elif ce and not kills:
        result = 'COMPILE ERROR (not a kill)'
    elif kills:
        result = 'KILLED'
    else:
        result = 'SURVIVED' if not pp and not ab else 'NOT COUNTED (product panic/abort only)'
    tests = sorted({'%s `%s` (%s:%d)' % (r[0], r[1].split('::')[-1], r[2].split('/')[-1], r[3]) for r in kills})
    extra = ''
    if pp:
        extra += ' — product panics: ' + '; '.join(sorted({'%s:%d' % (r[2].split('/')[-1], r[3]) for r in pp}))
    if ab:
        extra += ' — aborts: ' + '; '.join(sorted({'%s %s' % a for a in ab}))
    shown = '; '.join(tests[:4]) + (' (+%d more)' % (len(tests) - 4) if len(tests) > 4 else '')
    print('| %s | %s | %s | %s%s |' % (id_, SITES.get(id_, ''), result, shown or '—', extra))
