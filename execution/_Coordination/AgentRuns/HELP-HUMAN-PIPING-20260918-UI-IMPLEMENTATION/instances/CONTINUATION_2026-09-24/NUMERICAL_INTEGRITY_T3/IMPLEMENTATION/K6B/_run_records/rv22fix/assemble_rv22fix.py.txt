"""K6b D: move the scratch records into T3/IMPLEMENTATION/K6B/_run_records/ (standard library).

Usage: python3 assemble_records.py <scratch/i16> <_run_records destination>

- Placeholders only: machine paths become <scratch>, <wt>, <repo>, <home> and <tmp>; the host
  name becomes <host>. A final scan refuses to finish if any machine path or host name remains.
- The archived source copies (``src/``) are not records and are not moved; each folder's
  ``source.txt`` names the commit.
- A file over TRIM_BYTES is trimmed to its first HEAD_LINES lines, as ``<name>.head.txt``; its
  full size, line count and sha256 go in the folder's TRIMMED.txt.
- b2 keeps row 247 (the stop's evidence), its baseline, the index and the logs. Its other
  per-run files (W1-T1 to W1-T3, superseded by b3) are listed with their sha256 in DROPPED.txt.
- Scripts are renamed ``*.py.txt`` and ``*.sh.txt``, as K6's records are.
"""
import getpass
import hashlib
import os
import re
import socket
import subprocess
import sys

SRC, DST = sys.argv[1], sys.argv[2]
FOLDERS = ['rv22fix']
TRIM_BYTES = 512 * 1024
HEAD_LINES = 40
# The machine's own paths and names, found at run time so this script holds none of them.
T3 = os.path.dirname(os.path.dirname(os.path.abspath(SRC)))
HOME = os.path.expanduser('~')
REPO = T3.split(os.sep + '.claude' + os.sep)[0]
USER = getpass.getuser()
HOSTS = sorted({socket.gethostname(), subprocess.run(['scutil', '--get', 'LocalHostName'], stdout=subprocess.PIPE,
                                                     text=True).stdout.strip()} - {''})
PRIVATE, FOLDERS_TMP = '/' + 'private', '/' + 'var/folders'   # split so the script does not match itself
PATH_CHARS = '[A-Za-z0-9._/-]*'
SUBS = [(re.escape(T3 + os.sep + 'scratch'), '<scratch>'), (re.escape(T3), '<wt>'), (re.escape(REPO), '<repo>'),
        (re.escape(HOME), '<home>'), (re.escape(PRIVATE + '/tmp/') + PATH_CHARS, '<tmp>'),
        (re.escape(PRIVATE + FOLDERS_TMP + '/') + PATH_CHARS, '<tmp>'),
        (re.escape(FOLDERS_TMP + '/') + PATH_CHARS, '<tmp>')]
SUBS += [(re.escape(h) + r'(\.local)?', '<host>') for h in HOSTS]
FORBIDDEN = re.compile('|'.join([re.escape(HOME + os.sep), re.escape(PRIVATE + '/'), re.escape(FOLDERS_TMP)]
                                + [re.escape(h) for h in HOSTS] + [r'\b%s\b' % re.escape(USER)]),
                       re.I)


def sha256(path):
    h = hashlib.sha256()
    with open(path, 'rb') as fh:
        for block in iter(lambda: fh.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def scrub(text):
    for pattern, repl in SUBS:
        text = re.sub(pattern, repl, text)
    return text


def dest_name(name):
    for ext in ('.py', '.sh'):
        if name.endswith(ext):
            return name + '.txt'
    return name


def keep_b2(rel):
    parts = rel.split(os.sep)
    if parts[0] != 'records':
        return True
    name = parts[-1]
    return name in ('records.jsonl', 'metadata.json') or name.startswith(('247_', 'baseline_W1-T4'))


def main():
    if os.path.exists(DST):
        sys.exit('destination exists: ' + DST)
    for folder in FOLDERS:
        root = os.path.join(SRC, folder)
        trimmed, dropped = [], []
        for dirpath, dirnames, filenames in os.walk(root):
            dirnames[:] = sorted(d for d in dirnames if not (dirpath == root and d == 'src'))
            for name in sorted(filenames):
                path = os.path.join(dirpath, name)
                rel = os.path.relpath(path, root)
                if folder == 'b2' and not keep_b2(rel):
                    dropped.append((rel, os.path.getsize(path), sha256(path)))
                    continue
                out_dir = os.path.join(DST, folder, os.path.dirname(rel))
                os.makedirs(out_dir, exist_ok=True)
                with open(path, 'rb') as fh:
                    data = fh.read()
                text = scrub(data.decode('utf-8'))
                out = os.path.join(out_dir, dest_name(name))
                if len(data) > TRIM_BYTES:
                    lines = text.splitlines(keepends=True)
                    trimmed.append((rel, len(data), len(lines), sha256(path)))
                    text = ''.join(lines[:HEAD_LINES])
                    out = os.path.join(out_dir, name + '.head.txt')
                with open(out, 'w') as fh:
                    fh.write(text)
        if trimmed:
            with open(os.path.join(DST, folder, 'TRIMMED.txt'), 'w') as fh:
                fh.write('# Files over %d bytes, kept as their first %d lines (<name>.head.txt)\n'
                         '# path bytes lines sha256-of-the-full-file\n' % (TRIM_BYTES, HEAD_LINES))
                for rel, size, lines, digest in trimmed:
                    fh.write('%s %d %d %s\n' % (rel, size, lines, digest))
        if dropped:
            with open(os.path.join(DST, folder, 'DROPPED.txt'), 'w') as fh:
                fh.write('# b2: W1-T1 to W1-T3 per-run files, superseded by b3 (same source but the parity fix);\n'
                         '# records.jsonl indexes them and b2_report.txt summarizes them.\n'
                         '# path bytes sha256\n')
                for rel, size, digest in dropped:
                    fh.write('%s %d %s\n' % (rel, size, digest))
    bad = []
    for dirpath, _, filenames in os.walk(DST):
        for name in filenames:
            path = os.path.join(dirpath, name)
            with open(path) as fh:
                for n, line in enumerate(fh, 1):
                    if FORBIDDEN.search(line):
                        bad.append('%s:%d' % (os.path.relpath(path, DST), n))
    if bad:
        print('\n'.join(bad[:40]))
        sys.exit('machine paths or host names remain in %d lines' % len(bad))
    print('ok')


if __name__ == '__main__':
    main()
