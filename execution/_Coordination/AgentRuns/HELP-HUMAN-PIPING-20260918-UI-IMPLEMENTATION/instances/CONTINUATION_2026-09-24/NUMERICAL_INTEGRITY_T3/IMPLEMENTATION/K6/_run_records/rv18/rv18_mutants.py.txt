Usage: python3 rv18_mutants.py make <mutant> <copy root>   (applies the edit; fails unless it applies once)

    'RV18-M2': (MAIN, 'if repeat + 1 < args.repeats && projected.as_secs() >= budget {',
                'if repeat + 1 < args.repeats && projected.as_secs() >= budget.saturating_mul(1_000_000) {',
                'time-budget stop never fires (budget x 10^6)'),

    'K6-M4r': (RUNNER, "                    os.killpg(pgid, signal.SIGKILL)\n                    killed, kill_at = True, round(now, 3)",

}


def main():
    cmd, name, root = sys.argv[1:4]
    path, old, new, _ = MUTANTS[name]
    full = root + '/' + path
    text = open(full).read()
    assert text.count(old) == 1, (name, text.count(old))
    open(full, 'w').write(text.replace(old, new))
    print('applied', name)


if __name__ == '__main__':
    main()
