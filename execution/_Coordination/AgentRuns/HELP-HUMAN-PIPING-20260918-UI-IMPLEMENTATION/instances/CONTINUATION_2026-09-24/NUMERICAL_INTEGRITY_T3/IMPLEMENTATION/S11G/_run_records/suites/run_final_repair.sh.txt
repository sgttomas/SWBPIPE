#!/bin/sh
# T3 S11-G (I5) confirming run on the final repair tree: the full product_physics crate (includes s11g_tests and the site test), sequential, --no-fail-fast.
W=<wt>/s11g/projects/chirality-piping
O=$1
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=<s11g-target> CARGO_PROFILE_DEV_DEBUG=0 CARGO_PROFILE_TEST_DEBUG=0
A=$W/core/serialization/canonical_json/target/checked-json/release
export OPENPIPESTRESS_CHECKED_JSON_BIN=$A/openpipestress_jcs_ijson OPENPIPESTRESS_BINARY64_JSON_BIN=$A/openpipestress_jcs_binary64
export OPENPIPESTRESS_UNITS_BIN=$W/core/units/target/units-authority/release/openpipestress_units
for c in core/product_physics; do
  name=$(echo $c | tr '/' '_')
  if [ ! -f $W/$c/Cargo.toml ]; then echo "$c: no Cargo.toml" >> $O/summary.txt; continue; fi
  (cd $W/$c && nice cargo test --offline --locked --no-fail-fast > $O/$name.log 2>&1); rc=$?
  echo "$c exit=$rc $(grep -c '^test result: ok' $O/$name.log) ok-lines $(grep -c '^test result: FAILED' $O/$name.log) failed-lines" >> $O/summary.txt
  df -h / | tail -1 >> $O/summary.txt
done
echo DONE >> $O/summary.txt
