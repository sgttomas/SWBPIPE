"""K1 reference table: correctly rounded binary64 of exact Fraction sums/dots.
Independent of the Rust accumulator: uses Python's fractions.Fraction and a
separate nearest-even rounding (float(Fraction) is correctly rounded in CPython,
but we round explicitly to also apply the +0.0 underflow rule)."""
import random, struct, sys
from fractions import Fraction

def bits(x): return struct.unpack('<Q', struct.pack('<d', x))[0]
def frombits(b): return struct.unpack('<d', struct.pack('<Q', b))[0]

def round_fraction(q):
    """Nearest-even binary64 of exact rational q; +0.0 for zero/underflow; None if out of range."""
    if q == 0: return 0.0
    neg = q < 0
    a = -q if neg else q
    # find e with 2^e <= a < 2^(e+1)
    e = a.numerator.bit_length() - a.denominator.bit_length()
    if Fraction(2)**e > a: e -= 1
    if Fraction(2)**(e+1) <= a: e += 1
    quantum = max(e - 52, -1074)
    scaled = a / Fraction(2)**quantum
    m = scaled.numerator // scaled.denominator
    rem = scaled - m
    if rem > Fraction(1,2) or (rem == Fraction(1,2) and m % 2 == 1): m += 1
    if m == 0: return 0.0
    if m == 2**53: m //= 2; quantum += 1
    if m >= 2**52:
        biased = quantum + 52 + 1023
        if biased >= 2047: return None
        b = (biased << 52) | (m - 2**52)
    else:
        b = m
    v = frombits(b)
    return -v if neg else v

rng = random.Random(20260926)
def rand_double(lo, hi):
    e = rng.randint(lo, hi)
    m = rng.getrandbits(52)
    b = ((e + 1023) << 52) | m if e > -1023 else m
    v = frombits(b)
    return -v if rng.random() < 0.5 else v

sums = []
for k in range(160):
    n = rng.randint(1, 9)
    band = rng.choice([(-60, 60), (-1074 + 60, -1000), (900, 1000), (-300, 300), (-1022, -1015)])
    vals = [rand_double(*band) for _ in range(n)]
    if rng.random() < 0.4:  # force cancellation
        v = vals[0]; vals.append(-v); rng.shuffle(vals)
    if rng.random() < 0.2:
        vals.append(frombits(rng.randint(1, 2**52 - 1)) * rng.choice([1, -1]))
    exact = sum((Fraction(v) for v in vals), Fraction(0))
    r = round_fraction(exact)
    if r is None: continue
    sums.append((vals, r))

dots = []
for k in range(160):
    n = rng.randint(1, 7)
    band = rng.choice([(-60, 60), (-560, -520), (480, 510), (-1074 + 60, -1000), (-300, 300)])
    pairs = [(rand_double(*band), rand_double(*band)) for _ in range(n)]
    if rng.random() < 0.4:
        a, b = pairs[0]; pairs.append((-a, b)); rng.shuffle(pairs)
    exact = sum((Fraction(a) * Fraction(b) for a, b in pairs), Fraction(0))
    r = round_fraction(exact)
    if r is None: continue
    dots.append((pairs, r))

def h(x): return "0x%016x" % bits(x)
out = ["// Generated by gen_k1.py (Python fractions.Fraction, seed 20260926). Do not edit.",
       "pub(super) const RANDOM_SUMS: &[(&[u64], u64)] = &["]
for vals, r in sums:
    out.append("    (&[%s], %s)," % (", ".join(h(v) for v in vals), h(r)))
out.append("];")
out.append("pub(super) const RANDOM_DOTS: &[(&[(u64, u64)], u64)] = &[")
for pairs, r in dots:
    out.append("    (&[%s], %s)," % (", ".join("(%s, %s)" % (h(a), h(b)) for a, b in pairs), h(r)))
out.append("];")
# Named cases
named = {}
def named_case(name, exact_expr):
    named[name] = round_fraction(exact_expr)
named_case("probe_d", Fraction(2)**-110 + Fraction(2)**-53 + 1)
named_case("thermal", Fraction(1.3) + Fraction(4.1e7) - Fraction(4.1e7))
named_case("probe_x_products", Fraction(1.3) + Fraction(1.35)*Fraction(4.1e7) - Fraction(1.35)*Fraction(4.1e7))
named_case("three_2m1076", 3 * Fraction(2)**-1076)
named_case("tie_2m1075", Fraction(2)**-1075)
named_case("neg_2m1080", -Fraction(2)**-1080)
for name, v in named.items():
    out.append("pub(super) const %s: u64 = %s; // %r" % (name.upper(), h(v), v))
open(sys.argv[1], 'w').write("\n".join(out) + "\n")
print(len(sums), len(dots), named)
