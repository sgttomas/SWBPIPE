#!/usr/bin/env python3
"""K3 records (I11, checkpoint D): copy the run evidence into _run_records/.

Sanitizing, and nothing else:
- machine paths become placeholders: <VENV> (the repository venv), <scratch>
  (the TASK's scratch directory, <wt>/scratch/i11), <wt> (the T3 worktrees
  root), <home>;
- trailing spaces and tabs are stripped from every line, and trailing blank
  lines from every file (so `git diff --check` is clean).
No line is cut, and no other byte changes.

Usage: assemble_run_records.py <wt> <venv> <out-dir>
"""
import json
import os
import shutil
import sys
from pathlib import Path

WT, VENV, OUT = sys.argv[1], sys.argv[2], Path(sys.argv[3])
SCRATCH = os.path.join(WT, "scratch", "i11")
HOME = os.path.expanduser("~")


def clean(text):
    for real, placeholder in ((VENV, "<VENV>"), (SCRATCH, "<scratch>"), (WT, "<wt>"), (HOME, "<home>")):
        text = text.replace(real, placeholder)
    lines = [line.rstrip(" \t") for line in text.split("\n")]
    while lines and lines[-1] == "":
        lines.pop()
    return "\n".join(lines) + "\n"


def put(src, rel):
    dst = OUT / rel
    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_text(clean(Path(src).read_text(encoding="utf-8", errors="replace")))


def put_text(text, rel):
    dst = OUT / rel
    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_text(clean(text))


if OUT.exists():
    shutil.rmtree(OUT)
S = Path(SCRATCH)
R = S / "runs"

# The assembler itself.
put(__file__, "assemble_run_records.py.txt")

# Checkpoint A: the generators, the targeted runs in run order (the first,
# failing runs included), the heavy tests run one at a time with their wall
# times, the full FK suite at opt-level 0, and K-D5's suites in NI and PP.
put(S / "gen" / "generate.log", "checkpoint_a/generate.log")
put(R / "k3a_gen_check.log", "checkpoint_a/k3a_gen_wide_vectors_check.log")
put(R / "k3_gen_check.log", "checkpoint_a/k3_gen_wide_k3_vectors_check.log")
for name in ("fast1.log", "fast2.log", "fast3.log", "fk_lib_build.log", "fk_full.log",
             "ni_suite.log", "pp_formation_check_runtime.log", "pp_f1a_tests.log"):
    put(R / name, "checkpoint_a/" + name)
put(R / "heavy.sh", "checkpoint_a/heavy/heavy.sh.txt")
for p in sorted((R / "heavy").iterdir()):
    put(p, "checkpoint_a/heavy/" + p.name)

# Checkpoint B: the Q7 profile, the suites and T9.
B = R / "b"
for name in ("fk_o1_build.log", "fk_o1_full.log", "fk_guard_full.log"):
    put(B / name, "checkpoint_b/" + name)
put(B / "list_A.txt", "checkpoint_b/test_list_opt_level_0.txt")
put(B / "list_O1.txt", "checkpoint_b/test_list_opt_level_1.txt")
put(B / "run_suites_nff.sh", "checkpoint_b/suites/run_suites_nff.sh.txt")
put(S / "compare_suites.py", "checkpoint_b/suites/compare_suites.py.txt")
put(B / "suites_compare.txt", "checkpoint_b/suites/suites_compare.txt")
put(B / "suites_compare.json", "checkpoint_b/suites/suites_compare.json")
put(S / "suites_k3.log", "checkpoint_b/suites/suites_k3.log")
for p in sorted((S / "suites_k3").iterdir()):
    put(p, "checkpoint_b/suites/logs/" + p.name)
T = S / "t9"
put(T / "h_base" / "Cargo.toml", "checkpoint_b/t9/harness_base_Cargo.toml.txt")
put(T / "h_cand" / "Cargo.toml", "checkpoint_b/t9/harness_cand_Cargo.toml.txt")
for name in ("build_base.log", "build_cand.log", "run_base.log", "run_cand.log",
             "output_sha256_base.txt", "output_sha256_cand.txt"):
    put(T / name, "checkpoint_b/t9/" + name)
base = (T / "output_sha256_base.txt").read_text().splitlines()
cand = (T / "output_sha256_cand.txt").read_text().splitlines()
native = {}
for line in (Path(WT) / "scratch" / "calib" / "fixdiff" / "sha_native.txt").read_text().splitlines():
    digest, name = line.split(None, 1)
    native[name.strip()] = digest
base_map = {l.split(None, 1)[1].strip(): l.split(None, 1)[0] for l in base}
cand_map = {l.split(None, 1)[1].strip(): l.split(None, 1)[0] for l in cand}
same = sum(1 for k in base_map if cand_map.get(k) == base_map[k])
same_native = sum(1 for k in base_map if native.get(k) == base_map[k])
errs = sum(1 for p in (T / "out_base").rglob("*.out") if p.read_text().startswith(("ERR", "PANIC")))
put_text(
    "T9 (Mac-only): base = git archive of eb52114e9; candidate = git archive of 74add6078 with the then\n"
    "uncommitted FK/Cargo.toml test profile copied over it, a tree byte-identical to 8cacbfaf4's (diff -r);\n"
    "both without execution/, on this Mac.\n"
    "Harness: S11-K's fixdiff_main.rs (sha256 ec089c1d4a2a46dda689aad3ac55948cf858537ed5c986726c7accdd63cade53),\n"
    "ROOT's harness Cargo.lock (sha256 1c69935d85770777eb55ce77c03acfa023c444b86e46df1c49c9e459e9e0235f),\n"
    "--release --offline --locked, aarch64-apple-darwin, rustc 1.97.1.\n"
    "outputs: base %d, candidate %d\n"
    "base vs candidate: byte-identical %d/%d\n"
    "base vs ROOT Mac main (fixdiff/sha_native.txt): identical %d/%d\n"
    "ERR outputs (identical in both): %d; PANIC outputs: 0\n"
    "Never compared with Linux records (T3/PLATFORM_CALIBRATION_MAC/).\n"
    % (len(base), len(cand), same, len(base), same_native, len(base), errs),
    "checkpoint_b/t9/t9_summary.txt")

# Checkpoint C: the mutation driver, patches, logs and results, K-D5's suites
# against M17, and the P1/P2 guard re-run.
M = S / "mut"
put(M / "mutate.py", "mutations/mutate.py.txt")
for p in sorted((M / "out").iterdir()):
    put(p, "mutations/" + p.name)
for p in sorted((M / "out_guard").iterdir()):
    put(p, "mutations/guard/" + p.name)
summary = json.loads((M / "out" / "summary.json").read_text())
guard = json.loads((M / "out_guard" / "summary.json").read_text())
rows = ["id\tsite\tcompiled\tpassed\tfailed\tkilled\tbehavioural kills (test @ assertion)"]
for key, rec in list(summary.items()) + [("guard:" + k, v) for k, v in guard.items()]:
    site = "%s:%s" % (rec.get("file", "-").split("/")[-1], rec.get("line", "-"))
    kills = "; ".join("%s @ %s" % (k["test"].split("::")[-1], k["at"].split("/")[-1]) for k in rec["behavioural_kills"])
    rows.append("%s\t%s\t%s\t%d\t%d\t%s\t%s" % (key, site, rec["compiled"], rec["passed"], rec["failed"], rec["killed"], kills or "-"))
put_text("\n".join(rows), "mutations/MUTANTS.txt")

# The toolchain and the sources as delivered.
put_text(
    "platform: aarch64-apple-darwin (macOS 26.6.2, 18 logical CPUs, 128 GB, no swap)\n"
    "rustc 1.97.1 (8bab26f4f 2026-07-14); cargo 1.97.1 (c980f4866 2026-06-30)\n"
    "RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0; --offline --locked\n"
    "rustfmt 1.9.0-stable (the host's stable toolchain; applied to K3's new and changed files only)\n"
    "Python 3.13.14 (<VENV>), standard library only\n"
    "caps: cargo -j 8 at most (mutants -j 4, at most three at once); RUST_TEST_THREADS=4; at most two cargo jobs\n",
    "toolchain.txt")
FK = Path(WT) / "k3" / "projects/chirality-piping/core/solver/frame_kernel"
import hashlib
files = ["Cargo.toml", "src/structural/retained/mod.rs", "src/structural/retained/wide.rs",
         "src/structural/retained/wide/multi.rs"] + ["tests/retained_wide_k3/" + p.name for p in sorted((FK / "tests/retained_wide_k3").iterdir())]
put_text("\n".join("%s  FK/%s" % (hashlib.sha256((FK / f).read_bytes()).hexdigest(), f) for f in files),
         "source_sha256.txt")
print("assembled", sum(1 for _ in OUT.rglob("*") if _.is_file()), "files")
