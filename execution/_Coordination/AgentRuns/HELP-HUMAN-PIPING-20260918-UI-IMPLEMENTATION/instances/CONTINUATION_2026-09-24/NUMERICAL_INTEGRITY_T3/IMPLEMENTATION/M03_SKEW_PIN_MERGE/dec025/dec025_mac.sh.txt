#!/bin/bash
# DEC-025 under the owner's Mac decision (OWNER_DIRECTION.md, 2026-09-28): (1) the sweep invocation (fail-fast,
# expected to stop at the cargo surface's first Mac-platform failure); (2) all 39 cargo manifests --no-fail-fast,
# for comparison with the Mac baseline of main; (3) surfaces 2, 3 and 5 with the sweep's own commands.
SW=$1; L=$2; V=$3; R=$4; cd $SW
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_BUILD_JOBS=8 RUST_TEST_THREADS=4 PATH=$(dirname $V):$PATH
echo "start $(date -u +%FT%TZ) head $(git rev-parse HEAD)" > $L/meta.txt
$V tools/release/run_evidence_sweep.py --execute --only-capability sandboxed > $L/sweep.log 2>&1; echo "sweep exit $? $(date -u +%FT%TZ)" >> $L/meta.txt
cp validation/evidence/sweeps/SWEEP_*_$(git rev-parse HEAD | cut -c1-12).json $L/ 2>/dev/null
$R/scratch/calib/run_suites_nff.sh $(cd ../.. && pwd) $L/suites $R/sweep-skewpin-target > $L/suites.log 2>&1; echo "suites done $(date -u +%FT%TZ)" >> $L/meta.txt
run() { name=$1; shift; echo "== $name: $* ($(date -u +%T)Z)" >> $L/surfaces.txt; "$@" > $L/$name.log 2>&1; echo "   exit $? ($(date -u +%T)Z)" >> $L/surfaces.txt; }
run python_pytest $V -m pytest -q tests
run desktop_vitest_build_wasm npm run build:wasm:desktop
run desktop_vitest npm run test:desktop
run desktop_production_build npm run build:desktop
echo "ALL-DONE $(date -u +%FT%TZ)" >> $L/meta.txt
