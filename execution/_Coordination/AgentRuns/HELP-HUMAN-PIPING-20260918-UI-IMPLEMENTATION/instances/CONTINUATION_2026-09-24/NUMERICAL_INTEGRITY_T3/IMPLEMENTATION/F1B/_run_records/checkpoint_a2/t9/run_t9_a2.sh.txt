#!/bin/bash
# I13 F1b A2: T9 candidate side (the base side is A1's, unchanged: e7d930d49), S11-K's harness
# (fixdiff_main.rs, sha256 ec089c1d...), release, from a git archive of 94e543a24 with F1b's A2 files overlaid.
W=<wt>; A=$W/scratch/i13/a2; A1=$W/scratch/i13/a1
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0
( cd $A/t9/h_cand && CARGO_TARGET_DIR=$A/t9-target-cand cargo build --release --offline --locked -j 8 > $A/t9/build_cand.log 2>&1; echo "build rc=$?" >> $A/t9/build_cand.log )
BIN=$A/t9-target-cand/release/calib_fixdiff_native
: > $A/t9/run_cand.log
for root in core fixtures validation; do
  mkdir -p $A/t9/out_cand/$root
  $BIN $A/trees/t9cand/projects/chirality-piping/$root $A/t9/out_cand/$root >> $A/t9/run_cand.log 2>&1; echo "$root rc=$?" >> $A/t9/run_cand.log
done
( cd $A/t9/out_cand && find . -type f -name '*.out' | sort | xargs shasum -a 256 ) > $A/t9/sha_cand.txt
mkdir -p $A/t9/extra_cand; $BIN $A1/extra_corpus $A/t9/extra_cand; (cd $A/t9/extra_cand && ls | sort | xargs shasum -a 256) > $A/t9/extra_sha_cand.txt
echo T9-A2-DONE
