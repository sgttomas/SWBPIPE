For each mutant: a clean `git archive` of the candidate's `core/` into
<wt>/k3-mut/<id>/, one exact-string patch (asserted to apply once), its own
target directory, `cargo test --no-fail-fast` of frame_kernel with the
committed test profile, then the target is deleted. Each failing test is
classified by the source line it panicked at: a kill counts only when that
line is an assertion in a test file (not a panic in the code under test, not
an unwrap, not a compile failure).

Usage: mutate.py <wt> <commit> <out_dir> [mutant ids...]
"""

ENV = dict(os.environ, RUSTUP_TOOLCHAIN="1.97.1", RUSTUP_AUTO_INSTALL="0",
           CARGO_INCREMENTAL="0", RUST_TEST_THREADS="4")


def sanitize(text):
    return text.replace(str(WT), "<wt>")


def run(mid, desc, path, old, new):
    root = WT / "k3-mut" / mid
    if root.exists():
        shutil.rmtree(root)
    root.mkdir(parents=True)
    arch = subprocess.run(
        "git -C %s archive %s projects/chirality-piping/core | tar -x -C %s" % (WT / "k3", COMMIT, root),
        shell=True, capture_output=True, text=True)
    assert arch.returncode == 0, arch.stderr
    record = {"id": mid, "description": desc}
    # Uncommitted files copied over the archive (the P1/P2 guard re-run), recorded.
    overlay = [x for x in os.environ.get("OVERLAY", "").split(",") if x]
    for rel in overlay:
        shutil.copyfile(WT / "k3" / rel, root / rel)
    record["overlay"] = overlay
    if path is not None:
        f = root / path
        text = f.read_text()
        assert text.count(old) == 1, (mid, text.count(old))
        mutated = text.replace(old, new)
        f.write_text(mutated)
        record["file"] = path
        record["line"] = text[: text.index(old)].count("\n") + 1
        diff = "".join(difflib.unified_diff(text.splitlines(True), mutated.splitlines(True), "a/" + path, "b/" + path))
        (OUT / ("%s.patch" % mid)).write_text(diff)
    target = root / "target"
    start = time.time()
    proc = subprocess.run(
        ["cargo", "test", "--offline", "--locked", "-j", "4", "--no-fail-fast",
         "--manifest-path", str(root / FK / "Cargo.toml")],
        env=dict(ENV, CARGO_TARGET_DIR=str(target)), capture_output=True, text=True)
    log = proc.stdout + proc.stderr
    record["wall_s"] = round(time.time() - start, 1)
    record["rc"] = proc.returncode
    (OUT / ("%s.log" % mid)).write_text(sanitize(log))
    record["compiled"] = ("Compiling open_pipe_stress_frame_kernel" in log) and ("error[" not in log) and ("could not compile" not in log)
    results = re.findall(r"^test result: (\w+)\. (\d+) passed; (\d+) failed", log, re.M)
    record["passed"] = sum(int(r[1]) for r in results)
    record["failed"] = sum(int(r[2]) for r in results)
    failed_tests = re.findall(r"^test (\S+) \.\.\. FAILED$", log, re.M)
    record["failed_tests"] = failed_tests
    kills, other = [], []
    for block in re.finditer(r"^---- (\S+) stdout ----\n(.*?)(?=^---- |\Z|^failures:$)", log, re.M | re.S):
        name, body = block.group(1), block.group(2)
        # The last panic of the block is the one that failed the test (a panic
        # caught by catch_unwind inside the test is printed before it).
        found = re.findall(r"panicked at ([^:\n]+):(\d+):(\d+):\n(.*)", body)
        m = found[-1] if found else None
        if not m:
            other.append({"test": name, "why": "no panic location"})
            continue
        file, line, message = m[0], int(m[1]), m[3].strip()
        src = (root / FK / file).resolve()
        try:
            source_line = src.read_text().splitlines()[line - 1].strip()
        except Exception:
            source_line = "<unreadable>"
        norm = os.path.normpath(file)
        in_test_file = "/tests/" in "/" + norm or norm.startswith("tests/") or norm.endswith("_tests.rs")
        entry = {"test": name, "at": "%s:%d" % (norm, line), "source": source_line, "message": message[:200]}
        if in_test_file and "assert" in source_line:
            kills.append(entry)
        else:
            other.append(entry)
    record["behavioural_kills"] = kills
    record["other_failures"] = other
    record["killed"] = record["compiled"] and len(kills) > 0
    shutil.rmtree(target, ignore_errors=True)
    (OUT / ("%s.json" % mid)).write_text(json.dumps(record, indent=1))
    return record


def main():
    OUT.mkdir(parents=True, exist_ok=True)
    todo = [m for m in MUTANTS if not ONLY or m[0] in ONLY]
    results = {}
    if todo and todo[0][0] == "NONE":
        results["NONE"] = run(*todo[0])
        print("NONE", results["NONE"]["rc"], results["NONE"]["passed"], results["NONE"]["failed"], flush=True)
        assert results["NONE"]["rc"] == 0 and results["NONE"]["failed"] == 0, "control failed"
        todo = todo[1:]
    with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
        futures = {pool.submit(run, *m): m[0] for m in todo}
        for fut in concurrent.futures.as_completed(futures):
            r = fut.result()
            results[r["id"]] = r
            print(r["id"], "killed" if r["killed"] else "NOT-KILLED", "compiled=%s" % r["compiled"],
                  "failed=%d" % r["failed"], "kills=%d" % len(r["behavioural_kills"]), "wall=%s" % r["wall_s"], flush=True)
    (OUT / "summary.json").write_text(json.dumps(results, indent=1))
    print("MUTATIONS-DONE", flush=True)


if __name__ == "__main__":
    main()
