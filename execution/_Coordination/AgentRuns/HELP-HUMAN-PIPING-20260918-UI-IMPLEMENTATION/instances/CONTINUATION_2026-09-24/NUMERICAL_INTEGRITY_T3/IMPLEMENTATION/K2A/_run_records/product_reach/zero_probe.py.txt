"""K2a (I6): the exact-zero 12EI/L^3 probe (reach_zero/). Usage: python3 -B zero_probe.py <reach_requests.py> <outdir>; prints the case arithmetic."""
import math, json, sys
from fractions import Fraction as Fr
src=open(sys.argv[1]).read().split('out = sys.argv[1]')[0]
ns={}; exec(src,ns)
out=sys.argv[2]
def req(name, L, od, t, E, G, spring, u_target, gap):
    idd=od-2*t
    a=math.pi*(od*od-idd*idd)/4; i=math.pi*((od*od)*(od*od)-(idd*idd)*(idd*idd))/64
    e12=12.0*E*i
    ktrue=12*Fr(E)*Fr(i)/Fr(L)**3
    P=float('%.3g'%(spring*u_target))
    u_main=P/spring; u_true=Fr(P)/(Fr(spring)+ktrue)
    print(name,'E',E,'12EI',e12,'EA',E*a,'EA/L 2^%.1f'%math.log2(E*a/L),'GJ/L 2^%.1f'%math.log2(G*2*i/L),'true 12EI/L3 2^%.1f'%math.log2(float(ktrue)),'spring 2^%.1f'%math.log2(spring),'P',P,'u_main',u_main,'u_true',float(u_true),'rel',float(abs(Fr(u_main)-u_true)/u_true))
    json.dump(ns['request'](name,L,od,t,E,G,'UY',spring,P,gap),open('%s/%s.json'%(out,name),'w'))
L=2.0**-39
od=1e-11; t=1e-12; idd=od-2*t; i=math.pi*((od*od)*(od*od)-(idd*idd)*(idd*idd))/64
E=float('%.2g'%float(Fr(2)**-1075/(12*Fr(i))*Fr(9,10)))
ks=float('%.3g'%float(12*Fr(E)*Fr(i)/Fr(L)**3))
for gap in (False,True):
    req('Z1'+('-gap' if gap else ''),L,od,t,E,1e-100,ks,0.25,gap)
