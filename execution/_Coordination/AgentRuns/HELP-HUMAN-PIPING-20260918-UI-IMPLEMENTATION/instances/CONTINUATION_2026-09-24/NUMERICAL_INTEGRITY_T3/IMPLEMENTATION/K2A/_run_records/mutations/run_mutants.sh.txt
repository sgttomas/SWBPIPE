#!/usr/bin/env bash
# K2a (I6) mutation runner. Usage: run_mutants.sh <repo_root> <scratch_mut_dir> <target_dir> [mutant ...]
# Each mutant (and the no-mutation control NONE, run first) gets a fresh copy of the candidate's
# frame_kernel crate (tar -m: fresh mtimes) with the patch applied, a removed target directory, and
# `cargo test --offline --locked --test k2a_checked_formation`. frame_kernel has no dependencies.
# One line per mutant in MUTANTS.txt: <mutant> <killed|survived|BUILD-FAILED> <failing tests>.
set -u
repo="$1"; work="$2"; target="$3"; shift 3
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR="$target"
fk="$repo/projects/chirality-piping/core/solver/frame_kernel"
list=("$@")
if [ ${#list[@]} -eq 0 ]; then
  mapfile -t list < <(python3 -B "$work/mutate.py" --list | cut -f1)
fi
for m in NONE "${list[@]}"; do
  dir="$work/tree_$m"
  rm -rf "$dir" "$target"
  mkdir -p "$dir"
  (cd "$fk/.." && tar -cf - frame_kernel) | (cd "$dir" && tar -xmf -)
  if [ "$m" != NONE ]; then
    python3 -B "$work/mutate.py" "$dir/frame_kernel/src/lib.rs" "$m" || { echo "$m PATCH-FAILED" >> "$work/MUTANTS.txt"; continue; }
  fi
  log="$work/log_$m.txt"
  (cd "$dir/frame_kernel" && cargo test --offline --locked --test k2a_checked_formation) > "$log" 2>&1
  status=$?
  failing=$(grep -E '^    k2a_[a-z0-9_]+$' "$log" | tr -d ' ' | sort -u | tr '\n' ' ')
  if grep -q 'test result:' "$log"; then
    if [ $status -eq 0 ]; then verdict=survived; else verdict=killed; fi
  else
    verdict=BUILD-FAILED
  fi
  echo "$m $verdict $failing" >> "$work/MUTANTS.txt"
  echo "$m $verdict $failing"
  rm -rf "$dir"
done
rm -rf "$target"
