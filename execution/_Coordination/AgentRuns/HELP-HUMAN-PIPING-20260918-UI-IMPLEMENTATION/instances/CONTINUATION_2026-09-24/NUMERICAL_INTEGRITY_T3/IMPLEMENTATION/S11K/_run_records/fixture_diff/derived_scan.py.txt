"""Which committed files carry the raw values that change (token search, stdlib only)."""
import json, os, re, sys, difflib, collections
base_dir, cand_dir, repo = sys.argv[1], sys.argv[2], sys.argv[3]
NUM = re.compile(r'-?\d+\.\d+(?:e[-+]?\d+)?|-?\d+e[-+]?\d+|[0-9a-f]{64}')
tokens = collections.OrderedDict()
for name in sorted(os.listdir(base_dir)):
    a = open(os.path.join(base_dir, name)).read(); b = open(os.path.join(cand_dir, name)).read()
    if a == b: continue
    al, bl = a.splitlines(), b.splitlines()
    for tag, i1, i2, j1, j2 in difflib.SequenceMatcher(None, al, bl, autojunk=False).get_opcodes():
        if tag == 'equal': continue
        for x, y in zip(al[i1:i2], bl[j1:j2]):
            tx, ty = NUM.findall(x), NUM.findall(y)
            for p, q in zip(tx, ty):
                if p != q and len(p) >= 12: tokens[p] = q
files = []
for root in ['fixtures', 'validation', 'core', 'apps/desktop/src']:
    for dp, dn, fn in os.walk(os.path.join(repo, root)):
        dn[:] = [d for d in dn if d not in ('target', 'node_modules')]
        for f in fn:
            if f.endswith(('.json', '.ts', '.rs', '.md')): files.append(os.path.join(dp, f))
hits = {}
for f in files:
    try: text = open(f).read()
    except Exception: continue
    n = sum(text.count(t) for t in tokens)
    if n: hits[os.path.relpath(f, repo)] = n
print(json.dumps({"changed_tokens": len(tokens), "files_carrying_old_tokens": hits}, indent=1))
