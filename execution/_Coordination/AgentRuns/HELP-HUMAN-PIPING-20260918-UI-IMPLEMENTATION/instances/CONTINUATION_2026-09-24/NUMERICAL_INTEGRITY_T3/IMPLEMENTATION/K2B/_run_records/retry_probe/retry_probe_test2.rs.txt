
// Second candidate (the first is absorbed at N1 UX at every b >= 356: a scale-free refusal). An
// axial chain N0-N1-N2 with E = 2^440 for both bars (EA/L = 2^440 N/m, no absorption), UX loads
// 2^-1010 N at N1 and 1 N at N2. u1 = (2^-1010 + 1)/2^440 and u2 = u1 + 2^-440 are normal and
// b-invariant. The census span is [-1010, 443], the window [52, 572], the chosen b 312. The N1
// right-hand side in the equilibrated units is 2^(-1010 + b - (441 + b)/2) = 2^(-1230 + b/2):
// subnormal for b < 416, normal for b >= 416.
#[test]
fn k2b_scratch_retry_candidate_two() {
    let model = axial_chain(pow2(440), &[0.0, 1.0, 2.0], &[(1, pow2(-1010)), (2, 1.0)]);
    let force = model.ledger();
    let probe = model.case(&force, LinearSolveMode::DenseScrutiny, EvidenceRepresentation::Dense);
    let census = force_scale_census(&probe).unwrap();
    println!(
        "RETRY2 census span={:?} chosen={:?}",
        census.span(),
        census.force_scale().map(|s| s.exponent())
    );
    let summary = |r: &Result<ForceScaledSolution, ForceScaledError>| match r {
        Ok(s) => format!(
            "Ok quality={:?} u1={:e} u2={:e} records={:?}",
            s.solution.report.quality,
            s.solution.displacements[6],
            s.solution.displacements[12],
            s.records
                .iter()
                .map(|r| (r.record, r.global_dof, r.representability))
                .collect::<Vec<_>>()
        ),
        Err(e) => format!("Err {e:?}"),
    };
    for mode in MODES {
        for representation in REPRESENTATIONS {
            let today = model.existing(&force, mode, representation, true);
            println!(
                "RETRY2 {mode:?} {representation:?} b=0 today: {}",
                match &today {
                    Ok(s) => format!("Ok quality={:?}", s.report.quality),
                    Err(e) => format!("Err {e:?}"),
                }
            );
            let rule = solve_with_force_scaling(&model.case(&force, mode, representation));
            println!(
                "RETRY2 {mode:?} {representation:?} rule: {}",
                summary(&rule.map(|o| o.solution))
            );
            for b in [312, 414, 416, 500, 572] {
                let forced = model.forced(&force, scale(b), mode, representation, true);
                println!("RETRY2 {mode:?} {representation:?} forced b={b}: {}", summary(&forced));
            }
        }
    }
    let u1 = (pow2(-1010) + 1.0) / pow2(440);
    println!("RETRY2 reference u1={:e} u2={:e}", u1, u1 + pow2(-440));
}
