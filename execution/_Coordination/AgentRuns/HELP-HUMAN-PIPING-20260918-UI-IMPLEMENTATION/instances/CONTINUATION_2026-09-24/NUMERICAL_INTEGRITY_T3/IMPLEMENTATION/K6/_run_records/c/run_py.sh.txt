#!/bin/bash
# K6 checkpoint C, one runner run: a clean git archive copy of H and the DEC-025 wrapper at the
# K6 head, the mutant applied (none for NONE), and the wrapper's pytest run (the N15 invocation,
# with --noconftest: the tests conftest only builds unrelated tool binaries with cargo).
W=<wt>
C=$W/scratch/i15/c; HEAD_COMMIT=ada18de702b47e8bcca194754815ec53e4ecd161
VENV=<VENV>
MID=$1; D=$W/k6-mut/py-$MID; LOG=$C/logs/py-$MID.log
rm -rf "$D"; mkdir -p "$D"
{
  echo "=== py $MID: git archive $HEAD_COMMIT (performance_harness, tests/test_performance_harness_runner.py) into <wt>/k6-mut/py-$MID"
  git -C $W/k6 archive $HEAD_COMMIT projects/chirality-piping/core/solver/performance_harness \
      projects/chirality-piping/tests/test_performance_harness_runner.py | tar -x -C "$D"
  if [ -n "${OVERLAY:-}" ]; then
    cp "$OVERLAY" "$D/projects/chirality-piping/core/solver/performance_harness/runner/test_k6_runner.py"
    echo "=== overlay: runner/test_k6_runner.py replaced by the amended file (sha256 $(shasum -a 256 "$OVERLAY" | cut -c1-64))"
  fi
  [ "$MID" != NONE ] && python3 $C/mutants.py apply "$MID" "$D"
  echo "=== <VENV>/bin/python -m pytest -q -p no:cacheprovider --noconftest tests/test_performance_harness_runner.py"
} > "$LOG" 2>&1
cd "$D/projects/chirality-piping" || exit 9
start=$(date +%s)
"$VENV/bin/python" -m pytest -q -p no:cacheprovider --noconftest tests/test_performance_harness_runner.py >> "$LOG" 2>&1
rc=$?
echo "=== exit $rc elapsed $(( $(date +%s) - start )) s" >> "$LOG"
sed -i '' -e "s#$VENV#<VENV>#g" -e "s#$W#<wt>#g" -e 's/[[:space:]]*$//' "$LOG"
exit 0
