def bits(h):
    return struct.unpack('>d', bytes.fromhex(h))[0]


def pow2(k):
    return math.ldexp(1.0, k)


def request(model):
    return {'model': model, 'materials': []}


def frame_model(pid, nodes, members, supports, loads, od=0.2, wall=0.01, e=2.0e11, g=8.0e10, cases=None):
    def load(lid, node, dof, value):
        if dof.startswith('R'):
            return {'id': lid, 'category': 'concentrated_moment', 'target': {'type': 'node', 'node': node},
                    'direction': dof, 'magnitude': {'value': value, 'unit': 'N*m'}, 'dimension': 'moment', 'provenance': PROV}
        return {'id': lid, 'category': 'concentrated_force', 'target': {'type': 'node', 'node': node},
                'direction': 'global_' + dof[1:].lower(), 'magnitude': {'value': value, 'unit': 'N'}, 'dimension': 'force',
                'provenance': PROV}
    if cases is None:
        cases = [{'id': 'case', 'label': pid, 'kind': 'primitive_user_load',
                  'primitive_loads': [load(*l) for l in loads], 'provenance': PROV}]
    else:
        cases = [dict(c, primitive_loads=[load(*l) for l in c['primitive_loads']], provenance=PROV) for c in cases]
    return {
        'schema_version': '0.1.0', 'document_kind': 'openpipestress.product_preview.model',
        'analysis_status': {'mechanics': 'ready_for_preview_diagnostics', 'rule_check': 'not_performed_user_rule_inputs_missing',
                            'professional_acceptance': 'not_provided'},
        'project': {'id': 'invented:t3-f1b-a2:' + pid,
                    'units': {'length': 'm', 'force': 'N', 'angle': 'rad', 'pressure': 'Pa', 'temperature': 'degC', 'stress': 'Pa'}},
        'nodes': [{'id': n, 'position': {'x': p[0], 'y': p[1], 'z': p[2]}, 'provenance': PROV} for n, p in nodes],
        'pipe_segments': [{'id': m, 'from': a, 'to': b, 'material': 'mat:A2', 'y_reference': {'x': 0.0, 'y': 1.0, 'z': 0.0},
                           'section': {'outside_diameter': {'value': od, 'unit': 'm'}, 'wall_thickness': {'value': wall, 'unit': 'm'}},
                           'provenance': PROV} for m, a, b in members],
        'materials': [{'id': 'mat:A2', 'elastic_modulus': {'value': e, 'unit': 'Pa'}, 'shear_modulus': {'value': g, 'unit': 'Pa'},
                       'provenance': PROV}],
        'supports': supports,
        'load_cases': cases,
        'combinations': [],
    }


def anchor(node, restraints=ALL, sid=None):
    return {'id': sid or 'anchor:' + node, 'node': node, 'family': 'anchor', 'restraints': list(restraints), 'provenance': PROV}


def reach(pid, length, od, wall, e, g, free, spring, load, open_gap):
    supports = [anchor('N0'), anchor('N1', [d for d in ALL if d != free], 'rigid:N1')]
    if spring is not None:
        unit = 'N*m/rad' if free.startswith('R') else 'N/m'
        supports.append({'id': 'spring:N1:0', 'node': 'N1', 'family': 'spring', 'restraints': [free],
                         'stiffness': {'dof': free, 'value': {'value': spring, 'unit': unit}}, 'provenance': PROV})
    if open_gap:
        supports.append({'id': 'support:k2a-open-gap', 'node': 'N1', 'family': 'nonlinear', 'restraints': [],
                         'nonlinear': {'behavior': 'gap', 'dof': free, 'initial_state': 'inactive',
                                       'closes_when': 'positive_displacement', 'gap': {'value': 1000.0, 'unit': 'mm'}},
                         'provenance': PROV})
    return frame_model(pid, [('N0', (0, 0, 0)), ('N1', (length, 0, 0))], [('M1', 'N0', 'N1')], supports,
                       [('load:0', 'N1', free, load)], od=od, wall=wall, e=e, g=g)

def chain(members, pl=0, pm=0, pf=0):
    s = pow2(pl)
    nodes = [('N%d' % i, (i * s, 0.125 * i * s, -0.0625 * i * s)) for i in range(members + 1)]
    mem = [('M%d' % i, 'N%d' % (i - 1), 'N%d' % i) for i in range(1, members + 1)]
    tip = 'N%d' % members
    return frame_model('chain-%d-%d-%d-%d' % (members, pl, pm, pf), nodes, mem, [anchor('N0')],
                       [('load:tip-y', tip, 'UY', 1250.0 * pow2(pf)), ('load:tip-rx', tip, 'RX', 40.0 * pow2(pf + pl))],
                       od=0.2 * s, wall=0.01 * s, e=2.0e11 * pow2(pm), g=8.0e10 * pow2(pm))


def limitation_chain():
    free_ux = lambda n: anchor(n, ['UY', 'UZ', 'RX', 'RY', 'RZ'], 'rigid:' + n)
    return frame_model('limitation-chain', [('N0', (0, 0, 0)), ('N1', (1, 0, 0)), ('N2', (2, 0, 0))],
                       [('M1', 'N0', 'N1'), ('M2', 'N1', 'N2')], [anchor('N0'), free_ux('N1'), free_ux('N2')],
                       [('load:n1', 'N1', 'UX', pow2(-1010)), ('load:n2', 'N2', 'UX', 1.0)], e=pow2(440), g=pow2(439))

def tnode(i, x, y=0.0, z=0.0):
    return {'id': i, 'position': {'x': x, 'y': y, 'z': z}, 'provenance': PROV}


def tpipe(i, a, b, yref=(0.0, 1.0, 0.0), od=0.12, wall=0.01):
    return {'id': i, 'from': a, 'to': b, 'section': {'outside_diameter': {'value': od, 'unit': 'm'},
                                                    'wall_thickness': {'value': wall, 'unit': 'm'}},
            'material': 'material:m', 'y_reference': {'x': yref[0], 'y': yref[1], 'z': yref[2]}, 'provenance': PROV}


def tforce(i, node, d, v):
    return {'id': i, 'category': 'concentrated_force', 'target': {'type': 'node', 'node': node}, 'direction': d,
            'magnitude': {'value': v, 'unit': 'N'}, 'dimension': 'force', 'provenance': PROV}


def tsupport(i, node, restraints):
    return {'id': i, 'node': node, 'family': 'anchor', 'restraints': restraints, 'provenance': PROV}


def tcase(i, loads, **extra):
    return dict({'id': i, 'primitive_loads': loads, 'provenance': PROV}, **extra)


def family(name, escale, fscale):
    """A family model: moduli (and user stiffnesses) times escale, loads times fscale."""
    e, g = 200e9 * escale, 76.923076923e9 * escale
    tip = [tforce('f', 'node:b', 'global_y', 100.0 * fscale)]
    cant_nodes = [tnode('node:a', 0.0), tnode('node:b', 1.0)]
    cant_pipes = [tpipe('pipe:a-b', 'node:a', 'node:b')]
    anc = [tsupport('support:a', 'node:a', ALL)]
    if name == 'uniform':
        loads = tip + [{'id': 'w', 'category': 'distributed_force', 'target': {'type': 'element', 'pipe': 'pipe:a-b'},
                        'direction': 'global_z', 'magnitude': {'value': -50.0 * fscale, 'unit': 'N/m'},
                        'dimension': 'force_per_length', 'provenance': PROV}]
        return t0r_base(cant_nodes, cant_pipes, anc, [], [tcase('case:f', loads)], e, g)
    if name == 'thermal':
        loads = tip + [{'id': 't', 'category': 'thermal', 'target': {'type': 'element', 'pipe': 'pipe:a-b'},
                        'direction': 'global_z', 'magnitude': {'value': 12.5, 'unit': 'degC'},
                        'dimension': 'temperature_interval', 'provenance': PROV}]
        two = [tnode('node:a', 0.0), tnode('node:b', 1.0)]
        return t0r_base(two, cant_pipes, anc + [tsupport('support:b', 'node:b', ['UX'])], [], [tcase('case:f', loads)], e, g)
    if name == 'constant_effort':
        ce = {'id': 'support:ce', 'node': 'node:b', 'family': 'constant_effort_support', 'restraints': ['UY'],
              'hanger': {'hanger_type': 'constant_effort_support', 'constant_load': {'value': 375.0 * fscale, 'unit': 'N'},
                         'travel_range': {'value': 0.05, 'unit': 'm'}, 'source_reference': 'invented'}, 'provenance': PROV}
        return t0r_base(cant_nodes, cant_pipes, anc + [ce], [], [tcase('case:f', tip)], e, g)
    if name == 'user_element':
        joint = {'id': 'component:joint', 'label': 'invented joint', 'kind': 'expansion_joint', 'node': 'node:b',
                 'geometry': {'expansion_joint_pipe_ref': 'pipe:b-c', 'effective_area': {'value': 0.01, 'unit': 'm^2'},
                              'expansion_joint_source_reference': 'invented'},
                 'modifiers': {'axial_stiffness_user_value': {'value': 3.2e6 * escale, 'unit': 'N/m'},
                               'lateral_stiffness_user_value': {'value': 9.0e5 * escale, 'unit': 'N/m'},
                               'angular_stiffness_user_value': {'value': 4.8e5 * escale, 'unit': 'N*m/rad'},
                               'torsional_stiffness_user_value': {'value': 6.2e5 * escale, 'unit': 'N*m/rad'},
                               'source_reference': 'invented'},
                 'mechanics_interface': {'solver_consumption': 'mechanics_geometry_and_user_flexibility',
                                         'rule_check_consumption': 'user_rule_pack_inputs_only'}, 'provenance': PROV}
        nodes = [tnode('node:a', 0.0), tnode('node:b', 1.0), tnode('node:c', 1.0)]
        pipes = [tpipe('pipe:a-b', 'node:a', 'node:b'), tpipe('pipe:b-c', 'node:b', 'node:c')]
        sup = anc + [tsupport('support:c', 'node:c', ALL)]
        return t0r_base(nodes, pipes, sup, [joint], [tcase('case:f', [tforce('f', 'node:b', 'global_y', 100.0 * fscale)])], e, g)
    if name == 'curved_bend':
        comp = {'id': 'component:bend', 'label': 'invented arc', 'kind': 'bend', 'node': 'node:c',
                'geometry': {'bend_pipe_ref': 'pipe:b-c', 'bend_radius': {'value': 0.2, 'unit': 'm'},
                             'bend_angle': {'value': math.pi / 2, 'unit': 'rad'}, 'bend_plane_orientation': 'global_xy_preview',
                             'bend_geometry_source_reference': 'invented'},
                'modifiers': {'flexibility_factor_user_value': {'value': 1.0, 'unit': 'none'}, 'source_reference': 'invented'},
                'mechanics_interface': {'solver_consumption': 'curved_bend_macro_element',
                                        'rule_check_consumption': 'user_rule_pack_inputs_only'}, 'provenance': PROV}
        nodes = [tnode('node:a', 0.0), tnode('node:b', 1.0), tnode('node:c', 1.2, 0.2), tnode('node:d', 1.2, 1.2)]
        pipes = [tpipe('pipe:a-b', 'node:a', 'node:b'), tpipe('pipe:b-c', 'node:b', 'node:c', (0.0, 0.0, 1.0)),
                 tpipe('pipe:c-d', 'node:c', 'node:d', (1.0, 0.0, 0.0))]
        return t0r_base(nodes, pipes, anc, [comp], [tcase('case:f', [tforce('z', 'node:d', 'global_z', 1000.0 * fscale)])], e, g)
    raise ValueError(name)

def f_a2_series(ratio):
    """F-A2: two members in series along x, N0-M1-N1-M2-N2, N0 and N2 anchored, N1 free in UY only,
    a 2^-1021 N load at N1:UY. M1 is K2a's least-subnormal member; M2 has the same section with E
    2^-ratio times smaller, so its end shear (and N2's reaction) is about 2^-ratio times the load."""
    L = 1.8189894035458565e-12
    m = frame_model('fa2-%d' % ratio, [('N0', (0, 0, 0)), ('N1', (L, 0, 0)), ('N2', (2 * L, 0, 0))],
                    [('M1', 'N0', 'N1'), ('M2', 'N1', 'N2')],
                    [anchor('N0'), anchor('N1', [d for d in ALL if d != 'UY'], 'rigid:N1'), anchor('N2')],
                    [('load:0', 'N1', 'UY', pow2(-1021))], od=1.0e-11, wall=1.0e-12, e=9.6e-280, g=1.0e-100)
    m['materials'].append({'id': 'mat:soft', 'elastic_modulus': {'value': 9.6e-280 * pow2(-ratio), 'unit': 'Pa'},
                           'shear_modulus': {'value': 1.0e-100, 'unit': 'Pa'}, 'provenance': PROV})
    m['pipe_segments'][1]['material'] = 'mat:soft'
    return m

def overflow_cantilever(root_spring):
    """F-A2 / F-S (overflow direction): a 2^20 m cantilever along x, E = 2^83 Pa, G = 2^82 Pa,
    OD 0.2 m, wall 0.01 m, with a 2^1010 N tip load in UY. The root moment, about 2^1030 N*m, lies
    beyond binary64. With the root anchored it is a reaction (F-A2); with the root's RZ on a stiff
    rotational spring instead, it is that spring's action (F-S)."""
    supports = [anchor('N0', ALL if not root_spring else ['UX', 'UY', 'UZ', 'RX', 'RY'])]
    if root_spring:
        supports.append({'id': 'spring:N0:RZ', 'node': 'N0', 'family': 'spring', 'restraints': ['RZ'],
                         'stiffness': {'dof': 'RZ', 'value': {'value': pow2(120), 'unit': 'N*m/rad'}}, 'provenance': PROV})
    return frame_model('overflow-%s' % root_spring, [('N0', (0, 0, 0)), ('N1', (pow2(20), 0, 0))], [('M1', 'N0', 'N1')],
                       supports, [('load:y', 'N1', 'UY', pow2(1010))], e=pow2(83), g=pow2(82))


def zero_term():
    """Admission check 9 (OQ13 narrowed): K2a's exact-zero case with an extra authored 0 N load at N1:UY."""
    m = reach('zero-term', *REACH['exact-zero'], open_gap=False)
    m['load_cases'][0]['primitive_loads'].append({'id': 'load:zero', 'category': 'concentrated_force',
        'target': {'type': 'node', 'node': 'N1'}, 'direction': 'global_y', 'dimension': 'force',
        'magnitude': {'value': 0.0, 'unit': 'N'}, 'provenance': PROV})
    return m


def main():
    out = sys.argv[1]
    os.makedirs(os.path.join(out, 'requests'), exist_ok=True)
    cases = []

    def add(cid, model, members):
        path = 'requests/%s.json' % cid
        with open(os.path.join(out, path), 'w') as fh:
            json.dump(request(model), fh)
        cases.append({'id': cid, 'request_file': path, 'members': members})

    for name, args in REACH.items():
        add('A2-REACH-%s' % name, reach(name, *args, open_gap=False), 1)
        add('A2-REACH-%s-open-gap' % name, reach(name + '-open-gap', *args, open_gap=True), 1)
    add('A2-PHYS-R4-pressure', phys_r4(True), 1)
    add('A2-PHYS-R4-no-pressure', phys_r4(False), 1)
    add('A2-LIMITATION-CHAIN', limitation_chain(), 2)
    add('A2-LEF-ANALOGUE-base', chain(5), 5)
    add('A2-LEF-ANALOGUE-large', chain(5, 200, 300, 600), 5)
    for fam in ('uniform', 'thermal', 'constant_effort', 'user_element', 'curved_bend'):
        add('A2-FAMILY-%s-range' % fam, family(fam, pow2(-1000), 1.0), 3)

    add('A2-FAMILY-exact_pressure_operand-b0', exact_pressure_operand(1e-3), 1)
    for k in (1e-20, 1e-40, 1e-60):
        add('A2-FS-k%g' % k, spring_soft(k * 1.3e-292, 1.0e-307), 1)
    add('A2-SUBNORMAL-I-b0', subnormal_section(1.0), 1)
    add('A2-SUBNORMAL-I-range', subnormal_section(1e-300), 1)
    for load in (1e-310, 2.5e-308, 1e-300):
        add('A2-COEXIST-%g' % load, coexistence(load), 1)
    add('A2-COEXIST-spring-2.5e-308', coexistence(2.5e-308, 1e12), 1)
    add('A2-MIXED', mixed(1e-310, pow2(-1050), pow2(-1000)), 1)
    for fam in ('curved_bend', 'user_element'):
        add('A2-FAMILY-%s-tinyload' % fam, family(fam, 1.0, pow2(-1000)), 3)
        add('A2-FAMILY-%s-e990' % fam, family(fam, pow2(-990), 1.0), 3)
    add('A2-FAMILY-exact_pressure_operand-normal', exact_pressure_normal(), 1)
    add('A2-FS-least-subnormal-spring', f_s(), 1)
    for r in (54, 60, 66):
        add('A2-FA2-series-%d' % r, f_a2_series(r), 2)

    add('A2-FA2-overflow-reaction', overflow_cantilever(False), 1)
    add('A2-FS-overflow-spring', overflow_cantilever(True), 1)
    add('A2-ZERO-TERM', zero_term(), 1)
    for k in (900, 950, 980, 1010):
        add('A2-FAMILY-curved_bend-f%d' % k, family('curved_bend', 1.0, pow2(-k)), 3)
    for k in (900, 950, 970):
        add('A2-FAMILY-curved_bend-e%d' % k, family('curved_bend', pow2(-k), 1.0), 3)
    json.dump(cases, open(os.path.join(out, 'cases.json'), 'w'), indent=1)
