"""K-D5 (I3): timed comparison, main b24b3d536 probe against the candidate probe (ROOT's request),
on the two largest completed dense runs. Interleaved main, cand, main, cand per case, captured
entry, P1's run_one (fresh process, RLIMIT_AS 6 GiB, 1800 s). Records wall time, peak RSS, the
load average at start and end, and the published standing. Waits for idle cargo/sweeps before
each run. Usage: python3 -B timing_compare.py <main-probe> <cand-probe> <gen_out> <out.jsonl>
(committed as timing_compare.py.txt)"""
import json, os, subprocess, sys, time
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from run import run_one, TIMEOUT_LARGE  # noqa: E402
import compare as C  # noqa: E402

main_bin, cand_bin, gen, out = sys.argv[1:5]
cases = {c['id']: c for c in json.load(open(os.path.join(gen, 'cases.json')))}
plan = []
for case in ('RF-LARGE-CHAIN-n01000-AX', 'RF-LARGE-TREE-n01000-ROT'):
    for rep in (1, 2):
        plan += [(case, 'main', main_bin, rep), (case, 'cand', cand_bin, rep)]
with open(out, 'a') as fh:
    for case, label, binary, rep in plan:
        while subprocess.run(['pgrep', '-x', 'cargo'], capture_output=True).returncode == 0 or \
                subprocess.run(['pgrep', '-f', 'python[0-9.]* .*run_evidence_sweep'], capture_output=True).returncode == 0:
            time.sleep(30)
        la0 = open('/proc/loadavg').read().split()[:3]
        rec = run_one(binary, os.path.join(gen, cases[case]['request_file']), 'dense_scrutiny', False, TIMEOUT_LARGE)
        la1 = open('/proc/loadavg').read().split()[:3]
        run = rec.get('probe', {}).get('run', {})
        ident = C.identity(run['envelope']) if run.get('ok') else {}
        row = {'case': case, 'binary': label, 'rep': rep, 'wall_seconds': rec['wall_seconds'],
               'peak_rss_kib': rec['peak_rss_kib'], 'timed_out': rec['timed_out'], 'loadavg_start': la0,
               'loadavg_end': la1, 'quality': ident.get('numerical_quality_status'), 'standing': ident.get('standing')}
        fh.write(json.dumps(row) + '\n'); fh.flush()
        print(json.dumps(row), flush=True)
