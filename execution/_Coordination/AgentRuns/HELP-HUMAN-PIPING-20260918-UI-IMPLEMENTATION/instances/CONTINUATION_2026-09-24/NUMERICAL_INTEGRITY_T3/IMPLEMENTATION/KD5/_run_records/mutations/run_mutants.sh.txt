#!/bin/bash
# K-D5 (I3) mutation runner (committed as run_mutants.sh.txt).
# Usage: run_mutants.sh <kd5-worktree> <scratch> <mutant>...
# For each mutant: a fresh copy of projects/chirality-piping/core/{solver,loads} (no target dirs),
# the patch from mutate.py, then `cargo test --offline --locked kd5` in nonlinear_integration
# (which also builds frame_kernel) with its own target dir; the failing tests are the killers.
W=$1; X=$2; shift 2
export RUSTUP_TOOLCHAIN=1.97.1 CARGO_INCREMENTAL=0 RUSTUP_AUTO_INSTALL=0 CARGO_TARGET_DIR=$X/mut-target
for m in "$@"; do
  while pgrep -x cargo >/dev/null || pgrep -f 'python[0-9.]* .*run_evidence_sweep' >/dev/null; do sleep 60; done
  rm -rf $X/copy; mkdir -p $X/copy/core
  tar -C $W/projects/chirality-piping/core --exclude=target -cf - solver loads | tar -C $X/copy/core -xf -
  python3 $X/mutate.py $X/copy/core $m > $X/$m.log 2>&1 || { echo "$m PATCH-FAILED" >> $X/MUTANTS.txt; continue; }
  (cd $X/copy/core/solver/nonlinear_integration && cargo test --offline --locked kd5 -- --test-threads=1 >> $X/$m.log 2>&1); rc=$?
  killers=$(grep -E '^test .* FAILED$' $X/$m.log | awk '{print $2}' | tr '\n' ' ')
  comp=$(grep -c '^error\[' $X/$m.log)
  echo "$m exit=$rc compile_errors=$comp killed_by: ${killers:-NONE}" >> $X/MUTANTS.txt
  sleep 60
done
echo DONE >> $X/MUTANTS.txt
