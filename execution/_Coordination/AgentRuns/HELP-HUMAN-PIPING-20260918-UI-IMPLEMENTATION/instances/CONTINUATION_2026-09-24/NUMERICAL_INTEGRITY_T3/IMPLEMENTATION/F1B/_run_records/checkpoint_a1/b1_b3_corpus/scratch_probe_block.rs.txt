// ---------------------------------------------------------------- SCRATCH ONLY
// I13 F1b A1 scratch probe (never committed): B1-B3 corpus-wide. Every JSON
// request (or model document) under the directories in F1B_CORPUS with at
// most 1,001 nodes; the attempt comparison only up to 606 DOFs.
fn scratch_walk(dir: &std::path::Path, out: &mut Vec<std::path::PathBuf>) {
    let mut entries: Vec<_> = std::fs::read_dir(dir).unwrap().map(|e| e.unwrap().path()).collect();
    entries.sort();
    for p in entries {
        if p.is_dir() {
            if p.file_name().is_some_and(|n| n == "target" || n == "node_modules") { continue; }
            scratch_walk(&p, out);
        } else if p.extension().is_some_and(|e| e == "json") {
            out.push(p);
        }
    }
}

#[test]
fn scratch_corpus_b1_b3() {
    use std::io::Write;
    let dirs = std::env::var("F1B_CORPUS").unwrap();
    let out_path = std::env::var("F1B_CORPUS_OUT").unwrap();
    let mut out = std::fs::File::create(out_path).unwrap();
    let mut files = Vec::new();
    for d in dirs.split(':') { scratch_walk(std::path::Path::new(d), &mut files); }
    let (mut checked, mut failed, mut skipped) = (0, 0, 0);
    for file in files {
        let text = std::fs::read_to_string(&file).unwrap();
        let Ok(value) = serde_json::from_str::<Value>(&text) else { continue };
        let request = if value.get("model").is_some_and(Value::is_object) {
            value
        } else if value.get("document_kind").and_then(|k| k.as_str()) == Some("openpipestress.product_preview.model") {
            json!({"model": value, "materials": []})
        } else { continue };
        let nodes = request["model"]["nodes"].as_array().map_or(0, |n| n.len());
        let name = file.display().to_string();
        if nodes > 1001 {
            skipped += 1;
            writeln!(out, "{}", json!({"file": name, "nodes": nodes, "result": "skipped_above_1000_members"})).unwrap();
            continue;
        }
        let published = std::panic::catch_unwind(|| {
            let e = run_linear_static_preview_value_with_mode(request.clone(), PreviewSolverMode::SparseInteractive);
            e.map(|e| e.status.mechanics).unwrap_or_else(|err| format!("ERR {err}"))
        }).unwrap_or_else(|_| "PANIC".into());
        let r = request.clone();
        let n2 = name.clone();
        let result = std::panic::catch_unwind(move || assert_sparse_wiring_parity(&n2, &r));
        let row = match result {
            Ok((b, c)) => { checked += 1; json!({"file": name, "nodes": nodes, "product": published, "result": "parity", "bases": b, "cases": c}) }
            Err(panic) => {
                let msg = panic.downcast_ref::<String>().cloned().or_else(|| panic.downcast_ref::<&str>().map(|s| s.to_string())).unwrap_or_default();
                let front_end = msg.contains("called `Option::unwrap()`") || msg.contains("called `Result::unwrap()`") || msg.contains("no basis");
                if !front_end || published == "MECHANICS_SOLVED" { failed += 1; }
                json!({"file": name, "nodes": nodes, "product": published, "result": if front_end {"front_end_unwrap"} else {"MISMATCH"}, "message": msg.chars().take(400).collect::<String>()})
            }
        };
        writeln!(out, "{row}").unwrap();
    }
    writeln!(out, "{}", json!({"summary": {"checked": checked, "failed": failed, "skipped": skipped}})).unwrap();
    eprintln!("SCRATCH-CORPUS checked={checked} failed={failed} skipped={skipped}");
    assert_eq!(failed, 0);
}
