def reset():
    import shutil
    if os.path.exists(SCR):
        shutil.rmtree(SCR)
    shutil.copytree(WT, SCR, ignore=shutil.ignore_patterns("target"))
def patch(edits):
    for path, old, new in edits:
        full = os.path.join(SCR, path)
        text = open(full).read()
        assert text.count(old) == 1, (path, old[:60], text.count(old))
        open(full, "w").write(text.replace(old, new))
def run(crate_dir, args, extra=()):
    cmd = ["cargo", "test", "--offline", "--locked"] + list(args) + list(extra)
    p = subprocess.run(cmd, cwd=os.path.join(SCR, crate_dir), env=ENV, capture_output=True, text=True)
    out = p.stdout + p.stderr
    failed = sorted(set(re.findall(r"^test (\S+) \.\.\. FAILED", out, re.M)))
    compile_error = "error[" in out or "could not compile" in out and not failed
    return {"command": " ".join(cmd), "exit": p.returncode, "failed_tests": failed, "compile_error": bool(compile_error and not failed)}
results = []
only = sys.argv[2:]
for label, what, edits, crate in MUTANTS:
    if only and label not in only: continue
    crate_dir, args = CRATES[crate]
    entry = {"mutant": label, "mutation": what, "patch": [{"file": e[0], "old": e[1], "new": e[2]} for e in edits], "runs": {}}
    reset(); patch(edits)
    entry["runs"]["all"] = run(crate_dir, args)
    if re.fullmatch(r"M1[a-p]", label):
        entry["runs"]["site_table"] = run("solver/frame_kernel", ["--test", "s11_site_table"])
        for g, gedits in G_EDITS.items():
            reset(); patch(edits); patch([x for x in gedits if (x[0].startswith("solver/straight_pipe") and crate == "SP") or (x[0].startswith("loads/") and crate == "LCA")])
            entry["runs"]["G=" + g] = run(crate_dir, args)
    results.append(entry)
    print(label, {k: (v["exit"], len(v["failed_tests"])) for k, v in entry["runs"].items()}, flush=True)
reset()
json.dump(results, open(sys.argv[1] + "/mutation_results.json" if not only else sys.argv[1] + "/mutation_results_partial.json", "w"), indent=1)
