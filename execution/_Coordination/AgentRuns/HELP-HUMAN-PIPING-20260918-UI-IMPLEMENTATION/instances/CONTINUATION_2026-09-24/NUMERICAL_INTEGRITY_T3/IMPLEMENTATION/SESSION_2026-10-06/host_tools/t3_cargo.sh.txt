#!/bin/bash
# T3 host rule: one cargo job at a time across all T3 work (ROOT, 2026-10-06 UTC).
# Usage: WT/tools/t3_cargo.sh <cargo args...>   (blocks until the host-wide lock is free)
# Refuses to start if the memory guard is not running. While waiting, no process command line
# contains "cargo " (so DEC-025's quiet-host check does not count waiters as busy).
WT="${T3_WT:?set T3_WT to the T3 root, WT}"
if ! pgrep -f "$WT/guard/memguard.sh" >/dev/null; then
  echo "t3_cargo: memory guard not running; refusing to start" >&2; exit 97
fi
echo "$(date -u '+%FT%TZ') WAIT pid=$$ cwd=$PWD args: $*" >> "$WT/guard/cargo_jobs.log"
exec /usr/bin/lockf -k "$WT/guard/cargo_job.lock" "$WT/tools/t3_cargo_run.sh" "$@"
