"""Mirror of the Rust site-table scanner (S11 R3-3 / R3B-5), used to draft the table."""
import sys, re, json
PATTERNS = ["+=", "-=", ".sum(", ".sum::<", "fold("]
def lex(src):
    """Return code with comments removed and string/char literal contents blanked."""
    out = []; i = 0; n = len(src)
    while i < n:
        c = src[i]
        if src.startswith("//", i):
            j = src.find("\n", i); j = n if j < 0 else j
            i = j; continue
        if src.startswith("/*", i):
            depth = 1; i += 2
            while i < n and depth:
                if src.startswith("/*", i): depth += 1; i += 2
                elif src.startswith("*/", i): depth -= 1; i += 2
                else:
                    if src[i] == "\n": out.append("\n")
                    i += 1
            continue
        m = re.match(r'b?r(#*)"', src[i:])
        if m and (i == 0 or not (src[i-1].isalnum() or src[i-1] == '_')):
            hashes = m.group(1); end = '"' + hashes
            j = src.find(end, i + len(m.group(0)))
            seg = src[i:j + len(end)]
            out.append('""' + "\n" * seg.count("\n")); i = j + len(end); continue
        if c == '"' or (c == 'b' and i + 1 < n and src[i+1] == '"' and (i == 0 or not (src[i-1].isalnum() or src[i-1]=='_'))):
            if c == 'b': i += 1
            j = i + 1
            while j < n and src[j] != '"':
                if src[j] == '\\': j += 1
                j += 1
            seg = src[i:j+1]
            out.append('""' + "\n" * seg.count("\n")); i = j + 1; continue
        if c == "'":
            m = re.match(r"'(\\.[^']*|[^'\\])'", src[i:])
            if m:
                out.append("' '"); i += len(m.group(0)); continue
        out.append(c); i += 1
    return "".join(out)
def strip_cfg_test(code):
    out = []; i = 0
    while True:
        j = code.find("#[cfg(test)]", i)
        if j < 0: out.append(code[i:]); break
        out.append(code[i:j])
        k = j + len("#[cfg(test)]")
        # skip to end of the next item: first ';' before any '{', or matching brace block
        depth = 0; started = False
        while k < len(code):
            ch = code[k]
            if ch == '{': depth += 1; started = True
            elif ch == '}':
                depth -= 1
                if started and depth == 0: k += 1; break
            elif ch == ';' and not started: k += 1; break
            k += 1
        out.append("\n" * code[j:k].count("\n")); i = k
    return "".join(out)
def scan(src):
    code = strip_cfg_test(lex(src))
    counts = {}
    stack = []  # (name, depth)
    pending = None; depth = 0
    tokens = re.finditer(r"\bfn\s+([A-Za-z_][A-Za-z0-9_]*)|[{};()\[\]]|\+=|-=|\.sum\(|\.sum::<|\bfold\(", code)
    nest = 0
    for t in tokens:
        s = t.group(0)
        if t.group(1):
            pending = t.group(1)
        elif s in "([":
            nest += 1
        elif s in ")]":
            nest -= 1
        elif s == ';' and nest > 0:
            pass
        elif s == '{':
            depth += 1
            if pending is not None: stack.append((pending, depth)); pending = None
        elif s == '}':
            if stack and stack[-1][1] == depth: stack.pop()
            depth -= 1
        elif s == ';':
            pending = None  # fn declaration without body (trait)
        else:
            if s.endswith("("): nest += 1
            name = stack[-1][0] if stack else "<module>"
            counts[name] = counts.get(name, 0) + 1
    return counts
if __name__ == "__main__":
    res = {}
    for p in sys.argv[1:]:
        res[p] = scan(open(p).read())
    print(json.dumps(res, indent=1))
