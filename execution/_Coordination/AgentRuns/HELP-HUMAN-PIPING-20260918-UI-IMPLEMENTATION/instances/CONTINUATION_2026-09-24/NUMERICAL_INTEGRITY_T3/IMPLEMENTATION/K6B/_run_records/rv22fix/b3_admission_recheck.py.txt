"""K6b RV22-2: b3's admissions re-evaluated with the regenerated counts (the new E_max), each row
against the records b3 had before it (the same rule, ascent and baselines). Standard library.
Usage: python3 b3_admission_recheck.py <runner dir> <b3 records> <new counts.jsonl>"""
import sys
sys.dont_write_bytecode = True
sys.path.insert(0, sys.argv[1])
import k6_runner as r  # noqa: E402

records = r.read_records(sys.argv[2])
counts = r.read_counts(sys.argv[3])
rows = [x for x in r.run_entries(records)]
changed = worst_backstop = worst_half = 0
for i, x in enumerate(rows):
    before = rows[:i]
    run = next(s for s in r.schedule() if s['order'] == x['order'])
    run = dict(run, conditional=False)
    d = r.admission(run, counts, before, x.get('baseline_rss_bytes'),
                    baseline_footprint_bytes=x.get('baseline_footprint_bytes'))
    old = x['admission']
    if d['decision'] != old['decision']:
        changed += 1
        print('CHANGED', x['order'], x['model'], x['mode'], old['decision'], '->', d['decision'], d['reason'])
    if x['mode'] == 'w1a':
        worst_backstop = max(worst_backstop, d['estimate_adm_bytes'] / (run['heap_cap_bytes'] // 2))
        worst_half = max(worst_half, (d['footprint_estimate_bytes'] or 0) / d['half_cap_bytes'])
        if x['members'] == 10000 and x['pass'] == 1:
            print(x['order'], x['model'], 'E_adm %d (was %d)' % (d['estimate_adm_bytes'], old['estimate_adm_bytes']),
                  'rho %.3f' % d['rho'], 'E/(heap cap/2) %.3f' % (d['estimate_adm_bytes'] / (run['heap_cap_bytes'] // 2)),
                  'projected RSS %d <= %d' % (d['projected_rss_bytes'], d['projected_rss_limit_bytes']), d['decision'])
print('rows %d, decisions changed %d; largest w1a E/(heap cap/2) %.3f; largest w1a E*rho/(C/2) %.3f'
      % (len(rows), changed, worst_backstop, worst_half))
