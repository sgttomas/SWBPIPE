W=$1; S=$2; T=$S/combined/t9; M=$S/combined/mut; L2=$S/combined/logs

args=()
for f in check_frame_kernel.log check_sparse_direct.log check_nonlinear_integration.log test_fk_k1_k2a_1.log test_fk_k1_k2a_2.log test_fk_full.log test_sd_full.log test_ni_full.log test_pp_site_fc.log; do args+=("$L2/$f=checkpoint_a/$f"); done
