#!/bin/bash
W=<wt>; A=$W/scratch/i13/a1
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0
side=cand
( cd $A/t9/h_$side && CARGO_TARGET_DIR=$A/t9-target-$side cargo build --release --offline --locked -j 8 > $A/t9/build_${side}_final.log 2>&1; echo "build rc=$?" >> $A/t9/build_${side}_final.log )
BIN=$A/t9-target-$side/release/calib_fixdiff_native
: > $A/t9/run_${side}_final.log
for root in core fixtures validation; do
  mkdir -p $A/t9/out_$side/$root
  $BIN $A/trees/$side/projects/chirality-piping/$root $A/t9/out_$side/$root >> $A/t9/run_${side}_final.log 2>&1; echo "$root rc=$?" >> $A/t9/run_${side}_final.log
done
( cd $A/t9/out_$side && find . -type f -name '*.out' | sort | xargs shasum -a 256 ) > $A/t9/sha_$side.txt
mkdir -p $A/t9/extra_cand_final; $BIN $A/extra_corpus $A/t9/extra_cand_final; (cd $A/t9/extra_cand_final && ls | sort | xargs shasum -a 256) > $A/t9/extra_sha_cand_final.txt
echo T9-CAND-FINAL-DONE
