
// ---------------------------------------------------------------- SCRATCH ONLY (I13 C: F1B-M8 witness search)
#[test]
fn scratch_m8_witness_search() {
    // K2a's least-subnormal member with a ground spring k on N1:UY and a load F there.
    let f = pow2(-1021);
    let mut found = 0;
    for step in 0..4000 {
        let k = pow2(-958) * (1.0 + step as f64 / 1024.0);
        let request = reach_request(
            "m8", 1.8189894035458565e-12, 1.0e-11, 1.0e-12, 9.6e-280, 1.0e-100, "UY", Some(k), f,
        );
        let bases = product_bases(&request);
        let basis = &bases[0];
        let (_, force, _) = &basis.cases[0];
        let prescribed: Vec<(usize, f64)> = basis.restrained.iter().map(|&d| (d, 0.0)).collect();
        let inputs = attempt_inputs(&basis.built, &basis.springs, force, &prescribed);
        for mode in MODES {
            let Ok(outcome) = solve_with_force_scaling(&orchestrator_case(basis, &inputs, force, &prescribed, mode)) else { continue };
            if outcome.solution.force_scale.is_unscaled() { continue; }
            let u = &outcome.solution.solution.displacements;
            let dof = basis.springs[0].node_dof.global_index();
            let spring_k = basis.springs[0].stiffness.value;
            let Ok(helper) = force_scaled_spring_action((dof, spring_k), u, outcome.solution.force_scale) else { continue };
            let unchecked = -spring_k * u[dof];
            if helper.value.to_bits() != unchecked.to_bits() {
                println!("WITNESS step={step} k_bits={:#018x} f_bits={:#018x} mode={mode:?} b={} helper={:#018x} unchecked={:#018x} repr={:?}",
                    k.to_bits(), f.to_bits(), outcome.solution.force_scale.exponent(), helper.value.to_bits(), unchecked.to_bits(), helper.representability);
                found += 1;
            }
        }
        if found >= 6 { break; }
    }
    println!("FOUND {found}");
}
