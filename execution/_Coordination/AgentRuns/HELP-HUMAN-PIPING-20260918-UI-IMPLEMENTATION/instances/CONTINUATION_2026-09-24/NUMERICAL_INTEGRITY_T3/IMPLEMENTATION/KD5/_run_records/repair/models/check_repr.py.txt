import sys
from decimal import Decimal as D
sys.argv=[sys.argv[0], sys.argv[1]]
sys.path.insert(0, sys.argv[1])
import curved_ef as ce
def model():
    return {"name": "PP_UTM_2", "nodes": {"N0": [500000.0, 350000.0, 0.0], "N1": [500000.010469849, 350000.0001827519, 0.0]},
            "members": [dict(i="N0", j="N1", yref=[0.0, 1.0, 0.0], bend={"R": 0.3, "k": 1.0})],
            "rigid": {"N0": ["UX", "UY", "UZ"]}, "springs": [("N0", d, 1e6) for d in ("RX", "RY", "RZ")],
            "loads": [("N1", d, 1.0) for d in ("RX", "RY", "RZ")], "section": (0.2, 0.18, 200e9, 80e9)}
def uint():
    m = ce.build(model()); free = m["free"]
    u = ce.dsolve([[m["Ki"][i][j] for j in free] for i in free], [ce.dec(m["f"][i]) for i in free])
    return {i: float(u[r]) for r, i in enumerate(free)}, m["info"][0]["centre"]
exact, c = uint()
orig = ce.dec
ce.dec = lambda x: D(repr(float(x)))
rep, _ = uint()
ce.dec = orig
print("centre", c)
for i in sorted(exact): print(i, exact[i], rep[i])
