#!/usr/bin/env python3
"""F1b lane guard (ROOT's ruling on the gate's heap-cap finding): the required mutant.
F1B-LANE-UNGUARDED: the observation lane's guard removed (the lane always runs)."""
import sys
core, mid = sys.argv[1], sys.argv[2]
LIB = core + "/product_physics/src/lib.rs"
M = {"F1B-LANE-UNGUARDED": [(LIB, """    let lane_refusal = direct
        .as_ref()
        .and_then(|d| observation_lane_guard(d, dense_scrutiny_ceiling_bytes()).err());""",
                             "    let lane_refusal: Option<ObservationLaneRefusal> = None;")]}
if mid != "NONE":
    for path, old, new in M[mid]:
        text = open(path).read()
        assert text.count(old) == 1, (mid, old[:60])
        open(path, "w").write(text.replace(old, new))
print("applied", mid)
