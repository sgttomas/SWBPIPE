"""K-D5 (I3) no-Passed-breach gate: run driver (scratch; committed as gate_run.py.txt).

Usage: python3 -B gate_run.py <probe-binary> <gen_out-dir> <runs.jsonl>

Runs every authorable case of P1's generated R1 requests in both modes through BOTH entries
(captured and typed), each in a fresh probe process with P1's limits (run.py's run_one, imported
unchanged: RLIMIT_AS 6 GiB, 600 s, 1800 s for >= 1000 members). Before every run it waits while any
cargo process is running (shared host). Resumable: (case, mode, entry) already in runs.jsonl are skipped.
"""
import json
import os
import subprocess
import sys
import time

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from run import run_one, TIMEOUT_LARGE, TIMEOUT_SMALL  # noqa: E402

binary, gen, runs_path = sys.argv[1:4]
done = set()
if os.path.exists(runs_path):
    for line in open(runs_path):
        r = json.loads(line)
        done.add((r['case'], r['mode'], r['entry']))
cases = [c for c in json.load(open(os.path.join(gen, 'cases.json'))) if 'request_file' in c]
cases.sort(key=lambda c: (c.get('members', 0) >= 1000, c.get('members', 0), c['id']))
with open(runs_path, 'a') as fh:
    for c in cases:
        large = c.get('members', 0) >= 1000
        for mode in ('sparse_interactive', 'dense_scrutiny'):
            for entry, typed in (('captured', False), ('typed', True)):
                if (c['id'], mode, entry) in done:
                    continue
                while subprocess.run(['pgrep', '-x', 'cargo'], capture_output=True).returncode == 0 or \
                        subprocess.run(['pgrep', '-f', 'python[0-9.]* .*run_evidence_sweep'], capture_output=True).returncode == 0:
                    time.sleep(30)
                rec = run_one(binary, os.path.join(gen, c['request_file']), mode, typed,
                              TIMEOUT_LARGE if large else TIMEOUT_SMALL)
                rec.update(case=c['id'], mode=mode, entry=entry)
                fh.write(json.dumps(rec) + '\n')
                fh.flush()
                run = rec.get('probe', {}).get('run', {})
                print(c['id'], mode, entry, 'ok' if run.get('ok') else 'ERR', rec['wall_seconds'], rec['peak_rss_kib'], flush=True)
