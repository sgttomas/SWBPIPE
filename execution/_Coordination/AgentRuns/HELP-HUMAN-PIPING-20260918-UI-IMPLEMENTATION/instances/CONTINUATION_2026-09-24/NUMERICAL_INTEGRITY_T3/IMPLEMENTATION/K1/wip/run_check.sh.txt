#!/bin/bash
# K1 compile check (I8): cargo check --tests for the three kernel crates, K1's own target.
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=<wt>/k1-target
C=<wt>/k1/projects/chirality-piping/core/solver
OUT=$1
: > $OUT
for crate in frame_kernel sparse_direct nonlinear_integration; do
  echo "=== $crate $(date -u +%FT%TZ) disk_free_kb=$(df -k --output=avail /home | tail -1)" >> $OUT
  (cd $C/$crate && nice -n 10 cargo check --tests --offline --locked --message-format short) >> $OUT 2>&1
  echo "=== $crate exit=$?" >> $OUT
done
echo "=== done $(date -u +%FT%TZ) disk_free_kb=$(df -k --output=avail /home | tail -1)" >> $OUT
