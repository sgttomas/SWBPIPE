"""K6b B, slot K6B-S3: W1-T4 (10,000 members), as ROOT approved it ("K6b: B's W1-T1 to W1-T3
accepted; W1-T4 approved before KF1").

The runner is the archived one (f4d40dd17), unchanged. Two recorded overrides, both ROOT's:
- the tier's by-name deferral (CONDITIONAL_TIERS['W1-T4']) is lifted, because ROOT approved it;
- a heap-cap abort is a finding for KF1, not a stop: if run_tier stops and every reason is a
  heap-cap abort, the tier is resumed (run_tier skips runs already recorded). Any other stop
  ends the tier.
"""
import json
import sys

sys.dont_write_bytecode = True
runner, binary, counts, records, commit, tree = sys.argv[1:7]
sys.path.insert(0, runner)
import k6_runner as r  # noqa: E402

r.CONDITIONAL_TIERS = {}
print('override: CONDITIONAL_TIERS emptied for W1-T4 (ROOT approval)', flush=True)
findings = []
while True:
    out = r.run_tier('W1-T4', binary, counts, records, commit, tree,
                     log=lambda *a: print(*a, flush=True))
    print(json.dumps(out), flush=True)
    if not out['stop']:
        break
    if all('heap_cap_abort on an admitted run' in reason for reason in out['stop']):
        findings.extend(out['stop'])
        print('FINDING (KF1, not a stop): %s; resuming the tier' % out['stop'], flush=True)
        continue
    print('STOPPED: %s' % out['stop'], flush=True)
    sys.exit(1)
print('KF1 findings: %s' % findings, flush=True)
print('W1T4_DONE', flush=True)
