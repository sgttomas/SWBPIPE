#!/bin/bash
# Usage: run_suites.sh <project-root> <log-dir>
P=$1; L=$2
export RUSTUP_TOOLCHAIN=1.97.1 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=${T3_TARGET:-<t3-target>}
export PATH=<node24>/bin:$PATH
CRATES="core/solver/frame_kernel core/solver/straight_pipe core/solver/curved_bend core/loads/load_case_algebra core/loads/primitive_loads core/solver/nonlinear_integration core/solver/sparse_direct core/solver/linear_supports core/solver/nonlinear_supports core/solver/diagnostics core/solver/performance_harness core/loads/stress_recovery core/loads/user_loads core/loads/self_weight_wasm core/product_physics core/model_operations/operation_applier core/runner/headless core/reporting/result_export validation/benchmarks/mechanics validation/benchmarks/nonlinear validation/benchmarks/stress validation/benchmarks/physics_audit_regression validation/benchmarks/numerical_integrity apps/desktop/src-tauri"
for c in $CRATES; do
  while pgrep -x cargo >/dev/null; do sleep 20; done
  name=$(echo $c | tr / _)
  start=$(date +%s)
  (cd $P/$c && cargo test --offline --locked --no-fail-fast > $L/$name.log 2>&1); rc=$?
  end=$(date +%s)
  echo "$c exit=$rc secs=$((end-start)) $(grep -E '^test result' $L/$name.log | awk '{p+=$4; f+=$6; i+=$8} END {print "passed="p" failed="f" ignored="i}')" >> $L/SUMMARY.txt
  df -h <home> | tail -1 | awk '{print "  disk_free="$4}' >> $L/SUMMARY.txt
  avail=$(df --output=avail -k <home> | tail -1)
  if [ "$avail" -lt 9000000 ]; then rm -rf "$CARGO_TARGET_DIR/debug"; echo "  pruned own debug output" >> $L/SUMMARY.txt; fi
done
echo DONE >> $L/SUMMARY.txt
