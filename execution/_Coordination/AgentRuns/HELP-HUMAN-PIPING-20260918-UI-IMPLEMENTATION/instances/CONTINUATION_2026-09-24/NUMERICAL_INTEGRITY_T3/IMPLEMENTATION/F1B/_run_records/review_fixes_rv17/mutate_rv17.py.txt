#!/usr/bin/env python3
"""RV17's own mutants of F1b (9aeed9c22). Usage: python3 mutate_rv17.py <core-dir> <id>
Each patch is an exact text replacement that must match exactly once.
- RV17-M1: the basis's range deferral also on invocations with a nonlinear support (`linear`
  forced true; the W2 arm's own nonlinear filter kept, so W2 never engages).
- RV17-M2: every W2-published spring action is the first spring's.
- RV17-M3: the dense view for retained-source recovery is built at any size (the n <= 256
  condition removed; the budget refusal still precedes every read, so bytes are unchanged).
- RV17-M4: the finiteness scan reads only the force, not the stored stiffness values."""
import sys
core, mid = sys.argv[1], sys.argv[2]
LIB = core + "/product_physics/src/lib.rs"
M = {
    "RV17-M1": [(LIB, "    let linear = built.nonlinear_supports.is_empty();\n    let stiffness = match form_basis_stiffness(",
                 "    let linear = true;\n    let stiffness = match form_basis_stiffness(")],
    "RV17-M2": [(LIB, "Some(publication) => publication.spring_actions[spring_index].value,",
                 "Some(publication) => publication.spring_actions[0].value,")],
    "RV17-M3": [(LIB, """            if source_eligible
                && needs_source_recovery
                && formed.dimension() <= source_recovery::DENSE_SOURCE_DOF_LIMIT =>""",
                 """            if source_eligible
                && needs_source_recovery =>""")],
    "RV17-M4": [(LIB, """        formed
            .map_or(&[][..], SparseStiffness::values)
            .iter()
            .copied()
            .chain(force.values().iter().copied()),""",
                 """        (&[][..] as &[f64])
            .iter()
            .copied()
            .chain(force.values().iter().copied()),""")],
}
if mid != "NONE":
    for path, old, new in M[mid]:
        text = open(path).read()
        assert text.count(old) == 1, (mid, path, old[:80], text.count(old))
        open(path, "w").write(text.replace(old, new))
print("applied", mid)
