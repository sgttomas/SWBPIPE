#!/bin/bash
# F1b checkpoint C: NONE first, alone; then the listed ids, at most two at once (ROOT: the host is busy).
W=<wt>; C=$W/scratch/i13/c; REV=e215c6007
if [ "$1" = NONE ]; then
  : > $C/MUTANTS.txt
  $C/run_mutant.sh $W $C $REV NONE
  echo NONE-DONE >> $C/MUTANTS.txt
  exit 0
fi
for m in $(cat $C/$1); do
  while [ $(jobs -rp | wc -l) -ge 2 ]; do sleep 5; done
  $C/run_mutant.sh $W $C $REV $m &
done
wait
echo "DRIVE-DONE $1" >> $C/MUTANTS.txt
