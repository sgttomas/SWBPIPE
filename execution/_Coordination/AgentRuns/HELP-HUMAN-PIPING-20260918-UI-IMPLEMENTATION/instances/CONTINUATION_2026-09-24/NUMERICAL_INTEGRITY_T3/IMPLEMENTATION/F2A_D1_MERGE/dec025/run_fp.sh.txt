#!/bin/zsh
T=<WT>
S=$T/scratch/u9_dec025
V=<VENV>/bin/python
busy() { ps -axo pid,command | grep -E "cargo |rustc |pytest|vitest|/deps/[a-z_]+-[0-9a-f]{16}" | grep -v -E "grep|run_all.sh" | grep -v "^ *$$ " | wc -l | tr -d ' '; }
echo "wait-start $(date -u +%FT%TZ)" > $S/quiet_fp.log
q=0; while true; do n=$(busy); if [ "$n" = "0" ]; then q=$((q+1)); else q=0; fi; echo "$(date -u +%T) busy=$n quiet=$q" >> $S/quiet_fp.log; [ $q -ge 6 ] && break; sleep 20; done
echo "quiet $(date -u +%FT%TZ)" >> $S/quiet_fp.log
ps -p 5387 -o pid= > /dev/null || { echo "MEMGUARD DOWN" >> $S/quiet_fp.log; exit 9; }
echo "dec025-start $(date -u +%FT%TZ)" >> $S/quiet_fp.log
$S/dec025_mac.sh $T/sweep-skewpin/projects/chirality-piping $S/Fp $V $T
echo "dec025-done $(date -u +%FT%TZ)" >> $S/quiet_fp.log
