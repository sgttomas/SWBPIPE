"""I13 F1b A1 scratch: the targeted B4 run (every gate case except RF-LARGE,
792 runs, candidate P1 probe) against G1's Mac base of e7d930d49, per
(case, mode, entry): the envelope span bytes, the not-ok error text, the
exit code, and compare.classify's outcome. Standard library only."""
import json, os, sys, hashlib, collections
base, cand = sys.argv[1], sys.argv[2]
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import compare as C
def load(d):
    out = {}
    for line in open(os.path.join(d, 'runs.jsonl')):
        r = json.loads(line)
        out[(r['case'], r['mode'], r['entry'])] = r
    return out
b, c = load(base), load(cand)
keys = sorted(c)
diff = collections.Counter(); rows = []
for k in keys:
    rb, rc = b[k], c[k]
    key = '__'.join(k)
    def env(d):
        p = os.path.join(d, 'envelopes', key + '.json')
        return open(p, 'rb').read() if os.path.exists(p) else None
    eb, ec = env(base), env(cand)
    def err(r):
        run = (r.get('probe') or {}).get('run') or {}
        return None if run.get('ok') else json.dumps({k: v for k, v in run.items() if k != 'solve_seconds'}, sort_keys=True)
    same = (eb == ec) and (err(rb) == err(rc)) and rb['exit_code'] == rc['exit_code']
    cls_b = C.classify(rb) if hasattr(C, 'classify') else None
    cls_c = C.classify(rc) if hasattr(C, 'classify') else None
    if not same or cls_b != cls_c:
        diff['different'] += 1
        rows.append({'key': key, 'env_same': eb == ec, 'err_same': err(rb) == err(rc), 'exit': [rb['exit_code'], rc['exit_code']], 'class': [str(cls_b), str(cls_c)]})
    else:
        diff['identical'] += 1
    diff['outcome:' + str(cls_c)] += 1
print(json.dumps({'runs': len(keys), 'counts': diff, 'differences': rows}, indent=1, default=str))
