"""I13 F1b A2: product runs through P1's run_one, with full-envelope capture (scratch).

Derived from G1's gate_run_base_envelopes.py (sha256 in G1's SHA256SUMS). Unchanged from it:
run.py's run_one imported unchanged (resource.setrlimit a no-op; the probe caps its own heap at
6 GiB), both entries and both modes for every case, the timeouts, the tee of the child's pipes, the
envelope span rule, and the per-run check that the kept stdout line parses to the record's probe.

Added:
- The probe is F1b's A2 variant (P1's probe plus `run.envelope_sha256`, the sha256 of the complete
  serialized MechanicsEnvelope). The child's environment carries T3_FULL_ENVELOPE_PATH, set per run
  through a thread-local in the Popen wrapper (never os.environ), so the full envelope bytes land in
  full/<key>.json.
- Case selection: every case of <gen_out>/cases.json whose id starts with one of the prefixes, except
  cases of 1,000 or more members (skipped and counted). A2_ENTRIES=typed restricts the entries.
- No host sampler; memorystatus_level at each run's end is recorded.

Usage: python3 -B a2_run.py <run.py dir> <probe-binary> <gen_out-dir> <out-dir> <workers> <prefix> [prefix ...]
"""
import hashlib
import json
import os
import subprocess
import sys
import threading
import time
import types
from concurrent.futures import ThreadPoolExecutor, as_completed

sys.path.insert(0, os.path.abspath(sys.argv[1]))
import run as R  # noqa: E402

R.resource.setrlimit = lambda *a, **k: None

_tls = threading.local()


class _Tee:
    def __init__(self, raw):
        self._raw = raw
        self.kept = []

    def read(self, n=-1):
        b = self._raw.read(n)
        if b:
            self.kept.append(b)
        return b

    def close(self):
        return self._raw.close()

    def __getattr__(self, name):
        return getattr(self._raw, name)


class _TeePopen(subprocess.Popen):
    def __init__(self, *a, **k):
        env = dict(os.environ)
        env['T3_FULL_ENVELOPE_PATH'] = _tls.full_path
        k['env'] = env
        super().__init__(*a, **k)
        if self.stdout is not None:
            self.stdout = _Tee(self.stdout)
        if self.stderr is not None:
            self.stderr = _Tee(self.stderr)
        _tls.proc = self


_sp = types.ModuleType('subprocess')
_sp.__dict__.update(subprocess.__dict__)
_sp.Popen = _TeePopen
R.subprocess = _sp

binary, gen, outdir, workers = sys.argv[2], sys.argv[3], sys.argv[4], int(sys.argv[5])
prefixes = tuple(sys.argv[6:])
assert 1 <= workers <= 4, 'at most 4 workers'
assert prefixes, 'name the case prefixes'
cases = [c for c in json.load(open(os.path.join(gen, 'cases.json')))
         if 'request_file' in c and c['id'].startswith(prefixes)]
skipped = [c['id'] for c in cases if c.get('members', 0) >= 1000]
cases = [c for c in cases if c.get('members', 0) < 1000]  # never a 1,000-member case here (gate-scale)
if skipped:
    print('skipped (>= 1,000 members, gate-scale):', len(skipped), flush=True)
ENTRIES = tuple(os.environ.get('A2_ENTRIES', 'captured,typed').split(','))
jobs = []
for c in cases:
    for mode in ('sparse_interactive', 'dense_scrutiny'):
        for entry, typed in (('captured', False), ('typed', True)):
            if entry not in ENTRIES:
                continue
            jobs.append((c, mode, entry, typed, R.TIMEOUT_SMALL))
for d in ('stdout', 'stderr', 'envelopes', 'full'):
    os.makedirs(os.path.join(outdir, d), exist_ok=True)
runs_path = os.path.join(outdir, 'runs.jsonl')
assert not os.path.exists(runs_path), 'refusing to overwrite ' + runs_path


def memlevel():
    return int(os.popen('sysctl -n kern.memorystatus_level').read().strip() or -1)


def envelope_span(text, parsed):
    run = parsed.get('run', {})
    if 'envelope' not in run:
        return None
    prefix = '{"mode":' + json.dumps(parsed['mode']) + ',"run":{"entry":' + json.dumps(run['entry']) + ',"envelope":'
    if not text.startswith(prefix):
        raise ValueError('unexpected probe line layout')
    env, end = json.JSONDecoder().raw_decode(text, len(prefix))
    if env != run['envelope']:
        raise ValueError('envelope span does not parse to the recorded envelope')
    return text[len(prefix):end]


def go(j):
    c, mode, entry, typed, to = j
    key = '%s__%s__%s' % (c['id'], mode, entry)
    full_path = os.path.join(outdir, 'full', key + '.json')
    _tls.full_path = full_path
    _tls.proc = None
    rec = R.run_one(binary, os.path.join(gen, c['request_file']), mode, typed, to)
    proc = _tls.proc
    rec.update(case=c['id'], mode=mode, entry=entry, rlimit_as_enforced=False, heap_cap_bytes=6 * 1024 ** 3,
               memorystatus_level_end=memlevel())
    out_b = b''.join(proc.stdout.kept)
    err_b = b''.join(proc.stderr.kept)
    with open(os.path.join(outdir, 'stdout', key + '.out'), 'wb') as fh:
        fh.write(out_b)
    with open(os.path.join(outdir, 'stderr', key + '.err'), 'wb') as fh:
        fh.write(err_b)
    env_sha = '-'
    full_sha = '-'
    text = out_b.decode('utf-8', 'replace').strip()
    if text:
        parsed = json.loads(text.splitlines()[-1])
        if parsed != rec.get('probe'):
            raise RuntimeError('tee mismatch for ' + key)
        span = envelope_span(out_b.decode('utf-8').strip().splitlines()[-1], parsed)
        if span is not None:
            sb = span.encode('utf-8')
            with open(os.path.join(outdir, 'envelopes', key + '.json'), 'wb') as fh:
                fh.write(sb)
            env_sha = hashlib.sha256(sb).hexdigest()
            full = open(full_path, 'rb').read()
            full_sha = hashlib.sha256(full).hexdigest()
            if full_sha != parsed['run']['envelope_sha256']:
                raise RuntimeError('full envelope hash mismatch for ' + key)
    rec['a2'] = dict(key=key, envelope_span_sha256=env_sha, full_envelope_sha256=full_sha,
                     stdout_sha256=hashlib.sha256(out_b).hexdigest())
    return rec


t0 = time.time()
with open(runs_path, 'w') as fh:
    with ThreadPoolExecutor(workers) as ex:
        futs = [ex.submit(go, j) for j in jobs]
        for f in as_completed(futs):
            rec = f.result()
            fh.write(json.dumps(rec) + '\n')
            fh.flush()
print('done', len(jobs), 'runs in', round(time.time() - t0), 's; memorystatus_level', memlevel(), flush=True)
