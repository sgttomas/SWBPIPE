//! I14 (K5, A2) scratch harness, never committed: R1's nine RF-MECH models
//! (frame-only) through the adapter's public geometry and the four selected
//! entries, dense evidence up to 200 members and sparse evidence always.
//! Built from `git archive` copies of the base and the candidate; the two
//! outputs must be byte-identical (frame-only bodies never reach W4).
use open_pipe_stress_frame_kernel::load_ledger::LoadLedger;
use open_pipe_stress_frame_kernel::structural::{assemble_sparse_stiffness, SparseAssemblyOptions};
use open_pipe_stress_frame_kernel::{ForceScale, FrameElement, FrameNode, FrameSection};
use open_pipe_stress_nonlinear_integration::structural_adapter::{
    AssemblyEvidence, SparseAssemblyEvidence,
};
use open_pipe_stress_nonlinear_integration::LinearSolveMode;

struct Model {
    nodes: Vec<[f64; 3]>,
    frames: Vec<FrameElement>,
    rigid: Vec<usize>,
    springs: Vec<(usize, f64)>,
    loads: Vec<(usize, f64)>,
}

fn parse(text: &str) -> Model {
    let mut lines = text.lines();
    let mut count = |tag: &str, lines: &mut std::str::Lines| -> usize {
        let line = lines.next().unwrap();
        let (t, n) = line.split_once(' ').unwrap();
        assert_eq!(t, tag);
        n.parse().unwrap()
    };
    let n = count("nodes", &mut lines);
    let nodes: Vec<[f64; 3]> = (0..n)
        .map(|_| {
            let v: Vec<f64> = lines.next().unwrap().split(' ').map(|x| x.parse().unwrap()).collect();
            [v[0], v[1], v[2]]
        })
        .collect();
    let m = count("members", &mut lines);
    let frames = (0..m)
        .map(|_| {
            let v: Vec<&str> = lines.next().unwrap().split(' ').collect();
            let f = |k: usize| v[k].parse::<f64>().unwrap();
            let (i, j): (usize, usize) = (v[0].parse().unwrap(), v[1].parse().unwrap());
            let section = FrameSection::new(f(2), f(3), f(4), f(5), f(5), f(6)).unwrap();
            FrameElement::new(
                FrameNode::new(i, nodes[i]).unwrap(),
                FrameNode::new(j, nodes[j]).unwrap(),
                section,
                [f(7), f(8), f(9)],
            )
            .unwrap()
        })
        .collect();
    let r = count("rigid", &mut lines);
    let rigid = (0..r).map(|_| lines.next().unwrap().parse().unwrap()).collect();
    let pair = |line: &str| {
        let (a, b) = line.split_once(' ').unwrap();
        (a.parse().unwrap(), b.parse().unwrap())
    };
    let s = count("springs", &mut lines);
    let springs = (0..s).map(|_| pair(lines.next().unwrap())).collect();
    let l = count("loads", &mut lines);
    let loads = (0..l).map(|_| pair(lines.next().unwrap())).collect();
    Model {
        nodes,
        frames,
        rigid,
        springs,
        loads,
    }
}

#[test]
fn k5_rfmech_harness() {
    let dir = std::env::var("K5_RFMECH_DIR").expect("K5_RFMECH_DIR");
    let mut names: Vec<String> = std::fs::read_dir(&dir)
        .unwrap()
        .map(|e| e.unwrap().file_name().into_string().unwrap())
        .filter(|n| n.ends_with(".txt"))
        .collect();
    names.sort();
    for name in names {
        let m = parse(&std::fs::read_to_string(format!("{dir}/{name}")).unwrap());
        let n = m.nodes.len();
        let prescribed: Vec<(usize, f64)> = m.rigid.iter().map(|&d| (d, 0.0)).collect();
        let free: Vec<usize> = (0..6 * n).filter(|d| !m.rigid.contains(d)).collect();
        let mut ledger = LoadLedger::new();
        for (k, &(dof, v)) in m.loads.iter().enumerate() {
            ledger.push(format!("load:{k}"), dof, v);
        }
        let f = ledger.finish(6 * n).unwrap();
        let s2 = ForceScale::new(2).unwrap();
        let k = assemble_sparse_stiffness(n, &m.frames, &[], &[], &m.springs, &SparseAssemblyOptions::new()).unwrap();
        let k2 = assemble_sparse_stiffness(n, &m.frames, &[], &[], &m.springs, &SparseAssemblyOptions::new().with_force_scale(s2)).unwrap();
        let se = SparseAssemblyEvidence::new(k.pattern(), n, &m.frames, &[], &[], &m.springs).unwrap();
        let se2 = SparseAssemblyEvidence::new_force_scaled(k2.pattern(), n, &m.frames, &[], &[], &m.springs, s2).unwrap();
        let small = m.frames.len() <= 200;
        let modes: &[LinearSolveMode] = if small {
            &[LinearSolveMode::SparseInteractive, LinearSolveMode::DenseScrutiny]
        } else {
            &[LinearSolveMode::SparseInteractive]
        };
        let status = |g: &Result<(), open_pipe_stress_frame_kernel::structural::StructuralError>| match g {
            Ok(()) => "Restrained".to_string(),
            Err(e) => format!("{}", format!("{e:?}").split([' ', '{', '(']).next().unwrap()),
        };
        let geometry = se.geometry(&prescribed);
        println!("K5RFMECH {name} STATUS sparse-geometry={}", status(&geometry));
        println!("K5RFMECH {name} sparse geometry {geometry:?}");
        for &mode in modes {
            println!("K5RFMECH {name} sparse checked {mode:?} {:?}", se.solve_assembled_with_formation_check(&k, &f, &free, &prescribed, mode, &[], true));
            println!("K5RFMECH {name} sparse force-scaled checked b=0 {mode:?} {:?}", se.solve_force_scaled_with_formation_check(&k, &f, &free, &prescribed, mode, &[], true));
            println!("K5RFMECH {name} sparse force-scaled checked b=2 {mode:?} {:?}", se2.solve_force_scaled_with_formation_check(&k2, &f, &free, &prescribed, mode, &[], true));
        }
        if small {
            let dense = k.to_dense();
            let dense2 = k2.to_dense();
            let de = AssemblyEvidence::new(n, &m.frames, &[], &[], &m.springs).unwrap();
            let de2 = AssemblyEvidence::new_force_scaled(n, &m.frames, &[], &[], &m.springs, s2).unwrap();
            let geometry = de.geometry(&prescribed);
            println!("K5RFMECH {name} STATUS dense-geometry={}", status(&geometry));
            println!("K5RFMECH {name} dense geometry {geometry:?}");
            for &mode in modes {
                println!("K5RFMECH {name} dense checked {mode:?} {:?}", de.solve_assembled_with_formation_check(&dense, &f, &free, &prescribed, mode, &[], true));
                println!("K5RFMECH {name} dense force-scaled checked b=0 {mode:?} {:?}", de.solve_force_scaled_with_formation_check(&dense, &f, &free, &prescribed, mode, &[], true));
                println!("K5RFMECH {name} dense force-scaled checked b=2 {mode:?} {:?}", de2.solve_force_scaled_with_formation_check(&dense2, &f, &free, &prescribed, mode, &[], true));
            }
        }
    }
}
