#!/bin/bash
# RV11 delta fixes (I10): the new mutants, RV11's three delta survivors and K2b's three pin mutants
# (the pin list changed), from clean archives of f385a8bc8 with the delta fix overlaid (overlay.txt);
# the NONE control first, alone; then at most three at once, -j 4 each (run_mutant.sh).
W=<wt>; C=$W/scratch/i10/rv11d/mut; REV=f385a8bc8
: > $C/MUTANTS.txt
$C/run_mutant.sh $W $C $REV NONE $C/overlay.txt
for m in $(cat $C/ids.txt); do
  while [ $(jobs -rp | wc -l) -ge 3 ]; do sleep 5; done
  $C/run_mutant.sh $W $C $REV $m $C/overlay.txt &
done
wait
echo DRIVE-DONE >> $C/MUTANTS.txt
