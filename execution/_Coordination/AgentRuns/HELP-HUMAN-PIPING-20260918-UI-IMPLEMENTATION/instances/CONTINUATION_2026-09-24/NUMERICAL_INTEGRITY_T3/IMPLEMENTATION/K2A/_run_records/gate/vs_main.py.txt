"""K2a (I6): per-run standing of the K2a gate against main's (K-D5's combined-tree final gate result,
KD5/_run_records/combined/gate/final_result.json, whose product source equals main 5ae22926e's: only test
files differ, `git diff --stat 2409de83e 5ae22926e -- P/core P/fixtures P/validation P/schemas P/apps`).
Usage: python3 -B vs_main.py <k2a result.json> <kd5 final_result.json>
Compares outcome, quality, standing, trusted and breach count per (case, mode, entry)."""
import json, sys
a = {(r['case'], r['mode'], r['entry']): r for r in json.load(open(sys.argv[1]))['runs']}
b = {(r['case'], r['mode'], r['entry']): r for r in json.load(open(sys.argv[2]))['runs']}
keys = ('outcome', 'quality', 'standing', 'trusted', 'breaches')
common = sorted(set(a) & set(b))
print('runs in K2a result:', len(a), ' in main (K-D5 final):', len(b), ' compared:', len(common))
print('only in K2a:', sorted(set(a) - set(b)))
print('only in main:', sorted(set(b) - set(a)))
changes = [(k, {f: b[k].get(f) for f in keys}, {f: a[k].get(f) for f in keys}) for k in common
           if any(a[k].get(f) != b[k].get(f) for f in keys)]
print('standing changes (main -> K2a):', len(changes))
for c in changes:
    print(' ', c)
