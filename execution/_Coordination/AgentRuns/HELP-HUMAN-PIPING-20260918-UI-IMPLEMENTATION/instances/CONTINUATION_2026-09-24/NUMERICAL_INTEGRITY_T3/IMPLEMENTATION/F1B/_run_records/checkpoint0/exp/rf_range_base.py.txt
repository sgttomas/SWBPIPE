"""I13 checkpoint-0 read-only scan of the Mac gate baseline (G1, main e7d930d49,
<wt>/scratch/gate_base_e7d930d49/runs.jsonl): every RF-RANGE run's outcome and
blocking diagnostics, for F1b's proposed C1 list. Standard library only."""
import json, sys, collections
path = sys.argv[1]
out = []
with open(path) as fh:
    for line in fh:
        r = json.loads(line)
        if not str(r["case"]).startswith("RF-RANGE"):
            continue
        run = (r.get("probe") or {}).get("run") or {}
        env = run.get("envelope")
        status, blocking, nres = None, [], None
        if isinstance(env, dict):
            status = env["status"]["mechanics"]
            nres = env.get("result_count")
            blocking = [(d["code"], d["id"], d["message"][:170]) for d in env["diagnostics"] if d["severity"] == "blocking"]
        err = None if run.get("ok") else str(run.get("error") or run)[:160]
        out.append((r["case"], r["entry"], r["mode"], status, nres, blocking, err))
cnt = collections.Counter()
for o in sorted(out):
    key = o[5][0][0] if o[5] else ("ERR" if o[6] else o[3])
    cnt[(key, o[1])] += 1
    print(f"{o[0]:30s} {o[1]:8s} {o[2]:18s} {str(o[3]):22s} results={o[4]} {'ERR '+o[6] if o[6] else ''}")
    for b in o[5]:
        print(f"      {b[0]} [{b[1]}] {b[2]}")
print(cnt)
