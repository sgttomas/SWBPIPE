//! Scratch probe (T3 KF3 at D, finding KF3-B1; not committed): R7's
//! verification estimate (b) against its threshold, per candidate precision,
//! on one model file (K4T text format, env KF3_B1_MODEL).
use super::super::recover::{End, Kind, QuantityId};
use super::super::source::Component;
use super::super::verify::{build_verify_shared, e_hat, verify_state};
use super::*;
use std::time::Instant;

#[allow(dead_code)]
#[path = "models.rs"]
mod models;
#[allow(dead_code)]
#[path = "support.rs"]
mod support;

/// Ŵ_q / (2^(6−P)·ê) for every force and moment row: (index, ratio) sorted by
/// ratio, as f64 (upward).
fn ratios<const L: usize>(
    r: &super::super::verify::VerificationReport<L>,
    prep: &CasePrep,
) -> Vec<(usize, f64)>
where
    Wide<L>: SupportedWidth,
{
    let p = i64::from(r.precision);
    let hats: Vec<[f64; 2]> = r
        .resolution
        .iter()
        .zip(&prep.extents)
        .map(|(e, &x)| e_hat(*e, x))
        .collect();
    let mut out = Vec::new();
    for (index, meta) in prep.layout.iter().enumerate() {
        let k = match meta.kind {
            Kind::Force => 0,
            Kind::Moment => 1,
            _ => continue,
        };
        let Some(w) = &r.w[index] else { continue };
        let e = Wide::<L>::from_f64(hats[meta.body as usize][k]).unwrap();
        let thr = e.mul_pow2(6 - p).unwrap();
        let v = if w.is_zero() {
            0.0
        } else if thr.is_zero() {
            f64::INFINITY
        } else {
            let mut ctx = WideContext::<L>::new(64).unwrap();
            super::super::directed::binary64_up(&ctx.div(w, &thr).unwrap()).unwrap()
        };
        out.push((index, v));
    }
    out.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());
    out
}

macro_rules! pair {
    ($prep:expr, $group:expr, $p:expr, $q:expr, $L:literal, $R:literal, $P:expr, $Q:expr, $L2:literal, $R2:literal, $W:literal) => {{
        let t = Instant::now();
        let g = StageGuard::unlimited();
        let cand_shared = build_shared::<$L, $R>($p, $q, &$prep.source, &$group, g).result.unwrap();
        let cand = solve_case_at::<$L, $R>(&cand_shared, &$prep, &$group, g).result.unwrap();
        drop(cand_shared);
        let ver_shared = build_shared::<$L2, $R2>($P, $Q, &$prep.source, &$group, g).result.unwrap();
        let ver = solve_case_at::<$L2, $R2>(&ver_shared, &$prep, &$group, g).result.unwrap();
        let vs = build_verify_shared::<$L2, $R2, $W>(&ver_shared, &$prep.source, &$group, g).result.unwrap();
        let rep = verify_state::<$L2, $R2, $W>(&ver_shared, &vs, &$prep, &$group, &ver, g).result.unwrap();
        let d = decide(&$prep.layout, &$prep.extents, &cand.recovered.values, &ver.recovered.values, &rep, g);
        let rs = ratios(&rep, &$prep);
        let root = $prep
            .layout
            .iter()
            .position(|m| matches!(m.id, QuantityId::EndAction { member: 1, end: End::I, component: Component::Uy }))
            .unwrap_or(usize::MAX);
        let at_root = rs.iter().find(|x| x.0 == root).map(|x| x.1);
        println!(
            "PAIR {} vs {}: decision {:?} rejection {:?}; worst (b) ratio {:?} at {:?}; member 1 end I Uy (index {root}) ratio {:?}; B {:?}; theta {:?}; {:.1}s",
            $p, $P, d.result, d.rejection, rs.first().map(|x| x.1), rs.first().map(|x| $prep.layout[x.0].id), at_root,
            rep.bodies.iter().map(|b| b.b.map(|v| super::super::directed::binary64_up(&v).unwrap())).collect::<Vec<_>>(),
            rep.theta.iter().flatten().map(|t| super::super::directed::binary64_up(t).unwrap()).collect::<Vec<_>>(),
            t.elapsed().as_secs_f64()
        );
        let top: Vec<String> = rs.iter().take(5).map(|x| format!("{:?}={:.3}", $prep.layout[x.0].id, x.1)).collect();
        println!("  top5 {}", top.join(" "));
    }};
}

#[test]
#[ignore]
fn kf3_b1_probe() {
    let path = std::env::var("KF3_B1_MODEL").expect("KF3_B1_MODEL");
    let pairs = std::env::var("KF3_B1_PAIRS").unwrap_or_else(|_| "128,256,512".into());
    let text = std::fs::read_to_string(&path).unwrap();
    let m = &models::parse_models(&text)[0];
    let source = m.source();
    println!("MODEL {} K4SRC {}", m.name, support::sha256_hex(&source.encoding()));
    let prep = CasePrep::new(source).unwrap();
    let group = prepare_group(&prep.source).unwrap();
    for p in pairs.split(',') {
        match p {
            "128" => pair!(prep, group, 128, 192, 4, 4, 256, 320, 4, 8, 8),
            "256" => pair!(prep, group, 256, 320, 4, 8, 512, 576, 8, 16, 16),
            "512" => pair!(prep, group, 512, 576, 8, 16, 1024, 1024, 16, 16, 16),
            other => panic!("{other}"),
        }
    }
}
