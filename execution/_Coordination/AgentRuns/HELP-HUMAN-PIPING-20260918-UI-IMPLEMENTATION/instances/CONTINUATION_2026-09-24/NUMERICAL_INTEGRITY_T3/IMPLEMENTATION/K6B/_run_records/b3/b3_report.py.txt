"""K6b slot K6B-S4 (b3): the report of W1-T1 to W1-T4 from the runner's records (standard library).

Usage: python3 b_report.py <runner dir> <records dir> <projection.json> > b_report.txt
"""
import json
import os
import sys

sys.dont_write_bytecode = True
sys.path.insert(0, sys.argv[1])
import k6_runner as r  # noqa: E402
import k6b_analysis as a  # noqa: E402

records_dir, projection_path = sys.argv[2], sys.argv[3]
records = r.read_records(records_dir)
runs = [x for x in r.run_entries(records) if x['tier'].startswith('W1-') and x.get('classification') != 'not_run']
not_run = [x for x in r.run_entries(records) if x['tier'].startswith('W1-') and x.get('classification') == 'not_run']
with open(projection_path) as fh:
    projection = {(x['order']): x for x in json.load(fh)['rows']}


def objects(run_id):
    with open(os.path.join(records_dir, run_id + '.jsonl')) as fh:
        return r.jsonl(fh)


def mib(x):
    return None if x is None else round(x / 2 ** 20, 2)


print('# K6b slot K6B-S4 report (b3): W1-T1 to W1-T4 on 4eeb206c0 (the parity fix) (observations only; times carry their load)')
print('runs %d, not run %d, classifications %s' % (len(runs), len(not_run),
      sorted({x['classification'] for x in runs})))
print('parity failures: %s' % sum(len(x.get('parity_failures') or []) for x in runs))
print()
print('## per run')
print('order tier model mode pass class p heap_MiB heap_move_MiB rss_MiB footprint_MiB median_ms min_ms '
      's_per_LME_ns charged load_before load_after')
figures = {}
pair_time = {}
for x in runs:
    obj = objects(x['run_id'])
    rss = x.get('rss') or {}
    lb = (x.get('load_before') or [None])[0]
    la = (x.get('load_after') or [None])[0]
    if x['mode'] == 'w1a':
        f = a.w1_figures(obj)
        figures[x['order']] = (x, f, obj)
        print(x['order'], x['tier'], x['model'], 'w1a', x['pass'], f['class'], f['selected_precision'],
              mib(x.get('repeats_heap_peak')), mib(x.get('repeats_heap_peak_move')), mib(x.get('peak_rss_bytes')),
              mib(rss.get('time_peak_footprint_bytes')), round(f['solve_median_ns'] / 1e6, 1),
              round(f['solve_min_ns'] / 1e6, 1), round(f['s_per_lme_median'] * 1e9, 3), f['meter_charged'], lb, la)
        pair_time.setdefault((x['model'], x['pass']), {})['w1a'] = f['solve_median_ns']
    else:
        sp = a.sparse_figures(obj)
        print(x['order'], x['tier'], x['model'], 'sparse', x['pass'], r.observed_class(obj), '-',
              mib(x.get('repeats_heap_peak')), mib(x.get('repeats_heap_peak_move')), mib(x.get('peak_rss_bytes')),
              mib(rss.get('time_peak_footprint_bytes')), round(sp['staged_median_ns'] / 1e6, 2),
              round((sp['entry_checked_median_ns'] or 0) / 1e6, 2), '-', '-', lb, la)
        pair_time.setdefault((x['model'], x['pass']), {})['sparse_entry'] = sp['entry_checked_median_ns']
        pair_time[(x['model'], x['pass'])]['sparse_staged'] = sp['staged_median_ns']

print()
print('## W1/binary64 time multiple per pair (median w1_solve / median entry_checked; / staged total)')
for (model_id, p), t in sorted(pair_time.items()):
    if 'w1a' in t and t.get('sparse_entry'):
        print(model_id, 'pass', p, round(t['w1a'] / t['sparse_entry'], 1), round(t['w1a'] / t['sparse_staged'], 1))

print()
print('## work by stage and precision (repeat 0 attempts; own stages and shared stages, LME)')
for order, (x, f, obj) in sorted(figures.items()):
    if x['pass'] != 1:
        continue
    attempts = [o for o in obj if o.get('kind') == 'attempt' and o['repeat'] == 0]
    for att in attempts:
        own = {k[4:]: v for k, v in att.items() if k.startswith('own_') and k != 'own_total' and v}
        shared = {k[7:]: v for k, v in att.items() if k.startswith('shared_') and isinstance(v, int)
                  and not isinstance(v, bool) and k not in ('shared_work',) and v}
        print(x['model'], 'p%d' % att['precision'], att['role'], att['outcome'], 'own', own, 'shared', shared,
              'shift_factorizations', att.get('verification_shift_factorizations'))

print()
print('## W1-T3 and W1-T4 prefixes and the stop rule (pass 1; the stop rule only where the case is Selected)')
for order, (x, f, obj) in sorted(figures.items()):
    if x['tier'] not in ('W1-T3', 'W1-T4') or x['pass'] != 1:
        continue
    pr = projection.get(order, {})
    print(x['model'], 'call heap %.2f MiB' % mib(f['call_heap_peak']),
          'prefixes', [(p['through_segment'], mib(p['heap_peak']), mib(p['increment']),
                        round(p['elapsed_ns'] / 1e9, 3)) for p in f['prefixes']],
          'last-segment increment %s MiB' % mib(f.get('last_segment_increment')),
          'stop-rule increment %s MiB' % mib(f.get('decide_increment')),
          'kept equivalent %s' % (None if f.get('tracker_entries_equivalent') is None
                                  else round(f['tracker_entries_equivalent'])),
          'measured call %.2fs' % (f['solve_median_ns'] / 1e9))

print()
print('## kept equivalents by size (pass 1)')
for order, (x, f, obj) in sorted(figures.items()):
    if x['pass'] == 1 and x['model'].startswith('RF-LARGE'):
        if f.get('tracker_entries_equivalent') is None:
            print(x['model'], x['members'], f['class'], 'no stop rule ran (not Selected)')
            continue
        print(x['model'], x['members'], 'rows', f['rows'], 'kept_equiv %.0f' % f['tracker_entries_equivalent'],
              'kept/rows %s' % ('%.3f' % (f['tracker_entries_equivalent'] / f['rows']) if f['rows'] else 'n/a (not Selected)'))

print()
print('## growth fits 10 -> 10,000 members (log-log, pass 1)')
base = [x for x in records if x.get('run_id', '').startswith('baseline')]
series = {}
for x in runs:
    if x['pass'] != 1 or not x['model'].startswith('RF-LARGE'):
        continue
    key = (x['family'], x['mode'])
    s = series.setdefault(key, {'heap_move': [], 'rss_net': []})
    s['heap_move'].append((x['members'], x['repeats_heap_peak_move']))
    if x.get('peak_rss_bytes') and x.get('baseline_rss_bytes'):
        s['rss_net'].append((x['members'], max(1, x['peak_rss_bytes'] - x['baseline_rss_bytes'])))
for key, s in sorted(series.items()):
    print(key, 'heap_move', r.fit_loglog(s['heap_move']), 'rss_net', r.fit_loglog(s['rss_net']))

print()
print('## measured rho (footprint net of the baseline, or heap move, over E_adm)')
for x in runs:
    if x['pass'] != 1:
        continue
    rho = r.measured_footprint_ratio(x, x.get('baseline_footprint_bytes'))
    rho_rss = r.measured_ratio(x, x.get('baseline_rss_bytes'))
    print(x['model'], x['mode'], 'rho_fp %.4f rho_rss %.4f' % (rho, rho_rss),
          'heap/E %.4f' % (x['repeats_heap_peak_move'] / x['estimate_adm_bytes']))


# ---------------------------------------------------------------- W1-T4 at 10,000 members
F = ['formation', 'assembly', 'residual_formation', 'factor', 'condition', 'rhs', 'solve', 'refinement',
     'recovery', 'stop_rule', 'bounded_gate', 'scale', 'estimate', 'charge', 'bound', 'shift',
     'bounded_formation', 'wide_formation', 'uc']
VK = {'RF-LARGE-CHAIN-n10000-AX': 'Span', 'RF-LARGE-CHAIN-n10000-ROT': 'Span',
      'RF-LARGE-TREE-n10000-AX': 'Span', 'RF-LARGE-TREE-n10000-ROT': 'Span',
      'RF-LARGE-CONT-n10000-AX': 'Selected 128', 'RF-LARGE-CONT-n10000-ROT': 'Span'}

print()
print('## W1-T4 rows (10,000 members): outcome, charged work, heap, RSS, time, estimate, rho')
print('order model mode pass class reason p charged heap_MiB heap_move_MiB rss_MiB footprint_MiB median_s min_s '
      'E_adm_MiB rho_fp heap/E wall_s')
for x in r.run_entries(records):
    if x['tier'] != 'W1-T4':
        continue
    if x.get('classification') == 'not_run':
        print(x['order'], x['model'], x['mode'], x['pass'], 'NOT RUN', (x.get('admission') or {}).get('reason'))
        continue
    obj = objects(x['run_id'])
    rss = x.get('rss') or {}
    rho = r.measured_footprint_ratio(x, x.get('baseline_footprint_bytes'))
    heap_e = x['repeats_heap_peak_move'] / x['estimate_adm_bytes'] if x.get('estimate_adm_bytes') else None
    if x['mode'] == 'w1a':
        f = a.w1_figures(obj)
        first = r.first_of(obj, 'outcome')
        print(x['order'], x['model'], 'w1a', x['pass'], f['class'], first.get('reason'), f['selected_precision'],
              f['meter_charged'], mib(x.get('repeats_heap_peak')), mib(x.get('repeats_heap_peak_move')),
              mib(x.get('peak_rss_bytes')), mib(rss.get('time_peak_footprint_bytes')),
              round(f['solve_median_ns'] / 1e9, 3), round(f['solve_min_ns'] / 1e9, 3),
              mib(x.get('estimate_adm_bytes')), round(rho, 4) if rho else None,
              round(heap_e, 4) if heap_e else None, round(x.get('wall_s', 0), 1))
    else:
        sp = a.sparse_figures(obj)
        print(x['order'], x['model'], 'sparse', x['pass'], r.observed_class(obj), '-', '-', '-',
              mib(x.get('repeats_heap_peak')), mib(x.get('repeats_heap_peak_move')),
              mib(x.get('peak_rss_bytes')), mib(rss.get('time_peak_footprint_bytes')),
              round(sp['staged_median_ns'] / 1e9, 3), round((sp['entry_checked_median_ns'] or 0) / 1e9, 3),
              mib(x.get('estimate_adm_bytes')), round(rho, 4) if rho else None,
              round(heap_e, 4) if heap_e else None, round(x.get('wall_s', 0), 1))

print()
print("## W1-T4 against V-K's B at 10,000 members (pass 1 and 2 w1a; where each attempt stopped)")
for order, (x, f, obj) in sorted(figures.items()):
    if x['tier'] != 'W1-T4':
        continue
    first = r.first_of(obj, 'outcome')
    mine = 'Selected %s' % f['selected_precision'] if f['class'] == 'Selected' else \
        ('Span' if first.get('reason') == 'ExactSumSpan' else '%s %s' % (f['class'], first.get('reason')))
    print(x['model'], 'pass', x['pass'], 'mine', mine, "V-K's B", VK.get(x['model']),
          'AGREE' if mine == VK.get(x['model']) else 'DIFFER')
    for att in [o for o in obj if o.get('kind') == 'attempt' and o['repeat'] == 0]:
        own = sum(att['own_' + k] for k in F)
        shared = sum(att['shared_' + k] for k in F)
        print('   p%d %s %s stages_complete=%s own_total %d own_stages %d own_unstaged %s | shared_total %d '
              'shared_stages %d shared_unstaged %s | v-build bounded %d wide %d uc %d | charged_by %d'
              % (att['precision'], att['role'], att['outcome'], att.get('stages_complete'), att['own_total'], own,
                 att.get('own_unstaged'), att['shared_work'] + att['verification_shared_work'], shared,
                 att.get('shared_unstaged'), att['shared_bounded_formation'], att['shared_wide_formation'],
                 att['shared_uc'], att['charged_by']))
    parity = [(o['item'], o['equal']) for o in obj if o.get('kind') == 'parity' and o.get('repeat') == 0]
    print('   parity repeat 0:', parity)

print()
print('## CONT-AX at 10,000 members: the 128 candidate own stages and the stop rule (pass 1, repeat 0)')
for order, (x, f, obj) in sorted(figures.items()):
    if x['model'] != 'RF-LARGE-CONT-n10000-AX' or x['pass'] != 1:
        continue
    for att in [o for o in obj if o.get('kind') == 'attempt' and o['repeat'] == 0]:
        print('p%d %s %s' % (att['precision'], att['role'], att['outcome']),
              'own', {k: att['own_' + k] for k in F if att['own_' + k]},
              'own_total', att['own_total'], 'stop_rule_work', att['stop_rule_work'],
              'shared', {k: att['shared_' + k] for k in F if att['shared_' + k]},
              'gate', att.get('gate'), 'pivot_margin_min', att.get('pivot_margin_min'), 'rcond', att.get('rcond'),
              'residual_worst', att.get('residual_worst'))
    print('outcome', {k: v for k, v in r.first_of(obj, 'outcome').items() if k.startswith(('work_', 'rows', 'meter'))})
    print('call heap %s MiB; prefixes %s; last-segment (stop rule) increment %s MiB = %s B; kept equivalent %s'
          % (mib(f['call_heap_peak']), [(p['through_segment'], mib(p['heap_peak']), round(p['elapsed_ns'] / 1e9, 3))
                                          for p in f['prefixes']], mib(f.get('decide_increment')),
             f.get('decide_increment'), f.get('tracker_entries_equivalent')))
