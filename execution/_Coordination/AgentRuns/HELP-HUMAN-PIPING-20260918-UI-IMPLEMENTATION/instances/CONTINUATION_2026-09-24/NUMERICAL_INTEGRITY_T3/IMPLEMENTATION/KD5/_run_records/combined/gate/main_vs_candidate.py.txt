"""K-D5 (I3R): the combined candidate against main b24b3d536 over every (case, mode, entry) that
both runs files hold (main's attribution run covers the 836 runs under 1000 members): outcome,
numerical quality, standing, and whether the published displacements are identical. Standard
library; P1's compare.py unchanged. Usage: python3 -B main_vs_candidate.py <compare_dir> <main.jsonl> <cand.jsonl>"""
import json, sys
sys.path.insert(0, sys.argv[1])
import compare as C

def load(path):
    out = {}
    for line in open(path):
        r = json.loads(line)
        o, _ = C.classify(r)
        run = r.get('probe', {}).get('run', {})
        env = run.get('envelope') if run.get('ok') else None
        ident = C.identity(env) if env else {}
        disp = json.dumps([(x.get('id'), x.get('value'), x.get('unit')) for x in (env or {}).get('results', [])
                           if str(x.get('id', '')).startswith('result:disp:')])
        out[(r['case'], r['mode'], r['entry'])] = (o, ident.get('numerical_quality_status'), ident.get('standing'), disp)
    return out

main, cand = load(sys.argv[2]), load(sys.argv[3])
common = sorted(set(main) & set(cand))
print('main runs', len(main), 'candidate runs', len(cand), 'compared', len(common))
diff = [k for k in common if main[k][:3] != cand[k][:3]]
disp = [k for k in common if main[k][3] != cand[k][3]]
print('standing/outcome differences:', len(diff))
for k in diff:
    print(' ', k, main[k][:3], '->', cand[k][:3])
print('published-displacement differences:', len(disp), disp[:5])
