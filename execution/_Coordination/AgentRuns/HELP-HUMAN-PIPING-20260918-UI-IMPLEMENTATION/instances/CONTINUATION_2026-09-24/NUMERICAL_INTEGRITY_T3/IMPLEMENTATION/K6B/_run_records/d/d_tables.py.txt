"""K6b D: RETURN's tables from the records (standard library; observation only).

Usage: python3 d_tables.py <runner dir> <b3 records> <b records> > tables.md

Every figure is read from the runner's records and the binary's JSONL. b3 is the result
(slot K6B-S4, 4eeb206c0, after KF1); b is the pre-KF1 run of W1-T1 to W1-T3 (f4d40dd17).
W1-T4 (10,000 members) is pre-KF3.
"""
import json
import os
import sys

sys.dont_write_bytecode = True
sys.path.insert(0, sys.argv[1])
import k6_runner as r  # noqa: E402
import k6b_analysis as a  # noqa: E402

B3, B = sys.argv[2], sys.argv[3]
MIB = 2 ** 20
SOLVE_SIDE = ('rhs', 'solve', 'refinement', 'recovery')
PASS_SIDE = ('scale', 'estimate', 'charge', 'bound', 'shift', 'bounded_gate')
BUILD = ('formation', 'assembly', 'residual_formation', 'factor', 'condition')
VBUILD = ('bounded_formation', 'wide_formation', 'uc')


def objects(d, run_id):
    with open(os.path.join(d, run_id + '.jsonl')) as fh:
        return r.jsonl(fh)


def runs(d):
    return sorted((x for x in r.run_entries(r.read_records(d)) if x.get('classification') not in (None, 'not_run')),
                  key=lambda x: x['order'])


def mib(x):
    return '-' if x is None else '%.1f' % (x / MIB)


def short(model):
    return model.replace('RF-LARGE-', '').replace('DEC053:invented-', 'DEC053 ')


def attempts(obj):
    return [o for o in obj if o.get('kind') == 'attempt' and o['repeat'] == 0]


def ssum(att, prefix, names):
    return sum(att.get(prefix + n, 0) for n in names)


def outcome_text(f, first):
    if f['class'] == 'Selected':
        return 'Selected %s' % f['selected_precision']
    return '%s(%s)' % (f['class'], first.get('reason'))


def main():
    b3 = runs(B3)
    b = runs(B)
    w1 = {}
    for x in b3:
        if x['mode'] == 'w1a':
            obj = objects(B3, x['run_id'])
            w1[x['order']] = (x, a.w1_figures(obj), obj)

    print('# K6b tables (from the records; observation only)\n')
    print('b3: slot K6B-S4, source 4eeb206c0, %d processes. b: pre-KF1, source f4d40dd17, %d processes.'
          % (len(b3), len(b)))
    print('W1-T4 (10,000 members) is pre-KF3.\n')

    print('## T1. W1 per model and size (b3, pass 1)\n')
    print('| model | members | outcome | attempts | charged LME | stop rule LME (% of call) | shift fact. '
          '| profile | pattern | rows | heap MiB | footprint MiB | RSS MiB | call s (median) | ns/LME | load | '
          'E_adm MiB | heap/E | rho_fp |')
    print('|' + '---|' * 19)
    for order, (x, f, obj) in sorted(w1.items()):
        if x['pass'] != 1:
            continue
        first = r.first_of(obj, 'outcome')
        atts = attempts(obj)
        stop = sum(t['stop_rule_work'] for t in atts)
        shift = [t.get('verification_shift_factorizations') for t in atts
                 if t.get('verification_shift_factorizations') is not None]
        chain = ', '.join('%d %s' % (t['precision'], t['outcome'].split('(')[0]) for t in atts)
        rho = r.measured_footprint_ratio(x, x.get('baseline_footprint_bytes'))
        load = (x.get('load_before') or [None])[0]
        print('| %s | %d | %s | %s | %d | %d (%.1f%%) | %s | %d | %d | %s | %s | %s | %s | %.3f | %.3f | %.2f | '
              '%s | %.4f | %.4f |' % (
                  short(x['model']), x['members'], outcome_text(f, first), chain, f['meter_charged'], stop,
                  100.0 * stop / f['meter_charged'], shift[0] if shift else '-',
                  atts[0]['storage_profile_entries'], atts[0]['storage_pattern_entries'],
                  first.get('rows', '-'), mib(x.get('repeats_heap_peak')),
                  mib((x.get('rss') or {}).get('time_peak_footprint_bytes')), mib(x.get('peak_rss_bytes')),
                  f['solve_median_ns'] / 1e9, f['s_per_lme_median'] * 1e9, load, mib(x['estimate_adm_bytes']),
                  x['repeats_heap_peak_move'] / x['estimate_adm_bytes'], rho))

    print('\n## T2. Work by precision and part (b3, pass 1, repeat 0; LME)\n')
    print('Parts: solve = rhs + solve + refinement + recovery; build = formation + assembly + residual '
          'formation + factor + condition; pass = the verification pass (scale, estimate, charge, bound, '
          'shift); v-build = the verification\'s shared data (bounded formation, wide formation, uc); '
          'unstaged = charged work no stage records (a stopped build).\n')
    print('| model | p | role | outcome | solve | stop rule | pass | build | v-build (bounded / wide / uc) '
          '| unstaged | charged by |')
    print('|' + '---|' * 11)
    for order, (x, f, obj) in sorted(w1.items()):
        if x['pass'] != 1:
            continue
        for t in attempts(obj):
            print('| %s | %d | %s | %s | %d | %d | %d | %d | %d / %d / %d | %d | %d |' % (
                short(x['model']), t['precision'], t['role'], t['outcome'], ssum(t, 'own_', SOLVE_SIDE),
                t['own_stop_rule'], ssum(t, 'own_', PASS_SIDE), ssum(t, 'shared_', BUILD),
                t['shared_bounded_formation'], t['shared_wide_formation'], t['shared_uc'],
                (t.get('own_unstaged') or 0) + (t.get('shared_unstaged') or 0), t['charged_by']))

    print('\n## T3. Memory per precision: the prefixes (b3, pass 1; heap above the call\'s start, MiB; '
          'time s)\n')
    print('| model | solve_128 | solve_256 | verify_256 | call | last segment | stop-rule increment | '
          'kept-entry equivalent |')
    print('|' + '---|' * 8)
    for order, (x, f, obj) in sorted(w1.items()):
        if x['pass'] != 1 or x['members'] < 1000:
            continue
        pre = {p['through_segment']: p for p in f['prefixes']}

        def cell(k):
            p = pre.get(k)
            return '-' if p is None else '%s (+%s), %.2f s' % (mib(p['heap_peak']), mib(p['increment']),
                                                               p['elapsed_ns'] / 1e9)
        print('| %s | %s | %s | %s | %s, %.2f s | %s | %s | %s |' % (
            short(x['model']), cell('solve_128'), cell('solve_256'), cell('verify_256'),
            mib(f['call_heap_peak']), f['solve_median_ns'] / 1e9, mib(f.get('last_segment_increment')),
            mib(f.get('decide_increment')),
            '-' if f.get('tracker_entries_equivalent') is None else '%.0f' % f['tracker_entries_equivalent']))

    print('\n## T4. Before and after KF1 at 1,000 members (pass 1): b (pre-KF1) against b3\n')
    print('| model | stop rule LME b | b3 | call heap MiB b | b3 | stop-rule increment MiB b | b3 | '
          'kept equivalent b | b3 | call s b | b3 |')
    print('|' + '---|' * 11)
    bw1 = {}
    for x in b:
        if x['mode'] == 'w1a' and x['pass'] == 1 and x['members'] == 1000:
            obj = objects(B, x['run_id'])
            bw1[x['model']] = (a.w1_figures(obj), obj)
    for order, (x, f, obj) in sorted(w1.items()):
        if x['pass'] != 1 or x['members'] != 1000 or x['model'] not in bw1:
            continue
        fb, ob = bw1[x['model']]
        sb = sum(t['stop_rule_work'] for t in attempts(ob))
        s3 = sum(t['stop_rule_work'] for t in attempts(obj))
        print('| %s | %d | %d | %s | %s | %s | %s | %.0f | %.0f | %.2f | %.2f |' % (
            short(x['model']), sb, s3, mib(fb['call_heap_peak']), mib(f['call_heap_peak']),
            mib(fb.get('decide_increment')), mib(f.get('decide_increment')),
            fb.get('tracker_entries_equivalent') or 0, f.get('tracker_entries_equivalent') or 0,
            fb['solve_median_ns'] / 1e9, f['solve_median_ns'] / 1e9))

    print('\n## T5. W1/binary64 time multiple per pair (median w1_solve over the sparse median entry_checked, '
          'and over the sparse staged total; b3)\n')
    print('| model | pass 1 entry | pass 1 staged | pass 2 entry | pass 2 staged | W1 outcome |')
    print('|' + '---|' * 6)
    p = a.packet(B3)
    table = {}
    for pr in p['w1_pairs']:
        table.setdefault(pr['model'], {})[pr['pass']] = pr
    outcome = {x['model']: outcome_text(f, r.first_of(obj, 'outcome')) for (x, f, obj) in w1.values()}
    for model in sorted(table, key=lambda m: (m.startswith('DEC'), m)):
        t = table[model]

        def v(ps, k):
            val = t.get(ps, {}).get(k)
            return '-' if val is None else '%.1f' % val
        print('| %s | %s | %s | %s | %s | %s |' % (short(model), v(1, 'multiple_entry_checked'),
                                                   v(1, 'multiple_staged'), v(2, 'multiple_entry_checked'),
                                                   v(2, 'multiple_staged'), outcome.get(model)))

    print('\n## T6. Growth fits (the packet: log-log least squares against members, both passes, '
          'classification ok)\n')
    print('| family/mode | heap-move slope | points | net-RSS slope |')
    print('|---|---|---|---|')
    for key, v in sorted(p['fits'].items()):
        print('| %s | %s | %s | %s |' % (key, '%.3f' % v['heap_move']['slope'] if v['heap_move'] else '-',
                                         len(v['heap_move']['points']) if v['heap_move'] else 0,
                                         '%.3f' % v['rss_net']['slope'] if v['rss_net'] else '-'))

    print('\n## T7. rho and heap/E by size (b3, pass 1)\n')
    print('| mode | size | n | rho_fp min–max | heap/E min–max | largest rho_fp |')
    print('|---|---|---|---|---|---|')
    groups = {}
    for x in b3:
        if x['pass'] != 1:
            continue
        size = 'DEC053' if x['model'].startswith('DEC053') else str(x['members'])
        rho = r.measured_footprint_ratio(x, x.get('baseline_footprint_bytes'))
        he = x['repeats_heap_peak_move'] / x['estimate_adm_bytes']
        groups.setdefault((x['mode'], size), []).append((rho, he, x['model']))
    for (mode, size), v in sorted(groups.items(), key=lambda kv: (kv[0][0], kv[0][1] == 'DEC053',
                                                                  int(kv[0][1]) if kv[0][1].isdigit() else 0)):
        print('| %s | %s | %d | %.3f–%.3f | %.3f–%.3f | %s |' % (
            mode, size, len(v), min(t[0] for t in v), max(t[0] for t in v), min(t[1] for t in v),
            max(t[1] for t in v), short(max(v)[2])))

    print('\n## T8. The sparse runs at 10,000 members (b3; K6\'s own mode, for the record)\n')
    print('| model | pass | class | heap MiB | footprint MiB | RSS MiB | E_adm MiB | heap/E | rho_fp |')
    print('|' + '---|' * 9)
    for x in b3:
        if x['mode'] != 'sparse' or x['members'] != 10000:
            continue
        print('| %s | %d | %s | %s | %s | %s | %s | %.3f | %.3f |' % (
            short(x['model']), x['pass'], r.observed_class(objects(B3, x['run_id'])), mib(x['repeats_heap_peak']),
            mib((x.get('rss') or {}).get('time_peak_footprint_bytes')), mib(x['peak_rss_bytes']),
            mib(x['estimate_adm_bytes']), x['repeats_heap_peak_move'] / x['estimate_adm_bytes'],
            r.measured_footprint_ratio(x, x.get('baseline_footprint_bytes'))))


if __name__ == '__main__':
    main()
