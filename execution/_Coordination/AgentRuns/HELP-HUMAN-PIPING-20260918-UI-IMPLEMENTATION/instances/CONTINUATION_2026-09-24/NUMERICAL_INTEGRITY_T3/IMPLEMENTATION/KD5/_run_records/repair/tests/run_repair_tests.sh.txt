#!/bin/bash
# I3R: the repair's test runs on 2409de83e plus the repair working tree (committed as run_repair_tests.sh.txt).
# Usage: run_repair_tests.sh <kd5-worktree> <out-dir>
# One cargo job at a time, --offline --locked, own target dir; waits for idle cargo, DEC-025 sweeps and
# timed probe comparisons first (patterns anchored on the interpreter). --nocapture keeps each test's
# measured values (actual error, trigger) in the logs.
W=$1; O=$2; mkdir -p $O
export RUSTUP_TOOLCHAIN=1.97.1 RUSTUP_AUTO_INSTALL=0 CARGO_INCREMENTAL=0 CARGO_TARGET_DIR=<wt>/kd5-target
P=$W/projects/chirality-piping/core
{ echo "HEAD $(git -C $W rev-parse HEAD)"; git -C $W status --short -- projects/chirality-piping/core
  for f in solver/nonlinear_integration/src/structural_adapter/kd5_tests.rs solver/nonlinear_integration/src/structural_adapter/kd5_models.rs product_physics/tests/formation_check_runtime.rs; do
    echo "$(sha256sum < $P/$f | cut -c1-64)  core/$f"; done
  rustc --version; cargo --version; echo "RUSTUP_TOOLCHAIN=$RUSTUP_TOOLCHAIN CARGO_INCREMENTAL=0 RUSTUP_AUTO_INSTALL=0 CARGO_TARGET_DIR=<wt>/kd5-target"
} > $O/tree_and_toolchain.txt 2>&1
run() {  # <log> <crate-dir> <cargo test args...>
  local log=$1 dir=$2; shift 2
  while pgrep -x cargo >/dev/null || pgrep -f 'python[0-9.]* .*(run_evidence_sweep|timing_compare)' >/dev/null; do sleep 30; done
  echo "=== $dir: cargo test --offline --locked $* (start $(date -u +%FT%TZ), disk_free_kb=$(df -k --output=avail $O | tail -1))" > $O/$log
  (cd $P/$dir && cargo test --offline --locked "$@" >> $O/$log 2>&1); local rc=$?
  echo "=== exit=$rc end $(date -u +%FT%TZ)" >> $O/$log
  echo "$log exit=$rc $(grep -E '^test result:' $O/$log | awk '{p+=$4; f+=$6; i+=$8} END {print "passed="p" failed="f" ignored="i}')" >> $O/SUMMARY.txt
}
run ni_kd5.log solver/nonlinear_integration kd5 -- --nocapture --test-threads=1
run fk_kd5.log solver/frame_kernel kd5 -- --test-threads=1
run pp_formation_check_runtime.log product_physics --test formation_check_runtime -- --nocapture --test-threads=1
run ni_suite.log solver/nonlinear_integration
run pp_suite.log product_physics
echo DONE >> $O/SUMMARY.txt
