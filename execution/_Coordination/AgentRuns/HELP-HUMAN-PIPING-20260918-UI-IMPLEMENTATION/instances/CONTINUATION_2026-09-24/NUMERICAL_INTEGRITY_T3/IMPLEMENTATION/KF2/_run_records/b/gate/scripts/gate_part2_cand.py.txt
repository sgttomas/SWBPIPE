"""I20 (KF2, checkpoint B) gate part 2, candidate only (ROOT, "KF2: checkpoint A accepted; B granted": the
base's timeouts are already recorded and are not re-run). Derived from I13's gate_part2.py (F1b gate 2).
Changes: the usage line (no base binary), and the run loop calls only the candidate side; everything
else (run_one unchanged, the pass-through capture, T3_FULL_ENVELOPE_PATH, the envelope span rule, the
record fields, the full-envelope check, TIMEOUT_LARGE = 1800 s and the load wait) is unchanged.
Original docstring follows.
I13 (F1b) gate part 2 (scratch): the four dense 1,000-member runs (RF-LARGE-CHAIN-n01000-ROT and
RF-LARGE-TREE-n01000-AX, dense_scrutiny, both entries), interleaved base, candidate, base, candidate,
one at a time, each at run.py's TIMEOUT_LARGE (1800 s).

Derived from G1's gate_run_base_full.py (sha256 8cf952ae...). Unchanged from it: run.py's run_one
imported unchanged (resource.setrlimit a no-op; each probe caps its own heap at 6 GiB), the
pass-through Popen that keeps the child's pipes and gives the child T3_FULL_ENVELOPE_PATH, the
envelope span rule, the per-run record fields and the full-envelope hash check. Changed: the job
list (the four part-2 runs, each on both binaries in the order base, candidate), one worker, and the
host's load averages and kern.memorystatus_level recorded before and after each run.
Gate on 130445db2 (ROOT, 2026-09-28): a load wait before each run. If the 1-minute load average
exceeds 8 at a run's start, the driver waits, polling every 30 s, until it drops below 6, then starts
the run; the wait (its start, end, seconds and the loads that triggered and released it) is recorded
in that run's record as load_wait (null when no wait was needed).
Usage: python3 -B gate_part2_cand.py <run.py dir> <cand-binary> <gen_out> <out-dir>
"""
import hashlib
import json
import os
import subprocess
import sys
import threading
import time
import types

sys.path.insert(0, os.path.abspath(sys.argv[1]))
import run as R  # noqa: E402

R.resource.setrlimit = lambda *a, **k: None
_tls = threading.local()


class _Tee:
    def __init__(self, raw):
        self._raw = raw
        self.kept = []

    def read(self, n=-1):
        b = self._raw.read(n)
        if b:
            self.kept.append(b)
        return b

    def close(self):
        return self._raw.close()

    def __getattr__(self, name):
        return getattr(self._raw, name)


class _TeePopen(subprocess.Popen):
    def __init__(self, *a, **k):
        full_path = getattr(_tls, 'full_path', None)
        if full_path is not None:
            assert 'env' not in k
            k['env'] = dict(os.environ, T3_FULL_ENVELOPE_PATH=full_path)
        super().__init__(*a, **k)
        if self.stdout is not None:
            self.stdout = _Tee(self.stdout)
        if self.stderr is not None:
            self.stderr = _Tee(self.stderr)
        _tls.proc = self


_sp = types.ModuleType('subprocess')
_sp.__dict__.update(subprocess.__dict__)
_sp.Popen = _TeePopen
R.subprocess = _sp

cand_bin, gen, outdir = sys.argv[2:5]
cases = {c['id']: c for c in json.load(open(os.path.join(gen, 'cases.json')))}
RUNS = [('RF-LARGE-CHAIN-n01000-ROT', 'captured'), ('RF-LARGE-CHAIN-n01000-ROT', 'typed'),
        ('RF-LARGE-TREE-n01000-AX', 'captured'), ('RF-LARGE-TREE-n01000-AX', 'typed')]
for side in ('cand',):
    for d in ('stdout', 'stderr', 'envelopes', 'full'):
        os.makedirs(os.path.join(outdir, side, d), exist_ok=True)
    assert not os.path.exists(os.path.join(outdir, side, 'runs.jsonl'))


def utc():
    return time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())


def host():
    return dict(load=[round(x, 2) for x in os.getloadavg()],
                memorystatus_level=int(os.popen('sysctl -n kern.memorystatus_level').read().strip() or -1),
                cargo=len(subprocess.run(['pgrep', '-x', 'cargo'], capture_output=True, text=True).stdout.split()),
                rustc=len(subprocess.run(['pgrep', '-x', 'rustc'], capture_output=True, text=True).stdout.split()),
                probes=len(subprocess.run(['pgrep', '-x', 't3_p1_probe'], capture_output=True, text=True).stdout.split()))


def load_wait():
    l1 = os.getloadavg()[0]
    if l1 <= 8:
        return None
    w = dict(start_utc=utc(), trigger_load1=round(l1, 2))
    t0 = time.time()
    print(utc(), 'load1', round(l1, 2), '> 8: waiting for < 6', flush=True)
    while True:
        time.sleep(30)
        l1 = os.getloadavg()[0]
        if l1 < 6:
            break
    w.update(end_utc=utc(), seconds=round(time.time() - t0, 1), release_load1=round(l1, 2))
    print(utc(), 'load1', round(l1, 2), '< 6: starting after', w['seconds'], 's', flush=True)
    return w


def envelope_span(text, parsed):
    run = parsed.get('run', {})
    if 'envelope' not in run:
        return None
    prefix = '{"mode":' + json.dumps(parsed['mode']) + ',"run":{"entry":' + json.dumps(run['entry']) + ',"envelope":'
    if not text.startswith(prefix):
        raise ValueError('unexpected probe line layout')
    env, end = json.JSONDecoder().raw_decode(text, len(prefix))
    if env != run['envelope']:
        raise ValueError('envelope span does not parse to the recorded envelope')
    return text[len(prefix):end]


def go(side, binary, case_id, entry):
    c = cases[case_id]
    d = os.path.join(outdir, side)
    key = '%s__dense_scrutiny__%s' % (case_id, entry)
    full_path = os.path.join(d, 'full', key + '.json')
    wait = load_wait()
    before, t_start = host(), utc()
    _tls.proc, _tls.full_path = None, full_path
    try:
        rec = R.run_one(binary, os.path.join(gen, c['request_file']), 'dense_scrutiny', entry == 'typed', R.TIMEOUT_LARGE)
    finally:
        _tls.full_path = None
    proc = _tls.proc
    after, t_end = host(), utc()
    rec.update(case=case_id, mode='dense_scrutiny', entry=entry, side=side, rlimit_as_enforced=False,
               heap_cap_bytes=6 * 1024 ** 3, start_utc=t_start, end_utc=t_end, host_before=before, host_after=after,
               load_above_6=max(before['load'][0], after['load'][0]) > 6, load_wait=wait)
    out_b, err_b = b''.join(proc.stdout.kept), b''.join(proc.stderr.kept)
    open(os.path.join(d, 'stdout', key + '.out'), 'wb').write(out_b)
    open(os.path.join(d, 'stderr', key + '.err'), 'wb').write(err_b)
    text = out_b.decode('utf-8', 'replace').strip()
    rec['full_check'] = 'none'
    if text:
        parsed = json.loads(text.splitlines()[-1])
        if parsed != rec.get('probe'):
            raise RuntimeError('tee mismatch ' + key)
        span = envelope_span(text.splitlines()[-1], parsed)
        if span is not None:
            open(os.path.join(d, 'envelopes', key + '.json'), 'wb').write(span.encode())
            full = hashlib.sha256(open(full_path, 'rb').read()).hexdigest()
            rec['full_check'] = 'equal' if full == parsed['run'].get('envelope_sha256') else 'MISMATCH'
    with open(os.path.join(d, 'runs.jsonl'), 'a') as fh:
        fh.write(json.dumps(rec) + '\n')
    run = (rec.get('probe') or {}).get('run', {})
    print(utc(), side, key, 'timed_out' if rec['timed_out'] else ('ok' if run.get('ok') else 'not-ok'),
          rec['wall_seconds'], 'exit', rec['exit_code'], 'load', before['load'], after['load'],
          'mem%', after['memorystatus_level'], flush=True)


print(utc(), 'part 2 start: 4 runs, candidate only, one at a time', flush=True)
for case_id, entry in RUNS:
    go('cand', cand_bin, case_id, entry)
print(utc(), 'part 2 done', flush=True)
