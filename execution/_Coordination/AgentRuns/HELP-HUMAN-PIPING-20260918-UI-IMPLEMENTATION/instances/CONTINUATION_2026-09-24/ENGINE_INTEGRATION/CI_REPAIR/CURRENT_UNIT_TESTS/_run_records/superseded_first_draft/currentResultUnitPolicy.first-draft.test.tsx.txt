import { act, render, renderHook, within } from "@testing-library/react";
import { isDeepStrictEqual } from "node:util";
import { afterEach, expect, it, vi } from "vitest";
const invokeMock = vi.hoisted(() => vi.fn());
vi.mock("@tauri-apps/api/core", () => ({ invoke: invokeMock }));
import modelFixture from "../../../../../fixtures/product_preview/invented_preview_model.json";
import sourceFixture from "../../../../../fixtures/product_preview/invented_mechanics_result_precision_1_sparse.json";
import { buildCurrentSessionInputManifest } from "../../services/inputManifestService";
import {
  buildAnalysisRunPreview,
  hasNativeMechanicsInvocation,
  loadDesignKnowledge,
  pollPreviewMechanicsJob,
  startPreviewMechanicsJob,
} from "../../services/previewService";
import type { MechanicsResult, PreviewModel } from "../../types";
import { DesignWorkspacePanel } from "../design-workspace/DesignWorkspacePanel";
import { hasCurrentSourceContract } from "../results/numericalResultQuality";
import { SolvePanel } from "../solve/SolvePanel";
import { useResultsSessionState } from "./resultsSessionState";
import { completeSolveJob, initialSolveJob, startSolveJob } from "./solveJobAudit";

// Unit transport replay of recorded producer bytes through mocked IPC: the
// precision-1 sparse fixture is verbatim product stdout for the unchanged
// invented_preview_model.json (fixtures/product_preview/PRECISION_FIXTURES.md).
// This is NOT a native UI qualification witness. Current standing comes only
// from the unchanged session qualification gate in useResultsSessionState.
const mode = "sparse_interactive" as const;
const modelUnits = "model=angle=rad,force=N,length=m,pressure=Pa,stress=MPa,temperature=degC";
const recordedUnits = "MPa,N,N*m,N*m/rad,N/m,boolean,count,m,mm,mode_code,rad,state_code";

afterEach(() => {
  invokeMock.mockReset();
  delete (window as unknown as Record<string, unknown>).__TAURI_INTERNALS__;
});

function installRecordedProducerReplay(model: PreviewModel) {
  const jobId = "unit-transport-replay:invented-precision-1-sparse";
  invokeMock.mockImplementation(async (command: string, args?: unknown) => {
    if (command === "start_preview_mechanics_job_with_solver_mode") {
      // Replay only the exact recorded input/mode; anything else fails.
      if (!isDeepStrictEqual(args, { model, solverMode: mode })) throw new Error("REPLAY_REQUEST_MISMATCH");
      return { job_id: jobId, backend_cancellation_token: `${jobId}:token`, state: "queued", cancellation_scope: "unit_transport_replay_not_native_ui_qualification" };
    }
    if (command === "poll_preview_mechanics_job") {
      if (!isDeepStrictEqual(args, { jobId })) throw new Error("REPLAY_JOB_UNKNOWN");
      return { job_id: jobId, state: "completed", cancellation_requested: false, cancellation_status: "not_requested",
        cancellation_scope: "unit_transport_replay_not_native_ui_qualification", result: structuredClone(sourceFixture), error_message: null };
    }
    throw new Error(`REPLAY_COMMAND_UNSUPPORTED: ${command}`);
  });
  (window as unknown as Record<string, unknown>).__TAURI_INTERNALS__ = {};
}

// Mirrors the session's solve path: backend job start/poll, native invocation
// binding, then the manifest and analysis run the App builds for that output.
async function qualifiedCurrentSession() {
  const knowledge = await loadDesignKnowledge();
  const model = structuredClone(modelFixture) as PreviewModel;
  installRecordedProducerReplay(model);
  const receipt = await startPreviewMechanicsJob(model, mode);
  if (receipt.mode !== "backend_job") throw new Error("expected a backend job receipt");
  const terminal = await pollPreviewMechanicsJob(receipt.job_id);
  const output = terminal.result as MechanicsResult;
  expect(hasNativeMechanicsInvocation(output, model, mode)).toBe(true);
  expect(hasCurrentSourceContract(output)).toBe(true);
  const producer = output.producer!;
  const inputManifest = await buildCurrentSessionInputManifest({
    model,
    solver: {
      solver_name: producer.component_name,
      solver_version: producer.component_version,
      solver_build_ref: `${producer.component_name}@${producer.component_version}`,
      solver_mode: mode,
      settings: { nonlinear_iteration_policy: "DEC-046-CV-B-product-preview-active-set-count-v1", sparse_evidence_lane: true },
    },
    active_rule_packs: [],
    external_assets: [],
  });
  const analysisRun = await buildAnalysisRunPreview(output, { inputManifest });
  const session = renderHook(() => useResultsSessionState());
  act(() => {
    session.result.current.setResult(output);
    session.result.current.setInputManifest(inputManifest);
    session.result.current.setAnalysisRun(analysisRun);
  });
  const solveJob = completeSolveJob(startSolveJob(model, receipt), output, analysisRun);
  return { knowledge, model, output, analysisRun, session, solveJob };
}

function lineValue(container: HTMLElement, testId: string) {
  return within(container).getByTestId(testId).querySelector("strong")?.textContent;
}

it("reports a qualified Current result's recorded units, including N*m/rad and N/m, unconverted in the solve and design lines", async () => {
  const { knowledge, model, output, analysisRun, session, solveJob } = await qualifiedCurrentSession();
  const { result: sessionResult, currentSolvedResult, comparison } = session.result.current;
  expect(currentSolvedResult).toBe(output); // the unchanged qualification gate admitted it as Current
  expect(sessionResult).toBe(currentSolvedResult);
  expect(output.results).toHaveLength(830);
  expect([...new Set(output.results.map((row) => row.unit))].sort().join(",")).toBe(recordedUnits);
  const before = JSON.stringify(output);

  // App.tsx supplies SolvePanel the session result and DesignWorkspacePanel the qualified Current result.
  const solve = render(<SolvePanel analysisRun={analysisRun} model={model} result={sessionResult} running={false} solveJob={solveJob}
    solverMode={mode} onCancel={() => {}} onRun={() => {}} onSolverModeChange={() => {}} />);
  expect(lineValue(solve.container, "solve-job-unit-policy")).toBe(`${modelUnits}; results=${recordedUnits}; rows=830; conversion=false`);
  expect(within(solve.container).getByTestId("solve-job-summary")).toHaveTextContent("state=completed");
  expect(within(solve.container).getByTestId("solve-job-summary")).toHaveTextContent("result_rows=830");

  const design = render(<DesignWorkspacePanel model={model} knowledge={knowledge} result={currentSolvedResult} analysisRun={analysisRun}
    comparison={comparison} editorIntents={[]} proposal={null} selectedReviewTarget={null} />);
  const designUnits = lineValue(design.container, "design-workspace-units");
  expect(designUnits).toMatch(new RegExp(`^${escapeRegExp(`${modelUnits}; results=${recordedUnits}; comparison=`)}[^;]+; conversion=false$`));
  expect(designUnits).toContain("N*m/rad");
  expect(designUnits).toContain("N/m");
  expect(within(design.container).getByTestId("design-workspace-current")).toHaveTextContent("result_rows=830");
  expect(JSON.stringify(output)).toBe(before);
});

it("reads results=none in both lines without a Current result", async () => {
  const { knowledge, model, output, session } = await qualifiedCurrentSession();
  // A received copy of the same bytes carries no invocation registration, so the gate keeps it out of Current.
  act(() => session.result.current.setResult(structuredClone(output)));
  expect(session.result.current.currentSolvedResult).toBeNull();
  expect(session.result.current.comparison).toBeNull();

  const solve = render(<SolvePanel analysisRun={null} model={model} result={null} running={false} solveJob={initialSolveJob()}
    solverMode={mode} onCancel={() => {}} onRun={() => {}} onSolverModeChange={() => {}} />);
  expect(lineValue(solve.container, "solve-job-unit-policy")).toBe(`${modelUnits}; results=none; rows=0; conversion=false`);

  const design = render(<DesignWorkspacePanel model={model} knowledge={knowledge} result={session.result.current.currentSolvedResult}
    analysisRun={null} comparison={session.result.current.comparison} editorIntents={[]} proposal={null} selectedReviewTarget={null} />);
  expect(lineValue(design.container, "design-workspace-units")).toBe(`${modelUnits}; results=none; comparison=none; conversion=false`);
});

function escapeRegExp(value: string) {
  return value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}
